package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.HousingCategoryCodeAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.HousingCategoryCodeAcquisitionOutput;

/**
 * 住宅区分コード取得サービス実装。
 */
public class HousingCategoryCodeAcquisitionImpl implements HousingCategoryCodeAcquisition {

    private static final String BLANK = "";

    private final String housingUndecided;

    private final String housingDisplay;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     *
     **/
    public HousingCategoryCodeAcquisitionImpl(CiriusConstantService ciriusConstantService) {

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.housingUndecided = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Housing.TYPE,
                ConstantLookupKeys.Cirius.Housing.UNDECIDED);
        this.housingDisplay = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Housing.TYPE,
                ConstantLookupKeys.Cirius.Housing.DISPLAY);
    }

    /**
     * 概要: 引数の建物名カナをもとに判定し、住宅区分を取得する
     * 主な仕様:
     * - 引数の建物名カナが　BLANKかNullの場合　戻り値に【37200 0：未決定】を設定し返却
     * - 引数の建物名カナが　BLANKかNull以外の場合　戻り値に【37200 2：表示する】を設定し返却
     * @param bldnKn 建物名カナ
     * @return 住宅区分
     */
    @Override
    public HousingCategoryCodeAcquisitionOutput execute(String bldnKn) {
        //引数の建物名カナがBLANKorNullの場合
        if (bldnKn == null || BLANK.equals(bldnKn)) {
            return new HousingCategoryCodeAcquisitionOutput(housingUndecided);
        //引数の建物名カナがBLANKorNull以外の場合
        } else {
            return new HousingCategoryCodeAcquisitionOutput(housingDisplay);
        }
    }

}
