package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageAmountUsageDaysCalculation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageAmountUsageDaysCalculationOutput;

/**
 * 使用量・使用日数算出サービス実装。
 */
public final class UsageAmountUsageDaysCalculationImpl
        implements UsageAmountUsageDaysCalculation {

    private static final String METER_READING_REASON_TYPE =
            ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE;

    private static final String GAS_METER_STATUS_TYPE =
            ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final Set<String> closureReasonTypes;

    private final Set<String> regularReasonTypes;

    private final String openingStatus;

    /**
     * 概要: 判定に必要な DA と定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - DA-027-R004 の呼出し先データアクセスを保持する。
     * - 閉栓系事由、定例系事由、開栓中状態区分を必須定数として解決する。
     * 既知の制約:
     * - 必須依存や必須定数が不足している場合は遅延評価せず、コンストラクタ時点で fail-fast とする。
     * @param gmkskkdDataAccess ガスメーター検針結果＜導管＞データアクセス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public UsageAmountUsageDaysCalculationImpl(
            GmkskkdDataAccess gmkskkdDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.closureReasonTypes =
                Set.of(
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .RELOCATION_CLOSURE),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .UNUSED_CLOSURE),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .TEMPORARY),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .TEMPORARY_SW_CANCELLATION),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .SUPPLY_OUTAGE),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .FIRED_AND_LOST_CAP));
        this.regularReasonTypes =
                Set.of(
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .REGULAR),
                        requireReason(
                                resolvedConstantService,
                                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                        .REGULAR_SW_CANCELLATION));
        this.openingStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedConstantService,
                        GAS_METER_STATUS_TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 概要: 検針結果一覧から使用量・使用日数を算出して返却する。
     * 主な仕様:
     * - DA-027-R004 が 0 件の場合は使用量 0、使用日数 0 を返却する。
     * - 最初の対象レコードで初期値を設定し、以後は次の対象レコードを見つけるまで加算する。
     * - 返却順に依存しないよう、検針年月日降順・検針結果通し番号降順に並べ替えてから集計する。
     * - DA の null 戻り値は契約違反として fail-fast で検知する。
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @param tksuSvkyNo 託送サービス契約番号
     * @return 使用量・使用日数算出結果
     */
    @Override
    public UsageAmountUsageDaysCalculationOutput execute(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long tksuSvkyNo) {
        List<GmkskkdRecord> records =
                gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmdAndTksuSvkyNo(
                        gasMtstNo,
                        knsYmd,
                        tksuSvkyNo);
        if (records == null) {
            throw new IllegalStateException(
                    GmkskkdDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_KNS_YMD_AND_TKSU_SVKY_NO.identifier()
                            + " returned null");
        }
        if (records.isEmpty()) {
            return zeroOutput();
        }

        List<GmkskkdRecord> sortedRecords =
                records.stream()
                        .sorted(
                                Comparator.comparing(
                                                GmkskkdRecord::knsYmd,
                                                Comparator.nullsLast(Comparator.reverseOrder()))
                                        .thenComparing(
                                                GmkskkdRecord::knsKkaTsno,
                                                Comparator.nullsLast(Comparator.reverseOrder())))
                        .toList();

        boolean previousResultFound = false;
        BigDecimal usage = BigDecimal.ZERO;
        int usageDays = 0;
        for (GmkskkdRecord record : sortedRecords) {
            if (!previousResultFound) {
                if (isTargetRecord(record)) {
                    previousResultFound = true;
                    usage = resolveUsage(record);
                    usageDays = resolveUsageDays(record);
                }
                continue;
            }
            if (isTargetRecord(record)) {
                break;
            }
            usage = usage.add(resolveUsage(record));
            usageDays += resolveUsageDays(record);
        }
        return new UsageAmountUsageDaysCalculationOutput(usage, usageDays);
    }

    private String requireReason(CiriusConstantService ciriusConstantService, String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                METER_READING_REASON_TYPE,
                code);
    }

    private boolean isTargetRecord(GmkskkdRecord record) {
        String reasonType = record.knkkSksiJysb();
        return closureReasonTypes.contains(reasonType)
                || (regularReasonTypes.contains(reasonType)
                && openingStatus.equals(record.gmtKhsJtkb()));
    }

    private BigDecimal resolveUsage(GmkskkdRecord record) {
        return record.gasSyry() == null ? BigDecimal.ZERO : record.gasSyry();
    }

    private int resolveUsageDays(GmkskkdRecord record) {
        return record.shyDdsu() == null ? 0 : record.shyDdsu().intValue();
    }

    private UsageAmountUsageDaysCalculationOutput zeroOutput() {
        return new UsageAmountUsageDaysCalculationOutput(BigDecimal.ZERO, 0);
    }
}
