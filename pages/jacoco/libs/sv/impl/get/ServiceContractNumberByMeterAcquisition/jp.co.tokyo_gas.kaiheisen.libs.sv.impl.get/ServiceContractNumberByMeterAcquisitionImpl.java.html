package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DatabaseException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ServiceContractNumberByMeterAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ServiceContractNumberByMeterAcquisitionOutput;

/** メーターに紐づくサービス契約番号取得サービス実装. */
public final class ServiceContractNumberByMeterAcquisitionImpl
        implements ServiceContractNumberByMeterAcquisition {

    /** サービス契約・メーター設置場所関連データアクセス. */
    private final SvmstkrDataAccess svmstkrDataAccess;

    /** 専用ガスメーターを表す設置状況区分. */
    private final String dedicatedGasMeterStatus;

    /** 現役系検索で使用する関連状態区分一覧. */
    private final List<String> activeStatusList;

    /** 関連終了検索で使用する関連状態区分一覧. */
    private final List<String> endedStatusList;

    /**
     * コンストラクタ.
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ServiceContractNumberByMeterAcquisitionImpl(
            final SvmstkrDataAccess svmstkrDataAccessArg,
            final CiriusConstantService ciriusConstantService) {
        this.svmstkrDataAccess =
                Objects.requireNonNull(svmstkrDataAccessArg, "svmstkrDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.dedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                                .DEDICATED_GAS_METER);
        this.activeStatusList =
                List.of(
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus
                                        .RELATED_VALID),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus
                                        .BLANK_03));
        this.endedStatusList =
                List.of(
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus
                                        .RELATED_END));
    }

    @Override
    public ServiceContractNumberByMeterAcquisitionOutput execute(
            final Long gasMtstNo,
            final String ckrtGmscJtkb,
            final Long mtgMtstNo,
            final LocalDate hssgYmd) {
        QueryResult activeResult = findActive(gasMtstNo);
        if (activeResult.record().isPresent()) {
            return toOutput(activeResult.record().orElseThrow());
        }

        if (dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            QueryResult dedicatedActiveResult = findActive(mtgMtstNo);
            if (dedicatedActiveResult.record().isPresent()) {
                return toOutput(dedicatedActiveResult.record().orElseThrow());
            }
        }

        QueryResult endedResult = findEnded(gasMtstNo, hssgYmd);
        if (endedResult.record().isPresent()) {
            return toOutput(endedResult.record().orElseThrow());
        }

        if (dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            QueryResult dedicatedEndedResult = findEnded(mtgMtstNo, hssgYmd);
            if (dedicatedEndedResult.record().isPresent()) {
                return toOutput(dedicatedEndedResult.record().orElseThrow());
            }
            throw DataInconsistencySupport.create(
                    svmstkrDataAccess.tableName(),
                    dedicatedEndedResult.callSpec().identifier(),
                    dedicatedEndedResult.parameterJsonBuilder());
        }

        throw new DatabaseException(
                endedResult.callSpec().identifier(),
                null,
                new Object[] {
                    gasMtstNo,
                    endedStatusList,
                    hssgYmd
                },
                null);
    }

    private QueryResult findActive(final Long mtstNo) {
        return new QueryResult(
                SvmstkrDataAccess.CallSpecs.FIND_BY_MTST_NO_AND_RKKI_KNRN_JTKB_LIST,
                new FlatJsonBuilder()
                        .put("mtstNo", mtstNo)
                        .put("rkkiKnrnJtkbList", activeStatusList),
                svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb(mtstNo, activeStatusList));
    }

    private QueryResult findEnded(final Long mtstNo, final LocalDate hssgYmd) {
        return new QueryResult(
                SvmstkrDataAccess.CallSpecs.FIND_BY_MTST_NO_AND_RKKI_KNRN_JTKB_AND_RKKI_END_YMD,
                new FlatJsonBuilder()
                        .put("mtstNo", mtstNo)
                        .put("rkkiKnrnJtkbList", endedStatusList)
                        .put("rkkiEndYmd", hssgYmd),
                svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkbAndRkkiEndYmd(
                        mtstNo, endedStatusList, hssgYmd));
    }

    private static ServiceContractNumberByMeterAcquisitionOutput toOutput(
            final SvmstkrRecord record) {
        return new ServiceContractNumberByMeterAcquisitionOutput(
                record.mtstNo(),
                record.svkyNo(),
                record.skno(),
                record.suguKbn(),
                record.sknrFlg(),
                record.rkkiKnrnJtkb(),
                record.rkkiSttYmd(),
                record.rkkiEndYmd(),
                record.hishYtiYmd(),
                record.recUpdJnjNo());
    }

    private record QueryResult(
            DataAccessCallSpec callSpec,
            FlatJsonBuilder parameterJsonBuilder,
            Optional<SvmstkrRecord> record) { }
}
