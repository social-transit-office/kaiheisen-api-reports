package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisitionOutput;

/** コンスタントマスタ情報取得サービス実装。 */
public class ConstantMasterInfoAcquisitionImpl implements ConstantMasterInfoAcquisition {

    private static final String TABLE_NAME = "CSTMST";

    /** DA-155-R002 の「定義のみサイン <> '1'」に対応する条件値。 */
    private static final String DEFINITION_ONLY_SIGN = "1";

    private static final ConstantMasterInfoAcquisitionOutput EMPTY_OUTPUT =
            new ConstantMasterInfoAcquisitionOutput(null, null, null);

    private final CstmstDataAccess cstmstDataAccess;

    private final SystemDateTimeProvider clock;

    /**
     * このインスタンス内で正常取得した結果だけを再利用するキャッシュ。
     *
     * <p>呼出し元のCDI Producerは必ずリクエストスコープで本クラスを生成し、異なるリクエスト間で
     * キャッシュを共有しないこと。
     */
    private final Map<LookupCondition, ConstantMasterInfoAcquisitionOutput> requestCache =
            new HashMap<>();

    /**
     * コンストラクタ。
     *
     * @param cstmstDataAccess CSTMST データアクセス
     * @param clock システム日付プロバイダ
     */
    public ConstantMasterInfoAcquisitionImpl(
            CstmstDataAccess cstmstDataAccess, SystemDateTimeProvider clock) {
        this.cstmstDataAccess = Objects.requireNonNull(cstmstDataAccess, "cstmstDataAccess");
        this.clock = Objects.requireNonNull(clock, "clock");
    }

    /**
     * 概要: コンスタントコードが null でなければ CSTMST を検索し、戻り値へ詰め替えて返却する。 主な仕様: - コンスタントコードが null の場合は処理なしとして空の出力DTOを返却する。
     * - 検索基準日は SystemDateTimeProvider の currentDate() を利用する。 - 0件時は処理対象データ無しエラーを送出する。
     *
     * @param cnstShbtCd コンスタント種別コード
     * @param cnstCd コンスタントコード
     * @return コンスタントマスタ情報
     */
    @Override
    public ConstantMasterInfoAcquisitionOutput execute(String cnstShbtCd, String cnstCd) {

        if (cnstCd == null) {
            return EMPTY_OUTPUT;
        }

        LocalDate knskKjnYmd = clock.currentDate();

        LookupCondition condition = new LookupCondition(cnstShbtCd, cnstCd, knskKjnYmd);
        ConstantMasterInfoAcquisitionOutput cached = requestCache.get(condition);
        if (cached != null) {
            return cached;
        }

        CstmstRecord record =
                cstmstDataAccess
                        .findByCnstShbtCdAndCnstCdAndTkysYmdAndTyenYmdAndTignSgn(
                                cnstShbtCd, cnstCd, knskKjnYmd, DEFINITION_ONLY_SIGN)
                        .orElseThrow(() -> new DataNotFoundException(TABLE_NAME));

        ConstantMasterInfoAcquisitionOutput output = toOutput(record);
        requestCache.put(condition, output);
        return output;
    }

    /**
     * 概要: CSTMST レコードをサービス出力 DTO へ詰め替える。 主な仕様: - レコードの内容名を出力の cntn へ設定する。 - レコードの内容略称を出力の cnsr
     * へ設定する。 - レコードの登録時利用可否フラグを出力の trkjRykhFlg へ設定する。 既知の制約: - record が null の場合は呼出し時の参照で
     * NullPointerException が発生し得る。 - 値変換や補正は行わず、レコード値をそのまま出力 DTO へ設定する。
     *
     * @param record コンスタントマスタレコード
     * @return コンスタントマスタ情報出力 DTO
     */
    private static ConstantMasterInfoAcquisitionOutput toOutput(CstmstRecord record) {
        return new ConstantMasterInfoAcquisitionOutput(
                record.cntnMix(), record.cnstSnMix(), record.trkjRykhFlg());
    }

    private record LookupCondition(String cnstShbtCd, String cnstCd, LocalDate knskKjnYmd) { }

}
