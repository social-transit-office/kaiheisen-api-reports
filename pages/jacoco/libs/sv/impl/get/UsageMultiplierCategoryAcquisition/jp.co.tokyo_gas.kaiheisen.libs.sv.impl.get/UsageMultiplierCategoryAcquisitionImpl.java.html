package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.math.BigDecimal;
import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageMultiplierCategoryAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageMultiplierCategoryAcquisitionOutput;

/**
 * 使用量倍率分類取得サービス実装。
 */
public class UsageMultiplierCategoryAcquisitionImpl implements UsageMultiplierCategoryAcquisition {

    private static final BigDecimal USAGE_4 = BigDecimal.valueOf(4L);

    private static final BigDecimal USAGE_10 = BigDecimal.valueOf(10L);

    private static final BigDecimal USAGE_20 = BigDecimal.valueOf(20L);

    private static final BigDecimal USAGE_50 = BigDecimal.valueOf(50L);

    private static final BigDecimal USAGE_100 = BigDecimal.valueOf(100L);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_UNDER_4 =
            new UsageMultiplierCategoryAcquisitionOutput(12, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_UNDER_10 =
            new UsageMultiplierCategoryAcquisitionOutput(7, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_UNDER_20 =
            new UsageMultiplierCategoryAcquisitionOutput(4, -1, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_UNDER_50 =
            new UsageMultiplierCategoryAcquisitionOutput(4, 4, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_UNDER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(3, 3, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput APRIL_MAY_OVER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(2, 2, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_UNDER_4 =
            new UsageMultiplierCategoryAcquisitionOutput(10, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_UNDER_10 =
            new UsageMultiplierCategoryAcquisitionOutput(5, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_UNDER_20 =
            new UsageMultiplierCategoryAcquisitionOutput(3, -1, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_UNDER_50 =
            new UsageMultiplierCategoryAcquisitionOutput(3, 5, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_UNDER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(3, 4, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput DECEMBER_OVER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(2, 3, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_UNDER_4 =
            new UsageMultiplierCategoryAcquisitionOutput(10, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_UNDER_10 =
            new UsageMultiplierCategoryAcquisitionOutput(5, -1, 3, 3);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_UNDER_20 =
            new UsageMultiplierCategoryAcquisitionOutput(3, -1, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_UNDER_50 =
            new UsageMultiplierCategoryAcquisitionOutput(3, 4, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_UNDER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(3, 3, 2, 2);

    private static final UsageMultiplierCategoryAcquisitionOutput OTHER_MONTH_OVER_100 =
            new UsageMultiplierCategoryAcquisitionOutput(2, 2, 2, 2);

    /**
     * 概要: 検針年月日の月区分と今回使用量の閾値から使用量倍率分類を返却する。
     * 主な仕様:
     * - 4月または5月、12月、それ以外で倍率表を切り替える。
     * - 使用量閾値は小さい順に評価し、最初に成立した条件を採用する。
     * @param knsYmd 検針年月日
     * @param konSyry 今回使用量
     * @return 使用量倍率分類取得結果
     */
    @Override
    public UsageMultiplierCategoryAcquisitionOutput execute(LocalDate knsYmd, BigDecimal konSyry) {
        int monthValue = knsYmd.getMonthValue();
        if (monthValue == 4 || monthValue == 5) {
            return resolveAprilAndMayOutput(konSyry);
        }
        if (monthValue == 12) {
            return resolveDecemberOutput(konSyry);
        }
        return resolveOtherMonthOutput(konSyry);
    }

    /**
     * 概要: 4月または5月用の倍率を返却する。
     * 主な仕様:
     * - 使用量閾値を `4`、`10`、`20`、`50`、`100` の順で判定する。
     * @param konSyry 今回使用量
     * @return 4月または5月用の倍率
     */
    private UsageMultiplierCategoryAcquisitionOutput resolveAprilAndMayOutput(BigDecimal konSyry) {
        if (konSyry.compareTo(USAGE_4) <= 0) {
            return APRIL_MAY_UNDER_4;
        }
        if (konSyry.compareTo(USAGE_10) <= 0) {
            return APRIL_MAY_UNDER_10;
        }
        if (konSyry.compareTo(USAGE_20) <= 0) {
            return APRIL_MAY_UNDER_20;
        }
        if (konSyry.compareTo(USAGE_50) <= 0) {
            return APRIL_MAY_UNDER_50;
        }
        if (konSyry.compareTo(USAGE_100) <= 0) {
            return APRIL_MAY_UNDER_100;
        }
        return APRIL_MAY_OVER_100;
    }

    /**
     * 概要: 12月用の倍率を返却する。
     * 主な仕様:
     * - 使用量閾値を `4`、`10`、`20`、`50`、`100` の順で判定する。
     * @param konSyry 今回使用量
     * @return 12月用の倍率
     */
    private UsageMultiplierCategoryAcquisitionOutput resolveDecemberOutput(BigDecimal konSyry) {
        if (konSyry.compareTo(USAGE_4) <= 0) {
            return DECEMBER_UNDER_4;
        }
        if (konSyry.compareTo(USAGE_10) <= 0) {
            return DECEMBER_UNDER_10;
        }
        if (konSyry.compareTo(USAGE_20) <= 0) {
            return DECEMBER_UNDER_20;
        }
        if (konSyry.compareTo(USAGE_50) <= 0) {
            return DECEMBER_UNDER_50;
        }
        if (konSyry.compareTo(USAGE_100) <= 0) {
            return DECEMBER_UNDER_100;
        }
        return DECEMBER_OVER_100;
    }

    /**
     * 概要: 4月または5月、12月以外の月用の倍率を返却する。
     * 主な仕様:
     * - 使用量閾値を `4`、`10`、`20`、`50`、`100` の順で判定する。
     * @param konSyry 今回使用量
     * @return 通常月用の倍率
     */
    private UsageMultiplierCategoryAcquisitionOutput resolveOtherMonthOutput(BigDecimal konSyry) {
        if (konSyry.compareTo(USAGE_4) <= 0) {
            return OTHER_MONTH_UNDER_4;
        }
        if (konSyry.compareTo(USAGE_10) <= 0) {
            return OTHER_MONTH_UNDER_10;
        }
        if (konSyry.compareTo(USAGE_20) <= 0) {
            return OTHER_MONTH_UNDER_20;
        }
        if (konSyry.compareTo(USAGE_50) <= 0) {
            return OTHER_MONTH_UNDER_50;
        }
        if (konSyry.compareTo(USAGE_100) <= 0) {
            return OTHER_MONTH_UNDER_100;
        }
        return OTHER_MONTH_OVER_100;
    }
}
