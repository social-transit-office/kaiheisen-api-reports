package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstd.KiycstdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstd.KiycstdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstk.KiycstkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstk.KiycstkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.CustomerInfoAcquisition;

/**
 * 概要: SV-164（お客さま情報取得）の実装。
 *
 * 主な仕様:
 * - 公開引数として受け取るのは `kyoTrkNo`、`tgTkurKbn`、`kasJkKbn` のみである。
 * - `hukkSysShbt`、`correlationId`、`systemDatetime` は MA 共通項目だが、本実装では public 引数に含めず RequestContext 参照も行わない。
 *
 * 既知の制約:
 * - 設計上 RequestContext 管理対象の共通項目があっても、本実装の判定条件には使用しない。
 */
public final class CustomerInfoAcquisitionImpl implements CustomerInfoAcquisition {

    private static final String AFTER_OPENING_KBN = "2";

    private final KiycstdDataAccess kiycstdDataAccess;

    private final KiycstkDataAccess kiycstkDataAccess;

    private final String tgRetailKbn;

    /**
     * コンストラクタ。
     *
     * @param kiycstdDataAccessArg 契約お客さま＜導管＞データアクセス
     * @param kiycstkDataAccessArg 契約お客さま＜小売＞データアクセス
     * @param kaiheisenConstantServiceArg 開閉栓コンスタント取得サービス
     */
    public CustomerInfoAcquisitionImpl(
            KiycstdDataAccess kiycstdDataAccessArg,
            KiycstkDataAccess kiycstkDataAccessArg,
            KaiheisenConstantService kaiheisenConstantServiceArg) {
        this.kiycstdDataAccess = Objects.requireNonNull(kiycstdDataAccessArg, "kiycstdDataAccess");
        this.kiycstkDataAccess = Objects.requireNonNull(kiycstkDataAccessArg, "kiycstkDataAccess");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantServiceArg, "kaiheisenConstantService");
        this.tgRetailKbn =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TG_RETAIL);
    }

    @Override
    public CustomerInfoAcquisitionOutput execute(Long kyoTrkNo, String tgTkurKbn, String kasJkKbn) {
        Optional<KiycstdRecord> kiycstdRecord = kiycstdDataAccess.findByKyoTrkNo(kyoTrkNo);
        if (kiycstdRecord.isEmpty() && AFTER_OPENING_KBN.equals(kasJkKbn)) {
            throw DataInconsistencySupport.create(
                    kiycstdDataAccess.tableName(),
                    KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                    parameterJsonBuilder(kyoTrkNo));
        }

        KiycstkInf kiycstkInf = null;
        if (isTgRetail(tgTkurKbn)) {
            Optional<KiycstkRecord> kiycstkRecord = kiycstkDataAccess.findByKyoTrkNo(kyoTrkNo);
            if (kiycstkRecord.isEmpty()) {
                throw DataInconsistencySupport.create(
                        kiycstkDataAccess.tableName(),
                        KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                        parameterJsonBuilder(kyoTrkNo));
            }
            kiycstkInf = toKiycstkInf(kiycstkRecord.orElseThrow());
        }

        return new CustomerInfoAcquisitionOutput(
                kiycstdRecord.map(CustomerInfoAcquisitionImpl::toKiycstdInf).orElse(null),
                kiycstkInf);
    }

    private boolean isTgRetail(String tgTkurKbn) {
        return tgRetailKbn.equals(tgTkurKbn);
    }

    private static FlatJsonBuilder parameterJsonBuilder(Long kyoTrkNo) {
        return new FlatJsonBuilder().put("kyoTrkNo", kyoTrkNo);
    }

    private static KiycstdInf toKiycstdInf(KiycstdRecord record) {
        return new KiycstdInf(
                record.kyoTrkNo(),
                record.ctnmKj1(),
                record.ctnmKn1(),
                record.ctnm1HujnShbt(),
                record.ctnm1HshiShbt(),
                record.ctnmKj2(),
                record.ctnmKn2(),
                record.ctnm2HujnShbt(),
                record.ctnm2HshiShbt(),
                record.ctnmKjhkFlg(),
                record.shkiLvShbt(),
                record.yubnNo(),
                record.gkkCd(),
                record.mchCd(),
                record.cmazCd(),
                record.cbgSuj1(),
                record.cbgKn2(),
                record.cbgKn3(),
                record.cbgKn4(),
                record.bldNo(),
                record.bldnKj(),
                record.bldnKn(),
                record.bldnKjhkFlg(),
                record.jutKbn(),
                record.tunoKn(),
                record.hynKn(),
                record.htuFlg(),
                record.recUpdJnjNo());
    }

    private static KiycstkInf toKiycstkInf(KiycstkRecord record) {
        return new KiycstkInf(
                record.kyoTrkNo(),
                record.ctnmKj1(),
                record.ctnmKn1(),
                record.ctnm1HujnShbt(),
                record.ctnm1HshiShbt(),
                record.ctnmKj2(),
                record.ctnmKn2(),
                record.ctnm2HujnShbt(),
                record.ctnm2HshiShbt(),
                record.ctnmKjhkFlg(),
                record.kynmKj(),
                record.kynmKn(),
                record.kynmHujnShbt(),
                record.kynmHshiShbt(),
                record.kynmKjhkFlg(),
                record.shkiLvShbt(),
                record.yubnNo(),
                record.gkkCd(),
                record.mchCd(),
                record.cmazCd(),
                record.cbgSuj1(),
                record.cbgKn2(),
                record.cbgKn3(),
                record.cbgKn4(),
                record.bldNo(),
                record.bldnKj(),
                record.bldnKn(),
                record.bldnKjhkFlg(),
                record.jutKbn(),
                record.tunoKn(),
                record.hynKn(),
                record.htuFlg(),
                record.recUpdJnjNo());
    }
}
