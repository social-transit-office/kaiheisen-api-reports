package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultRetroactiveDateAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultRetroactiveDateAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KensakuTaishoukikanJouhou;

/** 検針結果情報遡及日取得サービス実装。 */
public class MeterReadingResultRetroactiveDateAcquisitionImpl
        implements MeterReadingResultRetroactiveDateAcquisition {

    private static final long RETROACTIVE_DAYS_DEFAULT = 810L;

    private static final long RETROACTIVE_DAYS_OPENING_NON_TEMPORARY = 100L;

    private final String openingStatusCode;

    private final String temporaryReasonCode;

    private final String temporarySwCancellationReasonCode;

    /**
     * 概要: 判定処理で使用する必須コンスタントを初期化する。 主な仕様: - CIRIUS コンスタント取得サービスの null を検知し、初期化時に失敗させる。 -
     * ガスメーター開閉栓状態区分（開栓中）を必須定義として取得・保持する。 - 検針結果作成事由種別（臨時/臨時(SW解約)）を必須定義として取得・保持する。 既知の制約: -
     * 必須コンスタント未定義時は RequiredConstantResolver の例外が送出される。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public MeterReadingResultRetroactiveDateAcquisitionImpl(
            CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.openingStatusCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.temporaryReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .TEMPORARY);
        this.temporarySwCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .TEMPORARY_SW_CANCELLATION);
    }

    /**
     * 概要: 入力値に基づき検索対象期間（開始日/終了日）を算出して返却する。 主な仕様: - 検針年月日が null の場合は専用分岐を置かず、処理途中で自然に例外となる。 - 基準日は
     * resolveBaseDate で決定し、遡及日数は resolveRetroactiveDays で決定する。 - 戻り値の開始日は「基準日 - 遡及日数」、終了日は基準日を設定する。
     * 既知の制約: - 日付演算は LocalDate の日単位計算に依存する。 - 状態区分/事由種別の空白・空値判定は normalize の仕様に従う。
     *
     * @param tsimKnsYmd 訂正前検針年月日
     * @param knsYmd 検針年月日
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param knkkSksiJysb 検針結果作成事由種別
     * @return 検索対象期間情報
     */
    @Override
    public MeterReadingResultRetroactiveDateAcquisitionOutput execute(
            LocalDate tsimKnsYmd, LocalDate knsYmd, String gmtKhsJtkb, String knkkSksiJysb) {
        LocalDate baseDate = resolveBaseDate(tsimKnsYmd, knsYmd);
        long retroactiveDays = resolveRetroactiveDays(gmtKhsJtkb, knkkSksiJysb);

        return new MeterReadingResultRetroactiveDateAcquisitionOutput(
                new KensakuTaishoukikanJouhou(baseDate.minusDays(retroactiveDays), baseDate));
    }

    /**
     * 概要: 検索対象期間の基準日（検針年月日または訂正前検針年月日）を決定する。 主な仕様: - 訂正前検針年月日が null の場合は検針年月日を基準日にする。 -
     * 両日付が存在する場合、経過日数（検針年月日 - 訂正前検針年月日）が 0 以上なら検針年月日を採用する。 - 経過日数が負の場合は訂正前検針年月日を採用する。 既知の制約: -
     * knsYmd が null の場合、本メソッド内での参照により NullPointerException が発生し得る （呼び出し側で knsYmd の null を除外しない場合）。
     *
     * @param tsimKnsYmd 訂正前検針年月日
     * @param knsYmd 検針年月日
     * @return 算出に使用する基準日
     */
    private static LocalDate resolveBaseDate(LocalDate tsimKnsYmd, LocalDate knsYmd) {
        if (tsimKnsYmd == null) {
            return knsYmd;
        }
        long elapsedDays = ChronoUnit.DAYS.between(tsimKnsYmd, knsYmd);
        return elapsedDays >= 0 ? knsYmd : tsimKnsYmd;
    }

    /**
     * 概要: 開閉栓状態区分と検針結果作成事由種別から遡及日数を判定する。 主な仕様: - 開栓中以外は既定遡及日数（810日）を返却する。 -
     * 開栓中かつ事由種別が臨時または臨時(SW解約)の場合は810日を返却する。 - 上記以外（開栓中かつ臨時系以外）の場合は100日を返却する。 既知の制約: - 入力値比較前に
     * normalize で trim と空値 null 化を行う。
     *
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param knkkSksiJysb 検針結果作成事由種別
     * @return 判定した遡及日数
     */
    private long resolveRetroactiveDays(String gmtKhsJtkb, String knkkSksiJysb) {
        String normalizedStatus = normalize(gmtKhsJtkb);
        if (!openingStatusCode.equals(normalizedStatus)) {
            return RETROACTIVE_DAYS_DEFAULT;
        }

        String normalizedReason = normalize(knkkSksiJysb);
        if (temporaryReasonCode.equals(normalizedReason)
                || temporarySwCancellationReasonCode.equals(normalizedReason)) {
            return RETROACTIVE_DAYS_DEFAULT;
        }
        return RETROACTIVE_DAYS_OPENING_NON_TEMPORARY;
    }

    /**
     * 概要: 文字列を trim し、空値を null に正規化する。 主な仕様: - 入力値が null の場合は null を返却する。 - trim 後の文字列が空文字の場合は null
     * を返却する。 - それ以外の場合は trim 後の文字列を返却する。 既知の制約: - 空白除去の挙動は Java 標準の trim 仕様に依存する。
     *
     * @param value 正規化対象の文字列
     * @return 正規化後の文字列（空値は null）
     */
    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }
}
