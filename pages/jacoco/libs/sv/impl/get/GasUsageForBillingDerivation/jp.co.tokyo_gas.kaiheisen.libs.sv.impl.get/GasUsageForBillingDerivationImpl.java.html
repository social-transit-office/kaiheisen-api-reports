package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasBillingCalculationInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasBillingCalculationInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasUsageForBillingDerivation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasUsageForBillingDerivationOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasUsageRelatedInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasUsageRelatedInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultRetroactiveDateAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultRetroactiveDateAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GasRknKsnInf;

/** TG導管ガス使用量（料金計算用）取得サービス実装。 */
public class GasUsageForBillingDerivationImpl implements GasUsageForBillingDerivation {

    private final MeterReadingResultRetroactiveDateAcquisition
            meterReadingResultRetroactiveDateAcquisition;

    private final MeterReadingResultAcquisition meterReadingResultAcquisition;

    private final GasUsageRelatedInfoAcquisition
            gasUsageRelatedInfoAcquisition;

    private final GasBillingCalculationInfoAcquisition
            gasBillingCalculationInfoAcquisition;

    /**
     * TG導管ガス使用量（料金計算用）取得サービス実装を生成する。
     *
     * @param meterReadingResultRetroactiveDateAcquisition 検針結果遡及日取得サービス
     * @param meterReadingResultAcquisition 検針結果取得サービス
     * @param gasUsageRelatedInfoAcquisition ガス使用量関連情報取得サービス
     * @param gasBillingCalculationInfoAcquisition ガス料金計算用情報取得サービス
     */
    public GasUsageForBillingDerivationImpl(
            MeterReadingResultRetroactiveDateAcquisition meterReadingResultRetroactiveDateAcquisition,
            MeterReadingResultAcquisition meterReadingResultAcquisition,
            GasUsageRelatedInfoAcquisition gasUsageRelatedInfoAcquisition,
            GasBillingCalculationInfoAcquisition gasBillingCalculationInfoAcquisition) {
        this.meterReadingResultRetroactiveDateAcquisition =
                Objects.requireNonNull(
                        meterReadingResultRetroactiveDateAcquisition,
                        "meterReadingResultRetroactiveDateAcquisition");
        this.meterReadingResultAcquisition =
                Objects.requireNonNull(meterReadingResultAcquisition, "meterReadingResultAcquisition");
        this.gasUsageRelatedInfoAcquisition =
                Objects.requireNonNull(
                        gasUsageRelatedInfoAcquisition,
                        "gasUsageRelatedInfoAcquisition");
        this.gasBillingCalculationInfoAcquisition =
                Objects.requireNonNull(
                        gasBillingCalculationInfoAcquisition,
                        "gasBillingCalculationInfoAcquisition");
    }

    /**
     * 概要: SV-313 から SV-316 を順に呼び出し、SV-312 の戻り値へ集約する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @param gmtShjSu ガスメーター指示数
     * @param knkkSsjyShbt 検針結果作成事由種別
     * @param sjskKbn 指示数回帰区分
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param jkarSu 実供給圧力数
     * @param sjsuMincSu 指示数最小値数
     * @param sjskHticSu 指示数回帰判定値数
     * @param sgyjKytiSyry 作業時協定使用量
     * @param knsKkaTseiMaeJh 検針結果訂正前情報
     * @return TG導管ガス使用量（料金計算用）取得結果
     */
    @Override
    public GasUsageForBillingDerivationOutput execute(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long gmtShjSu,
            String knkkSsjyShbt,
            String sjskKbn,
            String gmtKhsJtkb,
            Integer jkarSu,
            BigDecimal sjsuMincSu,
            Long sjskHticSu,
            BigDecimal sgyjKytiSyry,
            KnsKkaTseiMaeJh knsKkaTseiMaeJh) {
        LocalDate tsimKnsYmd = knsKkaTseiMaeJh == null ? null : knsKkaTseiMaeJh.knsYmd();
        Integer tsimKnsKkaTsno = knsKkaTseiMaeJh == null ? null : knsKkaTseiMaeJh.knsKkaTsno();

        MeterReadingResultRetroactiveDateAcquisitionOutput retroactiveDateOutput =
                meterReadingResultRetroactiveDateAcquisition.execute(
                        tsimKnsYmd, knsYmd, gmtKhsJtkb, knkkSsjyShbt);
        MeterReadingResultAcquisitionOutput meterReadingResultOutput =
                meterReadingResultAcquisition.execute(
                        gasMtstNo,
                        knsYmd,
                        retroactiveDateOutput.kensakuTaishoukikanJouhou().knskTsksYmd(),
                        retroactiveDateOutput.kensakuTaishoukikanJouhou().knskTskeYmd(),
                        knsKkaTseiMaeJh);
        List<KnsKkaInf> knsKkaInfList =
                meterReadingResultOutput.knkkJhList() == null
                        ? List.of()
                        : toKnsKkaInfList(meterReadingResultOutput.knkkJhList());
        GasUsageRelatedInfoAcquisitionOutput gasUsageRelatedInfoOutput =
                gasUsageRelatedInfoAcquisition.execute(
                        gasMtstNo,
                        knsYmd,
                        knkkSsjyShbt,
                        gmtShjSu,
                        sgyjKytiSyry,
                        sjskKbn,
                        tsimKnsYmd,
                        tsimKnsKkaTsno,
                        knsKkaInfList,
                        toBigDecimal(meterReadingResultOutput.tsimSyry()));
        List<KkyCushInf> kkyCushInfList =
                gasUsageRelatedInfoOutput.kkyCushInfList() == null
                        ? List.of()
                        : gasUsageRelatedInfoOutput.kkyCushInfList();
        GasBillingCalculationInfoAcquisitionOutput gasBillingCalculationInfoOutput =
                gasBillingCalculationInfoAcquisition.execute(
                        knsYmd,
                        gasUsageRelatedInfoOutput.znkiKnkkInf(),
                        kkyCushInfList,
                        knsKkaInfList,
                        createSv316InputGasRknKsnInf(gasUsageRelatedInfoOutput.gasRknKsnInf()));

        return new GasUsageForBillingDerivationOutput(
                gasUsageRelatedInfoOutput.znkiKnkkInf(),
                gasUsageRelatedInfoOutput.skinKnkkInf(),
                mergeGasRknKsnInf(
                        gasUsageRelatedInfoOutput.gasRknKsnInf(),
                        gasBillingCalculationInfoOutput.gasRknKsnInf()),
                gasUsageRelatedInfoOutput.konKnkkInf(),
                gasUsageRelatedInfoOutput.kyuMtrInf(),
                gasUsageRelatedInfoOutput.sjskKbn());
    }

    /**
     * 概要: SV-315 戻り値から SV-316 入力用のガス料金計算情報を組み立てる。
     *
     * @param gasRknKsnInf ガス料金計算情報
     * @return SV-316 入力用のガス料金計算情報
     */
    private GasRknKsnInf createSv316InputGasRknKsnInf(GasRknKsnInf gasRknKsnInf) {
        return new GasRknKsnInf(
                gasRknKsnInf == null ? null : gasRknKsnInf.kokiRkksYmd(),
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    /**
     * 概要: SV-315 戻り値をベースに、SV-316 由来の項目だけを上書きして最終戻り値を生成する。
     *
     * @param gasUsageRelatedInfoGasRknKsnInf ガス使用量関連情報取得サービスのガス料金計算情報
     * @param gasBillingCalculationInfoGasRknKsnInf ガス料金計算用情報取得サービスのガス料金計算情報
     * @return SV-312 の最終ガス料金計算情報
     */
    private GasRknKsnInf mergeGasRknKsnInf(
            GasRknKsnInf gasUsageRelatedInfoGasRknKsnInf,
            GasRknKsnInf gasBillingCalculationInfoGasRknKsnInf) {
        return new GasRknKsnInf(
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.kokiRkksYmd(),
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.jkarSuKnzgGasSyry(),
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.jkarSuKnzmGasSyry(),
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.tsmJkarSuKnzgGasSyry(),
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.shyDdsu(),
                gasBillingCalculationInfoGasRknKsnInf == null
                        ? null
                        : gasBillingCalculationInfoGasRknKsnInf.kkyCushDdsu(),
                gasBillingCalculationInfoGasRknKsnInf == null
                        ? null
                        : gasBillingCalculationInfoGasRknKsnInf.tksuSvkyNo(),
                gasUsageRelatedInfoGasRknKsnInf == null
                        ? null
                        : gasUsageRelatedInfoGasRknKsnInf.sghsFkumFlg());
    }

    private static List<KnsKkaInf> toKnsKkaInfList(List<KnkkJh> knkkJhList) {
        return knkkJhList.stream().map(GasUsageForBillingDerivationImpl::toKnsKkaInf).toList();
    }

    private static KnsKkaInf toKnsKkaInf(KnkkJh knkkJh) {
        return new KnsKkaInf(
                knkkJh.knkkKtmeFlg(),
                knkkJh.knsYmd(),
                knkkJh.knsKkaTsno(),
                knkkJh.knkkSksiJysb(),
                knkkJh.gmtShjSu(),
                toBigDecimal(knkkJh.gasSyry()),
                toBigDecimal(knkkJh.sgyjKytiSyry()),
                knkkJh.shyDdsu(),
                knkkJh.sjskKbn(),
                knkkJh.knsJsjkCd(),
                null);
    }

    private static BigDecimal toBigDecimal(Double value) {
        return value == null ? null : BigDecimal.valueOf(value);
    }
}
