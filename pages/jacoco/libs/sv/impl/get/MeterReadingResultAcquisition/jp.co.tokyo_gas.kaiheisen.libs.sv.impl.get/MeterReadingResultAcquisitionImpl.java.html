package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdKkmtKnsResultRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdKnsResultRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KnkkJh;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KnsKkaTseiMaeJh;

/** 検針結果情報取得サービス実装。 */
public class MeterReadingResultAcquisitionImpl implements MeterReadingResultAcquisition {

    private static final String ERROR_CODE_NO_METER_READING_RESULT =
            ErrorCodeConstants.ERROR_CODE_020000074;

    private static final String ERROR_CODE_TARGET_METER_READING_RESULT =
            ErrorCodeConstants.ERROR_CODE_020000077;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    /**
     * コンストラクタ。
     *
     * @param gmkskkdDataAccess ガスメーター検針結果データアクセス
     */
    public MeterReadingResultAcquisitionImpl(GmkskkdDataAccess gmkskkdDataAccess) {
        this.gmkskkdDataAccess = gmkskkdDataAccess;
    }

    @Override
    public MeterReadingResultAcquisitionOutput execute(
            Long gasMtstNo,
            LocalDate knsYmd,
            LocalDate knskTsksYmd,
            LocalDate knskTskeYmd,
            KnsKkaTseiMaeJh knsKkaTseiMaeJh) {
        boolean previousKnsYmdMissing = knsKkaTseiMaeJh == null || knsKkaTseiMaeJh.knsYmd() == null;
        SearchResult searchResult;
        if (previousKnsYmdMissing) {
            searchResult =
                    new SearchResult(
                            GmkskkdDataAccess.CallSpecs
                                    .FIND_ALL_BY_GAS_MTST_NO_AND_KNS_YMD_FROM_WITH_KKMT_FLG,
                            fromKkmtRecords(
                                    gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmdFromWithKkmtFlg(
                                            gasMtstNo, knskTsksYmd)));
        } else {
            searchResult =
                    loadWithPreviousInfo(
                            gasMtstNo, knsYmd, knskTsksYmd, knskTskeYmd, knsKkaTseiMaeJh);
        }

        if (searchResult.records().isEmpty()) {
            throw new BusinessRuleException(ERROR_CODE_NO_METER_READING_RESULT);
        }

        List<KnkkJh> knkkJhList = new ArrayList<>(searchResult.records());
        Double tsimSyry;
        if (previousKnsYmdMissing) {
            trimFutureRowsWithoutPrevious(
                    knkkJhList, knsYmd, searchResult.callSpec().identifier(), gasMtstNo);
            tsimSyry = null;
        } else {
            tsimSyry =
                    adjustRowsWithPrevious(
                            knkkJhList,
                            knsYmd,
                            knsKkaTseiMaeJh,
                            searchResult.callSpec().identifier(),
                            gasMtstNo);
        }

        return new MeterReadingResultAcquisitionOutput(List.copyOf(knkkJhList), tsimSyry);
    }

    /**
     * 概要: 訂正前情報の有無と状態に応じて DA-027 呼出し種別を選択する。 主な仕様: - 訂正前検針年月日が訂正後検針年月日以上の場合、確定前フラグで R011/R010
     * を切替える。 - 訂正前検針年月日が訂正後検針年月日より前の場合、R005 を使用する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 訂正後検針年月日
     * @param knskTsksYmd 検索対象期間開始年月日
     * @param knskTskeYmd 検索対象期間終了年月日
     * @param knsKkaTseiMaeJh 訂正前情報
     * @return DA 呼出し結果と不整合判定用メタ情報
     */
    private SearchResult loadWithPreviousInfo(
            Long gasMtstNo,
            LocalDate knsYmd,
            LocalDate knskTsksYmd,
            LocalDate knskTskeYmd,
            KnsKkaTseiMaeJh knsKkaTseiMaeJh) {
        LocalDate previousKnsYmd = knsKkaTseiMaeJh.knsYmd();

        if (!previousKnsYmd.isBefore(knsYmd)) {
            if (Boolean.FALSE.equals(knsKkaTseiMaeJh.knkkKtmeFlg())) {
                List<GmkskkdKnsResultRecord> records =
                        gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                                gasMtstNo, knskTsksYmd, knskTskeYmd, knsKkaTseiMaeJh.knsKkaTsno());
                return new SearchResult(
                        GmkskkdDataAccess.CallSpecs
                                .FIND_ALL_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO,
                        fromKnsRecords(records));
            }
            List<GmkskkdKkmtKnsResultRecord> records =
                    gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmdWithKkmtFlgAndKnsKkaTsno(
                            gasMtstNo, knskTsksYmd, knskTskeYmd, knsKkaTseiMaeJh.knsKkaTsno());
            return new SearchResult(
                    GmkskkdDataAccess.CallSpecs
                            .FIND_ALL_BY_GAS_MTST_NO_AND_KNS_YMD_WITH_KKMT_FLG_AND_KNS_KKA_TSNO,
                    fromKkmtRecords(records));
        }

        List<GmkskkdKkmtKnsResultRecord> records =
                gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmdWithKkmtFlg(
                        gasMtstNo, knskTsksYmd, knskTskeYmd);
        return new SearchResult(
                GmkskkdDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_KNS_YMD_WITH_KKMT_FLG,
                fromKkmtRecords(records));
    }

    /**
     * 概要: 訂正前検針年月日がない場合の未来データ除外と整合チェックを行う。 主な仕様: - 未来日かつ確定後データが存在する場合は不整合例外を送出する。 -
     * 未来日かつ確定前データは先頭から順に除外する。 既知の制約: - 先頭から条件を外れた時点でループを終了する。
     *
     * @param knkkJhList 検針結果情報
     * @param knsYmd 訂正後検針年月日
     * @param daIdentifier DA 識別子
     * @param gasMtstNo ガスメーター設置場所番号
     */
    private void trimFutureRowsWithoutPrevious(
            List<KnkkJh> knkkJhList, LocalDate knsYmd, String daIdentifier, Long gasMtstNo) {
        Iterator<KnkkJh> iterator = knkkJhList.iterator();
        while (iterator.hasNext()) {
            KnkkJh current = iterator.next();
            if (knsYmd.isBefore(current.knsYmd()) && Boolean.FALSE.equals(current.knkkKtmeFlg())) {
                throw businessRule(daIdentifier, gasMtstNo);
            }
            if (knsYmd.isBefore(current.knsYmd()) && Boolean.TRUE.equals(current.knkkKtmeFlg())) {
                iterator.remove();
                continue;
            }
            break;
        }
    }

    /**
     * 概要: 訂正前情報がある場合の範囲判定・除外・訂正前使用量抽出を行う。 主な仕様: - 処理対象期間を導出し、範囲に該当するデータを評価する。 -
     * 同一キーかつ同一確定前フラグの行は訂正前使用量として退避し、出力から除外する。 - 訂正後検針年月日より未来日のデータは除外する。
     *
     * @param knkkJhList 検針結果情報
     * @param knsYmd 訂正後検針年月日
     * @param previous 訂正前情報
     * @param daIdentifier DA 識別子
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 訂正前使用量
     */
    private Double adjustRowsWithPrevious(
            List<KnkkJh> knkkJhList,
            LocalDate knsYmd,
            KnsKkaTseiMaeJh previous,
            String daIdentifier,
            Long gasMtstNo) {
        ProcessingRange range = deriveProcessingRange(knsYmd, previous);
        Double tsimSyry = null;

        Iterator<KnkkJh> iterator = knkkJhList.iterator();
        while (iterator.hasNext()) {
            KnkkJh current = iterator.next();
            if (!isWithinProcessingRange(current, range)) {
                break;
            }

            if (isDifferentConfirmedRecord(current, previous)) {
                throw businessRule(daIdentifier, gasMtstNo);
            }

            if (isSameRecordAndSameFlag(current, previous)) {
                tsimSyry = current.gasSyry();
                iterator.remove();
                continue;
            }

            if (current.knsYmd().isAfter(knsYmd)) {
                iterator.remove();
            }
        }

        return tsimSyry;
    }

    /**
     * 概要: 処理対象期間を導出する。 主な仕様: - 訂正前検針年月日と訂正後検針年月日の前後関係で開始/終了キーを切り替える。 - 通し番号は設計記載どおり `1`
     * または訂正前通し番号を設定する。 既知の制約: - 比較対象値が null の場合は呼び出し元で事前に除外済みであることを前提とする。
     *
     * @param knsYmd 訂正後検針年月日
     * @param previous 訂正前情報
     * @return 処理対象期間
     */
    private ProcessingRange deriveProcessingRange(LocalDate knsYmd, KnsKkaTseiMaeJh previous) {
        LocalDate previousKnsYmd = previous.knsYmd();
        Integer previousTsno = previous.knsKkaTsno();

        if (previousKnsYmd.isEqual(knsYmd)) {
            return new ProcessingRange(previousKnsYmd, previousTsno, previousKnsYmd, previousTsno);
        }

        if (previousKnsYmd.isBefore(knsYmd)) {
            return new ProcessingRange(knsYmd.plusDays(1), 1, previousKnsYmd, previousTsno);
        }

        return new ProcessingRange(previousKnsYmd, previousTsno, knsYmd.plusDays(1), 1);
    }

    /**
     * 概要: 検針年月日および検針結果通し番号が、処理対象開始～終了の範囲内であることを確認する。 主な仕様: - 開始キー以前かつ終了キー以後を処理対象とみなす。 -
     * 日付同値時は通し番号で境界判定する。
     *
     * @param current 評価対象レコード
     * @param range 処理対象期間
     * @return 処理対象範囲に含まれる場合 true
     */
    private boolean isWithinProcessingRange(KnkkJh current, ProcessingRange range) {
        boolean beforeOrAtStart =
                current.knsYmd().isBefore(range.startKnsYmd())
                        || (current.knsYmd().isEqual(range.startKnsYmd())
                                && current.knsKkaTsno() <= range.startKnsKkaTsno());
        boolean afterOrAtEnd =
                current.knsYmd().isAfter(range.endKnsYmd())
                        || (current.knsYmd().isEqual(range.endKnsYmd())
                                && current.knsKkaTsno() >= range.endKnsKkaTsno());
        return beforeOrAtStart && afterOrAtEnd;
    }

    /**
     * 概要: 確定後検針結果の有無を確認する。 主な仕様: - 訂正前キーと異なる確定後レコードが対象範囲に存在する場合に不整合とする。 既知の制約: -
     * 日付比較と通し番号比較をともに不一致で評価する。
     *
     * @param current 評価対象レコード
     * @param previous 訂正前情報
     * @return 不整合に該当する場合 true
     */
    private boolean isDifferentConfirmedRecord(KnkkJh current, KnsKkaTseiMaeJh previous) {
        return !current.knsYmd().isEqual(previous.knsYmd())
                && !current.knsKkaTsno().equals(previous.knsKkaTsno())
                && Boolean.FALSE.equals(current.knkkKtmeFlg());
    }

    /**
     * 概要: 訂正前情報と同一キー・同一確定前フラグかを判定する。 主な仕様: - 日付・通し番号・確定前フラグがすべて一致した場合に true を返す。 既知の制約: - 確定前フラグは
     * null 安全のため Boolean.equals で比較する。
     *
     * @param current 評価対象レコード
     * @param previous 訂正前情報
     * @return 同一レコードの場合 true
     */
    private boolean isSameRecordAndSameFlag(KnkkJh current, KnsKkaTseiMaeJh previous) {
        return current.knsYmd().isEqual(previous.knsYmd())
                && current.knsKkaTsno().equals(previous.knsKkaTsno())
                && Boolean.valueOf(Boolean.TRUE.equals(current.knkkKtmeFlg()))
                        .equals(Boolean.valueOf(Boolean.TRUE.equals(previous.knkkKtmeFlg())));
    }

    /**
     * 概要: DA-027 の `R009/R010/R005` 取得結果を出力 DTO へ変換する。 主な仕様: - `knkkKktmFlg=1` を確定前 (`true`)
     * としてマップする。 既知の制約: - 数値型は BigDecimal から Double へ変換する。
     *
     * @param records DA 取得結果
     * @return 変換後リスト
     */
    private List<KnkkJh> fromKkmtRecords(List<GmkskkdKkmtKnsResultRecord> records) {
        List<KnkkJh> result = new ArrayList<>();
        for (GmkskkdKkmtKnsResultRecord record : records) {
            result.add(
                    new KnkkJh(
                            "1".equals(record.knkkKktmFlg()),
                            record.knsYmd(),
                            record.knsKkaTsno(),
                            record.knkkSksiJysb(),
                            record.gmtShjSu(),
                            toDouble(record.gasSyry()),
                            toDouble(record.sgyjKytiSyry()),
                            record.shyDdsu(),
                            record.sjskKbn(),
                            record.knsJsjkCd()));
        }
        return result;
    }

    /**
     * 概要: DA-027 `R011` 取得結果を出力 DTO へ変換する。 主な仕様: - R011 は GMKSKKD 単表のため、確定前フラグを false 固定で設定する。
     * 既知の制約: - 数値型は BigDecimal から Double へ変換する。
     *
     * @param records DA 取得結果
     * @return 変換後リスト
     */
    private List<KnkkJh> fromKnsRecords(List<GmkskkdKnsResultRecord> records) {
        List<KnkkJh> result = new ArrayList<>();
        for (GmkskkdKnsResultRecord record : records) {
            result.add(
                    new KnkkJh(
                            false,
                            record.knsYmd(),
                            record.knsKkaTsno(),
                            record.knkkSksiJysb(),
                            record.gmtShjSu(),
                            toDouble(record.gasSyry()),
                            toDouble(record.sgyjKytiSyry()),
                            record.shyDdsu(),
                            record.sjskKbn(),
                            record.knsJsjkCd()));
        }
        return result;
    }

    /**
     * 概要: BigDecimal を Double に変換する。 主な仕様: - null 入力時は null を返却する。 - 非 null 入力時は `doubleValue()`
     * を返却する。 既知の制約: - 精度丸めは `doubleValue()` の仕様に依存する。
     *
     * @param value 変換対象
     * @return 変換後値
     */
    private static Double toDouble(BigDecimal value) {
        return value == null ? null : value.doubleValue();
    }

    private static BusinessRuleException businessRule(String daIdentifier, Long gasMtstNo) {
        return new BusinessRuleException(
                ERROR_CODE_TARGET_METER_READING_RESULT,
                List.of("daIdentifier=" + daIdentifier, "gasMtstNo=" + gasMtstNo));
    }

    /**
     * DA 呼出し結果と不整合判定用メタ情報を保持する。
     *
     * @param callSpec DA 呼出し識別情報
     * @param records 変換後の検針結果一覧
     */
    private record SearchResult(DataAccessCallSpec callSpec, List<KnkkJh> records) { }

    /**
     * 処理対象期間を保持する値オブジェクト。
     *
     * @param startKnsYmd 開始検針年月日
     * @param startKnsKkaTsno 開始検針結果通し番号
     * @param endKnsYmd 終了検針年月日
     * @param endKnsKkaTsno 終了検針結果通し番号
     */
    private record ProcessingRange(
            LocalDate startKnsYmd,
            Integer startKnsKkaTsno,
            LocalDate endKnsYmd,
            Integer endKnsKkaTsno) { }
}
