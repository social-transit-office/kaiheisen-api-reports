package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.CustomerNameKanjiDisplayFlagAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.CustomerNameKanjiDisplayFlagAcquisitionOutput;

/**
 * <ul>
 *   <li>概要: お客さま名漢字1の設定有無から、おなまえ漢字表示可否フラグを判定するサービス実装。</li>
 *   <li>主な仕様: お客さま名漢字1が null または空白のみ（isBlank）の場合は未設定とみなし false を返し、それ以外は true を返す。</li>
 *   <li>既知の制約: お客さま名漢字2や RequestContext の値は判定に使用せず、空白のみの文字列も未設定扱いとする。</li>
 * </ul>
 */
public class CustomerNameKanjiDisplayFlagAcquisitionImpl
        implements CustomerNameKanjiDisplayFlagAcquisition {

    /**
     * <ul>
     *   <li>概要: お客さま名漢字1の設定有無から、おなまえ漢字表示可否フラグを判定し返却する。</li>
     *   <li>主な仕様: 入力値が null または空白のみ（isBlank）の場合は未設定とみなし false を返し、それ以外は true を返す。</li>
     *   <li>既知の制約: 空白のみの文字列も未設定扱いとする。</li>
     * </ul>
     *
     * @param ctnmKj1 お客さま名漢字1（null可）
     * @return おなまえ漢字表示可否フラグを持つ出力DTO（onmeKjhkFlgは常に非null）
     */
    @Override
    public CustomerNameKanjiDisplayFlagAcquisitionOutput execute(String ctnmKj1) {
        return new CustomerNameKanjiDisplayFlagAcquisitionOutput(hasCustomerNameKanji(ctnmKj1));
    }

    /**
     * <ul>
     *   <li>概要: お客さま名漢字1が設定済みかを判定する。</li>
     *   <li>主な仕様: 入力値が null または空白のみ（isBlank）の場合は未設定とみなして false を返し、それ以外は true を返す。</li>
     *   <li>既知の制約: 空白のみの文字列も未設定扱いとする。</li>
     * </ul>
     *
     * @param ctnmKj1 お客さま名漢字1
     * @return 設定済みの場合 true、未設定の場合 false
     */
    private static Boolean hasCustomerNameKanji(String ctnmKj1) {
        return (ctnmKj1 == null || ctnmKj1.isBlank()) ? Boolean.FALSE : Boolean.TRUE;
    }
}
