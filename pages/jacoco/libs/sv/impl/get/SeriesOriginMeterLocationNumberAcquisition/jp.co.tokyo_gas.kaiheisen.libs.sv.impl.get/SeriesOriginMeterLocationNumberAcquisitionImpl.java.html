package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.SeriesOriginMeterLocationNumberAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.SeriesOriginMeterLocationNumberAcquisitionOutput;

/** 直列元のガスメーター設置場所番号取得サービス実装。 */
public class SeriesOriginMeterLocationNumberAcquisitionImpl
        implements SeriesOriginMeterLocationNumberAcquisition {

    private final String openingStatus;

    private final String originalGasMeterStatus;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public SeriesOriginMeterLocationNumberAcquisitionImpl(
            CiriusConstantService ciriusConstantService) {
        this.openingStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.originalGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                                .ORIGINAL_GAS_METER);
    }

    @Override
    public SeriesOriginMeterLocationNumberAcquisitionOutput execute(
            Long gasMtstNo, Long mtgMtstNo, String gmtKhsJtkb, String ckrtGmscJtkb) {
        if (!openingStatus.equals(gmtKhsJtkb)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000068);
        }

        if (originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return new SeriesOriginMeterLocationNumberAcquisitionOutput(gasMtstNo);
        }

        return new SeriesOriginMeterLocationNumberAcquisitionOutput(mtgMtstNo);
    }
}
