package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingRelatedInfoEditProcess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingRelatedInfoEditProcessOutput;

/**
 * 検針関連情報編集処理サービス実装。
 */
public final class MeterReadingRelatedInfoEditProcessImpl
        implements MeterReadingRelatedInfoEditProcess {

    private final RequestContext requestContext;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final String openingRelatedInfoAcquisitionDivision;

    private final List<String> openingReasonTypes;

    private final List<String> closingReasonTypes;

    private final String supplyOutageClosureReason;

    private final String supplyStopPaymentUnnecessaryStatus;

    private final String supplyStopPaymentPendingStatus;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果データアクセス
     * @param kkytskrDataAccess 供給停止管理データアクセス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    @Inject
    public MeterReadingRelatedInfoEditProcessImpl(
            RequestContext requestContext,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.openingRelatedInfoAcquisitionDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.MeterReadingRelatedInfoEditProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.MeterReadingRelatedInfoEditProcessingDivision
                        .OPENING_METER_READING_RELATED_INFO_ACQUISITION);
        String openingReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.OPENING);
        String supplyResumedReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_RESUMED);
        String relocationClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.RELOCATION_CLOSURE);
        String unusedClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.UNUSED_CLOSURE);
        this.openingReasonTypes = List.of(openingReason, supplyResumedReason);
        this.closingReasonTypes = List.of(relocationClosureReason, unusedClosureReason);
        this.supplyOutageClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
        this.supplyStopPaymentUnnecessaryStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOPPAGE_PAYMENT_UNNECESSARY);
        this.supplyStopPaymentPendingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public MeterReadingRelatedInfoEditProcessOutput execute(
            Long gasMtstNo,
            LocalDate knskKjnYmd,
            String hesnJysb,
            String knsKnrnInfHnsShrKbn) {
        requireCorrelationId();
        requireHukkSysShbt();
        requireSystemDatetime();

        Optional<KtmgmkkRecord> preliminaryMeterReadingInfo =
                findPreliminaryMeterReadingInfo(gasMtstNo, knskKjnYmd, knsKnrnInfHnsShrKbn);
        Optional<KkytskrRecord> supplyStopManagementInfo =
                findSupplyStopManagementInfo(gasMtstNo, knskKjnYmd, knsKnrnInfHnsShrKbn);

        if (preliminaryMeterReadingInfo.isEmpty() && supplyStopManagementInfo.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000047);
        }

        if (supplyOutageClosureReason.equals(hesnJysb)) {
                        // 供給停止事由では供給停止管理情報を正とし、0件なら全項目 null を返却する。
            return mapFromSupplyStopManagementInfo(supplyStopManagementInfo.orElse(null));
        }

        if (preliminaryMeterReadingInfo.isPresent()) {
            return mapFromPreliminaryMeterReadingInfo(preliminaryMeterReadingInfo.get());
        }

        return mapFromSupplyStopManagementInfo(supplyStopManagementInfo.orElse(null));
    }

    /**
     * 確定前ガスメーター検針結果情報を取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @param knsKnrnInfHnsShrKbn 検針関連情報編集処理区分
     * @return 確定前ガスメーター検針結果情報
     */
    private Optional<KtmgmkkRecord> findPreliminaryMeterReadingInfo(
            Long gasMtstNo,
            LocalDate knskKjnYmd,
            String knsKnrnInfHnsShrKbn) {
        List<String> targetReasonTypes = isOpeningRelatedInfoAcquisitionDivision(knsKnrnInfHnsShrKbn)
                ? openingReasonTypes
                : closingReasonTypes;
        return ktmgmkkDataAccess.findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
                gasMtstNo,
                knskKjnYmd,
                targetReasonTypes);
    }

    /**
     * 供給停止管理情報を取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @param knsKnrnInfHnsShrKbn 検針関連情報編集処理区分
     * @return 供給停止管理情報
     */
    private Optional<KkytskrRecord> findSupplyStopManagementInfo(
            Long gasMtstNo,
            LocalDate knskKjnYmd,
            String knsKnrnInfHnsShrKbn) {
        if (isOpeningRelatedInfoAcquisitionDivision(knsKnrnInfHnsShrKbn)) {
            return kkytskrDataAccess.findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
                    gasMtstNo,
                    knskKjnYmd,
                    supplyStopPaymentUnnecessaryStatus);
        }
        return kkytskrDataAccess.findByGasMtstNoAndHesnYmdAndKkysSisnJtkb(
                gasMtstNo,
                knskKjnYmd,
                supplyStopPaymentPendingStatus);
    }

    /**
     * 検針関連情報編集処理区分が開栓の検針関連情報取得かどうかを判定する。
     *
     * @param knsKnrnInfHnsShrKbn 検針関連情報編集処理区分
     * @return 開栓の検針関連情報取得の場合 true
     */
    private boolean isOpeningRelatedInfoAcquisitionDivision(String knsKnrnInfHnsShrKbn) {
        return Objects.equals(openingRelatedInfoAcquisitionDivision, knsKnrnInfHnsShrKbn);
    }

    private String requireCorrelationId() {
        return requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 確定前ガスメーター検針結果情報を出力 DTO に変換する。
     *
     * @param record 確定前ガスメーター検針結果情報
     * @return 出力 DTO
     */
    private MeterReadingRelatedInfoEditProcessOutput mapFromPreliminaryMeterReadingInfo(
            KtmgmkkRecord record) {
        return new MeterReadingRelatedInfoEditProcessOutput(
                record.knkkSksiJysb(),
                null,
                record.gasSyry(),
                record.shyDdsu(),
                record.kgtsKknnFlg());
    }

    /**
     * 供給停止管理情報を出力 DTO に変換する。
     *
     * @param record 供給停止管理情報
     * @return 出力 DTO
     */
    private MeterReadingRelatedInfoEditProcessOutput mapFromSupplyStopManagementInfo(
            KkytskrRecord record) {
        if (record == null) {
            return new MeterReadingRelatedInfoEditProcessOutput(null, null, null, null, null);
        }
        return new MeterReadingRelatedInfoEditProcessOutput(
                null,
                record.kkysSisnJtkb(),
                record.gasSyry(),
                record.shyDdsu(),
                record.sgttKknnFlg());
    }
}
