    /**
     * 現在検針情報と履歴からガス使用量関連情報を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @param knkkSsjyShbt 検針結果作成事由種別
     * @param gmtShjSu ガスメーター指示数
     * @param sgyjKytiSyry 作業時協定使用量
     * @param sjskKbn 指示数回帰区分
     * @param tsimKnsYmd 訂正前検針年月日
     * @param tsimKnsKkaTsno 訂正前検針結果通し番号
     * @param knkkJhList 検針結果情報リスト
     * @param tsimSyry 訂正前使用量
     * @return ガス使用量指針間情報取得結果
     */
    @Override
    public GasUsageRelatedInfoAcquisitionOutput execute(
            Long gasMtstNo,
            LocalDate knsYmd,
            String knkkSsjyShbt,
            Long gmtShjSu,
            BigDecimal sgyjKytiSyry,
            String sjskKbn,
            LocalDate tsimKnsYmd,
            Integer tsimKnsKkaTsno,
            List<KnsKkaInf> knsKkaInfList,
            BigDecimal tsimSyry) {
        List<KnsKkaInf> meterReadingResults = new ArrayList<>();
        meterReadingResults.add(createCurrentMeterReading(knsYmd, knkkSsjyShbt, gmtShjSu, sgyjKytiSyry, sjskKbn));
        if (knsKkaInfList != null) {
            meterReadingResults.addAll(knsKkaInfList);
        }

        List<KkyCushInf> supplyStopInfos = new ArrayList<>();
        ZnkiKnkkInf previousMeterReading = null;
        SkinKnkkInf recentMeterReading = null;
        boolean meterReplaced = false;
        boolean disasterClosureOrRecoveryExists = false;
        Long removedMeterReading = null;
        Long installedMeterReading = null;
        KnsKkaInf installedMeterReadingResult = null;
        SkinKnkkInf oldMeterPreviousReading = null;
        LocalDate meterReplacementDate = null;
        String resolvedRegressionSegment = sjskKbn;
        KonKnkkInf currentMeterReadingInfo = new KonKnkkInf(null, null, 0);
        LocalDate chargeStartDate = null;
        BigDecimal totalConvertedGasUsage = BigDecimal.ZERO;
        BigDecimal totalPreConvertedGasUsage = BigDecimal.ZERO;
        BigDecimal totalCorrectedConvertedGasUsage = null;
        int totalUsageDays = 0;
        boolean intervalCalculated = false;
        boolean correctionInfoProvided =
            tsimKnsYmd != null && tsimKnsKkaTsno != null && tsimSyry != null;

        for (int index = 0; index < meterReadingResults.size(); index++) {
            KnsKkaInf meterReading = meterReadingResults.get(index);
            if (index > 0 && recentMeterReading == null) {
                recentMeterReading = toRecentMeterReading(meterReading);
            }
            if (isSupplyStopReason(meterReading.knkkSsjyShbt())) {
                supplyStopInfos.add(new KkyCushInf(meterReading.knsYmd(), meterReading.knkkSsjyShbt()));
                if (isSecuritySupplyStopReason(meterReading.knkkSsjyShbt())) {
                    continue;
                }
            }
            if (isDisasterRelatedSupplyStopReason(meterReading.knkkSsjyShbt())) {
                disasterClosureOrRecoveryExists = true;
                continue;
            }
            if (isInstallationSideReplacementReason(meterReading.knkkSsjyShbt())) {
                if (installedMeterReading == null) {
                    installedMeterReading = meterReading.gmtShjSu();
                    installedMeterReadingResult = meterReading;
                }
                meterReplaced = true;
                meterReplacementDate = meterReading.knsYmd();
                continue;
            }
            if (isRemovalSideReplacementReason(meterReading.knkkSsjyShbt())) {
                if (removedMeterReading == null) {
                    removedMeterReading = meterReading.gmtShjSu();
                    oldMeterPreviousReading = findPreviousIntervalReading(meterReadingResults, index + 1);
                }
                meterReplacementDate = meterReading.knsYmd();
                continue;
            }
            if (index > 0 && previousMeterReading == null && isPreviousMeterReadingReason(meterReading.knkkSsjyShbt())) {
                previousMeterReading = toPreviousMeterReading(meterReading);
                chargeStartDate = resolveChargeStartDate(meterReadingResults, index, previousMeterReading);
                if (shouldIncludePreviousReadingDay(meterReadingResults, index, previousMeterReading)) {
                    totalUsageDays += 1;
                    if (intervalCalculated && isSameReading(previousMeterReading, recentMeterReading)) {
                        currentMeterReadingInfo = new KonKnkkInf(
                                currentMeterReadingInfo.jkarSuKnzgGasSyry(),
                                currentMeterReadingInfo.jkarSuKnzmGasSyry(),
                                currentMeterReadingInfo.shyDdsu() + 1);
                    }
                }
                break;
            }

            ReplacementIntervalContext replacementIntervalContext =
                    index == 0 ? resolveReplacementIntervalContext(meterReadingResults) : null;
            if (shouldUseReplacementFollowUpInterval(meterReading, replacementIntervalContext)) {
                intervalCalculated = true;
                KnsKkaInf installationReading = replacementIntervalContext.installationReading();
                KnsKkaInf removalReading = replacementIntervalContext.removalReading();
                SkinKnkkInf replacementPreviousReading = replacementIntervalContext.previousReading();
                IntervalContext currentIntervalContext = resolveCurrentReplacementIntervalContext(
                        gasMtstNo,
                        meterReading,
                        installationReading);
                IntervalContext removalIntervalContext =
                    resolveRemovalReplacementIntervalContext(
                        gasMtstNo,
                        removalReading,
                        replacementPreviousReading);
                if (currentIntervalContext == null
                        || removalIntervalContext == null) {
                    continue;
                }
                UsageComputation currentUsageComputation =
                    resolveUsageComputation(
                        meterReading,
                        toRecentMeterReading(installationReading),
                        currentIntervalContext.currentMeter(),
                        currentIntervalContext.meterLocation());
                UsageComputation replacementUsageComputation =
                    resolveUsageComputation(
                        removalReading,
                        replacementPreviousReading,
                        removalIntervalContext.currentMeter(),
                        removalIntervalContext.meterLocation());
                int currentUsageDays = daysBetween(
                        installationReading.knsYmd(),
                        meterReading.knsYmd());
                int replacementUsageDays =
                        daysBetween(
                                replacementPreviousReading.knsYmd(),
                                removalReading.knsYmd());

                totalPreConvertedGasUsage = totalPreConvertedGasUsage
                        .add(currentUsageComputation.preConvertedUsage())
                        .add(replacementUsageComputation.preConvertedUsage());
                totalConvertedGasUsage = totalConvertedGasUsage
                        .add(currentUsageComputation.convertedUsage())
                        .add(replacementUsageComputation.convertedUsage());
                totalUsageDays += currentUsageDays + replacementUsageDays;
                recentMeterReading = replacementPreviousReading;
                chargeStartDate =
                    chargeStartDate == null
                        ? replacementPreviousReading.knsYmd()
                        : chargeStartDate;
                resolvedRegressionSegment =
                    currentUsageComputation.regressionSegment();
                currentMeterReadingInfo = new KonKnkkInf(
                        totalConvertedGasUsage,
                        totalPreConvertedGasUsage,
                        totalUsageDays);
                continue;
            }

            IntervalContext intervalContext = resolveIntervalContext(
                    gasMtstNo,
                    meterReadingResults,
                    index);
            if (intervalContext == null) {
                continue;
            }
            intervalCalculated = true;

            TgPipelineGasUsageBetweenReadingsAcquisitionOutput usageResult =
                    tgPipelineGasUsageBetweenReadingsAcquisition.execute(
                            meterReading.sjskKbn(),
                            meterReading.gmtShjSu(),
                            intervalContext.previousReading().gmtShjSu(),
                            intervalContext.currentMeter().sjskHticSu(),
                            intervalContext.currentMeter().sjsuMincSu(),
                            defaultBigDecimal(meterReading.sgyjKytiSyry()),
                            intervalContext.meterLocation() == null
                                    || intervalContext
                                    .meterLocation().jkarSu() == null
                                    ? 0
                                    : intervalContext.meterLocation().jkarSu());
            int usageDays =
                    daysBetween(
                            intervalContext.previousReading().knsYmd(),
                            meterReading.knsYmd());
            boolean confirmedMeterReading =
                    Boolean.FALSE.equals(meterReading.knkkKtmeFlg());
            BigDecimal preConvertedUsage = confirmedMeterReading
                    ? defaultBigDecimal(meterReading.gasSyry())
                    : defaultBigDecimal(usageResult.gasSyry());
            BigDecimal convertedUsage = confirmedMeterReading
                    ? defaultBigDecimal(meterReading.gasSyry())
                    : defaultBigDecimal(usageResult.knsngGasSyry());
            BigDecimal correctedConvertedUsage = resolveCorrectedConvertedUsage(
                    meterReading,
                    tsimKnsYmd,
                    tsimKnsKkaTsno,
                    tsimSyry,
                    convertedUsage);

            totalPreConvertedGasUsage =
                    totalPreConvertedGasUsage.add(preConvertedUsage);
            totalConvertedGasUsage = totalConvertedGasUsage.add(convertedUsage);
            if (correctedConvertedUsage != null) {
                totalCorrectedConvertedGasUsage =
                        totalCorrectedConvertedGasUsage == null
                                ? correctedConvertedUsage
                                : totalCorrectedConvertedGasUsage.add(
                                        correctedConvertedUsage);
            }
            totalUsageDays += usageDays;

            if (index == 0) {
                recentMeterReading = intervalContext.previousReading();
                chargeStartDate = chargeStartDate == null
                        ? intervalContext.previousReading().knsYmd()
                        : chargeStartDate;
                resolvedRegressionSegment = usageResult.sjskKbn();
                currentMeterReadingInfo = new KonKnkkInf(
                        convertedUsage,
                        preConvertedUsage,
                        usageDays);
            }
        }

        if (previousMeterReading == null) {
            throw previousMeterReadingDataInconsistency(gasMtstNo, knsYmd);
        }

        Long transportationServiceContractNumber =
                resolveTransportationServiceContractNumber(
                        gasMtstNo,
                        knsYmd,
                        meterReadingResults);

        GasRknKsnInf gasChargeCalculationInfo =
                new GasRknKsnInf(
                        null,
                        null,
                        null,
                        null,
                        0,
                        0,
                        transportationServiceContractNumber,
                        disasterClosureOrRecoveryExists);
        KyuMtrInf oldMeterInfo =
                new KyuMtrInf(
                        null,
                        null,
                        0,
                        removedMeterReading,
                        installedMeterReading);

        if (intervalCalculated) {
            int supplyStopDays = calculateSupplyStopDays(supplyStopInfos, knsYmd);
            gasChargeCalculationInfo = new GasRknKsnInf(
                    chargeStartDate,
                    totalConvertedGasUsage,
                    totalPreConvertedGasUsage,
                    correctionInfoProvided ? totalCorrectedConvertedGasUsage : null,
                    totalUsageDays,
                    supplyStopDays,
                    transportationServiceContractNumber,
                    disasterClosureOrRecoveryExists);
            Optional<GmtsetdRecord> meterLocationOpt =
                    gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knsYmd);
            int oldMeterUsageDays = meterReplaced ? totalUsageDays : 0;
            oldMeterInfo = resolveOldMeterInfo(
                    gasMtstNo,
                    meterReplacementDate,
                    meterLocationOpt.isPresent() ? meterLocationOpt.get() : null,
                    sjskKbn,
                    oldMeterPreviousReading,
                    new MeterReplacementReadings(
                            removedMeterReading,
                            installedMeterReading),
                    resolveOldMeterUsageDays(
                            oldMeterPreviousReading,
                            meterReplacementDate,
                            oldMeterUsageDays));
        }

        if (shouldPreferInstalledReadingAsRecent(
                gmtShjSu,
                installedMeterReadingResult,
                installedMeterReading)) {
            recentMeterReading = toRecentMeterReading(installedMeterReadingResult);
        }

        return new GasUsageRelatedInfoAcquisitionOutput(
                previousMeterReading,
                recentMeterReading,
                gasChargeCalculationInfo,
                currentMeterReadingInfo,
                oldMeterInfo,
                resolvedRegressionSegment,
                List.copyOf(supplyStopInfos));
    }

    private static KnsKkaInf createCurrentMeterReading(
            final LocalDate knsYmd,
            final String knkkSsjyShbt,
            final Long gmtShjSu,
            final BigDecimal sgyjKytiSyry,
            final String sjskKbn) {
        return new KnsKkaInf(
                Boolean.TRUE,
                knsYmd,
                DEFAULT_RESULT_SERIAL_NUMBER,
                knkkSsjyShbt,
                gmtShjSu,
                BigDecimal.ZERO,
                defaultBigDecimal(sgyjKytiSyry),
                0,
                sjskKbn,
                null,
                null);
    }

    private IntervalContext resolveIntervalContext(
            final Long gasMtstNo,
            final List<KnsKkaInf> meterReadingResults,
            final int currentIndex) {
        KnsKkaInf currentReading =
                meterReadingResults.get(currentIndex);
        SkinKnkkInf previousReading =
                findPreviousIntervalReading(
                        meterReadingResults,
                        currentIndex + 1);
        if (previousReading == null) {
            return null;
        }
        Optional<GmtsetdRecord> meterLocationOpt =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(
                        gasMtstNo,
                        currentReading.knsYmd());
        GmtsetdRecord meterLocation = meterLocationOpt.isPresent()
                ? meterLocationOpt.get()
                : null;
        GasmtrdRecord currentMeter =
                resolveCurrentMeter(gasMtstNo, currentReading.knsYmd());
        if (currentMeter == null) {
            return null;
        }
        return new IntervalContext(
                previousReading,
                currentMeter,
                meterLocation);
    }

    private SkinKnkkInf findPreviousIntervalReading(
            final List<KnsKkaInf> meterReadingResults,
            final int startIndex) {
        for (int index = startIndex;
                index < meterReadingResults.size();
                index++) {
            KnsKkaInf candidate = meterReadingResults.get(index);
            if (!isSupplyStopReason(candidate.knkkSsjyShbt())
                    && !isInstallationSideReplacementReason(
                            candidate.knkkSsjyShbt())
                    && !isRemovalSideReplacementReason(
                            candidate.knkkSsjyShbt())) {
                return toRecentMeterReading(candidate);
            }
        }
        return null;
    }

    private ReplacementIntervalContext resolveReplacementIntervalContext(
            final List<KnsKkaInf> meterReadingResults) {
        final int minimumReplacementIntervalSize = 4;
        final int previousReadingStartIndex = 3;
        if (meterReadingResults.size()
                < minimumReplacementIntervalSize) {
            return null;
        }

        KnsKkaInf installationReading = meterReadingResults.get(1);
        KnsKkaInf removalReading = meterReadingResults.get(2);
        if (!isInstallationSideReplacementReason(
                installationReading.knkkSsjyShbt())
                || !isRemovalSideReplacementReason(
                        removalReading.knkkSsjyShbt())
                || !Objects.equals(
                        installationReading.knsYmd(),
                        removalReading.knsYmd())) {
            return null;
        }

        SkinKnkkInf previousReading = findPreviousIntervalReading(
                meterReadingResults,
                previousReadingStartIndex);
        if (previousReading == null) {
            return null;
        }
        return new ReplacementIntervalContext(
                installationReading,
                removalReading,
                previousReading);
    }

    private boolean shouldUseReplacementFollowUpInterval(
            final KnsKkaInf currentReading,
            final ReplacementIntervalContext replacementIntervalContext) {
        return replacementIntervalContext != null
                && Objects.equals(
                        currentReading.knkkSsjyShbt(),
                        regularReasonCode)
                && currentReading.gmtShjSu() != null
                && currentReading.gmtShjSu() > 0;
    }

    private IntervalContext resolveCurrentReplacementIntervalContext(
            final Long gasMtstNo,
            final KnsKkaInf currentReading,
            final KnsKkaInf installationReading) {
        return resolveIntervalContext(
                gasMtstNo,
                toRecentMeterReading(installationReading),
                currentReading.knsYmd());
    }

    private IntervalContext resolveRemovalReplacementIntervalContext(
            final Long gasMtstNo,
            final KnsKkaInf removalReading,
            final SkinKnkkInf previousReading) {
        return resolveIntervalContext(
                gasMtstNo,
                previousReading,
                removalReading.knsYmd());
    }

    private IntervalContext resolveIntervalContext(
            final Long gasMtstNo,
            final SkinKnkkInf previousReading,
            final LocalDate currentReadingDate) {
        Optional<GmtsetdRecord> meterLocationOpt =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(
                        gasMtstNo,
                        currentReadingDate);
        GasmtrdRecord currentMeter =
                resolveCurrentMeter(gasMtstNo, currentReadingDate);
        if (currentMeter == null) {
            return null;
        }
        return new IntervalContext(
                previousReading,
                currentMeter,
                meterLocationOpt.isPresent()
                        ? meterLocationOpt.get()
                        : null);
    }

    private UsageComputation resolveUsageComputation(
            final KnsKkaInf meterReading,
            final SkinKnkkInf previousReading,
            final GasmtrdRecord currentMeter,
            final GmtsetdRecord meterLocation) {
        if (Boolean.FALSE.equals(meterReading.knkkKtmeFlg())) {
            BigDecimal fixedUsage =
                    defaultBigDecimal(meterReading.gasSyry());
            return new UsageComputation(
                    fixedUsage,
                    fixedUsage,
                    meterReading.sjskKbn());
        }
        TgPipelineGasUsageBetweenReadingsAcquisitionOutput usageResult =
                tgPipelineGasUsageBetweenReadingsAcquisition.execute(
                        meterReading.sjskKbn(),
                        meterReading.gmtShjSu(),
                        previousReading.gmtShjSu(),
                        currentMeter.sjskHticSu(),
                        currentMeter.sjsuMincSu(),
                        defaultBigDecimal(meterReading.sgyjKytiSyry()),
                        meterLocation == null
                                || meterLocation.jkarSu() == null
                                ? 0
                                : meterLocation.jkarSu());
        BigDecimal preConvertedUsage =
                defaultBigDecimal(usageResult.gasSyry());
        BigDecimal convertedUsage =
                defaultBigDecimal(usageResult.knsngGasSyry());
        return new UsageComputation(
                preConvertedUsage,
                convertedUsage,
                usageResult.sjskKbn());
    }

    private LocalDate resolveChargeStartDate(
            final List<KnsKkaInf> meterReadingResults,
            final int previousReadingIndex,
            final ZnkiKnkkInf resolvedPreviousMeterReading) {
        if (Objects.equals(
                resolvedPreviousMeterReading.knkkSsjyShbt(),
                supplyResumedReasonCode)
                && hasSameDateSupplyOutage(
                        meterReadingResults,
                        previousReadingIndex + 1,
                        resolvedPreviousMeterReading.knsYmd())) {
            return resolvedPreviousMeterReading.knsYmd().plusDays(1);
        }
        return resolvedPreviousMeterReading.knsYmd();
    }

    private boolean shouldIncludePreviousReadingDay(
            final List<KnsKkaInf> meterReadingResults,
            final int previousReadingIndex,
            final ZnkiKnkkInf previousMeterReading) {
        String reasonCode = previousMeterReading.knkkSsjyShbt();
        if (isAnyOf(
                reasonCode,
                meterInstallationReasonCode,
                openingReasonCode)) {
            return true;
        }
        return Objects.equals(reasonCode, supplyResumedReasonCode)
                && !hasSameDateSupplyOutage(
                        meterReadingResults,
                        previousReadingIndex + 1,
                        previousMeterReading.knsYmd());
    }

    private static boolean isSameReading(
            final ZnkiKnkkInf previousMeterReading,
            final SkinKnkkInf recentMeterReading) {
        if (previousMeterReading == null
                || recentMeterReading == null) {
            return false;
        }
        return Objects.equals(
                previousMeterReading.knsYmd(),
                recentMeterReading.knsYmd())
                && Objects.equals(
                        previousMeterReading.knsKkaTsno(),
                        recentMeterReading.knsKkaTsno());
    }

    private boolean hasSameDateSupplyOutage(
            final List<KnsKkaInf> meterReadingResults,
            final int startIndex,
            final LocalDate targetDate) {
        for (int index = startIndex;
                index < meterReadingResults.size();
                index++) {
            KnsKkaInf candidate = meterReadingResults.get(index);
            if (Objects.equals(candidate.knsYmd(), targetDate)
                    && Objects.equals(
                            candidate.knkkSsjyShbt(),
                            supplyOutageReasonCode)) {
                return true;
            }
        }
        return false;
    }

    private KyuMtrInf resolveOldMeterInfo(
            final Long gasMtstNo,
            final LocalDate meterReplacementDate,
            final GmtsetdRecord meterLocation,
            final String sjskKbn,
            final SkinKnkkInf oldMeterPreviousReading,
            final MeterReplacementReadings meterReplacementReadings,
            final Integer usageDays) {
        Long removedMeterReading =
                meterReplacementReadings.removedMeterReading();
        Long installedMeterReading =
                meterReplacementReadings.installedMeterReading();
        if (removedMeterReading == null
                || installedMeterReading == null
                || meterReplacementDate == null) {
            return new KyuMtrInf(
                    null,
                    null,
                    usageDays,
                    removedMeterReading,
                    installedMeterReading);
        }
        Optional<GasmtrdRecord> oldMeterOpt =
                gasmtrdDataAccess.findByTtkGmsbNoAndGmtThkgYmd(
                        gasMtstNo,
                        meterReplacementDate);
        if (!oldMeterOpt.isPresent()) {
            return new KyuMtrInf(
                    null,
                    null,
                    usageDays,
                    removedMeterReading,
                    installedMeterReading);
        }
        GasmtrdRecord oldMeter = oldMeterOpt.get();
        Long currentReading = installedMeterReading;
        Long previousReading = removedMeterReading;
        if (isZeroReading(installedMeterReading)
                && oldMeterPreviousReading != null) {
            currentReading = removedMeterReading;
            previousReading = oldMeterPreviousReading.gmtShjSu();
        }
        TgPipelineGasUsageBetweenReadingsAcquisitionOutput usageResult =
                tgPipelineGasUsageBetweenReadingsAcquisition.execute(
                        sjskKbn,
                        currentReading,
                        previousReading,
                        oldMeter.sjskHticSu(),
                        oldMeter.sjsuMincSu(),
                        BigDecimal.ZERO,
                        meterLocation == null || meterLocation.jkarSu() == null
                                ? 0
                                : meterLocation.jkarSu());
        return new KyuMtrInf(
                usageResult.knsngGasSyry(),
                usageResult.gasSyry(),
                usageDays,
                removedMeterReading,
                installedMeterReading);
    }

    private int resolveOldMeterUsageDays(
            final SkinKnkkInf oldMeterPreviousReading,
            final LocalDate meterReplacementDate,
            final int fallbackUsageDays) {
        if (oldMeterPreviousReading == null || meterReplacementDate == null) {
            return fallbackUsageDays;
        }
        return daysBetween(
                oldMeterPreviousReading.knsYmd(),
                meterReplacementDate);
    }

    private static BigDecimal resolveCorrectedConvertedUsage(
            final KnsKkaInf meterReading,
            final LocalDate tsimKnsYmd,
            final Integer tsimKnsKkaTsno,

            final BigDecimal defaultUsage) {

                && tsimKnsKkaTsno.equals(meterReading.knsKkaTsno())) {

        }
        return defaultUsage;
    }

    private static int calculateSupplyStopDays(
            final List<KkyCushInf> supplyStopInfos,
            final LocalDate currentReadingDate) {
        if (supplyStopInfos.isEmpty()) {
            return 0;
        }
        List<KkyCushInf> sortedSupplyStopInfos = supplyStopInfos.stream()
                .sorted(Comparator.comparing(KkyCushInf::knsYmd))
                .toList();
        LocalDate supplyStopStartDate = null;
        int totalDays = 0;
        for (KkyCushInf supplyStopInfo : sortedSupplyStopInfos) {
            if (Objects.equals(
                            supplyStopInfo.knkkSsjyShbt(),
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .CLOSED_IN_CASE_OF_DISASTER)
                    || Objects.equals(
                            supplyStopInfo.knkkSsjyShbt(),
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .SECURITY_CLOSURE)) {
                supplyStopStartDate = supplyStopInfo.knsYmd().plusDays(1);
                continue;
            }
            if (supplyStopStartDate != null) {
                totalDays += daysBetween(
                        supplyStopStartDate,
                        supplyStopInfo.knsYmd());
                supplyStopStartDate = null;
            }
        }
        if (supplyStopStartDate != null) {
            totalDays += daysBetween(supplyStopStartDate, currentReadingDate);
        }
        return totalDays;
    }

    private static int daysBetween(final LocalDate from, final LocalDate to) {
        return Math.max(0, Math.toIntExact(ChronoUnit.DAYS.between(from, to)));
    }

    private static BigDecimal defaultBigDecimal(final BigDecimal value) {
        return value == null ? BigDecimal.ZERO : value;
