/**
 * 概要: SV-123（総合他メーターの検針結果＆関連情報取得）の実装クラス。
 * 主な仕様:
 * - DA-041-R015 がメーター設置場所番号順で返す同一グループ候補を最大 300 件まで反復し、連続して出現した同一設置場所番号の重複候補はスキップする。
 * - 候補ごとに導管情報、検針結果関連情報、専用メーター情報を集約し、同一グループ内フラグを更新する。
 * - `systemDatetime` は RequestContext から取得し、未設定時は fail-fast する。
 * 既知の制約:
 * - public 引数の null ガードは SV 規約に従い実装しない。
 */
public final class ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionImpl
        implements ComprehensiveOtherMeterReadingAndRelatedInfoAcquisition {

    private static final int MAX_CANDIDATE_COUNT = 300;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final MeterReadingResultAcquisitionComprehensive meterReadingResultAcquisitionComprehensive;

    private final ComprehensiveOriginMeterDedicatedMeterInfoAcquisition
            comprehensiveOriginMeterDedicatedMeterInfoAcquisition;

    private final RequestContext requestContext;

    private final String dedicatedGasMeterStatus;

    private final String relatedValidStatus;

    private final String relatedEndWaitingStatus;

    private final String relatedEndStatus;

    private final String meterOpeningStatus;

    private final String meterClosedStatus;

    private final String securityClosureReason;

    private final String disasterClosureReason;

    /**
     * 概要: 依存サービスと必須定数を解決して初期化する。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param gmtsetdDataAccess ガスメーター設置場所導管 DA
     * @param meterReadingResultAcquisitionComprehensive SV-153
     * @param comprehensiveOriginMeterDedicatedMeterInfoAcquisition SV-124
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionImpl(
            SvmstkrDataAccess svmstkrDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            MeterReadingResultAcquisitionComprehensive meterReadingResultAcquisitionComprehensive,
            ComprehensiveOriginMeterDedicatedMeterInfoAcquisition
                    comprehensiveOriginMeterDedicatedMeterInfoAcquisition,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.meterReadingResultAcquisitionComprehensive = Objects.requireNonNull(
                meterReadingResultAcquisitionComprehensive,
                "meterReadingResultAcquisitionComprehensive");
        this.comprehensiveOriginMeterDedicatedMeterInfoAcquisition = Objects.requireNonNull(
                comprehensiveOriginMeterDedicatedMeterInfoAcquisition,
                "comprehensiveOriginMeterDedicatedMeterInfoAcquisition");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dedicatedGasMeterStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.relatedValidStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.relatedEndWaitingStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.relatedEndStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.meterOpeningStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.meterClosedStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
        this.securityClosureReason = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE);
        this.disasterClosureReason = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
    }

     /**
      * 概要: SV-123 を実行し、同一グループ内の検針結果と関連情報を返却する。
      *
      * @param svmstkrUmFlg サービス契約・メーター設置場所関連有無フラグ
      * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
      * @param svkyNo サービス契約番号
      * @param gasMtstNo ガスメーター設置場所番号
      * @param rkkiSttYmd 料金計算開始年月日
      * @param rkkiEndYmd 料金計算終了年月日
      * @param dgrpHsenYmd 同一グループ内閉栓年月日
      * @return 総合他メーターの検針結果＆関連情報取得結果
      */
    @Override
    public ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionOutput execute(
            Boolean svmstkrUmFlg,
            String ckrtGmscJtkb,
            Long svkyNo,
            Long gasMtstNo,
            LocalDate rkkiSttYmd,
            LocalDate rkkiEndYmd,
            LocalDate dgrpHsenYmd) {
        OutputAccumulator accumulator = new OutputAccumulator(dgrpHsenYmd);
        if (Boolean.FALSE.equals(svmstkrUmFlg) && dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            return accumulator.toOutput();
        }

        LocalDate currentDate = requireCurrentDate();
        List<SvmstkrRecord> candidates =
            svmstkrDataAccess.findAllBySvkyNoAndRkkiKnrnJtkbAndRkkiSttYmdAndRkkiEndYmdAndMtstNoForR015(
                svkyNo,
                List.of(relatedValidStatus, relatedEndWaitingStatus, relatedEndStatus),
                rkkiSttYmd,
                rkkiEndYmd,
                gasMtstNo);

        Long previousMeterLocationNumber = 0L;
        int limit = Math.min(candidates.size(), MAX_CANDIDATE_COUNT);
        for (int index = 0; index < limit; index++) {
            SvmstkrRecord candidate = candidates.get(index);
            if (Objects.equals(previousMeterLocationNumber, candidate.mtstNo())) {
                continue;
            }
            GmtsetdRecord pipelineInfo = findPipelineInfo(candidate.mtstNo(), currentDate);
            updateGroupFlags(accumulator, candidate, pipelineInfo);
            MeterReadingResultAcquisitionComprehensiveOutput meterReadingOutput =
                    meterReadingResultAcquisitionComprehensive.execute(
                        pipelineInfo == null ? null : pipelineInfo.kngmStbJtkb(),
                        candidate.mtstNo());
            ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput dedicatedMeterOutput =
                    comprehensiveOriginMeterDedicatedMeterInfoAcquisition.execute(
                        pipelineInfo == null ? null : pipelineInfo.ckrtGmscJtkb(),
                        pipelineInfo == null ? null : pipelineInfo.szcGkkCd(),
                        pipelineInfo == null ? null : pipelineInfo.szcCmazCd(),
                        pipelineInfo == null ? null : pipelineInfo.szcMchCd(),
                        candidate.mtstNo(),
                        candidate.rkkiKnrnJtkb());
            RecentMeterReadingInfo recentMeterReadingInfo = mapRecentMeterReadingInfo(meterReadingOutput);
            accumulator.mergeMeterReadingFlags(meterReadingOutput);
            accumulator.mergeDedicatedFlag(dedicatedMeterOutput);
            accumulator.addMeterInfo(new ComprehensiveOtherMeterInfo(
                    mapServiceContractMeterRelationInfo(candidate),
                    mapPipelineMeterLocationInfo(pipelineInfo),
                    recentMeterReadingInfo));
            previousMeterLocationNumber = candidate.mtstNo();
        }
        return accumulator.toOutput();
    }

    /**
     * 概要: 現在日付を RequestContext から取得する。
     *
     * @return DA 検索に用いる現在日付
     */
    private LocalDate requireCurrentDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    /**
     * 概要: 必須 CIRIUS 定数を解決する。
     *
     * @param ciriusConstantService 定数取得サービス
     * @param typeCode 定数種別コード
     * @param code 定数コード
     * @return 解決済み定数値
     */
    private static String requireConstant(
            CiriusConstantService ciriusConstantService,
            String typeCode,
            String code) {
        return RequiredConstantResolver.requireCiriusConstant(ciriusConstantService, typeCode, code);
    }

    /**
     * 概要: 候補設置場所番号に対応する導管情報を取得する。
     *
     * @param meterLocationNumber 候補設置場所番号
     * @param currentDate 検索基準日
     * @return 条件に一致した導管情報の先頭レコード。取得できない場合は null
     */
    private GmtsetdRecord findPipelineInfo(Long meterLocationNumber, LocalDate currentDate) {
        List<GmtsetdRecord> records =
            gmtsetdDataAccess.findAllByGasMtstNoAndTkysYmdAndTyenYmd(
                meterLocationNumber,
                currentDate);
        if (records.isEmpty()) {
            return null;
        }
        return records.getFirst();
    }

    /**
     * 概要: 候補と導管情報から同一グループ内のフラグ類を更新する。
     *
     * @param accumulator 出力集約状態
     * @param candidate 候補関連情報
     * @param pipelineInfo 導管情報
     */
    private void updateGroupFlags(
            OutputAccumulator accumulator,
            SvmstkrRecord candidate,
            GmtsetdRecord pipelineInfo) {
        if (pipelineInfo == null || !isPositiveMeterLocation(pipelineInfo.gasMtstNo())) {
            accumulator.dgrpMiRegFlg = true;
            return;
        }
        if (meterOpeningStatus.equals(pipelineInfo.gmtKhsJtkb())
                && !relatedEndStatus.equals(candidate.rkkiKnrnJtkb())) {
            accumulator.dgrpKasFlg = true;
        }
        if (isLaterDate(pipelineInfo.hesnYmd(), accumulator.dgrpHsenYmd)) {
            accumulator.dgrpHsenYmd = pipelineInfo.hesnYmd();
        }
        if (relatedEndStatus.equals(candidate.rkkiKnrnJtkb())
                && isLaterDate(candidate.rkkiEndYmd(), accumulator.dgrpHsenYmd)) {
            accumulator.dgrpHsenYmd = candidate.rkkiEndYmd();
        }
        if (meterClosedStatus.equals(pipelineInfo.gmtKhsJtkb())
                && isNonClosableReason(pipelineInfo.hesnJysb())) {
            accumulator.dgrpHkhesnJiyuFlg = true;
        }
    }

    /**
     * 概要: メーター設置場所番号が正数か判定する。
     *
     * @param meterLocationNumber 判定対象番号
     * @return 正数の場合 true
     */
    private static boolean isPositiveMeterLocation(Long meterLocationNumber) {
        return meterLocationNumber != null && meterLocationNumber > 0L;
    }

    /**
     * 概要: 日付が比較対象より後か判定する。
     *
     * @param candidateDate 判定対象日付
     * @param currentDate 比較基準日付
     * @return candidateDate が後日の場合 true
     */
    private static boolean isLaterDate(LocalDate candidateDate, LocalDate currentDate) {
        return candidateDate != null && (currentDate == null || candidateDate.isAfter(currentDate));
    }

    /**
     * 概要: 閉栓事由が全部閉栓不可対象か判定する。
     *
     * @param closureReason 閉栓事由種別
     * @return 保安閉栓または災害時閉栓の場合 true
     */
    private boolean isNonClosableReason(String closureReason) {
        return securityClosureReason.equals(closureReason)
                || disasterClosureReason.equals(closureReason);
    }

    /**
     * 概要: 候補関連情報を API モデルへ変換する。
     *
     * @param candidate 候補関連情報
     * @return API モデル
     */
    private static ServiceContractMeterRelationInfo mapServiceContractMeterRelationInfo(
            SvmstkrRecord candidate) {
        return new ServiceContractMeterRelationInfo(
                candidate.mtstNo(),
                candidate.svkyNo(),
                candidate.skno(),
                candidate.rkkiKnrnJtkb(),
                candidate.rkkiEndYmd(),
                candidate.recUpdJnjNo());
    }

    /**
     * 概要: 導管情報を API モデルへ変換する。
     *
     * @param pipelineInfo 導管情報
     * @return API モデル。入力が null の場合は null
     */
    private static PipelineMeterLocationInfo mapPipelineMeterLocationInfo(
            GmtsetdRecord pipelineInfo) {
        if (pipelineInfo == null) {
            return null;
        }
        return new PipelineMeterLocationInfo(
                pipelineInfo.kngmStbJtkb(),
                pipelineInfo.ckrtGmscJtkb());
    }

    /**
     * 概要: SV-153 の戻り値を最近検針結果情報へ変換する。
     *
     * @param output SV-153 戻り値
     * @return 最近検針結果情報。設定項目がない場合は null
     */
    private static RecentMeterReadingInfo mapRecentMeterReadingInfo(
            MeterReadingResultAcquisitionComprehensiveOutput output) {
        if (output.skinKnkkInf() == null && output.hsenHkkKnuKigenYmd() == null) {
            return null;
        }
        return new RecentMeterReadingInfo(
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().knsYmd(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().knsKkaTsno(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().gmtShjSu(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().knkkSsjyShbt(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().knkkKtmeFlg(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().sjskKbn(),
                output.skinKnkkInf() == null ? null : output.skinKnkkInf().knsJsjkCd(),
                output.hsenHkkKnuKigenYmd());
    }

    /** 出力集約状態。 */
    private static final class OutputAccumulator {

        private boolean dgrpKasFlg;

        private LocalDate dgrpHsenYmd;

        private boolean dgrpSyryKytiTyuFlg;

        private Long dgrpSyryKytiTyuMtstno;

        private boolean dgrpMiRegFlg;

        private boolean dgrpHkhesnJiyuFlg;

        private boolean snyKyktFlg;

        private final List<ComprehensiveOtherMeterInfo> sogoTmtrInfList = new ArrayList<>();

        private OutputAccumulator(LocalDate initialClosedDate) {
            this.dgrpHsenYmd = initialClosedDate;
        }

        private void mergeMeterReadingFlags(
                MeterReadingResultAcquisitionComprehensiveOutput output) {
            if (Boolean.TRUE.equals(output.dgrpSyryKytiTyuFlg())) {
                dgrpSyryKytiTyuFlg = true;
                if (output.dgrpSyryKytiTyuMtstno() != null) {
                    dgrpSyryKytiTyuMtstno = output.dgrpSyryKytiTyuMtstno();
                }
            }
        }

        private void mergeDedicatedFlag(
                ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput output) {
            if (Boolean.TRUE.equals(output.snyKyktFlg())) {
                snyKyktFlg = true;
            }
        }

        private void addMeterInfo(ComprehensiveOtherMeterInfo meterInfo) {
            sogoTmtrInfList.add(meterInfo);
        }

        private ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionOutput toOutput() {
            return new ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionOutput(
                    dgrpKasFlg,
                    dgrpHsenYmd,
                    dgrpSyryKytiTyuFlg,
                    dgrpSyryKytiTyuMtstno,
                    dgrpMiRegFlg,
                    dgrpHkhesnJiyuFlg,
                    snyKyktFlg,
                    List.copyOf(sogoTmtrInfList));
        }
    }
