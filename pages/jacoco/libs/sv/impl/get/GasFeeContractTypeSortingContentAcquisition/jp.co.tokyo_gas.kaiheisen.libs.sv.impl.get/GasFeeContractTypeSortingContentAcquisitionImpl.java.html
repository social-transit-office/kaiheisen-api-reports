package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasFeeContractTypeSortingContentAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasFeeContractTypeSortingContentAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GrkKsbHnsb;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GrkKsbHwsb;

/** ガス料金契約種別振分内容取得サービス実装. */
public final class GasFeeContractTypeSortingContentAcquisitionImpl
        implements GasFeeContractTypeSortingContentAcquisition {

    /** 料金地区コード未指定時の補完値. */
    private static final String DEFAULT_RKN_CHK_CD = "*";

    /** ガス料金契約種別振分データアクセス. */
    private final GrksbhwDataAccess grksbhwDataAccess;

    /**
     * コンストラクタ.
     *
     * @param grksbhwDataAccessArg ガス料金契約種別振分データアクセス
     */
    public GasFeeContractTypeSortingContentAcquisitionImpl(
            final GrksbhwDataAccess grksbhwDataAccessArg) {
        this.grksbhwDataAccess =
                Objects.requireNonNull(
                        grksbhwDataAccessArg,
                        "grksbhwDataAccess");
    }

    @Override
    public GasFeeContractTypeSortingContentAcquisitionOutput execute(
            final String grkKiyShbt,
            final List<GrkKsbHwsb> grkKsbHwsbList,
            final LocalDate knskKjnYmd) {
        List<String> searchGrkKsbHwsbList =
                new ArrayList<>(grkKsbHwsbList.size());
        List<GrkKsbHwsb> normalizedCriteriaList =
                new ArrayList<>(grkKsbHwsbList.size());
        for (GrkKsbHwsb grkKsbHwsb : grkKsbHwsbList) {
            searchGrkKsbHwsbList.add(grkKsbHwsb.grkKsbHwsb());
            normalizedCriteriaList.add(
                    new GrkKsbHwsb(
                            grkKsbHwsb.grkKsbHwsb(),
                            normalizeRknChkCd(grkKsbHwsb.rknChkCd())));
        }

        List<GrksbhwRecord> grksbhwRecords =
                grksbhwDataAccess
                        .findAllByGrkKsbHwsbListAndGrkKiyShbtAndKnskKjnYmd(
                                searchGrkKsbHwsbList,
                                grkKiyShbt,
                                knskKjnYmd);

        if (grksbhwRecords == null) {
            throw new IllegalStateException(
                    "findAllByGrkKsbHwsbListAndGrkKiyShbtAndKnskKjnYmd"
                            + " returned null");
        }

        if (grksbhwRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    grksbhwDataAccess.tableName(),
                    findAllCallSpecIdentifier(),
                    initialQueryParameterJson(
                            grkKiyShbt,
                            searchGrkKsbHwsbList,
                            knskKjnYmd));
        }

        Map<GrkKsbHwsb, GrksbhwRecord> latestRecordByCriteria =
                indexLatestRecordByCriteria(grksbhwRecords);

        List<GrkKsbHnsb> grkKsbHnsbList =
                new ArrayList<>(normalizedCriteriaList.size());
        for (GrkKsbHwsb normalizedCriteria : normalizedCriteriaList) {
            grkKsbHnsbList.add(
                    findMatchedGrkKsbHnsb(
                            grkKiyShbt,
                            normalizedCriteria,
                            knskKjnYmd,
                            latestRecordByCriteria));
        }

        return new GasFeeContractTypeSortingContentAcquisitionOutput(
                List.copyOf(grkKsbHnsbList));
    }

    /**
     * 入力条件に一致する振分内容種別を取得する.
     *
     * @param grkKiyShbt ガス料金契約種別
     * @param normalizedCriteria 正規化済み検索条件
     * @param knskKjnYmd 検索基準年月日
     * @param latestRecordByCriteria 条件ごとの最新振分情報
     * @return ガス料金契約種別振分内容種別情報
     */
    private GrkKsbHnsb findMatchedGrkKsbHnsb(
            final String grkKiyShbt,
            final GrkKsbHwsb normalizedCriteria,
            final LocalDate knskKjnYmd,
            final Map<GrkKsbHwsb, GrksbhwRecord> latestRecordByCriteria) {
        return Optional.ofNullable(
                        latestRecordByCriteria.get(normalizedCriteria))
                .map(record -> new GrkKsbHnsb(record.grkKsbHnsb()))
                .orElseThrow(
                        () ->
                                DataInconsistencySupport.create(
                                        grksbhwDataAccess.tableName(),
                                        findAllCallSpecIdentifier(),
                                        unmatchedParameterJson(
                                                grkKiyShbt,
                                                normalizedCriteria,
                                                knskKjnYmd)));
    }

    /**
     * 振分情報一覧を検索条件ごとの最新レコードに集約する.
     *
     * @param grksbhwRecords 振分情報一覧
     * @return 検索条件ごとの最新レコード
     */
    private static Map<GrkKsbHwsb, GrksbhwRecord> indexLatestRecordByCriteria(
            final List<GrksbhwRecord> grksbhwRecords) {
        Map<GrkKsbHwsb, GrksbhwRecord> latestRecordByCriteria =
                new HashMap<>();
        for (GrksbhwRecord grksbhwRecord : grksbhwRecords) {
            GrkKsbHwsb criteriaKey =
                    new GrkKsbHwsb(
                            grksbhwRecord.grkKsbHwsb(),
                            grksbhwRecord.rknChkCd());
            latestRecordByCriteria.merge(
                    criteriaKey,
                    grksbhwRecord,
                    (current, candidate) ->
                            current.tkysYmd()
                                            .compareTo(candidate.tkysYmd())
                                    >= 0
                                    ? current
                                    : candidate);
        }
        return latestRecordByCriteria;
    }

    /**
     * 一覧取得用の DA identifier を返す.
     *
     * @return DA identifier
     */
    private static String findAllCallSpecIdentifier() {
        return GrksbhwDataAccess.CallSpecs
                .FIND_ALL_BY_GRK_KSB_HWSB_LIST_AND_GRK_KIY_SHBT_AND_KNSK_KJN_YMD
                .identifier();
    }

    /**
     * 料金地区コードを検索用の値へ正規化する.
     *
     * @param rknChkCd 料金地区コード
     * @return 正規化済み料金地区コード
     */
    private static String normalizeRknChkCd(final String rknChkCd) {
        return rknChkCd == null || rknChkCd.isEmpty()
                ? DEFAULT_RKN_CHK_CD
                : rknChkCd;
    }

    /**
     * 初回 DA 呼出し失敗時のエラー出力 JSON を構築する.
     *
     * @param grkKiyShbt ガス料金契約種別
     * @param grkKsbHwsbList ガス料金契約種別振分種別一覧
     * @param knskKjnYmd 検索基準年月日
     * @return JSON ビルダー
     */
    private static FlatJsonBuilder initialQueryParameterJson(
            final String grkKiyShbt,
            final List<String> grkKsbHwsbList,
            final LocalDate knskKjnYmd) {
        return new FlatJsonBuilder()
                .put("grkKiyShbt", grkKiyShbt)
                .put("grkKsbHwsbList", grkKsbHwsbList)
                .put("knskKjnYmd", knskKjnYmd);
    }

    /**
     * 個別照合失敗時のエラー出力 JSON を構築する.
     *
     * @param grkKiyShbt ガス料金契約種別
     * @param normalizedCriteria 正規化済み検索条件
     * @param knskKjnYmd 検索基準年月日
     * @return JSON ビルダー
     */
    private static FlatJsonBuilder unmatchedParameterJson(
            final String grkKiyShbt,
            final GrkKsbHwsb normalizedCriteria,
            final LocalDate knskKjnYmd) {
        return new FlatJsonBuilder()
                .put("grkKiyShbt", grkKiyShbt)
                .put("grkKsbHwsb", normalizedCriteria.grkKsbHwsb())
                .put("rknChkCd", normalizedCriteria.rknChkCd())
                .put("knskKjnYmd", knskKjnYmd);
    }
}
