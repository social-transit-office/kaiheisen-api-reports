package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.retail.RetailTypeJudgmentCode;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dktksry.DktksryDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dktksry.DktksryRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultInfoAcquisitionOutput;

/**
 * 検針結果情報取得サービス実装
 */
public class MeterReadingResultInfoAcquisitionImpl implements MeterReadingResultInfoAcquisition {

    private final KkytskrDataAccess kkytskrDataAccess;

    private final DktksryDataAccess dktksryDataAccess;

    private final String closureReasonTypeSupplyOutage;

    private final String reasonTypeForCreatingMeterReadingResultsSupplyOutage;

    private final String supplyStopPaymentStatusSupplyOutage;

    /**
     * コンストラクタ
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kkytskrDataAccess 供給停止管理情報 データアクセス
     * @param dktksryDataAccess 導管通知確定使用量 データアクセス
     *
     */
    public MeterReadingResultInfoAcquisitionImpl(
            CiriusConstantService ciriusConstantService,
            KkytskrDataAccess kkytskrDataAccess,
            DktksryDataAccess dktksryDataAccess) {

        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.dktksryDataAccess = Objects.requireNonNull(dktksryDataAccess, "dktksryDataAccess");

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.closureReasonTypeSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);

        this.reasonTypeForCreatingMeterReadingResultsSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_OUTAGE);

        this.supplyStopPaymentStatusSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_OUTAGE);
    }

    /**
     * 概要:検針結果情報を取得する
     * 主な仕様:
     * - 各処理を呼び出す
     * - 処理1 recordsearch：KKYTSKR,DKTKSRYの取得を行う
     * - 処理2 kkysSisnJtkbSettings：戻り値の供給停止状態区分を設定する
     * - 処理3 responseResultSetting：戻り値の即検針年月日、ガスメーター指示数、検針結果作成事由種別を設定する
     * @param hesnJysb 閉栓事由種別
     * @param gasMtstNo ガスメーター設置場所番号
     * @param hntKkaCd 判定結果コード
     * @param knsYmd 検針年月日
     * @param gmtShjSu ガスメーター指示数
     * @param knkkSsjyShbt 検針結果作成事由種別
     */
    @Override
    public MeterReadingResultInfoAcquisitionOutput execute(
            String hesnJysb,
            Long gasMtstNo,
            String hntKkaCd,
            LocalDate knsYmd,
            Long gmtShjSu,
            String knkkSsjyShbt) {

        //処理1
        RecordSearchResult recordSearchResult = recordsearch(
                hesnJysb,
                gasMtstNo);
        //処理2
        String responseJtkb = kkysSisnJtkbSetting(
                hesnJysb,
                recordSearchResult.kkytskrRecords(),
                hntKkaCd);
        //処理3
        ResponseResult responseResult = responseResultSetting(
                recordSearchResult.kkytskrRecords(),
                recordSearchResult.dktksryRecords(),
                knsYmd,
                gmtShjSu,
                knkkSsjyShbt);

        return new MeterReadingResultInfoAcquisitionOutput(
                responseJtkb,
                responseResult.responseYmd(),
                responseResult.responseShjSu(),
                responseResult.responseShbt());
    }

    /**
     * 概要:KKYTSKR,DKTKSRYの取得を行う
     * 主な使用
     * - 引数の閉栓事由種別が【J0273 4：供給停止】の場合KKYTSKRの取得を行う
     * - 引数の閉栓事由種別が【J0273 4：供給停止】以外の場合処理なし
     * - KKYTSKRの取得結果がある場合DKTKSRYの取得を行う
     * @param hesnJysb 閉栓事由種別
     * @param gasMtstNo ガスメーター設置場所番号
     * @return kkytskrRecords KKYTSKRレコード
     * @return dktksryRecords DKTKSRYレコード
     */
    private RecordSearchResult recordsearch(String hesnJysb, Long gasMtstNo) {

        KkytskrRecord kkytskrRecords = null;
        DktksryRecord dktksryRecords = null;

        if (closureReasonTypeSupplyOutage.equals(hesnJysb)) {
            Optional<KkytskrRecord> kkytskrRecord = kkytskrDataAccess
                    .findByGasMtstNo(gasMtstNo);
            kkytskrRecords = kkytskrRecord.orElse(null);

            if (kkytskrRecords != null) {
                Optional<DktksryRecord> dktksryRecord = dktksryDataAccess.
                    findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
                            gasMtstNo,
                            kkytskrRecords.hesnYmd(),
                            reasonTypeForCreatingMeterReadingResultsSupplyOutage);
                dktksryRecords = dktksryRecord.orElse(null);
            }
        }
        return new RecordSearchResult(kkytskrRecords, dktksryRecords);
    }

    /**
     * 概要:戻り値の供給停止状態区分を設定する
     * 主な仕様:
     * - 引数の閉栓事由種別が【J0273 4：供給停止】かつ、引数の判定結果コードが〔KH031 1：TG小売〕の場合KKYTSKRで取得した供給停止精算状態区分を設定
     * - 引数の閉栓事由種別が【J0273 4：供給停止】かつ、引数の判定結果コードが〔KH031 1：TG小売〕ではない場合【J0275 3：供給停止】を設定する
     * - 引数の閉栓事由種別が【J0273 4：供給停止】ではない場合未設定を設定する
     * 既知の制約
     * - 未設定はnullとする
     * @param hesnJysb 閉栓事由種別
     * @param kkytskrRecords 供給停止管理レコード
     * @param hntKkaCd 判定結果コード
     * @return 供給停止精算状態区分
     */
    private String kkysSisnJtkbSetting(
            String hesnJysb,
            KkytskrRecord kkytskrRecords,
            String hntKkaCd) {

        if (closureReasonTypeSupplyOutage.equals(hesnJysb)) {
            if (RetailTypeJudgmentCode.TOKYO_GAS.code().equals(hntKkaCd)) {
                return kkytskrRecords.kkysSisnJtkb();
            } else {
                return supplyStopPaymentStatusSupplyOutage;
            }
        } else {
            return null;
        }
    }

    /**
     * 概要:戻り値の即検針年月日、ガスメーター指示数、検針結果作成事由種別を設定する
     * 主な仕様
     * - KKYTSKRの取得結果が存在する場合
     * -  引数の検針年月日＝KKYTSKRで取得した閉栓日の場合かつDKTKSRYの取得結果が存在する場合、引数の検針年月日、ガスメーター指示数、検針結果作成事由を設定
     * -  引数の検針年月日＝KKYTSKRで取得した閉栓日の場合かつDKTKSRYの取得結果が存在しない場合、KKYTSKRで取得した閉栓年月日、KKYTSKRで取得したガスメーター指示数、【07000 133：供給停止】を設定
     * -  引数の検針年月日＞KKYTSKRで取得した閉栓日の場合、引数の検針年月日、ガスメーター指示数、検針結果作成事由を設定
     * -  引数の検針年月日＜KKYTSKRで取得した閉栓日の場合、KKYTSKRで取得した閉栓年月日、KKYTSKRで取得したガスメーター指示数、【07000 133：供給停止】を設定
     * - KKYTSKRの取得結果が存在しない場合、引数の検針年月日、ガスメーター指示数、検針結果作成事由を設定
     * @param kkytskrRecords 供給停止管理レコード
     * @param dktksryRecords 導管通知確定使用量レコード
     * @param knsYmd 検針年月日
     * @param gmtShjSu ガスメーター指示数
     * @param knkkSsjyShbt 検針結果作成事由種別
     * @return responseYmd 戻り値：即検針年月日
     * @return responseShjSu 戻り値：ガスメーター指示数
     * @return responseShbt 戻り値：検針結果作成事由種別
     */
    private ResponseResult responseResultSetting(
            KkytskrRecord kkytskrRecords,
            DktksryRecord dktksryRecords,
            LocalDate knsYmd,
            Long gmtShjSu,
            String knkkSsjyShbt) {

        LocalDate responseYmd = null;
        Long responseShjSu = null;
        String responseShbt = null;

        if (kkytskrRecords != null) {
            if (knsYmd.equals(kkytskrRecords.hesnYmd())) {
                if (dktksryRecords != null) {
                    responseYmd = knsYmd;
                    responseShjSu = gmtShjSu;
                    responseShbt = knkkSsjyShbt;
                    return new ResponseResult(responseYmd, responseShjSu, responseShbt);
                } else {
                    responseYmd = kkytskrRecords.hesnYmd();
                    responseShjSu = kkytskrRecords.gmtShjSu();
                    responseShbt = reasonTypeForCreatingMeterReadingResultsSupplyOutage;
                    return new ResponseResult(responseYmd, responseShjSu, responseShbt);
                }
            } else {
                if (knsYmd.compareTo(kkytskrRecords.hesnYmd()) > 0) {
                    responseYmd = knsYmd;
                    responseShjSu = gmtShjSu;
                    responseShbt = knkkSsjyShbt;
                    return new ResponseResult(responseYmd, responseShjSu, responseShbt);
                } else {
                    responseYmd = kkytskrRecords.hesnYmd();
                    responseShjSu = kkytskrRecords.gmtShjSu();
                    responseShbt = reasonTypeForCreatingMeterReadingResultsSupplyOutage;
                    return new ResponseResult(responseYmd, responseShjSu, responseShbt);
                }
            }
        } else {
            responseYmd = knsYmd;
            responseShjSu = gmtShjSu;
            responseShbt = knkkSsjyShbt;
            return new ResponseResult(responseYmd, responseShjSu, responseShbt);
        }
    }

    /**
     * 概要：recordsearchメソッドの検索結果を保持する戻り値クラス
     *
     * @param kkytskrRecords KkytskrRecordの検索結果（存在しない場合は null）
     * @param dktksryRecords DktksryRecordの検索結果（存在しない場合は null）
     */
    public record RecordSearchResult(
            KkytskrRecord kkytskrRecords,
            DktksryRecord dktksryRecords
    ) { }

    /**
     * 概要：kkysSisnJtkbSettingメソッドの設定値を保持する戻り値クラス
     *
     * @param responseYmd 戻り値：即検針年月日
     * @param responseShjSu 戻り値：ガスメーター指示数
     * @param responseShbt 戻り値：検針結果作成事由種別
     */
    public record ResponseResult(
            LocalDate responseYmd,
            Long responseShjSu,
            String responseShbt
    ) { }
}
