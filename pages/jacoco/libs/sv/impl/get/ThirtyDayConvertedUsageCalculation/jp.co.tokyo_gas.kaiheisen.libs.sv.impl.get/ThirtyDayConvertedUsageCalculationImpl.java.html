package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.math.BigDecimal;
import java.math.MathContext;
import java.math.RoundingMode;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ThirtyDayConvertedUsageCalculation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ThirtyDayConvertedUsageCalculationOutput;

/**
 * ３０日換算後使用量算出サービス実装。
 */
public class ThirtyDayConvertedUsageCalculationImpl implements ThirtyDayConvertedUsageCalculation {

    private static final BigDecimal THIRTY = BigDecimal.valueOf(30L);

    /**
     * 概要: 使用量と使用日数から 30 日換算後使用量を返却する。
     * 主な仕様:
     * - 使用日数に応じた換算値を算出した後、0 以下の場合は 1 を返却する。
     * - 0 より大きい場合は算出した換算値をそのまま返却する。
     * @param syry 使用量
     * @param shyDdsu 使用日数
     * @return ３０日換算後使用量算出結果
     */
    @Override
    public ThirtyDayConvertedUsageCalculationOutput execute(BigDecimal syry, Integer shyDdsu) {
        BigDecimal convertedUsage = calculateConvertedUsage(syry, shyDdsu);
        BigDecimal outputUsage = resolveOutputUsage(convertedUsage);
        return new ThirtyDayConvertedUsageCalculationOutput(outputUsage);
    }

    /**
     * 概要: 使用日数に応じた 30 日換算使用量を算出する。
     * 主な仕様:
     * - 使用日数が 30 の場合は入力使用量をそのまま採用する。
     * - 使用日数が 0 の場合は入力使用量を 30 倍する。
     * - 上記以外は入力使用量を 30 倍して使用日数で除算する。
     * @param syry 使用量
     * @param shyDdsu 使用日数
     * @return 30 日換算使用量
     */
    private BigDecimal calculateConvertedUsage(BigDecimal syry, Integer shyDdsu) {
        int usageDays = shyDdsu.intValue();
        if (usageDays == 30) {
            return syry;
        }
        if (usageDays == 0) {
            return syry.multiply(THIRTY);
        }
        return syry.multiply(THIRTY)
                .divide(BigDecimal.valueOf(usageDays), MathContext.DECIMAL128);
    }

    /**
     * 概要: 設計書の下限制御に従って戻り値を確定する。
     * 主な仕様:
     * - 換算使用量が 0 以下の場合は 1 を返却する。
     * - 換算使用量が正の場合は換算使用量をそのまま返却する。
     * @param convertedUsage 30 日換算使用量
     * @return 戻り値に設定する 30 日換算後使用量
     */
    private BigDecimal resolveOutputUsage(BigDecimal convertedUsage) {
        if (convertedUsage.compareTo(BigDecimal.ZERO) <= 0) {
            return BigDecimal.ONE;
        }
        return convertedUsage.setScale(0, RoundingMode.DOWN);
    }
}
