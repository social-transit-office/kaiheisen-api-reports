package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ServiceContractNumberAcquisition;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ServiceContractNumberAcquisitionOutput;

/** サービス契約番号取得サービス実装。 */
public class ServiceContractNumberAcquisitionImpl implements ServiceContractNumberAcquisition {

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final String rkkiKnrnJtkbInvalid;

    private final String rkkiKnrnJtkbRelatedValid;

    private final String rkkiKnrnJtkbRelatedEndWaiting;

    private final String rkkiKnrnJtkbRelatedEnd;

    private final String dedicatedGasMeterStatus;

    private final List<String> rkkiKnrnJtkbListForShrKbn2;

    private final List<String> rkkiKnrnJtkbListForShrKbn3;

    /**
     * コンストラクタ。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ServiceContractNumberAcquisitionImpl(
            SvmstkrDataAccess svmstkrDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.rkkiKnrnJtkbInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.INVALID);
        this.rkkiKnrnJtkbRelatedValid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.rkkiKnrnJtkbRelatedEndWaiting =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.rkkiKnrnJtkbRelatedEnd =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.dedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                                .DEDICATED_GAS_METER);

        this.rkkiKnrnJtkbListForShrKbn2 =
                List.of(rkkiKnrnJtkbRelatedValid, rkkiKnrnJtkbRelatedEndWaiting);
        this.rkkiKnrnJtkbListForShrKbn3 =
                List.of(
                        rkkiKnrnJtkbRelatedValid,
                        rkkiKnrnJtkbRelatedEndWaiting,
                        rkkiKnrnJtkbRelatedEnd);
    }

    @Override
    public ServiceContractNumberAcquisitionOutput execute(
            Long gasMtstNo,
            Long mtgMtstNo,
            String shrKbn,
            String ckrtGmscJtkb,
            LocalDate knskKjnYmd) {
        Optional<QueryResult> primaryQueryResult = findByShrKbn(gasMtstNo, knskKjnYmd, shrKbn);
        if (primaryQueryResult.isPresent()) {
            Optional<SvmstkrRecord> primaryRecord = primaryQueryResult.orElseThrow().record();
            if (primaryRecord.isPresent()) {
                return toOutput(primaryRecord.orElseThrow().svkyNo());
            }
        }

        SerialSelection serialSelection =
                resolveSerialSelection(gasMtstNo, mtgMtstNo, knskKjnYmd, ckrtGmscJtkb);

        if (!dedicatedGasMeterStatus.equals(serialSelection.ckrtGmscJtkb())) {
            return handleNonDedicatedSerialStatus(primaryQueryResult);
        }

        Optional<QueryResult> serialQueryResult =
                findByShrKbn(serialSelection.mtgMtstNo(), knskKjnYmd, shrKbn);
        if (serialQueryResult.isEmpty()) {
            return toOutput(null);
        }
        QueryResult resolvedSerialQueryResult = serialQueryResult.orElseThrow();
        if (resolvedSerialQueryResult.record().isPresent()) {
            return toOutput(resolvedSerialQueryResult.record().orElseThrow().svkyNo());
        }

        throw DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                resolvedSerialQueryResult.callSpec().identifier(),
                resolvedSerialQueryResult.parameterJsonBuilder());
    }

    private ServiceContractNumberAcquisitionOutput handleNonDedicatedSerialStatus(
            Optional<QueryResult> primaryQueryResult) {
        if (primaryQueryResult.isEmpty()) {
            return toOutput(null);
        }

        QueryResult resolvedQueryResult = primaryQueryResult.orElseThrow();
        throw DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                resolvedQueryResult.callSpec().identifier(),
                resolvedQueryResult.parameterJsonBuilder());
    }

    private Optional<QueryResult> findByShrKbn(Long mtstNo, LocalDate knskKjnYmd, String shrKbn) {
        if (ServiceContractNumberAcquisitionConstants.ShrKbn.ACTIVE.equals(shrKbn)) {
            return Optional.of(
                    new QueryResult(
                            SvmstkrDataAccess.CallSpecs
                                    .FIND_BY_MTST_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB,
                            new FlatJsonBuilder()
                                    .put("mtstNo", mtstNo)
                                    .put("knskKjnYmd", knskKjnYmd)
                                    .put("rkkiKnrnJtkb", rkkiKnrnJtkbInvalid),
                            svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                                    mtstNo, knskKjnYmd, rkkiKnrnJtkbInvalid)));
        }
        if (ServiceContractNumberAcquisitionConstants.ShrKbn.NEAREST.equals(shrKbn)) {
            return Optional.of(
                    new QueryResult(
                            SvmstkrDataAccess.CallSpecs
                                    .FIND_BY_MTST_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB_LIST,
                            new FlatJsonBuilder()
                                    .put("mtstNo", mtstNo)
                                    .put("knskKjnYmd", knskKjnYmd)
                                    .put("rkkiKnrnJtkbList", rkkiKnrnJtkbListForShrKbn2),
                            svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                                    mtstNo, knskKjnYmd, rkkiKnrnJtkbListForShrKbn2)));
        }
        if (ServiceContractNumberAcquisitionConstants.ShrKbn.ACTIVE_OR_END.equals(shrKbn)) {
            return Optional.of(
                    new QueryResult(
                            SvmstkrDataAccess.CallSpecs
                                    .FIND_BY_MTST_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB_LIST,
                            new FlatJsonBuilder()
                                    .put("mtstNo", mtstNo)
                                    .put("knskKjnYmd", knskKjnYmd)
                                    .put("rkkiKnrnJtkbList", rkkiKnrnJtkbListForShrKbn3),
                            svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                                    mtstNo, knskKjnYmd, rkkiKnrnJtkbListForShrKbn3)));
        }
        return Optional.empty();
    }

    private SerialSelection resolveSerialSelection(
            Long gasMtstNo, Long mtgMtstNo, LocalDate knskKjnYmd, String ckrtGmscJtkb) {
        if (!isBlankSerialStatus(ckrtGmscJtkb)) {
            return new SerialSelection(ckrtGmscJtkb, mtgMtstNo);
        }

        Optional<GmtsetdRecord> gmtsetdRecord =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd);
        if (gmtsetdRecord.isEmpty()) {
            throw gmtsetdDataInconsistency(gasMtstNo, knskKjnYmd);
        }

        GmtsetdRecord resolvedRecord = gmtsetdRecord.orElseThrow();
        return new SerialSelection(resolvedRecord.ckrtGmscJtkb(), resolvedRecord.mtgMtstNo());
    }

    private static boolean isBlankSerialStatus(String ckrtGmscJtkb) {
        return ckrtGmscJtkb == null || ckrtGmscJtkb.isEmpty() || " ".equals(ckrtGmscJtkb);
    }

    private DataInconsistencyException gmtsetdDataInconsistency(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        return DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD
                        .identifier(),
                new FlatJsonBuilder().put("gasMtstNo", gasMtstNo).put("knskKjnYmd", knskKjnYmd));
    }

    private static ServiceContractNumberAcquisitionOutput toOutput(Long svkyNo) {
        return new ServiceContractNumberAcquisitionOutput(svkyNo);
    }

    private record SerialSelection(String ckrtGmscJtkb, Long mtgMtstNo) { }

    private record QueryResult(
            DataAccessCallSpec callSpec,
            FlatJsonBuilder parameterJsonBuilder,
            Optional<SvmstkrRecord> record) { }
}
