package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.BuildingKanjiDisplayFlagAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.BuildingKanjiDisplayFlagAcquisitionOutput;

/** 建物漢字表示可否フラグ取得サービス実装。 */
public class BuildingKanjiDisplayFlagAcquisitionImpl implements BuildingKanjiDisplayFlagAcquisition {
    private final String bldKjHyojihukaFlg;

    private final String bldKjHyojikaFlg;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS 建物漢字表示可否フラグ取得サービス
     */
    public BuildingKanjiDisplayFlagAcquisitionImpl(CiriusConstantService ciriusConstantService) {

        // コンストラクタの初期化
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(ciriusConstantService,
                "ciriusConstantService");

        this.bldKjHyojihukaFlg = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.BuildingKanjiDisplayFlag.TYPE,
                ConstantLookupKeys.Cirius.BuildingKanjiDisplayFlag.NOT_DISPLAYABLE);

        this.bldKjHyojikaFlg = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.BuildingKanjiDisplayFlag.TYPE,
                ConstantLookupKeys.Cirius.BuildingKanjiDisplayFlag.CAN_BE_DISPLAYED);

    }

    /**
     * 概要: 引数の建物名漢字をもとに建物漢字表示可否コードを取得する 主な仕様: -
     * 建物名漢字が未設定ではない場合、bldKjHyojiCd(true)を返し、未設定の場合、bldKjHyojiCd(false)を返す。
     *
     * @param bldnKj 建物名漢字
     * @return bldKjHyojiCd建物漢字表示可否コード
     */

    @Override
    public BuildingKanjiDisplayFlagAcquisitionOutput execute(String bldnKj) {

        // フラグをboolean型に変換
        Boolean bldKjHyojihukaFlg1 = "1".equals(bldKjHyojihukaFlg);

        Boolean bldKjHyojikaFlg1 = "1".equals(bldKjHyojikaFlg);

        // 建物名漢字が未設定の場合
        if (bldnKj == null || bldnKj.isBlank()) {
            return new BuildingKanjiDisplayFlagAcquisitionOutput(bldKjHyojihukaFlg1);
            // 建物名漢字に値が入っている場合
        } else {
            return new BuildingKanjiDisplayFlagAcquisitionOutput(bldKjHyojikaFlg1);
        }

    }
}
