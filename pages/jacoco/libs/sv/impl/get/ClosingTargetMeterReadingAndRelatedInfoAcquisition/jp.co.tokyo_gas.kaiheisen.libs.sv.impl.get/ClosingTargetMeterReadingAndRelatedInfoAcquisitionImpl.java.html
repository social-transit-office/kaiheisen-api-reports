/**
 * 概要: SV-122（閉栓対象メーターに紐づく元メーターと専用メーターの検針結果＆検針関連情報取得）の実装クラス。
 * 主な仕様:
 * - 直列ガスメーター設置状態区分に応じて元メーター分岐、専用メーター分岐、処理なし分岐を切り替える。
 * - 元メーター分岐では DA-033-R007 で専用メーター候補を取得し、各件に対して SV-151 を呼び出して出力を構成する。
 * - 専用メーター分岐では SV-152 を呼び出して元メーター情報を構成する。
 * 既知の制約:
 * - systemDatetime は RequestContext から取得する。
 * - public 引数の null ガードは SV 規約に従い実装しない。
 */
public final class ClosingTargetMeterReadingAndRelatedInfoAcquisitionImpl
        implements ClosingTargetMeterReadingAndRelatedInfoAcquisition {

    private final MeterReadingResultAcquisitionForOriginMeterDedicated originMeterDedicatedService;

    private final MeterReadingResultAcquisitionForDedicatedMeterOrigin dedicatedMeterOriginService;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final RequestContext requestContext;

    private final String originalGasMeterStatus;

    private final String dedicatedGasMeterStatus;

    private final String openingStatus;

    private final String securityClosureReason;

    private final String disasterClosureReason;

    /**
     * 概要: 依存サービスと必須定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 依存 SV / DA / RequestContext / 定数サービスの null を許容せず fail-fast する。
     * - J0244、02100、J0273 の必須判定値を保持する。
     *
     * @param originMeterDedicatedService SV-151
     * @param dedicatedMeterOriginService SV-152
     * @param gmtsetdDataAccess DA-033
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public ClosingTargetMeterReadingAndRelatedInfoAcquisitionImpl(
            MeterReadingResultAcquisitionForOriginMeterDedicated originMeterDedicatedService,
            MeterReadingResultAcquisitionForDedicatedMeterOrigin dedicatedMeterOriginService,
            GmtsetdDataAccess gmtsetdDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.originMeterDedicatedService = Objects.requireNonNull(
                originMeterDedicatedService,
                "originMeterDedicatedService");
        this.dedicatedMeterOriginService = Objects.requireNonNull(
                dedicatedMeterOriginService,
                "dedicatedMeterOriginService");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.securityClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE);
        this.disasterClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
    }

    /**
     * 概要: SV-122 を実行し、閉栓対象メーターに紐づく元メーター系・専用メーター系の検針結果と関連情報を返却する。
     * 主な仕様:
     * - 元ガスメーターの場合は DA-033-R007 と SV-151 を利用する。
     * - 専用ガスメーターの場合は SV-152 を利用する。
     * - それ以外は処理なしとして空の出力を返す。
     */
    @Override
    public ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput execute(
            String ckrtGmscJtkb,
            String szcGkkCd,
            String szcCmazCd,
            String szcMchCd,
            Long mtgMtstNo,
            Long gasMtstNo,
            String gmtKhsJtkb,
            String hesnJysb,
            String kngmStbJtkb,
            LocalDate hssgYmd) {
        if (originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return executeForOriginMeter(
                    szcGkkCd,
                    szcCmazCd,
                    szcMchCd,
                    mtgMtstNo,
                    gasMtstNo,
                    hssgYmd,
                                        requireSystemDatetime().toLocalDate());
        }
        if (dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            return executeForDedicatedMeter(kngmStbJtkb, mtgMtstNo);
        }
        return emptyOutput();
    }

    private ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput executeForOriginMeter(
            String szcGkkCd,
            String szcCmazCd,
            String szcMchCd,
            Long mtgMtstNo,
            Long gasMtstNo,
            LocalDate hssgYmd,
            LocalDate searchDate) {
        List<GmtsetdRecord> dedicatedMeterRecords = gmtsetdDataAccess
                .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                        szcGkkCd,
                        szcCmazCd,
                        szcMchCd,
                        gasMtstNo,
                        gasMtstNo,
                        searchDate);

        boolean openingDedicatedMeterExists = false;
        boolean futureClosingDateExists = false;
        boolean restrictedClosureReasonExists = false;
        boolean dedicatedMeterReadAfterClosingExists = false;
        boolean dedicatedContractExists = false;
        boolean dedicatedUsageAgreementExists = false;
        Long dedicatedUsageAgreementMeterLocationNumber = null;
        List<SkinKnkkInf> recentMeterReadingList = new ArrayList<>();
        List<SnyInf> dedicatedMeterInfoList = new ArrayList<>();

        for (GmtsetdRecord dedicatedMeterRecord : dedicatedMeterRecords) {
            dedicatedMeterInfoList.add(mapSnyInf(dedicatedMeterRecord));
            openingDedicatedMeterExists |= openingStatus.equals(dedicatedMeterRecord.gmtKhsJtkb());
            futureClosingDateExists |= dedicatedMeterRecord.hesnYmd() != null
                    && dedicatedMeterRecord.hesnYmd().isAfter(hssgYmd);
            restrictedClosureReasonExists |= securityClosureReason.equals(dedicatedMeterRecord.hesnJysb())
                    || disasterClosureReason.equals(dedicatedMeterRecord.hesnJysb());

            MeterReadingResultAcquisitionForOriginMeterDedicatedOutput childOutput =
                    originMeterDedicatedService.execute(
                            dedicatedMeterRecord.kngmStbJtkb(),
                            dedicatedMeterRecord.gasMtstNo(),
                            hssgYmd);
            if (childOutput.skinKnkkInf() != null) {
                recentMeterReadingList.add(childOutput.skinKnkkInf());
            }
            dedicatedMeterReadAfterClosingExists |= Boolean.TRUE.equals(childOutput.snyHsngKnsFlg());
            dedicatedContractExists |= Boolean.TRUE.equals(childOutput.snyKyktFlg());
            dedicatedUsageAgreementExists |= Boolean.TRUE.equals(childOutput.snySyryKytiTyuFlg());
            if (dedicatedUsageAgreementMeterLocationNumber == null
                    && childOutput.snySyryKytiTyuMtstno() != null) {
                dedicatedUsageAgreementMeterLocationNumber = childOutput.snySyryKytiTyuMtstno();
            }
        }

        return new ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput(
                openingDedicatedMeterExists,
                futureClosingDateExists,
                restrictedClosureReasonExists,
                new KnsKkaSnyInf(
                        List.copyOf(recentMeterReadingList),
                        dedicatedMeterReadAfterClosingExists,
                        dedicatedContractExists,
                        dedicatedUsageAgreementExists,
                        dedicatedUsageAgreementMeterLocationNumber),
                emptyOriginMeterInfo(),
                new GmtsetdSnyInf(List.copyOf(dedicatedMeterInfoList)));
    }

    private ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput executeForDedicatedMeter(
            String kngmStbJtkb,
            Long mtgMtstNo) {
        MeterReadingResultAcquisitionForDedicatedMeterOriginOutput childOutput =
                dedicatedMeterOriginService.execute(kngmStbJtkb, mtgMtstNo);
        return new ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput(
                Boolean.FALSE,
                Boolean.FALSE,
                Boolean.FALSE,
                emptyDedicatedMeterInfo(),
                new KnsKkaMtInf(
                        childOutput.skinKnkkInf(),
                        Boolean.TRUE.equals(childOutput.mtgSyryKytiTyuFlg()),
                        childOutput.mtgSyryKytiTyuMtstno()),
                new GmtsetdSnyInf(List.of()));
    }

    private ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput emptyOutput() {
        return new ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput(
                Boolean.FALSE,
                Boolean.FALSE,
                Boolean.FALSE,
                emptyDedicatedMeterInfo(),
                emptyOriginMeterInfo(),
                new GmtsetdSnyInf(List.of()));
    }

    private KnsKkaSnyInf emptyDedicatedMeterInfo() {
        return new KnsKkaSnyInf(List.of(), Boolean.FALSE, Boolean.FALSE, Boolean.FALSE, null);
    }

    private KnsKkaMtInf emptyOriginMeterInfo() {
        return new KnsKkaMtInf(null, Boolean.FALSE, null);
    }

    private java.time.LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime().orElseThrow(
                () -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
