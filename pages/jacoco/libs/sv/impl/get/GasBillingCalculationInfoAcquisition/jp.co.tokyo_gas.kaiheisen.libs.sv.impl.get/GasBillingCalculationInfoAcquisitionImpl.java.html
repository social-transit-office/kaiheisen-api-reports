package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasBillingCalculationInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasBillingCalculationInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GasRknKsnInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KkyCushInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KnsKkaInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.ZnkiKnkkInf;

/** ガス使用量指針間情報取得（前回検針日キー）サービス実装。 */
public class GasBillingCalculationInfoAcquisitionImpl
        implements GasBillingCalculationInfoAcquisition {

    private final String openingReasonCode;

    private final String disasterRestorationReasonCode;

    private final String securityReleaseReasonCode;

    private final String disasterClosureReasonCode;

    private final String securityClosureReasonCode;

    /**
     * 概要:
     * - 業務判定で利用する必須 CIRIUS 定数を初期化する。
     *
     * 主な仕様:
     * - 検針結果作成事由種別の必須定数を fail-fast で解決する。
     *
     * 既知の制約:
     * - 必須定数未定義時は RequiredConstantResolver の例外が送出される。
     *
    * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public GasBillingCalculationInfoAcquisitionImpl(CiriusConstantService ciriusConstantService) {
        this.openingReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.OPENING);
        this.disasterRestorationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .RESTORATION_OF_SHUTOFF_IN_CASE_OF_DISASTER);
        this.securityReleaseReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .SECURITY_RELEASE);
        this.disasterClosureReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .CLOSED_IN_CASE_OF_DISASTER);
        this.securityClosureReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .SECURITY_CLOSURE);
    }

    /**
     * 概要: 前回検針関連情報と供給中止/検針結果情報からガス料金計算用情報を返却する。
     * 主な仕様:
     * - 供給中止情報リストを優先して供給中止日数を算出する。
     * - 供給中止日数が 0 のままの場合のみ検針結果情報リストで再算出する。
     *
     * @param knsYmd 検針年月日
     * @param znkiKnkkInf 前回検針関連情報
     * @param kkyCushInfList 供給中止情報リスト
     * @param knsKkaInfList 検針結果情報リスト
     * @param gasRknKsnInf ガス料金計算用情報
     * @return ガス料金計算用情報取得結果
     */
    @Override
    public GasBillingCalculationInfoAcquisitionOutput execute(
            LocalDate knsYmd,
            ZnkiKnkkInf znkiKnkkInf,
            List<KkyCushInf> kkyCushInfList,
            List<KnsKkaInf> knsKkaInfList,
            GasRknKsnInf gasRknKsnInf) {
        int supplyStopDays =
            calculateSupplyStopDays(
                knsYmd,
                znkiKnkkInf,
                kkyCushInfList,
                knsKkaInfList,
                gasRknKsnInf);

        return new GasBillingCalculationInfoAcquisitionOutput(
                new GasRknKsnInf(
                        null,
                        null,
                        null,
                        null,
                        null,
                        supplyStopDays,
                        znkiKnkkInf.tksuSvkyNo(),
                        null));
    }

    /**
     * 概要:
     * - 供給中止情報リストと検針結果情報リストから供給中止日数を算出する。
     *
     * 主な仕様:
     * - 供給中止情報リストの閉栓/閉栓解除系パターンを優先して算出する。
     * - 算出結果が 0 の場合のみ検針結果情報リストで再算出する。
     *
     * 既知の制約:
     * - 供給中止日数ワークは各レコード処理開始時に 0 へ初期化する
     *
     * @param knsYmd 検針年月日
     * @param znkiKnkkInf 前回検針関連情報
     * @param kkyCushInfList 供給中止情報リスト
     * @param knsKkaInfList 検針結果情報リスト
     * @param gasRknKsnInf ガス料金計算用情報
     * @return 算出した供給中止日数
     */
    private int calculateSupplyStopDays(
            LocalDate knsYmd,
            ZnkiKnkkInf znkiKnkkInf,
            List<KkyCushInf> kkyCushInfList,
            List<KnsKkaInf> knsKkaInfList,
            GasRknKsnInf gasRknKsnInf) {
        int supplyStopDays = 0;
        int supplyStopRecordCount = kkyCushInfList.size();

        for (int index = 0; index < supplyStopRecordCount; index++) {
            int supplyStopDaysWork = 0;
            KkyCushInf currentRecord = kkyCushInfList.get(index);

            if (shouldCalculateSupplyStopDate(kkyCushInfList, index)) {
                LocalDate supplyStopDate = resolveSupplyStopDate(currentRecord, znkiKnkkInf);
                supplyStopDaysWork = calculateDaysBetween(supplyStopDate, knsYmd);
            }
            if (isLastRecord(index, supplyStopRecordCount)
                    && isReleaseReason(currentRecord.knkkSsjyShbt())) {
                supplyStopDaysWork =
                        calculateDaysBetween(gasRknKsnInf.kokiRkksYmd(), currentRecord.knsYmd());
            }
            supplyStopDays += supplyStopDaysWork;
        }

        if (supplyStopDays != 0) {
            return supplyStopDays;
        }
        return recalculateSupplyStopDaysFromMeterReadingResults(
                knsYmd, knsKkaInfList, gasRknKsnInf);
    }

    /**
     * 概要: 供給中止情報から供給中止日算出対象かを判定する。 主な仕様: - 1レコード目は閉栓系事由であれば対象とする。 -
     * 2レコード目以降は、現在が閉栓系かつ直前が閉栓解除系の場合のみ対象とする。 既知の制約: - リスト順は設計書どおりの時系列で渡される前提とする。
     *
     * @param kkyCushInfList 供給中止情報リスト
     * @param index 現在処理中の添字
     * @return 算出対象判定結果
     */
    private boolean shouldCalculateSupplyStopDate(List<KkyCushInf> kkyCushInfList, int index) {
        KkyCushInf currentRecord = kkyCushInfList.get(index);
        if (!isClosureReason(currentRecord.knkkSsjyShbt())) {
            return false;
        }
        if (index == 0) {
            return true;
        }
        return isReleaseReason(kkyCushInfList.get(index - 1).knkkSsjyShbt());
    }

    /**
     * 概要: 前回検針関連情報と供給中止情報から供給中止日を算出する。 主な仕様: - 前回検針関連情報が開栓かつ検針年月日不一致の場合は 2 日後を返す。 - 上記以外は 1 日後を返す。
     * 既知の制約: - 日付加算は LocalDate.plusDays を使用する。
     *
     * @param currentRecord 現在の供給中止情報
     * @param znkiKnkkInf 前回検針関連情報
     * @return 算出した供給中止日
     */
    private LocalDate resolveSupplyStopDate(KkyCushInf currentRecord, ZnkiKnkkInf znkiKnkkInf) {
        if (openingReasonCode.equals(normalize(znkiKnkkInf.knkkSsjyShbt()))
                && !znkiKnkkInf.knsYmd().isEqual(currentRecord.knsYmd())) {
            return currentRecord.knsYmd().plusDays(2);
        }
        return currentRecord.knsYmd().plusDays(1);
    }

    /**
     * 概要: 検針結果情報リストを用いて供給中止日数を再算出する。 主な仕様: - 閉栓系事由を検知した時点で今回料金起算年月日から検針年月日までの日数を返す。 -
     * 閉栓解除系事由を検知した時点で繰り返しを中断し 0 を返す。 既知の制約: - 供給中止情報リストで算出できた場合は本メソッドは呼ばれない。
     *
     * @param knsYmd 検針年月日
     * @param knsKkaInfList 検針結果情報リスト
     * @param gasRknKsnInf ガス料金計算用情報
     * @return 再算出した供給中止日数
     */
    private int recalculateSupplyStopDaysFromMeterReadingResults(
            LocalDate knsYmd,
            List<KnsKkaInf> knsKkaInfList,
            GasRknKsnInf gasRknKsnInf) {
        int meterReadingResultCount = knsKkaInfList.size();

        for (int index = 0; index < meterReadingResultCount; index++) {
            KnsKkaInf currentRecord = knsKkaInfList.get(index);
            if (isClosureReason(currentRecord.knkkSsjyShbt())) {
                return calculateDaysBetween(gasRknKsnInf.kokiRkksYmd(), knsYmd);
            }
            if (isReleaseReason(currentRecord.knkkSsjyShbt())) {
                break;
            }
        }
        return 0;
    }

    /**
     * 概要: 閉栓系事由かを判定する。 主な仕様: - 災害時閉栓または保安閉栓のみ true を返す。 既知の制約: - 比較前に trim と空値 null 化を行う。
     *
     * @param reasonCode 検針結果作成事由種別
     * @return 閉栓系判定結果
     */
    private boolean isClosureReason(String reasonCode) {
        String normalizedReasonCode = normalize(reasonCode);
        return disasterClosureReasonCode.equals(normalizedReasonCode)
                || securityClosureReasonCode.equals(normalizedReasonCode);
    }

    /**
     * 概要: 閉栓解除系事由かを判定する。 主な仕様: - 災害時閉栓復旧または保安閉栓解除のみ true を返す。 既知の制約: - 比較前に trim と空値 null 化を行う。
     *
     * @param reasonCode 検針結果作成事由種別
     * @return 閉栓解除系判定結果
     */
    private boolean isReleaseReason(String reasonCode) {
        String normalizedReasonCode = normalize(reasonCode);
        return disasterRestorationReasonCode.equals(normalizedReasonCode)
                || securityReleaseReasonCode.equals(normalizedReasonCode);
    }

    /**
     * 概要: 文字列を trim し、空値を null に正規化する。 主な仕様: - null は null のまま返す。 - trim 後に空文字の場合は null を返す。 既知の制約:
     * - 空白除去の挙動は Java 標準の trim に依存する。
     *
     * @param value 正規化対象
     * @return 正規化結果
     */
    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }

    /**
     * 概要: 2つの日付の差分日数を算出する。 主な仕様: - 開始日から終了日までの日数差を返す。 既知の制約: - 負数の差分もそのまま返却する。
     *
     * @param from 開始日
     * @param to 終了日
     * @return 日数差
     */
    private static int calculateDaysBetween(LocalDate from, LocalDate to) {
        return Math.toIntExact(ChronoUnit.DAYS.between(from, to));
    }

    /**
     * 概要: 現在レコードが最終走査レコードかを判定する。 主な仕様: - 走査件数ベースで末尾要素を判定する。 既知の制約: - count が 0 の場合は呼び出されない前提とする。
     *
     * @param index 現在添字
     * @param count 走査件数
     * @return 最終レコード判定結果
     */
    private static boolean isLastRecord(int index, int count) {
        return index == count - 1;
    }
}
