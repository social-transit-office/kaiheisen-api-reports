package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.SupplySuspensionSettlementStatusAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.SupplySuspensionSettlementStatusAcquisitionOutput;

/**
 * 供給停止精算状態区分取得サービス実装。
 */
public class SupplySuspensionSettlementStatusAcquisitionImpl implements SupplySuspensionSettlementStatusAcquisition {

    private static final String BLANK = "";

    private final String closureReasonTypeSupplyOutage;

    private final String retailerCodeTokyoGasCoLtd;

    private final String supplyStopPaymentStatusSupplyOutage;

    /**
     * コンストラクタ
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     *
     */
    public SupplySuspensionSettlementStatusAcquisitionImpl(CiriusConstantService ciriusConstantService) {

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.closureReasonTypeSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);

        this.retailerCodeTokyoGasCoLtd = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);

        this.supplyStopPaymentStatusSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_OUTAGE);
    }

    /**
     * 概要: 閉栓事由種別と託送供給依頼事業者コードを判定し供給停止精算状態区分を取得する
     * 主な仕様:
     * - 引数の閉栓事由種別が【J0273 4：供給停止】以外の場合 戻り値にBLANKを設定し返却
     * - 引数の閉栓事由種別が【J0273 4：供給停止】　託送サービスデータ有無フラグがtrue 託送供給依頼事業者コードが【J0580 A0012：東京ガス株式会社】の場合 戻り値に引数の供給停止精算状態区分を設定し返却
     * - 上記以外の場合 戻り値に【J0275 3：供給停止】を設定して返却
     * 既知の制約:
     * 各引数に対する null チェックや補正処理は行わない
     * @param hesnJysb 閉栓事由種別
     * @param tkssvkyUmFlg 託送サービス契約有無フラグ
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param　kkysSisnJtkb 供給停止精算状態区分
     * @return 供給停止精算状態区分
     */
    @Override
    public SupplySuspensionSettlementStatusAcquisitionOutput execute(
            String hesnJysb,
            Boolean tkssvkyUmFlg,
            String tksuKijsCd,
            String kkysSisnJtkb) {

        if (!closureReasonTypeSupplyOutage.equals(hesnJysb)) {
            return new SupplySuspensionSettlementStatusAcquisitionOutput(BLANK);
        } else {
            if (tkssvkyUmFlg.equals(true) && retailerCodeTokyoGasCoLtd.equals(tksuKijsCd)) {
                return new SupplySuspensionSettlementStatusAcquisitionOutput(kkysSisnJtkb);
            } else {
                return new SupplySuspensionSettlementStatusAcquisitionOutput(supplyStopPaymentStatusSupplyOutage);
            }
        }
    }
}
