package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisitionComprehensive;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.MeterReadingResultAcquisitionComprehensiveOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.SkinKnkkInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForClosing;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForClosingOutput;

/**
 * 概要: SV-153（検針結果＆検針関連情報取得_総合）の実装クラス。
 * 主な仕様:
 * - メーター保有状態のときのみ SV-162 を呼び出して最近検針結果情報を取得する。
 * - DA-063-R001 の結果を使い、同一グループ内使用量協定中フラグを判定する。
 */
public final class MeterReadingResultAcquisitionComprehensiveImpl
        implements MeterReadingResultAcquisitionComprehensive {

    private final MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing;

    private final KnsnkrnDataAccess knsnkrnDataAccess;

    private final String meterPossessionStatus;

    /**
     * 概要: 依存サービスと必須定数を解決してインスタンスを初期化する。
     *
     * @param meterReadingResultAcquisitionForClosing SV-162
     * @param knsnkrnDataAccess DA-063
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public MeterReadingResultAcquisitionComprehensiveImpl(
            MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing,
            KnsnkrnDataAccess knsnkrnDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.meterReadingResultAcquisitionForClosing = Objects.requireNonNull(
                meterReadingResultAcquisitionForClosing,
                "meterReadingResultAcquisitionForClosing");
        this.knsnkrnDataAccess = Objects.requireNonNull(knsnkrnDataAccess, "knsnkrnDataAccess");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.meterPossessionStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
    }

    /**
     * 概要: SV-153 を実行し、総合契約メーター向けの検針結果関連情報を返却する。
     *
     * @param systemDateKngmStbJtkb システム日時点供内管・ガスメーター設備状態区分
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 総合契約メーター向けの検針結果関連情報
     */
    @Override
    public MeterReadingResultAcquisitionComprehensiveOutput execute(
            String systemDateKngmStbJtkb,
            Long gasMtstNo) {
        SkinKnkkInf recentMeterReading = null;
        LocalDate closingReportDeadline = null;
        if (meterPossessionStatus.equals(systemDateKngmStbJtkb)) {
            MeterReadingResultAcquisitionForClosingState childState =
                    executeMeterReadingResultAcquisitionForClosing(gasMtstNo);
            if (!childState.startDateIdentificationError()) {
                MeterReadingResultAcquisitionForClosingOutput childOutput = childState.output();
                recentMeterReading = mapSkinKnkkInf(childOutput.skinKnkkInf());
                closingReportDeadline = childOutput.hsenHkkKnuKigenYmd();
            }
        }

        KnsnkrnRecord knsnkrnRecord = knsnkrnDataAccess.findByGasMtstNo(gasMtstNo).orElse(null);
        boolean usageAgreementInProgressFlag =
                knsnkrnRecord != null && Boolean.TRUE.equals(knsnkrnRecord.gmsrKtumFlg());
        Long usageAgreementMeterLocationNumber = usageAgreementInProgressFlag ? gasMtstNo : null;

        return new MeterReadingResultAcquisitionComprehensiveOutput(
                recentMeterReading,
                closingReportDeadline,
                usageAgreementInProgressFlag,
                usageAgreementMeterLocationNumber);
    }

    private MeterReadingResultAcquisitionForClosingState executeMeterReadingResultAcquisitionForClosing(
            Long gasMtstNo) {
        try {
            return MeterReadingResultAcquisitionForClosingState.success(
                    requireChildOutput(meterReadingResultAcquisitionForClosing.execute(gasMtstNo)));
        } catch (KaiheisenException exception) {
            if (ErrorCodeConstants.ERROR_CODE_020000074.equals(exception.getErrorCode())) {
                return MeterReadingResultAcquisitionForClosingState.startDateIdentificationErrorState();
            }
            throw exception;
        }
    }

    private MeterReadingResultAcquisitionForClosingOutput requireChildOutput(
            MeterReadingResultAcquisitionForClosingOutput childOutput) {
        if (childOutput == null) {
            throw new IllegalStateException("SV-162 returned null output");
        }
        return childOutput;
    }

    private SkinKnkkInf mapSkinKnkkInf(
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf childValue) {
        if (childValue == null) {
            return null;
        }
        return new SkinKnkkInf(
                childValue.knsYmd(),
                childValue.knsKkaTsno(),
                childValue.gmtShjSu(),
                childValue.knkkSsjyShbt(),
                childValue.knkkKtmeFlg(),
                childValue.sjskKbn(),
                childValue.knsJsjkCd());
    }

    private record MeterReadingResultAcquisitionForClosingState(
            MeterReadingResultAcquisitionForClosingOutput output,
            boolean startDateIdentificationError) {

        private static MeterReadingResultAcquisitionForClosingState success(
                MeterReadingResultAcquisitionForClosingOutput output) {
            return new MeterReadingResultAcquisitionForClosingState(output, false);
        }

        private static MeterReadingResultAcquisitionForClosingState startDateIdentificationErrorState() {
            return new MeterReadingResultAcquisitionForClosingState(null, true);
        }
    }
}
