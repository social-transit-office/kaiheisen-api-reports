/** 導管小売契約情報取得サービス実装。 */
public final class PipelineRetailContractInfoAcquisitionImpl
        implements PipelineRetailContractInfoAcquisition {

    private static final String SHR_KBN_PIPELINE = "2";

    private final RetailTypeJudgment retailTypeJudgment;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final SvckyaDataAccess svckyaDataAccess;

    private final ItensakDataAccess itensakDataAccess;

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final SihakyDataAccess sihakyDataAccess;

    private final String tgRetailerCode;

    private final String chargeCalculationRelatedStatusInvalid;

    private final String serviceContractStatusUnderContract;

    private final String paymentContractStatusInvalid;

    private final String transferDestinationCancellationFlagNoConcept;

    private final RequestContext requestContext;

    /**
         * コンストラクタ。
         *
         * @param retailTypeJudgmentArg 小売種別判定サービス
         * @param svmstkrDataAccessArg ガスメーター設置場所別料金計算関連状態情報データアクセス
         * @param svckyaDataAccessArg サービス契約情報データアクセス
         * @param itensakDataAccessArg 移転先管理情報データアクセス
         * @param svsihkrDataAccessArg サービス契約別支払契約関連情報データアクセス
         * @param sihakyDataAccessArg 支払契約情報データアクセス
         * @param requestContextArg リクエストコンテキスト
         * @param ciriusConstantServiceArg CIRIUS 定数サービス
         */
    public PipelineRetailContractInfoAcquisitionImpl(
            RetailTypeJudgment retailTypeJudgmentArg,
            SvmstkrDataAccess svmstkrDataAccessArg,
            SvckyaDataAccess svckyaDataAccessArg,
            ItensakDataAccess itensakDataAccessArg,
            SvsihkrDataAccess svsihkrDataAccessArg,
            SihakyDataAccess sihakyDataAccessArg,
            RequestContext requestContextArg,
            CiriusConstantService ciriusConstantServiceArg) {
        this.retailTypeJudgment = Objects.requireNonNull(retailTypeJudgmentArg, "retailTypeJudgment");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccessArg, "svmstkrDataAccess");
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccessArg, "svckyaDataAccess");
        this.itensakDataAccess = Objects.requireNonNull(itensakDataAccessArg, "itensakDataAccess");
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccessArg, "svsihkrDataAccess");
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccessArg, "sihakyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContextArg, "requestContext");

        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceArg, "ciriusConstantService");
        this.tgRetailerCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.chargeCalculationRelatedStatusInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.INVALID);
        this.serviceContractStatusUnderContract =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.paymentContractStatusInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
        this.transferDestinationCancellationFlagNoConcept =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferDestinationCancellationFlag.TYPE,
                        ConstantLookupKeys.Cirius.TransferDestinationCancellationFlag.NO_CONCEPT);
    }

    @Override
    public PipelineRetailContractInfoAcquisitionOutput execute(Long gasMtstNo, LocalDate knskKjnYmd) {
        RetailTypeJudgmentOutput retailTypeJudgmentOutput =
                retailTypeJudgment.execute(SHR_KBN_PIPELINE, gasMtstNo, knskKjnYmd);
        TkssvkyInf tkssvkyInf =
                Optional.ofNullable(retailTypeJudgmentOutput)
                        .map(RetailTypeJudgmentOutput::tkssvkyInf)
                        .map(PipelineRetailContractInfoAcquisitionImpl::toTkssvkyInf)
                        .orElse(null);

        if (!isTokyoGasRetail(tkssvkyInf)) {
            return new PipelineRetailContractInfoAcquisitionOutput(tkssvkyInf, null, null, null, null);
        }

        Optional<SvmstkrRecord> svmstkrRecord =
                svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                        gasMtstNo,
                        knskKjnYmd,
                        chargeCalculationRelatedStatusInvalid);

        Long svkyNo = svmstkrRecord.orElseThrow().svkyNo();
        SvckyaRecord svckyaRecord = findSvckyaRecord(svkyNo);
        ItensakInf itensakInf = findItensakInf(gasMtstNo, svkyNo);
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        SvsihkrRecord svsihkrRecord = findSvsihkrRecord(svkyNo, systemDate);
        SihakyRecord sihakyRecord = findSihakyRecord(svsihkrRecord.sihKiyNo(), systemDate);

        return new PipelineRetailContractInfoAcquisitionOutput(
                tkssvkyInf,
                toSvckyaInf(svckyaRecord),
                itensakInf,
                toSvsihkrInf(svsihkrRecord),
                toSihakyInf(sihakyRecord));
    }

    private boolean isTokyoGasRetail(TkssvkyInf tkssvkyInf) {
        return tkssvkyInf != null && tgRetailerCode.equals(tkssvkyInf.tksuKijsCd());
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext
                .getSystemDatetime()
                .orElseThrow(
                        () ->
                                new IllegalStateException(
                                        "RequestContext.systemDatetime is required"));
    }

    private SvckyaRecord findSvckyaRecord(Long svkyNo) {
        List<SvckyaRecord> svckyaRecords =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(svkyNo, serviceContractStatusUnderContract);
        if (svckyaRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                                        SvckyaDataAccess.CallSpecs.FIND_ALL_BY_SVKY_NO_AND_SVKY_JTKB.identifier(),
                    parameterJsonForSvckya(svkyNo));
        }
        return svckyaRecords.getFirst();
    }

    private ItensakInf findItensakInf(Long gasMtstNo, Long svkyNo) {
        return itensakDataAccess
                .findAllByGmtSetNoAndSyokyNoAndItenTksCd(
                        gasMtstNo,
                        svkyNo,
                        transferDestinationCancellationFlagNoConcept)
                .stream()
                .findFirst()
                .map(PipelineRetailContractInfoAcquisitionImpl::toItensakInf)
                .orElse(null);
    }

    private SvsihkrRecord findSvsihkrRecord(Long svkyNo, LocalDate systemDate) {
        return svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndTyenYmdAndNotSihKiyJtkb(
                        svkyNo,
                        systemDate,
                        paymentContractStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svsihkrDataAccess.tableName(),
                        SvsihkrDataAccess.CallSpecs
                                .FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_NOT_SIH_KIY_JTKB
                                .identifier(),
                        parameterJsonForSvsihkr(svkyNo, systemDate)));
    }

    private SihakyRecord findSihakyRecord(Long sihKiyNo, LocalDate systemDate) {
        return sihakyDataAccess
                .findBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
                        sihKiyNo,
                        systemDate,
                        paymentContractStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sihakyDataAccess.tableName(),
                        SihakyDataAccess.CallSpecs.FIND_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_NOT_SIH_KIY_JTKB.identifier(),
                        parameterJsonForSihaky(sihKiyNo, systemDate)));
    }
