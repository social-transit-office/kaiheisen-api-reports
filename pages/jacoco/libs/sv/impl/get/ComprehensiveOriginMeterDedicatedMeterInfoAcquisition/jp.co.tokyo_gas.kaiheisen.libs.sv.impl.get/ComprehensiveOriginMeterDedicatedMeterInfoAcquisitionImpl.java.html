package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ComprehensiveOriginMeterDedicatedMeterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput;

/**
 * 概要: SV-124（総合元メーター時の専用メーター情報取得）の実装クラス。
 * 主な仕様:
 * - 元ガスメーターの場合のみ総合先候補を取得し、候補ごとに専用契約中の有無を参照する。
 * - 同一の設置場所番号が連続した場合は重複候補としてスキップする。
 * - J0244 の元ガスメーター、および J0611 の関連有効 / 関連終了待ちをコンストラクタで必須解決する。
 * 既知の制約:
 * - `systemDatetime` は RequestContext から取得し、未設定時は fail-fast する。
 * - public 引数の null ガードは SV 規約に従い実装しない。
 */
public final class ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionImpl
        implements ComprehensiveOriginMeterDedicatedMeterInfoAcquisition {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final RequestContext requestContext;

    private final String originalGasMeterStatus;

    private final String relatedValidStatus;

    private final String relatedEndWaitingStatus;

    /**
     * 概要: 依存 DA、RequestContext、および必須定数を解決して初期化する。
     * 主な仕様:
     * - 依存 DA と RequestContext の null を許容せず fail-fast する。
     * - J0244 の元ガスメーター、J0611 の関連有効 / 関連終了待ちを保持する。
     *
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            SvmstkrDataAccess svmstkrDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.relatedValidStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.relatedEndWaitingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
    }

    /**
     * 概要: SV-124 を実行し、総合元メーター時の専用契約中フラグを返却する。
     * 主な仕様:
     * - 元ガスメーター以外の場合は下位依存を呼ばず false を返却する。
     * - 候補設置場所番号を順に評価し、専用契約中の候補が 1 件でもあれば true を返却する。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param szcGkkCd 所在地行政区画コード
     * @param szcCmazCd 所在地丁目・字コード
     * @param szcMchCd 所在地町コード
     * @param gasMtstNo ガスメーター設置場所番号
     * @param rkkiKnrnJtkb 料金計算関連状態区分
     * @return 総合元メーター時の専用メーター情報取得結果
     */
    @Override
    public ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput execute(
            String ckrtGmscJtkb,
            String szcGkkCd,
            String szcCmazCd,
            String szcMchCd,
            Long gasMtstNo,
            String rkkiKnrnJtkb) {
        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return toOutput(false);
        }

        LocalDate searchBaseDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();

        List<GmtsetdRecord> candidates = Objects.requireNonNull(
                gmtsetdDataAccess
                        .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                                szcGkkCd,
                                szcCmazCd,
                                szcMchCd,
                                gasMtstNo,
                                gasMtstNo,
                                searchBaseDate),
                "gmtsetdDataAccess.findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd");

        // 候補 gasMtstNo に 0 が返らない前提はコード上にないため、初回判定は null で表現する。
        Long previousMeterLocationNumber = null;
        List<String> relatedStatuses = List.of(relatedValidStatus, relatedEndWaitingStatus);

        for (GmtsetdRecord candidate : candidates) {
            Long candidateMeterLocationNumber = candidate.gasMtstNo();
            if (!Objects.equals(previousMeterLocationNumber, candidateMeterLocationNumber)) {
                Optional<SvmstkrRecord> dedicatedContract = Objects.requireNonNull(
                        svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb(
                                candidateMeterLocationNumber,
                                relatedStatuses),
                        "svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb(Long, List)");
                if (dedicatedContract.isPresent()) {
                    return toOutput(true);
                }
            }
            previousMeterLocationNumber = candidateMeterLocationNumber;
        }

        return toOutput(false);
    }

    private static ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput toOutput(
            boolean snyKyktFlg) {
        return new ComprehensiveOriginMeterDedicatedMeterInfoAcquisitionOutput(snyKyktFlg);
    }
}
