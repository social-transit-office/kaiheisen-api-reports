package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

/** 契約お客さま＜小売＞情報取得サービス実装。 */
public final class ContractCustomerRetailInfoAcquisitionImpl
        implements ContractCustomerRetailInfoAcquisition {

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final SihakyDataAccess sihakyDataAccess;

    private final KiycstkDataAccess kiycstkDataAccess;

    private final String paymentContractStatusApplying;

    private final String splitBill;

    /**
     * コンストラクタ。
     *
     * @param svsihkrDataAccessArg サービス契約・支払契約関連データアクセス
     * @param sihakyDataAccessArg 支払契約データアクセス
     * @param kiycstkDataAccessArg 契約お客さま＜小売＞データアクセス
     * @param ciriusConstantServiceArg CIRIUS コンスタント取得サービス
     */
    public ContractCustomerRetailInfoAcquisitionImpl(
            SvsihkrDataAccess svsihkrDataAccessArg,
            SihakyDataAccess sihakyDataAccessArg,
            KiycstkDataAccess kiycstkDataAccessArg,
            CiriusConstantService ciriusConstantServiceArg) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccessArg, "svsihkrDataAccess");
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccessArg, "sihakyDataAccess");
        this.kiycstkDataAccess = Objects.requireNonNull(kiycstkDataAccessArg, "kiycstkDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceArg, "ciriusConstantService");
        this.paymentContractStatusApplying =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.splitBill =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                        ConstantLookupKeys.Cirius.BillingServiceType.SPLIT_BILL);
    }

    @Override
    public ContractCustomerRetailInfoAcquisitionOutput execute(Long svkyNo) {
        SvsihkrRecord svsihkrRecord =
                svsihkrDataAccess
                        .findBySvkyNoAndSihKiyJtkb(svkyNo, paymentContractStatusApplying)
                        .orElseThrow(
                                () ->
                                        DataInconsistencySupport.create(
                                                svsihkrDataAccess.tableName(),
                                                SvsihkrDataAccess.CallSpecs
                                                        .FIND_BY_SVKY_NO_AND_SIH_KIY_JTKB
                                                        .identifier(),
                                                serviceContractJson(svkyNo)));

        SihakyRecord sihakyRecord =
                sihakyDataAccess
                        .findBySihKiyNoAndSihKiyJtkb(
                                svsihkrRecord.sihKiyNo(),
                                paymentContractStatusApplying)
                        .orElseThrow(
                                () ->
                                        DataInconsistencySupport.create(
                                                sihakyDataAccess.tableName(),
                                                SihakyDataAccess.CallSpecs.FIND_BY_SIH_KIY_NO_AND_SIH_KIY_JTKB
                                                        .identifier(),
                                                paymentContractJson(svsihkrRecord.sihKiyNo())));

        KiycstkInf kiycstkInf = null;
        if (!splitBill.equals(sihakyRecord.sikSvcShbt())) {
            KiycstkRecord kiycstkRecord =
                    kiycstkDataAccess
                            .findByKyoTrkNo(sihakyRecord.shsyKyoTrno())
                            .orElseThrow(
                                    () ->
                                            DataInconsistencySupport.create(
                                                    kiycstkDataAccess.tableName(),
                                                    KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO
                                                            .identifier(),
                                                    retailCustomerJson(
                                                            sihakyRecord.shsyKyoTrno())));
            kiycstkInf = toKiycstkInf(kiycstkRecord);
        }

        return new ContractCustomerRetailInfoAcquisitionOutput(
                kiycstkInf,
                toSvsihkrInf(svsihkrRecord),
                toSihakyInf(sihakyRecord));
    }

    private FlatJsonBuilder paymentContractJson(Long sihKiyNo) {
        return new FlatJsonBuilder()
                .put("sihKiyNo", sihKiyNo)
                .put("sihKiyJtkb", paymentContractStatusApplying);
    }

    private FlatJsonBuilder serviceContractJson(Long svkyNo) {
        return new FlatJsonBuilder()
                .put("svkyNo", svkyNo)
                .put("sihKiyJtkb", paymentContractStatusApplying);
    }

    private static FlatJsonBuilder retailCustomerJson(Long kyoTrkNo) {
        return new FlatJsonBuilder().put("kyoTrkNo", kyoTrkNo);
    }

    private static KiycstkInf toKiycstkInf(KiycstkRecord record) {

                record.ctnmKn1(),
                record.ctnm1HujnShbt(),
                record.ctnm1HshiShbt(),
                record.ctnmKj2(),
                record.ctnmKn2(),
                record.ctnm2HujnShbt(),
                record.ctnm2HshiShbt(),
                record.ctnmKjhkFlg(),
                record.kynmKj(),
                record.kynmKn(),
                record.kynmHujnShbt(),
                record.kynmHshiShbt(),
                record.kynmKjhkFlg(),
                record.shkiLvShbt(),
                record.yubnNo(),
                record.gkkCd(),
                record.mchCd(),
                record.cmazCd(),
                record.cbgSuj1(),
                record.cbgKn2(),
                record.cbgKn3(),
                record.cbgKn4(),
                record.bldNo(),
                record.bldnKj(),
                record.bldnKn(),
                record.bldnKjhkFlg(),
                record.jutKbn(),
                record.tunoKn(),
                record.hynKn(),
                record.htuFlg(),
                record.recUpdJnjNo());
    }

    private static SvsihkrInf toSvsihkrInf(SvsihkrRecord record) {
        return new SvsihkrInf(
                record.sihKiyNo(),
                record.svkyNo(),
                record.skno(),
                record.tkysYmd(),
                record.tyenYmd(),
                record.sihKiyJtkb(),
                record.snskSvkyNo(),
                record.ikskSttYm(),
                record.ikskEndYm(),
                record.ikskGrpNo(),
                record.tdssSbtShbt(),
                record.gsnKhFlg(),
                record.gsptShbt(),
                record.sikTmngShbt(),
                record.shziKbn(),
                record.regTsmp(),
                record.regUsrId(),
                record.updTsmp(),
                record.updUsrId(),
                record.updKtnKinoId(),
                record.recUpdJnjNo());
    }

    private static SihakyInf toSihakyInf(SihakyRecord record) {
        return new SihakyInf(
                record.sihKiyNo(),
                record.skno(),
                record.tkysYmd(),
                record.tyenYmd(),

                record.jduKykShbt(),
                record.jduKykYmd(),
                record.shsyKyoTrno(),
                record.byssUmFlg(),
                record.skshHyjsKrno(),
                record.regTsmp(),
                record.regUsrId(),
                record.updTsmp(),
                record.updUsrId(),
                record.updKtnKinoId(),
                record.recUpdJnjNo());
    }
