package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.time.temporal.TemporalAdjusters;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ntknsk.NtknskDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ntknsk.NtknskRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.NextMeterReadingScheduledDateConfirmation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.NextMeterReadingScheduledDateConfirmationOutput;

/** 次回検針予定日確定処理サービス実装。 */
public class NextMeterReadingScheduledDateConfirmationImpl
        implements NextMeterReadingScheduledDateConfirmation {

    private final NtknskDataAccess ntknskDataAccess;

    private final String designatedDateNotSpecified;

    private final String designatedBatchNotSpecified;

    private final String meterReadingMethodPhone;

    private final String meterReadingMethodSumame;

    private final String nwOperatorTokyoGas;

    private final String utilityGas;

    /**
     * コンストラクタ。
     *
     * @param ntknskDataAccess 検針日程＜小売＞データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public NextMeterReadingScheduledDateConfirmationImpl(
            NtknskDataAccess ntknskDataAccess, CiriusConstantService ciriusConstantService) {
        this.ntknskDataAccess = ntknskDataAccess;

        this.designatedDateNotSpecified =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.DesignatedMeterReadingDate.TYPE,
                        ConstantLookupKeys.Cirius.DesignatedMeterReadingDate.NOT_SPECIFIED);
        this.designatedBatchNotSpecified =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.DesignatedMeterReadingBatchNumber.TYPE,
                        ConstantLookupKeys.Cirius.DesignatedMeterReadingBatchNumber.NOT_SPECIFIED);
        this.meterReadingMethodPhone =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.MeterReadingMethod.TYPE,
                        ConstantLookupKeys.Cirius.MeterReadingMethod.PHONE);
        this.meterReadingMethodSumame =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.MeterReadingMethod.TYPE,
                        ConstantLookupKeys.Cirius.MeterReadingMethod.SUMAME);
        this.nwOperatorTokyoGas =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.NwOperatorCode.TYPE,
                        ConstantLookupKeys.Cirius.NwOperatorCode.TOKYO_GAS_CO_LTD);
        this.utilityGas =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.Utility.TYPE,
                        ConstantLookupKeys.Cirius.Utility.GAS);
    }

    /**
     * 検針方法、検針回分番号等をもとに定例検針予定年月日を返却する。
     *
     * @param knsTishYmd 検針対象年月日
     * @param stkbKbn 指定検針日区分
     * @param knsHuhuKbn 検針方法区分
     * @param shtiKskbNo 指定検針回分番号
     * @param tkkbNo 定例検針回分番号
     * @return 次回検針予定日確定処理結果
     */
    @Override
    public NextMeterReadingScheduledDateConfirmationOutput execute(
            LocalDate knsTishYmd,
            String stkbKbn,
            String knsHuhuKbn,
            String shtiKskbNo,
            String tkkbNo) {
        LocalDate kymYmdFr = knsTishYmd.withDayOfMonth(1);
        LocalDate kymYmdTo = knsTishYmd.with(TemporalAdjusters.lastDayOfMonth());

        SearchResult searchResult =
                findScheduledRecords(kymYmdFr, kymYmdTo, stkbKbn, knsHuhuKbn, shtiKskbNo, tkkbNo);

        if (searchResult.records().isEmpty()) {
            throw DataInconsistencySupport.create(
                    ntknskDataAccess.tableName(),
                    searchResult.callSpec().identifier(),
                    searchResult.parameterJsonBuilder());
        }
        return new NextMeterReadingScheduledDateConfirmationOutput(
                searchResult.records().getFirst().tkytYmd());
    }

    /**
     * 概要: 入力値に応じてDA-126を呼び分ける。 主な仕様: - 指定検針日区分・検針方法区分・指定検針回分番号の組み合わせで分岐する。 - 返却値は各DAの検索結果をそのまま返す。
     * 既知の制約: - 入力の妥当性チェックは上位層で実施済みである前提とする。
     *
     * @param kymYmdFr 暦年月日From
     * @param kymYmdTo 暦年月日To
     * @param stkbKbn 指定検針日区分
     * @param knsHuhuKbn 検針方法区分
     * @param shtiKskbNo 指定検針回分番号
     * @param tkkbNo 定例検針回分番号
     * @return DA-126検索結果
     */
    private SearchResult findScheduledRecords(
            LocalDate kymYmdFr,
            LocalDate kymYmdTo,
            String stkbKbn,
            String knsHuhuKbn,
            String shtiKskbNo,
            String tkkbNo) {
        if (designatedDateNotSpecified.equals(stkbKbn)) {
            return findWhenDesignatedDateNotSpecified(
                    kymYmdFr, kymYmdTo, knsHuhuKbn, shtiKskbNo, tkkbNo);
        }
        if (isPhoneOrSumame(knsHuhuKbn)) {
            return new SearchResult(
                    NtknskDataAccess.CallSpecs
                            .FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_SHTI_EKSD_KBN_AND_SHTI_EKSD_KBN2,
                    baseParameterJsonBuilder(kymYmdFr, kymYmdTo).put("shtiEksdKbn", stkbKbn),
                    ntknskDataAccess
                            .findAllByKymYmdAndNwjgCdAndUtltKbnAndShtiEksdKbnAndShtiEksdKbn2(
                                    kymYmdFr, kymYmdTo, nwOperatorTokyoGas, utilityGas, stkbKbn));
        }
        return new SearchResult(
                NtknskDataAccess.CallSpecs
                        .FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_SHTI_HKSD_KBN_AND_SHTI_HKSD_KBN2,
                baseParameterJsonBuilder(kymYmdFr, kymYmdTo).put("shtiHksdKbn", stkbKbn),
                ntknskDataAccess.findAllByKymYmdAndNwjgCdAndUtltKbnAndShtiHksdKbnAndShtiHksdKbn2(
                        kymYmdFr, kymYmdTo, nwOperatorTokyoGas, utilityGas, stkbKbn));
    }

    /**
     * 概要: 指定検針日区分が指定なしの場合のDA呼び分けを行う。 主な仕様: - 検針方法区分が電話/スマメの場合はDA-126-R002またはDA-126-R004を呼び出す。 -
     * 検針方法区分がそれ以外の場合はDA-126-R001を呼び出す。 既知の制約: - 2.d系仕様に従い、指定検針回分番号はDA-126-R001のtkkbNoへ設定する。
     *
     * @param kymYmdFr 暦年月日From
     * @param kymYmdTo 暦年月日To
     * @param knsHuhuKbn 検針方法区分
     * @param shtiKskbNo 指定検針回分番号
     * @param tkkbNo 定例検針回分番号
     * @return DA-126検索結果
     */
    private SearchResult findWhenDesignatedDateNotSpecified(
            LocalDate kymYmdFr,
            LocalDate kymYmdTo,
            String knsHuhuKbn,
            String shtiKskbNo,
            String tkkbNo) {
        if (isPhoneOrSumame(knsHuhuKbn)) {
            if (designatedBatchNotSpecified.equals(shtiKskbNo)) {
                return new SearchResult(
                        NtknskDataAccess.CallSpecs
                                .FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_TREK_SSKB_NO_AND_TREK_SDKB_NO,
                        baseParameterJsonBuilder(kymYmdFr, kymYmdTo)
                                .put("trekSskbNo", tkkbNo)
                                .put("trekSdkbNo", tkkbNo),
                        ntknskDataAccess
                                .findAllByKymYmdAndNwjgCdAndUtltKbnAndTrekSskbNoAndTrekSdkbNo(
                                        kymYmdFr,
                                        kymYmdTo,
                                        nwOperatorTokyoGas,
                                        utilityGas,
                                        tkkbNo,
                                        tkkbNo));
            }
            return new SearchResult(
                    NtknskDataAccess.CallSpecs
                            .FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_SHTI_EKKB_NO,
                    baseParameterJsonBuilder(kymYmdFr, kymYmdTo).put("shtiEkkbNo", shtiKskbNo),
                    ntknskDataAccess.findAllByKymYmdAndNwjgCdAndUtltKbnAndShtiEkkbNo(
                            kymYmdFr, kymYmdTo, nwOperatorTokyoGas, utilityGas, shtiKskbNo));
        }
        if (designatedBatchNotSpecified.equals(shtiKskbNo)) {
            return new SearchResult(
                    NtknskDataAccess.CallSpecs
                            .FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_TKKB_NO,
                    baseParameterJsonBuilder(kymYmdFr, kymYmdTo).put("tkkbNo", tkkbNo),
                    ntknskDataAccess.findAllByKymYmdAndNwjgCdAndUtltKbnAndTkkbNo(
                            kymYmdFr, kymYmdTo, nwOperatorTokyoGas, utilityGas, tkkbNo));
        }
        return new SearchResult(
                NtknskDataAccess.CallSpecs.FIND_ALL_BY_KYM_YMD_AND_NWJG_CD_AND_UTLT_KBN_AND_TKKB_NO,
                baseParameterJsonBuilder(kymYmdFr, kymYmdTo).put("tkkbNo", shtiKskbNo),
                ntknskDataAccess.findAllByKymYmdAndNwjgCdAndUtltKbnAndTkkbNo(
                        kymYmdFr, kymYmdTo, nwOperatorTokyoGas, utilityGas, shtiKskbNo));
    }

    /**
     * 概要: DA 呼出しパラメータ JSON 文字列を構築する。 主な仕様: - 共通パラメータ `kymYmdFr` / `kymYmdTo` / `nwjgCd` / `utltKbn`
     * を常に出力する。
     *
     * @param kymYmdFr 暦年月日 From
     * @param kymYmdTo 暦年月日 To
     * @return パラメータ JSON ビルダー
     */
    private FlatJsonBuilder baseParameterJsonBuilder(LocalDate kymYmdFr, LocalDate kymYmdTo) {
        return new FlatJsonBuilder()
                .put("kymYmdFr", kymYmdFr)
                .put("kymYmdTo", kymYmdTo)
                .put("nwjgCd", nwOperatorTokyoGas)
                .put("utltKbn", utilityGas);
    }

    /**
     * 概要: 検針方法区分が電話またはスマメかを判定する。 主な仕様: - 電話またはスマメのいずれかに一致した場合のみ true を返却する。 既知の制約: -
     * 比較値はコンストラクタで解決したコンスタント値を使用する。
     *
     * @param knsHuhuKbn 検針方法区分
     * @return 電話またはスマメ判定結果
     */
    private boolean isPhoneOrSumame(String knsHuhuKbn) {
        return meterReadingMethodPhone.equals(knsHuhuKbn)
                || meterReadingMethodSumame.equals(knsHuhuKbn);
    }

    /**
     * DA 呼出し結果とメタ情報を保持する。
     *
     * @param callSpec DA 呼出し識別情報
     * @param parameterJsonBuilder 呼出しパラメータ JSON ビルダー
     * @param records DA 検索結果
     */
    private record SearchResult(
            DataAccessCallSpec callSpec,
            FlatJsonBuilder parameterJsonBuilder,
            List<NtknskRecord> records) { }
}
