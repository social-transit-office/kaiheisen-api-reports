/** SV-023 の実装。 */
public class OpeningWorkLocationInfoAcquisitionImpl implements OpeningWorkLocationInfoAcquisition {

    private static final String ERROR_CODE_TARGET_SV023 = ErrorCodeConstants.ERROR_CODE_020100012;

    private final MtsetkDataAccess mtsetkDataAccess;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final GasmtrkDataAccess gasmtrkDataAccess;

    private final TatemonDataAccess tatemonDataAccess;

    private final KenssieDataAccess kenssieDataAccess;

    private final MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening;

    private final GasValveConfirmationWorkResultCodeAcquisition gasValveConfirmationWorkResultCodeAcquisition;

    private final NextMeterReadingScheduledDateAcquisition nextMeterReadingScheduledDateAcquisition;

    private final String holdingMeterState;

    private final String kenshinSupportCategoryCode;

    private final String noConceptMeterReadingMethod;

    /** コンストラクタ。 */
    public OpeningWorkLocationInfoAcquisitionImpl(
            MtsetkDataAccess mtsetkDataAccess,
            GmtsetkDataAccess gmtsetkDataAccess,
            GasmtrkDataAccess gasmtrkDataAccess,
            TatemonDataAccess tatemonDataAccess,
            KenssieDataAccess kenssieDataAccess,
            MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening,
            GasValveConfirmationWorkResultCodeAcquisition gasValveConfirmationWorkResultCodeAcquisition,
            NextMeterReadingScheduledDateAcquisition nextMeterReadingScheduledDateAcquisition,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.mtsetkDataAccess = Objects.requireNonNull(mtsetkDataAccess, "mtsetkDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.gasmtrkDataAccess = Objects.requireNonNull(gasmtrkDataAccess, "gasmtrkDataAccess");
        this.tatemonDataAccess = Objects.requireNonNull(tatemonDataAccess, "tatemonDataAccess");
        this.kenssieDataAccess = Objects.requireNonNull(kenssieDataAccess, "kenssieDataAccess");
        this.meterReadingInfoAcquisitionForOpening = Objects.requireNonNull(
            meterReadingInfoAcquisitionForOpening, "meterReadingInfoAcquisitionForOpening");
        this.gasValveConfirmationWorkResultCodeAcquisition = Objects.requireNonNull(
            gasValveConfirmationWorkResultCodeAcquisition,
            "gasValveConfirmationWorkResultCodeAcquisition");
        this.nextMeterReadingScheduledDateAcquisition = Objects.requireNonNull(
            nextMeterReadingScheduledDateAcquisition,
            "nextMeterReadingScheduledDateAcquisition");

        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.noConceptMeterReadingMethod = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.MeterReadingMethodCode.TYPE,
                ConstantLookupKeys.Kaiheisen.MeterReadingMethodCode.NO_CONCEPT);
        this.holdingMeterState = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
        this.kenshinSupportCategoryCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MeterReadingSupport.TYPE,
                ConstantLookupKeys.Cirius.MeterReadingSupport.SPECIAL_GAS_METER_MOUNTING_POSITION);
    }

    @Override
    public OpeningWorkLocationInfoAcquisitionOutput execute(Long gasMtstNo, LocalDate knskKjnYmd) {
        MtsetkRecord mtsetkRecord = findMtsetkRecord(gasMtstNo, knskKjnYmd);
        GmtsetkRecord gmtsetkRecord = gmtsetkDataAccess
                .findByGasMtstNoAndTkysYmd(gasMtstNo, mtsetkRecord.tkysYmd())
                .orElseThrow(OpeningWorkLocationInfoAcquisitionImpl::businessRule);

        GasMtrkResInf gasMtrkResInf = findGasMtrkResInf(gasMtstNo, knskKjnYmd, gmtsetkRecord.kngmStbJtkb());
        TatemonInf tatemonInf = findTatemonInf(mtsetkRecord.bldNo());
        String gmtItiInf = findGmtItiInf(gasMtstNo);
        Sv027ResInf sv027ResInf = findSv027ResInf(gasMtstNo);
        Sv290ResInf sv290ResInf = findSv290ResInf(gasMtstNo);
        Sv022ResInf sv022ResInf = findSv022ResInf(gasMtstNo, knskKjnYmd);

        return new OpeningWorkLocationInfoAcquisitionOutput(
                toGmtSetkResInf(gmtsetkRecord),
                toMtSetkResInf(mtsetkRecord),
                gasMtrkResInf,
                tatemonInf,
                gmtItiInf,
                sv027ResInf,
                sv290ResInf,
                sv022ResInf);
    }

    /**
     * DA-048-R004 用に SV 入力のガスメーター設置場所番号をメーター設置場所番号へ対応付けて取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return メーター設置場所情報
     */
    private MtsetkRecord findMtsetkRecord(Long gasMtstNo, LocalDate knskKjnYmd) {
        // DA-048-R004 の IF 項目定義に合わせ、gasMtstNo を mtstNo に同値で対応付ける。
        Long mtstNo = gasMtstNo;
        return mtsetkDataAccess.findByMtstNoAndTkysYmdAndTyenYmd(mtstNo, knskKjnYmd)
                .orElseThrow(OpeningWorkLocationInfoAcquisitionImpl::businessRule);
    }

    /**
     * ガスメーター情報を取得する。
     * <p>
     * {@code kngmStbJtkb} がメーター保有状態を示すコードと一致しない場合は {@code null} を返す。
     * 対象レコードが存在しない場合は業務例外をスローする。
     * </p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @param kngmStbJtkb 供内管・ガスメーター設備状態区分
     * @return ガスメーター情報、または取得対象外の場合 {@code null}
     */
    private GasMtrkResInf findGasMtrkResInf(Long gasMtstNo, LocalDate knskKjnYmd, String kngmStbJtkb) {
        if (!holdingMeterState.equals(kngmStbJtkb)) {
            return null;
        }

        List<GasmtrkRecord> records = gasmtrkDataAccess
                .findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(gasMtstNo, knskKjnYmd);
        if (records.isEmpty()) {
            throw businessRule();
        }

        GasmtrkRecord record = records.get(0);
        return new GasMtrkResInf(
                record.gmb(),
                record.gmkShbt(),
                record.gmtGusu(),
                record.ktyYm(),
                record.sjsuMincSu());
    }

    /**
     * 建物情報を取得する。
     * <p>
     * {@code bldNo} が {@code null} または空文字の場合は {@code null} を返す（任意項目扱い）。
     * 建物番号に対応するレコードが存在しない場合はデータ不整合例外をスローする。
     * </p>
     *
     * @param bldNo 建物番号
     * @return 建物情報、または建物番号が未設定の場合 {@code null}
     */
    private TatemonInf findTatemonInf(String bldNo) {
        if (!hasText(bldNo)) {
            return null;
        }

        TatemonRecord record = tatemonDataAccess.findByBldNo(bldNo)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        tatemonDataAccess.tableName(),
                        TatemonDataAccess.CallSpecs.FIND_BY_BLD_NO.identifier(),
                        new FlatJsonBuilder().put("bldNo", bldNo)));

        return new TatemonInf(
                record.bldKjHyojiCd(),
                record.bldNamKn(),
                record.bldNamKj(),
                record.jyutakuKbnCd(),
                record.jutakKbnStCd());
    }

    /**
     * 検針支援情報を取得する。
     * <p>
     * 対象レコードが存在しない場合は {@code null} を返す（任意項目扱い）。
     * </p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return メータ位置詳細情報、または存在しない場合 {@code null}
     */
    private String findGmtItiInf(Long gasMtstNo) {
        Optional<String> result = kenssieDataAccess.findByGmtSetNoAndKnsnSienBrCd(gasMtstNo, kenshinSupportCategoryCode)
            .map(KenssieRecord::torokuNaiyoMx);
        return result.isPresent() ? result.get() : null;
    }

    /**
     * SV-027（検針情報取得（開栓））を返す。
     * <p>
     * SV-027 の出力が {@code null} の場合は {@code null} を返す（任意項目扱い）。
     * </p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return SV-027 の応答情報、または取得結果なしの場合 {@code null}
     */
    private Sv027ResInf findSv027ResInf(Long gasMtstNo) {
        MeterReadingInfoAcquisitionForOpeningOutput output =
                meterReadingInfoAcquisitionForOpening.execute(gasMtstNo);
        if (output == null) {
            return null;
        }
        return new Sv027ResInf(output.gmtShjSu());
    }

    /**
     * SV-290（ガス栓確認作業結果コード取得）を呼び出し結果を返す。
     * <p>
     * SV-290 の出力が {@code null} の場合は {@code null} を返す（任意項目扱い）。
     * </p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return SV-290 の応答情報、または取得結果なしの場合 {@code null}
     */
    private Sv290ResInf findSv290ResInf(Long gasMtstNo) {
        GasValveConfirmationWorkResultCodeAcquisitionOutput output =
                gasValveConfirmationWorkResultCodeAcquisition.execute(gasMtstNo);
        if (output == null) {
            return null;
        }
        return new Sv290ResInf(output.gassKknKkaCd(), output.ulfKknKkaCd(), output.flxStzTktCd());
    }

    /**
     * SV-022（次回検針予定日情報取得）を呼び出し結果を返す。
     * <p>
     * SV-022 の出力が {@code null} の場合は {@code null} を返す（任意項目扱い）。
     * 処理対象データ無しの場合は設計書どおり概念なしの応答情報を返す。
     * </p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return SV-022 の応答情報、または取得結果なしの場合 {@code null}
     */
    private Sv022ResInf findSv022ResInf(Long gasMtstNo, LocalDate knskKjnYmd) {
        NextMeterReadingScheduledDateAcquisitionOutput output;
        try {
            output = nextMeterReadingScheduledDateAcquisition.execute(gasMtstNo, knskKjnYmd);
        } catch (DataNotFoundException exception) {
            return new Sv022ResInf(null, noConceptMeterReadingMethod);
        }
        if (output == null) {
            return null;
        }
        return new Sv022ResInf(output.tkytYmd(), output.knsHuhuKbn());
    }

    /**
     * {@link MtsetkRecord} を {@link MtSetkResInf} に変換する。
     *
     * @param record メーター設置場所レコード
     * @return メーター設置場所情報
     */
    private static MtSetkResInf toMtSetkResInf(MtsetkRecord record) {
        return new MtSetkResInf(
                record.szcGkkCd(),
                record.szcMchCd(),
                record.szcCmazCd(),
                record.cbgSuj1(),
                record.cbgKn2(),
                record.cbgKn3(),
                record.cbgKn4(),
                record.bldNo(),
                record.tunoKn(),
                record.hynKn(),
                record.htuFlg());
    }

    /**
     * {@link GmtsetkRecord} を {@link GmtSetkResInf} に変換する。
     *
     * @param record ガスメーター設置場所レコード
     * @return ガスメーター設置場所情報
     */
    private static GmtSetkResInf toGmtSetkResInf(GmtsetkRecord record) {
        return new GmtSetkResInf(
                record.hesnYmd(),
                record.kasYmd(),
                record.kasJysb(),
                record.hesnJysb(),
                record.gmtKhsJtkb(),
                record.ohmKj(),
                record.snstGkjYm(),
                record.gasYtCd(),
                record.ckrtGmscJtkb(),
                record.kkyuGmtKbn(),
                record.kngmStbJtkb(),
                record.kysfYmd(),
                record.gmicDbriShbt(),
                record.gmicSbriShbt(),
                record.knsKhFlg(),
                record.hesnSdnShbt());
    }

    /**
     * 文字列が {@code null} でなく、かつ空白のみでないかを判定する。
     *
     * @param value 判定対象の文字列
     * @return 空でない場合 {@code true}
     */
    private static boolean hasText(String value) {
        return value != null && !value.isBlank();
    }

    /**
     * SV-023 用の開栓共通エラーを生成する。
     *
     * @return エラーコード {@value ERROR_CODE_TARGET_SV023} を持つ {@link KaisenRuleException}
     */
    private static KaisenRuleException businessRule() {
        return new KaisenRuleException(ERROR_CODE_TARGET_SV023);
    }
