package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.DedicatedMeterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.DedicatedMeterInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.SnyInf;

/** 専用メーター情報取得サービス実装。 */
public final class DedicatedMeterInfoAcquisitionImpl implements DedicatedMeterInfoAcquisition {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final String meterPossessionStatus;

    private final String tokyoGasRetailerCode;

    private final String transportationServiceContractRelatedStatusInvalid;

    private final List<String> chargeCalculationRelatedStatuses;

    /**
     * コンストラクタ。
     *
     * @param gmtsetdDataAccessArg ガスメーター設置場所＜導管＞データアクセス
     * @param tskgmstDataAccessArg 託送サービス契約・ガスメーター設置場所関連データアクセス
     * @param svmstkrDataAccessArg サービス契約・メーター設置場所関連データアクセス
     * @param ciriusConstantServiceArg CIRIUS 定数サービス
     */
    public DedicatedMeterInfoAcquisitionImpl(
            GmtsetdDataAccess gmtsetdDataAccessArg,
            TskgmstDataAccess tskgmstDataAccessArg,
            SvmstkrDataAccess svmstkrDataAccessArg,
            CiriusConstantService ciriusConstantServiceArg) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccessArg, "gmtsetdDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccessArg, "tskgmstDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccessArg, "svmstkrDataAccess");

        CiriusConstantService ciriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceArg, "ciriusConstantService");
        this.meterPossessionStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
        this.tokyoGasRetailerCode =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.transportationServiceContractRelatedStatusInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.INVALID);
        this.chargeCalculationRelatedStatuses = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID),
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03));
    }

    /**
     * 専用メーター情報を取得する。
     *
     * @param mtgMtstNo 元ガスメーター設置場所番号
     * @param szcGkkCd 所在地行政区画コード
     * @param szcMchCd 所在地町コード
     * @param szcCmazCd 所在地丁目・字コード
     * @param knskKjnYmd 検索基準年月日
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @return 取得結果
     */
    @Override
    public DedicatedMeterInfoAcquisitionOutput execute(
            Long mtgMtstNo,
            String szcGkkCd,
            String szcMchCd,
            String szcCmazCd,
            LocalDate knskKjnYmd,
            String tksuKijsCd) {
        List<GmtsetdRecord> dedicatedMeters = Objects.requireNonNull(
                gmtsetdDataAccess
                        .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmdAndKngmStbJtkb(
                                szcGkkCd,
                                szcCmazCd,
                                szcMchCd,
                                mtgMtstNo,
                                mtgMtstNo,
                                knskKjnYmd,
                                meterPossessionStatus),
                "gmtsetdDataAccess.findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAnd"
                        + "MtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmdAndKngmStbJtkb "
                        + "(DA-033-R008) returned null");
        if (dedicatedMeters.isEmpty()) {
            return new DedicatedMeterInfoAcquisitionOutput(List.of(), Boolean.FALSE);
        }

        List<SnyInf> snyInfList = new ArrayList<>();
        for (GmtsetdRecord dedicatedMeter : dedicatedMeters) {
            snyInfList.add(toSnyInf(dedicatedMeter, knskKjnYmd, tksuKijsCd));
        }
        return new DedicatedMeterInfoAcquisitionOutput(List.copyOf(snyInfList), Boolean.TRUE);
    }

    private SnyInf toSnyInf(GmtsetdRecord dedicatedMeter, LocalDate knskKjnYmd, String tksuKijsCd) {
        Long gasMtstNo = dedicatedMeter.gasMtstNo();
        Long tksuSvkyNo = findTksuSvkyNo(gasMtstNo, knskKjnYmd);
        Long svkyNo = null;
        if (tksuSvkyNo != null && tokyoGasRetailerCode.equals(tksuKijsCd)) {
            svkyNo = findSvkyNo(gasMtstNo, knskKjnYmd);
        }

        return new SnyInf(
                gasMtstNo,
                dedicatedMeter.hesnJysb(),
                dedicatedMeter.gmtKhsJtkb(),
                tksuSvkyNo,
                svkyNo);
    }

    private Long findTksuSvkyNo(Long gasMtstNo, LocalDate knskKjnYmd) {
        return tskgmstDataAccess
                .findByGasMtstNoAndTkysYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        knskKjnYmd,
                        transportationServiceContractRelatedStatusInvalid)
                .filter(record -> Objects.equals(record.gasMtstNo(), gasMtstNo))
                .map(TskgmstRecord::tksuSvkyNo)
                .orElse(null);
    }

    private Long findSvkyNo(Long gasMtstNo, LocalDate knskKjnYmd) {
        return svmstkrDataAccess
                .findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                        gasMtstNo,
                        knskKjnYmd,
                        chargeCalculationRelatedStatuses)
                .filter(record -> Objects.equals(record.mtstNo(), gasMtstNo))
                .map(SvmstkrRecord::svkyNo)
                .orElse(null);
    }
}
