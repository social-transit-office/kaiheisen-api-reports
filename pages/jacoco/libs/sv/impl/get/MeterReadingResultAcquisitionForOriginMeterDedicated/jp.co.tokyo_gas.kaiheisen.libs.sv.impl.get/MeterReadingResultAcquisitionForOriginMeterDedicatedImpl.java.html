package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

/**
 * 概要: SV-151（検針結果＆検針関連情報取得_元メーターの専用メーター用）の実装クラス。
 * 主な仕様:
 * - メーター保有状態のときのみ SV-162 を呼び出し、最近検針結果情報と閉栓報告可能期限日を取得する。
 * - DA-041-R009 と DA-063-R001 を順に参照し、契約中フラグと使用量協定中フラグを組み立てる。
 * - CIRIUS 定数 05800 / J0611 はコンストラクタで必須解決し、判定に利用する。
 * 既知の制約:
 * - hukkSysShbt、correlationId、systemDatetime は RequestContext を通じて下位 SV 側で扱う。
 * - public 引数の null ガードは SV 規約に従い実装しない。
 */
public final class MeterReadingResultAcquisitionForOriginMeterDedicatedImpl
        implements MeterReadingResultAcquisitionForOriginMeterDedicated {

    private final MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final KnsnkrnDataAccess knsnkrnDataAccess;

    private final String meterPossessionStatus;

    private final String relatedValidStatus;

    private final String relatedEndWaitingStatus;

    /**
     * 概要: 依存サービスと必須定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 依存 SV / DA の null を許容せず fail-fast する。
     * - 05800 のメーター保有状態、および J0611 の関連有効 / 関連終了待ちコードを保持する。
     *
     * @param meterReadingResultAcquisitionForClosing SV-162
     * @param svmstkrDataAccess DA-041
     * @param knsnkrnDataAccess DA-063
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public MeterReadingResultAcquisitionForOriginMeterDedicatedImpl(
            MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing,
            SvmstkrDataAccess svmstkrDataAccess,
            KnsnkrnDataAccess knsnkrnDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.meterReadingResultAcquisitionForClosing = Objects.requireNonNull(
                meterReadingResultAcquisitionForClosing,
                "meterReadingResultAcquisitionForClosing");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.knsnkrnDataAccess = Objects.requireNonNull(knsnkrnDataAccess, "knsnkrnDataAccess");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.meterPossessionStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
        this.relatedValidStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.relatedEndWaitingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
    }

    /**
     * 概要: SV-151 を実行し、元メーターに紐づく専用メーター情報を返却する。
     * 主な仕様:
     * - メーター保有状態の場合のみ SV-162 を呼び出す。
     * - 閉栓日以降検針有無、契約中、使用量協定中の各フラグを設計どおりに判定する。
     *
     * @param systemDateKngmStbJtkb システム日時点供内管・ガスメーター設備状態区分
     * @param snyMtstNo 専用メーター設置場所番号
     * @param hssgYmd 閉栓作業年月日
     * @return 元メーターの専用メーター用検針結果・関連情報
     */
    @Override
    public MeterReadingResultAcquisitionForOriginMeterDedicatedOutput execute(
            String systemDateKngmStbJtkb,
            Long snyMtstNo,
            LocalDate hssgYmd) {
        jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf recentMeterReading = null;
        LocalDate closingReportDeadline = null;
        boolean dedicatedClosingAfterReadingFlag = false;

        if (meterPossessionStatus.equals(systemDateKngmStbJtkb)) {
            MeterReadingResultAcquisitionForClosingState childState =
                    executeMeterReadingResultAcquisitionForClosing(snyMtstNo);
            if (!childState.startDateIdentificationError()) {
                MeterReadingResultAcquisitionForClosingOutput childOutput = childState.output();
                recentMeterReading = childOutput.skinKnkkInf();
                closingReportDeadline = childOutput.hsenHkkKnuKigenYmd();
                if (recentMeterReading != null) {
                    dedicatedClosingAfterReadingFlag =
                            isFutureDate(hssgYmd, requireRecentMeterReadingDate(recentMeterReading));
                }
            }
        }

        boolean dedicatedContractInProgressFlag = svmstkrDataAccess
                .findByMtstNoAndRkkiKnrnJtkb(
                        snyMtstNo,
                        List.of(relatedValidStatus, relatedEndWaitingStatus))
                .isPresent();
        KnsnkrnRecord knsnkrnRecord = knsnkrnDataAccess.findByGasMtstNo(snyMtstNo).orElse(null);
        boolean usageAgreementInProgressFlag =
                knsnkrnRecord != null && Boolean.TRUE.equals(knsnkrnRecord.gmsrKtumFlg());
        Long usageAgreementMeterLocationNumber = usageAgreementInProgressFlag ? snyMtstNo : null;

        return new MeterReadingResultAcquisitionForOriginMeterDedicatedOutput(
                mapSkinKnkkInf(recentMeterReading),
                closingReportDeadline,
                dedicatedClosingAfterReadingFlag,
                dedicatedContractInProgressFlag,
                usageAgreementInProgressFlag,
                usageAgreementMeterLocationNumber);
    }

    private MeterReadingResultAcquisitionForClosingState executeMeterReadingResultAcquisitionForClosing(
            Long snyMtstNo) {
        try {
            return MeterReadingResultAcquisitionForClosingState.success(
                    requireChildOutput(meterReadingResultAcquisitionForClosing.execute(snyMtstNo)));
        } catch (KaiheisenException exception) {
            if (ErrorCodeConstants.ERROR_CODE_020000074.equals(exception.getErrorCode())) {
                return MeterReadingResultAcquisitionForClosingState.startDateIdentificationErrorState();
            }
            throw exception;
        }
    }

    private MeterReadingResultAcquisitionForClosingOutput requireChildOutput(
            MeterReadingResultAcquisitionForClosingOutput childOutput) {
        if (childOutput == null) {
            throw new IllegalStateException("SV-162 returned null output");
        }
        return childOutput;
    }

    private LocalDate requireRecentMeterReadingDate(
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf recentMeterReading) {
        if (recentMeterReading.knsYmd() == null) {
            throw new IllegalStateException("SV-162 returned null skinKnkkInf.knsYmd");
        }
        return recentMeterReading.knsYmd();
    }

    private boolean isFutureDate(LocalDate hssgYmd, LocalDate knsYmd) {
        return knsYmd.isAfter(hssgYmd);
    }

    private SkinKnkkInf mapSkinKnkkInf(
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf childValue) {
        if (childValue == null) {
            return null;
        }
        return new SkinKnkkInf(
                childValue.knsYmd(),
                childValue.knsKkaTsno(),
                childValue.gmtShjSu(),
                childValue.knkkSsjyShbt(),
                childValue.knkkKtmeFlg(),
                childValue.sjskKbn(),
                childValue.knsJsjkCd());
    }

    private record MeterReadingResultAcquisitionForClosingState(
            MeterReadingResultAcquisitionForClosingOutput output,
            boolean startDateIdentificationError) {

        private static MeterReadingResultAcquisitionForClosingState success(
                MeterReadingResultAcquisitionForClosingOutput output) {
            return new MeterReadingResultAcquisitionForClosingState(output, false);
        }

        private static MeterReadingResultAcquisitionForClosingState startDateIdentificationErrorState() {
            return new MeterReadingResultAcquisitionForClosingState(null, true);
        }
    }
}
