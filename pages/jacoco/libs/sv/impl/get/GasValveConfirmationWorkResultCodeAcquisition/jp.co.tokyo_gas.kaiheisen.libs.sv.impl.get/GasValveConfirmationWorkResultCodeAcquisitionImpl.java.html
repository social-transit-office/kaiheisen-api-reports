package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.get;

import java.util.Objects;
import java.util.Set;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.gssntkr.GssntkrDataAccess;

/** ガス栓確認作業結果コード取得サービス実装。 */
public class GasValveConfirmationWorkResultCodeAcquisitionImpl
        implements GasValveConfirmationWorkResultCodeAcquisition {

    private final GssntkrDataAccess gssntkrDataAccess;

    private final GasValveConfirmationResultCodeJudgment gasValveConfirmationResultCodeJudgment;

    private final String defaultGasValveConfirmationWorkResultCode;

    private final String defaultUlFlexConfirmationWorkResultCode;

    private final String defaultFlexPipeConnectionTargetCode;

    private final Set<String> ulFlexibleCustomerClassificationSigns;

    private final Set<String> flexiblePipeCustomerClassificationSigns;

    /**
     * コンストラクタ。
     *
     * @param gssntkrDataAccess ガス栓点検管理データアクセス
     * @param gasValveConfirmationResultCodeJudgment ガス栓確認結果コード判定サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public GasValveConfirmationWorkResultCodeAcquisitionImpl(
            GssntkrDataAccess gssntkrDataAccess,
            GasValveConfirmationResultCodeJudgment gasValveConfirmationResultCodeJudgment,
            CiriusConstantService ciriusConstantService) {
        this.gssntkrDataAccess = Objects.requireNonNull(gssntkrDataAccess, "gssntkrDataAccess");
        this.gasValveConfirmationResultCodeJudgment =
                Objects.requireNonNull(
                        gasValveConfirmationResultCodeJudgment,
                        "gasValveConfirmationResultCodeJudgment");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.defaultGasValveConfirmationWorkResultCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode.TYPE,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode
                                .GAS_VALVE_CONFIRMED);
        this.defaultUlFlexConfirmationWorkResultCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode.TYPE,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode
                                .SCREW_PRESENCE_CONFIRMATION_TARGET);
        // 現在、フレキ管デフォルトはガス栓確認と同じ初期結果コードを使用する。
        // ただし将来の仕様分岐に備え、初期化処理は意図的に分離している。
        this.defaultFlexPipeConnectionTargetCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode.TYPE,
                        ConstantLookupKeys.Cirius.GasValveConfirmationWorkResultCode
                                .GAS_VALVE_CONFIRMED);
        this.ulFlexibleCustomerClassificationSigns =
                Set.of(
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_UL_SCREW_GAS_VALVE_TARGET),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .OUTSIDE_EXTRACTION_TARGET_FLEX_UL_SCREW_GAS_VALVE));
        this.flexiblePipeCustomerClassificationSigns =
                Set.of(
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_INSPECTION_TARGET),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_CONFIRMATION_TARGET_SCHEDULED_MAINTENANCE),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_CONFIRMATION_TARGET_METRO),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_CONFIRMATION_TARGET_SPECIAL_PATROL_A),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_CONFIRMATION_TARGET_SPECIAL_PATROL_B),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign.TYPE,
                                ConstantLookupKeys.Cirius.CustomerClassificationSign
                                        .FLEX_PIPE_CONNECTION_CONFIRMATION_TARGET_SPECIAL_PATROL_C));
    }

    /**
     * 概要: ガス栓点検管理一覧を走査し、3 種の結果コードを決定して返却する。 主な仕様: - お客さま分類サインに応じてガス栓/ULフレキ/フレキ管接続部の対象フラグを立てる。 -
     * 対象フラグが立ち、かつ対応コードが未設定の場合のみ SV-288 を呼び出して値を補完する。 - 3 種すべての対象フラグが立った時点で走査を終了する。 -
     * 最後まで未設定のコードは規定値で補完する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return ガス栓確認作業結果コード取得結果
     */
    @Override
    public GasValveConfirmationWorkResultCodeAcquisitionOutput execute(Long gasMtstNo) {

        List<GssntkrRecord> records = gssntkrDataAccess.findAllByGmtSetNo(gasMtstNo);
        String gassKknKkaCd = null;
        String ulfKknKkaCd = null;
        String flxStzTktCd = null;
        boolean gasValveTarget = false;
        boolean ulFlexibleTarget = false;
        boolean flexiblePipeTarget = false;

        for (GssntkrRecord record : records) {
            if (record == null) {
                continue;
            }

            String cstBrSign = record.cstBrSign();
            if (ulFlexibleCustomerClassificationSigns.contains(cstBrSign)) {
                ulFlexibleTarget = true;
            } else if (flexiblePipeCustomerClassificationSigns.contains(cstBrSign)) {
                flexiblePipeTarget = true;
            } else {
                gasValveTarget = true;
            }

            if (shouldCallJudgment(
                    gasValveTarget,
                    gassKknKkaCd,
                    ulFlexibleTarget,
                    ulfKknKkaCd,
                    flexiblePipeTarget,
                    flxStzTktCd)) {
                GasValveConfirmationResultCodeJudgmentOutput judgmentOutput =
                        gasValveConfirmationResultCodeJudgment.execute(
                                cstBrSign,
                                record.stkJkCd(),
                                record.tsyGssnSign(),
                                record.tnknTsyKbCd());

                if (gasValveTarget && gassKknKkaCd == null) {
                    gassKknKkaCd = judgmentOutput.gassKknKkaCd();
                }
                if (ulFlexibleTarget && ulfKknKkaCd == null) {
                    ulfKknKkaCd = judgmentOutput.ulfKknKkaCd();
                }
                if (flexiblePipeTarget && flxStzTktCd == null) {
                    flxStzTktCd = judgmentOutput.flxStzTktCd();
                }
            }

            if (gasValveTarget && ulFlexibleTarget && flexiblePipeTarget) {
                break;
            }
        }

        return new GasValveConfirmationWorkResultCodeAcquisitionOutput(
                defaultIfNull(gassKknKkaCd, defaultGasValveConfirmationWorkResultCode),
                defaultIfNull(ulfKknKkaCd, defaultUlFlexConfirmationWorkResultCode),
                defaultIfNull(flxStzTktCd, defaultFlexPipeConnectionTargetCode));
    }

    /**
     * 概要: SV-288 呼び出しの要否を判定する。 主な仕様: - ガス栓、ULフレキ、フレキ管接続部の各対象フラグが true で、かつ対応コードが未設定の場合に true を返す。 -
     * 3 種のうち 1 つでも未設定の対象コードが残っていれば true を返す。 既知の制約: - コード値の妥当性判定や trim などの前処理は行わない。
     *
     * @param gasValveTarget ガス栓確認作業結果コード設定対象フラグ
     * @param gassKknKkaCd ガス栓確認作業結果コード
     * @param ulFlexibleTarget ULフレキ確認作業結果コード設定対象フラグ
     * @param ulfKknKkaCd ULフレキ確認作業結果コード
     * @param flexiblePipeTarget フレキ管接続部点検対象コード設定対象フラグ
     * @param flxStzTktCd フレキ管接続部点検対象コード
     * @return SV-288 を呼び出す必要がある場合は true
     */
    private static boolean shouldCallJudgment(
            boolean gasValveTarget,
            String gassKknKkaCd,
            boolean ulFlexibleTarget,
            String ulfKknKkaCd,
            boolean flexiblePipeTarget,
            String flxStzTktCd) {
        return (gasValveTarget && gassKknKkaCd == null)
                || (ulFlexibleTarget && ulfKknKkaCd == null)
                || (flexiblePipeTarget && flxStzTktCd == null);
    }

    /**
     * 概要: 値が未設定の場合に規定値を補完する。 主な仕様: - value が null の場合は defaultValue を返す。 - value が null でない場合は
     * value をそのまま返す。 既知の制約: - 空文字や空白文字列を未設定扱いにはしない。
     *
     * @param value 判定対象の値
     * @param defaultValue value が null の場合に返却する規定値
     * @return value または規定値
     */
    private static String defaultIfNull(String value, String defaultValue) {
        return value == null ? defaultValue : value;
    }
}
