package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OriginMeterClosingInProgressCheck;

/**
 * 概要: SV-179（元メーター閉栓中チェック）の実装クラス。
 * 主な仕様:
 * - 直列ガスメーター設置状態区分が専用メーターであることを CIRIUS 定数で判定する。
 * - 元メーターのガスメーター開閉栓状態区分が開栓中以外の場合に開栓共通エラー 020100025 を送出する。
 * - 上記条件に一致しない場合は何も返さず正常終了する。
 * 既知の制約:
 * - 入力値の null 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となり、インスタンス生成に失敗する。
 */
public final class OriginMeterClosingInProgressCheckImpl
        implements OriginMeterClosingInProgressCheck {

    /** 専用ガスメーターを表す直列ガスメーター設置状態区分。 */
    private final String dedicatedGasMeterStatus;

    /** 開栓中を表すガスメーター開閉栓状態区分。 */
    private final String openingStatus;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 直列ガスメーター設置状態区分 J0244 の「専用ガスメーター」を必須解決して保持する。
     * - ガスメーター開閉栓状態区分 02100 の「開栓中」を必須解決して保持する。
     * - CIRIUS 定数取得サービスの null は許容せず、fail-fast で検知する。
     * 既知の制約:
     * - 必須定数が未登録の場合は遅延評価せず、コンストラクタ実行時点で fail-fast として例外を送出する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public OriginMeterClosingInProgressCheckImpl(
            final CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.openingStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 概要: 元メーター閉栓中判定を実行する。
     * 主な仕様:
     * - 直列ガスメーター設置状態区分が専用メーターで、元メーターのガスメーター開閉栓状態区分が開栓中以外のときに業務例外を送出する。
     * - それ以外の組み合わせでは正常終了する。
     * 既知の制約:
     * - correlationId、hukkSysShbt、systemDatetime は SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードは追加せず、上位バリデーション済み前提で評価する。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param gmtKhsJtkb 元メーターのガスメーター開閉栓状態区分
     */
    @Override
    public void execute(final String ckrtGmscJtkb, final String gmtKhsJtkb) {
        if (dedicatedGasMeterStatus.equals(ckrtGmscJtkb)
                && !openingStatus.equals(gmtKhsJtkb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100025);
        }
    }
}
