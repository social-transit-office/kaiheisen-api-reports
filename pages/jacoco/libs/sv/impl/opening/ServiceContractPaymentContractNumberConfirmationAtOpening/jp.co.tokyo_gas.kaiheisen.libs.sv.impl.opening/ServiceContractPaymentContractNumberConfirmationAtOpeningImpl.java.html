package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.inject.Inject;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ServiceContractPaymentContractNumberConfirmationAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ServiceContractPaymentContractNumberConfirmationAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.ServiceContractPaymentContractNumberConfirmationMtgInf;

/**
 * 概要: 開栓時サービス契約番号・支払契約番号確定サービス実装。
 * 主な仕様: 主関連有無、直列専用通過、代替判定、継続使用の優先順位で確定契約番号群を決定する。
 * 既知の制約: RequestContext 項目はこの SV では参照せず、他-048 の採番可否は専用サポートへ委譲する。
 */
public final class ServiceContractPaymentContractNumberConfirmationAtOpeningImpl
        implements ServiceContractPaymentContractNumberConfirmationAtOpening {

    private final ServiceContractPaymentContractNumberingSupport numberingSupport;

    private final String continuedUseOpeningReason;

    /**
     * コンストラクタ。
          * 主な仕様: DI 依存を fail-fast に検知し、継続使用判定で利用する CIRIUS 定数を初期化時に解決する。
          * 既知の制約: 必須定数が欠落している場合はインスタンス生成時に例外となる。
     *
     * @param numberingSupport 他-048 採番サポート
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public ServiceContractPaymentContractNumberConfirmationAtOpeningImpl(
            ServiceContractPaymentContractNumberingSupport numberingSupport,
            CiriusConstantService ciriusConstantService) {
        this.numberingSupport = Objects.requireNonNull(numberingSupport, "numberingSupport");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.continuedUseOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
    }

    /** {@inheritDoc} */
    @Override
    public ServiceContractPaymentContractNumberConfirmationAtOpeningOutput execute(
            String kasJysb,
            Boolean sknrUmFlg,
            Boolean digwrHntFlg,
            Boolean snyTkmtFlg,
            Boolean syoriHnt2XFlg,
            Long svkyNo,
            Long sihKiyNo,
            String sqtiKiyNo,
            Long sikKrno,
            ServiceContractPaymentContractNumberConfirmationMtgInf mtgInf) {
        if (Boolean.TRUE.equals(sknrUmFlg)) {
            ServiceContractPaymentContractNumberingSupport.NumberingResult result = numberingSupport.issue2xSx9x();
            return new ServiceContractPaymentContractNumberConfirmationAtOpeningOutput(
                    svkyNo,
                    result.sihKiyNo(),
                    result.sqtiKiyNo(),
                    result.sikKrno());
        }

        if (Boolean.TRUE.equals(snyTkmtFlg)) {
            return new ServiceContractPaymentContractNumberConfirmationAtOpeningOutput(
                    mtgInf.svkyNo(),
                    mtgInf.sihKiyNo(),
                    mtgInf.sqtiKiyNo(),
                    mtgInf.sikKrno());
        }

        if (Boolean.TRUE.equals(digwrHntFlg)) {
            ServiceContractPaymentContractNumberingSupport.NumberingResult result =
                    numberingSupport.issue4x2xSx9x();
            return new ServiceContractPaymentContractNumberConfirmationAtOpeningOutput(
                    result.svkyNo(),
                    result.sihKiyNo(),
                    result.sqtiKiyNo(),
                    result.sikKrno());
        }

        if (isContinuedUse(kasJysb) && !Boolean.TRUE.equals(syoriHnt2XFlg)) {
            ServiceContractPaymentContractNumberingSupport.NumberingResult result = numberingSupport.issue2x9x();
            return new ServiceContractPaymentContractNumberConfirmationAtOpeningOutput(
                    svkyNo,
                    result.sihKiyNo(),
                    sqtiKiyNo,
                    result.sikKrno());
        }

        return new ServiceContractPaymentContractNumberConfirmationAtOpeningOutput(
                svkyNo,
                sihKiyNo,
                sqtiKiyNo,
                sikKrno);
    }

    /**
     * 概要: 開栓事由が継続使用かを判定する。
     * 主な仕様: CIRIUS 定数で解決した 04900 の継続使用コードと入力値を比較する。
                * 既知の制約: `kasJysb` が `null` の場合は `continuedUseOpeningReason.equals(null)` の評価となるため `false` を返す。
     *
     * @param kasJysb 開栓事由種別
     * @return 継続使用の場合 true
     */
    private boolean isContinuedUse(String kasJysb) {
        return continuedUseOpeningReason.equals(kasJysb);
    }
}
