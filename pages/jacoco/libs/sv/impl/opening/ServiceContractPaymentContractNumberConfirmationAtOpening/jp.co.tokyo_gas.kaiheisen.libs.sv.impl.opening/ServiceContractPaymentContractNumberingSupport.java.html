package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

/**
 * 概要: SV-218 で利用する他-048 採番サポート。
 * 主な仕様:
 * - SV-218 の 3 つの採番パターンごとに、必要な契約番号群を返す。
 * - 実装は他-048 の入出力差分をこの抽象へ閉じ込める。
 * 既知の制約: RequestContext や業務定数の解決はこの抽象の責務に含めない。
 */
interface ServiceContractPaymentContractNumberingSupport {

    /**
     * 概要: 2xSx9x パターンで採番する。
     * 主な仕様: 支払契約番号・請求対象契約番号・請求管理番号を返す。
     * 既知の制約: サービス契約番号は採番対象外のため `null` を返しうる。
     *
     * @return 採番結果
     */
    NumberingResult issue2xSx9x();

    /**
     * 概要: 4x2xSx9x パターンで採番する。
        * 主な仕様: 4 つの契約番号すべてを採番して返す。
        * 既知の制約: 他-048 応答の必須項目不足は実装側で例外化する。
     *
     * @return 採番結果
     */
    NumberingResult issue4x2xSx9x();

    /**
     * 概要: 2x9x パターンで採番する。
        * 主な仕様: 支払契約番号と請求管理番号のみを採番して返す。
        * 既知の制約: サービス契約番号と請求対象契約番号は採番対象外のため `null` を返しうる。
     *
     * @return 採番結果
     */
    NumberingResult issue2x9x();

    /**
     * 概要: SV-218 が利用する採番結果。
        * 主な仕様: SV-218 が最終出力へ転記する最小項目のみを保持する。
        * 既知の制約: 採番対象外の項目は `null` を許容する。
     *
     * @param svkyNo サービス契約番号
     * @param sihKiyNo 支払契約番号
     * @param sqtiKiyNo 請求対象契約番号
     * @param sikKrno 請求管理番号
     */
    record NumberingResult(
            Long svkyNo,
            Long sihKiyNo,
          String sqtiKiyNo,
            Long sikKrno) {
    }
}
