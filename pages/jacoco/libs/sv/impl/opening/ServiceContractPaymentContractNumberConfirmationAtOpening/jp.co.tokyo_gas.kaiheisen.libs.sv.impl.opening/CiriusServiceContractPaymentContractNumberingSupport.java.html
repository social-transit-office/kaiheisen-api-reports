package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering.CustomerNumberingCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering.CustomerNumberingRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering.CustomerNumberingResponse;

/**
 * 概要: SV-218 用の他-048 採番サポート実装。
 * 主な仕様:
 * - `param_他-048_*.tsv` に対応する 3 パターンの採番条件を固定で組み立てる。
 * - 応答に必要項目が存在しない場合は即座に失敗させる。
 * 既知の制約: 他-048 の既存 adapter `CustomerNumberingCiriusAdapter#number` を流用し、SV-218 専用 endpoint は追加しない。
 */
@ApplicationScoped
public final class CiriusServiceContractPaymentContractNumberingSupport
        implements ServiceContractPaymentContractNumberingSupport {

    private final CustomerNumberingCiriusAdapter customerNumberingCiriusAdapter;

    /**
     * コンストラクタ。
     *
     * @param customerNumberingCiriusAdapter 他-048 adapter
     */
    @Inject
    public CiriusServiceContractPaymentContractNumberingSupport(
            CustomerNumberingCiriusAdapter customerNumberingCiriusAdapter) {
        this.customerNumberingCiriusAdapter =
                Objects.requireNonNull(customerNumberingCiriusAdapter, "customerNumberingCiriusAdapter");
    }

    /** {@inheritDoc} */
    @Override
    public NumberingResult issue2xSx9x() {
        CustomerNumberingResponse response = number(
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ZERO,
                BigDecimal.ONE,
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ZERO);
        return new NumberingResult(
                null,
                requirePaymentContractNumber(response),
                requireBillingTargetContractNumber(response),
                requireBillingManagementNumber(response));
    }

    /** {@inheritDoc} */
    @Override
    public NumberingResult issue4x2xSx9x() {
        CustomerNumberingResponse response = number(
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ONE,
                BigDecimal.ONE,
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ZERO);
        return new NumberingResult(
                requireServiceContractNumber(response),
                requirePaymentContractNumber(response),
                requireBillingTargetContractNumber(response),
                requireBillingManagementNumber(response));
    }

    /** {@inheritDoc} */
    @Override
    public NumberingResult issue2x9x() {
        CustomerNumberingResponse response = number(
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ZERO,
                BigDecimal.ONE,
                BigDecimal.ZERO,
                BigDecimal.ZERO,
                BigDecimal.ZERO);
        return new NumberingResult(
                null,
                requirePaymentContractNumber(response),
                null,
                requireBillingManagementNumber(response));
    }

    /**
     * 概要: 指定された発番回数で他-048 を呼び出す。
        * 主な仕様: 既存 `CustomerNumberingCiriusAdapter#number` に `CustomerNumberingRequest` を組み立てて委譲する。
        * 既知の制約: 他-048 の RequestInfo には既存 adapter 実装どおり一部発番回数のみが出力される。
     *
     * @param sihKiyNoHbks 支払契約番号発番回数
     * @param kyoTrnoKurHbks 契約お客さま登録番号_小売発番回数
     * @param svkyNoHbks サービス契約番号発番回数
     * @param sikKrnoHbks 請求管理番号発番回数
     * @param sqtiKiyNoHbks 請求対象契約番号発番回数
     * @param kyoTrnoDuknHbks 契約お客さま登録番号_導管発番回数
     * @param tksuSvkyNoHbks 託送サービス契約番号発番回数
     * @return 他-048 レスポンス
     */
    private CustomerNumberingResponse number(
            BigDecimal sihKiyNoHbks,
            BigDecimal kyoTrnoKurHbks,
            BigDecimal svkyNoHbks,
            BigDecimal sikKrnoHbks,
            BigDecimal sqtiKiyNoHbks,
            BigDecimal kyoTrnoDuknHbks,
            BigDecimal tksuSvkyNoHbks) {
        return customerNumberingCiriusAdapter.number(
                new CustomerNumberingRequest(
                        new CustomerNumberingRequest.NumberingCondition(
                                sihKiyNoHbks,
                                kyoTrnoKurHbks,
                                svkyNoHbks,
                                sikKrnoHbks,
                                sqtiKiyNoHbks,
                                kyoTrnoDuknHbks,
                                tksuSvkyNoHbks)));
    }

    /**
     * 概要: サービス契約番号を必須取得する。
        * 主な仕様: 他-048 応答から `svkyNo` を取り出し、欠落時は fail-fast で例外化する。
        * 既知の制約: 応答項目名は `svky_no` または `svkyNo` のいずれかを前提とする。
     *
     * @param response 他-048 レスポンス
     * @return サービス契約番号
     */
    private Long requireServiceContractNumber(CustomerNumberingResponse response) {
        Long numberedValue = response == null ? null : response.serviceContractNumber();
        if (numberedValue == null) {
            throw new IllegalStateException(
                    "他-048 response did not contain svkyNo"
                            + " (サービス契約番号)");
        }
        return numberedValue;
    }

    /**
     * 概要: 支払契約番号を必須取得する。
        * 主な仕様: 他-048 応答から `sihKiyNo` を取り出し、欠落時は fail-fast で例外化する。
        * 既知の制約: 応答項目名は `sih_kiy_no` または `sihKiyNo` のいずれかを前提とする。
     *
     * @param response 他-048 レスポンス
     * @return 支払契約番号
     */
    private Long requirePaymentContractNumber(CustomerNumberingResponse response) {
        Long numberedValue = response == null ? null : response.paymentContractNumber();
        if (numberedValue == null) {
            throw new IllegalStateException(
                    "他-048 response did not contain sihKiyNo"
                            + " (支払契約番号)");
        }
        return numberedValue;
    }

    /**
     * 概要: 請求対象契約番号を必須取得する。
        * 主な仕様: 他-048 応答から `sqtiKiyNo` を取り出し、欠落時は fail-fast で例外化する。
        * 既知の制約: 応答項目名は `sqti_kiy_no` または `sqtiKiyNo` のいずれかを前提とする。
     *
     * @param response 他-048 レスポンス
     * @return 請求対象契約番号
     */
    private String requireBillingTargetContractNumber(CustomerNumberingResponse response) {
        String numberedValue = response == null ? null : response.billingTargetContractNumber();
        if (numberedValue == null) {
            throw new IllegalStateException(
                    "他-048 response did not contain sqtiKiyNo"
                            + " (請求対象契約番号)");
        }
        return numberedValue;
    }

    /**
     * 概要: 請求管理番号を必須取得する。
        * 主な仕様: 他-048 応答から `sikKrno` を取り出し、欠落時は fail-fast で例外化する。
        * 既知の制約: 応答項目名は `sik_krno` または `sikKrno` のいずれかを前提とする。
     *
     * @param response 他-048 レスポンス
     * @return 請求管理番号
     */
    private Long requireBillingManagementNumber(CustomerNumberingResponse response) {
        Long numberedValue = response == null ? null : response.billingManagementNumber();
        if (numberedValue == null) {
            throw new IllegalStateException(
                    "他-048 response did not contain sikKrno"
                            + " (請求管理番号)");
        }
        return numberedValue;
    }
}
