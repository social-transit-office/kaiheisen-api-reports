package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.BillingCollectionPlatformIntegrationNecessityJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.GmtsetdKsenSaymdInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KiyInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SvckyaInf;

/**
 * 概要:
 * SV-251（開栓時請求回収基盤連携要否判定）の実装。
 *
 * 主な仕様:
 * - 既定値として「連携なし / 連携なし」を返す。
 * - 契約有無フラグ=true かつ閉栓事由=供給停止 に一致した場合は step 2.a として評価を打ち切る。
 * - step 2.a のうち供給停止精算状態区分が設定済みの場合を最優先で判定する。
 * - 継続使用かつ 4x 判定=true かつ復活判定=true の場合は、2x 判定に応じて継続復活または開栓へ分岐する。
 * - 総合区分更新フラグ=true の場合は前契約継続 / 総合区分更新を返す。
 *
 * 既知の制約:
 * - public メソッド引数の業務妥当性検証は上位層責務とする。
 * - nested DTO は null-safe に扱い、内部判定では未設定時に false として扱う。
 */
public final class BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningImpl
        implements BillingCollectionPlatformIntegrationNecessityJudgmentAtOpening {

    private final String closureReasonSupplyOutage;

    private final String openingReasonContinuedUse;

    private final String noCoordination;

    private final String opening;

    private final String continuationOfPreviousContract;

    private final String previousContractNoCoordination;

    private final String previousContractContinuousRevival;

    private final String previousContractOutOfSupply;

    private final String previousContractGeneralUpdate;

    /**
     * 概要: SV-251 の判定に必要な定数値を解決して初期化する。
     * 主な仕様:
     * - CIRIUS 定数サービスから閉栓事由=供給停止と開栓事由=継続使用を解決する。
     * - 開閉栓定数サービスから KH023/KH024 の戻り値区分を解決する。
     * 既知の制約:
     * - 必須定数未登録時はコンストラクタ初期化時点で fail-fast する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.closureReasonSupplyOutage = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
        this.openingReasonContinuedUse = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.noCoordination = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory.TYPE,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory
                        .NO_COORDINATION);
        this.opening = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory.TYPE,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory
                        .OPENING);
        this.continuationOfPreviousContract = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory.TYPE,
                ConstantLookupKeys.Kaiheisen.BillingCollectionInfrastructureCooperationCategory
                        .CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.previousContractNoCoordination = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing
                        .NO_COORDINATION);
        this.previousContractContinuousRevival = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing
                        .CONTINUOUS_REVIVAL);
        this.previousContractOutOfSupply = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.OUT_OF_SUPPLY);
        this.previousContractGeneralUpdate = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.GENERAL_UPDATE);
    }

    /**
     * 概要: 開栓時の請求回収基盤連携要否と連携パターンを判定する。
     * 主な仕様:
     * - 設計 step 1 に従い、既定値「連携なし / 連携なし」を基準に判定する。
     * - 契約有無フラグ=true かつ閉栓事由=供給停止に一致した場合は、step 2.a としてその時点で評価を打ち切る。
     * - step 2.a で状態区分が設定済み、または未設定かつ開栓作業日が契約の消費税率判定用日より前なら前契約継続 / 供給停止中を返す。
     * - step 2.a に一致しない場合のみ step 2.b、2.c を順に評価し、最初に一致した結果を返却する。
     * - どの条件にも一致しない場合は既定値を返却する。
     * 既知の制約:
     * - Boolean は Boolean.TRUE.equals(...) で判定する。
     * - nested DTO は null-safe に扱い、未設定時は該当条件を不成立として扱う。
     *
     * @param kasJysb 開栓事由種別
     * @param kasgYmd 開栓作業年月日
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所＜導管＞情報（開栓作業日時点）
     * @param kiyInf 契約情報
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param syoriHnt2XFlg 処理判定_2x_フラグ
     * @param syoriHntFkktFlg 処理判定_復活_フラグ
     * @param suguKbnUpdFlg 総合区分更新フラグ
     * @param kiyUmFlg 契約有無フラグ
     * @return 判定結果
     */
    @Override
    public BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput execute(
            String kasJysb,
            LocalDate kasgYmd,
            GmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            KiyInf kiyInf,
            Boolean syoriHnt4XFlg,
            Boolean syoriHnt2XFlg,
            Boolean syoriHntFkktFlg,
            Boolean suguKbnUpdFlg,
            Boolean kiyUmFlg) {
        BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput defaultOutput =
                new BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput(
                        noCoordination,
                        previousContractNoCoordination);
        if (isSupplyOutageBranchEligible(kiyUmFlg, gmtsetdKsenSaymdInf)) {
            if (hasConfiguredSupplyStopPaymentStatus(kiyInf)
                    || isOpeningBeforeConsumptionTaxRateJudgmentDate(kasgYmd, kiyInf)) {
                return new BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput(
                        continuationOfPreviousContract,
                        previousContractOutOfSupply);
            }
            return defaultOutput;
        }
        if (shouldHandleContinuedUse(kasJysb, syoriHnt4XFlg, syoriHntFkktFlg)) {
            if (Boolean.TRUE.equals(syoriHnt2XFlg)) {
                return new BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput(
                        continuationOfPreviousContract,
                        previousContractContinuousRevival);
            }
            return new BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput(
                    opening,
                    previousContractNoCoordination);
        }
        if (Boolean.TRUE.equals(suguKbnUpdFlg)) {
            return new BillingCollectionPlatformIntegrationNecessityJudgmentAtOpeningOutput(
                    continuationOfPreviousContract,
                    previousContractGeneralUpdate);
        }
        return defaultOutput;
    }

    /**
     * 概要: step 2.a の外側条件である契約有無かつ閉栓事由=供給停止を判定する。
     * 主な仕様:
     * - 契約有無フラグ=true を前提とする。
     * - ガスメーター設置場所情報の閉栓事由が供給停止であることを確認する。
     * 既知の制約:
     * - ネスト DTO が null の場合は false とする。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所情報
     * @return step 2.a の外側条件に一致する場合 true
     */
    private boolean isSupplyOutageBranchEligible(
            Boolean kiyUmFlg,
            GmtsetdKsenSaymdInf gmtsetdKsenSaymdInf) {
        return Boolean.TRUE.equals(kiyUmFlg)
                && isSupplyOutageClosureReason(gmtsetdKsenSaymdInf);
    }

    /**
     * 概要: 継続使用パターンの主条件を判定する。
     * 主な仕様:
     * - 開栓事由=継続使用、4x 判定=true、復活判定=true のすべてを満たした場合に true を返す。
     * 既知の制約:
     * - 開栓事由が null の場合は false とする。
     *
     * @param kasJysb 開栓事由種別
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param syoriHntFkktFlg 処理判定_復活_フラグ
     * @return step 2.b に一致する場合 true
     */
    private boolean shouldHandleContinuedUse(
            String kasJysb,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg) {
        return openingReasonContinuedUse.equals(kasJysb)
                && Boolean.TRUE.equals(syoriHnt4XFlg)
                && Boolean.TRUE.equals(syoriHntFkktFlg);
    }

    /**
     * 概要: 閉栓事由が供給停止かを判定する。
     * 主な仕様:
     * - DTO または閉栓事由が未設定なら false を返す。
     * - 解決済み供給停止定数と一致する場合のみ true を返す。
     *
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所情報
     * @return 閉栓事由が供給停止なら true
     */
    private boolean isSupplyOutageClosureReason(GmtsetdKsenSaymdInf gmtsetdKsenSaymdInf) {
        return gmtsetdKsenSaymdInf != null
                && closureReasonSupplyOutage.equals(gmtsetdKsenSaymdInf.hesnJysb());
    }

    /**
     * 概要: 開栓作業日が、更新前のサービス契約の消費税率判定用日より前かを判定する。
     * 主な仕様:
     * - step 2.a.1.b.1 に従い、厳密な日付比較を行う。
     * - 同日と後日は追加の連携対象に含めない。
     * 既知の制約:
     * - 比較に必要な日付や契約情報が未設定の場合は条件不成立として扱う。
     *
     * @param kasgYmd 開栓作業年月日
     * @param kiyInf 更新前の契約情報
     * @return 開栓作業日が消費税率判定用日より前の場合 true
     */
    private boolean isOpeningBeforeConsumptionTaxRateJudgmentDate(LocalDate kasgYmd, KiyInf kiyInf) {
        if (kasgYmd == null || kiyInf == null || kiyInf.svckyaInf() == null) {
            return false;
        }
        LocalDate szirHtyoYmd = kiyInf.svckyaInf().szirHtyoYmd();
        return szirHtyoYmd != null && kasgYmd.isBefore(szirHtyoYmd);
    }

    /**
     * 概要: 供給停止精算状態区分が設定済みかを判定する。
     * 主な仕様:
     * - ネスト DTO が未設定なら false を返す。
     * - 値が null でも空白でもない場合に true を返す。
     * - step 2.a の外側条件成立後の内側条件として使用する。
     *
     * @param kiyInf 契約情報
     * @return 供給停止精算状態区分が設定済みなら true
     */
    private boolean hasConfiguredSupplyStopPaymentStatus(KiyInf kiyInf) {
        if (kiyInf == null) {
            return false;
        }
        SvckyaInf svckyaInf = kiyInf.svckyaInf();
        return svckyaInf != null
                && svckyaInf.kkysSisnJtkb() != null
                && !svckyaInf.kkysSisnJtkb().isBlank();
    }
}
