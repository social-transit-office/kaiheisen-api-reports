package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckRecentMeterReadingDate;

/**
 * 概要: SV-182（開栓日チェック（最近検針日判定））の業務判定を実装するクラス。
 *
 * 主な仕様:
 * - 開栓作業年月日が開栓報告可能期限日より前の場合は開栓共通エラーとして扱い、規定のエラーコードで例外を送出する。
 * - 判定を通過した場合は、何も返さずに正常終了する。
 *
 * 既知の制約:
 * - 本実装は入力値の妥当性検証や補正を行わず、渡された日付をそのまま比較する。
 * - 判定過程の比較結果や理由は保持せず、異常系は例外送出、正常系は正常終了でのみ表現する。
 */
public class OpeningDateCheckRecentMeterReadingDateImpl
        implements OpeningDateCheckRecentMeterReadingDate {

    @Override
    public void execute(
            LocalDate kasgYmd, LocalDate ksenHkkKnuKigenYmd) {
        if (kasgYmd.isBefore(ksenHkkKnuKigenYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }
    }
}
