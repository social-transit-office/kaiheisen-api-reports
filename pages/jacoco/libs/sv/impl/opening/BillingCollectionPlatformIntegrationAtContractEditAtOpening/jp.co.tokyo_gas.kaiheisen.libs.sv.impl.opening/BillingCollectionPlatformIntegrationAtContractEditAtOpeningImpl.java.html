/**
 * 概要: SV-253（前契約継続時請求回収基盤連携情報編集）の実装。
 * 主な仕様:
 * - DA-060-R001 と DA-062-R001 の取得結果、および前契約継続処理区分に応じて請求回収基盤連携 DTO を編集する。
 * - RequestContext の systemDatetime を処理年月日へ反映する。
 * 既知の制約:
 * - 公開引数の null 妥当性検証は上位層責務とし、本クラスでは最小限の fail-fast のみ行う。
 */
public final class BillingCollectionPlatformIntegrationAtContractEditAtOpeningImpl
        implements BillingCollectionPlatformIntegrationAtContractEditAtOpening {

    private static final long FIXED_GAS_METER_LOCATION_NO = 11111111111L;

    private static final long ZERO_CUSTOMER_ROLE_NO = 0L;

    private static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final KiycstkDataAccess kiycstkDataAccess;

    private final KyotelkDataAccess kyotelkDataAccess;

    private final RequestContext requestContext;

    private final String continuousRevivalProcessing;

    private final String outOfSupplyProcessing;

    private final String generalUpdateProcessing;

    private final String accountTransferPaymentMethod;

    private final String noPhoneType;

    private final String blankContactType;

    private final String tokyoGasOperatorCode;

    private final String paymentManagementTransferCode;

    private final String contractManagementTransferCode;

    private final String noServiceSuspensionOrCancellation;

    /**
     * 概要: SV-253 の依存と必須定数を初期化する。
     *
     * @param kiycstkDataAccess 契約お客さま＜小売＞データアクセス
     * @param kyotelkDataAccess 契約お客さま電話番号＜小売＞データアクセス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     * @param requestContext リクエストコンテキスト
     */
    public BillingCollectionPlatformIntegrationAtContractEditAtOpeningImpl(
            KiycstkDataAccess kiycstkDataAccess,
            KyotelkDataAccess kyotelkDataAccess,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService,
            RequestContext requestContext) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.kiycstkDataAccess = Objects.requireNonNull(kiycstkDataAccess, "kiycstkDataAccess");
        this.kyotelkDataAccess = Objects.requireNonNull(kyotelkDataAccess, "kyotelkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.continuousRevivalProcessing = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.CONTINUOUS_REVIVAL);
        this.outOfSupplyProcessing = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.OUT_OF_SUPPLY);
        this.generalUpdateProcessing = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.PreviousContractContinuationProcessing.GENERAL_UPDATE);
        this.accountTransferPaymentMethod = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ACCOUNT_TRANSFER);
        this.noPhoneType = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PhoneType.TYPE,
                ConstantLookupKeys.Cirius.PhoneType.NO_PHONE);
        this.blankContactType = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ContactType.TYPE,
                ConstantLookupKeys.Cirius.ContactType.BLANK);
        this.tokyoGasOperatorCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.NwOperatorCode.TYPE,
                ConstantLookupKeys.Cirius.NwOperatorCode.TOKYO_GAS_CO_LTD);
        this.paymentManagementTransferCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_PAYMENT_MANAGEMENT_TRANSFER);
        this.contractManagementTransferCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_CONTRACT_MANAGEMENT_TRANSFER);
        this.noServiceSuspensionOrCancellation = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.NO_SERVICE_SUSPENSION_OR_CANCELLATION);
    }

    /**
     * 概要: 前契約継続時の請求回収基盤連携 DTO 群を編集する。
     *
     * @param zkyKzkSyoriKbn 前契約継続処理区分
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所＜導管＞情報（開栓作業日時点）
     * @param svckyaKktInf サービス契約情報（確定）
     * @param sihakyKktInf 支払契約情報（確定）
     * @param syoTisiUmFlg 使用者訂正有無フラグ
     * @param sikSvcShbt 請求サービス種別
     * @param suguHtkFlg 総合判定結果フラグ
     * @param kiyUmFlg 契約有無フラグ
     * @return 編集済み DTO 群
     */
    @Override
    public BillingCollectionPlatformIntegrationAtContractEditAtOpeningOutput execute(
            String zkyKzkSyoriKbn,
            BillingCollectionPlatformIntegrationAtContractGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            BillingCollectionPlatformIntegrationAtContractSvckyaKktInf svckyaKktInf,
            BillingCollectionPlatformIntegrationAtContractSihakyKktInf sihakyKktInf,
            Boolean syoTisiUmFlg,
            String sikSvcShbt,
            Boolean suguHtkFlg,
            Boolean kiyUmFlg) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        Long payerCustomerRegistrationNo = sihakyKktInf.shsyKyoTrno();
        KiycstkRecord customerRecord = findCustomer(payerCustomerRegistrationNo);
        PhoneValues phoneValues = resolvePhoneValues(payerCustomerRegistrationNo);

        Long controlGasMeterLocationNo = gmtsetdKsenSaymdInf.gasMtstNo();
        Boolean paymentContractCancellationFlag = null;
        Boolean billingServiceTypeExistsFlag = null;
        Long requestSourceCustomerRoleNo = null;
        String accountNameKana = null;
        String accountNameKanji = null;
        String nwOperatorCode = null;
        LocalDate paymentManagementEndDate = null;

        if (continuousRevivalProcessing.equals(zkyKzkSyoriKbn)) {
            controlGasMeterLocationNo = FIXED_GAS_METER_LOCATION_NO;
            paymentContractCancellationFlag = Boolean.TRUE;
            requestSourceCustomerRoleNo = payerCustomerRegistrationNo;
            nwOperatorCode = tokyoGasOperatorCode;
            paymentManagementEndDate = sihakyKktInf.tyenYmd();
            billingServiceTypeExistsFlag = sikSvcShbt != null;
            if (accountTransferPaymentMethod.equals(sihakyKktInf.sihHuhuShbt())) {
                accountNameKana = customerRecord.ctnmKn1();
                accountNameKanji = customerRecord.ctnmKj1();
            }
        } else if (outOfSupplyProcessing.equals(zkyKzkSyoriKbn)) {
            controlGasMeterLocationNo = Boolean.TRUE.equals(suguHtkFlg)
                    ? svckyaKktInf.svkyNo()
                    : FIXED_GAS_METER_LOCATION_NO;
            paymentContractCancellationFlag = Boolean.FALSE;
            billingServiceTypeExistsFlag = Boolean.FALSE;
            requestSourceCustomerRoleNo = ZERO_CUSTOMER_ROLE_NO;
            phoneValues = defaultPhoneValues();
        } else if (generalUpdateProcessing.equals(zkyKzkSyoriKbn)) {
            controlGasMeterLocationNo = svckyaKktInf.svkyNo();
            paymentContractCancellationFlag = Boolean.FALSE;
            billingServiceTypeExistsFlag = Boolean.FALSE;
            requestSourceCustomerRoleNo = ZERO_CUSTOMER_ROLE_NO;
            paymentManagementEndDate = sihakyKktInf.tyenYmd();
        }

        Long noticeCustomerNumber = resolveNoticeCustomerNumber(
                kiyUmFlg,
                suguHtkFlg,
                gmtsetdKsenSaymdInf.gasMtstNo(),
                svckyaKktInf.svkyNo());

        return new BillingCollectionPlatformIntegrationAtContractEditAtOpeningOutput(
                new BillingCollectionPlatformIntegrationAtContractSkkbRkySgjhInf(
                        null,
                        sihakyKktInf.sikKrno(),
                        svckyaKktInf.sqtiKiyNo(),
                        svckyaKktInf.svkyNo(),
                        svckyaKktInf.skno(),
                        sihakyKktInf.sihKiyNo(),
                        sihakyKktInf.skno(),
                        controlGasMeterLocationNo,
                        svckyaKktInf.rkkiSttYmd(),
                        svckyaKktInf.rkkiEndYmd(),
                        Boolean.FALSE,
                        paymentContractCancellationFlag,
                        billingServiceTypeExistsFlag,
                        syoTisiUmFlg,
                        systemDate),
                new BillingCollectionPlatformIntegrationAtContractSktsSihTrirInf(
                        null,
                        null,
                        requestSourceCustomerRoleNo,
                        accountNameKana,
                        accountNameKanji,
                        phoneValues.contactType1(),
                        phoneValues.contactType2(),
                        phoneValues.phoneNumber1(),
                        phoneValues.phoneNumber2(),
                        phoneValues.phoneType1(),
                        phoneValues.phoneType2(),
                        paymentManagementTransferCode,
                        sihakyKktInf.tyenYmd(),
                        sihakyKktInf.tkysYmd(),
                        nwOperatorCode),
                new BillingCollectionPlatformIntegrationAtContractSktsSihHkirInf(
                        paymentManagementTransferCode,
                        paymentManagementEndDate),
                new BillingCollectionPlatformIntegrationAtContractSktsKykHkirInf(
                        noServiceSuspensionOrCancellation,
                        MAX_DATE,
                        MAX_DATE,
                        contractManagementTransferCode,
                        noticeCustomerNumber,
                        svckyaKktInf.szirHtyoYmd()),
                new BillingCollectionPlatformIntegrationAtContractSktsSihDlirInf(
                        paymentManagementTransferCode));
    }

    private KiycstkRecord findCustomer(Long kyoTrkNo) {
        return kiycstkDataAccess.findByKyoTrkNo(kyoTrkNo)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kiycstkDataAccess.tableName(),
                        KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                        new FlatJsonBuilder().put("kyoTrkNo", kyoTrkNo)));
    }

    private PhoneValues resolvePhoneValues(Long kyoTrkNo) {
        PhoneValues defaultValues = defaultPhoneValues();
        List<KyotelkRecord> phoneRecords = kyotelkDataAccess.findAllByKyoTrkNo(kyoTrkNo);
        if (phoneRecords.isEmpty()) {
            return defaultValues;
        }

        KyotelkRecord firstRecord = phoneRecords.getFirst();
        String contactType1 = firstRecord.telRrskShbt();
        String phoneNumber1 = firstRecord.kyoTelNo();
        String phoneType1 = firstRecord.telShbt();
        String contactType2 = defaultValues.contactType2();
        String phoneNumber2 = defaultValues.phoneNumber2();
        String phoneType2 = defaultValues.phoneType2();

        if (phoneRecords.size() > 1) {
            KyotelkRecord secondRecord = phoneRecords.get(1);
            contactType2 = secondRecord.telRrskShbt();
            phoneNumber2 = secondRecord.kyoTelNo();
            phoneType2 = secondRecord.telShbt();
        }

        return new PhoneValues(
                contactType1,
                contactType2,
                phoneNumber1,
                phoneNumber2,
                phoneType1,
                phoneType2);
    }

    private PhoneValues defaultPhoneValues() {
        return new PhoneValues(blankContactType, blankContactType, null, null, noPhoneType, noPhoneType);
    }

    private Long resolveNoticeCustomerNumber(
            Boolean kiyUmFlg,
            Boolean suguHtkFlg,
            Long openingGasMeterLocationNo,
            Long serviceContractNo) {
        if (Boolean.FALSE.equals(kiyUmFlg) || Boolean.FALSE.equals(suguHtkFlg)) {
            return openingGasMeterLocationNo;
        }
        return serviceContractNo;
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private record PhoneValues(
            String contactType1,
            String contactType2,
            String phoneNumber1,
            String phoneNumber2,
            String phoneType1,
            String phoneType2) {
    }
