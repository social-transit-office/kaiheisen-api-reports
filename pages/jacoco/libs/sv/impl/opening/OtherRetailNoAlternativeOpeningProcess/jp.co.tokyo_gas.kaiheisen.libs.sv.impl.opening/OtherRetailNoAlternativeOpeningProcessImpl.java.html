/**
 * 概要: SV-213（他小売代替なし開栓処理）の実装クラス。
 * 主な仕様:
 * - SV-214、SV-217、SV-220、SV-222、SV-019、SV-230、SV-232 を順次呼び出し、後続ステップへ必要値を受け渡す。
 * - SV-220 と開栓処理入力から得た値をそのまま後続 SV へ受け渡し、SV-217 の結果を専用メーター出力へ再構成する。
 */
public final class OtherRetailNoAlternativeOpeningProcessImpl
        implements OtherRetailNoAlternativeOpeningProcess {

    private final ApplicationEndStartDateConfirmation applicationEndStartDateConfirmation;

    private final TransportServiceContractNumberConfirmationAtOpening
            transportServiceContractNumberConfirmationAtOpening;

    private final ContractCustomerUpdatePatternJudgmentAtOpening
            contractCustomerUpdatePatternJudgmentAtOpening;

    private final ContractCustomerRegistrationUpdateOtherRetailAtOpening
            contractCustomerRegistrationUpdateOtherRetailAtOpening;

    private final PipelinePhoneNumberRegistration pipelinePhoneNumberRegistration;

    private final OpeningProcess openingProcess;

    private final PreConfirmationGasMeterReadingResultRegistrationAtOpening
            preConfirmationGasMeterReadingResultRegistrationAtOpening;

    /**
     * 概要: 必須依存を初期化する。
     *
     * @param applicationEndStartDateConfirmationService SV-214
     * @param transportServiceContractNumberConfirmationService SV-217
     * @param contractCustomerUpdatePatternJudgmentService SV-220
     * @param contractCustomerRegistrationUpdateOtherRetailService SV-222
     * @param pipelinePhoneNumberRegistrationService SV-019
     * @param openingProcessService SV-230
     * @param preConfirmationGasMeterReadingResultRegistrationService SV-232
     */
    @Inject
    public OtherRetailNoAlternativeOpeningProcessImpl(
            final ApplicationEndStartDateConfirmation
                    applicationEndStartDateConfirmationService,
            final TransportServiceContractNumberConfirmationAtOpening
                    transportServiceContractNumberConfirmationService,
            final ContractCustomerUpdatePatternJudgmentAtOpening
                    contractCustomerUpdatePatternJudgmentService,
            final ContractCustomerRegistrationUpdateOtherRetailAtOpening
                    contractCustomerRegistrationUpdateOtherRetailService,
            final PipelinePhoneNumberRegistration
                    pipelinePhoneNumberRegistrationService,
            final OpeningProcess openingProcessService,
            final PreConfirmationGasMeterReadingResultRegistrationAtOpening
                    preConfirmationGasMeterReadingResultRegistrationService) {
        this.applicationEndStartDateConfirmation = Objects.requireNonNull(
                applicationEndStartDateConfirmationService,
                "applicationEndStartDateConfirmation");
        this.transportServiceContractNumberConfirmationAtOpening = Objects.requireNonNull(
                transportServiceContractNumberConfirmationService,
                "transportServiceContractNumberConfirmationAtOpening");
        this.contractCustomerUpdatePatternJudgmentAtOpening = Objects.requireNonNull(
                contractCustomerUpdatePatternJudgmentService,
                "contractCustomerUpdatePatternJudgmentAtOpening");
        this.contractCustomerRegistrationUpdateOtherRetailAtOpening = Objects.requireNonNull(
                contractCustomerRegistrationUpdateOtherRetailService,
                "contractCustomerRegistrationUpdateOtherRetailAtOpening");
        this.pipelinePhoneNumberRegistration = Objects.requireNonNull(
                pipelinePhoneNumberRegistrationService,
                "pipelinePhoneNumberRegistration");
        this.openingProcess = Objects.requireNonNull(
                openingProcessService,
                "openingProcess");
        this.preConfirmationGasMeterReadingResultRegistrationAtOpening = Objects.requireNonNull(
                preConfirmationGasMeterReadingResultRegistrationService,
                "preConfirmationGasMeterReadingResultRegistrationAtOpening");
    }

    @Override
    public OtherRetailNoAlternativeOpeningProcessOutput execute(
            final Long gasMtstNo,
            final String tgTkurKbn,
            final LocalDate kasgYmd,
            final String kasJysb,
            final Long gmtShjSu,
            final String ohmKj,
            final String sgytSinNo,
            final Boolean snyTkmtFlg,
            final Boolean suguHtkFlg,
            final Boolean digwrHntFlg,
            final Boolean syoriHnt3XFlg,
            final Boolean syoriHntFkktFlg,
            final Boolean ksenYmdTskgmstUmFlg,
            final Boolean cstTkssvkyTrkZmFlg,
            final Long tksuSvkyNo,
            final KiycstInf kiycstInf,
            final GasmtrdInf gasmtrdInf,
            final OpeningProcessGmtsetdKsenSaymdInf gmtsetdInf,
            final OpeningProcessGmtsetdSystemDateInf gmtsetdSystemDateInf,
            final TatemonInf tatemonInf,
            final CstHsenjInf cstHsenjInf,
            final CstKsenjInf cstKsenjInf,
            final MtgInf mtgInf,
            final Long kktGmtShjSu,
            final JusyoKktInf jusyoKktInf,
            final Long uketkjCstRegNo,
            final List<TelnoUpdInf> telnoUpdInfList,
            final String kktGmtsetdGasYtCd,
            final List<SnyInf> snyInfList) {
        ApplicationEndStartDateConfirmationOutput applicationDates =
                applicationEndStartDateConfirmation.execute(
                        gasMtstNo,
                        kasgYmd,
                        tksuSvkyNo,
                        digwrHntFlg);
        TransportServiceContractNumberConfirmationAtOpeningOutput
                transportConfirmation =
                transportServiceContractNumberConfirmationAtOpening.execute(
                        gmtsetdInf.ckrtGmscJtkb(),
                        digwrHntFlg,
                        tksuSvkyNo,
                        toTransportServiceConfirmationSnyInfList(snyInfList));
        ContractCustomerUpdatePatternJudgmentAtOpeningOutput updatePattern =
                contractCustomerUpdatePatternJudgmentAtOpening.execute(
                        tgTkurKbn,
                        kasJysb,
                        snyTkmtFlg,
                        digwrHntFlg,
                        syoriHnt3XFlg,
                        syoriHntFkktFlg,
                        cstTkssvkyTrkZmFlg,
                        toPipelineOnlyClosingCustomerInfo(cstHsenjInf),
                        toPipelineOnlyOpeningCustomerInfo(cstKsenjInf),
                        mtgInf);
        ContractCustomerRegistrationUpdateOtherRetailAtOpeningOutput
                registration =
                contractCustomerRegistrationUpdateOtherRetailAtOpening.execute(
                        snyTkmtFlg,
                        updatePattern.cstUpdPtnKb(),
                        updatePattern.kktSyoKyoTrno(),
                        toConfirmedContractCustomerInfo(updatePattern.kiycstKktInf()),
                        toConfirmedAddressInfo(jusyoKktInf),
                        kiycstInf,
                        toGasMeterLocationInfo(gmtsetdInf),
                        toClosingCustomerInfo(cstHsenjInf));
        Long confirmedCustomerRegistrationNo = updatePattern.kktSyoKyoTrno();
        pipelinePhoneNumberRegistration.execute(
                uketkjCstRegNo,
                confirmedCustomerRegistrationNo,
                toArrayList(telnoUpdInfList));
        OpeningProcessOutput openingOutput = openingProcess.execute(
                gasMtstNo,
                kasgYmd,
                kasJysb,
                applicationDates.kktTyenYmd(),
                applicationDates.kktTkysYmd(),
                snyTkmtFlg,
                digwrHntFlg,
                suguHtkFlg,
                ksenYmdTskgmstUmFlg,
                gmtsetdInf,
                gmtsetdSystemDateInf,
                tatemonInf,
                ohmKj,
                kktGmtsetdGasYtCd,
                jusyoKktInf);
        preConfirmationGasMeterReadingResultRegistrationAtOpening.execute(
                gasMtstNo,
                kasgYmd,
                gmtShjSu,
                openingOutput.knsytiInf().knsHuhuKbn(),
                gmtsetdInf.ckrtGmscJtkb(),
                gmtsetdInf.mtgMtstNo(),
                gasmtrdInf.gmb(),
                gasmtrdInf.gmbTsno(),
                transportConfirmation.kktTksuSvkyNo(),
                sgytSinNo,
                kasJysb);
        return new OtherRetailNoAlternativeOpeningProcessOutput(
                applicationDates.kktTyenYmd(),
                applicationDates.kktTkysYmd(),
                transportConfirmation.kktTksuSvkyNo(),
                toConfirmedSnyInfList(snyInfList, transportConfirmation.snyInfList()),
                confirmedCustomerRegistrationNo,
                openingOutput.knsytiInf(),
                registration.kiycstKktInf());
    }

    private List<TransportServiceContractConfirmationDedicatedMeterInfo>
            toTransportServiceConfirmationSnyInfList(
            final List<SnyInf> snyInfList) {
        if (snyInfList == null) {
            return null;
        }
        return snyInfList.stream()
                .map(info -> new TransportServiceContractConfirmationDedicatedMeterInfo(
                        info.kktTksuSvkyNo(),
                        null))
                .toList();
    }

    private ArrayList<TelnoUpdInf> toArrayList(
            final List<TelnoUpdInf> telnoUpdInfList) {
        return new ArrayList<>(telnoUpdInfList);
    }

    private List<SnyInf> toConfirmedSnyInfList(
            final List<SnyInf> sourceSnyInfList,
            final List<TransportServiceContractConfirmationDedicatedMeterInfo>
                    confirmedDedicatedMeterInfoList) {
        if (sourceSnyInfList == null || confirmedDedicatedMeterInfoList == null) {
            return null;
        }
        int resultSize = Math.min(sourceSnyInfList.size(), confirmedDedicatedMeterInfoList.size());
        List<SnyInf> results = new ArrayList<>(resultSize);
        for (int index = 0; index < resultSize; index++) {
            results.add(new SnyInf(
                    sourceSnyInfList.get(index).gasMtstNo(),
                    confirmedDedicatedMeterInfoList.get(index).kktTksuSvkyNo()));
        }
        return results;
    }

    private ConfirmedAddressInfo toConfirmedAddressInfo(final JusyoKktInf jusyoKktInf) {
        if (jusyoKktInf == null) {
            return null;
        }
        return new ConfirmedAddressInfo(
                jusyoKktInf.kktCbgSuj1(),
                jusyoKktInf.kktCbgKn2(),
                jusyoKktInf.kktCbgKn3(),
                jusyoKktInf.kktCbgKn4(),
                jusyoKktInf.kktTunoKn(),
                jusyoKktInf.kktHynKn(),
                jusyoKktInf.kktHynUmFlg());
    }

    private GasMeterLocationInfo toGasMeterLocationInfo(
            final OpeningProcessGmtsetdKsenSaymdInf gmtsetdInf) {
        if (gmtsetdInf == null) {
            return null;
        }
        return new GasMeterLocationInfo(
                gmtsetdInf.szcGkkCd(),
                gmtsetdInf.szcMchCd(),
                gmtsetdInf.szcCmazCd(),
                gmtsetdInf.bldNo());
    }

    private ClosingCustomerInfo toClosingCustomerInfo(final CstHsenjInf cstHsenjInf) {
        if (cstHsenjInf == null) {
            return null;
        }
        if (cstHsenjInf.kiycstdInf() == null) {
            return new ClosingCustomerInfo(null);
        }
        return new ClosingCustomerInfo(
                new ClosingPipelineContractCustomerInfo(
                        cstHsenjInf.kiycstdInf().shkiLvShbt()));
    }

    private CstHsenjInf toPipelineOnlyClosingCustomerInfo(final CstHsenjInf source) {
        if (source == null) {
            return null;
        }
        return new CstHsenjInf(source.kiycstdInf(), null);
    }

    private CstKsenjInf toPipelineOnlyOpeningCustomerInfo(final CstKsenjInf source) {
        if (source == null) {
            return null;
        }
        return new CstKsenjInf(source.kiycstdInf(), null);
    }
