    OpeningReportInfoUpdateImpl(
            KbagmstDataAccess kbagmstDataAccess,
            OpeningReportInfoEdit openingReportInfoEdit,
            KasnhkDataAccess kasnhkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService,
            ExistingOpeningReportInfoReader existingOpeningReportInfoReader) {
        this.kbagmstDataAccess = Objects.requireNonNull(kbagmstDataAccess, "kbagmstDataAccess");
        this.openingReportInfoEdit = Objects.requireNonNull(openingReportInfoEdit, "openingReportInfoEdit");
        this.kasnhkDataAccess = Objects.requireNonNull(kasnhkDataAccess, "kasnhkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.searchStatusList = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.REPORTING_OPENING),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.FOLLOW_UP_ON_OPENING));
        this.existingOpeningReportInfoReader = existingOpeningReportInfoReader == null
                ? this::readExistingOpeningReportInfo
                : existingOpeningReportInfoReader;
    }

    @Override
    public OpeningReportInfoUpdateOutput execute(
            String sgyAnknNo,
            LocalDate kasHukkYmd,
            LocalDate kasgYmd,
            Long gasMtstNo,
            String gmb,
            String kasJysb,
            String tgTkurKbn,
            String sgytNwkCd,
            String ctnmKn1,
            String ctnmKj1,
            Long kktSyoKyoTrno,
            Long kktSvkyNo,
            Long kktSihKiyNo,
            String ckrtGmscJtkb,
            Boolean snyTkmtFlg,
            String svkySttJysb,
            Boolean digwrHntFlg,
            MtgInf mtgInf,
            Boolean suguHtkFlg,
            Boolean kiyUmFlg,
            Boolean syoriHnt4XFlg,
            KiycstKktInf kiycstKktInf,
            String isrrGymCd,
            KseflKeyInf kseflKeyInf) {
        AuditContext auditContext = resolveAuditContext();
        KbagmstRecord areaInfo = resolveAreaInfo(gasMtstNo, kasgYmd);
        OpeningReportInfoEditOutput edited = openingReportInfoEdit.execute(
                sgyAnknNo,
                kasHukkYmd,
                kasgYmd,
                gasMtstNo,
                gmb,
                kasJysb,
                tgTkurKbn,
                sgytNwkCd,
                ctnmKn1,
                ctnmKj1,
                kktSyoKyoTrno,
                kktSvkyNo,
                kktSihKiyNo,
                ckrtGmscJtkb,
                snyTkmtFlg,
                svkySttJysb,
                digwrHntFlg,
                mtgInf,
                suguHtkFlg,
                kiyUmFlg,
                syoriHnt4XFlg,
                kiycstKktInf,
                isrrGymCd,
                areaInfo.kbabCktkCd(),
                areaInfo.kbabKtnCd(),
                kseflKeyInf);

        ExistingOpeningReportSearchResult existing = existingOpeningReportInfoReader.read(edited);
        switch (existing.state()) {
            case NONE -> kasnhkDataAccess.insert(buildWriteRecord(edited), auditContext);
            case SINGLE -> updateExistingOpeningReportInfo(edited, existing.record(), auditContext);
        }
        return new OpeningReportInfoUpdateOutput(edited.kassSvkyNo(), sgyAnknNo);
    }

    private AuditContext resolveAuditContext() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        return AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
    }

    private KbagmstRecord resolveAreaInfo(Long gasMtstNo, LocalDate kasgYmd) {
        YearMonth knskKjnYm = YearMonth.of(kasgYmd.getYear(), kasgYmd.getMonthValue());
        String daIdentifier = KbagmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM
                .identifier();
        return kbagmstDataAccess.findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, knskKjnYm)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kbagmstDataAccess.tableName(),
                        daIdentifier,
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYm", knskKjnYm.toString())));
    }

    private ExistingOpeningReportSearchResult readExistingOpeningReportInfo(
            OpeningReportInfoEditOutput edited) {
        Optional<KasnhkRecord> existingRecord =
                kasnhkDataAccess.findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
                        edited.gmb(),
                        edited.kasgYmd(),
                        edited.kasHukkYmd(),
                        edited.lvltAnknNo(),
                        searchStatusList);
        if (existingRecord.isEmpty()) {
            return ExistingOpeningReportSearchResult.none();
        }
        return ExistingOpeningReportSearchResult.single(existingRecord.orElseThrow());
    }

    private void updateExistingOpeningReportInfo(
            OpeningReportInfoEditOutput edited,
            KasnhkRecord existingRecord,
            AuditContext auditContext) {
        KasnhkWriteRecord writeRecord = buildWriteRecord(edited);
        KasnhkU002UpdateKey updateKey = new KasnhkU002UpdateKey(
                edited.gmb(),
                edited.kasgYmd(),
                edited.kasHukkYmd(),
                edited.lvltAnknNo(),
                existingRecord.recUpdJnjNo());
        int updatedCount = kasnhkDataAccess
                .updateByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndRecUpdJnjNo(
                        writeRecord,
                        TARGET_COLUMNS,
                        updateKey,
                        auditContext);
        if (updatedCount != 1) {
            String daIdentifier = KasnhkDataAccess.CallSpecs
                    .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_LVLT_ANKN_NO_AND_REC_UPD_JNJ_NO_U002
                    .identifier();
            throw DataInconsistencySupport.create(
                    kasnhkDataAccess.tableName(),
                    daIdentifier,
                    buildUpdateParameterJson(edited, existingRecord.recUpdJnjNo(), auditContext));
        }
    }

    @FunctionalInterface
    interface ExistingOpeningReportInfoReader {
        ExistingOpeningReportSearchResult read(OpeningReportInfoEditOutput edited);
    }

    enum ExistingOpeningReportState {
        NONE,
                SINGLE
    }

    record ExistingOpeningReportSearchResult(
            ExistingOpeningReportState state,
            KasnhkRecord record) {

        static ExistingOpeningReportSearchResult none() {
            return new ExistingOpeningReportSearchResult(
                    ExistingOpeningReportState.NONE,
                    null);
        }

        static ExistingOpeningReportSearchResult single(KasnhkRecord record) {
            return new ExistingOpeningReportSearchResult(
                    ExistingOpeningReportState.SINGLE,
                    Objects.requireNonNull(record, "record"));
        }
    }
