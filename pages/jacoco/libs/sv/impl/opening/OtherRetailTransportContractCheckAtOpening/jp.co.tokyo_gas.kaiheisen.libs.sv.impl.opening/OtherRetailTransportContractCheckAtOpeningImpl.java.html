package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OtherRetailTransportContractCheckAtOpening;

/**
 * 概要: SV-191（開栓時他小売託送契約チェック）の実装クラス。
 * 主な仕様:
 * - 契約有無フラグが false の場合は、託送メータ情報本登録・仮解約有無フラグまたは仮登録有無フラグが true なら開栓共通エラー 020100033 を送出する。
 * - 契約有無フラグが false 以外の場合は、東京ガス事業者コードかつ、本登録・仮解約有無フラグが true、または仮登録有無フラグが true かつスイッチ申込以外なら開栓共通エラー 020100033 を送出する。
 * - 判定に使用する CIRIUS 定数はコンストラクタ初期化時に必須解決し、実行時は解決済み値のみで比較する。
 * - 下位 DA や他 SV 呼び出しは行わず、入力値だけで分岐判定を完結させる。
 * 既知の制約:
 * - 入力値の trim、blank 補正、null 補正は行わず、呼び出し元から渡された値をそのまま評価する。
 * - `correlationId`、`hukkSysShbt`、`systemDatetime` は SV 規約に従い公開引数へ含めない。
 */
public final class OtherRetailTransportContractCheckAtOpeningImpl
        implements OtherRetailTransportContractCheckAtOpening {

    private final String tokyoGasRetailerCode;

    private final String switchApplicationCisJudgmentCode;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 託送供給依頼事業者コード J0580 の東京ガス株式会社コードを必須解決して保持する。
     * - 託送契約ＣＩＳ判定コード 1AP00 のスイッチ申込コードを必須解決して保持する。
     * 既知の制約:
     * - CIRIUS 定数取得サービスの null は許容しない。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public OtherRetailTransportContractCheckAtOpeningImpl(
            CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.tokyoGasRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.switchApplicationCisJudgmentCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.SWITCH_APPLICATION);
    }

    /**
     * 概要: 開栓時他小売託送契約チェックを実行し、対象条件に該当する場合は開栓共通例外を送出する。
     * 主な仕様:
     * - 契約有無フラグが false の場合は、第1判定を優先し、託送メータ情報の有無だけで開栓共通エラー要否を決定する。
     * - 契約有無フラグが false 以外の場合は、東京ガス事業者コードに限定して第2判定を行う。
     * - 第2判定では、本登録・仮解約有無フラグが true の場合、または仮登録有無フラグが true かつスイッチ申込以外の場合に開栓共通エラーとする。
     * - 条件に該当しない場合は正常終了とする。
     * 既知の制約:
     * - `Boolean.FALSE.equals(...)` と `Boolean.TRUE.equals(...)` により null 入力も判定可能だが、業務上の null 補完は行わない。
     * - 事業者コードおよび託送契約ＣＩＳ判定コードは完全一致でのみ比較する。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param taksmtrHntrKrkykUmFlg 託送メータ情報本登録・仮解約有無フラグ
     * @param taksmtrKtrUmFlg 託送メータ情報仮登録有無フラグ
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param tkykCisjCd 託送契約ＣＩＳ判定コード
     */
    @Override
    public void execute(
            Boolean kiyUmFlg,
            Boolean taksmtrHntrKrkykUmFlg,
            Boolean taksmtrKtrUmFlg,
            String tksuKijsCd,
            String tkykCisjCd) {
        boolean hasNoContract = Boolean.FALSE.equals(kiyUmFlg);
        boolean hasMainOrTemporaryCancellationRegistration =
                Boolean.TRUE.equals(taksmtrHntrKrkykUmFlg);
        boolean hasTemporaryRegistration = Boolean.TRUE.equals(taksmtrKtrUmFlg);

        if (hasNoContract
                && (hasMainOrTemporaryCancellationRegistration || hasTemporaryRegistration)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100033);
        }

        if (!hasNoContract
                && tokyoGasRetailerCode.equals(tksuKijsCd)
                && (hasMainOrTemporaryCancellationRegistration
                        || (hasTemporaryRegistration
                                && !switchApplicationCisJudgmentCode.equals(tkykCisjCd)))) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100033);
        }
    }
}
