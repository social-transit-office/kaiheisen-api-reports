/**
 * 概要： SV-117(開栓エラーＰＦフォロー不要更新)実装クラス。
 *
 * 入力されたパラメータを元に、フォローステータスコード、システム日付、更新ユーザーID、最終更新起点機能ＩＤを更新する。 -
 * 排他エラーが発生し、更新件数が0件だった場合DA-139-R004を呼び出し、取得したフォローステータスコードを元にエラーをthrowする。 -
 * 入力値のnull 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 */
public class OpeningErrorPfFollowUnnecessaryUpdateImpl implements OpeningErrorPfFollowUnnecessaryUpdate {

    private KseflDataAccess kseflDataAccess;

    private final RequestContext requestContext;

    private KaiheisenConstantService kaiheisenConstantService;

    private final String followStatusMityakushu;

    private final String followStatusFollowHuyou;

    static final String USR_ID = "KHSS117";

    /**
     * 概要: フォローステータスコードの更新に必要な依存オブジェクトと定数を初期化する。
     * KSEFL更新、RequestContext、開閉栓定数サービスを保持する。 -
     * 依存オブジェクトがnullの場合は初期化時点でNullPointerException を送出する。 -
     * 必須CIRIUS定数が不足している場合はコンストラクタ初期化時点で失敗する。 -
     *
     * @param KseflDataAccess          KSEFL データアクセス
     * @param requestContext           リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService    CIRIUS 定数サービス
     */

    public OpeningErrorPfFollowUnnecessaryUpdateImpl(KseflDataAccess kseflDataAccess,
            CiriusConstantService ciriusConstantService, RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {

        this.kseflDataAccess = kseflDataAccess;
        this.requestContext = requestContext;
        this.kaiheisenConstantService = kaiheisenConstantService;

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(ciriusConstantService,
                "ciriusConstantService");

        this.followStatusMityakushu = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);
        this.followStatusFollowHuyou = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_NO_FOLLOW_REQUIRED);

        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");

        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TAXIS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.KRAKEN);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.EAGLE);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.HURRY);
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");

    }

    /**
     * 概要: フォローステータスコードの更新を実行する。-
     *
     * requestContextからシステム日時を取得する。- KseflWriteRecordに更新パラーメータを設定する。 -
     * kseflDataAccessを実施する。この時Responseとして更新件数を取得する -
     * 排他エラー(030400001)が返却された場合、DA-139-R004を呼び出し、該当テーブルのフォローステータスコードを取得する。 -
     * フォローステータスコードが未着手の場合はDBエラー、それ以外の場合は業務エラーをthrowする。 -
     * 引数のnull検証は明示的に行わず、下位処理で実行時例外となる可能性がある。 -
     *
     * @param gmb         ガスメーター社番
     * @param kasgYmd     開栓作業年月日
     * @param kasHukkYmd  開栓報告年月日
     * @param sgyAnknNo   作業案件番号
     * @param recUpdJnjNo レコード更新排他制御順序番号
     */
    @Override
    public void execute(String gmb, LocalDate kasgYmd, LocalDate kasHukkYmd, String sgyAnknNo, Integer recUpdJnjNo) {

        LocalDate fltkYmd = this.requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();

        String flwStCd = followStatusFollowHuyou;

        // DA-139-U002パラメータ設定
        KseflWriteRecord record = new KseflWriteRecord(gmb, kasgYmd, kasHukkYmd, null, sgyAnknNo, flwStCd, fltkYmd,
                null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null);
        Set<KseflColumn> targetColumns = Set.of(KseflColumn.FLW_ST_CD, KseflColumn.FLTK_YMD);
        KseflU002UpdateKey key = new KseflU002UpdateKey(gmb, kasgYmd, kasHukkYmd, sgyAnknNo, recUpdJnjNo);

        String hukkSysShbt = this.requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));

        String decodedHkkSysSyb = kaiheisenConstantService
                .findNameByTypeAndValue(ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE, hukkSysShbt);
        AuditContext context = new AuditContext(USR_ID, decodedHkkSysSyb);

        // DA-139-U002:GMB・KASG_YMD・KAS_HUKK_YMD・SGY_ANKN_NO・REC_UPD_JNJ_NOによる更新。(updateCountで更新件数を受け取る)
        try {
            kseflDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNoAndRecUpdJnjNo(record, targetColumns, key,
                    context);

        } catch (DatabaseException e) {
            String daIdentifier = e.getDaIdentifier();
            String sql = e.getSql();
            Object[] sqlBindPar = e.getSqlBindParameters();
            Throwable cause = e.getCause();

            // DA-139-U002で排他エラー（030400001）が返却された場合
            if (cause.getCause() instanceof ExclusiveException) {

                // DA-139-R004:ガスメーター社番・開栓作業年月日・開栓報告年月日・作業案件番号による取得。
                KseflRecord kseflRecord = findKseflRecord(gmb, kasgYmd, kasHukkYmd, sgyAnknNo);
                // フォロー状態の判定
                if (!Objects.equals(followStatusMityakushu, kseflRecord.flwStCd())) {

                    // 例外をスローする(業務エラー)
                    throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000043);

                } else {

                    // 例外をスローする(DBエラー)
                    throw new DatabaseException(daIdentifier, sql, sqlBindPar, cause);

                }
            } else {
                throw e;

            }
        }
    }

    private KseflRecord findKseflRecord(String gmb, LocalDate kasgYmd, LocalDate kasHukkYmd, String sgyAnknNo) {

        Optional<KseflRecord> record = kseflDataAccess.findByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNo(gmb, kasgYmd,
                kasHukkYmd, sgyAnknNo);
        if (!record.isEmpty()) {
            return record.get();
        }
        throw DataInconsistencySupport.create(kseflDataAccess.tableName(),
                KseflDataAccess.CallSpecs.FIND_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO.identifier(),
                new FlatJsonBuilder().put("gmb", gmb).put("kasgYmd", kasgYmd).put("kasHukkYmd", kasHukkYmd)
                        .put("sgyAnknNo", sgyAnknNo));

    }

}
