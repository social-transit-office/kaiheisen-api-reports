package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.BuildingTransferReceptionRegistration;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferReceptionRegistrationBuildingInfoChange;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferReceptionRegistrationBuildingNameChange;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KiycstKktInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TatemonKsenAtoInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TatemonKsenMaeInf;

/**
 * 概要: SV-236（建物異動受付登録）のサービス実装。
 * 主な仕様:
 * - 変更フラグと建物条件を評価し、`SV-237` と `SV-238` の呼出し要否を制御する。
 * - 下位 SV 呼出しが必要な経路では、RequestContext の必須値を事前に fail-fast で検証する。
 * 既知の制約:
 * - 建物条件不成立または変更なしの場合は、下位 SV を呼び出さずに終了する。
 */
public final class BuildingTransferReceptionRegistrationImpl
        implements BuildingTransferReceptionRegistration {

    private final TransferReceptionRegistrationBuildingNameChange buildingNameChangeService;

    private final TransferReceptionRegistrationBuildingInfoChange buildingInfoChangeService;

    private final RequestContext requestContext;

    private final String housingClassificationCodeNotSpecified;

    /**
     * 概要: 必須依存を受け取って SV-236 実装を初期化する。
     *
     * @param buildingNameChangeService SV-237
     * @param buildingInfoChangeService SV-238
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public BuildingTransferReceptionRegistrationImpl(
            TransferReceptionRegistrationBuildingNameChange buildingNameChangeService,
            TransferReceptionRegistrationBuildingInfoChange buildingInfoChangeService,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.buildingNameChangeService =
                Objects.requireNonNull(buildingNameChangeService, "buildingNameChangeService");
        this.buildingInfoChangeService =
                Objects.requireNonNull(buildingInfoChangeService, "buildingInfoChangeService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.housingClassificationCodeNotSpecified = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.HousingClassificationCode.TYPE,
                ConstantLookupKeys.Cirius.HousingClassificationCode.NOT_SPECIFIED);
    }

    @Override
    public void execute(
            Boolean bldNameHnkuFlg,
            Boolean bldTbgHnkuFlg,
            String bldNo,
            LocalDate uktkYmd,
            Integer uketkTime,
            String hkkEmpNo,
            LocalDate hkkKsenSaymd,
            String tgTkurKbn,
            String tgtnKn,
            String tgtnKj,
            String hkkNameKj,
            Long kktTksuSvkyNo,
            Long tksuSvkyNo,
            String duknIsrgCd,
            KiycstKktInf kiycstKktInf,
            Boolean suguHtkFlg,
            Long kktSvkyNo,
            LocalDate hesnYmd,
            Long kktKyoTrkNo,
            Long kyoTrkNo,
            Long hkkKsenShjSu,
            String kngmStbJtkb,
            String gmtThkgJysb,
            String stkbKbn,
            String shtiKskbNo,
            String ckrtGmscJtkb,
            String kkyuGmtKbn,
            String gmb,
            Long gasMtstNo,
            Boolean tatemonUmFlg,
            TatemonKsenMaeInf tatemonKsenMaeInf,
            TatemonKsenAtoInf tatemonKsenAtoInf) {
        if (!Boolean.TRUE.equals(bldNameHnkuFlg) && !Boolean.TRUE.equals(bldTbgHnkuFlg)) {
            return;
        }
        if (!isBuildingTransferApplicable(tatemonUmFlg, tatemonKsenMaeInf)) {
            return;
        }

        requireRequestContextValues();

        if (Boolean.TRUE.equals(bldNameHnkuFlg)) {
            buildingNameChangeService.execute(
                    bldNo,
                    uktkYmd,
                    uketkTime,
                    hkkEmpNo,
                    hkkKsenSaymd,
                    tgTkurKbn,
                    hkkNameKj,
                    kktTksuSvkyNo,
                    tksuSvkyNo,
                    duknIsrgCd,
                    kiycstKktInf,
                    suguHtkFlg,
                    kktSvkyNo,
                    hesnYmd,
                    kktKyoTrkNo,
                    kyoTrkNo,
                    hkkKsenShjSu,
                    kngmStbJtkb,
                    gmtThkgJysb,
                    stkbKbn,
                    shtiKskbNo,
                    ckrtGmscJtkb,
                    kkyuGmtKbn,
                    gmb);
            buildingInfoChangeService.execute(
                    bldNo,
                    uktkYmd,
                    uketkTime,
                    hkkEmpNo,
                    hkkKsenSaymd,
                    tgTkurKbn,
                    gasMtstNo,
                    hkkNameKj,
                    kktTksuSvkyNo,
                    duknIsrgCd,
                    kiycstKktInf,
                    tatemonKsenMaeInf,
                    tatemonKsenAtoInf);
            return;
        }

        if (Boolean.TRUE.equals(bldTbgHnkuFlg)) {
            buildingInfoChangeService.execute(
                    bldNo,
                    uktkYmd,
                    uketkTime,
                    hkkEmpNo,
                    hkkKsenSaymd,
                    tgTkurKbn,
                    gasMtstNo,
                    hkkNameKj,
                    kktTksuSvkyNo,
                    duknIsrgCd,
                    kiycstKktInf,
                    tatemonKsenMaeInf,
                    tatemonKsenAtoInf);
        }
    }

    /**
     * 概要: 建物異動受付を実施すべき建物条件かを判定する。
     *
     * @param tatemonUmFlg 建物有無フラグ
     * @param tatemonKsenMaeInf 建物情報（開栓前）
     * @return 建物異動受付の対象であれば true
     */
    private boolean isBuildingTransferApplicable(
            Boolean tatemonUmFlg,
            TatemonKsenMaeInf tatemonKsenMaeInf) {
        return Boolean.TRUE.equals(tatemonUmFlg)
                && tatemonKsenMaeInf != null
                && housingClassificationCodeNotSpecified.equals(tatemonKsenMaeInf.jutakKbnStCd());
    }

    /**
     * 概要: 下位 SV 呼出しに必要な RequestContext の必須値を検証する。
     */
    private void requireRequestContextValues() {
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
}
