    private static void appendPackedLong(StringBuilder builder, Long value, int digits) {
        appendPacked(builder, Long.toString(value), digits);
    }

    private static void appendPacked(StringBuilder builder, String rawValue, int digits) {
        String normalizedValue = rawValue.trim();
        if (normalizedValue.length() > digits) {
            throw new IllegalArgumentException(
                "Packed value exceeds fixed length: digits=" + digits + ", actualLength=" + normalizedValue.length());
        }
        String paddedDigits = normalizedValue.length() == digits
            ? normalizedValue
            : "0".repeat(digits - normalizedValue.length()) + normalizedValue;
        String nibbles = (digits % 2 == 0 ? "0" : "") + paddedDigits + "C";
        builder.append(nibbles.toUpperCase());
    }

    private static void appendChar(StringBuilder builder, String value, int length) {
        byte[] fixedLengthBytes = toFixedLengthEbcdicBytes(value, length, false);
        builder.append(HexFormat.of().formatHex(fixedLengthBytes).toUpperCase());
    }

    private static void appendGraphic(StringBuilder builder, String value, int length) {
        byte[] fixedLengthBytes = toFixedLengthEbcdicBytes(value, length, true);
        String hex = HexFormat.of().formatHex(fixedLengthBytes).toUpperCase();
        builder.append(hex.substring(2, hex.length() - 2));
    }

    private static byte[] toFixedLengthEbcdicBytes(String value, int length, boolean graphic) {
        byte[] fixedLengthBytes = new byte[graphic ? (length * 2) + 2 : length];
        Arrays.fill(fixedLengthBytes, EBCDIC_SPACE);
        if (graphic) {
            fixedLengthBytes[0] = 0x0E;
            fixedLengthBytes[fixedLengthBytes.length - 1] = 0x0F;
            Arrays.fill(fixedLengthBytes, 1, fixedLengthBytes.length - 1, EBCDIC_SPACE);
        }

        String normalized = value == null ? "" : value;
        if (normalized.isEmpty()) {
            return fixedLengthBytes;
        }

        int contentLength = graphic ? fixedLengthBytes.length - 2 : fixedLengthBytes.length;
        int acceptedEndIndex = 0;
        for (int index = 0; index < normalized.length();) {
            int nextIndex = index + Character.charCount(normalized.codePointAt(index));
            byte[] encodedCandidate = graphic
                    ? stripGraphicShiftCodes(normalized.substring(0, nextIndex).getBytes(EBCDIC_CHARSET))
                    : normalized.substring(0, nextIndex).getBytes(EBCDIC_CHARSET);
            if (encodedCandidate.length > contentLength) {
                break;
            }
            acceptedEndIndex = nextIndex;
            index = nextIndex;
        }

        byte[] encodedBytes = graphic
                ? stripGraphicShiftCodes(normalized.substring(0, acceptedEndIndex).getBytes(EBCDIC_CHARSET))
                : normalized.substring(0, acceptedEndIndex).getBytes(EBCDIC_CHARSET);
        if (graphic) {
            System.arraycopy(encodedBytes, 0, fixedLengthBytes, 1, encodedBytes.length);
            return fixedLengthBytes;
        }

        System.arraycopy(encodedBytes, 0, fixedLengthBytes, 0, encodedBytes.length);
        return fixedLengthBytes;
    }

    private static byte[] stripGraphicShiftCodes(byte[] encodedBytes) {
        byte[] stripped = new byte[encodedBytes.length];
        int writeIndex = 0;
        for (byte encodedByte : encodedBytes) {
            if (encodedByte == 0x0E || encodedByte == 0x0F) {
                continue;
            }
            stripped[writeIndex++] = encodedByte;
        }
        return Arrays.copyOf(stripped, writeIndex);
    }

    private static <T, R> R valueOf(T source, java.util.function.Function<T, R> accessor) {
        return source == null ? null : accessor.apply(source);
    }

    private static String toFlag(Boolean flag) {
        if (flag == null) {
            return null;
        }
        return flag ? "1" : "0";
    }

    private static Charset resolveEbcdicCharset() {
        for (String candidate : List.of("Cp930", "x-IBM930", "IBM930")) {
            try {
                return Charset.forName(candidate);
            } catch (IllegalCharsetNameException | UnsupportedCharsetException ignored) {
                // try next alias
            }
        }
        throw new IllegalStateException("Cp930 compatible charset is required");
    }
