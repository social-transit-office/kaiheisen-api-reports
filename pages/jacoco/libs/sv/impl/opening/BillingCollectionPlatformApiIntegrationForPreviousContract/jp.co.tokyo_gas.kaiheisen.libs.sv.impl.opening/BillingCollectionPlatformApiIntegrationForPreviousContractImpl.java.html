package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.keizokushiyou_kaisenkannrenji_idoujouhou_touroku.KeizokushiyouKaisenkannrenjiIdoujouhouTourokuCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.keizokushiyou_kaisenkannrenji_idoujouhou_touroku.KeizokushiyouKaisenkannrenjiIdoujouhouTourokuRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.BillingCollectionPlatformApiIntegrationForPreviousContract;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.billingcollectionplatformapiintegrationforpreviouscontract.model.SktsSihHkirInf;

/**
 * 概要: SV-058（請求回収基盤API連携（前契約継続））のサービス実装。
 * 主な仕様:
 * - grouped input を CIRIUS 他-046 request へマッピングして連携する。
 */
public final class BillingCollectionPlatformApiIntegrationForPreviousContractImpl
        implements BillingCollectionPlatformApiIntegrationForPreviousContract {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final int ZERO_DATE = 0;

    private final KeizokushiyouKaisenkannrenjiIdoujouhouTourokuCiriusAdapter adapter;

    private final String presenceFlagYes;

    private final String presenceFlagNothing;

    /**
     * 概要: CDI から依存を受け取って初期化する。
     *
     * @param adapter 他-046 adapter
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public BillingCollectionPlatformApiIntegrationForPreviousContractImpl(
            KeizokushiyouKaisenkannrenjiIdoujouhouTourokuCiriusAdapter adapter,
            CiriusConstantService ciriusConstantService) {
        this(
                adapter,
                RequiredConstantResolver.requireCiriusConstant(
                        Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                        ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                        ConstantLookupKeys.Cirius.PresenceFlag.YES),
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                        ConstantLookupKeys.Cirius.PresenceFlag.NOTHING));
    }

    BillingCollectionPlatformApiIntegrationForPreviousContractImpl(
            KeizokushiyouKaisenkannrenjiIdoujouhouTourokuCiriusAdapter adapter,
            String presenceFlagYes,
            String presenceFlagNothing) {
        this.adapter = Objects.requireNonNull(adapter, "adapter");
        this.presenceFlagYes = Objects.requireNonNull(presenceFlagYes, "presenceFlagYes");
        this.presenceFlagNothing = Objects.requireNonNull(presenceFlagNothing, "presenceFlagNothing");
    }

    /**
     * 概要: 前契約継続パターンの請求回収基盤連携を実行する。
     */
    @Override
    public void execute(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihTrirInf sktsSihTrirInf,
            SktsSihHkirInf sktsSihHkirInf,
            SktsKykHkirInf sktsKykHkirInf,
            SktsSihDlirInf sktsSihDlirInf) {
        adapter.register(buildRequest(
                skkbRkySgjhInf,
                sktsSihTrirInf,
                sktsSihHkirInf,
                sktsKykHkirInf,
                sktsSihDlirInf));
    }

    private KeizokushiyouKaisenkannrenjiIdoujouhouTourokuRequest buildRequest(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihTrirInf sktsSihTrirInf,
            SktsSihHkirInf sktsSihHkirInf,
            SktsKykHkirInf sktsKykHkirInf,
            SktsSihDlirInf sktsSihDlirInf) {
        return new KeizokushiyouKaisenkannrenjiIdoujouhouTourokuRequest(
            new KeizokushiyouKaisenkannrenjiIdoujouhouTourokuRequest
                .MaekeiyakukeizokujiKeizokushiyouKaisenkannrenIdoujouhou(
                skkbRkySgjhInf.skrkGrpKrno(),
                toOptionalLong(skkbRkySgjhInf.sikKrno()),
                skkbRkySgjhInf.sqtiKiyNo(),
                skkbRkySgjhInf.svkyNo(),
                skkbRkySgjhInf.svkySkno(),
                skkbRkySgjhInf.sihKiyNo(),
                skkbRkySgjhInf.sihKiySkno(),
                skkbRkySgjhInf.gasMtstNo(),
                toYmd(skkbRkySgjhInf.tkysYmd()),
                toYmd(skkbRkySgjhInf.khsYmd()),
                toPresenceFlag(skkbRkySgjhInf.trksUmFlg()),
                toPresenceFlag(skkbRkySgjhInf.sihakyKykUmFlg()),
                toPresenceFlag(skkbRkySgjhInf.ssbShbtUmFlg()),
                toPresenceFlag(skkbRkySgjhInf.syoTisiUmFlg()),
                toYmd(skkbRkySgjhInf.shrYmd()),
                sktsSihTrirInf.vtkzNo(),
                sktsSihTrirInf.vtstCd(),
                toOptionalLong(sktsSihTrirInf.irimCstrNo()),
                sktsSihTrirInf.kznKn(),
                sktsSihTrirInf.kznKj(),
                sktsSihTrirInf.shsyRrskShbt1(),
                sktsSihTrirInf.shsyRrskShbt2(),
                sktsSihTrirInf.shsyTelNo1(),
                sktsSihTrirInf.shsyTelNo2(),
                sktsSihTrirInf.shsyTelShbt1(),
                sktsSihTrirInf.shsyTelShbt2(),
                sktsSihTrirInf.isrrGymCd(),
                toYmd(sktsSihTrirInf.sikEndYmd()),
                toYmd(sktsSihTrirInf.sikSttYmd()),
                sktsSihTrirInf.nwjgCd(),
                sktsSihHkirInf.isrrGymCd(),
                toYmd(sktsSihHkirInf.sikEndYmd()),
                sktsKykHkirInf.svkjyShbt(),
                toYmd(sktsKykHkirInf.svtkYmd()),
                toYmd(sktsKykHkirInf.kiyEndYmd()),
                sktsKykHkirInf.isrrGymCd(),
                toOptionalLong(sktsKykHkirInf.uwhyCstuNo()),
                toYmd(sktsKykHkirInf.szirHtyoYmd()),
                sktsSihDlirInf.isrrGymCd()));
    }

    private String toPresenceFlag(Boolean value) {
        return Boolean.TRUE.equals(value) ? presenceFlagYes : presenceFlagNothing;
    }

    private Long toOptionalLong(Long value) {
        if (value == null || value.longValue() == 0L) {
            return null;
        }
        return value;
    }

    private Integer toYmd(LocalDate value) {
        return value == null ? ZERO_DATE : Integer.valueOf(BASIC_DATE.format(value));
    }
}
