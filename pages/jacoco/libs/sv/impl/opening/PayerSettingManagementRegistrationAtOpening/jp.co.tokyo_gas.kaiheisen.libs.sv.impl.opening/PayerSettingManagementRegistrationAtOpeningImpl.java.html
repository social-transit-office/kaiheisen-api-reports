package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.KsshstkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.write.KsshstkWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.PayerSettingManagementRegistrationAtOpening;

/**
 * 概要: SV-235（開栓時支払者設定管理登録）のサービス実装。
 * 主な仕様:
 * - DA-115-R001 で開栓時支払者設定管理の有無を確認する。
 * - 取得結果が 0 件の場合のみ DA-115-C001 で 1 件登録する。
 * 既知の制約:
 * - 報告システム種別は RequestContext から取得し、最終更新起点機能 ID へデコードして渡す。
 * - correlationId と systemDatetime は本 SV の public 引数では扱わない。
 */
public final class PayerSettingManagementRegistrationAtOpeningImpl
        implements PayerSettingManagementRegistrationAtOpening {

    static final String SERVICE_USER_ID = "KHSS235";

    private final KsshstkDataAccess ksshstkDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * 概要: SV-235 の実行に必要な依存オブジェクトを初期化する。
     *
     * @param ksshstkDataAccess 開栓時支払者設定管理データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public PayerSettingManagementRegistrationAtOpeningImpl(
            KsshstkDataAccess ksshstkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.ksshstkDataAccess = Objects.requireNonNull(ksshstkDataAccess, "ksshstkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
    }

    /**
     * 概要: 開栓時支払者設定管理が未登録の場合に 1 件登録する。
     *
     * @param kassSvkyNo 開栓先サービス契約番号
     * @param sgyAnknNo 作業案件番号
     */
    @Override
    public void execute(Long kassSvkyNo, String sgyAnknNo) {
        if (ksshstkDataAccess.findBySvkyNo(kassSvkyNo).isPresent()) {
            return;
        }

        ksshstkDataAccess.insert(
            new KsshstkWriteRecord(kassSvkyNo, sgyAnknNo, null),
                AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId()));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }
}
