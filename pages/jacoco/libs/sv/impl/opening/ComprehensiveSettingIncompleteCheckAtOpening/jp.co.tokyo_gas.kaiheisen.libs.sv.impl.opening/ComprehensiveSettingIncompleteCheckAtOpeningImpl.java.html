package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleSougouSetteiMisaiException;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ComprehensiveSettingIncompleteCheckAtOpening;

/**
 * 概要: 開栓時総合設定未済チェックサービス実装。
 * 主な仕様:
 * - 同型式並列メーター確認フラグが false の場合のみ総合設定未済判定を行う。
 * - ＴＧ他小売区分が他小売以外、メーターグループ＜導管＞有無フラグが true、サービス契約状態区分が契約適用中以外のとき総合設定未済エラーを送出する。
 * 既知の制約:
 * - RequestContext 値は public 引数へ含めず、本クラスでも参照しない。
 */
public final class ComprehensiveSettingIncompleteCheckAtOpeningImpl
        implements ComprehensiveSettingIncompleteCheckAtOpening {

    private final String otherRetailCategory;

    private final String serviceContractStatusUnderContract;

    /**
     * 概要: 判定に必要な定数値を初期化する。
     * 主な仕様:
     * - KH005 の他小売区分と J0566 の契約適用中を constructor で fail-fast 解決する。
     * 既知の制約:
     * - 必須定数不足時は初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public ComprehensiveSettingIncompleteCheckAtOpeningImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.otherRetailCategory = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
        this.serviceContractStatusUnderContract = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
    }

    /**
     * 概要: 開栓時の総合設定未済条件を判定する。
     * 主な仕様:
     * - 同型式並列メーター確認フラグが false のときのみ総合設定未済判定へ進む。
     * - 条件一致時はガスメーター設置場所番号を埋め込んだ総合設定未済エラーを送出する。
     * @param gasMtstNo ガスメーター設置場所番号
     * @param hrtMtrKknFlg 同型式並列メーター確認フラグ
     * @param tgTkurKbn ＴＧ他小売区分
     * @param mtrgrpdUmFlg メーターグループ＜導管＞有無フラグ
     * @param svkyJtkb サービス契約状態区分
     * @throws KaisenRuleSougouSetteiMisaiException 総合設定未済条件に該当する場合
     */
    @Override
        public void execute(
            Long gasMtstNo,
            Boolean hrtMtrKknFlg,
            String tgTkurKbn,
            Boolean mtrgrpdUmFlg,
            String svkyJtkb) {
        if (Boolean.FALSE.equals(hrtMtrKknFlg)
                && !otherRetailCategory.equals(tgTkurKbn)
                && Boolean.TRUE.equals(mtrgrpdUmFlg)
                && !serviceContractStatusUnderContract.equals(svkyJtkb)) {
            throw new KaisenRuleSougouSetteiMisaiException(toCustomerNumber(gasMtstNo));
        }

    }

    /**
     * 概要: 例外へ埋め込むお客さま番号文字列へ変換する。
     * 主な仕様:
     * - null はそのまま例外側へ委譲し、非 null は文字列化する。
     * 既知の制約:
     * - ガスメーター設置場所番号をそのままお客さま番号埋込み値として扱う。
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 例外埋込み用お客さま番号
     */
    private String toCustomerNumber(Long gasMtstNo) {
        return gasMtstNo == null ? null : String.valueOf(gasMtstNo);
    }
}
