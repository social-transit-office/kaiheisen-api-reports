package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

/**
 * 概要: SV-266（開栓エラーフォローフォロー済更新）のサービス実装。
 * 主な仕様:
 * - DA-066-R001 で検針分担エリア情報を取得し、未取得時はデータ不整合例外を送出する。
 * - 確定サービス契約番号が null 以外の場合はそれを採用し、null の場合は確定託送サービス契約番号を採用する。
 * - DA-139-U002 で KSEFL をフォロー済へ更新し、更新件数 0 件はデータ不整合例外を送出する。
 * 既知の制約:
 * - RequestContext の systemDatetime と hukkSysShbt は上位層で設定済みであることを前提とする。
 * - `correlationId` は本 SV のメソッド引数に持たず、RequestContext/MDC 側で扱う。
 */
public class OpeningErrorFollowCompletedUpdateImpl implements OpeningErrorFollowCompletedUpdate {

    static final String SERVICE_USER_ID = "KHSS266";

    private static final Set<KseflColumn> TARGET_COLUMNS = Set.of(
            KseflColumn.FLW_ST_CD,
            KseflColumn.FLTK_YMD,
            KseflColumn.TSIG_KASG_YMD,
            KseflColumn.TSIG_GMB,
            KseflColumn.TSIG_GMSB_NO,
            KseflColumn.TSGK_KYO_TRNO,
            KseflColumn.TSIG_TKYK_NO,
            KseflColumn.SIH_KIY_NO,
            KseflColumn.TSIG_CKTK_CD,
            KseflColumn.SVKY_NO);

    private final KbagmstDataAccess kbagmstDataAccess;

    private final KseflDataAccess kseflDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String followCompletedStatusCode;

    /**
     * 概要: SV-266 の実行に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - DA、RequestContext、開閉栓定数サービスを保持する。
     * - フォロー済更新に使う CIRIUS 定数をコンストラクタ初期化時に検証する。
     * @param kbagmstDataAccess KBAGMST データアクセス
     * @param kseflDataAccess KSEFL データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public OpeningErrorFollowCompletedUpdateImpl(
            KbagmstDataAccess kbagmstDataAccess,
            KseflDataAccess kseflDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.kbagmstDataAccess = Objects.requireNonNull(kbagmstDataAccess, "kbagmstDataAccess");
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.followCompletedStatusCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_FOLLOW_COMPLETED);
    }

    /**
     * 概要: 開栓エラーフォロー件名をフォロー済へ更新する。
     * 主な仕様:
     * - RequestContext の必須値を使って監査情報と付帯作業年月日を解決する。
     * - DA-066-R001 の地域統括コードを DA-139-U002 の更新値へ転記する。
     * - 更新対象列は設計書の SET 対象に限定する。
     * @param kseflKeyInf 開栓エラーフォロー件名キー情報
     * @param kasgYmd 開栓作業年月日
     * @param tkykNo 託送契約番号
     * @param gmb ガスメーター社番
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
     * @param kktSihKiyNo 確定支払契約番号
     * @param kktSvkyNo 確定サービス契約番号
     * @param kktTksuSvkyNo 確定託送サービス契約番号
     */
    @Override
    public void execute(
            KseflKeyInf kseflKeyInf,
            LocalDate kasgYmd,
            String tkykNo,
            String gmb,
            Long gasMtstNo,
            Long kktSyoKyoTrno,
            Long kktSihKiyNo,
            Long kktSvkyNo,
            Long kktTksuSvkyNo) {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        String decodedOriginFunctionId = decodeOriginFunctionId();
        YearMonth knskKjnYm = YearMonth.of(kasgYmd.getYear(), kasgYmd.getMonthValue());
        KbagmstRecord kbagmstRecord = kbagmstDataAccess.findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, knskKjnYm)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kbagmstDataAccess.tableName(),
                        KbagmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM.identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYm", knskKjnYm.toString())));

        Long resolvedSvkyNo = resolveServiceContractNumber(kktSvkyNo, kktTksuSvkyNo);
        LocalDate fltkYmd = systemDatetime.toLocalDate();
        int updatedCount = kseflDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNoAndRecUpdJnjNo(
                buildWriteRecord(
                        kasgYmd,
                        gmb,
                        gasMtstNo,
                        kktSyoKyoTrno,
                        tkykNo,
                        kktSihKiyNo,
                        kbagmstRecord.kbabCktkCd(),
                        resolvedSvkyNo,
                        fltkYmd),
                TARGET_COLUMNS,
                new KseflU002UpdateKey(
                        kseflKeyInf.gmb(),
                        kseflKeyInf.kasgYmd(),
                        kseflKeyInf.kasHukkYmd(),
                        kseflKeyInf.sgyAnknNo(),
                        kseflKeyInf.recUpdJnjNo()),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kseflDataAccess.tableName(),
                    KseflDataAccess.CallSpecs.UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("gmb", kseflKeyInf.gmb())
                            .put("kasgYmd", kseflKeyInf.kasgYmd())
                            .put("kasHukkYmd", kseflKeyInf.kasHukkYmd())
                            .put("sgyAnknNo", kseflKeyInf.sgyAnknNo())
                            .put("recUpdJnjNo", kseflKeyInf.recUpdJnjNo())
                            .put("flwStCd", followCompletedStatusCode)
                            .put("fltkYmd", fltkYmd)
                            .put("tsigKasgYmd", kasgYmd)
                            .put("tsigGmb", gmb)
                            .put("tsigGmsbNo", gasMtstNo)
                            .put("tsgkKyoTrno", kktSyoKyoTrno)
                            .put("tsigTkykNo", tkykNo)
                            .put("sihKiyNo", kktSihKiyNo)
                            .put("tsigCktkCd", kbagmstRecord.kbabCktkCd())
                            .put("svkyNo", resolvedSvkyNo)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private Long resolveServiceContractNumber(Long kktSvkyNo, Long kktTksuSvkyNo) {
        if (kktSvkyNo != null) {
            return kktSvkyNo;
        }
        return kktTksuSvkyNo;
    }

    private KseflWriteRecord buildWriteRecord(
            LocalDate kasgYmd,
            String gmb,
            Long gasMtstNo,
            Long kktSyoKyoTrno,
            String tkykNo,
            Long kktSihKiyNo,
            String tsigCktkCd,
            Long svkyNo,
            LocalDate fltkYmd) {
        return KseflWriteRecord.forFollowCompletedUpdate(
                followCompletedStatusCode,
                fltkYmd,
                kasgYmd,
                gmb,
                gasMtstNo,
                kktSyoKyoTrno,
                tkykNo,
                tsigCktkCd,
                kktSihKiyNo,
                svkyNo);
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
