package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ComprehensiveMeterJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ComprehensiveMeterJudgmentAtOpeningOutput;

/**
 * 概要: 開栓時総合メーター判定サービス実装。
 * 主な仕様:
 * - 料金計算関連状態区分が関連有効または関連終了待ちの場合は DA-041-R003 の件数で総合判定を行う。
 * - それ以外の場合は DA-041-R004 の取得結果から最大料金計算終了年月日を求め、継続使用扱いコードを決定する。
 * - 判定に必要な CIRIUS / 開閉栓定数はコンストラクタで fail-fast に解決する。
 * 既知の制約:
 * - public execute 引数の null / blank 判定は上位層責務とし、本クラスでは追加ガードしない。
 */
public final class ComprehensiveMeterJudgmentAtOpeningImpl
        implements ComprehensiveMeterJudgmentAtOpening {

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final String chargeCalculationRelatedStatusValid;

    private final String chargeCalculationRelatedStatusEndWaiting;

    private final String chargeCalculationRelatedStatusEnded;

    private final String openingReasonContinuedUse;

    private final String continuousUseHandlingCodeComprehensiveWarningTarget;

    private final String continuousUseHandlingCodeAlternativeOpeningTarget;

    private final List<String> activeOrEndWaitingStatuses;

    private final List<String> activeWaitingOrEndedStatuses;

    /**
     * 概要: 判定に必要な依存と定数値を初期化する。
     * 主な仕様:
     * - SVMSTKR 参照 DA を保持する。
     * - J0611、04900、KH007 の必須定数をコンストラクタで fail-fast 解決する。
     * 既知の制約:
     * - DI 依存不足は Objects.requireNonNull によりインスタンス生成時に失敗する。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public ComprehensiveMeterJudgmentAtOpeningImpl(
            SvmstkrDataAccess svmstkrDataAccess,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.chargeCalculationRelatedStatusValid = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.chargeCalculationRelatedStatusEndWaiting = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.chargeCalculationRelatedStatusEnded = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.openingReasonContinuedUse = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.continuousUseHandlingCodeComprehensiveWarningTarget =
                RequiredConstantResolver.requireKaiheisenConstant(
                        nonNullKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode
                                .COMPREHENSIVE_WARNING_TARGET);
        this.continuousUseHandlingCodeAlternativeOpeningTarget =
                RequiredConstantResolver.requireKaiheisenConstant(
                        nonNullKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode
                                .ALTERNATIVE_OPENING_TARGET);
        this.activeOrEndWaitingStatuses = List.of(
                chargeCalculationRelatedStatusValid,
                chargeCalculationRelatedStatusEndWaiting);
        this.activeWaitingOrEndedStatuses = List.of(
                chargeCalculationRelatedStatusValid,
                chargeCalculationRelatedStatusEndWaiting,
                chargeCalculationRelatedStatusEnded);
    }

    /**
     * 概要: 開栓前時点で総合メーターだったかを判定する。
     * 主な仕様:
     * - 関連有効または関連終了待ちでは DA-041-R003 の件数が 1 件なら false、複数件なら true と判定する。
     * - それ以外では DA-041-R004 の取得有無で総合判定し、継続使用時は最大料金計算終了年月日との比較で扱いコードを返す。
     * 既知の制約:
     * - DA 返却レコードの料金計算終了年月日が null の場合は最大値計算から除外する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param svkyNo サービス契約番号
     * @param kasJysb 開栓事由種別
     * @param hkkKsenSaymd 報告開栓作業年月日
     * @param rkkiKnrnJtkb 料金計算関連状態区分
     * @param rkkiEndYmd 料金計算終了年月日
     * @return 総合メーター判定結果
     */
    @Override
    public ComprehensiveMeterJudgmentAtOpeningOutput execute(
            Long gasMtstNo,
            Long svkyNo,
            String kasJysb,
            LocalDate hkkKsenSaymd,
            String rkkiKnrnJtkb,
            LocalDate rkkiEndYmd) {
        if (isTargetForCountBranch(rkkiKnrnJtkb)) {
            List<String> activeOrEndWaitingStatusList = activeOrEndWaitingStatusList();
            long relationCount = svmstkrDataAccess.findBySvkyNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                    svkyNo,
                    hkkKsenSaymd,
                    activeOrEndWaitingStatusList);
            if (relationCount == 0) {
                throw createZeroCountException(svkyNo, hkkKsenSaymd, activeOrEndWaitingStatusList);
            }
            return new ComprehensiveMeterJudgmentAtOpeningOutput(relationCount > 1, null);
        }

        List<String> activeWaitingOrEndedStatusList = activeWaitingOrEndedStatusList();
        List<SvmstkrRecord> relatedRecords =
                svmstkrDataAccess.findAllBySvkyNoAndRkkiKnrnJtkbAndRkkiSttYmdAndRkkiEndYmdAndMtstNo(
                        svkyNo,
                        activeWaitingOrEndedStatusList,
                        hkkKsenSaymd,
                        gasMtstNo);
        boolean suguHtkFlg = !relatedRecords.isEmpty();
        LocalDate maximumChargeCalculationEndDate =
                findMaximumChargeCalculationEndDate(relatedRecords);

        return new ComprehensiveMeterJudgmentAtOpeningOutput(
                suguHtkFlg,
                resolveContinuousUseHandlingCode(
                        kasJysb,
                        suguHtkFlg,
                        rkkiEndYmd,
                        maximumChargeCalculationEndDate));
    }

    /**
     * 概要: DA-041-R003 件数分岐の対象状態かを判定する。
     * 主な仕様:
     * - 関連有効または関連終了待ちに一致する場合に true を返す。
     *
     * @param rkkiKnrnJtkb 料金計算関連状態区分
     * @return 件数分岐の対象であれば true
     */
    private boolean isTargetForCountBranch(String rkkiKnrnJtkb) {
        return rkkiKnrnJtkb.equals(chargeCalculationRelatedStatusValid)
                || rkkiKnrnJtkb.equals(chargeCalculationRelatedStatusEndWaiting);
    }

    /**
     * 概要: DA-041-R003 へ渡す料金計算関連状態区分一覧を返す。
     * 主な仕様:
     * - J0611 の関連有効、関連終了待ちを設計順で返す。
     *
     * @return 判定対象の状態区分一覧
     */
    private List<String> activeOrEndWaitingStatusList() {
        return activeOrEndWaitingStatuses;
    }

    /**
     * 概要: DA-041-R004 へ渡す料金計算関連状態区分一覧を返す。
     * 主な仕様:
     * - J0611 の関連有効、関連終了待ち、関連終了を設計順で返す。
     *
     * @return 判定対象の状態区分一覧
     */
    private List<String> activeWaitingOrEndedStatusList() {
        return activeWaitingOrEndedStatuses;
    }

    /**
     * 概要: DA-041-R003 が 0 件だった場合のデータ不整合エラーを生成する。
     * 主な仕様:
     * - daIdentifier は CallSpecs.COUNT_BY... の identifier を利用する。
     * - parameterJsonString には DA 呼出しパラメータ（サービス契約番号・検索基準年月日・状態区分リスト）を保持する。
     * - tableName は SVMSTKR を設定する。
     *
     * @param svkyNo サービス契約番号
     * @param hkkKsenSaymd 報告開栓作業年月日
     * @param rkkiKnrnJtkbList 料金計算関連状態区分一覧
     * @return DataInconsistencyException（データ不整合エラー）
     */
    private DataInconsistencyException createZeroCountException(
            Long svkyNo,
            LocalDate hkkKsenSaymd,
            List<String> rkkiKnrnJtkbList) {
        return DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                SvmstkrDataAccess.CallSpecs
                        .COUNT_BY_SVKY_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB
                        .identifier(),
                new FlatJsonBuilder()
                        .put("svkyNo", svkyNo)
                        .put("knskKjnYmd", hkkKsenSaymd)
                        .put("rkkiKnrnJtkbList", rkkiKnrnJtkbList));
    }

    /**
     * 概要: 取得結果から最大料金計算終了年月日を算出する。
     * 主な仕様:
     * - レコード順に走査し、より大きい料金計算終了年月日を保持する。
     * - 料金計算終了年月日が null のレコードは比較対象外とする。
     *
     * @param relatedRecords DA-041-R004 の取得結果
     * @return 最大料金計算終了年月日。比較対象がない場合は null
     */
    private LocalDate findMaximumChargeCalculationEndDate(List<SvmstkrRecord> relatedRecords) {
        LocalDate maximumChargeCalculationEndDate = null;
        for (SvmstkrRecord relatedRecord : relatedRecords) {
            LocalDate currentChargeCalculationEndDate = relatedRecord.rkkiEndYmd();
            if (currentChargeCalculationEndDate != null
                    && (maximumChargeCalculationEndDate == null
                            || currentChargeCalculationEndDate.isAfter(maximumChargeCalculationEndDate))) {
                maximumChargeCalculationEndDate = currentChargeCalculationEndDate;
            }
        }
        return maximumChargeCalculationEndDate;
    }

    /**
     * 概要: 継続使用扱いコードを判定する。
     * 主な仕様:
     * - 継続使用かつ総合判定結果フラグ true の場合のみ判定対象とする。
     * - 入力料金計算終了年月日が最大料金計算終了年月日より前なら代替開栓対象、それ以外は総合ワーニング対象を返す。
     *
     * @param kasJysb 開栓事由種別
     * @param suguHtkFlg 総合判定結果フラグ
     * @param rkkiEndYmd 入力の料金計算終了年月日
     * @param maximumChargeCalculationEndDate 取得結果から求めた最大料金計算終了年月日
     * @return 継続使用扱いコード。条件不成立時は null
     */
    private String resolveContinuousUseHandlingCode(
            String kasJysb,
            boolean suguHtkFlg,
            LocalDate rkkiEndYmd,
            LocalDate maximumChargeCalculationEndDate) {
        if (!openingReasonContinuedUse.equals(kasJysb) || !suguHtkFlg) {
            return null;
        }
        if (maximumChargeCalculationEndDate != null
                && rkkiEndYmd.isBefore(maximumChargeCalculationEndDate)) {
            return continuousUseHandlingCodeAlternativeOpeningTarget;
        }
        return continuousUseHandlingCodeComprehensiveWarningTarget;
    }
}
