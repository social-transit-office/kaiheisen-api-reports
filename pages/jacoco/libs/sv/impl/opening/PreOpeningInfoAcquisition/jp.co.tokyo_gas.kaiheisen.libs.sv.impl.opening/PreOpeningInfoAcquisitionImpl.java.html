    /**
     * 概要: 開栓報告前に必要な情報を取得する。
     * 主な仕様: SV-163 の設計書に従い、ガスメーター設置場所番号の解決後にシステム日時点と開栓作業日時点の情報を組み合わせて返却値を構成する。
     * 既知の制約: RequestContext では systemDatetime のみを必須値として利用し、他の文脈値は本 SV の public execute では直接参照しない。
     *
     * @param gmb ガスメーター社番
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kasgYmd 開栓作業年月日
     * @param tgTkurKbn 東京ガス小売区分
     * @param gmtShjSu ガスメーター指示数
     * @return 開栓前情報取得結果
     */
    @Override
    public PreOpeningInfoAcquisitionOutput execute(
        final String gmb,
        final Long gasMtstNo,
        final LocalDate kasgYmd,
        final String tgTkurKbn,
        final Long gmtShjSu) {
        LocalDateTime systemDatetime = requireSystemDatetime();
        LocalDate systemDate = systemDatetime.toLocalDate();
        Long resolvedGasMtstNo = resolveGasMtstNo(gmb, gasMtstNo);

        GmtsetdRecord systemDateGmtsetd =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(resolvedGasMtstNo, systemDate)
                        .orElse(null);
        GmtsetdRecord openingDateGmtsetd =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(resolvedGasMtstNo, kasgYmd)
                        .orElse(null);
        GmtsetdRecord previousGenerationGmtsetd =
                openingDateGmtsetd == null
                        ? null
                        : gmtsetdDataAccess
                                .findByGasMtstNoAndTkysYmd(resolvedGasMtstNo, openingDateGmtsetd.tkysYmd())
                                .orElse(null);
        TatemonRecord tatemonRecord =

    /**
     * 概要: 確定ガスメーター指示数を決定する。
     * 主な仕様: 入力値と最近指示数を比較し、大きい方を返す。片方が `null` の場合は他方を採用する。
     * 既知の制約: 両方 `null` の場合は `null` を返し、数値妥当性の追加検証は行わない。
     */
    private static Long determineConfirmedMeterInstructionCount(
        Long inputGmtShjSu, Long recentGmtShjSu) {
        if (inputGmtShjSu == null) {
            return recentGmtShjSu;
        }
        if (recentGmtShjSu == null) {
            return inputGmtShjSu;
        }
        return inputGmtShjSu >= recentGmtShjSu ? inputGmtShjSu : recentGmtShjSu;
    }
