    @Inject
    public ContractCustomerRegistrationUpdateOtherRetailAtOpeningImpl(
            CustomerNameKanjiDisplayFlagAcquisition customerNameKanjiDisplayFlagAcquisition,
            KiycstdDataAccess kiycstdDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.customerNameKanjiDisplayFlagAcquisition = Objects.requireNonNull(
                customerNameKanjiDisplayFlagAcquisition,
                "customerNameKanjiDisplayFlagAcquisition");
        this.kiycstdDataAccess = Objects.requireNonNull(kiycstdDataAccess, "kiycstdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.newCustomerPattern = RequiredConstantResolver.requireKaiheisenConstant(
                this.kaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.TYPE,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.NEW_CUSTOMER);

                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.USE_EXISTING_CUSTOMER_NUMBER);
        this.defaultInquiryLevelCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.InquiryLevelType.TYPE,
                ConstantLookupKeys.Cirius.InquiryLevelType.ALL_CAN_BE_QUERIED_AND_DISPLAYED);
    }

        /**
         * 概要: 開栓時契約お客さま登録・更新（他小売）を実行する。
         * 主な仕様:
         * - 契約お客さま情報.お客さま名カナ1 が null または空文字の場合は、入力の契約お客さま情報（確定）を返して終了する。
         * - それ以外では SV-075 でおなまえ漢字表示可否を取得し、3x更新パターン区分に応じて内部変数を編集する。
         * - 直列専用通過メーター対象では外部更新を行わず、戻り値編集のみで終了する。
         * - 新規お客さまでは確定使用者契約お客さま登録番号を使用して DA-104-C001 を実行し、それ以外は DA-104-U001 を実行する。
         * - 更新時に DA-104-U001 の更新件数が 0 件の場合は、当該 DA の呼出し情報を保持した
         *   {@code DataInconsistencyException} を送出する。
         *
         * @param snyTkmtFlg 直列専用通過メーターフラグ
         * @param cstUpdPtnKb 3x更新パターン区分
         * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
         * @param confirmedContractCustomerInfo 契約お客さま情報（確定）
         * @param confirmedAddressInfo 住所情報（確定）
         * @param kiycstInf 契約お客さま情報
         * @param gasMeterLocationInfo ガスメーター設置場所＜導管＞情報
         * @param closingCustomerInfo お客さま情報（閉栓時）
         * @return 契約お客さま情報（確定）の編集結果
         */
    @Override
    public ContractCustomerRegistrationUpdateOtherRetailAtOpeningOutput execute(
            Boolean snyTkmtFlg,
            String cstUpdPtnKb,
            Long kktSyoKyoTrno,
            ConfirmedContractCustomerInfo confirmedContractCustomerInfo,
            ConfirmedAddressInfo confirmedAddressInfo,
            KiycstInf kiycstInf,
            GasMeterLocationInfo gasMeterLocationInfo,
            ClosingCustomerInfo closingCustomerInfo) {
        requireRequestContext();

        if (kiycstInf.ctnmKn1() == null || kiycstInf.ctnmKn1().isEmpty()) {
            return new ContractCustomerRegistrationUpdateOtherRetailAtOpeningOutput(
                    confirmedContractCustomerInfo);
        }

        Boolean customerNameKanjiDisplayFlag = customerNameKanjiDisplayFlagAcquisition
                .execute(kiycstInf.ctnmKj1())
                .onmeKjhkFlg();
        InternalValues internalValues = buildInternalValues(
                cstUpdPtnKb,
                kktSyoKyoTrno,
                confirmedContractCustomerInfo,
                confirmedAddressInfo,
                kiycstInf,
                gasMeterLocationInfo,
                closingCustomerInfo,
                customerNameKanjiDisplayFlag);

        if (Boolean.TRUE.equals(snyTkmtFlg)) {
            return toOutput(internalValues);
        }

        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        if (newCustomerPattern.equals(cstUpdPtnKb)) {
            kiycstdDataAccess.insert(toWriteRecord(internalValues, kktSyoKyoTrno, null), auditContext);
            return toOutput(internalValues);
        }

        int updatedCount = kiycstdDataAccess.updateByKyoTrkNo(
                toWriteRecord(internalValues, internalValues.kyoTrkNo(), confirmedContractCustomerInfo.recUpdJnjNo()),
                UPDATE_TARGET_COLUMNS,
                new KiycstdUpdateKey(internalValues.kyoTrkNo(), confirmedContractCustomerInfo.recUpdJnjNo()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kiycstdDataAccess.tableName(),
                    KiycstdDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO.identifier(),
                    buildParameterJson(internalValues, confirmedContractCustomerInfo.recUpdJnjNo(), decodedOriginFunctionId));
        }
        return toOutput(internalValues);
    }

    private InternalValues buildInternalValues(
            String cstUpdPtnKb,

            ConfirmedAddressInfo confirmedAddressInfo,
            KiycstInf kiycstInf,
            GasMeterLocationInfo gasMeterLocationInfo,
            ClosingCustomerInfo closingCustomerInfo,
            Boolean customerNameKanjiDisplayFlag) {
        if (newCustomerPattern.equals(cstUpdPtnKb)) {
            String closingInquiryLevel = (closingCustomerInfo != null
                    && closingCustomerInfo.kiycstdInf() != null)
                            ? closingCustomerInfo.kiycstdInf().shkiLvShbt()
                            : null;
            return new InternalValues(
                    null,
                    kiycstInf.ctnmKj1(),
                    kiycstInf.ctnmKn1(),
                    kiycstInf.ctnm1HujnShbt(),
                    kiycstInf.ctnm1HshiShbt(),
                    kiycstInf.ctnmKj2(),
                    kiycstInf.ctnmKn2(),
                    kiycstInf.ctnm2HujnShbt(),
                    kiycstInf.ctnm2HshiShbt(),
                    customerNameKanjiDisplayFlag,
                    closingInquiryLevel == null || closingInquiryLevel.isEmpty()
                            ? defaultInquiryLevelCode
                            : closingInquiryLevel,
                    kiycstInf.yubnNo(),
                    gasMeterLocationInfo.gkkCd(),
                    gasMeterLocationInfo.mchCd(),
                    gasMeterLocationInfo.cmazCd(),
                    confirmedAddressInfo.kktCbgSuj1(),
                    confirmedAddressInfo.kktCbgKn2(),
                    confirmedAddressInfo.kktCbgKn3(),
                    confirmedAddressInfo.kktCbgKn4(),
                    gasMeterLocationInfo.bldNo(),
                    null,
                    null,
                    null,
                    null,
                    confirmedAddressInfo.kktTunoKn(),
                    confirmedAddressInfo.kktHynKn(),
                    confirmedAddressInfo.kktHynUmFlg());
        }
        return new InternalValues(
                kktSyoKyoTrno,
                kiycstInf.ctnmKj1(),
                kiycstInf.ctnmKn1(),
                kiycstInf.ctnm1HujnShbt(),
                kiycstInf.ctnm1HshiShbt(),
                kiycstInf.ctnmKj2(),
                kiycstInf.ctnmKn2(),
                kiycstInf.ctnm2HujnShbt(),
                kiycstInf.ctnm2HshiShbt(),
                customerNameKanjiDisplayFlag,
                confirmedContractCustomerInfo.shkiLvShbt(),
                kiycstInf.yubnNo(),
                confirmedContractCustomerInfo.gkkCd(),
                confirmedContractCustomerInfo.mchCd(),
                confirmedContractCustomerInfo.cmazCd(),
                confirmedAddressInfo.kktCbgSuj1(),
                confirmedAddressInfo.kktCbgKn2(),
                confirmedAddressInfo.kktCbgKn3(),
                confirmedAddressInfo.kktCbgKn4(),
                confirmedContractCustomerInfo.bldNo(),
                confirmedContractCustomerInfo.bldnKj(),
                confirmedContractCustomerInfo.bldnKn(),
                confirmedContractCustomerInfo.bldnKjhkFlg(),
                confirmedContractCustomerInfo.jutKbn(),
                confirmedAddressInfo.kktTunoKn(),

    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (originFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return originFunctionId;
    }
