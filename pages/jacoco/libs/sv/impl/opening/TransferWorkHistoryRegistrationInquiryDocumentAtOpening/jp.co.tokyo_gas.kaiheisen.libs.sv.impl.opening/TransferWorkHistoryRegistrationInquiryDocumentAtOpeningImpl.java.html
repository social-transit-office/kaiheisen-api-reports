package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferWorkHistoryCommonRetailEditAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferWorkHistoryCommonRetailEditAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferWorkHistoryRegistrationInquiryDocumentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.IsrkkInsInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TransferWorkHistoryInquirySvckyaKktInf;

/**
 * 概要: SV-245（開栓時異動・作業履歴登録（照会文書用））の実装。
 * 主な仕様:
 * - 処理対象条件を満たす場合のみ SV-247 を呼び出し、一般契約適用の異動・作業履歴＜小売＞登録を行う。
 * - サービス契約開始事由種別が新規または前契約継続でない場合は開栓共通エラーを送出する。
 * - step 1.a.3 の補助ソース定義に従い、他-004 リクエストは SV-247 戻り値と固定値を合成して組み立てる。
 * 既知の制約:
 * - public メソッド引数の null 判定や、下位 SV 戻り値 DTO に対する設計外の null 判定は追加しない。
 */
public final class TransferWorkHistoryRegistrationInquiryDocumentAtOpeningImpl
        implements TransferWorkHistoryRegistrationInquiryDocumentAtOpening {

    private static final String INVALID_SERVICE_CONTRACT_START_REASON_ERROR_CODE =
            ErrorCodeConstants.ERROR_CODE_020100041;

    private static final String EMPTY_STRING = "";

    private final TransferWorkHistoryCommonRetailEditAtOpening commonRetailEditAtOpening;

    private final IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter;

    private final RequestContext requestContext;

    private final String continuedUseOpeningReason;

    private final String newServiceContractStartReason;

    private final String continuationOfPreviousContract;

    private final String generalContractApplicable;

    private final String newOpeningDetailCode;

    private final String previousContractContinuationDetailCode;

    private final String transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable;

    private final String customerInfoAcquisitionPossible;

    /**
     * 概要: SV-245 実行に必要な依存と定数を初期化する。
     * 主な仕様:
     * - SV-247、他-004 adapter、RequestContext を必須依存として保持する。
     * - 分岐判定と固定値設定に必要な CIRIUS 定数をコンストラクタ初期化時に fail-fast で解決する。
     *
     * @param commonRetailEditAtOpening SV-247
     * @param ciriusAdapter 他-004 adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public TransferWorkHistoryRegistrationInquiryDocumentAtOpeningImpl(
            TransferWorkHistoryCommonRetailEditAtOpening commonRetailEditAtOpening,
            IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.commonRetailEditAtOpening = Objects.requireNonNull(
                commonRetailEditAtOpening,
                "commonRetailEditAtOpening");
        this.ciriusAdapter = Objects.requireNonNull(ciriusAdapter, "ciriusAdapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.continuedUseOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.newServiceContractStartReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.NEW);
        this.continuationOfPreviousContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.generalContractApplicable = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.GENERAL_CONTRACT_APPLICABLE);
        this.newOpeningDetailCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.NEW_CCBG5101);
        this.previousContractContinuationDetailCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.NOT_APPLICABLE_TO_DETAILS_ONLY);
        this.customerInfoAcquisitionPossible = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PossibilityFlag.TYPE,
                ConstantLookupKeys.Cirius.PossibilityFlag.POSSIBLE);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void execute(
            String isrrHkkrKbn,
            String hukkUsrId,
            String hukkUnmKj,
            String tgTkurKbn,
            Long gasMtstNo,
            LocalDate kasgYmd,
            String kasJysb,
            Boolean digwrHntFlg,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg,
            TransferWorkHistoryInquirySvckyaKktInf svckyaKktInf) {
        if (!isRegistrationTarget(kasJysb, digwrHntFlg, syoriHnt4XFlg, syoriHntFkktFlg)) {
            return;
        }
        requireSystemDatetime();
        TransferWorkHistoryCommonRetailEditAtOpeningOutput commonEditOutput = commonRetailEditAtOpening.execute(
                gasMtstNo,
                kasgYmd,
                hukkUsrId,
                hukkUnmKj,
                tgTkurKbn);
        String detailCode = resolveDetailCode(svckyaKktInf.svkySttJysb());
        ciriusAdapter.register(toRequest(commonEditOutput.isrkkInsInf(), detailCode, isrrHkkrKbn));
    }

    /**
     * 概要: 処理対象かどうかを判定する。
     *
     * @param kasJysb 開栓事由種別
     * @param digwrHntFlg 代替判定結果フラグ
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param syoriHntFkktFlg 処理判定_復活_フラグ
     * @return 処理対象の場合 true
     */
    private boolean isRegistrationTarget(
            String kasJysb,
            Boolean digwrHntFlg,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg) {
        return Boolean.TRUE.equals(digwrHntFlg)
                || (continuedUseOpeningReason.equals(kasJysb)
                && Boolean.TRUE.equals(syoriHnt4XFlg)
                && Boolean.TRUE.equals(syoriHntFkktFlg));
    }

    /**
     * 概要: サービス契約開始事由種別から異動・作業履歴業務詳細コードを解決する。
     *
     * @param svkySttJysb サービス契約開始事由種別
     * @return 異動・作業履歴業務詳細コード
     */
    private String resolveDetailCode(String svkySttJysb) {
        if (newServiceContractStartReason.equals(svkySttJysb)) {
            return newOpeningDetailCode;
        }
        if (continuationOfPreviousContract.equals(svkySttJysb)) {
            return previousContractContinuationDetailCode;
        }
        throw new KaisenRuleException(INVALID_SERVICE_CONTRACT_START_REASON_ERROR_CODE);
    }

        /**
         * 概要: SV-247 の戻り値と固定値から他-004 リクエストを生成する。
         *
         * @param commonEditResult SV-247 の編集結果
         * @param detailCode 異動・作業履歴業務詳細コード
         * @param isrrHkkrKbn 異動・作業履歴報告経路区分
         * @return 他-004 リクエスト
         */
    private IdousagyourirekiKouriJouhouRequest toRequest(
            IsrkkInsInf commonEditResult,
            String detailCode,
            String isrrHkkrKbn) {
        return new IdousagyourirekiKouriJouhouRequest(
                new IdousagyourirekiKouriJouhouRequest.IdousagyourirekiKouriJouhouRenkei(
                        new IdousagyourirekiKouriJouhouRequest.IdousagyourirekigyoumuKyoutsuuKouri(
                                formatCustomerNumber(commonEditResult.artyCstNo()),
                                commonEditResult.akrkKbn(),
                                commonEditResult.akrkTrskShbt(),
                                generalContractApplicable,
                                detailCode,
                                isrrHkkrKbn,
                                transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable,
                                toDateText(commonEditResult.isrrTriYmd()),
                                toTimeText(commonEditResult.isrrTriHms()),
                                toDateText(commonEditResult.idsgYmd()),
                                emptyIfNull(commonEditResult.isrrHusrId()),
                                emptyIfNull(commonEditResult.isrrHkunKj()),
                                EMPTY_STRING),
                        List.of(),
                        customerInfoAcquisitionPossible,
                        EMPTY_STRING,
                        EMPTY_STRING));
    }

    /**
     * 概要: RequestContext から必須のシステム日時を取得する。
     *
     * @return RequestContext の必須システム日時
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 登録依頼用お客さま番号を 11 桁文字列へ変換する。
     *
     * @param customerNumber 登録依頼用お客さま番号
     * @return 11 桁文字列。null の場合は null
     */
    private String formatCustomerNumber(final Long customerNumber) {
        if (customerNumber == null) {
            return null;
        }
        return "%011d".formatted(customerNumber);
    }

    /**
     * 概要: LocalDate を yyyyMMdd 文字列へ変換する。
     *
     * @param date 日付
     * @return yyyyMMdd 文字列。null の場合は null
     */
    private String toDateText(LocalDate date) {
        return date == null ? null : date.format(java.time.format.DateTimeFormatter.BASIC_ISO_DATE);
    }

    /**
     * 概要: HHmmss 整数を 6 桁文字列へ変換する。
     *
     * @param hhmmss HHmmss 整数
     * @return 6 桁文字列。null の場合は null
     */
    private String toTimeText(Integer hhmmss) {
        return hhmmss == null ? null : "%06d".formatted(hhmmss);
    }

    /**
     * 概要: null を空文字へ補正する。
     *
     * @param value 入力文字列
     * @return null の場合は空文字、それ以外は入力値
     */
    private String emptyIfNull(String value) {
        return value == null ? EMPTY_STRING : value;
    }

}
