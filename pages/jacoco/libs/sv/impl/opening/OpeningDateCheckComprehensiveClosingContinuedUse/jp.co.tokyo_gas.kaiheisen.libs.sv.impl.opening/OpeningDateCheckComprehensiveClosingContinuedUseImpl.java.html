package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckComprehensiveClosingContinuedUse;

/**
 * 概要: SV-185（開栓日チェック（総合閉栓時の継続使用判定））の実装クラス。
 * 主な仕様:
 * - 総合判定結果フラグ、継続使用扱いコード、サービス契約有無フラグ、日付条件を組み合わせて判定する。
 * - 条件がすべて成立する場合は開栓共通エラー 020100018（KaisenRuleException）を送出する。
 * 既知の制約:
 * - 公開メソッド引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 */
public final class OpeningDateCheckComprehensiveClosingContinuedUseImpl
        implements OpeningDateCheckComprehensiveClosingContinuedUse {

    /**
     * 概要: SV 実行時のリクエストコンテキスト。
     * 主な仕様:
     * - 設計書上の共通入力である correlationId、hukkSysShbt、systemDatetime の受け渡し先となる。
     * - SV-185 の判定ロジック自体では参照しないが、既存 SV 実装と同じ依存注入形を維持する。
     * 既知の制約:
     * - 本フィールドはコンストラクタで非 null を必須とする。
     */
    @SuppressWarnings("unused")
    private final RequestContext requestContext;

    /**
     * 継続使用扱いコード: 総合ワーニング対象（KH007-1）
     * fail-fast: 必須定数が取得できない場合はコンストラクタで例外
     */
    private final String comprehensiveWarningTargetCode;

    /**
     * 概要: 共通入力・定数サービスを受け取って初期化する。
     * 主な仕様:
     * - 設計書上の共通入力は RequestContext で扱う。
     * - 継続使用扱いコード（KH007-1）は KaiheisenConstantService + RequiredConstantResolver でfail-fast解決。
     * - 必須依存（requestContext, kaiheisenConstantService）はnull禁止。
     *
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public OpeningDateCheckComprehensiveClosingContinuedUseImpl(
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        // fail-fast: KH007-1 必須定数
        this.comprehensiveWarningTargetCode = RequiredConstantResolver.requireKaiheisenConstant(
                kaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.TYPE,
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.COMPREHENSIVE_WARNING_TARGET);
    }

    /**
     * 概要: 総合閉栓時の継続使用開栓における開栓日エラー判定を実行する。
     * 主な仕様:
     * - 総合判定結果フラグが true、継続使用扱いコードが総合ワーニング対象、サービス契約有無フラグが true、
    *   開栓作業年月日が料金計算開始年月日より前のときに開栓共通エラーを送出する。
     * - それ以外の場合は何も返さずに正常終了する。
     * 既知の制約:
     * - null を含む入力の扱いは上位層責務であり、必要に応じて NullPointerException となる。
     *
     * @param suguHtkFlg 総合判定結果フラグ
     * @param kzksyAtkCd 継続使用扱いコード
     * @param svckyaUmuFlg サービス契約有無フラグ
     * @param rkkiSttYmd 料金計算開始年月日
     * @param kasgYmd 開栓作業年月日
         * @throws KaisenRuleException 開栓共通エラー 020100018 を送出する場合
     */
    @Override
     public void execute(
            Boolean suguHtkFlg,
            String kzksyAtkCd,
            Boolean svckyaUmuFlg,
            LocalDate rkkiSttYmd,
            LocalDate kasgYmd) {
        if (suguHtkFlg
                && isComprehensiveWarningTarget(kzksyAtkCd)
                && svckyaUmuFlg
                && kasgYmd.isBefore(rkkiSttYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }

    }

    /**
     * 概要: 継続使用扱いコードが総合ワーニング対象かを判定する。
     * 主な仕様:
     * - 解決済みの総合ワーニング対象コードと完全一致する場合に true を返却する。
     * 既知の制約:
     * - 引数が null の場合は false を返す（NullPointerException とはならない）。
     *
     * @param kzksyAtkCd 継続使用扱いコード（null可）
     * @return 総合ワーニング対象の場合 true、nullの場合はfalse
     */
    private boolean isComprehensiveWarningTarget(String kzksyAtkCd) {
        return comprehensiveWarningTargetCode.equals(kzksyAtkCd);
    }
}
