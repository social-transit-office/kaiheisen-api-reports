package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferReceptionCommonEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferReceptionCommonEditOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.IdouktkInsInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KiycstKktInf;

/**
 * 概要: SV-239（異動受付共通編集）のサービス実装。
 * 主な仕様:
 * - 建物に関する異動受付の共通項目を編集し、異動受付情報を返却する。
 * - 業務コード、漢字表示可否、ＴＧ他小売区分の3つの判定で戻り値項目を編集する。
 * 既知の制約:
 * - 建物番号は数値文字列である前提で Long へ変換する。
 */
public final class TransferReceptionCommonEditImpl implements TransferReceptionCommonEdit {

    private final String openingRegistration;

    private final String openingSystemProcessing;

    private final String openingErrorFollowUp;

    private final String routeOtherSystems;

    private final String routeOnlineReportingErrorFollowUp;

    private final String routeTransferSlip;

    private final String transferReportOpeningReport;

    private final String transferReportOpeningErrorFollowUp;

    private final String transferReportOpeningBatchFollow;

    private final String kanjiNameDisplay;

    private final String kanjiNameHidden;

    private final String otherRetail;

    private final String consignmentContract;

    private final String gasRateBusinessMenu;

    private final Map<String, String> transferReportWorkCodesByConduitWorkCode;

    private final Map<String, String> reportingRoutesByConduitWorkCode;

    /**
     * 概要: 必須定数を解決してインスタンスを初期化する。
     *
     * @param ciriusConstantService CIRIUS定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public TransferReceptionCommonEditImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.openingRegistration =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .OPENING_REGISTRATION);
        this.openingSystemProcessing =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .OPENING_SYSTEM_PROCESSING);
        this.openingErrorFollowUp =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .OPENING_ERROR_FOLLOW_UP_CCJG03);
        this.routeOtherSystems =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.OTHER_SYSTEMS);
        this.routeOnlineReportingErrorFollowUp =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute
                                .ONLINE_REPORTING_ERROR_FOLLOW_UP);
        this.routeTransferSlip =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.TRANSFER_SLIP);
        this.transferReportOpeningReport =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.OPENING_REPORT);
        this.transferReportOpeningErrorFollowUp =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.OPENING_ERROR_FOLLOW_UP);
        this.transferReportOpeningBatchFollow =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.TransferReportWorkCode.OPENING_BATCH_FOLLOW);
        this.kanjiNameDisplay =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.TYPE,
                        ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.KANJI_NAME_DISPLAY);
        this.kanjiNameHidden =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.TYPE,
                        ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.KANJI_NAME_HIDDEN);
        this.otherRetail =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
        this.consignmentContract =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.GasRateContractTypeCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.GasRateContractTypeCode.CONSIGNMENT_CONTRACT);
        this.gasRateBusinessMenu =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.UsageSubsystemInformationCode.TYPE,
                        ConstantLookupKeys.Kaiheisen.UsageSubsystemInformationCode.GAS_RATE_BUSINESS_MENU);
        this.transferReportWorkCodesByConduitWorkCode =
                Map.of(
                        openingRegistration, transferReportOpeningReport,
                        openingErrorFollowUp, transferReportOpeningErrorFollowUp,
                        openingSystemProcessing, transferReportOpeningBatchFollow);
        this.reportingRoutesByConduitWorkCode =
                Map.of(
                        openingRegistration, routeOtherSystems,
                        openingErrorFollowUp, routeOnlineReportingErrorFollowUp,
                        openingSystemProcessing, routeTransferSlip);
    }

    /**
     * 概要: 異動受付の共通項目を編集して返却する。
     *
     * @param bldNo 建物番号
     * @param uktkYmd 受付年月日
     * @param uketkTime 受付時刻
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     * @param kiycstKktInf 契約お客さま情報（確定）。氏名・法人種別・漢字表示可否を含む
     * @param hkkEmpNo 報告者番号
     * @param hkkNameKj 報告者名漢字
     * @param kktTksuSvkyNo 確定託送サービス契約番号
     * @param hkkKsenSaymd 報告開栓作業年月日
     * @param tgTkurKbn ＴＧ他小売区分
     * @param idoHkNySbCd 異動報告内容識別コード
     * @param syokaiTsyCd 照会対象コード
     * @return 異動受付共通編集結果
     */
    @Override
    public TransferReceptionCommonEditOutput execute(
            String bldNo,
            LocalDate uktkYmd,
            Integer uketkTime,
            String duknIsrgCd,
            KiycstKktInf kiycstKktInf,
            String hkkEmpNo,
            String hkkNameKj,
            Long kktTksuSvkyNo,
            LocalDate hkkKsenSaymd,
            String tgTkurKbn,
            String idoHkNySbCd,
            String syokaiTsyCd) {
        Long customerNo = Long.valueOf(bldNo);
        LocalDate transferReportDate = uktkYmd;
        Integer transferReportTime = uketkTime;
        String transferReportWorkCode = resolveTransferReportWorkCode(duknIsrgCd);
        String reportingRoute = resolveReportingRoute(duknIsrgCd);
        String customerNameKana1 = kiycstKktInf.ctnmKn1();
        String customerNameKanji1 = kiycstKktInf.ctnmKj1();
        String customerNameCorporateType = kiycstKktInf.ctnm1HujnShbt();
        String customerNameCorporateDisplayType = kiycstKktInf.ctnm1HshiShbt();
        String kanjiDisplayCode = resolveKanjiDisplayCode(kiycstKktInf.ctnmKjhkFlg());
        String businessOfficeCode = null;
        String sectionCode = null;
        String blockCode = null;
        String usageSubsystemInformationCode = gasRateBusinessMenu;
        String terminalNo = null;
        String reporterEmployeeNo = hkkEmpNo;
        String reporterNameKanji = hkkNameKj;
        Long serviceContractNo = kktTksuSvkyNo;
        String networkConstructionCode = null;
        LocalDate openingWorkDate = hkkKsenSaymd;
        String kStKnsnJoho = null;
        String sStKnsnJoho = null;
        LocalDate hkzSsyMryYmd = null;
        LocalDate hkgSsyMryYmd = null;
        String tjhkTrksCd = null;
        String gasRateContractTypeCode = resolveGasRateContractTypeCode(tgTkurKbn);
        String transferReportContentTypeCode = idoHkNySbCd;
        String inquiryTargetCode = syokaiTsyCd;

        IdouktkInsInf output = new IdouktkInsInf(
                customerNo,
                transferReportDate,
                transferReportTime,
                transferReportWorkCode,
                reportingRoute,
                customerNameKana1,
                customerNameKanji1,
                customerNameCorporateType,
                customerNameCorporateDisplayType,
                kanjiDisplayCode,
                businessOfficeCode,
                sectionCode,
                blockCode,
                usageSubsystemInformationCode,
                terminalNo,
                reporterEmployeeNo,
                reporterNameKanji,
                serviceContractNo,
                networkConstructionCode,
                openingWorkDate,
                kStKnsnJoho,
                sStKnsnJoho,
                hkzSsyMryYmd,
                hkgSsyMryYmd,
                tjhkTrksCd,
                gasRateContractTypeCode,
                transferReportContentTypeCode,
                inquiryTargetCode);
        return new TransferReceptionCommonEditOutput(output);
    }

    private String resolveTransferReportWorkCode(String duknIsrgCd) {
        return transferReportWorkCodesByConduitWorkCode.get(duknIsrgCd);
    }

    private String resolveReportingRoute(String duknIsrgCd) {
        return reportingRoutesByConduitWorkCode.get(duknIsrgCd);
    }

    private String resolveKanjiDisplayCode(Boolean ctnmKjhkFlg) {
        return Boolean.TRUE.equals(ctnmKjhkFlg) ? kanjiNameDisplay : kanjiNameHidden;
    }

    private String resolveGasRateContractTypeCode(String tgTkurKbn) {
        return otherRetail.equals(tgTkurKbn) ? consignmentContract : null;
    }
}
