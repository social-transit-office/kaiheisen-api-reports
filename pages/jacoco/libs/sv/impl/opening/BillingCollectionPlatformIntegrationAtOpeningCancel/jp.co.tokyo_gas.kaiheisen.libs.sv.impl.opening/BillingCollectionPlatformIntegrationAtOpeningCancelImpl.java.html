    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            GmtsetdOgaInf gmtsetdOgaInf,
            Long tksuSvkyNoOga,
            Long tksuSvkyNo,
            SvckyaInf svckyaInf,
            SihakyInf sihakyInf,
            Boolean ksenYukuFlg,
            List<SuguGmtInf> suguGmtInfList,
            Boolean ksenFlg,
            Boolean kkytskrKktUmFlg,
            Boolean kkytskrUmFlg,
            Boolean suguKbnUpdFlg) {
        LocalDate previousDay = trkKnsYmd.minusDays(1);
        CollaborationDecision decision = determineCollaborationDecision(
                gasMtstNo,
                tksuSvkyNoOga,
                tksuSvkyNo,
                svckyaInf,
                sihakyInf,
                ksenYukuFlg,
                gmtsetdOgaInf,
                kkytskrKktUmFlg,
                kkytskrUmFlg,
                previousDay);

        if (decision.collaborationCategory() == null) {
            return;
        }

        if (isSv055Category(decision.collaborationCategory())) {
            executeSv055(
                    decision.collaborationCategory(),
                    gasMtstNo,
                    gmtsetdOgaInf,
                    svckyaInf,
                    sihakyInf,
                    ksenFlg,
                    suguGmtInfList,
                    kkytskrKktUmFlg,
                    kkytskrUmFlg,
                    suguKbnUpdFlg,
                    decision.previousContext());
            return;
        }

        executeSv196(gmtsetdOgaInf, svckyaInf, sihakyInf, decision.previousContext());
    }

    /**
     * 概要: 設計書 stepNo2 に従って請求回収基盤連携区分と前契約文脈を判定する。
     * 主な仕様:
     * - 開栓有効フラグが true の場合は供給再開分岐のみを評価し、非該当時は連携対象外とする。
     * - 前契約継続候補のときだけ DA-044-R005 と DA-075-R004 を実行する。
     * - 支払契約不一致時は開栓取消区分とし、SV-196 用の前契約文脈を取得する。
     * 既知の制約:
     * - 前契約文脈の取得では前日基準日のデータが必須で、不足時はデータ不整合例外となる。
     */
    private CollaborationDecision determineCollaborationDecision(
            Long gasMtstNo,
            Long tksuSvkyNoOga,
            Long tksuSvkyNo,
            SvckyaInf svckyaInf,
            SihakyInf sihakyInf,
            Boolean ksenYukuFlg,
            GmtsetdOgaInf gmtsetdOgaInf,
            Boolean kkytskrKktUmFlg,
            Boolean kkytskrUmFlg,
            LocalDate previousDay) {
        if (Boolean.TRUE.equals(ksenYukuFlg)) {
            if (isSupplyResumedBranch(gmtsetdOgaInf, kkytskrKktUmFlg, kkytskrUmFlg)) {
                return new CollaborationDecision(
                        categorySupplyResumed,
                        loadPreviousContext(gasMtstNo, previousDay));
            }
            return new CollaborationDecision(null, null);
        }
        if (!isPreviousContractCandidate(tksuSvkyNoOga, tksuSvkyNo, svckyaInf)) {
            return new CollaborationDecision(categoryCancellationOfOpening, null);
        }

        SvsihkrRecord previousRelation = findPreviousRelation(svckyaInf.svkyNo(), previousDay);
        SihakyRecord previousPaymentContract =
                findPreviousPaymentContract(previousRelation.sihKiyNo(), previousDay);
        PreviousContext previousContext = loadPreviousContext(gasMtstNo, previousDay);
        if (Objects.equals(sihakyInf.sihKiyNo(), previousPaymentContract.sihKiyNo())) {
            return new CollaborationDecision(categoryContinuationOfPreviousContract, previousContext);
        }
        return new CollaborationDecision(categoryCancellationOfOpening, previousContext);
    }

    /**
     * 概要: 供給再開区分に該当するかを判定する。
     * 主な仕様:
     * - 閉栓事由が供給停止以外であれば true を返す。
     * - 供給停止でも供給再開関連フラグのいずれかが true の場合は true を返す。
     * 既知の制約:
     * - `gmtsetdOgaInf` の事由種別は呼出元で正しく設定済みであることを前提とする。
     */
    private boolean isSupplyResumedBranch(
            GmtsetdOgaInf gmtsetdOgaInf,
            Boolean kkytskrKktUmFlg,
            Boolean kkytskrUmFlg) {
        return !closureReasonSupplyOutage.equals(gmtsetdOgaInf.hesnJysb())
                || Boolean.TRUE.equals(kkytskrUmFlg)
                || Boolean.TRUE.equals(kkytskrKktUmFlg);
    }

    /**
     * 概要: 前契約継続判定の前提条件を満たすかを判定する。
     * 主な仕様:
     * - 1世代前託送サービス契約番号と託送サービス契約番号の一致を確認する。
     * - あわせてサービス契約開始事由が前契約継続であることを確認する。
     * 既知の制約:
     * - 開始事由種別の比較は定数値との完全一致で判定する。
     */
    private boolean isPreviousContractCandidate(
            Long tksuSvkyNoOga,
            Long tksuSvkyNo,
            SvckyaInf svckyaInf) {
        return Objects.equals(tksuSvkyNoOga, tksuSvkyNo)
                && isPreviousContractStartReason(svckyaInf);
    }

    /**
     * 概要: サービス契約開始事由が前契約継続かを判定する。
     * 主な仕様:
     * - Cirius 定数 `前契約継続` と入力 DTO の開始事由種別を比較する。
     * 既知の制約:
     * - 入力値が null の場合は false 扱いになる。
     */
    private boolean isPreviousContractStartReason(SvckyaInf svckyaInf) {
        return serviceContractStartReasonContinuation.equals(svckyaInf.svkySttJysb());
    }

    /**
     * 概要: SV-055 呼出し対象の連携区分かを判定する。
     * 主な仕様:
     * - 供給再開または前契約継続の区分のみ true を返す。
     * 既知の制約:
     * - 連携区分はコンストラクタで解決した定数値を前提とする。
     */
    private boolean isSv055Category(String collaborationCategory) {
        return categorySupplyResumed.equals(collaborationCategory)
                || categoryContinuationOfPreviousContract.equals(collaborationCategory);
    }

    /**
     * 概要: SV-055 へ grouped input を組み立てて委譲する。
     * 主な仕様:
     * - 即時開栓情報の先頭要素を `svmstkrInf.mtstNo` に転送する。
     * - 現契約 DTO と前契約 DTO を SV-055 用モデルへ詰め替える。
     * 既知の制約:
     * - `suguGmtInfList` は 1 件以上必須で、空の場合は `IllegalArgumentException` とする。

                        sihakyInf.skno(),
                        sihakyInf.tkysYmd(),

                        svckyaInf.rkkiEndYmd(),

                        previousContext.previousSvckyaRecord().svkyEndJysb(),
                        previousContext.previousSvckyaRecord().rkkiEndYmd(),
                        previousContext.previousSvckyaRecord().szirHtyoYmd()),
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.GmtsetdOgaInf(
                        gmtsetdOgaInf.hesnYmd(),
                        gmtsetdOgaInf.hesnJysb()),

    }

    /**
     * 概要: SV-196 呼出し用の内部変数を構築して委譲する。
     * 主な仕様:
     * - 前契約継続事由の場合は SV-035 を実行する。
     * - 前契約文脈がある場合のみ CCFX08 変更依頼情報の補完項目を設定する。
     * 既知の制約:
     * - ガスメータ設置場所番号は補助資料に合わせた固定値を使用する。
     */
    private void executeSv196(
            GmtsetdOgaInf gmtsetdOgaInf,
            SvckyaInf svckyaInf,
            SihakyInf sihakyInf,
            PreviousContext previousContext) {
        String transferBusinessCode = null;
        String serviceSuspensionCancellationReasonType = null;
        LocalDate serviceSuspensionCancellationDate = null;
        LocalDate contractEndDate = null;
        LocalDate consumptionTaxDeterminationDate = null;

        if (isPreviousContractStartReason(svckyaInf)) {
            transferBusinessCode = billableContractManagementTransfer;
            serviceSuspensionCancellationReasonType =
                    serviceSuspensionCancellationReasonTypeAcquisition.execute(
                                    previousContext == null
                                            ? null
                                            : previousContext.previousSvckyaRecord().svkyEndJysb(),
                                    gmtsetdOgaInf.hesnJysb())
                            .svkjyShbt();
            if (previousContext != null) {
                SvckyaRecord previousSvckyaRecord = previousContext.previousSvckyaRecord();
                serviceSuspensionCancellationDate = previousSvckyaRecord.rkkiEndYmd();
                contractEndDate = previousSvckyaRecord.rkkiEndYmd();
                consumptionTaxDeterminationDate = previousSvckyaRecord.szirHtyoYmd();
            }
        }

        billingCollectionPlatformApiIntegrationForOpeningCancel.execute(
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SkkbRkySgjhInf(
                        null,
                        sihakyInf.sikKrno(),
                        svckyaInf.sqtiKiyNo(),
                        svckyaInf.svkyNo(),
                        svckyaInf.skno(),
                        sihakyInf.sihKiyNo(),
                        sihakyInf.skno(),
                        FIXED_GAS_METER_LOCATION_NO,
                        svckyaInf.rkkiSttYmd(),
                        svckyaInf.rkkiEndYmd()),
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsSihDlirInf(
                        billablePaymentManagementTransfer),
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsKykDlirInf(
                        billableContractManagementTransfer),
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsKykHkirInf(
                        transferBusinessCode,
                        serviceSuspensionCancellationReasonType,
                        serviceSuspensionCancellationDate,
                        contractEndDate,
                        consumptionTaxDeterminationDate));
    }

    /**
     * 概要: SV-055/SV-196 で利用する前契約文脈を取得する。
     * 主な仕様:
     * - DA-041-R010 相当のメータ契約取得後、DA-038-R012 相当の前契約サービス契約を取得する。
     * - いずれかが 0 件の場合は DA 識別子付きのデータ不整合例外を送出する。
     * 既知の制約:
     * - 基準日は常に引数の前日で評価される。
     */
    private PreviousContext loadPreviousContext(Long gasMtstNo, LocalDate previousDay) {
        SvmstkrRecord previousSvmstkrRecord = svmstkrDataAccess
                .findByMtstNoAndRkkiSttYmdAndRkkiKnrnJtkb(
                        gasMtstNo,
                        previousDay,
                        chargeCalculationRelatedStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svmstkrDataAccess.tableName(),
                        SvmstkrDataAccess.CallSpecs
                                .FIND_BY_MTST_NO_AND_RKKI_STT_YMD_AND_RKKI_KNRN_JTKB.identifier(),
                        new FlatJsonBuilder()
                                .put("mtstNo", gasMtstNo)
                                .put("kkiSttYmd", previousDay)
                                .put("rkkiKnrnJtkb", chargeCalculationRelatedStatusInvalid)));
        SvckyaRecord previousSvckyaRecord = svckyaDataAccess
                .findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNotOrderByRkkiSttYmd(
                        previousSvmstkrRecord.svkyNo(),
                        previousDay,
                        serviceContractStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svckyaDataAccess.tableName(),
                        SvckyaDataAccess.CallSpecs
                                .FIND_LATEST_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_NOT_ORDER_BY_RKKI_STT_YMD
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("svkyNo", previousSvmstkrRecord.svkyNo())
                                .put("knskKjnYmd", previousDay)
                                .put("svkyJtkb", serviceContractStatusInvalid)));
        return new PreviousContext(previousSvmstkrRecord, previousSvckyaRecord);
    }

    /**
     * 概要: 前契約のサービス・支払契約関連を取得する。
     * 主な仕様:
     * - DA-044-R005 を実行し、対象日時点で有効な支払契約紐付を 1 件取得する。
     * - 0 件時は検索条件を含むデータ不整合例外を送出する。
     * 既知の制約:
     * - 無効ステータス除外はコンストラクタで解決した定数値に依存する。
     */
    private SvsihkrRecord findPreviousRelation(Long svkyNo, LocalDate previousDay) {
        return svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndNotSihKiyJtkb(
                        svkyNo,
                        previousDay,

    /**
     * 概要: 前契約側の支払契約情報を取得する。
     * 主な仕様:
     * - DA-075-R004 を実行し、対象日時点で有効な支払契約を 1 件取得する。

    private SihakyRecord findPreviousPaymentContract(Long sihKiyNo, LocalDate previousDay) {

                        SihakyDataAccess.CallSpecs

                        new FlatJsonBuilder()
                                .put("sihKiyNo", sihKiyNo)
                                .put("tkysYmd", previousDay)
                                .put("sihKiyJtkb", paymentContractStatusInvalid)));
    }

    /**
     * 概要: 開閉栓定数を必須取得する。
     * 主な仕様:
     * - `type` と `value` で名称解決できることを検証し、戻り値には `value` 自体を返す。
     * - 未定義時は `ConstantDefinitionNotFoundException` を送出する。
     * 既知の制約:
     * - 名称自体は使用せず、存在確認のためだけに検索する。
     */
    private static String requireKaiheisenConstant(
            KaiheisenConstantService kaiheisenConstantService,
            String type,
            String value) {
        String name = kaiheisenConstantService.findNameByTypeAndValue(type, value);
        if (name == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + type + ", value=" + value);
        }
        return value;
    }
