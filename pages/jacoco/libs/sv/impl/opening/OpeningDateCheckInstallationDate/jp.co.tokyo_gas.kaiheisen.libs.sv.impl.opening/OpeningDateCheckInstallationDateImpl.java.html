package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckInstallationDate;

/** 開栓日チェック（取付日判定）サービス実装。 */
public class OpeningDateCheckInstallationDateImpl implements OpeningDateCheckInstallationDate {

    /**
     * 開栓日と取付日を比較し、取付日が開栓日より後の場合は開栓共通エラーを送出する。
     *
     * @param kasgYmd 開栓日
     * @param tusoGmtkYmd 取付日
     * @throws KaisenRuleException 取付日が開栓日より後の場合
     */
    @Override
    public void execute(
            LocalDate kasgYmd, LocalDate tusoGmtkYmd) {
        if (tusoGmtkYmd.isAfter(kasgYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }
    }
}
