package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDefinitionNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkRecord;

/**
 * 概要: SV-232（開栓時確定前ガスメーター検針結果登録）のサービス実装。
 * 主な仕様:
 * - `DA-052-R002` で既存の確定前ガスメーター検針結果を取得できた場合は、その最新 1 件を登録内容へ採用する。
 * - `DA-052-R002` で取得できない場合は `DA-027-R001` で導管検針結果を取得し、取得できた場合のみその最新 1 件を登録内容へ採用する。
 * - 開栓事由種別から検針結果作成事由種別を決定し、`DA-052-C001` へ 1 件登録する。
 * - 監査ユーザー ID は `KHSS232`、最終更新起点機能 ID は RequestContext の報告システム種別を開閉栓定数でデコードした値を利用する。
 * 既知の制約:
 * - RequestContext の `systemDatetime` と `hukkSysShbt` が未設定の場合は早期失敗する。
 * - 固定設定値はコンストラクタ初期化時に必須 CIRIUS 定数として検証する。
 */
public class PreConfirmationGasMeterReadingResultRegistrationAtOpeningImpl
    implements PreConfirmationGasMeterReadingResultRegistrationAtOpening {

    static final String SERVICE_USER_ID = "KHSS232";

    private static final Set<String> SUPPLY_RESUMED_REASON_TYPES = Set.of(
            ConstantLookupKeys.Cirius.OpeningReasonType.SUPPLY_RESUMED,
            ConstantLookupKeys.Cirius.OpeningReasonType.START_SUPPLYING_TO_OTHER_PARTIES);

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String openingReasonCode;

    private final String supplyResumedReasonCode;

    private final String disasterRestorationReasonCode;

    private final String securityReleaseReasonCode;

    private final String noRegressionCode;

    private final String noChangeCode;

    private final String onlineReportingMethodCode;

    private final String irregularMeterReadingCode;

    private final String actualInspectionStatusCode;

    private final String openingStatusCode;

    /**
     * 概要: SV-232 実行に必要な依存オブジェクトと必須定数を初期化する。
     *
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果データアクセス
     * @param gmkskkdDataAccess ガスメーター検針結果＜導管＞データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PreConfirmationGasMeterReadingResultRegistrationAtOpeningImpl(
                KtmgmkkDataAccess ktmgmkkDataAccess,
                GmkskkdDataAccess gmkskkdDataAccess,
                RequestContext requestContext,
                KaiheisenConstantService kaiheisenConstantService,
                CiriusConstantService ciriusConstantService) {
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.openingReasonCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.OPENING);
        this.supplyResumedReasonCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_RESUMED);
        this.disasterRestorationReasonCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                        .RESTORATION_OF_SHUTOFF_IN_CASE_OF_DISASTER);
        this.securityReleaseReasonCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SECURITY_RELEASE);
        this.noRegressionCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.NONE);
        this.noChangeCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChangeDetection.TYPE,
                ConstantLookupKeys.Cirius.ChangeDetection.NO_CHANGE);
        this.onlineReportingMethodCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MeterReadingResultReportingMethod.TYPE,
                ConstantLookupKeys.Cirius.MeterReadingResultReportingMethod.ONLINE);
        this.irregularMeterReadingCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RegularMeterReading.TYPE,
                ConstantLookupKeys.Cirius.RegularMeterReading.IRREGULAR);
        this.actualInspectionStatusCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MeterReadingImplementationStatusCode.TYPE,
                ConstantLookupKeys.Cirius.MeterReadingImplementationStatusCode
                        .ACTUAL_INSPECTION_NORMAL_METER_READING);
        this.openingStatusCode = requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 概要: SV-232 を実行し、確定前ガスメーター検針結果を 1 件登録する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param hkkKsenSaymd 報告開栓作業年月日
     * @param kktGmtShjSu 確定ガスメーター指示数
     * @param knsHuhuKbn 検針方法区分
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param mtgMtstNo 元ガスメーター設置場所番号
     * @param gmb ガスメーター社番
     * @param gmbTsno ガスメーター社番通し番号
     * @param kktTksuSvkyNo 確定託送サービス契約番号
     * @param sgytSinNo 作業担当社員番号
     * @param kasJysb 開栓事由種別
     */
    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate hkkKsenSaymd,
            Long kktGmtShjSu,
            String knsHuhuKbn,
            String ckrtGmscJtkb,
            Long mtgMtstNo,
            String gmb,
            Integer gmbTsno,
            Long kktTksuSvkyNo,
            String sgytSinNo,
            String kasJysb) {
        RequestState requestState = requireRequestState();
        int maxSequenceNumber = resolveMaxSequenceNumber(gasMtstNo, hkkKsenSaymd);
        String reasonTypeForCreate = resolveReasonTypeForCreate(kasJysb);
        ktmgmkkDataAccess.insert(
                new KtmgmkkWriteRecord(
                        gasMtstNo,
                        hkkKsenSaymd,
                        maxSequenceNumber + 1,
                        null,
                        reasonTypeForCreate,
                        null,
                        kktGmtShjSu,
                        null,
                        null,
                        null,
                        noRegressionCode,
                        noChangeCode,
                        onlineReportingMethodCode,
                        false,
                        irregularMeterReadingCode,
                        actualInspectionStatusCode,
                        openingStatusCode,
                        knsHuhuKbn,
                        ckrtGmscJtkb,
                        mtgMtstNo,
                        null,
                        null,
                        gmb,
                        gmbTsno,
                        kktTksuSvkyNo,
                        sgytSinNo,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        requestState.systemDate(),
                        false,
                        null),
                AuditContext.of(SERVICE_USER_ID, requestState.originFunctionId()));
    }

    /**
     * 概要: 既存データから最大検針結果通し番号を決定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @return 最大検針結果通し番号
     */
    private int resolveMaxSequenceNumber(Long gasMtstNo, LocalDate knsYmd) {
        var preConfirmationResult = ktmgmkkDataAccess.findByGasMtstNoAndKnsYmd(gasMtstNo, knsYmd);
        if (preConfirmationResult.isPresent()) {
            return preConfirmationResult.map(KtmgmkkRecord::knsKkaTsno).orElse(0);
        }
        return gmkskkdDataAccess.findByGasMtstNoAndKnsYmd(gasMtstNo, knsYmd)
                .map(GmkskkdRecord::knsKkaTsno)
                .orElse(0);
    }

    /**
     * 概要: 開栓事由種別から検針結果作成事由種別を決定する。
     *
     * @param kasJysb 開栓事由種別
     * @return 検針結果作成事由種別
     */
    private String resolveReasonTypeForCreate(String kasJysb) {
        if (SUPPLY_RESUMED_REASON_TYPES.contains(kasJysb)) {
            return supplyResumedReasonCode;
        }
        return switch (kasJysb) {
            case ConstantLookupKeys.Cirius.OpeningReasonType.SECURITY_RESTORATION -> securityReleaseReasonCode;
            case ConstantLookupKeys.Cirius.OpeningReasonType.DISASTER_RECOVERY -> disasterRestorationReasonCode;
            default -> openingReasonCode;
        };
    }

    /**
     * 概要: 実行に必要な RequestContext 値を取得し、起点機能 ID を解決する。
     *
     * @return 実行時に利用するリクエスト状態
     */
    private RequestState requireRequestState() {
        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (originFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return new RequestState(systemDate, originFunctionId);
    }

    /**
     * 概要: 必須 CIRIUS 定数を検証済みコードとして返却する。
     *
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param type 定数種別
     * @param code 定数コード
     * @return 検証済みコード
     */
    private static String requireCiriusConstant(
            CiriusConstantService ciriusConstantService,
            String type,
            String code) {
        return RequiredConstantResolver.requireCiriusConstant(
            ciriusConstantService,
            type,
            code);
    }

    /** 実行に必要な RequestContext 由来値。 */
    private record RequestState(
            LocalDate systemDate,
            String originFunctionId) {
    }
}
