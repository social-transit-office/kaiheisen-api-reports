    /**
     * 前契約継続の開栓時契約情報更新を実行する。
     */
    @Override
    public ContractInfoUpdatePreviousContractContinuationAtOpeningOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            LocalDate kktTkysYmd,
            Long kktSvkyNo,
            Long kktSihKiyNo,
            Long kktSikKrno,
            Long kktSyoKyoTrno,
            Long kktTksuSvkyNo,
            String kktSvckyaGasYtCd,
            PreviousContractContinuationKiyInf kiyInf,
            String ckrtGmscJtkb,
            Integer snyMtrKnsu,
                        List<SnyInf> snyInfList,
            Boolean syoriHnt2XFlg) {
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId());

        sv227.execute(
                gasMtstNo,
                kktTksuSvkyNo,
                kktTkysYmd,
                kktSyoKyoTrno,
                kktSvkyNo,
                ckrtGmscJtkb,
                snyInfList);

        SvckyaKktInf svckyaKktInf = registerServiceContract(
                kktSvkyNo,
                kasgYmd,
                kktTkysYmd,
                kktSyoKyoTrno,
                kktSvckyaGasYtCd,
                kiyInf.svckyaInf(),
                auditContext);
        registerServiceContractRelation(gasMtstNo, kktSvkyNo, kktTkysYmd, kiyInf.svmstkrInf(), auditContext);

        if (isDutyFree(kiyInf.svckyaInf())) {
            sv071.execute(
                    DUTY_FREE_CREATE,
                    kktSvkyNo,
                    kktTkysYmd,
                    null,
                    buildDutyFreeContract(kiyInf.mnzkiyInf()));
        }

        if (!Boolean.TRUE.equals(syoriHnt2XFlg)) {
            SihakyKktInf sihakyKktInf = registerNewPaymentContract(kktSihKiyNo, kktSikKrno, kktSyoKyoTrno, kktTkysYmd, auditContext);
            SvsihkrKktInf svsihkrKktInf = registerNewServicePaymentRelation(kktSihKiyNo, kktSvkyNo, kktTkysYmd, auditContext);
            registerInitialPaymentSlip(sihakyKktInf, auditContext);
            registerDunning(kktSihKiyNo, auditContext);
            return new ContractInfoUpdatePreviousContractContinuationAtOpeningOutput(
                    svckyaKktInf,
                    svsihkrKktInf,
                    sihakyKktInf);
        }

        SvsihkrKktInf svsihkrKktInf = updateOrInsertServicePaymentRelation(
                kasgYmd,
                kktTkysYmd,
                kktSihKiyNo,
                kktSvkyNo,
                kiyInf.svsihkrInf(),
                auditContext);
        SihakyKktInf sihakyKktInf = updateOrInsertPaymentContract(
                kasgYmd,
                kktTkysYmd,
                kktSihKiyNo,
                kiyInf.sihakyInf(),
                auditContext);

        return new ContractInfoUpdatePreviousContractContinuationAtOpeningOutput(
                svckyaKktInf,
                svsihkrKktInf,
                sihakyKktInf);
    }

    private SvckyaKktInf registerServiceContract(
            Long kktSvkyNo,
            LocalDate kasgYmd,
            LocalDate kktTkysYmd,
            Long kktSyoKyoTrno,
            String kktSvckyaGasYtCd,
            PreviousContractContinuationSvckyaInf input,
            AuditContext auditContext) {
        Integer currentSkno = svckyaDataAccess.findLatestBySvkyNoAndSvkyJtkbNot(kktSvkyNo)
                .map(SvckyaRecord::skno)
                .orElseThrow(() -> createDataInconsistency(
                        svckyaDataAccess.tableName(),
                        SvckyaDataAccess.CallSpecs.FIND_LATEST_BY_SVKY_NO_AND_SVKY_JTKB_NOT.identifier(),
                        new FlatJsonBuilder().put("svkyNo", kktSvkyNo)));
        SvckyaKktInf output = buildServiceContractOutput(
                currentSkno,
                kktSvkyNo,
                kasgYmd,
                kktTkysYmd,
                kktSyoKyoTrno,
                kktSvckyaGasYtCd,
                input);
        svckyaDataAccess.insert(toSvckyaWriteRecord(output), auditContext);
        return output;
    }

        return new SvckyaKktInf(

                null,
                kktTkysYmd,

                MAX_DATE,
                kktTkysYmd,
                MAX_DATE,
                null,
                input.svcRyksYmd(),
                input.svkyTiktYmd(),
                kktTkysYmd,
                kasgYmd,
                input.sqtiKiyNo(),
                kktSyoKyoTrno,

                input.byssUmFlg(),

                kktSvckyaGasYtCd,

                input.rknKhumFlg(),
                null,
                input.kiyKnrkCd(),

                input.tsziKysbNo());
    }

    private void registerServiceContractRelation(
            Long gasMtstNo,
            Long kktSvkyNo,
            LocalDate kktTkysYmd,
            PreviousContractContinuationSvmstkrInf input,
            AuditContext auditContext) {
        Integer currentSkno = svmstkrDataAccess.findBySvkyNoAndMtstNo(kktSvkyNo, gasMtstNo)
                .map(SvmstkrRecord::skno)
                .orElseThrow(() -> createDataInconsistency(
                        svmstkrDataAccess.tableName(),
                        SvmstkrDataAccess.CallSpecs.FIND_BY_SVKY_NO_AND_MTST_NO.identifier(),
                        new FlatJsonBuilder().put("svkyNo", kktSvkyNo).put("mtstNo", gasMtstNo)));
        svmstkrDataAccess.insert(
                buildSvmstkrWriteRecord(gasMtstNo, kktSvkyNo, currentSkno + 1, kktTkysYmd, input),
                auditContext);
    }

                skno,
                "0",
                input.sknrFlg(),

                kktTkysYmd,
                MAX_DATE,
                null);
    }

    private boolean isDutyFree(PreviousContractContinuationSvckyaInf svckyaInf) {
        return "3".equals(svckyaInf.sziKziKbn());
    }

    private MnzkiyInf buildDutyFreeContract(MnzkiyInf input) {
        return new MnzkiyInf(

                null,
                input == null ? null : input.sseTiskNo(),

                input == null ? null : input.mnziTysjYmd(),
                input == null ? null : input.mnziHksjYmd(),
                input == null ? null : input.mzkhHjonKj(),
                input == null ? null : input.mzkhHjonKn());
    }

    private SvsihkrKktInf updateOrInsertServicePaymentRelation(
            LocalDate kasgYmd,
            LocalDate kktTkysYmd,
            Long kktSihKiyNo,
            Long kktSvkyNo,
            PreviousContractContinuationSvsihkrInf input,
            AuditContext auditContext) {
        if (isWithinContinuationWindow(kasgYmd, input.tkysYmd(), input.tyenYmd())) {
            int updated = svsihkrDataAccess.updateBySihKiyNoAndSvkyNoAndSknoAndRecUpdJnjNo(
                    new SvsihkrWriteRecord(
                            input.sihKiyNo(),
                            input.svkyNo(),
                            input.skno(),
                            input.tkysYmd(),
                            MAX_DATE,
                            ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING,
                            input.snskSvkyNo(),
                            input.ikskSttYm(),
                            input.ikskEndYm(),
                            input.ikskGrpNo(),
                            input.tdssSbtShbt(),
                            input.gsnKhFlg(),
                            input.gsptShbt(),
                            input.sikTmngShbt(),
                            input.shziKbn()),
                    Set.of(SvsihkrColumn.TYEN_YMD, SvsihkrColumn.SIH_KIY_JTKB),
                    new SvsihkrSknoJnjUpdateKey(input.sihKiyNo(), input.svkyNo(), input.skno(), input.recUpdJnjNo()),
                    auditContext);
            if (updated == 0) {
                throw createDataInconsistency(
                        svsihkrDataAccess.tableName(),
                        SvsihkrDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                        new FlatJsonBuilder()
                                .put("sihKiyNo", input.sihKiyNo())
                                .put("svkyNo", input.svkyNo())
                                .put("skno", input.skno())
                                .put("recUpdJnjNo", input.recUpdJnjNo()));
            }
            return new SvsihkrKktInf(
                    input.sihKiyNo(),
                    input.svkyNo(),
                    input.skno(),
                    input.tkysYmd(),
                    MAX_DATE,
                    ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING,
                    input.snskSvkyNo(),
                    input.ikskSttYm(),
                    input.ikskEndYm(),
                    input.ikskGrpNo(),
                    input.tdssSbtShbt(),
                    input.gsnKhFlg(),
                    input.gsptShbt(),
                    input.sikTmngShbt(),
                    input.shziKbn());
        }

    private boolean isWithinContinuationWindow(LocalDate baseDate, LocalDate startDate, LocalDate endDate) {
        boolean withinRange = !baseDate.isBefore(startDate) && !baseDate.isAfter(endDate) && !MAX_DATE.equals(endDate);
        boolean beforeOrEqualStart = !baseDate.isAfter(startDate);
        return withinRange || beforeOrEqualStart;
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        if (hukkSysShbt.isBlank()) {
            throw new IllegalStateException("RequestContext.hukkSysShbt is required");
        }
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null || decodedOriginFunctionId.isBlank()) {
            throw new IllegalStateException("Origin function id constant is required");
        }
        return decodedOriginFunctionId;
    }
