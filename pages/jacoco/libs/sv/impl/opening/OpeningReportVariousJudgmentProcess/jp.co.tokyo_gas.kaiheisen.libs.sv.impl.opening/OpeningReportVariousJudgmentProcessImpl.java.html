/**
 * 概要: SV-165（開栓報告各種判定処理）の実装クラス.
 * 主な仕様:
 * - SV-171 で契約有無を判定し、継続使用時のみ SV-172 を実行する。
 * - SV-173 と、継続使用・供給再開・保安復旧・災害復旧時に 2 回の SV-174 を実行し、下位 SV の戻り値を合成して返却する。
 * - RequestContext の必須値は下位 SV 呼出し前に fail-fast で検査する。
 * 既知の制約:
 * - output_patterns.condition の文言は不整合があるが、Phase B 合意に従い継続使用時のみ SV-172 を実行する。
 */
public final class OpeningReportVariousJudgmentProcessImpl
        implements OpeningReportVariousJudgmentProcess {

        /** 開栓時契約有無判定サービス. */
    private final ContractExistenceJudgmentAtOpening
            contractExistenceJudgmentAtOpening;

        /** 開栓時継続使用契約継続判定サービス. */
    private final ContinuedUseContractContinuationJudgmentAtOpening
            continuedUseContractContinuationJudgmentAtOpening;

        /** 開栓時代替判定サービス. */
    private final AlternativeJudgmentAtOpening alternativeJudgmentAtOpening;

        /** 開栓時チェック用おなまえ編集サービス. */
    private final CustomerNameEditForOpeningCheck
            customerNameEditForOpeningCheck;

        /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 継続使用の定数値. */
    private final String continuedUseReason;

    /** 供給再開の定数値. */
    private final String supplyResumedReason;

    /** 保安復旧の定数値. */
    private final String securityRestorationReason;

    /** 災害復旧の定数値. */
    private final String disasterRecoveryReason;

    /**
     * 概要: 必須依存と開栓事由種別判定用定数を初期化する.
     * 主な仕様:
     * - 下位 SV と RequestContext を fail-fast で保持する。
     * - 報告開栓事由種別 04900 の継続使用・供給再開・保安復旧・災害復旧コードをコンストラクタで解決する。
     *
     * @param contractExistenceJudgmentService 開栓時契約有無判定サービス
     * @param continuedUseJudgmentService 開栓時継続使用契約継続判定サービス
     * @param alternativeJudgmentService 開栓時代替判定サービス
     * @param customerNameEditService 開栓時チェック用おなまえ編集サービス
     * @param currentRequestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public OpeningReportVariousJudgmentProcessImpl(
            final ContractExistenceJudgmentAtOpening
                    contractExistenceJudgmentService,
            final ContinuedUseContractContinuationJudgmentAtOpening
                    continuedUseJudgmentService,
            final AlternativeJudgmentAtOpening alternativeJudgmentService,
            final CustomerNameEditForOpeningCheck customerNameEditService,
            final RequestContext currentRequestContext,
            final CiriusConstantService ciriusConstantService) {
        this.contractExistenceJudgmentAtOpening = Objects.requireNonNull(
                contractExistenceJudgmentService,
                "contractExistenceJudgmentAtOpening");
        this.continuedUseContractContinuationJudgmentAtOpening =
                Objects.requireNonNull(
                        continuedUseJudgmentService,
                        "continuedUseContractContinuationJudgmentAtOpening");
        this.alternativeJudgmentAtOpening = Objects.requireNonNull(
                alternativeJudgmentService,
                "alternativeJudgmentAtOpening");
        this.customerNameEditForOpeningCheck = Objects.requireNonNull(
                customerNameEditService,
                "customerNameEditForOpeningCheck");
        this.requestContext = Objects.requireNonNull(
                currentRequestContext,
                "requestContext");
        CiriusConstantService resolvedService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.continuedUseReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .CONTINUED_USE);
        this.supplyResumedReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .SUPPLY_RESUMED);
        this.securityRestorationReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .SECURITY_RESTORATION);
        this.disasterRecoveryReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .DISASTER_RECOVERY);
    }

    /**
     * 概要: 開栓報告各種判定処理を実行する.
     * 主な仕様:
     * - RequestContext の必須値を先に検証し、以降の子SV呼出し条件を確定する。
     * - 継続使用以外では SV-172 をスキップし、該当フラグは false で返却する。
     * - SV-174 は継続使用・供給再開・保安復旧・災害復旧時に、閉栓時と開栓時で 2 回呼び出す。
     * 既知の制約:
     * - public 引数の null 妥当性は上位層責務とし、必要な箇所での自然な fail-fast に委ねる。
     */
    @Override
    public OpeningReportVariousJudgmentProcessOutput
            execute(
            final String kasJysb,
            final String tgTkurKbn,
            final LocalDate kasgYmd,
            final String ctnmKn1Ksen,
            final String ctnm1HujnShbtKsen,
            final String ctnm1HshiShbtKsen,
            final TskgmstInf tskgmstInf,
            final TkssvkyInf tkssvkyInf,
            final OpeningReportJudgmentSvmstkrInf svmstkrInf,
            final OpeningReportJudgmentSvckyaInf svckyaInf,
            final GmtsetdInf gmtsetdInf,
            final OpeningReportJudgmentSihakyInf sihakyInf,
            final KiycstInf kiycstInf,
            final Boolean ksenYmdTskgmstUmFlg,
            final Boolean ksenYmdSvmstkrUmFlg,
            final Boolean snyTkmtFlg,
            final Boolean sknrUmFlg,
            final String kzksyAtkCd) {
        validateRequestContext();
        boolean continuedUse = isContinuedUse(kasJysb);
        boolean requiresCustomerNameCheck = requiresCustomerNameCheck(kasJysb);

        ContractExistenceJudgmentAtOpeningOutput contractExistenceOutput =
                contractExistenceJudgmentAtOpening.execute(
                        ksenYmdTskgmstUmFlg,
                        tskgmstInf,
                        ksenYmdSvmstkrUmFlg,
                        toContractExistenceSvmstkrInf(svmstkrInf),
                        svckyaInf.svkyJtkb());

        Boolean contractExists = contractExistenceOutput.kiyUmFlg();
        ContinuedUseContractContinuationJudgmentAtOpeningOutput
                continuedUseOutput = null;
        if (continuedUse) {
            continuedUseOutput =
                    continuedUseContractContinuationJudgmentAtOpening
                            .execute(
                                    tgTkurKbn,
                                    kasgYmd,
                                    contractExists,
                                    kzksyAtkCd,
                                    tkssvkyInf,
                                    new ContinuedUseContractSvckyaInf(
                                            svckyaInf.kiyShbt()),
                                    gmtsetdInf,
                                    new ContinuedUseContractSihakyInf(
                                            sihakyInf.sikSvcShbt()));
        }

        Boolean judgment4xFlag = Boolean.TRUE.equals(extractJudgment4xFlag(continuedUseOutput));
        AlternativeJudgmentAtOpeningOutput alternativeOutput =
                alternativeJudgmentAtOpening.execute(
                        kasJysb,
                        snyTkmtFlg,
                        judgment4xFlag,
                        sknrUmFlg);
        CustomerNameEditForOpeningCheckOutput closingCustomerNameOutput =
                requiresCustomerNameCheck
                        ? executeCustomerNameEditIfPresent(
                                kiycstInf.ctnmKn1(),
                                kiycstInf.ctnm1HujnShbt(),
                                kiycstInf.ctnm1HshiShbt())
                        : new CustomerNameEditForOpeningCheckOutput(null);
        CustomerNameEditForOpeningCheckOutput openingCustomerNameOutput =
                requiresCustomerNameCheck
                        ? executeCustomerNameEditIfPresent(
                                ctnmKn1Ksen,
                                ctnm1HujnShbtKsen,
                                ctnm1HshiShbtKsen)
                        : new CustomerNameEditForOpeningCheckOutput(null);

        return new OpeningReportVariousJudgmentProcessOutput(
                contractExists,
                Boolean.TRUE.equals(judgment4xFlag),
                Boolean.TRUE.equals(extractJudgment3xFlag(continuedUseOutput)),
                Boolean.TRUE.equals(extractJudgment2xFlag(continuedUseOutput)),
                Boolean.TRUE.equals(extractRevivalFlag(continuedUseOutput)),
                extractBillingServiceCategory(continuedUseOutput),
                alternativeOutput.digwrHntFlg(),
                alternativeOutput.sknrUmFlg(),
                closingCustomerNameOutput.chkCtnm1Kn(),
                openingCustomerNameOutput.chkCtnm1Kn());
    }

    private CustomerNameEditForOpeningCheckOutput executeCustomerNameEditIfPresent(
            String ctnmKn1,
            String ctnm1HujnShbt,
            String ctnm1HshiShbt) {
        if (isBlank(ctnmKn1) && isBlank(ctnm1HujnShbt) && isBlank(ctnm1HshiShbt)) {
            return new CustomerNameEditForOpeningCheckOutput(null);
        }
        return customerNameEditForOpeningCheck.execute(
                ctnmKn1,
                ctnm1HujnShbt,
                ctnm1HshiShbt);
    }

    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }

    private Boolean extractJudgment4xFlag(
            final ContinuedUseContractContinuationJudgmentAtOpeningOutput
                    continuedUseOutput) {
        return continuedUseOutput == null
                ? null
                : continuedUseOutput.syoriHnt4XFlg();
    }

    private Boolean extractJudgment3xFlag(
            final ContinuedUseContractContinuationJudgmentAtOpeningOutput
                    continuedUseOutput) {
        return continuedUseOutput == null
                ? null
                : continuedUseOutput.syoriHnt3XFlg();
    }

    private Boolean extractJudgment2xFlag(
            final ContinuedUseContractContinuationJudgmentAtOpeningOutput
                    continuedUseOutput) {
        return continuedUseOutput == null
                ? null
                : continuedUseOutput.syoriHnt2XFlg();
    }

    private Boolean extractRevivalFlag(
            final ContinuedUseContractContinuationJudgmentAtOpeningOutput
                    continuedUseOutput) {
        return continuedUseOutput == null
                ? null
                : continuedUseOutput.syoriHntFkktFlg();
    }

    private String extractBillingServiceCategory(
            final ContinuedUseContractContinuationJudgmentAtOpeningOutput
                    continuedUseOutput) {
        return continuedUseOutput == null
                ? null
                : continuedUseOutput.sqsvsHnsb();
    }

    private boolean isContinuedUse(final String kasJysb) {
        return kasJysb.equals(continuedUseReason);
    }

    private boolean requiresCustomerNameCheck(final String kasJysb) {
        return isContinuedUse(kasJysb)
                || kasJysb.equals(supplyResumedReason)
                || kasJysb.equals(securityRestorationReason)
                || kasJysb.equals(disasterRecoveryReason);
    }

    private SvmstkrInf toContractExistenceSvmstkrInf(
            final OpeningReportJudgmentSvmstkrInf svmstkrInf) {
        return svmstkrInf == null
                ? null
                : new SvmstkrInf(svmstkrInf.rkkiKnrnJtkb());
    }

    private void validateRequestContext() {
        requireCorrelationIdentifier();
        requireReportingSystemType();
        requireRequestSystemDatetime();
    }

    private String requireCorrelationIdentifier() {
        return requestContext.getCorrelationId().orElseThrow(
                () -> new IllegalStateException(
                        "RequestContext.correlationId is required"));
    }

    private String requireReportingSystemType() {
        return requestContext.getHukkSysShbt().orElseThrow(
                () -> new IllegalStateException(
                        "RequestContext.hukkSysShbt is required"));
    }

    private LocalDateTime requireRequestSystemDatetime() {
        return requestContext.getSystemDatetime().orElseThrow(
                () -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
}
