package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.SameDayOpeningDateCheckMultipleOpening;

/** 同日開栓日チェック（同日複数開栓）サービス実装。 */
public class SameDayOpeningDateCheckMultipleOpeningImpl
        implements SameDayOpeningDateCheckMultipleOpening {

    /**
     * 開栓年月日と開栓作業年月日が同日の場合は同日開栓エラーを送出する。
     *
     * @param kasYmd 開栓年月日
     * @param kasgYmd 開栓作業年月日
     * @throws KaisenRuleException 開栓年月日と開栓作業年月日が同日の場合
     */
    @Override
    public void execute(
            LocalDate kasYmd, LocalDate kasgYmd) {
        if (kasgYmd.equals(kasYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100047);
        }
    }
}
