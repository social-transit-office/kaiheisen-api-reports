package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckSeriesInstallationStatus;

/**
 * 概要: SV-178（開栓日チェック（直列設置状態判定））の実装クラス。
 * 主な仕様:
 * - システム日時点と開栓日時点の直列ガスメーター設置状態区分を比較する。
 * - 2 つの区分が不一致の場合は開栓業務エラー 020100018 を送出する。
 * - 一致する場合は何も返さずに正常終了する。
 * 既知の制約:
 * - 入力値の trim や正規化は行わず、呼び出し元から渡された値をそのまま比較する。
 * - RequestContext や下位依存呼び出しは使用しない。
 */
public class OpeningDateCheckSeriesInstallationStatusImpl
        implements OpeningDateCheckSeriesInstallationStatus {

    /**
     * システム日時点と開栓日時点の直列設置状態区分を比較して SV-178 を実行する。
     *
     * @param systemDateCkrtGmscJtkb システム日時点直列ガスメーター設置状態区分
     * @param ksenYmdCkrtGmscJtkb 開栓日時点直列ガスメーター設置状態区分
     * @throws KaisenRuleException 2 つの設置状態区分が不一致の場合
     */
    @Override
    public void execute(
            String systemDateCkrtGmscJtkb,
            String ksenYmdCkrtGmscJtkb) {
        if (!Objects.equals(systemDateCkrtGmscJtkb, ksenYmdCkrtGmscJtkb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }
    }
}
