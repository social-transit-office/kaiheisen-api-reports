package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalTime;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.idouktk.IdouktkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.idouktk.write.IdouktkWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferReceptionRegistration;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TransferReceptionRegistrationIdouktkInsInf;

/**
 * SV-240（異動受付登録）の実装。
 */
public final class TransferReceptionRegistrationImpl implements TransferReceptionRegistration {

    static final String SERVICE_USER_ID = "KHSS240";

    private static final DateTimeFormatter TIME_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    private final IdouktkDataAccess idouktkDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * コンストラクタ。
     *
     * @param idouktkDataAccess 異動受付 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public TransferReceptionRegistrationImpl(
            IdouktkDataAccess idouktkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.idouktkDataAccess = Objects.requireNonNull(idouktkDataAccess, "idouktkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
    }

    @Override
    public void execute(TransferReceptionRegistrationIdouktkInsInf idouktkInsInf) {
        Objects.requireNonNull(idouktkInsInf, "idouktkInsInf");

        requireCorrelationId();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        requireSystemDatetime();

        LocalTime reportTime = toReportTime(idouktkInsInf.idoHkokuJkk());
        Integer nextSequenceNo = idouktkDataAccess
                .findByOkyakuNoAndIdohHkkYmdAndIdoHkokuJkk(
                        idouktkInsInf.cstNo(),
                        idouktkInsInf.ihkYmd(),
                        reportTime)
                .map(sequenceNo -> sequenceNo + 1)
                .orElse(1);

        IdouktkWriteRecord record = new IdouktkWriteRecord(
                idouktkInsInf.cstNo(),
                idouktkInsInf.ihkYmd(),
                reportTime,
                nextSequenceNo,
                idouktkInsInf.idoHkkGymCd(),
                idouktkInsInf.idoHkNySbCd(),
                idouktkInsInf.idoUserNamKn(),
                idouktkInsInf.idoUserNamKj(),
                idouktkInsInf.usNHjiSybCd(),
                idouktkInsInf.usNHjiHyjCd(),
                idouktkInsInf.usNKjhKhiCd(),
                idouktkInsInf.jgshCd(),
                idouktkInsInf.kCd(),
                idouktkInsInf.skbCd(),
                idouktkInsInf.riySubsysJCd(),
                idouktkInsInf.tanmatuNo(),
                idouktkInsInf.hkoksEmpNo(),
                idouktkInsInf.hkoksEmpNam(),
                idouktkInsInf.syokyNo(),
                idouktkInsInf.uktkKeiroCd(),
                idouktkInsInf.syokaiTsyCd(),
                idouktkInsInf.nwKsyCd(),
                idouktkInsInf.idohYmd(),
                idouktkInsInf.kStKnsnJoho(),
                idouktkInsInf.sStKnsnJoho(),
                idouktkInsInf.hkzSsyMryYmd(),
                idouktkInsInf.hkgSsyMryYmd(),
                idouktkInsInf.gasKyksybCd(),
                idouktkInsInf.tjhkTrksCd(),
                idouktkInsInf.hkkNaiyoAr(),
                idouktkInsInf.teiseiFilAr());
        idouktkDataAccess.insert(record, AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requireHukkSysShbt());
    }

    private void requireCorrelationId() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private void requireSystemDatetime() {
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private static LocalTime toReportTime(Integer reportTime) {
        return LocalTime.parse("%06d".formatted(reportTime), TIME_FORMATTER);
    }
}
