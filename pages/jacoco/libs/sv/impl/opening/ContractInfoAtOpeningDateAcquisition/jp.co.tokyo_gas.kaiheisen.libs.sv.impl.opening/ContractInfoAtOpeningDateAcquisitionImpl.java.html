        /**
         * 概要: 開栓作業日時点の契約情報を取得する.
         * 主な仕様:
         * - 託送供給依頼事業者コードが東京ガス以外の場合は、
         *   後続依存を呼び出さず直列専用通過メーターフラグのみ判定して返却する。
         * - 東京ガス分岐では DA-041-R013 で
         *   サービス契約・メーター設置場所関連情報を取得し、0件時は例外にせず空の契約情報で返却する。
         * - DA-041-R013 で取得できた場合は SV-170、DA-038-R009、
         *   DA-004-R003、DA-044-R004、DA-075-R004 を順に評価し、取得結果を出力 DTO へ詰め替える。
         * 既知の制約:
         * - RequestContext の systemDatetime が未設定の場合は
         *   fail-fast する。
         * - サービス契約、支払契約関連、支払契約が取得必須の箇所で
         *   0件だった場合は DataInconsistencyException を送出する。
         * - 引数の null/blank の妥当性判断は上位層責務とし、本実装では追加の必須チェックを行わない。
         *
         * @param gasMtstNo ガスメーター設置場所番号
         * @param kasgYmd 開栓作業年月日
         * @param tksuKijsCd 託送供給依頼事業者コード
         * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
         * @param kasJysb 開栓事由種別
         * @return 開栓日時点契約情報取得結果
         */
    @Override
    public ContractInfoAtOpeningDateAcquisitionOutput execute(
            final Long gasMtstNo,
            final LocalDate kasgYmd,
            final String tksuKijsCd,
            final String ckrtGmscJtkb,
            final String kasJysb) {
        boolean snyTkmtFlg = isDedicatedGasMeter(ckrtGmscJtkb);
        List<String> rkkiKnrnJtkbList =
                List.of(
                        chargeCalculationRelatedStatusRelatedValid,
                        chargeCalculationRelatedStatusRelatedEndWait);
        if (!isTokyoGasRetailer(tksuKijsCd)) {
            return emptyContractInfoAtOpeningOutput(snyTkmtFlg);
        }

        SvmstkrRecord svmstkrRecord =
                svmstkrDataAccess
                        .findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                                gasMtstNo,
                                kasgYmd,
                                rkkiKnrnJtkbList)
                        .orElse(null);
        if (svmstkrRecord == null) {
            return emptyContractInfoAtOpeningOutput(snyTkmtFlg);
        }

        LocalDateTime systemDatetime = requireSystemDatetime();
        ComprehensiveMeterJudgmentAtOpeningOutput comprehensiveOutput =
                comprehensiveMeterJudgmentAtOpening.execute(
                        svmstkrRecord.mtstNo(),
                        svmstkrRecord.svkyNo(),
                        kasJysb,
                        svmstkrRecord.rkkiSttYmd(),
                        svmstkrRecord.rkkiKnrnJtkb(),
                        svmstkrRecord.rkkiEndYmd());
        SvckyaRecord svckyaRecord = findSvckyaRecord(svmstkrRecord.svkyNo());
        ContractInfoAtOpeningDateItensakInf itensakInf =
                findItensakInf(
                        gasMtstNo,
                        svmstkrRecord.svkyNo());
        SvsihkrRecord svsihkrRecord =
                findSvsihkrRecord(
                        svmstkrRecord.svkyNo(),
                        systemDatetime.toLocalDate());
        SihakyRecord sihakyRecord =
                findSihakyRecord(
                        svsihkrRecord.sihKiyNo(),
                        systemDatetime.toLocalDate());

        return new ContractInfoAtOpeningDateAcquisitionOutput(
                toSvmstkrInf(svmstkrRecord),
                toSvckyaInf(svckyaRecord),
                itensakInf,
                toSvsihkrInf(svsihkrRecord),
                toSihakyInf(sihakyRecord),
                comprehensiveOutput == null
                        ? null
                        : comprehensiveOutput.suguHtkFlg(),
                comprehensiveOutput == null
                        ? null
                        : comprehensiveOutput.kzksyAtkCd(),
                false,
                itensakInf != null);
    }

    private boolean isTokyoGasRetailer(final String tksuKijsCd) {

    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(
                        () ->
                                new IllegalStateException(
                                        "RequestContext.systemDatetime is"
                                                + " required"));
    }

    private SvckyaRecord findSvckyaRecord(final Long svkyNo) {
        List<SvckyaRecord> svckyaRecords =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(
                        svkyNo,
                        serviceContractStatusUnderContract);
        if (svckyaRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs
                            .FIND_ALL_BY_SVKY_NO_AND_SVKY_JTKB
                            .identifier(),
                    parameterJsonForSvckya(svkyNo));
        }
        return svckyaRecords.stream()
                .max(
                        Comparator.comparing(
                                SvckyaRecord::skno,
                                Comparator.nullsLast(Integer::compareTo)))
                .orElseThrow();
    }

    private ContractInfoAtOpeningDateItensakInf findItensakInf(
            final Long gasMtstNo,
            final Long svkyNo) {
        return itensakDataAccess
                .findAllByGmtSetNoAndSyokyNoAndItenTksCd(
                        gasMtstNo,
                        svkyNo,
                        transferDestinationCancellationFlagNoConcept)
                .stream()
                .findFirst()
                .map(ContractInfoAtOpeningDateAcquisitionImpl::toItensakInf)
                .orElse(null);
    }

    private SvsihkrRecord findSvsihkrRecord(
            final Long svkyNo,
            final LocalDate systemDate) {
        String svsihkrCallSpecIdentifier =
                SvsihkrDataAccess.CallSpecs
                        .FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_SIH_KIY_JTKB
                        .identifier();
        return svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndSihKiyJtkb(
                        svkyNo,
                        systemDate,
                        paymentContractStatusApplying)
                .orElseThrow(
                        () ->
                                DataInconsistencySupport.create(
                                        svsihkrDataAccess.tableName(),
                                        svsihkrCallSpecIdentifier,
                                        parameterJsonForSvsihkr(
                                                svkyNo,
                                                systemDate)));
    }

    private SihakyRecord findSihakyRecord(
            final Long sihKiyNo,
            final LocalDate systemDate) {
        String sihakyCallSpecIdentifier =
                SihakyDataAccess.CallSpecs
                        .FIND_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_NOT_SIH_KIY_JTKB
                        .identifier();
        return sihakyDataAccess
                .findBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
                        sihKiyNo,
                        systemDate,
                        paymentContractStatusInvalid)
                .orElseThrow(
                        () ->
                                DataInconsistencySupport.create(
                                        sihakyDataAccess.tableName(),
                                        sihakyCallSpecIdentifier,
                                        parameterJsonForSihaky(
                                                sihKiyNo,
                                                systemDate)));
    }
