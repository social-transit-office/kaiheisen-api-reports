package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferWorkHistoryCommonPipelineEditAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransferWorkHistoryCommonPipelineEditAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.IsrkdInsInf;

/**
 * 概要: SV-246（開栓時異動・作業履歴（共通）＜導管＞共通編集）の実装。
 * 主な仕様:
 * - 入力値と RequestContext、定数ライブラリの値から異動・作業履歴登録項目を編集する。
 * - お客さま情報取得可否フラグは常に true を設定する。
 * 既知の制約:
 * - public メソッド引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 */
public final class TransferWorkHistoryCommonPipelineEditAtOpeningImpl
        implements TransferWorkHistoryCommonPipelineEditAtOpening {

    private static final String TRANSFER_WORK_HISTORY_DETAIL_SCREEN_ID = "CXXG0082";

    private static final DateTimeFormatter HMS_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    private final RequestContext requestContext;

    private final String caseHistoryResultTransfer;

    private final String itemHistoryRegistrationDestinationPlace;

    private final String transferWorkHistoryInquiryCodeSubject;

    /**
     * 概要: 出力編集に必要な依存と定数を初期化する。
     * 主な仕様:
     * - RequestContext は必須依存として保持する。
     * - 案件履歴区分、案件履歴登録先種別、異動・作業履歴照会対象コードを fail-fast に解決する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param requestContext リクエストコンテキスト
     */
    public TransferWorkHistoryCommonPipelineEditAtOpeningImpl(
            CiriusConstantService ciriusConstantService,
            RequestContext requestContext) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.caseHistoryResultTransfer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.CaseHistory.TYPE,
                ConstantLookupKeys.Cirius.CaseHistory.RESULT_TRANSFER);
        this.itemHistoryRegistrationDestinationPlace = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.TYPE,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.PLACE);
        this.transferWorkHistoryInquiryCodeSubject = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.SUBJECT);
    }

    /**
     * 概要: 開栓時異動・作業履歴（共通）＜導管＞の登録項目を編集する。
     * 主な仕様:
     * - RequestContext からシステム日時を取得し、登録依頼年月日と時分秒を組み立てる。
     * - お客さま情報取得可否フラグは常に true を設定する。
     * 既知の制約:
        * - RequestContext の値設定と妥当性確認は上位層責務とし、本メソッドでは補完しない。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kasgYmd 開栓作業年月日
     * @param hukkUsrId 報告ユーザーＩＤ
     * @param hukkUnmKj 報告ユーザー名漢字
     * @return 異動・作業履歴（共通）＜導管＞編集結果
     */
    @Override
    public TransferWorkHistoryCommonPipelineEditAtOpeningOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            String hukkUsrId,
            String hukkUnmKj) {
        LocalDateTime requestSystemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));

        IsrkdInsInf output = new IsrkdInsInf(
                gasMtstNo,
                requestSystemDatetime.toLocalDate(),
                caseHistoryResultTransfer,
                itemHistoryRegistrationDestinationPlace,
                transferWorkHistoryInquiryCodeSubject,
                toHms(requestSystemDatetime),
                kasgYmd,
                hukkUsrId,
                hukkUnmKj,
                TRANSFER_WORK_HISTORY_DETAIL_SCREEN_ID,
                Boolean.TRUE);

        return new TransferWorkHistoryCommonPipelineEditAtOpeningOutput(output);
    }

    /**
     * 概要: システム日時から HHmmss 形式の整数値を生成する。
     * 主な仕様:
     * - DateTimeFormatter("HHmmss") でゼロ埋めした 6 桁整数を返す。
     * 既知の制約:
     * - LocalDateTime は呼び出し前に null ではない前提で扱う。
     *
     * @param systemDatetime システム日時
     * @return HHmmss 形式の時分秒整数
     */
    private int toHms(LocalDateTime systemDatetime) {
        return Integer.parseInt(HMS_FORMATTER.format(systemDatetime));
    }
}
