package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.inject.Inject;
import java.time.LocalDateTime;
import java.util.Locale;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerResponse;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CustomerNameEditForOpeningCheck;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CustomerNameEditForOpeningCheckOutput;

/**
 * 概要: 開栓時チェック用おなまえ編集サービス実装。
 * 主な仕様: CIRIUS お客さま情報編集結果取得 API の編集結果から編集パターン3のお客さま名カナ2＋法人種別を取得し、ブランクサプレスと小文字大文字変換を適用して返却する。
 * 既知の制約: CIRIUS お客さま情報編集結果取得 API 呼出しの詳細な request 展開は gateway に委譲し、RequestContext の必須値不足時は fail-fast する。
 */
public final class CustomerNameEditForOpeningCheckImpl implements CustomerNameEditForOpeningCheck {

    private final EditedCustomerInfoGateway editedCustomerInfoGateway;

    private final String customerNameKanjiHiddenFlag;

    private final RequestContext requestContext;

    /**
     * 概要: CDI から依存を受け取って初期化する。
    * 主な仕様: CIRIUS お客さま情報編集結果取得 API の adapter を SV 用 gateway へ変換して委譲する。
    * 既知の制約: CIRIUS お客さま情報編集結果取得 API の adapter と RequestContext と CIRIUS 定数サービスが必須である。
     *
    * @param editCustomerCiriusAdapter CIRIUS お客さま情報編集結果取得 API の adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public CustomerNameEditForOpeningCheckImpl(
            EditCustomerCiriusAdapter editCustomerCiriusAdapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this(
                new EditedCustomerInfoCiriusGateway(editCustomerCiriusAdapter),
                requestContext,
                resolveCustomerNameKanjiHiddenFlag(ciriusConstantService));
    }

    /**
     * 概要: 依存を初期化する。
    * 主な仕様: CIRIUS お客さま情報編集結果取得 API 呼出し gateway と RequestContext をフィールドへ保持する。
     * 既知の制約: 必須依存が null の場合はコンストラクタ初期化時点で fail-fast する。
     *
    * @param editedCustomerInfoGateway CIRIUS お客さま情報編集結果取得 API 呼出し gateway
     * @param requestContext リクエストコンテキスト
     * @param customerNameKanjiHiddenFlag お客さま名漢字非表示フラグ
     */
    CustomerNameEditForOpeningCheckImpl(
            EditedCustomerInfoGateway editedCustomerInfoGateway,
            RequestContext requestContext,
            String customerNameKanjiHiddenFlag) {
        this.editedCustomerInfoGateway = Objects.requireNonNull(
                editedCustomerInfoGateway,
                "editedCustomerInfoGateway");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.customerNameKanjiHiddenFlag = Objects.requireNonNull(
            customerNameKanjiHiddenFlag,
            "customerNameKanjiHiddenFlag");
    }

    /**
     * 概要: 開栓時チェック用のお客さま名カナ1を編集する。
    * 主な仕様: RequestContext の必須値を解決して CIRIUS お客さま情報編集結果取得 API を呼び出し、編集パターン3のお客さま名カナ2＋法人種別へブランクサプレスと小文字大文字変換を適用して返却する。
    * 既知の制約: public 引数の null は補正せず下位 gateway へ引き渡し、CIRIUS お客さま情報編集結果取得 API の例外は握りつぶさず伝播する。
     *
     * @param ctnmKn1 お客さま名カナ1
     * @param ctnm1HujnShbt お客さま名1＿法人種別
     * @param ctnm1HshiShbt お客さま名1＿法人種別表示位置種別
     * @return チェック用お客さま名1カナ
     */
    @Override
    public CustomerNameEditForOpeningCheckOutput execute(
            String ctnmKn1,
            String ctnm1HujnShbt,
            String ctnm1HshiShbt) {
        requireCorrelationId();
        requireHukkSysShbt();
        requireSystemDatetime();
        EditedCustomerInfoResult editedCustomerInfoResult = editedCustomerInfoGateway.execute(
                new EditedCustomerInfoRequest(
                        ctnmKn1,
                        ctnm1HujnShbt,
                        ctnm1HshiShbt,
                        customerNameKanjiHiddenFlag));
        return new CustomerNameEditForOpeningCheckOutput(
                normalizeEditedCustomerNameKana(
                editedCustomerInfoResult.editedPattern3CustomerNameKana2WithCorporateType()));
    }

    /**
     * 概要: お客さま名漢字非表示フラグを CIRIUS 定数から解決する。
     * 主な仕様: 01500 の 0 を必須 CIRIUS 定数として検証し、コード値を返却する。
     * 既知の制約: 必須定数が未定義の場合は下位例外をそのまま送出する。
     *
     * @param ciriusConstantService CIRIUS 定数サービス
     * @return お客さま名漢字非表示フラグ
     */
    private static String resolveCustomerNameKanjiHiddenFlag(
            CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.TYPE,
                ConstantLookupKeys.Cirius.NameKanjiDisplayFlag.KANJI_NAME_HIDDEN);
    }

    /**
     * 概要: 相関IDを取得する。
     * 主な仕様: RequestContext から相関IDを取得し、未設定時は fail-fast する。
     * 既知の制約: Optional の空値は許容しない。
     *
     * @return 相関ID
     */
    private String requireCorrelationId() {
        return requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
    }

    /**
     * 概要: 報告システム種別を取得する。
     * 主な仕様: RequestContext から報告システム種別を取得し、未設定時は fail-fast する。
     * 既知の制約: Optional の空値は許容しない。
     *
     * @return 報告システム種別
     */
    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    /**
     * 概要: システム日時を取得する。
     * 主な仕様: RequestContext からシステム日時を取得し、未設定時は fail-fast する。
     * 既知の制約: Optional の空値は許容しない。
     *
     * @return システム日時
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 編集済みお客さま名カナを正規化する。
     * 主な仕様: ブランクサプレスを行った後に小文字大文字変換を適用する。
     * 既知の制約: null 入力は null のまま返却し、日本語小文字の正規化は JDK 標準変換で扱えない範囲を補正しない。
     *
     * @param editedCustomerNameKanaWithCorporateType 編集済みお客さま名カナ1＋法人種別
     * @return 正規化後のチェック用お客さま名1カナ
     */
    private static String normalizeEditedCustomerNameKana(String editedCustomerNameKanaWithCorporateType) {
        if (editedCustomerNameKanaWithCorporateType == null) {
            return null;
        }
        return suppressBlanks(editedCustomerNameKanaWithCorporateType).toUpperCase(Locale.ROOT);
    }

    /**
     * 概要: 文字列から空白文字を除去する。
     * 主な仕様: 半角空白、全角空白、改行やタブを含む Java whitespace を除去する。
     * 既知の制約: 空白以外の記号や文字種変換は行わない。
     *
     * @param value 入力文字列
     * @return 空白除去後文字列
     */
    private static String suppressBlanks(String value) {
        StringBuilder builder = new StringBuilder(value.length());
        value.codePoints().forEach(codePoint -> {
            if (!Character.isWhitespace(codePoint) && codePoint != '　') {
                builder.appendCodePoint(codePoint);
            }
        });
        return builder.toString();
    }

    /** CIRIUS お客さま情報編集結果取得 API 呼出しを抽象化する。 */
    private interface EditedCustomerInfoGateway {

        /**
         * CIRIUS お客さま情報編集結果取得 API を呼び出して編集済みお客さま情報を取得する。
         *
         * @param request CIRIUS お客さま情報編集結果取得 API 呼出し入力
         * @return 編集済みお客さま情報
         */
        EditedCustomerInfoResult execute(EditedCustomerInfoRequest request);
    }

    /** CIRIUS お客さま情報編集結果取得 API の request body に必要な最小入力。 */
    private record EditedCustomerInfoRequest(
            String ctnmKn1,
            String ctnm1HujnShbt,
            String ctnm1HshiShbt,
            String ctnmKjhkFlg) {
    }

    /** CIRIUS お客さま情報編集結果取得 API 応答から SV-174 が必要とする最小出力。 */
    private record EditedCustomerInfoResult(
            String editedPattern3CustomerNameKana2WithCorporateType) {
    }

    /** CIRIUS お客さま情報編集結果取得 API の adapter を SV-174 用 gateway に変換する。 */
    private static final class EditedCustomerInfoCiriusGateway implements EditedCustomerInfoGateway {

        private final EditCustomerCiriusAdapter editCustomerCiriusAdapter;

        private EditedCustomerInfoCiriusGateway(EditCustomerCiriusAdapter editCustomerCiriusAdapter) {
            this.editCustomerCiriusAdapter = Objects.requireNonNull(
                    editCustomerCiriusAdapter,
                    "editCustomerCiriusAdapter");
        }

        @Override
        public EditedCustomerInfoResult execute(EditedCustomerInfoRequest request) {
            EditCustomerResponse response = editCustomerCiriusAdapter.editCustomer(
                    new EditCustomerRequest(
                            new EditCustomerRequest.BeforeCustomerInfo(
                                    new EditCustomerRequest.BeforeNameInfo(
                                            "",
                                            "",
                                            "",
                                            "",
                                            "",
                                            request.ctnmKn1(),
                                            request.ctnm1HujnShbt(),
                                            request.ctnm1HshiShbt(),
                                            request.ctnmKjhkFlg(),
                                            "",
                                            "",
                                            "",
                                            "",
                                            ""),
                                    null)));
            return new EditedCustomerInfoResult(resolveEditedPattern3CustomerNameKana2(response));
        }

        private static String resolveEditedPattern3CustomerNameKana2(EditCustomerResponse response) {
            if (response == null) {
                return null;
            }
            return response.editedPattern3CustomerNameKana2();
        }
    }
}
