package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningInProgressCheckAtOpening;

/**
 * 概要: SV-177（開栓時開栓中チェック）の実装クラス。
 * 主な仕様:
 * - 判定に必要な CIRIUS 定数「02100 1:開栓中」をコンストラクタで fail-fast に解決する。
 * - ガスメーター開閉栓状態区分が開栓中の場合は開栓共通エラー 020100016 を送出する。
 * - それ以外の場合は正常終了する。
 * 既知の制約:
 * - RequestContext および下位依存呼び出しは使用しない。
 * - 入力値の trim や blank 判定は追加しない。
 */
public final class OpeningInProgressCheckAtOpeningImpl implements OpeningInProgressCheckAtOpening {

    private final String openingStatus;

    /**
     * 概要: 判定に必要な CIRIUS 定数を初期化する。
     * 主な仕様:
     * - ガスメーター開閉栓状態区分の「開栓中」を必須定数として解決する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public OpeningInProgressCheckAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 概要: ガスメーター開閉栓状態区分を判定して SV-177 を実行する。
     * 主な仕様:
     * - ガスメーター開閉栓状態区分が開栓中の場合は開栓共通エラー 020100016 を送出する。
     * - それ以外の場合は何も返さずに正常終了する。
     * 既知の制約:
     * - 引数が null の場合は開栓中以外として扱う。
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @throws KaisenRuleException ガスメーター開閉栓状態区分が開栓中の場合
     */
    @Override
    public void execute(String gmtKhsJtkb) {
        if (openingStatus.equals(gmtKhsJtkb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100016);
        }
    }
}
