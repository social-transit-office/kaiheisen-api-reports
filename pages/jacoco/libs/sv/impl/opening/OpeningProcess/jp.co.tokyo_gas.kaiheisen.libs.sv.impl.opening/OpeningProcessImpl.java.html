    /**
     * 開栓処理を実行する。
     */
    @Override
    public OpeningProcessOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            String kasJysb,
            LocalDate kktTyenYmd,
            LocalDate kktTkysYmd,
            Boolean snyTkmtFlg,
            Boolean digwrHntFlg,
            Boolean suguHtkFlg,
            Boolean ksenYmdTskgmstUmFlg,
            OpeningProcessGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            OpeningProcessGmtsetdSystemDateInf gmtsetdSystemDateInf,
            TatemonInf tatemonInf,
            String ohmKj,
            String kktGmtsetdGasYtCd,
            JusyoKktInf jusyoKktInf) {
        if (Boolean.TRUE.equals(digwrHntFlg)) {
            supportInfoDeletion.execute(digwrHntFlg, gasMtstNo);
            knsssvcDataAccess.deleteByGasMtstNoAndIdujMskhFlg(gasMtstNo, Boolean.TRUE, null);
            cstsvcDataAccess.deleteByGmtSetNoAndIdoMasyKhiCd(gasMtstNo, deleteCode, null);
        }

        GmtsetkRecord currentGmtsetk = requireGmtsetk(gmtsetdKsenSaymdInf.gasMtstNo(), kasgYmd);

        if (gmtsetdKsenSaymdInf.tkysYmd() != null && gmtsetdKsenSaymdInf.tkysYmd().isEqual(kktTkysYmd)) {
            deleteCurrentGeneration(
                    gmtsetdKsenSaymdInf,
                    currentGmtsetk);
        } else if (gmtsetdKsenSaymdInf.tkysYmd() != null && gmtsetdKsenSaymdInf.tkysYmd().isBefore(kktTkysYmd)) {
            gasMeterLocationInfoTermination.execute(
                    noExclusionDivision,
                    new GmtsetdTerminationInf(
                            gmtsetdKsenSaymdInf.gasMtstNo(),
                            gmtsetdKsenSaymdInf.tkysYmd(),
                            kktTyenYmd,
                            null),
                    new GmtsetkTerminationInf(
                            gmtsetdKsenSaymdInf.gasMtstNo(),
                            gmtsetdKsenSaymdInf.tkysYmd(),
                            kktTyenYmd,
                            null),
                    new MtsetkTerminationInf(
                            gmtsetdKsenSaymdInf.gasMtstNo(),
                            gmtsetdKsenSaymdInf.tkysYmd(),
                            kktTyenYmd,
                            null));
        }

        String currentJutKbn = resolveCurrentJutKbn(gmtsetdSystemDateInf, tatemonInf);
        registerGeneration(
                kktTkysYmd,
                kasgYmd,
                kasJysb,
                kktGmtsetdGasYtCd,
                snyTkmtFlg,
                digwrHntFlg,
                suguHtkFlg,
                ksenYmdTskgmstUmFlg,
                gmtsetdKsenSaymdInf,
                gmtsetdSystemDateInf,
                ohmKj,
                jusyoKktInf,
                currentGmtsetk,
                currentJutKbn);

        rebuildFutureGenerations(
                gmtsetdKsenSaymdInf.gasMtstNo(),
                kasgYmd,
                kasJysb,
                kktGmtsetdGasYtCd,
                snyTkmtFlg,
                digwrHntFlg,
                suguHtkFlg,
                ksenYmdTskgmstUmFlg,
                gmtsetdSystemDateInf,
                ohmKj,
                jusyoKktInf);

        var scheduledDateUpdateOutput = nextRegularMeterReadingScheduledDateUpdate.execute(gasMtstNo);
        return new OpeningProcessOutput(toKnsytiInf(scheduledDateUpdateOutput));
    }

    private GmtsetkRecord requireGmtsetk(Long gasMtstNo, LocalDate knskKjnYmd) {
        return gmtsetkDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        gmtsetkDataAccess.tableName(),
                        GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                        new FlatJsonBuilder().put("gasMtstNo", gasMtstNo).put("knskKjnYmd", knskKjnYmd)));
    }

    private void deleteCurrentGeneration(
            OpeningProcessGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            GmtsetkRecord currentGmtsetk) {
        int gmtsetdDeletedCount = gmtsetdDataAccess.deleteByGasMtstNoAndTkysYmd(
                gmtsetdKsenSaymdInf.gasMtstNo(),
                gmtsetdKsenSaymdInf.tkysYmd(),
                gmtsetdKsenSaymdInf.recUpdJnjNo());
        ensureDeletedCount(
                gmtsetdDeletedCount,
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gmtsetdKsenSaymdInf.gasMtstNo())
                        .put("tkysYmd", gmtsetdKsenSaymdInf.tkysYmd())
                        .put("recUpdJnjNo", gmtsetdKsenSaymdInf.recUpdJnjNo()));

        int gmtsetkDeletedCount = gmtsetkDataAccess.deleteByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
                currentGmtsetk.gasMtstNo(),
                currentGmtsetk.tkysYmd(),
                currentGmtsetk.recUpdJnjNo());
        ensureDeletedCount(
                gmtsetkDeletedCount,
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", currentGmtsetk.gasMtstNo())
                        .put("tkysYmd", currentGmtsetk.tkysYmd())
                        .put("recUpdJnjNo", currentGmtsetk.recUpdJnjNo()));

        int mtsetkDeletedCount = mtsetkDataAccess.deleteByMtstNoAndTkysYmd(
                gmtsetdKsenSaymdInf.gasMtstNo(),
                gmtsetdKsenSaymdInf.tkysYmd(),
                null);
        ensureDeletedCount(
                mtsetkDeletedCount,
                mtsetkDataAccess.tableName(),
                MtsetkDataAccess.CallSpecs.DELETE_BY_MTST_NO_AND_TKYS_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("mtstNo", gmtsetdKsenSaymdInf.gasMtstNo())
                        .put("tkysYmd", gmtsetdKsenSaymdInf.tkysYmd())
                        .put("recUpdJnjNo", null));
    }

    private void rebuildFutureGenerations(
            Long gasMtstNo,
            LocalDate kasgYmd,
            String kasJysb,
            String kktGmtsetdGasYtCd,
            Boolean snyTkmtFlg,
            Boolean digwrHntFlg,
            Boolean suguHtkFlg,
            Boolean ksenYmdTskgmstUmFlg,
            OpeningProcessGmtsetdSystemDateInf gmtsetdSystemDateInf,
            String ohmKj,
            JusyoKktInf jusyoKktInf) {
        for (GmtsetdRecord futureGmtsetd : gmtsetdDataAccess.findAllByGasMtstNoAndTkysYmd(gasMtstNo, kasgYmd)) {
            OpeningProcessGmtsetdKsenSaymdInf futureSource = toOpeningSource(futureGmtsetd);
            GmtsetkRecord futureGmtsetk =
                    requireGmtsetkByTkysYmd(futureGmtsetd.gasMtstNo(), futureGmtsetd.tkysYmd());
            MtsetkRecord futureMtsetk =
                    requireMtsetk(futureSource.gasMtstNo(), futureGmtsetd.tkysYmd());

            int gmtsetdDeletedCount = gmtsetdDataAccess.deleteByGasMtstNoAndTkysYmd(
                    futureGmtsetd.gasMtstNo(),
                    futureGmtsetd.tkysYmd(),
                    futureGmtsetd.recUpdJnjNo());
            ensureDeletedCount(
                    gmtsetdDeletedCount,
                    gmtsetdDataAccess.tableName(),
                    GmtsetdDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", futureGmtsetd.gasMtstNo())
                            .put("tkysYmd", futureGmtsetd.tkysYmd())
                            .put("recUpdJnjNo", futureGmtsetd.recUpdJnjNo()));

            int gmtsetkDeletedCount = gmtsetkDataAccess.deleteByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
                    futureGmtsetk.gasMtstNo(),
                    futureGmtsetk.tkysYmd(),
                    futureGmtsetk.recUpdJnjNo());
            ensureDeletedCount(
                    gmtsetkDeletedCount,
                    gmtsetkDataAccess.tableName(),
                    GmtsetkDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", futureGmtsetk.gasMtstNo())
                            .put("tkysYmd", futureGmtsetk.tkysYmd())
                            .put("recUpdJnjNo", futureGmtsetk.recUpdJnjNo()));

            int mtsetkDeletedCount = mtsetkDataAccess.deleteByMtstNoAndTkysYmd(
                    futureMtsetk.mtstNo(),
                    futureMtsetk.tkysYmd(),
                    futureMtsetk.recUpdJnjNo());
            ensureDeletedCount(
                    mtsetkDeletedCount,
                    mtsetkDataAccess.tableName(),
                    MtsetkDataAccess.CallSpecs.DELETE_BY_MTST_NO_AND_TKYS_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("mtstNo", futureMtsetk.mtstNo())
                            .put("tkysYmd", futureMtsetk.tkysYmd())
                            .put("recUpdJnjNo", futureMtsetk.recUpdJnjNo()));

            registerGeneration(
                    futureGmtsetd.tkysYmd(),
                    kasgYmd,
                    kasJysb,
                    kktGmtsetdGasYtCd,
                    snyTkmtFlg,
                    digwrHntFlg,
                    suguHtkFlg,
                    ksenYmdTskgmstUmFlg,
                    futureSource,
                    gmtsetdSystemDateInf,
                    ohmKj,
                    jusyoKktInf,
                    futureGmtsetk,
                    resolveFutureJutKbn(futureSource));
        }
    }

    private GmtsetkRecord requireGmtsetkByTkysYmd(Long gasMtstNo, LocalDate tkysYmd) {
        return gmtsetkDataAccess.findByGasMtstNoAndTkysYmd(gasMtstNo, tkysYmd)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        gmtsetkDataAccess.tableName(),
                        GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD.identifier(),
                        new FlatJsonBuilder().put("gasMtstNo", gasMtstNo).put("tkysYmd", tkysYmd)));
    }

    private void ensureDeletedCount(int deletedCount, String tableName, String daIdentifier, FlatJsonBuilder jsonBuilder) {
        if (deletedCount == 0) {
            throw DataInconsistencySupport.create(tableName, daIdentifier, jsonBuilder);
        }
    }
