package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CancellationIncompleteCheckContractValid;

/** 開栓時解約未済チェック（契約有効）サービス実装. */
public final class CancellationIncompleteCheckContractValidImpl
        implements CancellationIncompleteCheckContractValid {

    /**
     * 解約未済条件を判定し、該当時は開栓共通エラーを送出する.
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param suguHtkFlg 総合判定結果フラグ
     * @param snyTkmtFlg 直列専用通過メーターフラグ
     * @throws KaisenRuleException 解約未済条件に該当する場合
     */
    @Override
    public void execute(
            final Boolean kiyUmFlg,
            final Boolean suguHtkFlg,
            final Boolean snyTkmtFlg) {
        if (!suguHtkFlg && !snyTkmtFlg && kiyUmFlg) {
            throw new KaisenRuleException(
                    ErrorCodeConstants.ERROR_CODE_020100036);
        }
    }
}
