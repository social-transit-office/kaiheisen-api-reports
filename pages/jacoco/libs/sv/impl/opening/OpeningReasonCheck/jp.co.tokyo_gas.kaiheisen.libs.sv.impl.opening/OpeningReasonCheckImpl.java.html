package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReasonCheck;

/**
 * 概要: SV-202（開栓事由チェック）の実装クラス。
 * 主な仕様:
 * - 契約有無、氏名カナ、報告開栓事由種別、託送関連有無フラグ、閉栓事由種別の組み合わせを判定する。
 * - TSV / YAML 設計で禁止されている開栓事由パターンを検知した場合は開栓共通エラー 020100041 を送出する。
 * - 判定に使用する CIRIUS 定数はコンストラクタ初期化時に必須解決し、検証済みコードで比較する。
 * - 下位 DA や他 SV 呼び出しは行わず、入力値のみで分岐判定を完結させる。
 * 既知の制約:
 * - 入力値の null 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となり、インスタンス生成に失敗する。
 */
public final class OpeningReasonCheckImpl implements OpeningReasonCheck {

    // 継続使用を表す解決済み報告開栓事由種別コード。
    private final String openingReasonContinuedUse;

    // 供給再開を表す解決済み報告開栓事由種別コード。
    private final String openingReasonSupplyResumed;

    // 保安復旧を表す解決済み報告開栓事由種別コード。
    private final String openingReasonSecurityRestoration;

    // 保安閉栓を表す解決済み閉栓事由種別コード。
    private final String closureReasonSecurityClosure;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 報告開栓事由種別 04900 の継続使用、供給再開、保安復旧を必須解決して保持する。
     * - 閉栓事由種別 J0273 の保安閉栓を必須解決して保持する。
     * 既知の制約:
     * - CIRIUS 定数取得サービスの null は許容しない。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public OpeningReasonCheckImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.openingReasonContinuedUse =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.openingReasonSupplyResumed =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.SUPPLY_RESUMED);
        this.openingReasonSecurityRestoration =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.SECURITY_RESTORATION);
        this.closureReasonSecurityClosure =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE);
    }

    /**
     * 概要: 開栓事由チェックを実行し、禁止パターンに該当する場合は開栓共通エラーを送出する。
     * 主な仕様:
     * - 契約なし時は氏名不一致かつ継続使用、保安閉栓かつ保安復旧、供給再開の各パターンを判定する。
     * - 契約あり時は保安閉栓と継続使用または供給再開の組み合わせを判定する。
     * - いずれかの禁止条件に一致した場合は `KaisenRuleException` により 020100041 を送出する。
     * 既知の制約:
     * - `correlationId`、`hukkSysShbt`、`systemDatetime` は SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードは追加せず、上位バリデーション済み前提で評価する。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param ctnmKn1Hsen お客さま名カナ１（閉栓時）
     * @param ctnmKn1Ksen お客さま名カナ１（開栓時）
     * @param hkkKasJysb 報告開栓事由種別
     * @param tskgmstUmFlg 託送サービス契約・ガスメーター設置場所関連有無フラグ
     * @param hesnJysb 閉栓事由種別
     */
    @Override
    public void execute(
            boolean kiyUmFlg,
            String ctnmKn1Hsen,
            String ctnmKn1Ksen,
            String hkkKasJysb,
            boolean tskgmstUmFlg,
            String hesnJysb) {
        if (shouldRejectForNoContractNameMismatchAndContinuedUseAndConsignmentRelated(
                        kiyUmFlg, ctnmKn1Hsen, ctnmKn1Ksen, hkkKasJysb, tskgmstUmFlg)
                || shouldRejectForNoContractSecurityRecovery(kiyUmFlg, hesnJysb, hkkKasJysb)
                || shouldRejectForNoContractSupplyResumed(kiyUmFlg, hkkKasJysb)
                || shouldRejectForContractedSecurityClosure(kiyUmFlg, hesnJysb, hkkKasJysb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100041);
        }
    }

    /**
     * 概要: 契約なし時の氏名不一致かつ継続使用パターンを判定する。
     * 主な仕様:
     * - 契約有無フラグが false、閉栓時/開栓時のお客さま名カナが不一致、開栓事由が継続使用、託送関連有無が true の全条件成立時に true を返す。
     * - step 1.a.1.a の禁止パターン判定をそのままメソッド化している。
     * 既知の制約:
     * - 文字列の正規化や大文字小文字変換は行わず、`String#equals` による完全一致で比較する。
     * - 入力文字列の null 許可/不許可の妥当性は呼び出し元で担保されている前提とする。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param ctnmKn1Hsen お客さま名カナ１（閉栓時）
     * @param ctnmKn1Ksen お客さま名カナ１（開栓時）
     * @param hkkKasJysb 報告開栓事由種別
     * @param tskgmstUmFlg 託送サービス契約・ガスメーター設置場所関連有無フラグ
     * @return 禁止パターンに一致する場合 true
     */
    private boolean shouldRejectForNoContractNameMismatchAndContinuedUseAndConsignmentRelated(
            boolean kiyUmFlg,
            String ctnmKn1Hsen,
            String ctnmKn1Ksen,
            String hkkKasJysb,
            boolean tskgmstUmFlg) {
        return !kiyUmFlg
                && !Objects.equals(ctnmKn1Hsen, ctnmKn1Ksen)
                && openingReasonContinuedUse.equals(hkkKasJysb)
                && tskgmstUmFlg;
    }

    /**
     * 概要: 契約なし時の保安閉栓かつ保安復旧パターンを判定する。
     * 主な仕様:
     * - 契約有無フラグが false、閉栓事由が保安閉栓、開栓事由が保安復旧の全条件成立時に true を返す。
     * - step 1.a.1.b.1.a の禁止パターン判定に対応する。
     * 既知の制約:
     * - 契約なし時の先行判定を通過した後でも本メソッド単体では排他制御を持たず、純粋に条件式のみを評価する。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param hesnJysb 閉栓事由種別
     * @param hkkKasJysb 報告開栓事由種別
     * @return 禁止パターンに一致する場合 true
     */
    private boolean shouldRejectForNoContractSecurityRecovery(
            boolean kiyUmFlg,
            String hesnJysb,
            String hkkKasJysb) {
        return !kiyUmFlg
                && closureReasonSecurityClosure.equals(hesnJysb)
                && openingReasonSecurityRestoration.equals(hkkKasJysb);
    }

    /**
     * 概要: 契約なし時の供給再開パターンを判定する。
     * 主な仕様:
     * - 契約有無フラグが false かつ開栓事由が供給再開である場合に true を返す。
     * - step 1.a.1.b.1.b.1.a の禁止パターン判定に対応する。
     * 既知の制約:
     * - 閉栓事由との組み合わせは本メソッドでは見ず、供給再開単独条件のみを評価する。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param hkkKasJysb 報告開栓事由種別
     * @return 禁止パターンに一致する場合 true
     */
    private boolean shouldRejectForNoContractSupplyResumed(
            boolean kiyUmFlg,
            String hkkKasJysb) {
        return !kiyUmFlg && openingReasonSupplyResumed.equals(hkkKasJysb);
    }

    /**
     * 概要: 契約あり時の保安閉栓かつ継続使用/供給再開パターンを判定する。
     * 主な仕様:
     * - 契約有無フラグが true、閉栓事由が保安閉栓、開栓事由が継続使用または供給再開のとき true を返す。
     * - step 1.b.1.a の禁止パターン判定に対応する。
     * 既知の制約:
     * - 継続使用と供給再開以外の開栓事由は、契約あり分岐では一律に正常終了側として扱う。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param hesnJysb 閉栓事由種別
     * @param hkkKasJysb 報告開栓事由種別
     * @return 禁止パターンに一致する場合 true
     */
    private boolean shouldRejectForContractedSecurityClosure(
            boolean kiyUmFlg,
            String hesnJysb,
            String hkkKasJysb) {
        return kiyUmFlg
                && closureReasonSecurityClosure.equals(hesnJysb)
                && (openingReasonContinuedUse.equals(hkkKasJysb)
                        || openingReasonSupplyResumed.equals(hkkKasJysb));
    }
}
