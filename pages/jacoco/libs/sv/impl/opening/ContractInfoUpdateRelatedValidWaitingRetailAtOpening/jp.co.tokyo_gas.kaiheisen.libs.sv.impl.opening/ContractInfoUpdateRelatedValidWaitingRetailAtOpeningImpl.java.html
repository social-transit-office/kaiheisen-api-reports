    /**
     * 概要: 関連有効待ちの主関連・契約情報を更新し、確定契約情報 DTO を返却する。
     * 主な仕様:
     * - 主関連更新件数 0 件、最大世代管理番号未取得、最大契約特定世代管理番号未取得、契約適用待ち更新件数 0 件はデータ不整合とする。
     * - RequestContext からは監査用の報告システム種別のみを参照し、未設定時は fail-fast する。
     *
     * @param kasgYmd 開栓作業年月日
     * @param kktTyenYmd 確定適用終了年月日
     * @param kktTkysYmd 確定適用開始年月日
     * @param kktSvkyNo 確定サービス契約番号
     * @param kktSihKiyNo 確定支払契約番号
     * @param kktSqtiKiyNo 確定請求対象契約番号
     * @param kktSikKrno 確定請求管理番号
     * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
     * @param kktSvckyaGasYtCd 確定サービス契約ガス用途コード
     * @param kktSziKziKbn 確定消費税課税区分
     * @param svckyaMchUmuFlg サービス契約適用待ち有無フラグ
     * @param svmstkrYukuMchInf サービス契約・メーター設置場所関連情報（関連有効待ち）
     * @param svckyaKiyKtymInf サービス契約情報（契約適用待ち）
     * @return 登録した契約情報 DTO
     */
    @Override
    public ContractInfoUpdateRelatedValidWaitingRetailAtOpeningOutput execute(
            LocalDate kasgYmd,
            LocalDate kktTyenYmd,
            LocalDate kktTkysYmd,
            Long kktSvkyNo,
            Long kktSihKiyNo,
            String kktSqtiKiyNo,
            Long kktSikKrno,
            Long kktSyoKyoTrno,
            String kktSvckyaGasYtCd,
            String kktSziKziKbn,
            Boolean svckyaMchUmuFlg,
            SvmstkrYukuMchInf svmstkrYukuMchInf,
            SvckyaKiyKtymInf svckyaKiyKtymInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        LocalDate applicableEndDate = requireMaximumApplicableEndDate(kktTyenYmd);
        Long serviceContractNumber = resolveServiceContractNumber(
                kktSvkyNo,
                svmstkrYukuMchInf,
                svckyaMchUmuFlg,
                svckyaKiyKtymInf);

        svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                buildPrimaryRelationUpdateRecord(svmstkrYukuMchInf, kktTkysYmd),
                EnumSet.of(SvmstkrColumn.RKKI_KNRN_JTKB, SvmstkrColumn.RKKI_STT_YMD),
                new SvmstkrSknoOptionalUpdateKey(
                        svmstkrYukuMchInf.mtstNo(),
                        svmstkrYukuMchInf.svkyNo(),
                        svmstkrYukuMchInf.skno(),
                        svmstkrYukuMchInf.recUpdJnjNo()),
                auditContext);

        SvckyaRecord latestServiceContract = svckyaDataAccess
                .findLatestBySvkyNoAndSvkyJtkbNot(serviceContractNumber)
                .orElseThrow(() -> createLatestServiceContractMissingException(serviceContractNumber));
        int maxSkno = latestServiceContract.skno() + 1;
        int maxKyttSkno = svckyaDataAccess.findMaxKyttSknoBySvkyNo(serviceContractNumber)
                .map(value -> value + 1)
                .orElseThrow(() -> createMaxKyttSknoMissingException(serviceContractNumber));

        SvckyaWriteRecord insertedServiceContract = buildInsertedServiceContractRecord(
                kktTkysYmd,
                serviceContractNumber,
                kktSqtiKiyNo,
                kktSyoKyoTrno,
                kktSvckyaGasYtCd,
                kktSziKziKbn,
                kasgYmd,
                maxSkno,
                maxKyttSkno);
        svckyaDataAccess.insert(insertedServiceContract, auditContext);

        if (Boolean.TRUE.equals(svckyaMchUmuFlg)) {
            svckyaDataAccess.updateBySvkyNoAndSknoAndRecUpdJnjNo(
                    buildWaitingServiceContractUpdateRecord(
                            serviceContractNumber,
                            svckyaKiyKtymInf,
                            kktTkysYmd,
                            kasgYmd,
                            kktSvckyaGasYtCd),
                    EnumSet.of(
                            SvckyaColumn.SVC_RYKS_YMD,
                            SvckyaColumn.SVKY_TIKT_YMD,
                            SvckyaColumn.SZIR_HTYO_YMD,
                            SvckyaColumn.BYSS_UM_FLG,
                            SvckyaColumn.GAS_YT_CD,
                            SvckyaColumn.RKN_KHUM_FLG,
                            SvckyaColumn.KKYS_SISN_JTKB,
                            SvckyaColumn.TSZI_KYSB_NO),
                    new SvckyaSknoJnjUpdateKey(
                            serviceContractNumber,
                            svckyaKiyKtymInf.skno(),
                            svckyaKiyKtymInf.recUpdJnjNo()),
                    auditContext);
        }

        SvsihkrWriteRecord insertedServicePaymentRelation =
                buildInsertedServicePaymentRelationRecord(
                        kktTkysYmd,
                        applicableEndDate,
                        serviceContractNumber,
                        kktSihKiyNo);
        svsihkrDataAccess.insert(insertedServicePaymentRelation, auditContext);

        SihakyWriteRecord insertedPaymentContract = buildInsertedPaymentContractRecord(
                kktTkysYmd,
                applicableEndDate,
                kktSihKiyNo,
                kktSikKrno,
                kktSyoKyoTrno);
        sihakyDataAccess.insert(insertedPaymentContract, auditContext);

        hrikmiDataAccess.insert(
                buildInsertedPaymentRelatedInfoRecord(kktTkysYmd, applicableEndDate, kktSihKiyNo),
                auditContext);
        sihtkskDataAccess.insert(buildInsertedDunningRecord(kktSihKiyNo), auditContext);

        return new ContractInfoUpdateRelatedValidWaitingRetailAtOpeningOutput(
                toSvckyaKktInf(insertedServiceContract),
                toSvsihkrKktInf(insertedServicePaymentRelation),
                toSihakyKktInf(insertedPaymentContract));
    }

    /**
     * 概要: 入力オブジェクト間で扱うサービス契約番号の整合性を検証する。
     * 主な仕様:
     * - 主関連情報のサービス契約番号を基準値とし、確定サービス契約番号との不一致は fail-fast する。
     * - 契約適用待ち更新を行う場合は、契約適用待ち情報のサービス契約番号も同一値であることを要求する。
     *
     * @param kktSvkyNo 確定サービス契約番号
     * @param svmstkrYukuMchInf サービス契約・メーター設置場所関連情報（関連有効待ち）
     * @param svckyaMchUmuFlg サービス契約適用待ち有無フラグ
     * @param svckyaKiyKtymInf サービス契約情報（契約適用待ち）
     * @return 検証済みのサービス契約番号
     */
    private Long resolveServiceContractNumber(
            Long kktSvkyNo,
            SvmstkrYukuMchInf svmstkrYukuMchInf,
            Boolean svckyaMchUmuFlg,
            SvckyaKiyKtymInf svckyaKiyKtymInf) {
        Long serviceContractNumber = svmstkrYukuMchInf.svkyNo();
        if (!Objects.equals(serviceContractNumber, kktSvkyNo)) {
            throw new IllegalArgumentException(
                    "Service contract number mismatch between kktSvkyNo and svmstkrYukuMchInf.svkyNo: "
                            + "kktSvkyNo="
                            + kktSvkyNo
                            + ", svmstkrYukuMchInf.svkyNo="
                            + serviceContractNumber);
        }
        if (Boolean.TRUE.equals(svckyaMchUmuFlg)
                && !Objects.equals(serviceContractNumber, svckyaKiyKtymInf.svkyNo())) {
            throw new IllegalArgumentException(
                    "Service contract number mismatch between svmstkrYukuMchInf.svkyNo and svckyaKiyKtymInf.svkyNo: "
                            + "svmstkrYukuMchInf.svkyNo="
                            + serviceContractNumber
                            + ", svckyaKiyKtymInf.svkyNo="
                            + svckyaKiyKtymInf.svkyNo());
        }
        return serviceContractNumber;
    }

    /**
     * 概要: 適用終了年月日が設計どおり最大日付であることを検証する。
     * 主な仕様:
     * - DA-044-C001、DA-075-C001、DA-093-C001 の適用終了年月日は最大日付固定である。
     * - 最大日付以外が渡された場合は登録前に fail-fast する。
     *
     * @param kktTyenYmd 確定適用終了年月日
     * @return 検証済みの最大日付
     */
    private LocalDate requireMaximumApplicableEndDate(LocalDate kktTyenYmd) {
        if (!Objects.equals(MAX_DATE, kktTyenYmd)) {
            throw new IllegalArgumentException(
                    "Applicable end date must be MAX_DATE for DA-044-C001/DA-075-C001/DA-093-C001: "
                            + "kktTyenYmd="
                            + kktTyenYmd);
        }
        return MAX_DATE;
    }
