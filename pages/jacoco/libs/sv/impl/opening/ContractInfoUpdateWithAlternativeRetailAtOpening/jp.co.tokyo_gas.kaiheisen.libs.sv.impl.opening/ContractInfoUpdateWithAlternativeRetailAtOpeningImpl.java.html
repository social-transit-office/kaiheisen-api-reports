    /**
     * 概要: 開栓時代替ありの小売契約情報を登録し、確定値を返却する。
     * 主な仕様:
     * - 設計書どおり 6 件の DA insert を順番に実行する。
     * - 報告システム種別の復号値を監査コンテキストの kinoId に設定する。
     * - 戻り値は入力値・定数値・固定値から組み立てる。
     * - 確定適用終了年月日は source 入力互換のため受け取るが、各 DA の終了日は補助シートどおり最大日付固定とする。
     */
    @Override
    public ContractInfoUpdateWithAlternativeRetailAtOpeningOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            LocalDate kktTyenYmd,
            LocalDate kktTkysYmd,
            Long kktSvkyNo,
            Long kktSihKiyNo,
            String kktSqtiKiyNo,
            Long kktSikKrno,
            Long kktSyoKyoTrno,
            String kktSvckyaGasYtCd,
            String kktSziKziKbn) {
        requireRequestContext();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        // source 設計の入力互換のため kktTyenYmd は受け取るが、各 DA 補助シートでは終了日は最大日付固定。
        LocalDate applicableEndDate = MAX_DATE;
        SvmstkrWriteRecord svmstkrWriteRecord = new SvmstkrWriteRecord(
                gasMtstNo,
                kktSvkyNo,
                INITIAL_SKNO,
                nonComprehensive,
                Boolean.FALSE,
                relatedValid,
                kktTkysYmd,
                applicableEndDate,
                null);
        SvckyaWriteRecord svckyaWriteRecord = new SvckyaWriteRecord(
                kktSvkyNo,
                INITIAL_SKNO,
                INITIAL_SKNO,
                INITIAL_SKNO,
                serviceContractStatusUnderContract,
                serviceContractStartReasonNew,
                contractTypeApplicableReasonNewContract,
                null,
                kktTkysYmd,
                applicableEndDate,
                kktTkysYmd,
                applicableEndDate,
                kktTkysYmd,
                applicableEndDate,
                null,
                kktTkysYmd,
                kktTkysYmd,
                kktTkysYmd,
                kasgYmd,
                kktSqtiKiyNo,
                kktSyoKyoTrno,
                kktSyoKyoTrno,
                Boolean.FALSE,
                gasProduct,
                generalContract,
                kktSvckyaGasYtCd,
                kktSziKziKbn,
                Boolean.FALSE,
                null,
                contractManagementDepartmentGeneral,
                otherThanCompanyUse,
                INITIAL_OTHER_PRODUCT_CONTRACT_ID,
                null);
        SvsihkrWriteRecord svsihkrWriteRecord = new SvsihkrWriteRecord(
                kktSihKiyNo,
                kktSvkyNo,
                INITIAL_SKNO,
                kktTkysYmd,
                applicableEndDate,
                paymentContractStatusApplying,
                INITIAL_REFERENCE_SERVICE_CONTRACT_NO,
                null,
                null,
                INITIAL_GROUP_NO,
                singleBillingPartyTotal,
                Boolean.TRUE,
                null,
                immediateBilling,
                gasProduct);
        SihakyWriteRecord sihakyWriteRecord = new SihakyWriteRecord(
                kktSihKiyNo,
                INITIAL_SKNO,
                kktTkysYmd,
                applicableEndDate,
                paymentContractStatusApplying,
                paymentMethod,
                null,
                null,
                nextMonthCarryOver,
                null,
                INITIAL_BILL_CYCLE_NO,
                kktSikKrno,
                null,
                automaticCancellationNone,
                null,
                kktSyoKyoTrno,
                Boolean.FALSE,
                INITIAL_BILL_DISPLAY_CONTROL_NO);
        HrikmiWriteRecord hrikmiWriteRecord = new HrikmiWriteRecord(
                kktSihKiyNo,
                INITIAL_SKNO,
                kktTkysYmd,
                applicableEndDate,
                paymentFormShippingGeneral,
                paymentFormStandard,
                null,
                null,
                null,
                null,
                null,
                null);
        SihtkskWriteRecord sihtkskWriteRecord = new SihtkskWriteRecord(
                kktSihKiyNo,
                normalReminder,
                batchIssueTarget);

        svmstkrDataAccess.insert(svmstkrWriteRecord, auditContext);

        svckyaDataAccess.insert(svckyaWriteRecord, auditContext);

        svsihkrDataAccess.insert(svsihkrWriteRecord, auditContext);

        sihakyDataAccess.insert(sihakyWriteRecord, auditContext);

        hrikmiDataAccess.insert(hrikmiWriteRecord, auditContext);

        sihtkskDataAccess.insert(sihtkskWriteRecord, auditContext);

        return new ContractInfoUpdateWithAlternativeRetailAtOpeningOutput(
                toSvckyaKktInf(svckyaWriteRecord),
                toSvsihkrKktInf(svsihkrWriteRecord),
                toSihakyKktInf(sihakyWriteRecord));
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     * 主な仕様:
     * - correlationId、hukkSysShbt、systemDatetime の存在を確認する。
     * - 欠落がある場合は最初に検出した項目で処理を中断する。
     */
    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: RequestContext の報告システム種別を監査用機能 ID に復号する。
     * 主な仕様:
     * - hukkSysShbt は requireRequestContext で検証済みである前提で取得する。
     * - 開閉栓定数サービスを用いて名称へ変換する。
     * - 定数定義が未解決の場合は定数サービス側の例外をそのまま伝播する。
     *
     * @return 監査用の最終更新起点機能ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt().get());
    }
