package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;

/**
 * SV-227（開栓時契約情報更新処理（代替なし）（導管））のサービス実装。
 */
public class ContractInfoUpdateNoAlternativePipelineAtOpeningImpl
        implements ContractInfoUpdateNoAlternativePipelineAtOpening {

    static final String SERVICE_USER_ID = "KHSS227";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private static final String DEEMED_METER_READING_REQUIRED = "1";

    private static final Integer DEEMED_METER_READING_SEQUENCE = 0;

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final DeemedMeterReadingRegistration deemedMeterReadingRegistration;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String invalidTransportationServiceContractStatus;

    private final String invalidTransportationServiceContractRelatedStatus;

    private final String underContractStatus;

    private final String applyingRelationStatus;

    private final String previousContractContinuationReason;

    private final String consignmentContractType;

    private final String tokyoGasRetailerCode;

    private final String originalGasMeterStatus;

    private final String closedGasMeterStatus;

    /**
     * 依存オブジェクトと必須定数を初期化する。
     *
     * @param tkssvkyDataAccess 託送サービス契約データアクセス
     * @param tskgmstDataAccess 託送サービス契約・ガスメーター設置場所関連データアクセス
     * @param deemedMeterReadingRegistration みなし検針登録サービス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public ContractInfoUpdateNoAlternativePipelineAtOpeningImpl(
            final TkssvkyDataAccess tkssvkyDataAccess,
            final TskgmstDataAccess tskgmstDataAccess,
            final DeemedMeterReadingRegistration deemedMeterReadingRegistration,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.deemedMeterReadingRegistration =
                Objects.requireNonNull(deemedMeterReadingRegistration, "deemedMeterReadingRegistration");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.invalidTransportationServiceContractStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.INVALID);
        this.invalidTransportationServiceContractRelatedStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.INVALID);
        this.underContractStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.applyingRelationStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.previousContractContinuationReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType
                                .CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.consignmentContractType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ConsignmentContractType.TYPE,
                        ConstantLookupKeys.Cirius.ConsignmentContractType.CONSIGNMENT_CONTRACT);
        this.tokyoGasRetailerCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.originalGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.closedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
    }

    /**
     * 開栓時の導管契約情報を更新する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kktTksuSvkyNo 確定託送サービス契約番号
     * @param kktTkysYmd 確定適用開始年月日
     * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
     * @param kktSvkyNo 確定サービス契約番号
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param snyInfList 専用メーター情報リスト
     */
    @Override
    public void execute(
            final Long gasMtstNo,
            final Long kktTksuSvkyNo,
            final LocalDate kktTkysYmd,
            final Long kktSyoKyoTrno,
            final Long kktSvkyNo,
            final String ckrtGmscJtkb,
            final List<SnyInf> snyInfList) {
        requireCorrelationId();
        requireSystemDatetime();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        registerContract(gasMtstNo, kktTksuSvkyNo, kktTkysYmd, kktSyoKyoTrno, auditContext);

        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return;
        }

        for (SnyInf dedicatedMeterInfo : normalizeDedicatedMeterInfoList(snyInfList)) {
            registerDedicatedMeter(
                    dedicatedMeterInfo,
                    kktTkysYmd,
                    kktSyoKyoTrno,
                    kktSvkyNo,
                    auditContext);
        }
    }

    private void registerDedicatedMeter(
            final SnyInf dedicatedMeterInfo,
            final LocalDate kktTkysYmd,
            final Long kktSyoKyoTrno,
            final Long kktSvkyNo,
            final AuditContext auditContext) {
        Long dedicatedGasMtstNo = dedicatedMeterInfo.gasMtstNo();
        Long dedicatedTksuSvkyNo = dedicatedMeterInfo.kktTksuSvkyNo();
        registerContract(
                dedicatedGasMtstNo,
                dedicatedTksuSvkyNo,
                kktTkysYmd,
                kktSyoKyoTrno,
                auditContext);
        deemedMeterReadingRegistration.execute(
                DEEMED_METER_READING_REQUIRED,
                kktTkysYmd,
                dedicatedGasMtstNo,
                DEEMED_METER_READING_SEQUENCE,
                kktSvkyNo,
                closedGasMeterStatus);
    }

    private void registerContract(
            final Long gasMtstNo,
            final Long tksuSvkyNo,
            final LocalDate tkysYmd,
            final Long syoKyoTrno,
            final AuditContext auditContext) {
        TkssvkyRecord currentTransportContract = requireTransportContract(tksuSvkyNo);
        tkssvkyDataAccess.insert(
                buildTkssvkyWriteRecord(currentTransportContract, tkysYmd, syoKyoTrno),
                auditContext);

        TskgmstRecord currentTransportRelation = requireTransportRelation(gasMtstNo, tksuSvkyNo);
        tskgmstDataAccess.insert(
                buildTskgmstWriteRecord(currentTransportRelation, tkysYmd),
                auditContext);
    }

    private TkssvkyRecord requireTransportContract(final Long tksuSvkyNo) {
        return tkssvkyDataAccess
                .findByTksuSvkyNoAndTksuSvkyJtkb(tksuSvkyNo, invalidTransportationServiceContractStatus)
                .orElseThrow(() -> createTkssvkyNotFoundException(tksuSvkyNo));
    }

    private TskgmstRecord requireTransportRelation(
            final Long gasMtstNo,
            final Long tksuSvkyNo) {
        return tskgmstDataAccess
                .findByGasMtstNoAndTksuSvkyNoAndTksuSvkkJtkb(
                        gasMtstNo,
                        tksuSvkyNo,
                        invalidTransportationServiceContractRelatedStatus)
                .orElseThrow(() -> createTskgmstNotFoundException(gasMtstNo, tksuSvkyNo));
    }

    private TkssvkyWriteRecord buildTkssvkyWriteRecord(
            final TkssvkyRecord currentTransportContract,
            final LocalDate tkysYmd,
            final Long syoKyoTrno) {
        return new TkssvkyWriteRecord(
                currentTransportContract.tksuSvkyNo(),
                currentTransportContract.skno() + 1,
                underContractStatus,
                previousContractContinuationReason,
                null,
                tkysYmd,
                MAX_DATE,
                currentTransportContract.hesnYtiYmd(),
                consignmentContractType,
                tokyoGasRetailerCode,
                Boolean.TRUE,
                syoKyoTrno);
    }

    private TskgmstWriteRecord buildTskgmstWriteRecord(
            final TskgmstRecord currentTransportRelation,
            final LocalDate tkysYmd) {
        return new TskgmstWriteRecord(
                currentTransportRelation.gasMtstNo(),
                currentTransportRelation.tksuSvkyNo(),
                currentTransportRelation.skno() + 1,
                applyingRelationStatus,
                tkysYmd,
                MAX_DATE);
    }

    private List<SnyInf> normalizeDedicatedMeterInfoList(final List<SnyInf> snyInfList) {
        return snyInfList == null ? List.of() : snyInfList;
    }

    private DataInconsistencyException createTkssvkyNotFoundException(final Long tksuSvkyNo) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                TkssvkyDataAccess.CallSpecs.FIND_BY_TKSU_SVKY_NO_AND_TKSU_SVKY_JTKB.identifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", tksuSvkyNo)
                        .put("tksuSvkyJtkb", invalidTransportationServiceContractStatus));
    }

    private DataInconsistencyException createTskgmstNotFoundException(
            final Long gasMtstNo,
            final Long tksuSvkyNo) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                TskgmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_TKSU_SVKK_JTKB.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("tksuSvkyNo", tksuSvkyNo)
                        .put("tksuSvkkJtkb", invalidTransportationServiceContractRelatedStatus));
    }

    private String requireCorrelationId() {
        String correlationId = requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        if (correlationId.isBlank()) {
            throw new IllegalStateException("RequestContext.correlationId is required");
        }
        return correlationId;
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        if (hukkSysShbt.isBlank()) {
            throw new IllegalStateException("RequestContext.hukkSysShbt is required");
        }
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null || decodedOriginFunctionId.isBlank()) {
            throw new IllegalStateException("Origin function id constant is required");
        }
        return decodedOriginFunctionId;
    }
}
