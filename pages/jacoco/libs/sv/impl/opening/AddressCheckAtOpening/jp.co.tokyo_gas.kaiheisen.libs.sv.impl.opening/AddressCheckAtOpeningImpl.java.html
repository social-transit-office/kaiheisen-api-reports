package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.AddressCheckAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.GmtsetdSystemDateInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.JusyoKsenjInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TsigInf;

/** 開栓時住所チェックサービス実装。 */
public class AddressCheckAtOpeningImpl implements AddressCheckAtOpening {

    private static final String ZERO_FILLED_KANA3 = "00000";

    /** 開栓時住所と比較対象住所を照合し、住所エラーに該当する場合は業務例外を送出する。 */
    @Override
    public void execute(
            Boolean ksenskJuteiFlg,
            JusyoKsenjInf jusyoKsenjInf,
            GmtsetdSystemDateInf gmtsetdSystemDateInf,
            TsigInf tsigInf) {
        if (jusyoKsenjInf.gkkCd() == null || jusyoKsenjInf.gkkCd().isEmpty()) {
            return;
        }

        if (!Boolean.TRUE.equals(ksenskJuteiFlg)) {
            validateAddressMatch(
                    jusyoKsenjInf,
                    gmtsetdSystemDateInf.szcGkkCd(),
                    gmtsetdSystemDateInf.szcMchCd(),
                    gmtsetdSystemDateInf.cbgSuj1(),
                    gmtsetdSystemDateInf.cbgKn2());
        } else {
            validateAddressMatch(
                    jusyoKsenjInf,
                    tsigInf.tsigGkkCd(),
                    tsigInf.tsigMchCd(),
                    tsigInf.tsigCbgSuj1(),
                    tsigInf.tsigCbgKn2());
        }

    }

    /** 開栓時住所と比較対象住所の一致を検証し、不一致時は業務例外を送出する。 */
    private static void validateAddressMatch(
            JusyoKsenjInf jusyoKsenjInf,
            String targetGkkCd,
            String targetMchCd,
            String targetCbgSuj1,
            String targetCbgKn2) {
        boolean hasMismatch = !Objects.equals(jusyoKsenjInf.gkkCd(), targetGkkCd)
                || !Objects.equals(jusyoKsenjInf.mchCd(), targetMchCd)
                || !Objects.equals(jusyoKsenjInf.cbgSuj1(), targetCbgSuj1)
                || !Objects.equals(jusyoKsenjInf.cbgKn2(), targetCbgKn2)
                || ZERO_FILLED_KANA3.equals(jusyoKsenjInf.cbgKn3());
        if (hasMismatch) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100014);
        }
    }
}
