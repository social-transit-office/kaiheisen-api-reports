package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.AddressEditAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.AddressEditAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.GmtsetdSystemDateInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.LivalitInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TsigInf;

/**
 * 概要: SV-216（開栓時住所編集）の実装。
 * 主な仕様:
 * - 訂正後情報.丁目－番地－号数字１が null でも空文字でもない場合は訂正後情報を返す。
 * - それ以外で LIVALIT連携情報.丁目－番地－号数字１が null でも空文字でもない場合は LIVALIT連携情報を返す。
 * - 上記以外はガスメーター設置場所＜導管＞情報（システム日時点）を返す。
 * - 返却値は選択された入力ソースの 7 項目をそのまま出力 DTO へ移送する。
 * 既知の制約:
 * - public メソッド引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 */
public final class AddressEditAtOpeningImpl implements AddressEditAtOpening {

    /**
     * 概要: 入力住所情報の優先順位に応じて確定住所情報を返却する。
     * 主な仕様:
     * - 訂正後情報.丁目－番地－号数字１が設定されていれば訂正後情報を返す。
     * - そうでない場合に LIVALIT連携情報.丁目－番地－号数字１が設定されていれば LIVALIT連携情報を返す。
     * - 両方とも未設定ならガスメーター設置場所＜導管＞情報（システム日時点）を返す。
     * 既知の制約:
     * - 返却元として選択された入力モデルが null の場合は、そのまま null 参照例外となる。
     *
     * @param livalitInf LIVALIT連携情報
     * @param tsigInf 訂正後情報
     * @param gmtsetdSystemDateInf ガスメーター設置場所＜導管＞情報（システム日時点）
     * @return 確定住所情報
     */
    @Override
    public AddressEditAtOpeningOutput execute(
            LivalitInf livalitInf,
            TsigInf tsigInf,
            GmtsetdSystemDateInf gmtsetdSystemDateInf) {
        if (hasText(tsigInf.cbgSuj1())) {
            return toOutput(tsigInf);
        }
        if (hasText(livalitInf.cbgSuj1())) {
            return toOutput(livalitInf);
        }
        return toOutput(gmtsetdSystemDateInf);
    }

    /**
     * 概要: 文字列が null でも空文字でもないことを判定する。
     *
     * @param value 判定対象文字列
     * @return 判定対象文字列が null でも空文字でもない場合は true
     */
    private boolean hasText(String value) {
        return value != null && !value.isEmpty();
    }

    /**
     * 概要: LIVALIT連携情報を出力 DTO へ変換する。
     *
     * @param livalitInf LIVALIT連携情報
     * @return 確定住所情報
     */
    private AddressEditAtOpeningOutput toOutput(LivalitInf livalitInf) {
        return new AddressEditAtOpeningOutput(
                livalitInf.cbgSuj1(),
                livalitInf.cbgKn2(),
                livalitInf.cbgKn3(),
                livalitInf.cbgKn4(),
                livalitInf.tunoKn(),
                livalitInf.hynKn(),
                livalitInf.htuFlg());
    }

    /**
     * 概要: 訂正後情報を出力 DTO へ変換する。
     *
     * @param tsigInf 訂正後情報
     * @return 確定住所情報
     */
    private AddressEditAtOpeningOutput toOutput(TsigInf tsigInf) {
        return new AddressEditAtOpeningOutput(
                tsigInf.cbgSuj1(),
                tsigInf.cbgKn2(),
                tsigInf.cbgKn3(),
                tsigInf.cbgKn4(),
                tsigInf.tunoKn(),
                tsigInf.hynKn(),
                tsigInf.htuFlg());
    }

    /**
     * 概要: ガスメーター設置場所＜導管＞情報（システム日時点）を出力 DTO へ変換する。
     *
     * @param gmtsetdSystemDateInf ガスメーター設置場所＜導管＞情報（システム日時点）
     * @return 確定住所情報
     */
    private AddressEditAtOpeningOutput toOutput(GmtsetdSystemDateInf gmtsetdSystemDateInf) {
        return new AddressEditAtOpeningOutput(
                gmtsetdSystemDateInf.cbgSuj1(),
                gmtsetdSystemDateInf.cbgKn2(),
                gmtsetdSystemDateInf.cbgKn3(),
                gmtsetdSystemDateInf.cbgKn4(),
                gmtsetdSystemDateInf.tunoKn(),
                gmtsetdSystemDateInf.hynKn(),
                gmtsetdSystemDateInf.htuFlg());
    }
}
