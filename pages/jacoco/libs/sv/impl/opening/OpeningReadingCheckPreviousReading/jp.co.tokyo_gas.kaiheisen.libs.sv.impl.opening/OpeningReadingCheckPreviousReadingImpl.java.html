package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReadingCheckPreviousReading;

/**
 * 概要: 開栓指示数チェック（前回指示数判定）サービス実装。
 * 主な仕様:
 * - 災害時閉栓以外の場合に、報告開栓指示数とガスメーター指示数の差分を判定する。
 * - 判定で使用する CIRIUS 定数はコンストラクタで fail-fast に解決する。
 * 既知の制約:
 * - RequestContext 値は本 SV の public 引数へ含めず、処理でも参照しない。
 */
public final class OpeningReadingCheckPreviousReadingImpl implements OpeningReadingCheckPreviousReading {

    private static final long ALLOWED_DIFFERENCE_MIN = -1L;

    private static final long ALLOWED_DIFFERENCE_MAX = 1L;

    private final String disasterClosureReason;

    /**
     * 概要: 判定に必要な CIRIUS 定数を初期化する。
     * 主な仕様:
     * - 閉栓事由種別 J0273 の災害時閉栓を必須解決して保持する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public OpeningReadingCheckPreviousReadingImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.disasterClosureReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
    }

    /**
     * 概要: 前回指示数との差分を判定する。
     * 主な仕様:
     * - 閉栓事由種別が災害時閉栓以外で、差分が -1 未満または 1 超の場合に開栓共通エラー 020100019 を送出する。
    * - 条件不成立時は何も返さずに正常終了する。
     * 既知の制約:
     * - Long 引数が null の場合は Java の unboxing により NullPointerException となる。
     * @param hesnJysb 閉栓事由種別
     * @param hkkKsenShjSu 報告開栓指示数
     * @param gmtShjSu ガスメーター指示数
     * @throws KaisenRuleException 差分が許容範囲外の場合
     */
    @Override
    public void execute(
            String hesnJysb,
            Long hkkKsenShjSu,
            Long gmtShjSu) {
        if (!disasterClosureReason.equals(hesnJysb)) {
            long difference = hkkKsenShjSu - gmtShjSu;
            if (difference < ALLOWED_DIFFERENCE_MIN || difference > ALLOWED_DIFFERENCE_MAX) {
                throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100019);
            }
        }

    }
}
