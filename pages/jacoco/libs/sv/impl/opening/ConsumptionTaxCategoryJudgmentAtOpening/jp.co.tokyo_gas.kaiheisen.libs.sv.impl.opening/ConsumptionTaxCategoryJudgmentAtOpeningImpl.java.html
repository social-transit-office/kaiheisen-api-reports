package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ConsumptionTaxCategoryJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ConsumptionTaxCategoryJudgmentAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.ConsumptionTaxCategoryJudgmentAtOpeningMtgInf;

/**
 * 概要: 開栓時消費税課税区分判定サービスを実装する。
 * 主な仕様: 直列専用通過メーターフラグを最優先で評価し、非直列時は代替判定結果フラグに応じて課税定数または入力値を返却する。
 * 既知の制約: 必須定数の不足はコンストラクタ初期化時点で fail-fast し、Boolean 引数が null の場合は false として扱う。
 */
public class ConsumptionTaxCategoryJudgmentAtOpeningImpl
        implements ConsumptionTaxCategoryJudgmentAtOpening {

    private final String taxation;

    /**
     * 概要: 開栓時消費税課税区分判定に使用する CIRIUS 定数を初期化する。
     * 主な仕様: 課税区分の課税値を `CiriusConstantService` から解決して保持する。
     * 既知の制約: 必須依存が null の場合と必須定数が取得できない場合は、コンストラクタ初期化時点で fail-fast する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ConsumptionTaxCategoryJudgmentAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.taxation = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ConsumptionTax.TYPE,
                ConstantLookupKeys.Cirius.ConsumptionTax.TAXATION);
    }

    /**
     * 概要: 開栓時の確定消費税課税区分を判定する。
     * 主な仕様: 直列専用通過メーターフラグを最優先で評価し、true の場合は元メーター情報から、false の場合は代替判定結果フラグに応じて課税定数または入力値から確定消費税課税区分を返却する。
     * 既知の制約: Boolean 引数が null の場合は false として扱い、`mtgInf` が null のまま直列専用通過分岐に入った場合は `NullPointerException` となる。
     *
     * @param digwrHntFlg 代替判定結果フラグ
     * @param snyTkmtFlg 直列専用通過メーターフラグ
     * @param sziKziKbn 消費税課税区分
     * @param mtgInf 元メーター情報
     * @return 確定消費税課税区分判定結果
     */
    @Override
    public ConsumptionTaxCategoryJudgmentAtOpeningOutput execute(
            Boolean digwrHntFlg,
            Boolean snyTkmtFlg,
            String sziKziKbn,
            ConsumptionTaxCategoryJudgmentAtOpeningMtgInf mtgInf) {
        boolean seriesDedicatedMeterFlag = Boolean.TRUE.equals(snyTkmtFlg);
        boolean alternativeJudgmentResult = Boolean.TRUE.equals(digwrHntFlg);

        if (seriesDedicatedMeterFlag) {
            return output(mtgInf.sziKziKbn());
        }

        if (alternativeJudgmentResult) {
            return output(taxation);
        }

        return output(sziKziKbn);
    }

    /**
     * 概要: 確定消費税課税区分の出力 DTO を生成する。
     * 主な仕様: 判定済みの消費税課税区分をそのまま出力 DTO に格納して返却する。
     * 既知の制約: 出力値の追加補正は行わない。
     *
     * @param kktSziKziKbn 確定消費税課税区分
     * @return 確定消費税課税区分判定結果
     */
    private static ConsumptionTaxCategoryJudgmentAtOpeningOutput output(String kktSziKziKbn) {
        return new ConsumptionTaxCategoryJudgmentAtOpeningOutput(kktSziKziKbn);
    }
}
