package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CustomerNumberAssignmentTargetJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CustomerNumberAssignmentTargetJudgmentAtOpeningOutput;

/**
 * 開栓時お客さま番号採番対象判定サービス実装。
 */
public class CustomerNumberAssignmentTargetJudgmentAtOpeningImpl
        implements CustomerNumberAssignmentTargetJudgmentAtOpening {

    private final String dedicatedGasMeter;

    private final String newlyOpendTapCode;

    private final String alternativeOpeningCode;

    private final String customerOpening;

    private Boolean cstNoSibntiFlg;

    private final List<String> targetCodes;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public CustomerNumberAssignmentTargetJudgmentAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(ciriusConstantService,
                "ciriusConstantService");
        this.dedicatedGasMeter = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.newlyOpendTapCode = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.NEWLY_OPENED_TAP);
        this.alternativeOpeningCode = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.ALTERNATIVE_OPENING);
        this.customerOpening = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CUSTOMER_OPENING);
        this.targetCodes = List.of(newlyOpendTapCode, alternativeOpeningCode, customerOpening);
    }

    /**
     * 概要: おなまえ訂正フラグ、ガスメーター設置場所状態区分、開栓事由種別によって判定を行う
     * 主な仕様:
     * - 引数のおなまえ訂正フラグがTRUE、または直列ガスメーター設置状態区分が専用ガスメーター、開栓事由種別が新設開栓、代替開栓、お客さま開栓のいずれかの場合
     *   お客さま番号採番対象フラグをTRUEで返却
     * - 上記以外の場合 お客さま番号採番対象フラグをFALSEで返却
     *
     * @param namTisiFlg おなまえ訂正フラグ
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param kasJysb 開栓事由種別
     */
    @Override
    public CustomerNumberAssignmentTargetJudgmentAtOpeningOutput execute(Boolean namTisiFlg, String ckrtGmscJtkb, String kasJysb) {
        // 開栓時お客さま番号採番対象判定
        if (namTisiFlg
                || (!this.dedicatedGasMeter.equals(ckrtGmscJtkb) && targetCodes.contains(kasJysb))) {
            cstNoSibntiFlg = true;
        } else {
            cstNoSibntiFlg = false;
        }
        return new CustomerNumberAssignmentTargetJudgmentAtOpeningOutput(cstNoSibntiFlg);
    }
}
