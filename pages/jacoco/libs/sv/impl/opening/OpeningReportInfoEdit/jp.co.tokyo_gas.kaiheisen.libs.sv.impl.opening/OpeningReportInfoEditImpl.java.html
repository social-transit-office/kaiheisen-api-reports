package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReportInfoEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReportInfoEditOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KiycstKktInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.MtgInf;

/** 開栓報告情報編集サービス実装。 */
public final class OpeningReportInfoEditImpl implements OpeningReportInfoEdit {

    private final RequestContext requestContext;

    private final String tgRetail;

    private final String dedicatedGasMeter;

    private final String newServiceContractStartReason;

    private final String continuationOfPreviousContract;

    private final String openingReported;

    private final String openingFollowed;

    private final String openingRegistration;

    private final String openingSystemProcessing;

    private final String openingErrorFollowUp;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param requestContext リクエストコンテキスト
     */
    public OpeningReportInfoEditImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService,
            RequestContext requestContext) {
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.tgRetail =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                        ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TG_RETAIL);
        this.dedicatedGasMeter =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.newServiceContractStartReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractStartReasonType.NEW);
        this.continuationOfPreviousContract =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractStartReasonType.CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.openingReported =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.OPENING_REPORTED);
        this.openingFollowed =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReportStatus.OPENING_FOLLOWED);
        this.openingRegistration =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.OPENING_REGISTRATION);
        this.openingSystemProcessing =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.OPENING_SYSTEM_PROCESSING);
        this.openingErrorFollowUp =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.OPENING_ERROR_FOLLOW_UP_CCBG50);
    }

    @Override
    public OpeningReportInfoEditOutput execute(
            String sgyAnknNo,
            LocalDate kasHukkYmd,
            LocalDate kasgYmd,
            Long gasMtstNo,
            String gmb,
            String kasJysb,
            String tgTkurKbn,
            String sgytNwkCd,
            String ctnmKn1,
            String ctnmKj1,
            Long kktSyoKyoTrno,
            Long kktSvkyNo,
            Long kktSihKiyNo,
            String ckrtGmscJtkb,
            Boolean snyTkmtFlg,
            String svkySttJysb,
            Boolean digwrHntFlg,
            MtgInf mtgInf,
            Boolean suguHtkFlg,
            Boolean kiyUmFlg,
            Boolean syoriHnt4XFlg,
            KiycstKktInf kiycstKktInf,
            String isrrGymCd,
            String cktkCd,
            String ktnCd) {
        String requestHukkSysShbt =
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        LocalDate systemDate =
                requestContext.getSystemDatetime()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                        .toLocalDate();
        boolean tgRetailJudgmentResult = isTgRetail(tgTkurKbn);
        String resolvedServiceContractStartReason =
                resolveServiceContractStartReason(
                        svkySttJysb,
                        digwrHntFlg,
                        ckrtGmscJtkb,
                        snyTkmtFlg,
                        mtgInf,
                        suguHtkFlg,
                        kiyUmFlg,
                        syoriHnt4XFlg);

        String outputCtnmKn1 = null;
        String outputCtnmKj1 = null;
        String outputOpeningReportStatus = null;
        Long correctedGasMeterInstallationLocationNo = null;
        Long correctedOpeningDestinationCustomerRegistrationNo = null;
        String correctedOpeningReason = null;
        String correctedServiceContractStartReason = null;
        String correctedCustomerNameKana1 = null;
        String correctedCustomerNameKanji1 = null;

        if (isSystemProcessingOrErrorFollowUp(isrrGymCd)) {
            outputCtnmKn1 = ctnmKn1;
            outputCtnmKj1 = ctnmKj1;
            outputOpeningReportStatus = openingFollowed;
            correctedGasMeterInstallationLocationNo = gasMtstNo;
            correctedOpeningDestinationCustomerRegistrationNo = kktSyoKyoTrno;
            correctedOpeningReason = kasJysb;
            correctedServiceContractStartReason = resolvedServiceContractStartReason;
            correctedCustomerNameKana1 = kiycstKktInf == null ? null : kiycstKktInf.ctnmKn1();
            correctedCustomerNameKanji1 = kiycstKktInf == null ? null : kiycstKktInf.ctnmKj1();
        } else if (openingRegistration.equals(isrrGymCd)) {
            outputCtnmKn1 = kiycstKktInf == null ? null : kiycstKktInf.ctnmKn1();
            outputCtnmKj1 = kiycstKktInf == null ? null : kiycstKktInf.ctnmKj1();
            outputOpeningReportStatus = openingReported;
        }

        return new OpeningReportInfoEditOutput(
                gmb,
                kasgYmd,
                kasHukkYmd,
                requestHukkSysShbt,
                sgyAnknNo,
                outputOpeningReportStatus,
                systemDate,
                kasgYmd,
                sgytNwkCd,
                cktkCd,
                ktnCd,
                gasMtstNo,
                kktSyoKyoTrno,
                kktSvkyNo,
                kktSihKiyNo,
                tgRetailJudgmentResult,
                kasJysb,
                resolvedServiceContractStartReason,
                outputCtnmKn1,
                outputCtnmKj1,
                correctedGasMeterInstallationLocationNo,
                correctedOpeningDestinationCustomerRegistrationNo,
                correctedOpeningReason,
                correctedServiceContractStartReason,
                correctedCustomerNameKana1,
                correctedCustomerNameKanji1);
    }

    private boolean isTgRetail(String tgTkurKbn) {
        return tgRetail.equals(tgTkurKbn);
    }

    private String resolveServiceContractStartReason(
            String svkySttJysb,
            Boolean digwrHntFlg,
            String ckrtGmscJtkb,
            Boolean snyTkmtFlg,
            MtgInf mtgInf,
            Boolean suguHtkFlg,
            Boolean kiyUmFlg,
            Boolean syoriHnt4XFlg) {
        if (Boolean.TRUE.equals(digwrHntFlg)) {
            return newServiceContractStartReason;
        }
        if (dedicatedGasMeter.equals(ckrtGmscJtkb) && Boolean.TRUE.equals(snyTkmtFlg)) {
            return mtgInf == null ? null : mtgInf.svkySttJysb();
        }
        if (Boolean.TRUE.equals(suguHtkFlg) && Boolean.TRUE.equals(kiyUmFlg)) {
            return svkySttJysb;
        }
        if (Boolean.TRUE.equals(syoriHnt4XFlg)) {
            return continuationOfPreviousContract;
        }
        return svkySttJysb;
    }

    private boolean isSystemProcessingOrErrorFollowUp(String isrrGymCd) {
        return openingSystemProcessing.equals(isrrGymCd) || openingErrorFollowUp.equals(isrrGymCd);
    }
}
