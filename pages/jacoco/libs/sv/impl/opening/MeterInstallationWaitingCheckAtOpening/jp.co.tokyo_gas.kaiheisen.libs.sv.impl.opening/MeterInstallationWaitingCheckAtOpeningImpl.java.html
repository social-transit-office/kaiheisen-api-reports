package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.MeterInstallationWaitingCheckAtOpening;

/**
 * 概要: 開栓時メーター取付待ちチェックサービス実装。
 * 主な仕様:
 * - ガスメーター設置場所＜導管＞有無フラグ、当初ガスメーター取付年月日、設備状態区分を順次判定する。
 * - 判定に必要な CIRIUS 定数はコンストラクタで fail-fast に解決する。
 * 既知の制約:
 * - RequestContext 値は本 SV の public 引数へ含めず、処理でも参照しない。
 */
public final class MeterInstallationWaitingCheckAtOpeningImpl implements MeterInstallationWaitingCheckAtOpening {

    private final String meterNotInstalledStatus;

    private final String suspendedMeterRemovedStatus;

    /**
     * 概要: 判定に必要な CIRIUS 定数を初期化する。
     * 主な仕様:
     * - 供内管・ガスメーター設備状態区分の未取付状態と休止状態を解決する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public MeterInstallationWaitingCheckAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.meterNotInstalledStatus = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_NOT_INSTALLED_STATUS);
        this.suspendedMeterRemovedStatus = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.AT_REST_METER_REMOVED_STATE);
    }

    /**
     * 概要: 開栓時のメーター取付待ち判定を実行する。
     * 主な仕様:
        * - ガスメーター設置場所＜導管＞有無フラグが false の場合は開栓共通エラー
        *   020100012（KaisenRuleException）を送出する。
        * - 当初ガスメーター取付年月日が null の場合は開栓共通エラー
        *   020100012（KaisenRuleException）を送出する。
        * - 供内管・ガスメーター設備状態区分が未取付または休止の場合は開栓共通エラー
        *   020100012（KaisenRuleException）を送出する。
     * 既知の制約:
     * - Boolean 引数が null の場合は Java の unboxing により NullPointerException となる。
     * @param gmtsetdUmFlg ガスメーター設置場所＜導管＞有無フラグ
     * @param tusoGmtkYmd 当初ガスメーター取付年月日
     * @param kngmStbJtkb 供内管・ガスメーター設備状態区分
        * @throws KaisenRuleException メーター取付待ち条件に該当し、開栓共通エラーを送出する場合
     */
    @Override
     public void execute(
            Boolean gmtsetdUmFlg,
            LocalDate tusoGmtkYmd,
            String kngmStbJtkb) {
        if (!gmtsetdUmFlg) {
            throwMeterInstallationWaitingError();
        }
        if (tusoGmtkYmd == null) {
            throwMeterInstallationWaitingError();
        }
        if (isMeterInstallationWaitingStatus(kngmStbJtkb)) {
            throwMeterInstallationWaitingError();
        }
    }

    /**
     * 概要: メーター取付待ち判定対象の設備状態区分かを判定する。
     * 主な仕様:
     * - メーター未取付状態または休止・メーター取外状態を対象とする。
     * 既知の制約:
     * - 入力が null の場合は false として扱う。
     * @param kngmStbJtkb 供内管・ガスメーター設備状態区分
     * @return 対象状態の場合は true
     */
    private boolean isMeterInstallationWaitingStatus(String kngmStbJtkb) {
        return meterNotInstalledStatus.equals(kngmStbJtkb)
                || suspendedMeterRemovedStatus.equals(kngmStbJtkb);
    }

    /**
     * 概要: メーター取付待ちの開栓共通エラーを送出する。
     * 主な仕様:
     * - 開栓共通エラー 020100012 を KaisenRuleException として送出する。
     * 既知の制約:
     * - 例外変換や追加情報付与は行わない。
     */
    private void throwMeterInstallationWaitingError() {
        throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100012);
    }
}
