package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jakarta.inject.Inject;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransportServiceContractNumberConfirmationAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.TransportServiceContractNumberConfirmationAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TransportServiceContractConfirmationDedicatedMeterInfo;

/**
 * 開栓時託送サービス契約番号確定サービス実装。
 */
public final class TransportServiceContractNumberConfirmationAtOpeningImpl
        implements TransportServiceContractNumberConfirmationAtOpening {

    private final NumberingAllInOneSupport numberingAllInOneSupport;

    private final String originalGasMeterStatus;

    /**
     * コンストラクタ。
     *
     * @param numberingAllInOneSupport 他-048 採番ゲートウェイ
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public TransportServiceContractNumberConfirmationAtOpeningImpl(
            NumberingAllInOneSupport numberingAllInOneSupport,
            CiriusConstantService ciriusConstantService) {
        this.numberingAllInOneSupport = Objects.requireNonNull(numberingAllInOneSupport, "numberingAllInOneSupport");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
    }

    /**
     * 代替判定結果フラグと直列ガスメーター設置状態区分に応じて確定託送サービス契約番号を返却する。
     * 代替ありの場合は主契約番号を採番し、元ガスメーターの場合は専用メーター件数分も採番する。
     * 代替なしの場合は入力の託送サービス契約番号を採用し、元ガスメーター時のみ専用メーター情報を確定番号へ移送する。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param digwrHntFlg 代替判定結果フラグ
     * @param tksuSvkyNo 託送サービス契約番号
     * @param snyInfList 専用メーター情報リスト
     * @return 確定託送サービス契約番号と専用メーター情報
     */
    @Override
    public TransportServiceContractNumberConfirmationAtOpeningOutput execute(
            String ckrtGmscJtkb,
            Boolean digwrHntFlg,
            Long tksuSvkyNo,
            List<TransportServiceContractConfirmationDedicatedMeterInfo> snyInfList) {
        boolean alternative = Boolean.TRUE.equals(digwrHntFlg);
        Long confirmedTransportServiceContractNumber = alternative
                ? numberingAllInOneSupport.issueTransportServiceContractNumber()
                : tksuSvkyNo;

        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return new TransportServiceContractNumberConfirmationAtOpeningOutput(
                    confirmedTransportServiceContractNumber,
                    List.of());
        }

        List<TransportServiceContractConfirmationDedicatedMeterInfo> normalizedSnyInfList =
                normalizeSnyInfList(snyInfList);
        List<TransportServiceContractConfirmationDedicatedMeterInfo> confirmedDedicatedMeters = alternative
                ? issueDedicatedMeterNumbers(normalizedSnyInfList)
                : mapDedicatedMeterNumbers(normalizedSnyInfList);

        return new TransportServiceContractNumberConfirmationAtOpeningOutput(
                confirmedTransportServiceContractNumber,
                confirmedDedicatedMeters);
    }

    /**
     * 専用メーター情報リストの件数分だけ託送サービス契約番号を採番し、確定結果へ詰め替える。
     *
     * @param snyInfList 専用メーター情報リスト
     * @return 採番後の専用メーター情報リスト
     */
    private List<TransportServiceContractConfirmationDedicatedMeterInfo> issueDedicatedMeterNumbers(
            List<TransportServiceContractConfirmationDedicatedMeterInfo> snyInfList) {
        List<TransportServiceContractConfirmationDedicatedMeterInfo> confirmedDedicatedMeters = new ArrayList<>();
        for (TransportServiceContractConfirmationDedicatedMeterInfo snyInf : snyInfList) {
            confirmedDedicatedMeters.add(
                    new TransportServiceContractConfirmationDedicatedMeterInfo(
                            null,
                            numberingAllInOneSupport.issueTransportServiceContractNumber()));
        }
        return List.copyOf(confirmedDedicatedMeters);
    }

    /**
     * 入力の専用メーター情報リストから託送サービス契約番号を確定結果へ移送する。
     *
     * @param snyInfList 専用メーター情報リスト
     * @return 確定後の専用メーター情報リスト
     */
    private List<TransportServiceContractConfirmationDedicatedMeterInfo> mapDedicatedMeterNumbers(
            List<TransportServiceContractConfirmationDedicatedMeterInfo> snyInfList) {
        List<TransportServiceContractConfirmationDedicatedMeterInfo> confirmedDedicatedMeters = new ArrayList<>();
        for (TransportServiceContractConfirmationDedicatedMeterInfo snyInf : snyInfList) {
            confirmedDedicatedMeters.add(new TransportServiceContractConfirmationDedicatedMeterInfo(
                    null,
                    snyInf == null ? null : snyInf.tksuSvkyNo()));
        }
        return List.copyOf(confirmedDedicatedMeters);
    }

    /**
     * 専用メーター情報リストが null の場合は空リストへ正規化する。
     *
     * @param snyInfList 専用メーター情報リスト
     * @return 正規化後の専用メーター情報リスト
     */
    private List<TransportServiceContractConfirmationDedicatedMeterInfo> normalizeSnyInfList(
            List<TransportServiceContractConfirmationDedicatedMeterInfo> snyInfList) {
        return snyInfList == null ? List.of() : snyInfList;
    }
}
