package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisenkannrenji_idoujouhou_touroku.KaisenkannrenjiIdoujouhouTourokuCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisenkannrenji_idoujouhou_touroku.KaisenkannrenjiIdoujouhouTourokuRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.BillingCollectionPlatformApiIntegrationForOpening;

/**
 * 概要: SV-028（請求回収基盤API連携（開栓））の実装。
 * 主な仕様:
 * - 3 つの入力モデルを他-044の 52 項目へ編集し、請求回収基盤（開栓）-API を 1 回呼び出す。
 * - 日付は YYYYMMDD、年月は YYYYMM の数値へ変換する。
 * - 有無フラグと合算可否フラグは CIRIUS 定数を検証済みコードへ変換する。
 * 既知の制約:
 * - `他-044呼出しパラメータ.tsv` の No.27 以降は best-effort で `sktsKykTrirInf` を正として扱う。
 */
public final class BillingCollectionPlatformApiIntegrationForOpeningImpl
        implements BillingCollectionPlatformApiIntegrationForOpening {

    private static final DateTimeFormatter BASIC_DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private final KaisenkannrenjiIdoujouhouTourokuCiriusAdapter ciriusAdapter;

    private final CiriusConstantService ciriusConstantService;

    /**
     * 概要: SV-028 の依存を保持して初期化する。
     * 主な仕様:
     * - 外部API adapter と CIRIUS 定数取得サービスを必須依存として保持する。
     * 既知の制約:
     * - 依存の null は許容しない。
     *
     * @param ciriusAdapter 請求回収基盤（開栓）-API adapter
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public BillingCollectionPlatformApiIntegrationForOpeningImpl(
            KaisenkannrenjiIdoujouhouTourokuCiriusAdapter ciriusAdapter,
            CiriusConstantService ciriusConstantService) {
        this.ciriusAdapter = Objects.requireNonNull(ciriusAdapter, "ciriusAdapter");
        this.ciriusConstantService = Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
    }

    /**
     * 概要: SV-028 を実行し、請求回収基盤（開栓）-API を呼び出す。
     * 主な仕様:
     * - 他-044の flat request へ編集した後、外部 API を 1 回呼び出す。
     * - RequestContext 項目は公開引数に含めず、この Service では利用しない。
     *
     * @param skkbRkySgjhInf 請求回収基盤連携用制御情報
     * @param sktsSihTrirInf 請求対象支払管理登録依頼情報
     * @param sktsKykTrirInf 請求対象契約管理登録依頼情報
     */
    @Override
    public void execute(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihTrirInf sktsSihTrirInf,
            SktsKykTrirInf sktsKykTrirInf) {
        ciriusAdapter.register(toRequest(skkbRkySgjhInf, sktsSihTrirInf, sktsKykTrirInf));
    }

    /**
     * 概要: SV 入力を請求回収基盤（開栓）-API（他-044）のリクエストへ変換する。
     *
     * 主な仕様:
     * - 各入力モデルのフィールドを外部API仕様に従い型・値域・変換ルールで編集する。
     * - 日付型（LocalDate）はYYYYMMDD数値、年/月型（YearMonth）はYYYYMM数値へ変換。
     * - 有無フラグ・合算可否フラグはCIRIUS定数へ変換し、存在検証を行う。
     * - best-effortで契約系項目は`sktsKykTrirInf`配下を正として扱う。
     *
     * 既知の制約:
     * - 入力値の妥当性検証は本メソッドでは行わない。
     * - 外部API仕様変更時は必ず変換ロジックの見直しが必要。
     *
     * @param skkbRkySgjhInf 請求回収基盤連携用制御情報
     * @param sktsSihTrirInf 請求対象支払管理登録依頼情報
     * @param sktsKykTrirInf 請求対象契約管理登録依頼情報
     * @return 外部 API リクエスト
     */
    private KaisenkannrenjiIdoujouhouTourokuRequest toRequest(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihTrirInf sktsSihTrirInf,
            SktsKykTrirInf sktsKykTrirInf) {
        return new KaisenkannrenjiIdoujouhouTourokuRequest(
                new KaisenkannrenjiIdoujouhouTourokuRequest.KaisenkannrenjiIdoujouhou(
                        skkbRkySgjhInf.skrkGrpKrno(),
                skkbRkySgjhInf.sikKrno(),
                skkbRkySgjhInf.sqtiKiyNo(),
                skkbRkySgjhInf.svkyNo(),
                skkbRkySgjhInf.svkySkno(),
                skkbRkySgjhInf.sihKiyNo(),
                skkbRkySgjhInf.sihKiySkno(),
                skkbRkySgjhInf.gasMtstNo(),
                toYmd(skkbRkySgjhInf.tkysYmd()),
                toYmd(skkbRkySgjhInf.khsYmd()),
                toPresenceFlag(skkbRkySgjhInf.syoTisiUmFlg()),
                sktsSihTrirInf.vtkzNo(),
                sktsSihTrirInf.vtstCd(),
                sktsSihTrirInf.irimCstrNo(),
                sktsSihTrirInf.kznKn(),
                sktsSihTrirInf.kznKj(),
                sktsSihTrirInf.shsyRrskShbt1(),
                sktsSihTrirInf.shsyRrskShbt2(),
                sktsSihTrirInf.shsyTelNo1(),
                sktsSihTrirInf.shsyTelNo2(),
                sktsSihTrirInf.shsyTelShbt1(),
                sktsSihTrirInf.shsyTelShbt2(),
                sktsSihTrirInf.isrrGymCd(),
                toYmd(sktsSihTrirInf.sikEndYmd()),
                toYmd(sktsSihTrirInf.sikSttYmd()),
                sktsSihTrirInf.nwjgCd(),
                sktsKykTrirInf.svkjyShbt(),
                toYmd(sktsKykTrirInf.svtkYmd()),
                sktsKykTrirInf.ikskGrpNo(),
                toYm(sktsKykTrirInf.ikskEndYm()),
                toYm(sktsKykTrirInf.ikskSttYm()),
                sktsKykTrirInf.syoRrskShbt1(),
                sktsKykTrirInf.syoRrskShbt2(),
                sktsKykTrirInf.syoTelNo1(),
                sktsKykTrirInf.syoTelNo2(),
                sktsKykTrirInf.syoTelShbt1(),
                sktsKykTrirInf.syoTelShbt2(),
                sktsKykTrirInf.uwhyCstuNo(),
                sktsKykTrirInf.tdssSbtShbt(),
                sktsKykTrirInf.snskSqtkNo(),
                sktsKykTrirInf.gsptShbt(),
                toSummabilityFlag(sktsKykTrirInf.gsnKhFlg()),
                sktsKykTrirInf.kiyShri(),
                toYmd(sktsKykTrirInf.kiyEndYmd()),
                toYmd(sktsKykTrirInf.kiySttYmd()),
                sktsKykTrirInf.rknTtkbNo(),
                sktsKykTrirInf.isrrGymCd(),
                sktsKykTrirInf.sikTmngShbt(),
                toYmd(sktsKykTrirInf.sikSttYmd()),
                sktsKykTrirInf.nwjgCd(),
                sktsKykTrirInf.sziKziKbn(),
                toYmd(sktsKykTrirInf.szirHtyoYmd())));
    }

    /**
     * 概要: 日付（LocalDate）をYYYYMMDD形式の数値へ変換する。
     *
     * 主な仕様:
     * - nullの場合はnullを返す。
     * - BASIC_ISO_DATE（yyyyMMdd）でフォーマットしint変換。
     *
     * 既知の制約:
     * - 日付範囲外・不正値は上位で検証すること。
     *
     * @param value 日付
     * @return YYYYMMDD 数値
     */
    private Integer toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        return Integer.valueOf(value.format(BASIC_DATE_FORMATTER));
    }

    /**
     * 概要: 年月（YearMonth）をYYYYMM形式の数値へ変換する。
     *
     * 主な仕様:
     * - nullの場合はnullを返す。
     * - YearMonth#toYmInt()で変換。
     *
     * 既知の制約:
     * - 年月範囲外・不正値は上位で検証すること。
     *
     * @param value 年月
     * @return YYYYMM 数値
     */
    private Integer toYm(YearMonth value) {
        if (value == null) {
            return null;
        }
        return value.toYmInt();
    }

    /**
     * 概要: Boolean値を有無フラグ（CIRIUS定数）へ変換する。
     *
     * 主な仕様:
     * - nullの場合はnullを返す。
     * - true→PresenceFlag.YES、false→PresenceFlag.NOTHINGへ変換。
     * - CiriusConstantService経由で定数存在検証を行う。
     *
     * 既知の制約:
     * - 定数未定義時はRequiredConstantResolverで例外発生。
     *
     * @param value Boolean 値
     * @return J0253 の定数コード
     */
    private String toPresenceFlag(Boolean value) {
        if (value == null) {
            return null;
        }
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                value
                        ? ConstantLookupKeys.Cirius.PresenceFlag.YES
                        : ConstantLookupKeys.Cirius.PresenceFlag.NOTHING);
    }

    /**
     * 概要: Boolean値を合算可否フラグ（CIRIUS定数）へ変換する。
     *
     * 主な仕様:
     * - nullの場合はnullを返す。
     * - true→SummabilityFlag.CAN_BE_COMBINED、false→SummabilityFlag.CANNOT_BE_COMBINEDへ変換。
     * - CiriusConstantService経由で定数存在検証を行う。
     *
     * 既知の制約:
     * - 定数未定義時はRequiredConstantResolverで例外発生。
     *
     * @param value Boolean 値
     * @return J0130 の定数コード
     */
    private String toSummabilityFlag(Boolean value) {
        if (value == null) {
            return null;
        }
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.SummabilityFlag.TYPE,
                value
                        ? ConstantLookupKeys.Cirius.SummabilityFlag.CAN_BE_COMBINED
                        : ConstantLookupKeys.Cirius.SummabilityFlag.CANNOT_BE_COMBINED);
    }
}
