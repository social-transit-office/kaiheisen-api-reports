package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckSwitchingCompletion;

/**
 * 概要: 開栓日チェック（スイッチング完了判定）サービス実装。
 * 主な仕様:
 * - スイッチング検針有無フラグが true の場合は開栓共通エラー 020100018 を送出する。
 * - それ以外の場合は何も返さずに正常終了する。
 * 既知の制約:
 * - 公開引数の null 妥当性検証は上位層責務とし、本クラスでは追加の防御判定を行わない。
 */
public final class OpeningDateCheckSwitchingCompletionImpl
        implements OpeningDateCheckSwitchingCompletion {

    /**
     * 概要: スイッチング完了判定を実行する。
     * 主な仕様:
     * - スイッチング検針有無フラグが true の場合は開栓共通エラーを送出する。
    * - 条件に該当しない場合は何も返さずに正常終了する。
     * 既知の制約:
     * - null 入力は上位層バリデーション済み前提とし、Boolean の自動アンボクシング結果に従う。
     *
     * @param swKnsUmFlg スイッチング検針有無フラグ
     * @throws KaisenRuleException スイッチング完了判定に該当する場合
     */
    @Override
    public void execute(final Boolean swKnsUmFlg) {
        if (swKnsUmFlg) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }
    }
}
