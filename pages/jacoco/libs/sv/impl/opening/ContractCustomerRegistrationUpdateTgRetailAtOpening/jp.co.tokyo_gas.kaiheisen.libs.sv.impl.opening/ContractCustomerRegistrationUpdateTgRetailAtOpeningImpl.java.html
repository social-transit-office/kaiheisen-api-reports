    public ContractCustomerRegistrationUpdateTgRetailAtOpeningImpl(

            KiycstkDataAccess kiycstkDataAccess,

    /**
     * 開栓時契約お客さま登録・更新（TG小売）を実行する。
     *
     * @param snyTkmtFlg 直列専用通過メーターフラグ
     * @param suguHtkFlg 総合判定結果フラグ
     * @param cstUpdPtnKb 3x更新パターン区分
     * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
     * @param confirmedContractCustomerInfo 契約お客さま情報（確定）
     * @param confirmedAddressInfo 住所情報（確定）
     * @param contractCustomerInfo 契約お客さま情報
     * @param gasMeterLocationInfo ガスメーター設置場所＜導管＞情報
     * @param closingCustomerInfo お客さま情報（閉栓時）
     * @return 契約お客さま情報（確定）と契約お客さま電話番号＜小売＞情報リスト
     */
    @Override
    public ContractCustomerRegistrationUpdateTgRetailAtOpeningOutput execute(
            Boolean snyTkmtFlg,
            Boolean suguHtkFlg,
            String cstUpdPtnKb,
            Long kktSyoKyoTrno,
            ConfirmedContractCustomerInfo confirmedContractCustomerInfo,
            ConfirmedPipelineContractCustomerInfo confirmedPipelineContractCustomerInfo,
            ConfirmedAddressInfo confirmedAddressInfo,
            ContractCustomerInfo contractCustomerInfo,
            GasMeterLocationInfo gasMeterLocationInfo,
            ClosingCustomerInfo closingCustomerInfo) {
        requireRequestContext();
        if (contractCustomerInfo.ctnmKn1() == null || contractCustomerInfo.ctnmKn1().isEmpty()) {
            return new ContractCustomerRegistrationUpdateTgRetailAtOpeningOutput(
                    confirmedContractCustomerInfo,
                    List.of());
        }

        Boolean customerNameKanjiDisplayFlag = customerNameKanjiDisplayFlagAcquisition
                .execute(contractCustomerInfo.ctnmKj1())
                .onmeKjhkFlg();
        String inquiryLevel = resolveInquiryLevel(closingCustomerInfo);
        AddressValues addressValues = resolveAddressValues(
                suguHtkFlg,
                confirmedContractCustomerInfo,
                confirmedAddressInfo);

        if (Boolean.TRUE.equals(snyTkmtFlg)) {
            return new ContractCustomerRegistrationUpdateTgRetailAtOpeningOutput(
                    confirmedContractCustomerInfo,
                    List.of());
        }

        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        if (newCustomerPattern.equals(cstUpdPtnKb)) {
            InternalValues newCustomerValues = buildNewCustomerValues(
                    kktSyoKyoTrno,
                    confirmedContractCustomerInfo,
                    contractCustomerInfo,
                    gasMeterLocationInfo,
                    addressValues,
                    inquiryLevel,
                    customerNameKanjiDisplayFlag);
            kiycstkDataAccess.insert(toRetailWriteRecord(newCustomerValues), auditContext);
            kiycstdDataAccess.insert(toPipelineWriteRecord(newCustomerValues), auditContext);
            return toOutput(newCustomerValues, List.of());
        }

        if (useExistingCustomerNumberPattern.equals(cstUpdPtnKb)) {
            InternalValues internalValues = buildExistingCustomerValues(
                    kktSyoKyoTrno,
                    confirmedContractCustomerInfo,
                    contractCustomerInfo,
                    addressValues,
                    inquiryLevel,
                    customerNameKanjiDisplayFlag);
            updateRetailCustomer(internalValues, decodedOriginFunctionId, auditContext);
            updatePipelineCustomer(
                    internalValues,
                    confirmedPipelineContractCustomerInfo,
                    decodedOriginFunctionId,
                    auditContext);
            List<KyotelkInf> kyotelkInfList = kyotelkDataAccess.findAllByKyoTrkNo(internalValues.kyoTrkNo())
                    .stream()
                    .map(this::toKyotelkInf)
                    .toList();
            return toOutput(internalValues, kyotelkInfList);
        }

        if (newNumberAssignedPattern.equals(cstUpdPtnKb)) {
            InternalValues internalValues = buildNewNumberAssignedValues(
                    kktSyoKyoTrno,
                    confirmedContractCustomerInfo,
                    contractCustomerInfo,
                    gasMeterLocationInfo,
                    addressValues,
                    inquiryLevel,
                    customerNameKanjiDisplayFlag);
            updateRetailCustomer(internalValues, decodedOriginFunctionId, auditContext);
            InternalValues pipelineInsertValues = internalValues.withRecUpdJnjNo(INITIAL_REC_UPD_JNJ_NO);
            kiycstdDataAccess.insert(toPipelineWriteRecord(pipelineInsertValues), auditContext);
            return toOutput(internalValues, List.of());
        }

        return new ContractCustomerRegistrationUpdateTgRetailAtOpeningOutput(
                confirmedContractCustomerInfo,
                List.of());
    }

    private String resolveInquiryLevel(ClosingCustomerInfo closingCustomerInfo) {
        String inquiryLevel = closingCustomerInfo == null || closingCustomerInfo.kiycstdInf() == null
                ? null
                : closingCustomerInfo.kiycstdInf().shkiLvShbt();
        return inquiryLevel == null || inquiryLevel.isEmpty() ? defaultInquiryLevelCode : inquiryLevel;
    }

    private void updateRetailCustomer(
            InternalValues internalValues,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = kiycstkDataAccess.updateByKyoTrkNo(
                toRetailWriteRecord(internalValues),
                RETAIL_UPDATE_TARGET_COLUMNS,
                new KiycstkUpdateKey(internalValues.kyoTrkNo(), internalValues.recUpdJnjNo()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kiycstkDataAccess.tableName(),
                    KiycstkDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO.identifier(),
                    buildRetailUpdateParameterJson(internalValues, decodedOriginFunctionId));
        }
    }

    private void updatePipelineCustomer(
            InternalValues internalValues,
            ConfirmedPipelineContractCustomerInfo confirmedPipelineContractCustomerInfo,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        Integer pipelineRecUpdJnjNo = confirmedPipelineContractCustomerInfo == null
                ? null
                : confirmedPipelineContractCustomerInfo.recUpdJnjNo();
        int updatedCount = kiycstdDataAccess.updateByKyoTrkNo(
                toPipelineWriteRecord(internalValues, pipelineRecUpdJnjNo),
                PIPELINE_UPDATE_TARGET_COLUMNS,
                new KiycstdUpdateKey(internalValues.kyoTrkNo(), pipelineRecUpdJnjNo),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kiycstdDataAccess.tableName(),
                    KiycstdDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO.identifier(),
                    buildPipelineUpdateParameterJson(internalValues, pipelineRecUpdJnjNo, decodedOriginFunctionId));
        }
    }

    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (originFunctionId == null) {
            throw new IllegalStateException(
                    "Reporting system constant is not defined: type="
                            + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value="
                            + hukkSysShbt);
        }
        return originFunctionId;
    }
