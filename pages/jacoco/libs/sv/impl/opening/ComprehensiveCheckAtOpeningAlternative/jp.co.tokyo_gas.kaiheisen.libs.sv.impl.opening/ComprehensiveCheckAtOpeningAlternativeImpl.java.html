package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ComprehensiveCheckAtOpeningAlternative;

/**
 * 概要:
 * SV-200（開栓時総合チェック（代替））における代替開栓系の判定を実装する。
 *
 * 主な仕様:
 * - 契約有無フラグと総合判定フラグがともに true の場合のみ判定対象とする。
 * - 料金計算関連状態区分が「関連有効」または「終了待ち」かつ、報告開栓事由種別が代替開栓、新設開栓、お客さま開栓のいずれかのとき開栓共通エラーを送出する。
 * - 判定に必要な定数値はコンストラクタで解決し、実行時は解決済み値を参照する。
 *
 * 既知の制約:
 * - public メソッド引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 */
public final class ComprehensiveCheckAtOpeningAlternativeImpl
        implements ComprehensiveCheckAtOpeningAlternative {

    private final String chargeCalculationRelatedStatusValid;

    private final String chargeCalculationRelatedStatusEndWaiting;

    private final String alternativeOpeningReason;

    private final String newlyOpenedTapReason;

    private final String customerOpeningReason;

    /**
     * 概要: 判定に必要な定数値を解決して初期化する。
     * 主な仕様:
     * - CIRIUS 定数サービスから料金計算関連状態区分の判定値を解決する。
     * - CIRIUS 定数サービスから報告開栓事由種別の対象 3 事由を解決する。
     * 既知の制約:
     * - 定数コード自体の妥当性検証は RequiredConstantResolver と CIRIUS 定数サービスに委譲する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ComprehensiveCheckAtOpeningAlternativeImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.chargeCalculationRelatedStatusValid = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.chargeCalculationRelatedStatusEndWaiting = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.alternativeOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.ALTERNATIVE_OPENING);
        this.newlyOpenedTapReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.NEWLY_OPENED_TAP);
        this.customerOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CUSTOMER_OPENING);
    }

    /**
     * 概要: 開栓時の代替開栓系包括チェックを実行する。
     * 主な仕様:
     * - 契約有無フラグ、総合判定フラグ、料金計算関連状態区分、報告開栓事由種別の組み合わせを評価する。
     * - 条件をすべて満たした場合はエラーコード 020100022 の開栓共通エラーを送出する。
     * 既知の制約:
     * - 引数が null の場合の扱いは呼び出し側責務であり、本メソッドでは null 安全化しない。
     *
     * @param kiyUmFlg 契約有無フラグ
     * @param suguHntFlg 総合判定フラグ
     * @param rkkiKnrnJtkb 料金計算関連状態区分
     * @param hkkKasJysb 報告開栓事由種別
     */
    @Override
    public void execute(
            Boolean kiyUmFlg,
            Boolean suguHntFlg,
            String rkkiKnrnJtkb,
            String hkkKasJysb) {
        if (kiyUmFlg
                && suguHntFlg
                && isRelatedStatus(rkkiKnrnJtkb)
                && isTargetOpeningReason(hkkKasJysb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100022);
        }
    }

    /**
     * 概要: 料金計算関連状態区分が判定対象かを判定する。
     * 主な仕様:
     * - 「関連有効」または「終了待ち」のいずれかに一致する場合に true を返す。
     * 既知の制約:
     * - 引数が null の場合は NullPointerException となる。
     *
     * @param rkkiKnrnJtkb 料金計算関連状態区分
     * @return 判定対象の料金計算関連状態区分であれば true
     */
    private boolean isRelatedStatus(String rkkiKnrnJtkb) {
        return rkkiKnrnJtkb.equals(chargeCalculationRelatedStatusValid)
                || rkkiKnrnJtkb.equals(chargeCalculationRelatedStatusEndWaiting);
    }

    /**
     * 概要: 報告開栓事由種別が対象 3 事由かを判定する。
     * 主な仕様:
     * - 代替開栓、新設開栓、お客さま開栓のいずれかに一致する場合に true を返す。
     * 既知の制約:
     * - 引数が null の場合は NullPointerException となる。
     *
     * @param hkkKasJysb 報告開栓事由種別
     * @return 対象 3 事由のいずれかであれば true
     */
    private boolean isTargetOpeningReason(String hkkKasJysb) {
        return hkkKasJysb.equals(alternativeOpeningReason)
                || hkkKasJysb.equals(newlyOpenedTapReason)
                || hkkKasJysb.equals(customerOpeningReason);
    }
}
