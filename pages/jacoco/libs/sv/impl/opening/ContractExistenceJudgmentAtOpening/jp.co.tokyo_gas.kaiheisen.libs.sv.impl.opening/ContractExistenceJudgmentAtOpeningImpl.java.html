package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContractExistenceJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContractExistenceJudgmentAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SvmstkrInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TskgmstInf;

/**
 * 概要: 開栓時契約有無判定サービス実装。
 * 主な仕様:
 * - 託送サービス契約関連有無フラグが true かつ、託送サービス契約関連状態区分が適用中の場合は契約ありと判定する。
 * - サービス契約関連有無フラグが true かつ、料金計算関連状態区分が関連終了かつサービス契約状態区分が契約適用中の場合も契約ありと判定する。
 * - 判定に利用する CIRIUS 定数はコンストラクタで fail-fast に解決する。
 */
public final class ContractExistenceJudgmentAtOpeningImpl
        implements ContractExistenceJudgmentAtOpening {

    private final String transportationServiceContractRelatedStatusApplying;

    private final String chargeCalculationRelatedStatusEnded;

    private final String serviceContractStatusUnderContract;

    /**
     * 概要: 判定に必要な CIRIUS 定数を初期化する。
     * 主な仕様:
     * - 託送サービス契約関連状態区分「適用中」を解決する。
     * - 料金計算関連状態区分「関連終了」とサービス契約状態区分「契約適用中」を解決する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ContractExistenceJudgmentAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.transportationServiceContractRelatedStatusApplying =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.chargeCalculationRelatedStatusEnded = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.serviceContractStatusUnderContract = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
    }

    /**
     * 概要: 開栓時点の契約有無を判定する。
     * 主な仕様:
     * - 託送サービス契約関連有無フラグが true かつ託送サービス契約関連状態区分が適用中なら契約ありを返す。
     * - サービス契約関連有無フラグが true かつ料金計算関連状態区分が関連終了、かつサービス契約状態区分が契約適用中なら契約ありを返す。
     * 既知の制約:
     * - `Boolean` フラグや DTO 参照が null の場合は条件不成立として扱う。
     * - 状態区分の比較は完全一致のみで行う。
     *
     * @param ksenYmdTskgmstUmFlg 開栓日時点託送サービス契約・ガスメーター設置場所関連有無フラグ
     * @param tskgmstInf 託送サービス契約・ガスメーター設置場所関連情報
     * @param ksenYmdSvmstkrUmFlg 開栓日時点サービス契約・メーター設置場所関連有無フラグ
     * @param svmstkrInf サービス契約・メーター設置場所関連情報
     * @param svkyJtkb サービス契約状態区分
     * @return 契約有無判定結果
     */
    @Override
    public ContractExistenceJudgmentAtOpeningOutput execute(
            Boolean ksenYmdTskgmstUmFlg,
            TskgmstInf tskgmstInf,
            Boolean ksenYmdSvmstkrUmFlg,
            SvmstkrInf svmstkrInf,
            String svkyJtkb) {
        boolean hasConsignmentContract = Boolean.TRUE.equals(ksenYmdTskgmstUmFlg)
                && tskgmstInf != null
                && Objects.equals(
                        tskgmstInf.tksuSvkkJtkb(),
                        transportationServiceContractRelatedStatusApplying);
        boolean hasServiceContract = Boolean.TRUE.equals(ksenYmdSvmstkrUmFlg)
                && svmstkrInf != null
                && Objects.equals(svmstkrInf.rkkiKnrnJtkb(), chargeCalculationRelatedStatusEnded)
                && Objects.equals(svkyJtkb, serviceContractStatusUnderContract);

        return new ContractExistenceJudgmentAtOpeningOutput(
                hasConsignmentContract || hasServiceContract);
    }
}
