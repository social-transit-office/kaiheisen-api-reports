/**
 * 概要: 開栓時建物更新サービス実装。
 * 主な仕様: 住宅区分指定コード、住所差分、建物名称差分を判定し、必要な場合のみ SV-076 と DA-007-U001 を使って建物情報を更新する。
 * 既知の制約: 相関IDとシステム日時は public 引数へ露出せず、報告システム種別のみ RequestContext から監査用途で利用する。
 */
public final class BuildingUpdateAtOpeningImpl implements BuildingUpdateAtOpening {

    static final String SERVICE_USER_ID = "KHSS293";

    private final TatemonDataAccess tatemonDataAccess;

    private final BuildingKanjiDisplayFlagAcquisition buildingKanjiDisplayFlagAcquisition;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String housingClassificationCodeNotSpecified;

    private final String housingClassificationCodeSpecified;

    /**
     * 概要: CDI から依存を受け取って初期化する。
     * 主な仕様: DA、下位 SV、RequestContext、定数サービスを保持する。
     * 既知の制約: 必須依存が null の場合はコンストラクタ初期化時点で fail-fast する。
     *
     * @param tatemonDataAccess 建物データアクセス
     * @param buildingKanjiDisplayFlagAcquisition 建物漢字表示可否コード取得 SV
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    @Inject
    public BuildingUpdateAtOpeningImpl(
            TatemonDataAccess tatemonDataAccess,
            BuildingKanjiDisplayFlagAcquisition buildingKanjiDisplayFlagAcquisition,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.tatemonDataAccess = Objects.requireNonNull(tatemonDataAccess, "tatemonDataAccess");
        this.buildingKanjiDisplayFlagAcquisition = Objects.requireNonNull(
                buildingKanjiDisplayFlagAcquisition,
                "buildingKanjiDisplayFlagAcquisition");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.housingClassificationCodeNotSpecified = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.HousingClassificationCode.TYPE,
                ConstantLookupKeys.Cirius.HousingClassificationCode.NOT_SPECIFIED);
        this.housingClassificationCodeSpecified = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.HousingClassificationCode.TYPE,
                ConstantLookupKeys.Cirius.HousingClassificationCode.SPECIFIED);
    }

    /**
     * 概要: 開栓報告時に訂正された建物情報を判定し、必要な場合のみ建物更新を実行する。
     * 主な仕様: 初期出力へ入力建物情報コピーを設定し、更新要否がある場合だけ確定建物情報と DA 更新パラメータを再構成する。
     * 既知の制約: public 引数の null は補正せず、そのまま比較または下位依存へ引き渡す。
     *
     * @param tatemonInf 建物情報
     * @param tatemonHkkInf 建物報告情報
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所＜導管＞情報（開栓作業日時点）
     * @param jusyoKsenjInf 住所情報（開栓時）
     * @return 建物更新結果
     */
    @Override
    public BuildingUpdateAtOpeningOutput execute(
            TatemonInf tatemonInf,
            TatemonHkkInf tatemonHkkInf,
            BuildingUpdateGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            JusyoKsenjInf jusyoKsenjInf) {
        TatemonKktInf initialConfirmedBuildingInfo = toConfirmedBuildingInfo(tatemonInf);
        if (!housingClassificationCodeNotSpecified.equals(tatemonInf.jutakKbnStCd())) {
            return new BuildingUpdateAtOpeningOutput(Boolean.FALSE, Boolean.FALSE, initialConfirmedBuildingInfo);
        }

        boolean addressChanged = hasAddressDifference(tatemonInf, jusyoKsenjInf);
        boolean nameChanged = hasNameDifference(tatemonInf, tatemonHkkInf);
        if (!addressChanged && !nameChanged) {
            return new BuildingUpdateAtOpeningOutput(Boolean.FALSE, Boolean.FALSE, initialConfirmedBuildingInfo);
        }

        TatemonKktInf confirmedBuildingInfo = resolveConfirmedBuildingInfo(
                tatemonInf,
                tatemonHkkInf,
                gmtsetdKsenSaymdInf,
                jusyoKsenjInf,
                addressChanged,
                nameChanged);
        String decodedOriginFunctionId = decodeOriginFunctionId();
        TatemonWriteRecord writeRecord = toWriteRecord(confirmedBuildingInfo);
        LinkedHashSet<TatemonColumn> targetColumns = buildTargetColumns(addressChanged, nameChanged);
        TatemonUpdateKey updateKey = new TatemonUpdateKey(tatemonInf.bldNo(), tatemonInf.recUpdJnjNo());
        tatemonDataAccess.updateByBldNoAndRecUpdJnjNo(
                writeRecord,
                Set.copyOf(targetColumns),
                updateKey,
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        return new BuildingUpdateAtOpeningOutput(
                Boolean.valueOf(nameChanged),
                Boolean.valueOf(addressChanged),
                confirmedBuildingInfo);
    }

    /**
     * 概要: 入力建物情報を返却用の建物情報（確定）へ変換する。
     * 主な仕様: 返却対象外の監査項目と排他制御項目を除いた値をそのまま転記する。
     * 既知の制約: null 項目は補完せず、そのまま返却 DTO に保持する。
     *
     * @param tatemonInf 建物情報
     * @return 建物情報（確定）
     */
    private static TatemonKktInf toConfirmedBuildingInfo(TatemonInf tatemonInf) {
        return new TatemonKktInf(
                tatemonInf.bldNo(),
                tatemonInf.bldnKn(),
                tatemonInf.bldnKj(),
                tatemonInf.tymkBantGo_1(),
                tatemonInf.tymkBantGo_2(),
                tatemonInf.tymkBantGo_3(),
                tatemonInf.autolockCd(),
                tatemonInf.autolockAnsNo(),
                tatemonInf.autolockComent(),
                tatemonInf.bldKjhkFlg(),
                tatemonInf.jyutakuKbnCd(),
                tatemonInf.jutakKbnStCd(),
                tatemonInf.szcGkkCd(),
                tatemonInf.szcMchCd(),
                tatemonInf.szcCmazCd());
    }

    /**
     * 概要: 建物丁目番地号の差分有無を判定する。
     * 主な仕様: 3 項目のいずれかが不一致であれば住所変更ありと判定する。
     * 既知の制約: 前後空白の除去や正規化は行わず、完全一致比較のみを行う。
     *
     * @param tatemonInf 建物情報
     * @param jusyoKsenjInf 住所情報（開栓時）
     * @return 差分がある場合 true
     */
    private static boolean hasAddressDifference(TatemonInf tatemonInf, JusyoKsenjInf jusyoKsenjInf) {
        return !Objects.equals(tatemonInf.tymkBantGo_1(), jusyoKsenjInf.kktCbgSuj1())
                || !Objects.equals(tatemonInf.tymkBantGo_2(), jusyoKsenjInf.kktCbgKn2())
                || !Objects.equals(tatemonInf.tymkBantGo_3(), jusyoKsenjInf.kktCbgKn3());
    }

    /**
     * 概要: 建物名称の差分有無を判定する。
     * 主な仕様: 建物報告情報.建物名カナが null または空文字以外であり、かつ建物名カナまたは建物名漢字が不一致のとき true を返す。
     * 既知の制約: 建物報告情報.建物名カナが空文字のときは建物名漢字だけが不一致でも名称変更とみなさない。
     *
     * @param tatemonInf 建物情報
     * @param tatemonHkkInf 建物報告情報
     * @return 差分がある場合 true
     */
    private static boolean hasNameDifference(TatemonInf tatemonInf, TatemonHkkInf tatemonHkkInf) {
        return !isNullOrEmpty(tatemonHkkInf.bldnKn())
                && (!Objects.equals(tatemonInf.bldnKn(), tatemonHkkInf.bldnKn())
                || !Objects.equals(tatemonInf.bldnKj(), tatemonHkkInf.bldnKj()));
    }

    /**
     * 概要: null または空文字かを判定する。
     * 主な仕様: 空白のみ文字列は空文字扱いしない。
     * 既知の制約: YAML 設計に合わせて blank 判定ではなく empty 判定のみを行う。
     *
     * @param value 判定対象文字列
     * @return null または空文字の場合 true
     */
    private static boolean isNullOrEmpty(String value) {
        return value == null || value.isEmpty();
    }

    /**
     * 概要: 更新後の建物情報（確定）を構築する。
     * 主な仕様: 名称変更時は SV-076 を使って建物漢字表示可否フラグと住宅区分指定コードを更新し、住所変更時は住所項目と所在地コード項目を更新する。
     * 既知の制約: 名称変更と住所変更は独立に評価し、両方成立する場合は両方の更新を反映する。
     *
     * @param tatemonInf 建物情報
     * @param tatemonHkkInf 建物報告情報
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所＜導管＞情報（開栓作業日時点）
     * @param jusyoKsenjInf 住所情報（開栓時）
     * @param addressChanged 住所差分有無
     * @param nameChanged 名称差分有無
     * @return 更新後の建物情報（確定）
     */
    private TatemonKktInf resolveConfirmedBuildingInfo(
            TatemonInf tatemonInf,
            TatemonHkkInf tatemonHkkInf,
            BuildingUpdateGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            JusyoKsenjInf jusyoKsenjInf,
            boolean addressChanged,
            boolean nameChanged) {
        String resolvedBuildingNameKana = tatemonInf.bldnKn();
        String resolvedBuildingNameKanji = tatemonInf.bldnKj();
        Boolean resolvedBuildingKanjiDisplayFlag = tatemonInf.bldKjhkFlg();
        String resolvedHousingCategoryCode = tatemonInf.jyutakuKbnCd();
        String resolvedHousingCategorySpecifiedCode = tatemonInf.jutakKbnStCd();
        String resolvedAddress1 = tatemonInf.tymkBantGo_1();
        String resolvedAddress2 = tatemonInf.tymkBantGo_2();
        String resolvedAddress3 = tatemonInf.tymkBantGo_3();
        String resolvedGkkCd = tatemonInf.szcGkkCd();
        String resolvedMchCd = tatemonInf.szcMchCd();
        String resolvedCmazCd = tatemonInf.szcCmazCd();

        if (nameChanged) {
            resolvedBuildingNameKana = tatemonHkkInf.bldnKn();
            resolvedBuildingNameKanji = tatemonHkkInf.bldnKj();
            resolvedBuildingKanjiDisplayFlag = buildingKanjiDisplayFlagAcquisition
                    .execute(tatemonHkkInf.bldnKj())
                    .bldKjHyojiCd();
            resolvedHousingCategorySpecifiedCode = housingClassificationCodeSpecified;
        }
        if (addressChanged) {
            resolvedAddress1 = jusyoKsenjInf.kktCbgSuj1();
            resolvedAddress2 = jusyoKsenjInf.kktCbgKn2();
            resolvedAddress3 = jusyoKsenjInf.kktCbgKn3();
            resolvedGkkCd = gmtsetdKsenSaymdInf.gkkCd();
            resolvedMchCd = gmtsetdKsenSaymdInf.mchCd();
            resolvedCmazCd = gmtsetdKsenSaymdInf.cmazCd();
        }
        return new TatemonKktInf(
                tatemonInf.bldNo(),
                resolvedBuildingNameKana,
                resolvedBuildingNameKanji,
                resolvedAddress1,
                resolvedAddress2,
                resolvedAddress3,
                tatemonInf.autolockCd(),
                tatemonInf.autolockAnsNo(),
                tatemonInf.autolockComent(),
                resolvedBuildingKanjiDisplayFlag,
                resolvedHousingCategoryCode,
                resolvedHousingCategorySpecifiedCode,
                resolvedGkkCd,
                resolvedMchCd,
                resolvedCmazCd);
    }

    /**
     * 概要: 更新対象列集合を構築する。
     * 主な仕様: 名称変更時は名称系 5 列、住所変更時は住所系 6 列を追加し、両方成立する場合は union を返す。
     * 既知の制約: 列の追加順は更新意図を読み取りやすいよう固定する。
     *
     * @param addressChanged 住所差分有無
     * @param nameChanged 名称差分有無
     * @return 更新対象列
     */
    private static LinkedHashSet<TatemonColumn> buildTargetColumns(boolean addressChanged, boolean nameChanged) {
        LinkedHashSet<TatemonColumn> targetColumns = new LinkedHashSet<>();
        if (nameChanged) {
            targetColumns.add(TatemonColumn.BLD_NAM_KN);
            targetColumns.add(TatemonColumn.BLD_NAM_KJ);
            targetColumns.add(TatemonColumn.BLD_KJ_HYOJI_CD);
            targetColumns.add(TatemonColumn.JYUTAKU_KBN_CD);
            targetColumns.add(TatemonColumn.JUTAK_KBN_ST_CD);
        }
        if (addressChanged) {
            targetColumns.add(TatemonColumn.TYMK_BANT_GO_1);
            targetColumns.add(TatemonColumn.TYMK_BANT_GO_2);
            targetColumns.add(TatemonColumn.TYMK_BANT_GO_3);
            targetColumns.add(TatemonColumn.SYOZA_GYOK_CD);
            targetColumns.add(TatemonColumn.SYOZA_MATI_CD);
            targetColumns.add(TatemonColumn.SYOZA_TYMKAZ_CD);
        }
        return targetColumns;
    }

    /**
     * 概要: 建物情報（確定）を DA 更新レコードへ変換する。
     * 主な仕様: 監査列と主キーを除く TATEMON 更新対象列へ対応する値を転記する。
     * 既知の制約: targetColumns に含まれない項目もレコードへ保持するが、DA 側で更新対象外として扱われる。
     *
     * @param confirmedBuildingInfo 建物情報（確定）
     * @return DA 更新レコード
     */
    private static TatemonWriteRecord toWriteRecord(TatemonKktInf confirmedBuildingInfo) {
        return new TatemonWriteRecord(
                confirmedBuildingInfo.bldnKn(),
                confirmedBuildingInfo.bldnKj(),
                confirmedBuildingInfo.tymkBantGo_1(),
                confirmedBuildingInfo.tymkBantGo_2(),
                confirmedBuildingInfo.tymkBantGo_3(),
                confirmedBuildingInfo.autolockCd(),
                confirmedBuildingInfo.autolockAnsNo(),
                confirmedBuildingInfo.autolockComent(),
                confirmedBuildingInfo.bldKjhkFlg(),
                confirmedBuildingInfo.jyutakuKbnCd(),
                confirmedBuildingInfo.jutakKbnStCd(),
                confirmedBuildingInfo.szcGkkCd(),
                confirmedBuildingInfo.szcMchCd(),
                confirmedBuildingInfo.szcCmazCd());
    }

    /**
     * 概要: 報告元機能 ID を復元する。
     * 主な仕様: RequestContext から取得した報告システム種別を使って開閉栓定数サービスで名称解決する。
     * 既知の制約: 報告システム種別の妥当性確認は定数解決側の契約に委ねる。
     *
     * @return 報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt().orElse(null));
    }
