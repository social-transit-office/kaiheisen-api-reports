/**
 * 概要: 前住者契約情報取得サービスを実装する。
 * 主な仕様: 東京ガス分岐時のみ前住者契約情報を辿り、総合メーター判定、サービス契約、免税契約、移転先管理、支払契約関連を取得して返却する。
 * 既知の制約: RequestContext の systemDatetime は支払関連取得時のみ必須であり、DA record は API モデルへ変換して返却する。
 */

    /**
     * 概要: 依存オブジェクトと必須定数を初期化する。
     * 主な仕様: 前住者契約取得で必要な下位 SV、各 DA、RequestContext、CIRIUS 定数を fail-fast に解決する。
     *
     * @param comprehensiveMeterJudgmentAtOpeningArg 開栓時総合メーター判定サービス
     * @param svmstkrDataAccessArg サービス契約・メーター設置場所関連データアクセス
     * @param svckyaDataAccessArg サービス契約データアクセス
     * @param mnzkiyDataAccessArg 免税契約データアクセス
     * @param itensakDataAccessArg 移転先管理データアクセス
     * @param svsihkrDataAccessArg サービス契約・支払契約関連データアクセス
     * @param sihakyDataAccessArg 支払契約データアクセス
     * @param requestContextArg リクエストコンテキスト
     * @param ciriusConstantServiceArg CIRIUS 定数取得サービス
     */
    public PreviousResidentContractInfoAcquisitionImpl(
            ComprehensiveMeterJudgmentAtOpening comprehensiveMeterJudgmentAtOpeningArg,
            SvmstkrDataAccess svmstkrDataAccessArg,
            SvckyaDataAccess svckyaDataAccessArg,
            MnzkiyDataAccess mnzkiyDataAccessArg,
            ItensakDataAccess itensakDataAccessArg,
            SvsihkrDataAccess svsihkrDataAccessArg,
            SihakyDataAccess sihakyDataAccessArg,
            RequestContext requestContextArg,
            CiriusConstantService ciriusConstantServiceArg) {
        this.comprehensiveMeterJudgmentAtOpening = Objects.requireNonNull(
                comprehensiveMeterJudgmentAtOpeningArg,
                "comprehensiveMeterJudgmentAtOpening");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccessArg, "svmstkrDataAccess");
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccessArg, "svckyaDataAccess");
        this.mnzkiyDataAccess = Objects.requireNonNull(mnzkiyDataAccessArg, "mnzkiyDataAccess");
        this.itensakDataAccess = Objects.requireNonNull(itensakDataAccessArg, "itensakDataAccess");
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccessArg, "svsihkrDataAccess");
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccessArg, "sihakyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContextArg, "requestContext");

        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceArg, "ciriusConstantService");
        this.tgRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.chargeCalculationRelatedStatusInvalid = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.INVALID);
        this.serviceContractStatusInvalid = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.INVALID);
        this.serviceContractStatusUnderContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.taxExempt = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsumptionTax.TYPE,
                ConstantLookupKeys.Cirius.ConsumptionTax.DUTY_FREE);
        this.paymentContractStatusInvalid = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
        this.transferDestinationCancellationFlagNoConcept = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferDestinationCancellationFlag.TYPE,
                ConstantLookupKeys.Cirius.TransferDestinationCancellationFlag.NO_CONCEPT);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
    }

    /**
     * 概要: 前住者契約情報を取得する。
     * 主な仕様: 東京ガス以外または前住者契約が取得できない場合は関連情報を未設定で返し、取得できる場合のみ後続依存を連鎖実行する。
     * 既知の制約: 支払関連取得時は RequestContext.systemDatetime を必須とし、必須取得 DA の 0 件はデータ不整合として扱う。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kasgYmd 開栓作業年月日
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param kasJysb 開栓事由種別
     * @return 前住者契約情報取得結果
     */
    @Override
    public PreviousResidentContractInfoAcquisitionOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            String tksuKijsCd,
            String ckrtGmscJtkb,
            String kasJysb) {
        boolean dedicatedMeterFlag = dedicatedGasMeterStatus.equals(ckrtGmscJtkb);
        if (!tgRetailerCode.equals(tksuKijsCd)) {
            return emptyOutput(dedicatedMeterFlag);
        }

        Optional<SvmstkrRecord> svmstkrRecord =
                svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiKnrnJtkb(
                        gasMtstNo,
                        kasgYmd,
                        chargeCalculationRelatedStatusInvalid);
        if (svmstkrRecord.isEmpty()) {
            return emptyOutput(dedicatedMeterFlag);
        }

        SvmstkrRecord previousResidentRecord = svmstkrRecord.orElseThrow();
        ComprehensiveMeterJudgmentAtOpeningOutput comprehensiveOutput =
                comprehensiveMeterJudgmentAtOpening.execute(
                        previousResidentRecord.mtstNo(),
                        previousResidentRecord.svkyNo(),
                        kasJysb,
                        kasgYmd,
                        previousResidentRecord.rkkiKnrnJtkb(),
                        previousResidentRecord.rkkiEndYmd());
        SvckyaRecord svckyaKsenSaymdRecord =
                findSvckyaKsenSaymdRecord(previousResidentRecord.svkyNo(), kasgYmd);
        List<SvckyaRecord> svckyaTkyTyuRecords =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(
                        previousResidentRecord.svkyNo(),
                        serviceContractStatusUnderContract);
        MnzkiyInf mnzkiyInf = null;
        if (taxExempt.equals(svckyaKsenSaymdRecord.sziKziKbn())) {
            mnzkiyInf = toMnzkiyInf(
                    findMnzkiyRecord(
                            svckyaKsenSaymdRecord.svkyNo(),
                            svckyaKsenSaymdRecord.rkkiEndYmd()));
        }
        Optional<ItensakRecord> itensakRecord =
                itensakDataAccess.findAllByGmtSetNoAndSyokyNoAndItenTksCd(
                                gasMtstNo,
                                previousResidentRecord.svkyNo(),
                                transferDestinationCancellationFlagNoConcept)
                        .stream()
                        .findFirst();
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        SvsihkrRecord svsihkrRecord = findSvsihkrRecord(previousResidentRecord.svkyNo(), systemDate);
        SihakyRecord sihakyRecord = findSihakyRecord(svsihkrRecord.sihKiyNo(), systemDate);

        return new PreviousResidentContractInfoAcquisitionOutput(
                toSvmstkrInf(previousResidentRecord),
                toSvckyaKsenSaymdInf(svckyaKsenSaymdRecord),
                toSvckyaTkyTyuInf(svckyaTkyTyuRecords),
                itensakRecord.map(PreviousResidentContractInfoAcquisitionImpl::toItensakInf).orElse(null),
                toSvsihkrInf(svsihkrRecord),
                toSihakyInf(sihakyRecord),
                comprehensiveOutput == null ? null : comprehensiveOutput.suguHtkFlg(),
                comprehensiveOutput == null ? null : comprehensiveOutput.kzksyAtkCd(),
                dedicatedMeterFlag,
                mnzkiyInf,
                itensakRecord.isPresent());
    }

    private PreviousResidentContractInfoAcquisitionOutput emptyOutput(boolean dedicatedMeterFlag) {

                Boolean.FALSE,
                null,
                dedicatedMeterFlag,
                emptyMnzkiyInf(),
                Boolean.FALSE);
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime().orElseThrow(
                () -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private SvckyaRecord findSvckyaKsenSaymdRecord(Long svkyNo, LocalDate kasgYmd) {
        return svckyaDataAccess
                .findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNotOrderByRkkiSttYmd(
                        svkyNo,
                        kasgYmd,
                        serviceContractStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svckyaDataAccess.tableName(),
                        SvckyaDataAccess.CallSpecs
                                .FIND_LATEST_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_NOT_ORDER_BY_RKKI_STT_YMD
                                .identifier(),
                        parameterJsonForSvckyaKsenSaymd(svkyNo, kasgYmd)));
    }

    private MnzkiyRecord findMnzkiyRecord(Long svkyNo, LocalDate rkkiEndYmd) {
        return mnzkiyDataAccess
                .findBySvkyNoAndMnziTyenYmd(svkyNo, rkkiEndYmd)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        mnzkiyDataAccess.tableName(),
                        MnzkiyDataAccess.CallSpecs.FIND_BY_SVKY_NO_AND_MNZI_TYEN_YMD.identifier(),
                        parameterJsonForMnzkiy(svkyNo, rkkiEndYmd)));
    }

    private SvsihkrRecord findSvsihkrRecord(Long svkyNo, LocalDate systemDate) {
        return svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndNotSihKiyJtkb(
                        svkyNo,
                        systemDate,
                        paymentContractStatusInvalid)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svsihkrDataAccess.tableName(),
                        SvsihkrDataAccess.CallSpecs
                                .FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_NOT_SIH_KIY_JTKB
                                .identifier(),
                        parameterJsonForSvsihkr(svkyNo, systemDate)));
    }

    private static SvckyaTkyTyuInf toSvckyaTkyTyuInf(List<SvckyaRecord> records) {
        if (records.isEmpty()) {
            return new SvckyaTkyTyuInf(Boolean.FALSE, null);
        }
        LocalDate latestRkkiSttYmd = records.stream()
                .map(SvckyaRecord::rkkiSttYmd)
                .filter(Objects::nonNull)
                .max(LocalDate::compareTo)
                .orElse(null);
        return new SvckyaTkyTyuInf(Boolean.TRUE, latestRkkiSttYmd);
    }
