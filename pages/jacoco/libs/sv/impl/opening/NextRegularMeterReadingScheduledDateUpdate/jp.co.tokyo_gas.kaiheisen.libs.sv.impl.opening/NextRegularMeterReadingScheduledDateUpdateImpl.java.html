/**
 * 概要: SV-231（次回定例検針予定日更新）のサービス実装。
 * 主な仕様:
 * - ガスメーター定例検針対象管理がある場合は KNSYTI を検索して次回定例検針予定年月日を更新する。
 * - ガスメーター定例検針対象管理がない場合は当月の検針結果有無に応じて検索基準日を切り替え、SV-022 から補完した結果で更新する。
 * - KNSYTI 取得 0 件は DataInconsistencyException として送出する。
 * - 排他あり更新で対象なしとなる場合は DA 共通 helper 由来の例外をそのまま上位へ委譲する。
 * 既知の制約:
 * - RequestContext の hukkSysShbt と systemDatetime が未設定の場合は IllegalStateException を送出する。
 * - DA-027-R006 の検索事由種別はコンストラクタ初期化時に CIRIUS 定数から fail-fast で解決する。
 */
public class NextRegularMeterReadingScheduledDateUpdateImpl
        implements NextRegularMeterReadingScheduledDateUpdate {

    static final String SERVICE_USER_ID = "KHSS231";

    private final GmtkstsDataAccess gmtkstsDataAccess;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final KnsytiDataAccess knsytiDataAccess;

    private final NextMeterReadingScheduledDateAcquisition nextMeterReadingScheduledDateAcquisition;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String regularReasonType;

    private final String regularSwCancellationReasonType;

    /**
     * 概要: SV-231 実行に必要な依存オブジェクトと必須定数を初期化する。
     *
     * @param gmtkstsDataAccess ガスメーター定例検針対象管理データアクセス
     * @param gmkskkdDataAccess ガスメーター検針結果＜導管＞データアクセス
     * @param knsytiDataAccess 検針予定データアクセス
     * @param nextMeterReadingScheduledDateAcquisition 次回検針予定年月日取得サービス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public NextRegularMeterReadingScheduledDateUpdateImpl(
            GmtkstsDataAccess gmtkstsDataAccess,
            GmkskkdDataAccess gmkskkdDataAccess,
            KnsytiDataAccess knsytiDataAccess,
            NextMeterReadingScheduledDateAcquisition nextMeterReadingScheduledDateAcquisition,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gmtkstsDataAccess = Objects.requireNonNull(gmtkstsDataAccess, "gmtkstsDataAccess");
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        this.knsytiDataAccess = Objects.requireNonNull(knsytiDataAccess, "knsytiDataAccess");
        this.nextMeterReadingScheduledDateAcquisition = Objects.requireNonNull(
                nextMeterReadingScheduledDateAcquisition,
                "nextMeterReadingScheduledDateAcquisition");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.regularReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR);
        this.regularSwCancellationReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR_SW_CANCELLATION);
    }

    /**
     * 概要: SV-231 を実行し、更新対象として採用した検針予定情報を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 次回定例検針予定日更新結果
     */
    @Override
    public NextRegularMeterReadingScheduledDateUpdateOutput execute(Long gasMtstNo) {
        RequestState requestState = requireRequestState();
        return gmtkstsDataAccess.findByGasMtstNo(gasMtstNo)
                .map(record -> executeUsingRegularScheduleTarget(gasMtstNo, record, requestState))
                .orElseGet(() -> executeUsingFallbackSearch(gasMtstNo, requestState));
    }

    /**
     * 概要: ガスメーター定例検針対象管理が存在する通常分岐を実行する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param gmtkstsRecord ガスメーター定例検針対象管理レコード
     * @param requestState リクエスト状態
     * @return 更新後に返却する出力 DTO
     */
    private NextRegularMeterReadingScheduledDateUpdateOutput executeUsingRegularScheduleTarget(
            Long gasMtstNo,
            GmtkstsRecord gmtkstsRecord,
            RequestState requestState) {
        KnsytiRecord knsytiRecord = knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(
                        gasMtstNo,
                        toYearMonth(requestState.systemDate()))
                .stream()
                .findFirst()
                .orElseThrow(() -> createKnsytiReadInconsistency(
                        gasMtstNo,
                        toYearMonth(requestState.systemDate())));
        updateScheduledDate(
                gasMtstNo,
                knsytiRecord.tkysYm(),
                knsytiRecord.recUpdJnjNo(),
                gmtkstsRecord.tkytYmd(),
                requestState.originFunctionId());
        return toOutput(knsytiRecord);
    }

    /**
     * 概要: ガスメーター定例検針対象管理が存在しない場合のフォールバック分岐を実行する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param requestState リクエスト状態
     * @return 更新後に返却する出力 DTO
     */
    private NextRegularMeterReadingScheduledDateUpdateOutput executeUsingFallbackSearch(
            Long gasMtstNo,
            RequestState requestState) {
        LocalDate systemDate = requestState.systemDate();
        LocalDate firstDayOfMonth = systemDate.withDayOfMonth(1);
        LocalDate lastDayOfMonth = systemDate.withDayOfMonth(systemDate.lengthOfMonth());
        List<String> reasonTypes = List.of(regularReasonType, regularSwCancellationReasonType);
        boolean hasCurrentMonthMeterReading = gmkskkdDataAccess.findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
                        gasMtstNo,
                        firstDayOfMonth,
                        lastDayOfMonth,
                        reasonTypes)
                .isPresent();
        LocalDate searchBaseDate = hasCurrentMonthMeterReading
                ? systemDate.plusMonths(1).withDayOfMonth(1)
                : systemDate;
        NextMeterReadingScheduledDateAcquisitionOutput nextMeterReading =
                nextMeterReadingScheduledDateAcquisition.execute(gasMtstNo, searchBaseDate);
        if (nextMeterReading == null) {
            throw new IllegalStateException("SV-022 output is required");
        }
        updateScheduledDate(
                gasMtstNo,
                nextMeterReading.tkysYm(),
                nextMeterReading.recUpdJnjNo(),
                nextMeterReading.tkytYmd(),
                requestState.originFunctionId());
        return toOutput(nextMeterReading);
    }

    /**
     * 概要: KNSYTI の次回定例検針予定年月日を更新し、0 件更新時はデータ不整合例外を送出する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param tkysYm 適用開始年月
     * @param recUpdJnjNo レコード更新排他制御順序番号
     * @param nextScheduledDate 更新後の次回定例検針予定年月日
     * @param originFunctionId 監査用の起点機能 ID
     */
    private void updateScheduledDate(
            Long gasMtstNo,
            YearMonth tkysYm,
            Integer recUpdJnjNo,
            LocalDate nextScheduledDate,
            String originFunctionId) {
        if (tkysYm == null) {
            throw new IllegalStateException("Update tkysYm is required");
        }
        if (nextScheduledDate == null) {
            throw new IllegalStateException("Update jkiTkytYmd is required");
        }
        knsytiDataAccess.updateByGasMtstNoAndTkysYmAndRecUpdJnjNo(
                new KnsytiWriteRecord(
                        gasMtstNo,
                        tkysYm,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        nextScheduledDate,
                        null,
                        null,
                        recUpdJnjNo),
                Set.of(KnsytiColumn.JKI_TKYT_YMD),
                new KnsytiUpdateKey(gasMtstNo, tkysYm, recUpdJnjNo),
                AuditContext.of(SERVICE_USER_ID, originFunctionId));
    }

    /**
     * 概要: 実行に必要な RequestContext 値を取得し、監査用機能 ID を解決する。
     *
     * @return 実行時に利用するリクエスト状態
     */
    private RequestState requireRequestState() {
        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String originFunctionId = requireOriginFunctionId(hukkSysShbt);
        return new RequestState(systemDate, originFunctionId);
    }

    /**
     * 概要: 報告システム種別に対応する起点機能 ID を必須定数として解決する。
     *
     * @param hukkSysShbt 報告システム種別
     * @return 起点機能 ID
     */
    private String requireOriginFunctionId(String hukkSysShbt) {
        String type = ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE;
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(type, hukkSysShbt);
        if (originFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + type + ", value=" + hukkSysShbt);
        }
        return originFunctionId;
    }

    /**
     * 概要: KNSYTI 取得 0 件時のデータ不整合例外を生成する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYm 検索基準年月
     * @return データ不整合例外
     */
    private RuntimeException createKnsytiReadInconsistency(Long gasMtstNo, YearMonth knskKjnYm) {
        return DataInconsistencySupport.create(
                knsytiDataAccess.tableName(),
                KnsytiDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYm", knskKjnYm.toString()));
    }

    /**
     * 概要: KNSYTI レコードから出力 DTO を構築する。
     *
     * @param knsytiRecord 検針予定レコード
     * @return 次回定例検針予定日更新結果
     */
    private static NextRegularMeterReadingScheduledDateUpdateOutput toOutput(
            KnsytiRecord knsytiRecord) {
        return new NextRegularMeterReadingScheduledDateUpdateOutput(
                new KnsytiInf(
                        knsytiRecord.knsHuhuKbn(),
                        knsytiRecord.stkbKbn(),
                        knsytiRecord.shtiKskbNo()));
    }

    /**
     * 概要: SV-022 の出力から出力 DTO を構築する。
     *
     * @param nextMeterReading SV-022 出力
     * @return 次回定例検針予定日更新結果
     */
    private static NextRegularMeterReadingScheduledDateUpdateOutput toOutput(
            NextMeterReadingScheduledDateAcquisitionOutput nextMeterReading) {
        return new NextRegularMeterReadingScheduledDateUpdateOutput(
                new KnsytiInf(
                        nextMeterReading.knsHuhuKbn(),
                        nextMeterReading.stkbKbn(),
                        nextMeterReading.shtiKskbNo()));
    }

    /**
     * 概要: LocalDate の年月を YearMonth へ変換する。
     *
     * @param date 対象日付
     * @return YearMonth
     */
    private static YearMonth toYearMonth(LocalDate date) {
        return YearMonth.of(date.getYear(), date.getMonthValue());
    }

    /** 実行に必要な RequestContext 由来値。 */
    private record RequestState(
            LocalDate systemDate,
            String originFunctionId) {
    }
}
