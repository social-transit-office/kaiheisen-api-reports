/**
 * 概要: SV-057（開栓取消実施可否判定）のサービス実装。
 * 主な仕様: 開栓取消対象の供給停止管理、確定前ガスメーター検針結果、サービス契約関連を取得し、設計書で定義された業務エラーとデータ不整合を判定する。
 * 既知の制約: RequestContext.systemDatetime は必須扱いとし、DA-041-R009 再検索 0 件のみデータ不整合として扱う。
 */
public final class OpeningCancelFeasibilityJudgmentImpl implements OpeningCancelFeasibilityJudgment {

    private final KkytskrDataAccess kkytskrDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final SvckyaDataAccess svckyaDataAccess;

    private final RequestContext requestContext;

    private final String tgRetailerCode;

    private final String dedicatedGasMeterStatus;

    private final String supplyStopPaymentUnnecessary;

    private final String relatedValid;

    private final String relatedEndWaiting;

    private final String serviceContractUnderContract;

    private final String corporateForCompanyUse;

    private final String dutyFree;

    private final String meterReadingReasonOpening;

    private final String meterReadingReasonSupplyResumed;

    /**
     * コンストラクタ。
     *
     * @param kkytskrDataAccessArg 供給停止管理 DA
     * @param ktmgmkkDataAccessArg 確定前ガスメーター検針結果 DA
     * @param svmstkrDataAccessArg サービス契約・メーター設置場所関連 DA
     * @param svckyaDataAccessArg サービス契約 DA
     * @param requestContextArg リクエストコンテキスト
     * @param ciriusConstantServiceArg CIRIUS コンスタント取得サービス
     */
    public OpeningCancelFeasibilityJudgmentImpl(
            KkytskrDataAccess kkytskrDataAccessArg,
            KtmgmkkDataAccess ktmgmkkDataAccessArg,
            SvmstkrDataAccess svmstkrDataAccessArg,
            SvckyaDataAccess svckyaDataAccessArg,
            RequestContext requestContextArg,
            CiriusConstantService ciriusConstantServiceArg) {
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccessArg, "kkytskrDataAccess");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccessArg, "ktmgmkkDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccessArg, "svmstkrDataAccess");
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccessArg, "svckyaDataAccess");
        this.requestContext = Objects.requireNonNull(requestContextArg, "requestContext");

        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceArg, "ciriusConstantService");
        this.tgRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.supplyStopPaymentUnnecessary = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOPPAGE_PAYMENT_UNNECESSARY);
        this.relatedValid = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.relatedEndWaiting = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.serviceContractUnderContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.corporateForCompanyUse = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Corporate.TYPE,
                ConstantLookupKeys.Cirius.Corporate.FOR_COMPANY_USE);
        this.dutyFree = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsumptionTax.TYPE,
                ConstantLookupKeys.Cirius.ConsumptionTax.DUTY_FREE);
        this.meterReadingReasonOpening = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
            ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.OPENING);
        this.meterReadingReasonSupplyResumed = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
            ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_RESUMED);
    }

    /**
     * 概要: 開栓取消実施可否を判定し、関連情報を返却する。
     * 主な仕様: 供給停止管理、確定前ガスメーター検針結果、サービス契約・メーター設置場所関連、サービス契約情報を順に参照する。
     * 既知の制約: サービス契約取得結果が 0 件の場合は正常継続せず、後続の社用・免税判定で
     * 実行時例外（NullPointerException）となり得る。
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param mtgMtstNo 元ガスメーター設置場所番号
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param tksuSvkyNo 託送サービス契約番号
     * @return 開栓取消実施可否判定結果
     */
    @Override
    public OpeningCancelFeasibilityJudgmentOutput execute(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            Long mtgMtstNo,
            String tksuKijsCd,
            String ckrtGmscJtkb,
            Long tksuSvkyNo) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        Optional<KkytskrRecord> kkytskrRecord =
                kkytskrDataAccess.findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
                        gasMtstNo,
                        trkKnsYmd,
                        supplyStopPaymentUnnecessary);
        Optional<KtmgmkkRecord> ktmgmkkRecord = ktmgmkkDataAccess.findByGasMtstNo(gasMtstNo);

        if (kkytskrRecord.isEmpty()) {
            validatePreconfirmedMeterReading(ktmgmkkRecord, systemDate, trkKnsYmd, tksuSvkyNo);
        }

        SvmstkrRecord svmstkrRecord = null;
        SvckyaRecord svckyaRecord = null;
        boolean sny4XUmFlg = false;

        if (tgRetailerCode.equals(tksuKijsCd)) {
            Optional<SvmstkrRecord> primaryRelation = findSvmstkrByMtstNo(gasMtstNo);
            if (primaryRelation.isPresent()) {
                svmstkrRecord = primaryRelation.orElseThrow();
            } else if (dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
                svmstkrRecord = findSvmstkrByMtstNoOrThrow(mtgMtstNo);
                sny4XUmFlg = true;
            } else {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
            }

            validateSupplyStopServiceContract(kkytskrRecord, svmstkrRecord.svkyNo());

            svckyaRecord = findSvckyaBySvkyNo(svmstkrRecord.svkyNo()).orElse(null);
            if (corporateForCompanyUse.equals(svckyaRecord.syaKbn())) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000025);
            }
            if (dutyFree.equals(svckyaRecord.sziKziKbn())) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000026);
            }
        }

        return new OpeningCancelFeasibilityJudgmentOutput(
                toSvmstkrInf(svmstkrRecord),
                ktmgmkkRecord.map(OpeningCancelFeasibilityJudgmentImpl::toKtmgmkkInf).orElse(null),
                toSvckyaInf(svckyaRecord),
                kkytskrRecord.map(OpeningCancelFeasibilityJudgmentImpl::toKkytskrInf).orElse(null),
                sny4XUmFlg,
                kkytskrRecord.isPresent());
    }

    private void validatePreconfirmedMeterReading(
            Optional<KtmgmkkRecord> ktmgmkkRecord,
            LocalDate systemDate,
            LocalDate trkKnsYmd,
            Long tksuSvkyNo) {
        if (ktmgmkkRecord.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000023);
        }

        KtmgmkkRecord record = ktmgmkkRecord.orElseThrow();
        if (!Objects.equals(record.kgtsHukkYmd(), systemDate)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000021);
        }
        if (!isTargetMeterReadingReason(record.knkkSksiJysb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000048);
        }
        if (!Objects.equals(record.knsYmd(), trkKnsYmd)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000048);
        }
        if (!Objects.equals(record.tksuSvkyNo(), tksuSvkyNo)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
        }
    }

    private boolean isTargetMeterReadingReason(String knkkSksiJysb) {
        return meterReadingReasonOpening.equals(knkkSksiJysb)
                || meterReadingReasonSupplyResumed.equals(knkkSksiJysb);
    }

    private void validateSupplyStopServiceContract(Optional<KkytskrRecord> kkytskrRecord, Long svkyNo) {
        if (kkytskrRecord.isPresent() && !Objects.equals(kkytskrRecord.orElseThrow().svkyNo(), svkyNo)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
        }
    }

    private Optional<SvmstkrRecord> findSvmstkrByMtstNo(Long mtstNo) {
        return svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb(
                mtstNo,
                List.of(relatedValid, relatedEndWaiting));
    }

    private SvmstkrRecord findSvmstkrByMtstNoOrThrow(Long mtstNo) {
        return findSvmstkrByMtstNo(mtstNo).orElseThrow(() -> DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                SvmstkrDataAccess.CallSpecs.FIND_BY_MTST_NO_AND_RKKI_KNRN_JTKB_LIST.identifier(),
                new FlatJsonBuilder()
                        .put("mtstNo", mtstNo)
                        .put("rkkiKnrnJtkbList", List.of(relatedValid, relatedEndWaiting))));
    }

    private Optional<SvckyaRecord> findSvckyaBySvkyNo(Long svkyNo) {
        return svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(svkyNo, serviceContractUnderContract)
                .stream()
                .findFirst();
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private static OpeningCancelFeasibilityJudgmentSvmstkrInf toSvmstkrInf(SvmstkrRecord record) {
        if (record == null) {
            return null;
        }
        return new OpeningCancelFeasibilityJudgmentSvmstkrInf(
                record.mtstNo(),
                record.svkyNo(),
                record.skno(),
                record.sknrFlg());
    }

    private static KtmgmkkInf toKtmgmkkInf(KtmgmkkRecord record) {
        return new KtmgmkkInf(record.knsYmd(), record.knsKkaTsno(), record.tksuSvkyNo());
    }

    private static OpeningCancelFeasibilityJudgmentSvckyaInf toSvckyaInf(SvckyaRecord record) {
        if (record == null) {
            return null;
        }
        return new OpeningCancelFeasibilityJudgmentSvckyaInf(
                record.svkyNo(),
                record.skno(),
                record.syoKyoTrno(),
                record.rkkiSttYmd(),
                record.rkkiEndYmd(),
                record.svkySttJysb(),
                record.svkyEndJysb(),
                record.sqtiKiyNo());
    }

    private static OpeningCancelFeasibilityJudgmentKkytskrInf toKkytskrInf(KkytskrRecord record) {
        return new OpeningCancelFeasibilityJudgmentKkytskrInf(record.hesnYmd());
    }
}
