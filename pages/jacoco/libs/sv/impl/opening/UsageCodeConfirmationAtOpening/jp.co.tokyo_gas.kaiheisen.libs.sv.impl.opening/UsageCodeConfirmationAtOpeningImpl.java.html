package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.UsageCodeConfirmationAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.UsageCodeConfirmationAtOpeningOutput;

/**
 * 概要: SV-215（開栓時用途コード確定）の実装。
 * 主な仕様:
 * - 報告ガス用途コードが未設定の場合は導管用途コードを基準に確定する。
 * - 他小売の場合はサービス契約用途コードを null とする。
 * - 報告ガス用途コードが設定済みの場合は報告コードを基準に確定する。
 * 既知の制約:
 * - public 引数の業務妥当性検証は上位層責務とする。
 * - Boolean は Boolean.TRUE.equals(...) で判定する。
 */
public final class UsageCodeConfirmationAtOpeningImpl
        implements UsageCodeConfirmationAtOpening {

    private final String otherRetailCategory;

    /**
     * 概要: SV-215 の判定に必要な開閉栓定数を解決して初期化する。
     * 主な仕様:
     * - KH005 の「他小売」をコンストラクタ初期化時点で fail-fast に解決する。
     * 既知の制約:
     * - 必須定数未登録時はコンストラクタ初期化時点で失敗する。
     *
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public UsageCodeConfirmationAtOpeningImpl(
            KaiheisenConstantService kaiheisenConstantService) {
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.otherRetailCategory = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
    }

    /**
     * 概要: 開栓時の確定用途コードを返却する。
     * 主な仕様:
     * - 報告ガス用途コード未設定時は導管用途コードを基準に返却する。
     * - 他小売以外では代替判定結果フラグを評価し、報告コード設定済みかつ代替=false のときだけ総合判定結果フラグを評価する。
     * - 他小売の場合はサービス契約用途コードを null とする。
     * 既知の制約:
     * - hukkSysShbt、correlationId、systemDatetime は公開引数に含めない。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @param digwrHntFlg 代替判定結果フラグ
     * @param suguHtkFlg 総合判定結果フラグ
     * @param hkkGasYtCd 報告ガス用途コード
     * @param svckyaGasYtCd サービス契約ガス用途コード
     * @param gmtsetdGasYtCd ガスメーター設置場所＜導管＞ガス用途コード
     * @return 確定用途コード
     */
    @Override
    public UsageCodeConfirmationAtOpeningOutput execute(
            String tgTkurKbn,
            Boolean digwrHntFlg,
            Boolean suguHtkFlg,
            String hkkGasYtCd,
            String svckyaGasYtCd,
            String gmtsetdGasYtCd) {
        if (isUnsetUsageCode(hkkGasYtCd)) {
            return determineWhenReportedUsageCodeIsUnset(
                    tgTkurKbn,
                    digwrHntFlg,
                    svckyaGasYtCd,
                    gmtsetdGasYtCd);
        }
        return determineWhenReportedUsageCodeIsSet(
                tgTkurKbn,
                digwrHntFlg,
                suguHtkFlg,
                hkkGasYtCd,
                svckyaGasYtCd);
    }

    /**
     * 概要: 報告ガス用途コード未設定時の確定ロジックを実行する。
     * 主な仕様:
     * - 他小売の場合は導管用途コードと null を返す。
     * - 他小売以外で代替=true の場合は導管用途コードを両項目へ返す。
     * - その他は導管用途コードとサービス契約用途コードを返す。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @param digwrHntFlg 代替判定結果フラグ
     * @param svckyaGasYtCd サービス契約ガス用途コード
     * @param gmtsetdGasYtCd ガスメーター設置場所＜導管＞ガス用途コード
     * @return 確定用途コード
     */
    private UsageCodeConfirmationAtOpeningOutput determineWhenReportedUsageCodeIsUnset(
            String tgTkurKbn,
            Boolean digwrHntFlg,
            String svckyaGasYtCd,
            String gmtsetdGasYtCd) {
        if (isOtherRetail(tgTkurKbn)) {
            return new UsageCodeConfirmationAtOpeningOutput(gmtsetdGasYtCd, null);
        }
        if (Boolean.TRUE.equals(digwrHntFlg)) {
            return new UsageCodeConfirmationAtOpeningOutput(gmtsetdGasYtCd, gmtsetdGasYtCd);
        }
        return new UsageCodeConfirmationAtOpeningOutput(gmtsetdGasYtCd, svckyaGasYtCd);
    }

    /**
     * 概要: 報告ガス用途コード設定済み時の確定ロジックを実行する。
     * 主な仕様:
     * - 他小売の場合は報告用途コードと null を返す。
     * - 他小売以外で代替=true の場合は報告用途コードを両項目へ返す。
     * - 他小売以外で代替=false かつ総合=true の場合は報告用途コードとサービス契約用途コードを返す。
     * - その他は報告用途コードを両項目へ返す。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @param digwrHntFlg 代替判定結果フラグ
     * @param suguHtkFlg 総合判定結果フラグ
     * @param hkkGasYtCd 報告ガス用途コード
     * @param svckyaGasYtCd サービス契約ガス用途コード
     * @return 確定用途コード
     */
    private UsageCodeConfirmationAtOpeningOutput determineWhenReportedUsageCodeIsSet(
            String tgTkurKbn,
            Boolean digwrHntFlg,
            Boolean suguHtkFlg,
            String hkkGasYtCd,
            String svckyaGasYtCd) {
        if (isOtherRetail(tgTkurKbn)) {
            return new UsageCodeConfirmationAtOpeningOutput(hkkGasYtCd, null);
        }
        if (Boolean.TRUE.equals(digwrHntFlg)) {
            return new UsageCodeConfirmationAtOpeningOutput(hkkGasYtCd, hkkGasYtCd);
        }
        if (Boolean.TRUE.equals(suguHtkFlg)) {
            return new UsageCodeConfirmationAtOpeningOutput(hkkGasYtCd, svckyaGasYtCd);
        }
        return new UsageCodeConfirmationAtOpeningOutput(hkkGasYtCd, hkkGasYtCd);
    }

    /**
     * 概要: 報告ガス用途コードが未設定かを判定する。
     * 主な仕様:
     * - null または空白のみを未設定として扱う。
     *
     * @param hkkGasYtCd 報告ガス用途コード
     * @return 未設定なら true
     */
    private boolean isUnsetUsageCode(String hkkGasYtCd) {
        return hkkGasYtCd == null || hkkGasYtCd.isBlank();
    }

    /**
     * 概要: ＴＧ他小売区分が他小売かを判定する。
     * 主な仕様:
     * - KH005 の「他小売」定数と一致する場合のみ true を返す。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @return 他小売なら true
     */
    private boolean isOtherRetail(String tgTkurKbn) {
        return otherRetailCategory.equals(tgTkurKbn);
    }
}
