package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OtherRetailTransportContractNumberDiscrepancyCheckAtOpening;

/** 開栓時他小売託送契約番号相違チェックサービス実装。 */
public final class OtherRetailTransportContractNumberDiscrepancyCheckAtOpeningImpl
        implements OtherRetailTransportContractNumberDiscrepancyCheckAtOpening {

    private final String otherRetailCategory;

    private final String alternativeOpeningReason;

    private final String newlyOpenedTapReason;

    private final String continuedUseReason;

    private final String customerOpeningReason;

    /**
     * 判定に必要な他小売区分と開栓事由定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public OtherRetailTransportContractNumberDiscrepancyCheckAtOpeningImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.otherRetailCategory = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
        this.alternativeOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.ALTERNATIVE_OPENING);
        this.newlyOpenedTapReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.NEWLY_OPENED_TAP);
        this.continuedUseReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.customerOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CUSTOMER_OPENING);
    }

    /**
     * 他小売の開栓報告時に比較対象の託送契約番号を切り替えて相違判定を行う。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @param hkkKasJysb 報告開栓事由種別
     * @param takskyNoKtr 託送契約番号（仮登録）
     * @param takskyNoHntrKrkyk 託送契約番号（本登録・仮解約）
     * @param hkkTakskyNo 報告託送契約番号
     */
    @Override
    public void execute(
            String tgTkurKbn,
            String hkkKasJysb,
            String takskyNoKtr,
            String takskyNoHntrKrkyk,
            String hkkTakskyNo) {
        if (!isOtherRetail(tgTkurKbn)) {
            return;
        }

        if (isTargetOpeningReason(hkkKasJysb)) {
            throwIfTransportContractNumberDiffers(takskyNoKtr, hkkTakskyNo);
        } else {
            throwIfTransportContractNumberDiffers(takskyNoHntrKrkyk, hkkTakskyNo);
        }
    }

    /**
     * TG/他小売区分が他小売かを判定する。
     *
     * @param tgTkurKbn TG/他小売区分
     * @return 他小売の場合は true
     */
    private boolean isOtherRetail(String tgTkurKbn) {
        return otherRetailCategory.equals(tgTkurKbn);
    }

    /**
     * 報告開栓事由種別が仮登録側比較の対象かを判定する。
     *
     * @param hkkKasJysb 報告開栓事由種別
     * @return 仮登録側比較の対象の場合は true
     */
    private boolean isTargetOpeningReason(String hkkKasJysb) {
        return alternativeOpeningReason.equals(hkkKasJysb)
                || newlyOpenedTapReason.equals(hkkKasJysb)
                || continuedUseReason.equals(hkkKasJysb)
                || customerOpeningReason.equals(hkkKasJysb);
    }

    /**
     * 比較対象の託送契約番号が不一致の場合に開栓共通エラーを送出する。
     *
     * @param expectedTransportContractNumber 比較元の託送契約番号
     * @param reportedTransportContractNumber 報告託送契約番号
     */
    private void throwIfTransportContractNumberDiffers(
            String expectedTransportContractNumber,
            String reportedTransportContractNumber) {
        if (!Objects.equals(expectedTransportContractNumber, reportedTransportContractNumber)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100032);
        }
    }
}
