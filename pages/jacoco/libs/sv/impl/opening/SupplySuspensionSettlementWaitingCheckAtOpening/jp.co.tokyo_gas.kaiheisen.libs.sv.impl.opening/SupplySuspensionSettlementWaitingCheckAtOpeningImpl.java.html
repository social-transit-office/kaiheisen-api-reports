package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.SupplySuspensionSettlementWaitingCheckAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KkytskrInf;

/**
 * 概要: SV-204（開栓時供給停止精算待ちチェック）の実装クラス。
 * 主な仕様:
 * - 東京ガス小売事業者、供給停止管理あり、供給停止精算保留、開栓年月日未設定、開栓作業年月日が閉栓年月日+5日超過の全条件成立時に開栓共通エラー 020100030 を送出する。
 * - いずれかの条件が不成立の場合は正常終了する。
 * - 判定に使用する CIRIUS 定数はコンストラクタ初期化時点で解決し、実行時は解決済み値のみを参照する。
 * 既知の制約:
 * - 入力値の trim、blank 補正、null 補正は行わず、呼び出し元から渡された値をそのまま評価する。
 * - RequestContext は共通ルールとの整合のため保持するが、本ロジックでは参照しない。
 */
public final class SupplySuspensionSettlementWaitingCheckAtOpeningImpl
        implements SupplySuspensionSettlementWaitingCheckAtOpening {

    private static final long TARGET_DATE_OFFSET_DAYS = 5L;

    @SuppressWarnings("unused")
    private final RequestContext requestContext;

    private final String tokyoGasRetailerCode;

    private final String supplyStopPaymentPendingStatus;

    /**
     * 概要: 依存と必須定数を初期化するコンストラクタ。
     * 主な仕様:
     * - RequestContext と CiriusConstantService を必須依存として受け取り、null を許容しない。
     * - 東京ガス小売事業者コードと供給停止精算保留コードを初期化時に解決する。
     * 既知の制約:
     * - いずれかの必須定数が未登録の場合はインスタンス生成時点で例外送出となる。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param requestContext リクエストコンテキスト
     */
    public SupplySuspensionSettlementWaitingCheckAtOpeningImpl(
            CiriusConstantService ciriusConstantService,
            RequestContext requestContext) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.tokyoGasRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.supplyStopPaymentPendingStatus = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
    }

         /**
          * 概要: SV-204 の条件判定を実行する。
          * 主な仕様:
          * - 設計 step 3.a の 5 条件がすべて成立した場合に開栓共通エラー 020100030 を送出する。
          * - いずれかの条件が不成立の場合は正常終了する。
          * 既知の制約:
          * - nested DTO や日付が未設定の場合は該当条件を不成立として扱う。
          * - 開栓共通エラー以外の例外変換は行わない。
          *
          * @param kkytskrUmFlg 供給停止管理有無フラグ
          * @param tksuKijsCd 託送供給依頼事業者コード
          * @param kkytskrInf 供給停止管理情報
          * @param kasgYmd 開栓作業年月日
          * @throws KaisenRuleException 全条件成立時に開栓共通エラー 020100030 を送出する
          */
    @Override
    public void execute(
            Boolean kkytskrUmFlg,
            String tksuKijsCd,
            KkytskrInf kkytskrInf,
            LocalDate kasgYmd) {
        if (matchesBusinessRuleViolation(kkytskrUmFlg, tksuKijsCd, kkytskrInf, kasgYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100030);
        }
    }

    private boolean matchesBusinessRuleViolation(
            Boolean kkytskrUmFlg,
            String tksuKijsCd,
            KkytskrInf kkytskrInf,
            LocalDate kasgYmd) {
        return Boolean.TRUE.equals(kkytskrUmFlg)
                && tokyoGasRetailerCode.equals(tksuKijsCd)
                && hasPendingSettlementStatus(kkytskrInf)
                && isOpeningDateMissing(kkytskrInf)
                && isOpeningWorkDateAfterTargetDate(kkytskrInf, kasgYmd);
    }

    private boolean hasPendingSettlementStatus(KkytskrInf kkytskrInf) {
        return kkytskrInf != null
                && supplyStopPaymentPendingStatus.equals(kkytskrInf.kkysSisnJtkb());
    }

    private boolean isOpeningDateMissing(KkytskrInf kkytskrInf) {
        return kkytskrInf != null
                && kkytskrInf.kasYmd() == null;
    }

    private boolean isOpeningWorkDateAfterTargetDate(KkytskrInf kkytskrInf, LocalDate kasgYmd) {
        return kkytskrInf != null
                && kkytskrInf.hesnYmd() != null
                && kasgYmd != null
                && kasgYmd.isAfter(kkytskrInf.hesnYmd().plusDays(TARGET_DATE_OFFSET_DAYS));
    }
}
