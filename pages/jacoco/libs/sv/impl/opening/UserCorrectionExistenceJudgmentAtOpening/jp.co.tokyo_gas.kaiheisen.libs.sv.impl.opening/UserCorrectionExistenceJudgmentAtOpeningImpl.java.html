package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.KyotelkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.KyotelkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.UserCorrectionExistenceJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.UserCorrectionExistenceJudgmentAtOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KyotelkInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.Sv252KiycstKktInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.usercorrectionexistencejudgmentatopening.model.UserCorrectionCstGetHsenjInf;

/**
 * 概要: SV-252（開栓時使用者訂正有無判定）の実装。
 * 主な仕様:
 * - 既定値 false を基準とし、既存客番利用パターンで完全一致しない場合のみ true を返す。
 * - 契約お客さま電話番号＜小売＞情報は DA-062-R001 で取得し、補助TSVの比較条件で評価する。
 * - `hukkSysShbt`、`correlationId`、`systemDatetime` は source 設計上の入力項目だが、本 SV の判定条件では参照しない。
 * 既知の制約:
 * - public メソッド引数に対する null チェックは行わない。
 * - 電話番号比較対象不足は完全一致不成立として扱う。
 */
public final class UserCorrectionExistenceJudgmentAtOpeningImpl
        implements UserCorrectionExistenceJudgmentAtOpening {

    private final KyotelkDataAccess kyotelkDataAccess;

    private final UserCorrectionExistenceJudgmentAtOpeningMatcher matcher;

    private final String useExistingCustomerNumberPattern;

    /**
     * 概要: SV-252 の依存オブジェクトと必須定数を初期化する。
     *
     * @param kyotelkDataAccess 契約お客さま電話番号＜小売＞データアクセス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public UserCorrectionExistenceJudgmentAtOpeningImpl(
            KyotelkDataAccess kyotelkDataAccess,
            KaiheisenConstantService kaiheisenConstantService) {
        this(
                kyotelkDataAccess,
                kaiheisenConstantService,
                new UserCorrectionExistenceJudgmentAtOpeningMatcher());
    }

    UserCorrectionExistenceJudgmentAtOpeningImpl(
            KyotelkDataAccess kyotelkDataAccess,
            KaiheisenConstantService kaiheisenConstantService,
            UserCorrectionExistenceJudgmentAtOpeningMatcher matcher) {
        this.kyotelkDataAccess = Objects.requireNonNull(kyotelkDataAccess, "kyotelkDataAccess");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.matcher = Objects.requireNonNull(matcher, "matcher");
        this.useExistingCustomerNumberPattern = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.TYPE,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.USE_EXISTING_CUSTOMER_NUMBER);
    }

    /**
     * 概要: 使用者訂正有無を判定する。
     *
     * @param cstUpdPtnKb 3x更新パターン区分
     * @param kiycstKktInf 契約お客さま情報（確定）
     * @param cstGetHsenjInf お客さま情報取得情報（閉栓時）
     * @param bldNameHnkuFlg 建物名称変更フラグ
     * @param bldTbgHnkuFlg 建物丁目番地号変更フラグ
     * @return 使用者訂正有無判定結果
     */
    @Override
    public UserCorrectionExistenceJudgmentAtOpeningOutput execute(
            String cstUpdPtnKb,
            Sv252KiycstKktInf kiycstKktInf,
            UserCorrectionCstGetHsenjInf cstGetHsenjInf,
            Boolean bldNameHnkuFlg,
            Boolean bldTbgHnkuFlg) {
        boolean userCorrectionExists = false;
        if (useExistingCustomerNumberPattern.equals(cstUpdPtnKb)) {
            List<KyotelkInf> phoneInfoList = kyotelkDataAccess
                    .findAllByKyoTrkNo(kiycstKktInf.kyoTrkNo())
                    .stream()
                    .map(this::toPhoneInfo)
                    .toList();
            if (!matcher.isFullyMatched(
                    kiycstKktInf,
                    cstGetHsenjInf,
                    phoneInfoList,
                    bldNameHnkuFlg,
                    bldTbgHnkuFlg)) {
                userCorrectionExists = true;
            }
        }
        return new UserCorrectionExistenceJudgmentAtOpeningOutput(userCorrectionExists);
    }

    private KyotelkInf toPhoneInfo(KyotelkRecord kyotelkRecord) {
        return new KyotelkInf(
                kyotelkRecord.kyoTelNo(),
                kyotelkRecord.telShbt(),
                kyotelkRecord.telRrskShbt());
    }
}
