package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.KyotelkInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.Sv252KiycstKktInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.usercorrectionexistencejudgmentatopening.model.KiycstkInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.usercorrectionexistencejudgmentatopening.model.UserCorrectionCstGetHsenjInf;

/**
 * 概要: SV-252 の補助TSV比較条件を評価する matcher。
 * 主な仕様:
 * - 氏名、住所、電話番号、建物変更フラグを完全一致で評価する。
 * - 電話番号は閉栓時情報と DA 取得結果の件数および各要素を比較する。
 */
class UserCorrectionExistenceJudgmentAtOpeningMatcher {

    boolean isFullyMatched(
            Sv252KiycstKktInf confirmedCustomerInfo,
            UserCorrectionCstGetHsenjInf customerInfoAtClosing,
            List<KyotelkInf> phoneInfoList,
            Boolean bldNameHnkuFlg,
            Boolean bldTbgHnkuFlg) {
        KiycstkInf closingCustomerInfo = customerInfoAtClosing.kiycstkInf();
        List<KyotelkInf> closingPhoneInfoList = customerInfoAtClosing.kyotelkInfList();
        return isCustomerInfoMatched(confirmedCustomerInfo, closingCustomerInfo)
                && isPhoneInfoMatched(closingPhoneInfoList, phoneInfoList)
                && Objects.equals(Boolean.FALSE, bldNameHnkuFlg)
                && Objects.equals(Boolean.FALSE, bldTbgHnkuFlg);
    }

    private boolean isCustomerInfoMatched(
            Sv252KiycstKktInf confirmedCustomerInfo,
            KiycstkInf closingCustomerInfo) {
        return Objects.equals(closingCustomerInfo.ctnmKj1(), confirmedCustomerInfo.ctnmKj1())
                && Objects.equals(closingCustomerInfo.ctnmKn1(), confirmedCustomerInfo.ctnmKn1())
                && Objects.equals(
                        closingCustomerInfo.ctnm1HujnShbt(),
                        confirmedCustomerInfo.ctnm1HujnShbt())
                && Objects.equals(
                        closingCustomerInfo.ctnm1HshiShbt(),
                        confirmedCustomerInfo.ctnm1HshiShbt())
                && Objects.equals(closingCustomerInfo.ctnmKj2(), confirmedCustomerInfo.ctnmKj2())
                && Objects.equals(closingCustomerInfo.ctnmKn2(), confirmedCustomerInfo.ctnmKn2())
                && Objects.equals(
                        closingCustomerInfo.ctnm2HujnShbt(),
                        confirmedCustomerInfo.ctnm2HujnShbt())
                && Objects.equals(
                        closingCustomerInfo.ctnm2HshiShbt(),
                        confirmedCustomerInfo.ctnm2HshiShbt())
                && Objects.equals(
                        closingCustomerInfo.ctnmKjhkFlg(),
                        confirmedCustomerInfo.ctnmKjhkFlg())
                && Objects.equals(closingCustomerInfo.yubnNo(), confirmedCustomerInfo.yubnNo())
                && Objects.equals(closingCustomerInfo.cbgSuj1(), confirmedCustomerInfo.cbgSuj1())
                && Objects.equals(closingCustomerInfo.cbgKn2(), confirmedCustomerInfo.cbgKn2())
                && Objects.equals(closingCustomerInfo.cbgKn3(), confirmedCustomerInfo.cbgKn3())
                && Objects.equals(closingCustomerInfo.cbgKn4(), confirmedCustomerInfo.cbgKn4())
                && Objects.equals(closingCustomerInfo.tunoKn(), confirmedCustomerInfo.tunoKn())
                && Objects.equals(closingCustomerInfo.hynKn(), confirmedCustomerInfo.hynKn())
                && Objects.equals(closingCustomerInfo.htuFlg(), confirmedCustomerInfo.htuFlg());
    }

    private boolean isPhoneInfoMatched(
            List<KyotelkInf> closingPhoneInfoList,
            List<KyotelkInf> phoneInfoList) {
        if (closingPhoneInfoList == null || phoneInfoList == null) {
            return closingPhoneInfoList == phoneInfoList;
        }
        if (closingPhoneInfoList.size() != phoneInfoList.size()) {
            return false;
        }
        for (int index = 0; index < closingPhoneInfoList.size(); index++) {
            if (!isPhoneInfoRowMatched(closingPhoneInfoList.get(index), phoneInfoList.get(index))) {
                return false;
            }
        }
        return true;
    }

    private boolean isPhoneInfoRowMatched(KyotelkInf closingPhoneInfo, KyotelkInf phoneInfo) {
        return Objects.equals(closingPhoneInfo.kyoTelNo(), phoneInfo.kyoTelNo())
                && Objects.equals(closingPhoneInfo.telShbt(), phoneInfo.telShbt())
                && Objects.equals(closingPhoneInfo.telRrskShbt(), phoneInfo.telRrskShbt());
    }
}
