/**
 * 概要: SV-248（開栓時受付情報変更通知API連携）のサービス実装。
 * 主な仕様:
 * - 受付インデックス番号が有効な場合のみ SV-087 と 他-042 を連鎖呼び出しする。
 * - 報告システム種別、建物名漢字、サービス契約異動年月日は設計の条件に従って編集する。
 * - `hukkSysShbt`、`systemDatetime` は RequestContext から取得する。
 * 既知の制約:
 * - 戻り値は持たず、下位依存で発生した例外はそのまま上位へ伝播する。
 */
public class ReceptionInfoChangeNotificationApiIntegrationAtOpeningImpl
        implements ReceptionInfoChangeNotificationApiIntegrationAtOpening {

    static final LocalDate CIRIUS_ZERO_DATE = LocalDate.of(1900, 1, 1);

    private static final DateTimeFormatter BASIC_DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private final ApplicationTypeAcquisition applicationTypeAcquisition;

    private final KaiheisenChangeNotificationCiriusAdapter adapter;

    private final RequestContext requestContext;

    private final String tgRetailFlag;

    private final String gasMerchandiseWorkType;

    private final String procedureCompleted;

    private final String workCompletedCompletion;

    private final String presenceFlagNothing;

    private final String customerRegistrationNumberKeyType;

    private final String gasUsageContractNumberKeyType;

    private final String gasMeterInstallationLocationNumberKeyType;

    private final String gasServiceContractType;

    /**
     * 概要: SV-248 の依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - SV-087、他-042 adapter、RequestContext、定数サービスを保持する。
     * - リクエスト編集で使用する必須定数を初期化時点で解決する。
     *
     * @param applicationTypeAcquisition 申込種別取得サービス
     * @param adapter 開閉栓変更通知登録 API adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public ReceptionInfoChangeNotificationApiIntegrationAtOpeningImpl(
            ApplicationTypeAcquisition applicationTypeAcquisition,
            KaiheisenChangeNotificationCiriusAdapter adapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.applicationTypeAcquisition = Objects.requireNonNull(
                applicationTypeAcquisition,
                "applicationTypeAcquisition");
        this.adapter = Objects.requireNonNull(adapter, "adapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.tgRetailFlag = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.TYPE,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.TG_RETAIL);
        this.gasMerchandiseWorkType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MerchandiseWorkType.TYPE,
                ConstantLookupKeys.Cirius.MerchandiseWorkType.GAS);
        this.workCompletedCompletion = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkDetailsCode.TYPE,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkDetailsCode.WORK_COMPLETED_COMPLETION);
        this.presenceFlagNothing = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.NOTHING);
        this.procedureCompleted = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ProcedureStatus.TYPE,
                ConstantLookupKeys.Kaiheisen.ProcedureStatus.PROCEDURE_COMPLETED);
        this.customerRegistrationNumberKeyType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE_OF_CUSTOMER_REGISTRATION_NUMBER);
        this.gasUsageContractNumberKeyType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE_OF_GAS_USAGE_CONTRACT_NUMBER);
        this.gasMeterInstallationLocationNumberKeyType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE_OF_GAS_METER_INSTALLATION_LOCATION_NUMBER);
        this.gasServiceContractType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ServiceContractType.TYPE,
                ConstantLookupKeys.Kaiheisen.ServiceContractType.GAS);
    }

    /**
     * 概要: 開栓時受付情報変更通知 API 連携を実行する。
     * 主な仕様:
     * - 受付インデックス番号が null または空文字の場合は何もせず終了する。
      * - 利用する RequestContext の必須値を確認し、SV-087 の戻り値で request を補完して他-042 を呼び出す。
     */
    @Override
    public void execute(
            String utidxNo,
            String kasJysb,
            String sgyAnknNo,
            Long kktSvkyNo,
            Long gasMtstNo,
            ReceptionInfoChangeKiycstKktInf kiycstKktInf,
            String hujnHnsgCtnmKn1,
            String hujnHnsgCtnmKj1,
            String hujnHnsgCtnmKn2,
            String hujnHnsgCtnmKj2,
            String cbg,
            ReceptionInfoChangeTatemonKktInf tatemonKktInf,
            ReceptionInfoChangeNwkseigInf nwkseigInf,
            String sagyoSyaName,
            Boolean digwrHntFlg,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg,
            ReceptionInfoChangeKseflInf kseflInf,
            LocalDate kasgYmd) {
        if (utidxNo == null || utidxNo.isEmpty()) {
            return;
        }
        String requestContextReportSystemType = requireHukkSysShbt();
        LocalDateTime systemDatetime = requireSystemDatetime();
        String reportSystemType = resolveReportSystemType(
                kasJysb,
                kseflInf,
                requestContextReportSystemType);
        String mkSbt = applicationTypeAcquisition.execute(kasJysb, null, reportSystemType).mkSbt();
        adapter.registerNotification(buildRequest(
                utidxNo,
                sgyAnknNo,
                kktSvkyNo,
                gasMtstNo,
                kiycstKktInf,
                hujnHnsgCtnmKn1,
                hujnHnsgCtnmKj1,
                hujnHnsgCtnmKn2,
                hujnHnsgCtnmKj2,
                cbg,
                tatemonKktInf,
                nwkseigInf,
                sagyoSyaName,
                digwrHntFlg,
                syoriHnt4XFlg,
                syoriHntFkktFlg,
                kasJysb,
                kasgYmd,
                systemDatetime,
                mkSbt));
    }

    private KaiheisenChangeNotificationRequest buildRequest(
            String utidxNo,
            String sgyAnknNo,
            Long kktSvkyNo,
            Long gasMtstNo,
            ReceptionInfoChangeKiycstKktInf kiycstKktInf,
            String hujnHnsgCtnmKn1,
            String hujnHnsgCtnmKj1,
            String hujnHnsgCtnmKn2,
            String hujnHnsgCtnmKj2,
            String cbg,
            ReceptionInfoChangeTatemonKktInf tatemonKktInf,
            ReceptionInfoChangeNwkseigInf nwkseigInf,
            String sagyoSyaName,
            Boolean digwrHntFlg,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg,
            String kasJysb,
            LocalDate kasgYmd,
            LocalDateTime systemDatetime,
            String mkSbt) {
        LocalDate serviceContractChangeDate = resolveServiceContractChangeDate(
                kasJysb,
                digwrHntFlg,
                syoriHnt4XFlg,
                syoriHntFkktFlg,
                kasgYmd,
                systemDatetime.toLocalDate());
        String buildingName = resolveBuildingName(tatemonKktInf);
        String administrativeCode = kiycstKktInf == null ? null : kiycstKktInf.gkkCd();
        return new KaiheisenChangeNotificationRequest(
                new KaiheisenChangeNotificationRequest.NotificationInfo(
                        tgRetailFlag,
                        utidxNo,
                        gasMerchandiseWorkType,
                        mkSbt,
                        sgyAnknNo,
                        toBigDecimal(kktSvkyNo),
                        procedureCompleted,
                        toBigDecimal(gasMtstNo),
                        kiycstKktInf == null ? null : toBigDecimal(kiycstKktInf.kyoTrkNo()),
                        toBigDecimal(kktSvkyNo),
                        hujnHnsgCtnmKn1,
                        "",
                        nullToEmpty(hujnHnsgCtnmKj1),
                        "",
                        extractPrefectureCode(administrativeCode),
                        extractMunicipalityCode(administrativeCode),
                        kiycstKktInf == null ? null : kiycstKktInf.mchCd(),
                        kiycstKktInf == null ? null : kiycstKktInf.cmazCd(),
                        cbg,
                        nullToEmpty(buildingName),
                        suppressBlanksOrEmpty(kiycstKktInf == null ? null : kiycstKktInf.tunoKn()),
                        suppressBlanksOrEmpty(kiycstKktInf == null ? null : kiycstKktInf.hynKn()),
                        workCompletedCompletion,
                        nullToEmpty(hujnHnsgCtnmKn2),
                        nullToEmpty(hujnHnsgCtnmKj2),
                        nullToEmpty(nwkseigInf == null ? null : nwkseigInf.nwknKj()),
                        nullToEmpty(sagyoSyaName),
                        toYmd(systemDatetime.toLocalDate()),
                        presenceFlagNothing,
                        customerRegistrationNumberKeyType,
                        kiycstKktInf == null ? null : toBigDecimal(kiycstKktInf.kyoTrkNo()),
                        gasServiceContractType,
                        toYmd(serviceContractChangeDate),
                        gasUsageContractNumberKeyType,
                        toBigDecimal(kktSvkyNo),
                        gasMeterInstallationLocationNumberKeyType,
                        toBigDecimal(gasMtstNo)));
    }

    private String resolveReportSystemType(
            String kasJysb,
            ReceptionInfoChangeKseflInf kseflInf,
            String requestContextReportSystemType) {
        String errorFollowReportSystemType = kseflInf == null ? null : kseflInf.hukkSysShbt();
        if (ConstantLookupKeys.Cirius.OpeningReasonType.SUPPLY_RESUMED.equals(kasJysb)
                && errorFollowReportSystemType != null
                && !errorFollowReportSystemType.isEmpty()) {
            return errorFollowReportSystemType;
        }
        return requestContextReportSystemType;
    }

    private String resolveBuildingName(ReceptionInfoChangeTatemonKktInf tatemonKktInf) {
        if (tatemonKktInf == null || !Boolean.TRUE.equals(tatemonKktInf.bldKjhkFlg())) {
            return null;
        }
        return tatemonKktInf.bldnKj();
    }

    private LocalDate resolveServiceContractChangeDate(
            String kasJysb,
            Boolean digwrHntFlg,
            Boolean syoriHnt4XFlg,
            Boolean syoriHntFkktFlg,
            LocalDate kasgYmd,
            LocalDate systemDate) {
        if (Boolean.TRUE.equals(digwrHntFlg)) {
            return kasgYmd;
        }
        if (ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE.equals(kasJysb)
                && Boolean.TRUE.equals(syoriHnt4XFlg)
                && Boolean.TRUE.equals(syoriHntFkktFlg)) {
            return kasgYmd;
        }
        return systemDate;
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private static String extractPrefectureCode(String administrativeCode) {
        if (administrativeCode == null || administrativeCode.length() < 2) {
            return null;
        }
        return administrativeCode.substring(0, 2);
    }

    private static String extractMunicipalityCode(String administrativeCode) {
        if (administrativeCode == null || administrativeCode.length() < 5) {
            return null;
        }
        return administrativeCode.substring(administrativeCode.length() - 3);
    }

    private static String suppressBlanks(String value) {
        if (value == null) {
            return null;
        }
        StringBuilder builder = new StringBuilder(value.length());
        value.codePoints().forEach(codePoint -> {
            if (!Character.isWhitespace(codePoint) && codePoint != '　') {
                builder.appendCodePoint(codePoint);
            }
        });
        return builder.toString();
    }

    private static String suppressBlanksOrEmpty(String value) {
        return nullToEmpty(suppressBlanks(value));
    }

    private static String nullToEmpty(String value) {
        return value == null ? "" : value;
    }

    private static BigDecimal toBigDecimal(Long value) {
        return value == null ? null : BigDecimal.valueOf(value);
    }

    private static Integer toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        if (CIRIUS_ZERO_DATE.equals(value)) {
            return 0;
        }
        return Integer.valueOf(BASIC_DATE_FORMATTER.format(value));
    }
}
