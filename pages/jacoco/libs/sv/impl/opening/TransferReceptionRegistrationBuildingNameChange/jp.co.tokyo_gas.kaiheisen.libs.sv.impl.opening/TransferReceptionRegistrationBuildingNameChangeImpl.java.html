    @Override
    public void execute(
            String bldNo,
            LocalDate uktkYmd,
            Integer uketkTime,
            String hkkEmpNo,
            LocalDate hkkKsenSaymd,
            String tgTkurKbn,
            String hkkNameKj,
            Long kktTksuSvkyNo,
            Long tksuSvkyNo,
            String duknIsrgCd,
            KiycstKktInf kiycstKktInf,
            Boolean suguHntFlg,
            Long kktSvkyNo,
            LocalDate hesnYmd,
            Long kktKyoTrkNo,
            Long kyoTrkNo,
            Long hkkKsenShjSu,
            String kngmStbJtkb,
            String gmtThkgJysb,
            String stkbKbn,
            String shtiKskbNo,
            String ckrtGmscJtkb,
            String kkyuGmtKbn,
            String gmb) {
        requireHukkSysShbt();
        requireCorrelationId();
        requireSystemDatetime();

        TransferReceptionCommonEditOutput commonEditOutput = transferReceptionCommonEdit.execute(
                bldNo,
                uktkYmd,
                uketkTime,
                duknIsrgCd,
                kiycstKktInf,
                hkkEmpNo,
                hkkNameKj,
                kktTksuSvkyNo,
                hkkKsenSaymd,
                tgTkurKbn,
                buildingNameChange,
                notApplicableInquiryCode);

        int comprehensiveMeterCount = 0;
        if (Boolean.TRUE.equals(suguHntFlg)) {
            List<String> relatedStatuses = List.of(relatedValid, relatedEndWaiting);
            long count = svmstkrDataAccess.findBySvkyNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                    kktSvkyNo,
                    hkkKsenSaymd,
                    relatedStatuses);
            if (count == 0L) {
                throw DataInconsistencySupport.create(
                        svmstkrDataAccess.tableName(),
                        SvmstkrDataAccess.CallSpecs
                                .COUNT_BY_SVKY_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("svkyNo", kktSvkyNo)
                                .put("knskKjnYmd", hkkKsenSaymd)
                                .put("rkkiKnrnJtkbList", relatedStatuses));
            }
            comprehensiveMeterCount = Math.toIntExact(count);
        }

        String ghpMeterInstallationCode = ghpMeterInstallationCodeEdit
                .execute(ckrtGmscJtkb, kkyuGmtKbn)
                .ghpmSchCd();

        TransferReceptionRegistrationIdouktkInsInf registrationInfo = toRegistrationInfo(
                commonEditOutput.idouktkInsInf(),
                buildCorrectionInfo(
                        hkkKsenSaymd,
                        tksuSvkyNo,
                        kktTksuSvkyNo,
                        comprehensiveMeterCount,
                        hesnYmd,
                        kyoTrkNo,
                        kktKyoTrkNo,
                        gmb,
                        hkkKsenShjSu,
                        kngmStbJtkb,
                        gmtThkgJysb,
                        stkbKbn,
                        shtiKskbNo,
                        ghpMeterInstallationCode));
        transferReceptionRegistration.execute(registrationInfo);
    }

    private String buildCorrectionInfo(
            LocalDate openingWorkDate,
            Long currentContractNo,
            Long confirmedContractNo,
            int comprehensiveMeterCount,
            LocalDate closingDate,
            Long currentCustomerNo,
            Long confirmedCustomerNo,
            String gasMeterBusinessNo,
            Long openingInstructionCount,
            String equipmentStatus,
            String removalReason,
            String meterReadingDayCategory,
            String meterReadingSequenceNo,
            String ghpMeterInstallationCode) {
        // docs/application_design/service/source/SV-237/異動訂正情報編集 の P/C/G 変換規則に従う。
        StringBuilder builder = new StringBuilder();
        appendPackedDate(builder, openingWorkDate, 9);
        appendPackedLong(builder, currentContractNo, 11);
        appendPackedLong(builder, confirmedContractNo, 11);
        appendChar(builder, installmentSettlementRequiredCode, 1);
        appendChar(builder, "", 1);
        appendPacked(builder, Integer.toString(comprehensiveMeterCount), 3);
        appendChar(builder, temporaryPaymentReasonNotSetCode, 1);
        appendPackedDate(builder, closingDate, 9);
        appendPackedLong(builder, currentCustomerNo, 11);
        appendPackedLong(builder, confirmedCustomerNo, 11);
        appendPacked(builder, gasMeterBusinessNo, 9);
        appendPacked(builder, gasMeterBusinessNo, 9);
        appendChar(builder, "", 3);
        appendChar(builder, "", 3);
        appendPacked(builder, "0", 9);
        appendPacked(builder, "0", 7);
        appendPackedLong(builder, openingInstructionCount, 9);
        appendChar(builder, gasMeterOpeningStatusCode, 1);
        appendChar(builder, equipmentStatus, 1);
        appendChar(builder, removalReason, 1);
        appendChar(builder, "", 2);
        appendChar(builder, "", 2);
        appendChar(builder, meterReadingDayCategory, 2);
        appendChar(builder, meterReadingSequenceNo, 2);
        appendChar(builder, meterReadingDayCategory, 2);
        appendChar(builder, meterReadingSequenceNo, 2);
        appendChar(builder, ghpMeterInstallationCode, 1);
        appendChar(builder, "", 11);
        appendChar(builder, "", 2);
        appendChar(builder, "", 12);
        appendChar(builder, "", 1);
        appendChar(builder, "", 1);
        appendPacked(builder, "0", 11);
        appendChar(builder, noConceptUsageCode, 5);
        return builder.toString();
    }

    private static void appendPackedDate(StringBuilder builder, LocalDate value, int digits) {
        appendPacked(builder, value.format(java.time.format.DateTimeFormatter.BASIC_ISO_DATE), digits);
    }

    private static void appendPackedLong(StringBuilder builder, Long value, int digits) {
        appendPacked(builder, Long.toString(value), digits);
    }

    private static void appendPacked(StringBuilder builder, String rawValue, int digits) {
        String normalizedValue = rawValue.trim();
        String paddedDigits = normalizedValue.length() >= digits
                ? normalizedValue
                : "0".repeat(digits - normalizedValue.length()) + normalizedValue;
        String nibbles = (digits % 2 == 0 ? "0" : "") + paddedDigits + "C";
        builder.append(nibbles.toUpperCase());
    }

    private static void appendChar(StringBuilder builder, String value, int length) {
        byte[] fixedLengthBytes = toFixedLengthEbcdicBytes(value, length);
        builder.append(HexFormat.of().formatHex(fixedLengthBytes).toUpperCase());
    }

    private static byte[] toFixedLengthEbcdicBytes(String value, int length) {
        byte[] fixedLengthBytes = new byte[length];
        Arrays.fill(fixedLengthBytes, EBCDIC_SPACE);

        String normalized = value == null ? "" : value;
        if (normalized.isEmpty()) {
            return fixedLengthBytes;
        }

        int acceptedEndIndex = 0;
        for (int index = 0; index < normalized.length();) {
            int nextIndex = index + Character.charCount(normalized.codePointAt(index));
            byte[] encodedCandidate = normalized.substring(0, nextIndex).getBytes(EBCDIC_CHARSET);
            if (encodedCandidate.length > length) {
                break;
            }
            acceptedEndIndex = nextIndex;
            index = nextIndex;
        }

        byte[] encodedPrefix = normalized.substring(0, acceptedEndIndex).getBytes(EBCDIC_CHARSET);
        System.arraycopy(encodedPrefix, 0, fixedLengthBytes, 0, encodedPrefix.length);
        return fixedLengthBytes;
    }

    private static Charset resolveEbcdicCharset() {
        for (String candidate : List.of("Cp930", "x-IBM930", "IBM930")) {
            try {
                return Charset.forName(candidate);
            } catch (IllegalCharsetNameException | UnsupportedCharsetException ignored) {
                // try next alias
            }
        }
        throw new IllegalStateException("Cp930 compatible charset is required");
    }
