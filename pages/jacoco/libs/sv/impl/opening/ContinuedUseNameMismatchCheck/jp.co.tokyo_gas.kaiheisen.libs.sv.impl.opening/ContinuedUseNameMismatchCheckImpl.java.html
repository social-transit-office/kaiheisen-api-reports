package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContinuedUseNameMismatchCheck;

/** 継続使用名義不一致チェックサービス実装。 */
public class ContinuedUseNameMismatchCheckImpl implements ContinuedUseNameMismatchCheck {

    private final String continuedUseReason;

    private final String supplyResumedReason;

    /**
     * 継続使用系の開栓事由定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ContinuedUseNameMismatchCheckImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.continuedUseReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.supplyResumedReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.SUPPLY_RESUMED);
    }

    /**
     * 継続使用名義不一致条件を判定し、該当時は開栓共通エラーを送出する。
     *
     * @param kzkMgitKknFlg 継続使用名義確認フラグ
     * @param snyTkmtFlg 直列専用通過メーターフラグ
     * @param kiyUmFlg 契約有無フラグ
     * @param ctnmKn1Hsen お客さま名カナ１（閉栓時）
     * @param ctnmKn1Ksen お客さま名カナ１（開栓時）
     * @param hkkKasJysb 報告開栓事由種別
     * @throws KaisenRuleException 継続使用名義不一致に該当する場合
     */
    @Override
        public void execute(
            Boolean kzkMgitKknFlg,
            Boolean snyTkmtFlg,
            Boolean kiyUmFlg,
            String ctnmKn1Hsen,
            String ctnmKn1Ksen,
            String hkkKasJysb) {
        boolean isContinuationCheckDisabled = !kzkMgitKknFlg;
        boolean isSerialMeterFlagDisabled = !snyTkmtFlg;
        boolean hasContract = kiyUmFlg;

        if (isContinuationCheckDisabled
                && isSerialMeterFlagDisabled
                && hasContract
                && !Objects.equals(ctnmKn1Hsen, ctnmKn1Ksen)
                && (continuedUseReason.equals(hkkKasJysb)
                        || supplyResumedReason.equals(hkkKasJysb))) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100023);
        }

    }
}
