    /**
     * 開栓時の契約情報更新を実行し、確定契約情報を返却する。
     *
     * <p>専用通過メーター経路では元メーター情報から確定契約情報を組み立てる。</p>
     *
     * <p>通常経路の戻り値のサービス契約情報は、設計書どおり入力の契約情報.サービス契約情報を返す。</p>
     */
    @Override
    public ContractInfoUpdateNoAlternativeRetailAtOpeningOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            String kasJysb,
            LocalDate kktTkysYmd,
            Long kktSyoKyoTrno,
            Long kktSvkyNo,
            Long kktTksuSvkyNo,
            String ckrtGmscJtkb,
            Boolean snyTkmtFlg,
            Boolean digwrHntFlg,
            Boolean suguHtkFlg,
            String kktSvckyaGasYtCd,
            String kktSziKziKbn,
            ContractInfoUpdateNoAlternativeRetailAtOpeningTskgmstInf tskgmstInf,
            ContractInfoUpdateNoAlternativeRetailAtOpeningKiyInf kiyInf,
            ContractInfoUpdateNoAlternativeRetailAtOpeningMtgInf mtgInf,
            List<SnyInf> snyInfList,
            LocalDate ksenHkkKnuKigenYmd) {
        if (Boolean.TRUE.equals(snyTkmtFlg)) {
            return buildOutputFromOriginMeterInfo(mtgInf);
        }

        if (kiyInf == null || kiyInf.svckyaInf() == null) {
            return buildOutputFromAvailableContractInfo(kiyInf, mtgInf);
        }

        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId());

        ContractInfoUpdateNoAlternativeRetailAtOpeningSvckyaInf currentSvckyaInf = kiyInf.svckyaInf();
        LocalDate kktSzirHtyoYmd = consumptionTaxRateDeterminationDateConfirmationAtOpening
                .execute(
                        gasMtstNo,
                        kktSvkyNo,
                        kasgYmd,
                        kasJysb,
                        currentSvckyaInf.szirHtyoYmd(),
                        digwrHntFlg,
                        currentSvckyaInf.kkysSisnJtkb(),
                        ksenHkkKnuKigenYmd)
                .kktSzirHtyoYmd();
        updateCurrentServiceContract(
                currentSvckyaInf,
                kktSzirHtyoYmd,
                kktSvckyaGasYtCd,
                kktSziKziKbn,
                auditContext);

        boolean suguFkktFlg = false;
        boolean suguKbnUpdFlg = false;
        ContractInfoUpdateNoAlternativeRetailAtOpeningSvmstkrInf currentSvmstkrInf = kiyInf.svmstkrInf();

        if (isComprehensiveCorrectionTarget(ckrtGmscJtkb, suguHtkFlg)
                && currentSvmstkrInf != null) {
            if (chargeCalculationRelatedEnd.equals(currentSvmstkrInf.rkkiKnrnJtkb())) {
                reviveComprehensiveRelation(
                        gasMtstNo,
                        kktSvkyNo,
                        kktTkysYmd,
                        currentSvmstkrInf,
                        auditContext);
                suguFkktFlg = true;
            } else if (tskgmstInf != null
                    && transportationApplicableEnds.equals(tskgmstInf.tksuSvkkJtkb())) {
                SvmstkrRecord relatedRecord =
                        findFirstRelatedRecord(kktSvkyNo, gasMtstNo, kasgYmd);
                contractInfoUpdateNoAlternativePipelineAtOpening.execute(
                        gasMtstNo,
                        kktTksuSvkyNo,
                        kktTkysYmd,
                        kktSyoKyoTrno,
                        kktSvkyNo,
                        ckrtGmscJtkb,
                        snyInfList);
                if (relatedRecord != null && nonComprehensive.equals(relatedRecord.suguKbn())) {
                    updateCurrentMeterGeneralCategory(
                            relatedRecord,
                            generalRepresentative,
                            auditContext);
                    suguKbnUpdFlg = true;
                }
            }
        }

        List<SvckyaRecord> waitingContracts =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(kktSvkyNo, waitingForContractApplication);
        if (!waitingContracts.isEmpty()) {
            updateWaitingServiceContract(
                    waitingContracts.get(0),
                    kktSzirHtyoYmd,
                    kktSvckyaGasYtCd,
                    auditContext);
        }

        return new ContractInfoUpdateNoAlternativeRetailAtOpeningOutput(
                toSvckyaKktInf(
                        kiyInf.svckyaInf(),
                        kktSzirHtyoYmd,
                        kiyInf.svckyaInf().recUpdJnjNo()),
                toSvsihkrKktInf(kiyInf.svsihkrInf()),
                toSihakyKktInf(kiyInf.sihakyInf()),
                suguFkktFlg,
                suguKbnUpdFlg);
    }

    private ContractInfoUpdateNoAlternativeRetailAtOpeningOutput buildOutputFromAvailableContractInfo(
            ContractInfoUpdateNoAlternativeRetailAtOpeningKiyInf kiyInf,
            ContractInfoUpdateNoAlternativeRetailAtOpeningMtgInf mtgInf) {
        ContractInfoUpdateNoAlternativeRetailAtOpeningDknKurKiyInf originKiyInf =
                mtgInf == null ? null : mtgInf.dknKurKiyInf();
        return new ContractInfoUpdateNoAlternativeRetailAtOpeningOutput(
                kiyInf != null && kiyInf.svckyaInf() != null
                        ? toSvckyaKktInf(kiyInf.svckyaInf())
                        : toSvckyaKktInf(originKiyInf == null ? null : originKiyInf.svckyaInf(), null),
                kiyInf != null && kiyInf.svsihkrInf() != null
                        ? toSvsihkrKktInf(kiyInf.svsihkrInf())
                        : toSvsihkrKktInf(originKiyInf == null ? null : originKiyInf.svsihkrInf(), null),
                kiyInf != null && kiyInf.sihakyInf() != null
                        ? toSihakyKktInf(kiyInf.sihakyInf())
                        : toSihakyKktInf(originKiyInf == null ? null : originKiyInf.sihakyInf(), null),
                false,
                false);
    }

    private boolean isComprehensiveCorrectionTarget(String ckrtGmscJtkb, Boolean suguHtkFlg) {
        return !dedicatedGasMeterStatus.equals(ckrtGmscJtkb) && Boolean.TRUE.equals(suguHtkFlg);
    }

    private SvmstkrRecord findFirstRelatedRecord(Long kktSvkyNo, Long gasMtstNo, LocalDate kasgYmd) {
        List<SvmstkrRecord> records = svmstkrDataAccess.findAllBySvkyNoAndMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                kktSvkyNo,
                gasMtstNo,
                kasgYmd,
                chargeCalculationRelatedValid);
        return records.isEmpty() ? null : records.get(0);
    }

    private void reviveComprehensiveRelation(
            Long gasMtstNo,
            Long kktSvkyNo,
            LocalDate kktTkysYmd,
            ContractInfoUpdateNoAlternativeRetailAtOpeningSvmstkrInf currentSvmstkrInf,
            AuditContext auditContext) {
        int nextSkno = svmstkrDataAccess.findBySvkyNoAndMtstNoAndRkkiKnrnJtkb(
                        kktSvkyNo,
                        gasMtstNo,
                        chargeCalculationRelatedInvalid)
                .map(SvmstkrRecord::skno)
                .orElse(0) + 1;
        svmstkrDataAccess.insert(
                new SvmstkrWriteRecord(
                        gasMtstNo,
                        kktSvkyNo,
                        nextSkno,
                        synthesized,
                        currentSvmstkrInf.sknrFlg(),
                        chargeCalculationRelatedValid,
                        kktTkysYmd,
                        MAX_DATE,
                        null),
                auditContext);
    }
