package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.AlternativeJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.AlternativeJudgmentAtOpeningOutput;

/**
 * 概要: 開栓時代替判定サービスを実装する。
 * 主な仕様: CIRIUS 定数から開栓事由種別を解決し、直列専用通過メーターフラグ、開栓事由種別、処理判定_4x_フラグに応じて代替判定結果を返却する。
 * 既知の制約: 必須定数の不足はコンストラクタ初期化時点で fail-fast し、Boolean 引数はアンボクシングにより null を許容しない。
 */
public class AlternativeJudgmentAtOpeningImpl implements AlternativeJudgmentAtOpening {

    private final String continuedUseReason;

    private final Set<String> falseResultReasons;

    /**
     * 概要: 開栓事由判定に使用する CIRIUS 定数を初期化する。
     * 主な仕様: 継続使用と false 判定対象 4 事由の定数値を `CiriusConstantService` から解決して保持する。
     * 既知の制約: 必須依存が null の場合と必須定数が取得できない場合は、コンストラクタ初期化時点で fail-fast する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public AlternativeJudgmentAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.continuedUseReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.falseResultReasons = Set.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.SUPPLY_RESUMED),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.START_SUPPLYING_TO_OTHER_PARTIES),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.SECURITY_RESTORATION),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.DISASTER_RECOVERY));
    }

    /**
     * 概要: 開栓時の代替判定を行う。
     * 主な仕様: 直列専用通過メーターフラグを最優先で評価し、継続使用時は処理判定_4x_フラグ、その他の特定 4 事由は false 固定、それ以外は主関連有無フラグを維持して返却する。
     * 既知の制約: 必須定数不足はコンストラクタで fail-fast 済みであることを前提とし、Boolean 引数はアンボクシングにより null の場合 `NullPointerException` となる。
     *
     * @param kasJysb 開栓事由種別
     * @param snyTkmtFlg 直列専用通過メーターフラグ
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param sknrUmFlg 主関連有無フラグ
     * @return 代替判定結果
     */
    @Override
    public AlternativeJudgmentAtOpeningOutput execute(
            String kasJysb,
            Boolean snyTkmtFlg,
            Boolean syoriHnt4XFlg,
            Boolean sknrUmFlg) {
        boolean seriesDedicatedMeterFlag = snyTkmtFlg;
        boolean fourXJudgmentFlag = syoriHnt4XFlg;

        if (seriesDedicatedMeterFlag) {
            return falseResult();
        }

        if (continuedUseReason.equals(kasJysb)) {
            if (!fourXJudgmentFlag) {
                return trueResult(sknrUmFlg);
            }
            return falseResult();
        }

        if (falseResultReasons.contains(kasJysb)) {
            return falseResult();
        }

        return trueResult(sknrUmFlg);
    }

    /**
     * 概要: false 固定の判定結果を生成する。
     * 主な仕様: 代替判定結果フラグと主関連有無フラグの両方に `false` を設定して返却する。
     * 既知の制約: 呼出元の入力値は参照せず、常に固定値を返却する。
     *
     * @return false 固定の判定結果
     */
    private static AlternativeJudgmentAtOpeningOutput falseResult() {
        return new AlternativeJudgmentAtOpeningOutput(Boolean.FALSE, Boolean.FALSE);
    }

    /**
     * 概要: true 判定の結果を生成する。
     * 主な仕様: 代替判定結果フラグに `true` を設定し、主関連有無フラグには入力値をそのまま設定して返却する。
     * 既知の制約: 主関連有無フラグの null は補正せず、そのまま出力 DTO へ格納する。
     *
     * @param sknrUmFlg 主関連有無フラグ
     * @return true 判定の結果
     */
    private static AlternativeJudgmentAtOpeningOutput trueResult(Boolean sknrUmFlg) {
        return new AlternativeJudgmentAtOpeningOutput(Boolean.TRUE, sknrUmFlg);
    }
}
