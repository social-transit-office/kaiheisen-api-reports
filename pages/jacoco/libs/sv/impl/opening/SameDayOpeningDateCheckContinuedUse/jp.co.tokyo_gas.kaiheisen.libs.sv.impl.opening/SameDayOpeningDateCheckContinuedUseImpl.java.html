package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.SameDayOpeningDateCheckContinuedUse;

/** 同日開栓日チェック（継続使用）サービス実装。 */
public class SameDayOpeningDateCheckContinuedUseImpl
        implements SameDayOpeningDateCheckContinuedUse {

    /**
     * 閉栓日、開栓作業日、当初ガスメーター取付日、料金計算終了年月日を判定し、
     * 同日開栓エラー条件に該当する場合は開栓共通エラー（KaisenRuleException）を送出する。
     *
     * @param hesnYmd 閉栓年月日
     * @param kasgYmd 開栓作業年月日
     * @param tusoGmtkYmd 当初ガスメーター取付年月日
     * @param rkkiEndYmd 料金計算終了年月日
     * @throws KaisenRuleException 開栓共通エラー 020100047 に該当する場合
     */
    @Override
    public void execute(
            LocalDate hesnYmd,
            LocalDate kasgYmd,
            LocalDate tusoGmtkYmd,
            LocalDate rkkiEndYmd) {
        boolean sameDayReopening = hesnYmd.isEqual(kasgYmd) && !tusoGmtkYmd.isEqual(hesnYmd);
        boolean sameDayBillingEnd = rkkiEndYmd.isEqual(kasgYmd);

        if (sameDayReopening || sameDayBillingEnd) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100047);
        }

    }
}
