/**
 * 概要: SV-250（開栓時請求回収基盤連携情報編集）の実装。
 * 主な仕様:
 * - DA-062-R001 で契約お客さま電話番号＜小売＞を取得し、先頭 2 レコードを支払者・使用者電話情報へ反映する。
 * - 固定値と CIRIUS 定数を組み合わせて、請求回収基盤連携用の 3 つの出力 DTO を編集する。
 * 既知の制約:
 * - 設計に基づき、電話情報の 3 件目以降は参照しない。
 * - 電話番号は null 時に空文字へ補完し、電話連絡先種別と電話種別は null または空文字時に既定値へ補完する。
 */
public final class BillingCollectionPlatformIntegrationEditAtOpeningImpl
        implements BillingCollectionPlatformIntegrationEditAtOpening {

    private static final String EMPTY_PHONE_NUMBER = "";

    private static final Long FIXED_GAS_METER_LOCATION_NO = 11111111111L;

    private final KyotelkDataAccess kyotelkDataAccess;

    private final String blankContactType;

    private final String noPhoneType;

    private final String paymentManagementTransferBusinessCode;

    private final String contractManagementTransferBusinessCode;

    private final String tokyoGasNetworkOperatorCode;

    private final String noServiceSuspensionCancellationReason;

    private final String gasFeeContractType;

    /**
     * 概要: SV-250 の依存と必須定数を初期化する。
     * 主な仕様:
     * - DA と CIRIUS 定数取得サービスを必須依存として保持する。
     * - 戻り値編集で使用する固定 CIRIUS 定数をコンストラクタで fail-fast 解決する。
     *
     * @param kyotelkDataAccess 契約お客さま電話番号＜小売＞ DA
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public BillingCollectionPlatformIntegrationEditAtOpeningImpl(
            KyotelkDataAccess kyotelkDataAccess,
             CiriusConstantService ciriusConstantService) {
        this.kyotelkDataAccess = Objects.requireNonNull(kyotelkDataAccess, "kyotelkDataAccess");
        Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.blankContactType = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ContactType.TYPE,
                ConstantLookupKeys.Cirius.ContactType.BLANK);
        this.noPhoneType = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.PhoneType.TYPE,
                ConstantLookupKeys.Cirius.PhoneType.NO_PHONE);
        this.paymentManagementTransferBusinessCode = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_PAYMENT_MANAGEMENT_TRANSFER);
        this.contractManagementTransferBusinessCode = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_CONTRACT_MANAGEMENT_TRANSFER);
        this.tokyoGasNetworkOperatorCode = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.NwOperatorCode.TYPE,
                ConstantLookupKeys.Cirius.NwOperatorCode.TOKYO_GAS_CO_LTD);
        this.noServiceSuspensionCancellationReason = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.NO_SERVICE_SUSPENSION_OR_CANCELLATION);
        this.gasFeeContractType = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ContractType__N2.TYPE,
                ConstantLookupKeys.Cirius.ContractType__N2.GAS_FEE);
    }

    /**
     * 概要: SV-250 を実行し、請求回収基盤連携向けの 3 つの出力 DTO を編集する。
     * 主な仕様:
     * - `kyoTrkNo` を用いて DA-062-R001 から電話情報を取得し、支払者・使用者電話情報へ反映する。
     * - `gasMtstNo` と各種契約入力、固定値、CIRIUS 定数を組み合わせて戻り値を組み立てる。
     *
     * @param gmtsetdKsenSaymdInf ガスメーター設置場所＜導管＞情報（開栓作業日時点）
     * @param kiycstKktInf 契約お客さま情報（確定）
     * @param svckyaKktInf サービス契約情報（確定）
     * @param svsihkrKktInf サービス契約・支払契約関連情報（確定）
     * @param sihakyKktInf 支払契約情報（確定）
     * @param syoTisiUmFlg 使用者訂正有無フラグ
     * @return 開栓時請求回収基盤連携情報編集結果
     */
    @Override
    public BillingCollectionPlatformIntegrationEditAtOpeningOutput execute(
            BillingCollectionPlatformIntegrationGmtsetdKsenSaymdInf gmtsetdKsenSaymdInf,
            BillingCollectionPlatformIntegrationKiycstKktInf kiycstKktInf,
            BillingCollectionPlatformIntegrationSvckyaKktInf svckyaKktInf,
            BillingCollectionPlatformIntegrationSvsihkrKktInf svsihkrKktInf,
            BillingCollectionPlatformIntegrationSihakyKktInf sihakyKktInf,
            Boolean syoTisiUmFlg) {
        List<KyotelkRecord> phoneRecords = kyotelkDataAccess.findAllByKyoTrkNo(kiycstKktInf.kyoTrkNo());
        return new BillingCollectionPlatformIntegrationEditAtOpeningOutput(
                buildControlInfo(
                    sihakyKktInf.sikKrno(),
                    svckyaKktInf.sqtiKiyNo(),
                    svckyaKktInf.svkyNo(),
                    svckyaKktInf.skno(),
                    sihakyKktInf.sihKiyNo(),
                    sihakyKktInf.skno(),
                    svckyaKktInf.rkkiSttYmd(),
                    svckyaKktInf.rkkiEndYmd(),
                        syoTisiUmFlg),
                buildPaymentInfo(
                    phoneRecords,
                    sihakyKktInf.shsyKyoTrno(),
                    sihakyKktInf.tyenYmd(),
                    sihakyKktInf.tkysYmd()),
                buildContractInfo(
                        phoneRecords,
                    gmtsetdKsenSaymdInf.gasMtstNo(),
                    svsihkrKktInf.ikskGrpNo(),
                    svsihkrKktInf.tdssSbtShbt(),
                    svsihkrKktInf.gsptShbt(),
                    svsihkrKktInf.gsnKhFlg(),
                    svckyaKktInf.rkkiEndYmd(),
                    svckyaKktInf.svcRyksYmd(),
                    svsihkrKktInf.sikTmngShbt(),
                    sihakyKktInf.tkysYmd(),
                    svckyaKktInf.sziKziKbn(),
                    svckyaKktInf.szirHtyoYmd()));
    }

    /**
     * 概要: 請求回収基盤連携用制御情報を編集する。
     * 主な仕様:
     * - 設計書の固定値・入力値に従い `SkkbRkySgjhInf` を組み立てる。
     * - 請求回収連携グループ管理番号は null、ガスメーター設置場所番号は固定値 11111111111 を設定する。
     *
     * @param sikKrno 請求管理番号
     * @param sqtiKiyNo 請求対象契約番号
     * @param svkyNo サービス契約番号
     * @param svkySkno サービス契約世代管理番号
     * @param sihKiyNo 支払契約番号
     * @param sihKiySkno 支払契約世代管理番号
     * @param rkkiSttYmd 料金計算開始年月日
     * @param rkkiEndYmd 料金計算終了年月日
     * @param syoTisiUmFlg 使用者訂正有無フラグ
     * @return 請求回収基盤連携用制御情報
     */
    private SkkbRkySgjhInf buildControlInfo(
            Long sikKrno,
            String sqtiKiyNo,
            Long svkyNo,
            Integer svkySkno,
            Long sihKiyNo,
            Integer sihKiySkno,
            LocalDate rkkiSttYmd,
            LocalDate rkkiEndYmd,
            Boolean syoTisiUmFlg) {
        return new SkkbRkySgjhInf(
                null,
                sikKrno,
                sqtiKiyNo,
                svkyNo,
                svkySkno,
                sihKiyNo,
                sihKiySkno,
                FIXED_GAS_METER_LOCATION_NO,
                rkkiSttYmd,
                rkkiEndYmd,
                syoTisiUmFlg);
    }

    /**
     * 概要: 請求対象支払管理登録依頼情報を編集する。
     * 主な仕様:
     * - DA-062-R001 の先頭 2 レコードを支払者電話情報へ反映する。
     * - 電話番号は null 時に空文字へ補完し、電話連絡先種別と電話種別は null または空文字時に既定値へ補完する。
     *
     * @param phoneRecords 契約お客さま電話番号＜小売＞
     * @param shsyKyoTrno 支払者契約お客さま登録番号
     * @param tyenYmd 適用終了年月日
     * @param tkysYmd 適用開始年月日
     * @return 請求対象支払管理登録依頼情報
     */
    private SktsSihTrirInf buildPaymentInfo(
            List<KyotelkRecord> phoneRecords,
            Long shsyKyoTrno,
            LocalDate tyenYmd,
            LocalDate tkysYmd) {
        KyotelkRecord firstPhoneRecord = phoneRecordAt(phoneRecords, 0);
        KyotelkRecord secondPhoneRecord = phoneRecordAt(phoneRecords, 1);
        return new SktsSihTrirInf(
                null,
                null,
                shsyKyoTrno,
                null,
                null,
                contactTypeOf(firstPhoneRecord),
                contactTypeOf(secondPhoneRecord),
                phoneNumberOf(firstPhoneRecord),
                phoneNumberOf(secondPhoneRecord),
                phoneTypeOf(firstPhoneRecord),
                phoneTypeOf(secondPhoneRecord),
                paymentManagementTransferBusinessCode,
                tyenYmd,
                tkysYmd,
                tokyoGasNetworkOperatorCode);
    }

    /**
     * 概要: 請求対象契約管理登録依頼情報を編集する。
     * 主な仕様:
     * - DA-062-R001 の先頭 2 レコードを使用者電話情報へ反映する。
     * - 設計書で固定指定された契約種類、異動・作業履歴業務コード、NW 事業者コードを設定する。
     *
     * @param phoneRecords 契約お客さま電話番号＜小売＞
     * @param gasMtstNo ガスメーター設置場所番号
     * @param ikskGrpNo 一括請求グループ番号
     * @param tdssSbtShbt 単独請求先識別種別
     * @param gsptShbt 合算パターン種別
     * @param gsnKhFlg 合算可否フラグ
     * @param rkkiEndYmd 料金計算終了年月日
     * @param svcRyksYmd サービス利用開始年月日
     * @param sikTmngShbt 請求タイミング種別
     * @param tkysYmd 適用開始年月日
     * @param sziKziKbn 消費税課税区分
     * @param szirHtyoYmd 消費税率判定用年月日
     * @return 請求対象契約管理登録依頼情報
     */
    private SktsKykTrirInf buildContractInfo(
            List<KyotelkRecord> phoneRecords,
            Long gasMtstNo,
            Long ikskGrpNo,
            String tdssSbtShbt,
            String gsptShbt,
            Boolean gsnKhFlg,
            LocalDate rkkiEndYmd,
            LocalDate svcRyksYmd,
            String sikTmngShbt,
            LocalDate tkysYmd,
            String sziKziKbn,
            LocalDate szirHtyoYmd) {
        KyotelkRecord firstPhoneRecord = phoneRecordAt(phoneRecords, 0);
        KyotelkRecord secondPhoneRecord = phoneRecordAt(phoneRecords, 1);
        return new SktsKykTrirInf(
                noServiceSuspensionCancellationReason,
                null,
                ikskGrpNo,
                null,
                null,
                contactTypeOf(firstPhoneRecord),
                contactTypeOf(secondPhoneRecord),
                phoneNumberOf(firstPhoneRecord),
                phoneNumberOf(secondPhoneRecord),
                phoneTypeOf(firstPhoneRecord),
                phoneTypeOf(secondPhoneRecord),
                gasMtstNo,
                tdssSbtShbt,
                null,
                gsptShbt,
                gsnKhFlg,
                gasFeeContractType,
                rkkiEndYmd,
                svcRyksYmd,
                null,
                contractManagementTransferBusinessCode,
                sikTmngShbt,
                tkysYmd,
                tokyoGasNetworkOperatorCode,
                sziKziKbn,
                szirHtyoYmd);
    }

    /**
     * 概要: 指定位置の電話レコードを取得する。
     * 主な仕様:
     * - 対象位置にレコードが存在しない場合は null を返す。
     *
     * @param phoneRecords 契約お客さま電話番号＜小売＞一覧
     * @param index 取得位置
     * @return 指定位置のレコード、存在しない場合は null
     */
    private KyotelkRecord phoneRecordAt(List<KyotelkRecord> phoneRecords, int index) {
        if (phoneRecords.size() <= index) {
            return null;
        }
        return phoneRecords.get(index);
    }

    /**
     * 概要: 電話レコードから連絡先種別を取得する。
     * 主な仕様:
     * - 電話連絡先種別が null または空文字の場合は CIRIUS 定数【18000 0】のコード 0 を返す。
     * - レコードが存在しない場合も CIRIUS 定数【18000 0】のコード 0 を返す。
     *
     * @param record 契約お客さま電話番号＜小売＞レコード
     * @return 連絡先種別
     */
    private String contactTypeOf(KyotelkRecord record) {
        if (record == null) {
            return blankContactType;
        }
        return applyNullOrEmptyFallback(record.telRrskShbt(), blankContactType);
    }

    /**
     * 概要: 電話レコードから電話番号を取得する。
     * 主な仕様:
     * - 電話番号が null、またはレコードが存在しない場合は空文字を返す。
     *
     * @param record 契約お客さま電話番号＜小売＞レコード
     * @return 電話番号
     */
    private String phoneNumberOf(KyotelkRecord record) {
        if (record == null) {
            return EMPTY_PHONE_NUMBER;
        }
        return applyNullOrEmptyFallback(record.kyoTelNo(), EMPTY_PHONE_NUMBER);
    }

    /**
     * 概要: 電話レコードから電話種別を取得する。
     * 主な仕様:
     * - 電話種別が null または空文字の場合は CIRIUS 定数【14500 0】のコード 0 を返す。
     * - レコードが存在しない場合も CIRIUS 定数【14500 0】のコード 0 を返す。
     *
     * @param record 契約お客さま電話番号＜小売＞レコード
     * @return 電話種別
     */
    private String phoneTypeOf(KyotelkRecord record) {
        if (record == null) {
            return noPhoneType;
        }
        return applyNullOrEmptyFallback(record.telShbt(), noPhoneType);
    }

    /**
     * 概要: null または空文字を既定値へ補完する。
     * 主な仕様:
     * - 値が null または空文字の場合は既定値を返す。
     * - それ以外の値はそのまま返す。
     *
     * @param value 判定対象値
     * @param fallbackValue 補完値
     * @return 補完後の値
     */
    private String applyNullOrEmptyFallback(String value, String fallbackValue) {
        if (value == null || value.isEmpty()) {
            return fallbackValue;
        }
        return value;
    }
