/**
 * 概要: 開栓時契約お客さま更新パターン判定サービスを実装する。
 * 主な仕様:
 * - 直列専用通過を最優先で評価し、次に他小売経路、継続使用経路、通常経路の順で返却パターンを決定する。
 * - 他-048 の採番が必要な分岐では、導管向け・小売向けの採番 support を呼び出して確定使用者契約お客さま登録番号へ反映する。
 * - 設計上 `RequestContext` は参照せず、分岐に必要な値は公開引数のみから判定する。
 * 既知の制約:
 * - Boolean 引数は null の場合に false として扱い、分岐で参照する必須 DTO は参照時点で fail-fast する。
 * - 小売契約お客さま情報が欠ける場合は、利用可能な導管情報から確定情報を補完して継続する。
 * - 導管情報から契約お客さま情報（確定）を組み立てる場合、契約者名系の小売専用項目は null を設定する。
 */
public final class ContractCustomerUpdatePatternJudgmentAtOpeningImpl
        implements ContractCustomerUpdatePatternJudgmentAtOpening {

    private final NumberingAllInOneSupport numberingAllInOneSupport;

    private final String otherRetailCategory;

    private final String continuedUseOpeningReason;

    private final String newCustomerPattern;

    private final String useExistingCustomerNumberPattern;

    private final String newNumberAssignedPattern;

    /**
     * コンストラクタ。
     *
     * @param numberingAllInOneSupport 他-048 採番ゲートウェイ
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public ContractCustomerUpdatePatternJudgmentAtOpeningImpl(
            NumberingAllInOneSupport numberingAllInOneSupport,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.numberingAllInOneSupport = Objects.requireNonNull(numberingAllInOneSupport, "numberingAllInOneSupport");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.otherRetailCategory = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
        this.continuedUseOpeningReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.newCustomerPattern = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.TYPE,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.NEW_CUSTOMER);
        this.useExistingCustomerNumberPattern = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.TYPE,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.USE_EXISTING_CUSTOMER_NUMBER);
        this.newNumberAssignedPattern = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.TYPE,
                ConstantLookupKeys.Kaiheisen.N3xUpdatePattern.NEW_NUMBER_ASSIGNED);
    }

    /** {@inheritDoc} */
    @Override
    public ContractCustomerUpdatePatternJudgmentAtOpeningOutput execute(
            String tgTkurKbn,
            String kasJysb,
            Boolean snyTkmtFlg,
            Boolean digwrHntFlg,
            Boolean syoriHnt3XFlg,
            Boolean syoriHntFkktFlg,
            Boolean cstTkssvkyTrkZmFlg,
            CstHsenjInf cstHsenjInf,
            CstKsenjInf cstKsenjInf,
            MtgInf mtgInf) {
        boolean serialMeterFlag = Boolean.TRUE.equals(snyTkmtFlg);
        boolean alternativeJudgmentResult = Boolean.TRUE.equals(digwrHntFlg);
        boolean process3xFlag = Boolean.TRUE.equals(syoriHnt3XFlg);
        boolean revivalFlag = Boolean.TRUE.equals(syoriHntFkktFlg);
        boolean registeredTransportContract = Boolean.TRUE.equals(cstTkssvkyTrkZmFlg);

        if (serialMeterFlag) {
            return isOtherRetail(tgTkurKbn)
                    ? output(useExistingCustomerNumberPattern, mtgInf.kiycstdInf())
                    : output(useExistingCustomerNumberPattern, mtgInf.kiycstkInf(), mtgInf.kiycstdInf());
        }

        if (isOtherRetail(tgTkurKbn)) {
            if (alternativeJudgmentResult) {
                if (registeredTransportContract) {
                    return output(
                            newCustomerPattern,
                            numberingAllInOneSupport.issuePipelineContractCustomerRegistrationNumber(),
                            null,
                            null);
                }
                return output(
                        newNumberAssignedPattern,
                        cstKsenjInf.kiycstdInf());
            }
            return output(useExistingCustomerNumberPattern, cstHsenjInf.kiycstdInf());
        }

        if (isContinuedUse(kasJysb)) {
            if (process3xFlag) {
                return output(
                        useExistingCustomerNumberPattern,
                        cstHsenjInf.kiycstkInf(),
                        cstHsenjInf.kiycstdInf());
            }
            Long numberedRetailCustomerNumber = numberingAllInOneSupport.issueRetailContractCustomerRegistrationNumber();
            if (revivalFlag) {
                return output(
                        newCustomerPattern,
                        numberedRetailCustomerNumber,
                        toConfirmedContractCustomerInfo(cstHsenjInf.kiycstkInf()),
                        toConfirmedPipelineContractCustomerInfo(cstHsenjInf.kiycstdInf()));
            }
            return output(newCustomerPattern, numberedRetailCustomerNumber, null, null);
        }

        if (alternativeJudgmentResult) {
            if (registeredTransportContract) {
                return output(
                        newCustomerPattern,
                        numberingAllInOneSupport.issueRetailContractCustomerRegistrationNumber(),
                        null,
                        null);
            }
            return output(
                    newNumberAssignedPattern,
                    cstKsenjInf.kiycstkInf().kyoTrkNo(),
                    toConfirmedContractCustomerInfo(cstKsenjInf.kiycstkInf()),
                    null);
        }

        if (cstHsenjInf == null) {
            if (cstKsenjInf == null) {
                return output(useExistingCustomerNumberPattern, (Long) null, null, null);
            }
            if (cstKsenjInf.kiycstkInf() == null) {
                return output(useExistingCustomerNumberPattern, cstKsenjInf.kiycstdInf());
            }
            return output(
                    useExistingCustomerNumberPattern,
                    cstKsenjInf.kiycstkInf(),
                    cstKsenjInf.kiycstdInf());
        }

        if (cstHsenjInf.kiycstkInf() == null) {
            return output(useExistingCustomerNumberPattern, cstHsenjInf.kiycstdInf());
        }

        return output(
                useExistingCustomerNumberPattern,
                cstHsenjInf.kiycstkInf(),
                cstHsenjInf.kiycstdInf());
    }

    /**
     * 概要: 他小売区分を判定する。
     * 主な仕様: 開閉栓定数で解決した KH005 の他小売コードと入力値を比較する。
     * 既知の制約: 入力値が null の場合は false と同等ではなく単純に不一致扱いとする。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @return 他小売の場合 true
     */
    private boolean isOtherRetail(String tgTkurKbn) {
        return otherRetailCategory.equals(tgTkurKbn);
    }

    /**
     * 概要: 開栓事由が継続使用かを判定する。
     * 主な仕様: CIRIUS 定数で解決した 04900 の継続使用コードと入力値を比較する。
     * 既知の制約: 入力値が null の場合は単純に不一致扱いとする。
     *
     * @param kasJysb 開栓事由種別
     * @return 継続使用の場合 true
     */
    private boolean isContinuedUse(String kasJysb) {
        return continuedUseOpeningReason.equals(kasJysb);
    }

    /**
     * 概要: 導管情報ベースの既存客番利用/新規採番済出力を組み立てる。
     * 主な仕様: 導管情報を契約お客さま情報（確定）と契約お客さま＜導管＞情報（確定）の両方へ反映する。
     * 既知の制約: 契約者名系の小売専用項目は null を設定する。
     *
     * @param pattern 3x更新パターン区分
     * @param pipelineInfo 契約お客さま＜導管＞情報
     * @return 判定結果
     */
    private ContractCustomerUpdatePatternJudgmentAtOpeningOutput output(
            String pattern,
            KiycstdInf pipelineInfo) {
        return output(
                pattern,
                pipelineInfo.kyoTrkNo(),
                toConfirmedContractCustomerInfo(pipelineInfo),
                toConfirmedPipelineContractCustomerInfo(pipelineInfo));
    }

    /**
     * 概要: 小売情報と導管情報を併せて出力を組み立てる。
     * 主な仕様: 契約お客さま情報（確定）は小売情報から、導管情報（確定）は導管情報から構築する。
     * 既知の制約: 小売情報が null の場合は参照時点で fail-fast する。
     *
     * @param pattern 3x更新パターン区分
     * @param retailInfo 契約お客さま＜小売＞情報
     * @param pipelineInfo 契約お客さま＜導管＞情報
     * @return 判定結果
     */
    private ContractCustomerUpdatePatternJudgmentAtOpeningOutput output(
            String pattern,
            KiycstkInf retailInfo,
            KiycstdInf pipelineInfo) {
        return output(
                pattern,
                retailInfo.kyoTrkNo(),
                toConfirmedContractCustomerInfo(retailInfo),
                toConfirmedPipelineContractCustomerInfo(pipelineInfo));
    }

    /**
     * 概要: 完成済みの各項目から出力 DTO を生成する。
     * 主な仕様: 3x更新パターン区分、確定番号、確定情報をそのまま出力へ格納する。
     * 既知の制約: null 補完や追加変換は行わない。
     *
     * @param pattern 3x更新パターン区分
     * @param confirmedCustomerRegistrationNumber 確定使用者契約お客さま登録番号
     * @param confirmedContractCustomerInfo 契約お客さま情報（確定）
     * @param confirmedPipelineContractCustomerInfo 契約お客さま＜導管＞情報（確定）
     * @return 判定結果
     */
    private static ContractCustomerUpdatePatternJudgmentAtOpeningOutput output(
            String pattern,
            Long confirmedCustomerRegistrationNumber,
            KiycstKktInf confirmedContractCustomerInfo,
            KiycstdKktInf confirmedPipelineContractCustomerInfo) {
        return new ContractCustomerUpdatePatternJudgmentAtOpeningOutput(
                pattern,
                confirmedCustomerRegistrationNumber,
                confirmedContractCustomerInfo,
                confirmedPipelineContractCustomerInfo);
    }

    /**
