package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.CompanyUseCheckAtOpening;

/**
 * 概要: 開栓時社用チェックサービス実装。
 * 主な仕様:
 * - 契約終了時からの継続使用で社用に該当する場合は開栓共通エラー 020100046 を送出する。
 * - 判定で使用する CIRIUS 定数はコンストラクタで fail-fast に解決する。
 * 既知の制約:
 * - RequestContext 値は本 SV の public 引数へ含めず、処理でも参照しない。
 */
public class CompanyUseCheckAtOpeningImpl implements CompanyUseCheckAtOpening {

    private final String tokyoGasRetailerCode;

    private final String continuedUseReason;

    private final String relocationClosureReason;

    private final String unusedClosureReason;

    private final String supplyOutageReason;

    private final String securityClosureReason;

    private final String companyUseDivision;

    /**
     * 概要: 判定に必要な CIRIUS 定数を初期化する。
     * 主な仕様:
     * - 判定に必要な小売事業者コード、開栓事由、閉栓事由、社用区分を解決する。
     * 既知の制約:
     * - 必須定数不足時はコンストラクタ初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public CompanyUseCheckAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.tokyoGasRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.continuedUseReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.relocationClosureReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE);
        this.unusedClosureReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.UNUSED_CLOSURE);
        this.supplyOutageReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
        this.securityClosureReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE);
        this.companyUseDivision = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.Corporate.TYPE,
                ConstantLookupKeys.Cirius.Corporate.FOR_COMPANY_USE);
    }

    /**
     * 概要: 契約終了時からの継続使用で社用に該当するかを判定する。
     * 主な仕様:
     * - 社用確認フラグが false、東京ガス、契約終了、継続使用、対象閉栓事由、社用区分の全条件成立時に開栓共通エラーを送出する。
        * - 条件不成立時は何も返さずに正常終了する。
     * 既知の制約:
     * - Boolean 引数が null の場合は Java の unboxing により NullPointerException となる。
     * @param syayoKknFlg 社用確認フラグ
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param kiyUmFlg 契約有無フラグ
     * @param hkkKasJysb 報告開栓事由種別
     * @param hesnJysb 閉栓事由種別
     * @param syaKbn 社用区分
     * @throws KaisenRuleException 社用継続不可条件に該当する場合
     */
    @Override
        public void execute(
            Boolean syayoKknFlg,
            String tksuKijsCd,
            Boolean kiyUmFlg,
            String hkkKasJysb,
            String hesnJysb,
            String syaKbn) {
        boolean isUnconfirmedCompanyUse = !syayoKknFlg;
        boolean isTokyoGasRetailer = tokyoGasRetailerCode.equals(tksuKijsCd);
        boolean isNoContract = !kiyUmFlg;
        boolean isContinuedUse = continuedUseReason.equals(hkkKasJysb);
        boolean isTargetClosureReason = isTargetClosureReason(hesnJysb);
        boolean isCompanyUse = companyUseDivision.equals(syaKbn);

        if (isUnconfirmedCompanyUse
                && isTokyoGasRetailer
                && isNoContract
                && isContinuedUse
                && isTargetClosureReason
                && isCompanyUse) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100046);
        }

    }

    /**
     * 概要: 開栓共通エラー対象となる閉栓事由かを判定する。
     * 主な仕様:
     * - 移転閉栓、不使用閉栓、供給停止、保安閉栓を対象とする。
     * 既知の制約:
     * - 入力が null の場合は false として扱う。
     * @param hesnJysb 閉栓事由種別
     * @return 対象閉栓事由であれば true
     */
    private boolean isTargetClosureReason(String hesnJysb) {
        return relocationClosureReason.equals(hesnJysb)
                || unusedClosureReason.equals(hesnJysb)
                || supplyOutageReason.equals(hesnJysb)
                || securityClosureReason.equals(hesnJysb);
    }
}
