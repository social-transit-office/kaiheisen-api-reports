package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tkssvky.TkssvkyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tkssvky.write.TkssvkyWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.write.TskgmstWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContractInfoUpdateWithAlternativePipelineAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SnyInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.TaksmtrKtrInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.DeemedMeterReadingRegistration;

/**
 * 開栓時契約情報更新処理（代替あり）（導管）サービス実装。
 */
public class ContractInfoUpdateWithAlternativePipelineAtOpeningImpl
        implements ContractInfoUpdateWithAlternativePipelineAtOpening {

    static final String SERVICE_USER_ID = "KHSS224";

    private static final String DEEMED_METER_READING_REQUIRED = "1";

    private static final int DEEMED_METER_READING_SEQUENCE = 0;

    private static final int GENERATION_NUMBER = 1;

    private static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final DeemedMeterReadingRegistration deemedMeterReadingRegistration;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String otherRetailCategory;

    private final String originalGasMeterStatus;

    private final String tokyoGasCode;

    private final String applyingContractStatus;

    private final String newStartReasonType;

    private final String consignmentContractType;

    private final String applyingRelatedStatus;

    private final String closedGasMeterStatus;

    /**
     * コンストラクタ。
     *
     * @param tkssvkyDataAccess TKSSVKY データアクセス
     * @param tskgmstDataAccess TSKGMST データアクセス
     * @param deemedMeterReadingRegistration SV-015
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ContractInfoUpdateWithAlternativePipelineAtOpeningImpl(
            TkssvkyDataAccess tkssvkyDataAccess,
            TskgmstDataAccess tskgmstDataAccess,
            DeemedMeterReadingRegistration deemedMeterReadingRegistration,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.deemedMeterReadingRegistration =
                Objects.requireNonNull(deemedMeterReadingRegistration, "deemedMeterReadingRegistration");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.otherRetailCategory = RequiredConstantResolver.requireKaiheisenConstant(
                this.kaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.tokyoGasCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.applyingContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.newStartReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType.NEW);
        this.consignmentContractType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractType.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractType.CONSIGNMENT_CONTRACT);
        this.applyingRelatedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.closedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
    }

    /**
     * 主契約の託送サービス契約・ガスメーター設置場所関連を登録し、元ガスメーターの場合は専用メーター分を反復登録する。
     * 他小売以外かつ元ガスメーターの場合は、各専用メーターに対してみなし検針登録サービスも呼び出す。
     *
     * @param tgTkurKbn ＴＧ他小売区分
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kktTksuSvkyNo 確定託送サービス契約番号
     * @param kktTkysYmd 確定適用開始年月日
     * @param kktSyoKyoTrno 確定使用者契約お客さま登録番号
     * @param kktSvkyNo 確定サービス契約番号
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param snyInfList 専用メーター情報リスト
     * @param taksmtrKtrInf 託送メータ情報（仮登録）
     */
    @Override
    public void execute(
            String tgTkurKbn,
            Long gasMtstNo,
            Long kktTksuSvkyNo,
            LocalDate kktTkysYmd,
            Long kktSyoKyoTrno,
            Long kktSvkyNo,
            String ckrtGmscJtkb,
            List<SnyInf> snyInfList,
            TaksmtrKtrInf taksmtrKtrInf) {
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId());
        String supplierCode = resolveSupplierCode(tgTkurKbn, taksmtrKtrInf);

        insertTkssvky(kktTksuSvkyNo, kktTkysYmd, kktSyoKyoTrno, supplierCode, auditContext);
        insertTskgmst(gasMtstNo, kktTksuSvkyNo, kktTkysYmd, auditContext);

        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return;
        }

        for (SnyInf snyInf : normalizeSnyInfList(snyInfList)) {
            insertTkssvky(
                    snyInf.kktTksuSvkyNo(),
                    kktTkysYmd,
                    kktSyoKyoTrno,
                    supplierCode,
                    auditContext);
            insertTskgmst(
                    snyInf.gasMtstNo(),
                    snyInf.kktTksuSvkyNo(),
                    kktTkysYmd,
                    auditContext);
            if (!otherRetailCategory.equals(tgTkurKbn)) {
                deemedMeterReadingRegistration.execute(
                        DEEMED_METER_READING_REQUIRED,
                        kktTkysYmd,
                        snyInf.gasMtstNo(),
                        DEEMED_METER_READING_SEQUENCE,
                        kktSvkyNo,
                        closedGasMeterStatus);
            }
        }
    }

    private void insertTkssvky(
            Long tksuSvkyNo,
            LocalDate tkysYmd,
            Long syoKyoTrno,
            String supplierCode,
            AuditContext auditContext) {
        tkssvkyDataAccess.insert(
                new TkssvkyWriteRecord(
                        tksuSvkyNo,
                        GENERATION_NUMBER,
                        applyingContractStatus,
                        newStartReasonType,
                        null,
                        tkysYmd,
                        MAX_DATE,
                        null,
                        consignmentContractType,
                        supplierCode,
                        Boolean.TRUE,
                        syoKyoTrno),
                auditContext);
    }

    private void insertTskgmst(
            Long gasMtstNo,
            Long tksuSvkyNo,
            LocalDate tkysYmd,
            AuditContext auditContext) {
        tskgmstDataAccess.insert(
                new TskgmstWriteRecord(
                        gasMtstNo,
                        tksuSvkyNo,
                        GENERATION_NUMBER,
                        applyingRelatedStatus,
                        tkysYmd,
                        MAX_DATE),
                auditContext);
    }

    private String resolveSupplierCode(String tgTkurKbn, TaksmtrKtrInf taksmtrKtrInf) {
        if (!otherRetailCategory.equals(tgTkurKbn)) {
            return tokyoGasCode;
        }
        if (taksmtrKtrInf == null) {
            throw new IllegalStateException(
                    "taksmtrKtrInf is required when tgTkurKbn indicates other retail: tgTkurKbn="
                            + tgTkurKbn);
        }
        String supplierCode = isBlank(taksmtrKtrInf.lskkJgysCd())
                ? taksmtrKtrInf.gasKjgsCd()
                : taksmtrKtrInf.lskkJgysCd();
        if (isBlank(supplierCode)) {
            throw new IllegalStateException(
                    "Supplier code is required when tgTkurKbn indicates other retail: tgTkurKbn="
                            + tgTkurKbn
                            + ", missingFields=[lskkJgysCd, gasKjgsCd]");
        }
        return supplierCode;
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    private List<SnyInf> normalizeSnyInfList(List<SnyInf> snyInfList) {
        return snyInfList == null ? List.of() : snyInfList;
    }

    private boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}
