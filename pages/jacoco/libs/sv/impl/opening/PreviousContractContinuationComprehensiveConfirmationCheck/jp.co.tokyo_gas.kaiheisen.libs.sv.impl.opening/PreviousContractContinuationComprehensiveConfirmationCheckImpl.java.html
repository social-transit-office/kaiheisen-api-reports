package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.PreviousContractContinuationComprehensiveConfirmationCheck;

/**
 * 概要: 前契約継続（総合）確認チェックサービス実装。
 * 主な仕様:
 * - 前契約継続確認フラグが false かつ継続使用扱いコードが総合ワーニング対象の場合のみ詳細判定へ進む。
 * - 契約終了判定と異動・作業履歴業務コード（導管）の組み合わせに応じて業務例外を送出する。
 * 既知の制約:
 * - RequestContext 値は public 引数へ含めず、本クラスでも参照しない。
 */
public final class PreviousContractContinuationComprehensiveConfirmationCheckImpl
        implements PreviousContractContinuationComprehensiveConfirmationCheck {

    private final String comprehensiveWarningTarget;

    private final String contractEnded;

    private final String openingRegistration;

    private final String openingSystemProcessing;

    /**
     * 概要: 判定に必要な定数値を初期化する。
     * 主な仕様:
     * - KH007、J0566、J0497 の必須定数を constructor で fail-fast 解決する。
     * 既知の制約:
     * - 必須定数不足時は初期化時点で失敗する。
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public PreviousContractContinuationComprehensiveConfirmationCheckImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.comprehensiveWarningTarget = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.TYPE,
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode
                        .COMPREHENSIVE_WARNING_TARGET);
        this.contractEnded = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.END_OF_CONTRACT);
        this.openingRegistration = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.OPENING_REGISTRATION);
        this.openingSystemProcessing = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                        .OPENING_SYSTEM_PROCESSING);
    }

    /**
     * 概要: 前契約継続に関する総合確認を行う。
     * 主な仕様:
     * - 判定入口を満たさない場合は正常終了する。
     * - 判定入口を満たす場合は step 設計どおりに業務例外を送出する。
     * @param zkyKzkKknFlg 前契約継続確認フラグ
     * @param kzksyAtkCd 継続使用扱いコード
     * @param svkyJtkb サービス契約状態区分
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     */
    @Override
    public void execute(
            Boolean zkyKzkKknFlg,
            String kzksyAtkCd,
            String svkyJtkb,
            Boolean syoriHnt4XFlg,
            String duknIsrgCd) {
        if (!shouldEvaluate(zkyKzkKknFlg, kzksyAtkCd)) {
            return;
        }

        if (isContractEnded(svkyJtkb)) {
            evaluateContractEndedBranch(syoriHnt4XFlg, duknIsrgCd);
            return;
        }

        evaluateNotContractEndedBranch(duknIsrgCd);
    }

    /**
     * 概要: 詳細判定に進むべきかを判定する。
     * 主な仕様:
     * - 前契約継続確認フラグが false かつ継続使用扱いコードが総合ワーニング対象のときのみ true を返す。
     * @param zkyKzkKknFlg 前契約継続確認フラグ
     * @param kzksyAtkCd 継続使用扱いコード
     * @return 詳細判定に進む場合 true
     */
    private boolean shouldEvaluate(Boolean zkyKzkKknFlg, String kzksyAtkCd) {
        return Boolean.FALSE.equals(zkyKzkKknFlg)
                && comprehensiveWarningTarget.equals(kzksyAtkCd);
    }

    /**
     * 概要: 契約終了状態かどうかを判定する。
     * 主な仕様:
     * - J0566 の契約終了コードとの一致で判定する。
     * @param svkyJtkb サービス契約状態区分
     * @return 契約終了の場合 true
     */
    private boolean isContractEnded(String svkyJtkb) {
        return contractEnded.equals(svkyJtkb);
    }

    /**
     * 概要: 契約終了側の分岐を評価する。
     * 主な仕様:
     * - 処理判定_4x_フラグが true で、異動業務コードが開栓登録系以外なら 020110051 を送出する。
     * - 処理判定_4x_フラグが true で、異動業務コードが開栓登録系なら 020100051 を送出する。
     * - 処理判定_4x_フラグが false の場合は正常終了する。
     * @param syoriHnt4XFlg 処理判定_4x_フラグ
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     */
    private void evaluateContractEndedBranch(Boolean syoriHnt4XFlg, String duknIsrgCd) {
        if (!Boolean.TRUE.equals(syoriHnt4XFlg)) {
            return;
        }
        if (isNonOpeningWorkCode(duknIsrgCd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020110051);
        }
        throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100051);
    }

    /**
     * 概要: 契約終了以外側の分岐を評価する。
     * 主な仕様:
     * - 異動業務コードが開栓登録系以外なら 020120051 を送出する。
     * - 異動業務コードが開栓登録系なら 020100051 を送出する。
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     */
    private void evaluateNotContractEndedBranch(String duknIsrgCd) {
        if (isNonOpeningWorkCode(duknIsrgCd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020120051);
        }
        throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100051);
    }

    /**
     * 概要: 開栓登録系以外の異動業務コードかどうかを判定する。
     * 主な仕様:
     * - J0497 の開栓登録および開栓（システム処理）以外を true とする。
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     * @return 開栓登録系以外の場合 true
     */
    private boolean isNonOpeningWorkCode(String duknIsrgCd) {
        return !openingRegistration.equals(duknIsrgCd)
                && !openingSystemProcessing.equals(duknIsrgCd);
    }
}
