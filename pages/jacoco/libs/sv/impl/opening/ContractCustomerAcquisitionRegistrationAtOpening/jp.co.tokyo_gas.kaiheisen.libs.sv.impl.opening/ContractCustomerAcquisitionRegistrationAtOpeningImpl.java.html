package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstd.KiycstdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstd.write.KiycstdWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstk.KiycstkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstk.write.KiycstkWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContractCustomerAcquisitionRegistrationAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ContractCustomerAcquisitionRegistrationAtOpeningOutput;

/**
 * 開栓時契約お客さま取得・登録サービス実装。
 */
public class ContractCustomerAcquisitionRegistrationAtOpeningImpl
        implements ContractCustomerAcquisitionRegistrationAtOpening {

    private final KiycstkDataAccess kiycstkDataAccess;

    private final KiycstdDataAccess kiycstdDataAccess;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final NumberingAllInOneSupport numberingAllInOneSupport;

    private final RequestContext requestContext;

    private final String syokaiLevel;

    private final String decResOther;

    private final String usrId = "KHSS275";

    /**
     * コンストラクタ。
     *
     * @param requestContext           コンテキスト情報
     * @param kiycstkDataAccess        契約お客さま＜小売＞ データアクセス
     * @param kiycstdDataAccess        契約お客さま＜導管＞ データアクセス
     * @param ciriusConstantService    CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓API独自コンスタント取得サービス
     */
    public ContractCustomerAcquisitionRegistrationAtOpeningImpl(RequestContext requestContext,
            KiycstkDataAccess kiycstkDataAccess, KiycstdDataAccess kiycstdDataAccess,
            NumberingAllInOneSupport numberingAllInOneSupport, CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        RequestContext nonNullRequestContext = Objects.requireNonNull(requestContext, "requestContext");

        this.requestContext = nonNullRequestContext;
        this.kiycstkDataAccess = kiycstkDataAccess;
        this.kiycstdDataAccess = kiycstdDataAccess;
        this.numberingAllInOneSupport = Objects.requireNonNull(numberingAllInOneSupport, "numberingAllInOneSupport");
        // コンスタントマスタ
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(ciriusConstantService,
                "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.GASTA);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TAXIS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.KRAKEN);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.EAGLE);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.HURRY);
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        // DI
        this.syokaiLevel = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.InquiryLevelType.TYPE,
                ConstantLookupKeys.Cirius.InquiryLevelType.ALL_CAN_BE_QUERIED_AND_DISPLAYED);
        this.decResOther = RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
    }

    /**
     * 概要: 開栓時の契約お客さま情報を取得・登録をする 主な仕様: -
     * お客さま番号採番対象フラグがTrue、他小売区分が他小売の場合契約お客さま＜導管＞にデータを登録する
     * お客さま番号採番対象フラグがTrue、他小売区分が小売の場合契約お客さま＜導管＞、契約お客さま＜小売＞にデータを登録する
     * お客さま番号採番対象フラグがFalseの場合、契約お客さま＜導管＞情報を取得する
     *
     * @param cstNoSibntiFlg お客さま番号採番対象フラグ
     * @param tkourKbn       他小売区分
     * @param kyoTrkNo       契約お客さま登録番号
     * @param ctnmKj1        お客さま名漢字１
     * @param ctnmKn1        お客さま名カナ１
     * @param ctnm1HujnShbt  お客さま名１＿法人種別
     * @param ctnm1HshiShbt  お客さま名１＿法人種別表示位置種別
     * @param ctnmKj2        お客さま名漢字２
     * @param ctnmKn2        お客さま名カナ２
     * @param ctnm2HujnShbt  お客さま名２＿法人種別
     * @param ctnm2HshiShbt  お客さま名２＿法人種別表示位置種別
     * @param ctnmKjhkFlg    お客さま名漢字表示可否フラグ
     * @param yubnNo         郵便番号
     * @param gkkCd          行政区画コード
     * @param mchCd          町コード
     * @param cmazCd         丁目・字コード
     * @param cbgSuj1        丁目－番地－号数字１
     * @param cbgKn2         丁目－番地－号カナ２
     * @param cbgKn3         丁目－番地－号カナ３
     * @param cbgKn4         丁目－番地－号カナ４
     * @param bldNo          建物番号
     * @param tunoKn         棟番号カナ
     * @param hynKn          部屋番号カナ
     * @param htuFlg         部屋番号登録有無フラグ
     * @return 開栓時契約お客さま取得登録結果
     */
    @Override
    public ContractCustomerAcquisitionRegistrationAtOpeningOutput execute(Boolean cstNoSibntiFlg, String tkourKbn,
            Long kyoTrkNo, String ctnmKj1, String ctnmKn1, String ctnm1HujnShbt, String ctnm1HshiShbt, String ctnmKj2,
            String ctnmKn2, String ctnm2HujnShbt, String ctnm2HshiShbt, Boolean ctnmKjhkFlg, String yubnNo,
            String gkkCd, String mchCd, String cmazCd, String cbgSuj1, String cbgKn2, String cbgKn3, String cbgKn4,
            String bldNo, String tunoKn, String hynKn, Boolean htuFlg) {
        if (cstNoSibntiFlg) {
            // 他048を呼び出し、契約お客さま登録番号を採番
            Long assignedkyoTrkNo = numberingAllInOneSupport.issueRetailContractCustomerRegistrationNumber();
            // DA-104-C001パラメータ設定
            KiycstdWriteRecord kiycstdWriteRecord = new KiycstdWriteRecord(assignedkyoTrkNo, ctnmKj1, ctnmKn1,
                    ctnm1HujnShbt, ctnm1HshiShbt, ctnmKj2, ctnmKn2, ctnm2HujnShbt, ctnm2HshiShbt, ctnmKjhkFlg,
                    this.syokaiLevel, yubnNo, gkkCd, mchCd, cmazCd, cbgSuj1, cbgKn2, cbgKn3, cbgKn4, bldNo, null, null,
                    false, null, tunoKn, hynKn, htuFlg, 0);
            String decodedHkkSysSyb = kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE, this.requestContext.getHukkSysShbt()
                            .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
            AuditContext context = new AuditContext(usrId, decodedHkkSysSyb);
            // 契約お客さま＜導管＞にお客さま情報登録
            kiycstdDataAccess.insert(kiycstdWriteRecord, context);
            if (!decResOther.equals(tkourKbn)) {
                // DA-060-C001パラメータ設定
                KiycstkWriteRecord kiycstkWriteRecord = new KiycstkWriteRecord(assignedkyoTrkNo, ctnmKj1, ctnmKn1,
                        ctnm1HujnShbt, ctnm1HshiShbt, ctnmKj2, ctnmKn2, ctnm2HujnShbt, ctnm2HshiShbt, ctnmKjhkFlg, null,
                        null, null, null, Boolean.FALSE, this.syokaiLevel, yubnNo, gkkCd, mchCd, cmazCd, cbgSuj1, cbgKn2,
                        cbgKn3, cbgKn4, bldNo, null, null, Boolean.FALSE, null, tunoKn, hynKn, htuFlg, 0);
                // 契約お客さま＜小売＞にお客さま情報登録
                kiycstkDataAccess.insert(kiycstkWriteRecord, context);
            }
            return new ContractCustomerAcquisitionRegistrationAtOpeningOutput(assignedkyoTrkNo);
        } else {
            return new ContractCustomerAcquisitionRegistrationAtOpeningOutput(kyoTrkNo);
        }
    }
}
