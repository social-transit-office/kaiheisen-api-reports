package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.MeterCompanyNumberCheckAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.MeterCompanyNumberCheckAtOpeningOutput;

/**
 * 概要: 開栓時メーター社番チェックサービス実装。
 * 主な仕様:
 * - ガスメーター＜導管＞有無フラグを先に判定し、false の場合は開栓共通エラー 020100013 を送出する。
 * - 報告ガスメーター社番とガスメーター社番が不一致の場合は開栓共通エラー 020100013 を送出する。
 * 既知の制約:
 * - public 引数に対する追加の null ガードは行わず、Boolean の unboxing による fail-fast を許容する。
 */
public final class MeterCompanyNumberCheckAtOpeningImpl implements MeterCompanyNumberCheckAtOpening {

    /**
     * 概要: 開栓時のメーター社番整合性を判定する。
     * 主な仕様:
     * - ガスメーター＜導管＞有無フラグが false の場合は開栓共通エラー 020100013 を送出する。
     * - 報告ガスメーター社番とガスメーター社番が不一致の場合は開栓共通エラー 020100013 を送出する。
     * 既知の制約:
     * - 両方の社番が null または空文字で等しい場合は正常終了する。
     * @param gasmtrdUmFlg ガスメーター＜導管＞有無フラグ
     * @param hkkGmb 報告ガスメーター社番
     * @param gmb ガスメーター社番
     */
    @Override
    public void execute(Boolean gasmtrdUmFlg, String hkkGmb, String gmb) {
        if (!gasmtrdUmFlg) {
            throwMeterCompanyNumberError();
        }
        if (!Objects.equals(hkkGmb, gmb)) {
            throwMeterCompanyNumberError();
        }
    }

    /**
     * 概要: メーター社番エラーに対応する開栓共通エラーを送出する。
     * 主な仕様:
     * - エラーコード 020100013 を `KaisenRuleException` として送出する。
     * 既知の制約:
     * - 追加の補足情報付与や例外変換は行わない。
     */
    private void throwMeterCompanyNumberError() {
        throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100013);
    }
}
