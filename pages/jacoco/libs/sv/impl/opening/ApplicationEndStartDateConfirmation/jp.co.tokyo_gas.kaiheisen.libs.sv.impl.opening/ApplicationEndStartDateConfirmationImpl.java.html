package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ApplicationEndStartDateConfirmation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.ApplicationEndStartDateConfirmationOutput;

/**
 * 概要: 適用終了・開始日確定サービス（SV-214）の実装。
 * 主な仕様:
 * - 代替判定結果フラグと下位DA取得結果に応じて適用終了年月日を決定する。
 * - 検針結果作成事由種別の対象集合は CIRIUS 定数ライブラリから解決する。
 * - 下位層例外は握りつぶさずに上位へ伝播する。
 * 既知の制約:
 * - 入力値の妥当性検証と HTTP 変換は本クラスでは行わない。
 */
public class ApplicationEndStartDateConfirmationImpl
        implements ApplicationEndStartDateConfirmation {

    private final KkytskrDataAccess kkytskrDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final Set<String> sameDayReasonTypes;

    /**
     * 概要: SV-214 実行に必要な依存を受け取り初期化する。
     * 主な仕様:
     * - 3 件の下位DAと CIRIUS 定数サービスを保持する。
     * - 検針結果作成事由種別の対象集合を初期化時に解決し、以後の execute で再利用する。
     * 既知の制約:
     * - 必須定数が未登録の場合はコンストラクタ初期化時点で fail-fast する。
     *
     * @param kkytskrDataAccess 供給停止管理データアクセス
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果データアクセス
     * @param gmkskkdDataAccess ガスメーター検針結果＜導管＞データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ApplicationEndStartDateConfirmationImpl(
            KkytskrDataAccess kkytskrDataAccess,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            GmkskkdDataAccess gmkskkdDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.sameDayReasonTypes = Set.of(
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR_SW_CANCELLATION),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TEMPORARY),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.CONTRACT_TYPE_CHANGE),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.RELOCATION_CLOSURE),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_OUTAGE),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.UNUSED_CLOSURE),
                requireReasonType(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.FIRED_AND_LOST_CAP));
    }

    @Override
    public ApplicationEndStartDateConfirmationOutput execute(
            Long gasMtstNo,
            LocalDate kasgYmd,
            Long tksuSvkyNo,
            Boolean digwrHntFlg) {
        if (!Boolean.TRUE.equals(digwrHntFlg)) {
            return buildOutput(kasgYmd.minusDays(1), kasgYmd);
        }
        if (kkytskrDataAccess.findByGasMtstNoAndHesnYmd(gasMtstNo, kasgYmd).isPresent()) {
            return buildOutput(kasgYmd, kasgYmd);
        }
        Optional<String> reasonType = resolveMeterReadingReasonType(gasMtstNo, kasgYmd, tksuSvkyNo);
        if (reasonType.filter(sameDayReasonTypes::contains).isPresent()) {
            return buildOutput(kasgYmd, kasgYmd);
        }
        return buildOutput(kasgYmd.minusDays(1), kasgYmd);
    }

    /**
     * 概要: 確定前ガスメーター検針結果情報またはガスメーター検針結果＜導管＞情報から検針結果作成事由種別を解決する。
     * 主な仕様:
     * - まず DA-052-R003 を参照し、未取得時のみ DA-027-R002 を参照する。
     * - いずれも未取得時は空を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kasgYmd 開栓作業年月日
     * @param tksuSvkyNo 託送サービス契約番号
     * @return 解決した検針結果作成事由種別
     */
    private Optional<String> resolveMeterReadingReasonType(
            Long gasMtstNo,
            LocalDate kasgYmd,
            Long tksuSvkyNo) {
        Optional<KtmgmkkRecord> ktmgmkkRecord =
                ktmgmkkDataAccess.findByGasMtstNoAndKnsYmdAndTksuSvkyNo(gasMtstNo, kasgYmd, tksuSvkyNo);
        if (ktmgmkkRecord.isPresent()) {
            return ktmgmkkRecord.flatMap(record -> Optional.ofNullable(record.knkkSksiJysb()));
        }
        return gmkskkdDataAccess.findByGasMtstNoAndKnsYmdAndTksuSvkyNo(gasMtstNo, kasgYmd, tksuSvkyNo)
                .flatMap(record -> Optional.ofNullable(record.knkkSksiJysb()));
    }

    private static ApplicationEndStartDateConfirmationOutput buildOutput(
            LocalDate kktTyenYmd,
            LocalDate kktTkysYmd) {
        return new ApplicationEndStartDateConfirmationOutput(kktTyenYmd, kktTkysYmd);
    }

    private static String requireReasonType(
            CiriusConstantService ciriusConstantService,
            String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                code);
    }
}
