package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.DisasterClosingCheckAtOpening;

/**
 * 概要: SV-190（開栓時災害時閉栓チェック）の実装クラス。
 * 主な仕様:
 * - 閉栓事由種別が災害時閉栓であり、報告開栓事由種別が災害復旧以外の場合を禁止パターンとして判定する。
 * - 禁止パターンに一致した場合は開栓共通エラー 020100041 を送出する。
 * - 判定に使用する CIRIUS 定数はコンストラクタ初期化時に必須解決し、解決済みコードで比較する。
 * - 下位 DA や他 SV 呼び出しは行わず、入力値のみで分岐判定を完結させる。
 * 既知の制約:
 * - 入力値の null 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となり、インスタンス生成に失敗する。
 */
public final class DisasterClosingCheckAtOpeningImpl implements DisasterClosingCheckAtOpening {

    private final String closureReasonClosedInCaseOfDisaster;

    private final String openingReasonDisasterRecovery;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 閉栓事由種別 J0273 の災害時閉栓を必須解決して保持する。
     * - 報告開栓事由種別 04900 の災害復旧を必須解決して保持する。
     * 既知の制約:
     * - CIRIUS 定数取得サービスの null は許容しない。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public DisasterClosingCheckAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.closureReasonClosedInCaseOfDisaster =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
        this.openingReasonDisasterRecovery =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.DISASTER_RECOVERY);
    }

    /**
     * 概要: 開栓時災害時閉栓チェックを実行し、禁止パターンに該当する場合は開栓共通エラーを送出する。
     * 主な仕様:
     * - 閉栓事由種別が災害時閉栓であり、報告開栓事由種別が災害復旧以外のときに開栓共通エラーを送出する。
     * - 上記以外の組み合わせでは正常終了する。
     * 既知の制約:
     * - `correlationId`、`hukkSysShbt`、`systemDatetime` は SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードは追加せず、上位バリデーション済み前提で評価する。
     *
     * @param hesnJysb 閉栓事由種別
     * @param hkkKasJysb 報告開栓事由種別
     */
    @Override
    public void execute(String hesnJysb, String hkkKasJysb) {
        if (closureReasonClosedInCaseOfDisaster.equals(hesnJysb)
                && !openingReasonDisasterRecovery.equals(hkkKasJysb)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100041);
        }
    }
}
