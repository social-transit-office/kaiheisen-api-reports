package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;

/**
 * 概要: SV-172（開栓時継続使用契約継続判定）の実装クラス。
 * 主な仕様:
 * - 契約あり時は託送供給依頼事業者コード、TG他小売区分、継続使用扱いコードを評価して処理判定_4x/3x/2x を設定する。
 * - 契約なし時は契約種別が一般契約であることを判定し、DA-157-R001 を呼び出して復活フラグと処理判定_4x/3x/2x を決定する。
 * - systemDatetime は RequestContext から取得し、未設定時は fail-fast する。
 * 既知の制約:
 * - 一般契約（18200-01101）、J0600-0023、J0601-0023002 は必須定数として解決する。
 * - 公開引数の null ガードや blank 補正は追加せず、上位層から渡された値をそのまま評価する。
 */
public final class ContinuedUseContractContinuationJudgmentAtOpeningImpl
        implements ContinuedUseContractContinuationJudgmentAtOpening {

    private final SsvsbhwDataAccess ssvsbhwDataAccess;

    private final RequestContext requestContext;

    private final String tokyoGasRetailerCode;

    private final String otherRetailCategory;

    private final String alternativeOpeningTarget;

    private final String generalContractType;

    private final String paymentContractContinuationDecisionType;

    private final String paymentContractContinuedUseDecisionContent;

    /**
     * 概要: 依存と定数サービスを受け取ってインスタンスを初期化する。
     * 主な仕様: SV-172 の判定で使用する CIRIUS / 開閉栓定数をコンストラクタ初期化時点で fail-fast 解決する。
     * 既知の制約: 依存と定数サービスの null は許容しない。
     *
     * @param ssvsbhwDataAccess DA-157-R001 API
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public ContinuedUseContractContinuationJudgmentAtOpeningImpl(
            SsvsbhwDataAccess ssvsbhwDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this(
                ssvsbhwDataAccess,
                requestContext,
                resolveTokyoGasRetailerCode(ciriusConstantService),
                resolveOtherRetailCategory(kaiheisenConstantService),
                resolveAlternativeOpeningTarget(kaiheisenConstantService),
                resolveGeneralContractType(ciriusConstantService),
                resolvePaymentContractContinuationDecisionType(ciriusConstantService),
                resolvePaymentContractContinuedUseDecisionContent(ciriusConstantService));
    }

    /**
     * 概要: テスト差し替え用に解決済み依存を受け取って初期化する。
     * 主な仕様: 定数解決済みコード値と依存をそのまま保持し、execute で再利用する。
     * 既知の制約: null の依存やコード値は許容しない。
     */
    ContinuedUseContractContinuationJudgmentAtOpeningImpl(
            SsvsbhwDataAccess ssvsbhwDataAccess,
            RequestContext requestContext,
            String tokyoGasRetailerCode,
            String otherRetailCategory,
            String alternativeOpeningTarget,
            String generalContractType,
            String paymentContractContinuationDecisionType,
            String paymentContractContinuedUseDecisionContent) {
        this.ssvsbhwDataAccess = Objects.requireNonNull(ssvsbhwDataAccess, "ssvsbhwDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.tokyoGasRetailerCode = Objects.requireNonNull(tokyoGasRetailerCode, "tokyoGasRetailerCode");
        this.otherRetailCategory = Objects.requireNonNull(otherRetailCategory, "otherRetailCategory");
        this.alternativeOpeningTarget = Objects.requireNonNull(
                alternativeOpeningTarget,
                "alternativeOpeningTarget");
        this.generalContractType = Objects.requireNonNull(generalContractType, "generalContractType");
        this.paymentContractContinuationDecisionType = Objects.requireNonNull(
                paymentContractContinuationDecisionType,
                "paymentContractContinuationDecisionType");
        this.paymentContractContinuedUseDecisionContent = Objects.requireNonNull(
                paymentContractContinuedUseDecisionContent,
                "paymentContractContinuedUseDecisionContent");
    }

    /**
     * 概要: 開栓時継続使用契約継続判定を行う。
     * 主な仕様: systemDatetime を取得したうえで契約あり判定を優先し、不成立時は契約なし側の復活・継続判定へ進む。
     * 既知の制約: 公開引数の null ガードは行わず、必要な箇所では NullPointerException がそのまま発生する。
     */
    @Override
    public ContinuedUseContractContinuationJudgmentAtOpeningOutput execute(
            String tgTkurKbn,
            LocalDate kasgYmd,
            Boolean kiyUmFlg,
            String kzksyAtkCd,
            TkssvkyInf tkssvkyInf,
            ContinuedUseContractSvckyaInf svckyaInf,
            GmtsetdInf gmtsetdInf,
            ContinuedUseContractSihakyInf sihakyInf) {
        LocalDateTime systemDatetime = requireSystemDatetime();

        String tksuKijsCd = tkssvkyInf.tksuKijsCd();
        if (shouldReturnContractContinuation(
                tksuKijsCd,
                kiyUmFlg,
                tgTkurKbn,
                kzksyAtkCd)) {
            return new ContinuedUseContractContinuationJudgmentAtOpeningOutput(
                    Boolean.TRUE,
                    Boolean.TRUE,
                    Boolean.TRUE,
                    Boolean.FALSE,
                    null);
        }

        if (!shouldEvaluateReactivation(
                tksuKijsCd,
                kiyUmFlg,
                tgTkurKbn)) {
            return defaultOutput();
        }

        return judgeReactivation(
                kasgYmd,
                kzksyAtkCd,
                svckyaInf,
                gmtsetdInf,
                sihakyInf,
                systemDatetime.toLocalDate());
    }

    /**
     * 概要: 契約あり側の継続判定成立条件を評価する。
     * 主な仕様: 東京ガス、契約あり、他小売以外、代替開栓対象以外の全条件が成立した場合に true を返す。
     * 既知の制約: 契約有無フラグの null はアンボクシングにより NullPointerException となる。
     */
    private boolean shouldReturnContractContinuation(
            String tksuKijsCd,
            Boolean kiyUmFlg,
            String tgTkurKbn,
            String kzksyAtkCd) {
        return tksuKijsCd != null
                && !tksuKijsCd.isEmpty()
                && tokyoGasRetailerCode.equals(tksuKijsCd)
                && kiyUmFlg
                && !otherRetailCategory.equals(tgTkurKbn)
                && !alternativeOpeningTarget.equals(kzksyAtkCd);
    }

    /**
     * 概要: 契約なし側の復活判定に進む条件を評価する。
     * 主な仕様: 東京ガス、契約なし、他小売以外のときに true を返す。
     * 既知の制約: 契約有無フラグの null はアンボクシングにより NullPointerException となる。
     */
    private boolean shouldEvaluateReactivation(
            String tksuKijsCd,
            Boolean kiyUmFlg,
            String tgTkurKbn) {
        return tksuKijsCd != null
                && !tksuKijsCd.isEmpty()
                && tokyoGasRetailerCode.equals(tksuKijsCd)
                && !kiyUmFlg
                && !otherRetailCategory.equals(tgTkurKbn);
    }

    /**
     * 概要: 契約なし側の復活・継続判定を行う。
     * 主な仕様: 一般契約判定、代替開栓対象判定、13ヶ月境界判定、DA-157 判定の順に評価して出力を組み立てる。
     * 既知の制約: 13ヶ月境界を超える場合は DA-157 を呼び出さず、復活フラグのみ true にする。
     */
    private ContinuedUseContractContinuationJudgmentAtOpeningOutput judgeReactivation(
            LocalDate kasgYmd,
            String kzksyAtkCd,
            ContinuedUseContractSvckyaInf svckyaInf,
            GmtsetdInf gmtsetdInf,
            ContinuedUseContractSihakyInf sihakyInf,
            LocalDate systemDate) {
        if (!generalContractType.equals(svckyaInf.kiyShbt())
                || alternativeOpeningTarget.equals(kzksyAtkCd)) {
            return defaultOutput();
        }

        if (!kasgYmd.isBefore(resolveThirteenMonthExceededDate(gmtsetdInf.hesnYmd()))) {
            return new ContinuedUseContractContinuationJudgmentAtOpeningOutput(
                    Boolean.FALSE,
                    Boolean.FALSE,
                    Boolean.FALSE,
                    Boolean.TRUE,
                    null);
        }

        String sqsvsHnsb = findBillingServiceTypeDistributionContent(
                sihakyInf.sikSvcShbt(),
                systemDate);
        return new ContinuedUseContractContinuationJudgmentAtOpeningOutput(
                Boolean.TRUE,
                Boolean.TRUE,
                paymentContractContinuedUseDecisionContent.equals(sqsvsHnsb),
                Boolean.TRUE,
                sqsvsHnsb);
    }

    /**
     * 概要: DA-157-R001 を呼び出して請求サービス種別振分内容種別を取得する。
     * 主な仕様: J0600-0023 を種別として最新の SSVSBHW を 1 件取得し、内容種別を返す。
     * 既知の制約: DA 結果が空の場合は null を返す。
     */
    private String findBillingServiceTypeDistributionContent(String sikSvcShbt, LocalDate systemDate) {
        return ssvsbhwDataAccess
                .findBySqsvsHwsbAndSikSvcShbtAndKnskKjnYmdOrderByTkysYmdDesc(
                        paymentContractContinuationDecisionType,
                        sikSvcShbt,
                        systemDate)
                .map(SsvsbhwRecord::sqsvsHnsb)
                .orElse(null);
    }

    /**
     * 概要: 13ヶ月超過となる境界日を算出する。
     * 主な仕様: 閉栓年月日の13ヶ月後の月初日を返し、その日以降を13ヶ月超過として扱う。
     * 既知の制約: 閉栓年月日が null の場合は NullPointerException となる。
     */
    private static LocalDate resolveThirteenMonthExceededDate(LocalDate hesnYmd) {
        return hesnYmd.plusMonths(13).withDayOfMonth(1);
    }

    /**
     * 概要: 初期値出力を返す。
     * 主な仕様: 4 種の処理判定フラグを false、請求サービス種別振分内容種別を null で返す。
     * 既知の制約: 設計書の初期値設定ステップに対応する固定値を返す。
     */
    private static ContinuedUseContractContinuationJudgmentAtOpeningOutput defaultOutput() {
        return new ContinuedUseContractContinuationJudgmentAtOpeningOutput(
                Boolean.FALSE,
                Boolean.FALSE,
                Boolean.FALSE,
                Boolean.FALSE,
                null);
    }

    /**
     * 概要: RequestContext からシステム日時を取得する。
     * 主な仕様: 未設定時は IllegalStateException で fail-fast する。
     * 既知の制約: Optional の空値は許容しない。
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 東京ガス小売事業者コードを解決する。
     * 主な仕様: J0580-A0020 を必須 CIRIUS 定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolveTokyoGasRetailerCode(CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
    }

    /**
     * 概要: 他小売区分コードを解決する。
     * 主な仕様: KH005-1 を必須開閉栓定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolveOtherRetailCategory(KaiheisenConstantService kaiheisenConstantService) {
        return RequiredConstantResolver.requireKaiheisenConstant(
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService"),
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.TYPE,
                ConstantLookupKeys.Kaiheisen.TgAndOtherRetailCategories.OTHER_RETAIL);
    }

    /**
     * 概要: 代替開栓対象コードを解決する。
     * 主な仕様: KH007-2 を必須開閉栓定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolveAlternativeOpeningTarget(KaiheisenConstantService kaiheisenConstantService) {
        return RequiredConstantResolver.requireKaiheisenConstant(
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService"),
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.TYPE,
                ConstantLookupKeys.Kaiheisen.ContinuousUseHandlingCode.ALTERNATIVE_OPENING_TARGET);
    }

    /**
     * 概要: 一般契約コードを解決する。
     * 主な仕様: 18200-01101 を必須 CIRIUS 定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolveGeneralContractType(CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.ContractType.TYPE,
                ConstantLookupKeys.Cirius.ContractType.GENERAL_CONTRACT);
    }

    /**
     * 概要: 支払契約継続使用判断の請求サービス種別振分種別を解決する。
     * 主な仕様: J0600-0023 を必須 CIRIUS 定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolvePaymentContractContinuationDecisionType(
            CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.BillingServiceTypeDistributionType.TYPE,
                ConstantLookupKeys.Cirius.BillingServiceTypeDistributionType
                        .DECISION_ON_CONTINUED_USE_OF_PAYMENT_CONTRACT);
    }

    /**
     * 概要: 支払契約継続使用判断の継続使用内容種別を解決する。
     * 主な仕様: J0601-0023002 を必須 CIRIUS 定数として解決する。
     * 既知の制約: 定数サービスの null は許容しない。
     */
    private static String resolvePaymentContractContinuedUseDecisionContent(
            CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.BillingServiceTypeDistributionContentType.TYPE,
                ConstantLookupKeys.Cirius.BillingServiceTypeDistributionContentType
                        .PAYMENT_CONTRACT_CONTINUED_USE_DECISION_CONTINUED_USE);
    }
