package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.SameDayOpeningDateCheckContractStartDateDuplicate;

/**
 * 概要: SV-189（同日開栓日チェック（契約開始日重複））の業務判定を実装するクラス。
 * 主な仕様:
 * - 代替判定結果フラグが true の場合に、適用開始年月日および料金計算開始年月日の順で開栓作業年月日との同日判定を行う。
 * - いずれかが同日の場合は開栓共通エラー 020100047 を送出し、どちらも同日でない場合は何も返さずに正常終了する。
 * 既知の制約:
 * - 入力値の補完や null ガードは行わず、実行時例外はそのまま表出する。
 * - 両方の日付が同日の場合は、適用開始年月日の判定を先に評価する。
 */
public class SameDayOpeningDateCheckContractStartDateDuplicateImpl
        implements SameDayOpeningDateCheckContractStartDateDuplicate {

    /**
     * 代替判定結果フラグと各開始年月日をもとに、同日開栓共通エラー要否を判定する。
     *
     * @param digwrHntFlg 代替判定結果フラグ
     * @param tkysYmd 適用開始年月日
     * @param rkkiSttYmd 料金計算開始年月日
     * @param kasgYmd 開栓作業年月日
     * @throws KaisenRuleException 代替判定結果フラグが true かつ適用開始年月日または料金計算開始年月日が開栓作業年月日と同日の場合
     * @throws NullPointerException 代替判定結果フラグが null の場合、または true かつ判定に必要な日付入力が null の場合
     */
    @Override
    public void execute(
            Boolean digwrHntFlg,
            LocalDate tkysYmd,
            LocalDate rkkiSttYmd,
            LocalDate kasgYmd) {
        if (digwrHntFlg && kasgYmd.equals(tkysYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100047);
        }

        if (digwrHntFlg && kasgYmd.equals(rkkiSttYmd)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100047);
        }

    }
}
