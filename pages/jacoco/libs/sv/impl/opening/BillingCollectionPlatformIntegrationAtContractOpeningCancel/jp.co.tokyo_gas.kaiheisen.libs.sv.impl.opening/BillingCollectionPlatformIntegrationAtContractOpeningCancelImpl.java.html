/**
 * 概要: SV-055（前契約継続開栓取消時請求回収基盤連携）のサービス実装。
 * 主な仕様:
 * - 判定、補完、下位 SV/DA 呼出しのみを行う void サービスとして実装する。
 * - `SV-058` 実行後のみ `DA-030-C001` 登録を続行する。
 */
public final class BillingCollectionPlatformIntegrationAtContractOpeningCancelImpl
        implements BillingCollectionPlatformIntegrationAtContractOpeningCancel {

    private static final long FIXED_GAS_METER_LOCATION_NO = 11111111111L;

    private static final int PREVIOUS_CONTRACT_PHONE_FETCH_LIMIT = 2;

    private static final String SERVICE_USER_ID = "KHSS055";

    private static final DateTimeFormatter REQUEST_TIME_FORMAT = DateTimeFormatter.ofPattern("HHmmss");

    private final KyotelkDataAccess kyotelkDataAccess;

    private final ServiceSuspensionCancellationReasonTypeAcquisition
            serviceSuspensionCancellationReasonTypeAcquisition;

    private final BillingCollectionPlatformApiIntegrationForPreviousContract
            billingCollectionPlatformApiIntegrationForPreviousContract;

    private final IrslkkzDataAccess irslkkzDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String previousContractCategory;

    private final String noServiceSuspensionOrCancellationReasonType;

    private final String supplyOutageReasonType;

    private final String blankContactType;

    private final String noPhoneType;

    private final String tokyoGasNetworkOperatorCode;

    private final String billablePaymentManagementTransferCode;

    private final String billableContractManagementTransferCode;

    /**
     * 概要: CDI から依存を受け取って初期化する。
     */
    @Inject
    public BillingCollectionPlatformIntegrationAtContractOpeningCancelImpl(
            KyotelkDataAccess kyotelkDataAccess,
            ServiceSuspensionCancellationReasonTypeAcquisition
                    serviceSuspensionCancellationReasonTypeAcquisition,
            BillingCollectionPlatformApiIntegrationForPreviousContract
                    billingCollectionPlatformApiIntegrationForPreviousContract,
            IrslkkzDataAccess irslkkzDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.kyotelkDataAccess = Objects.requireNonNull(kyotelkDataAccess, "kyotelkDataAccess");
        this.serviceSuspensionCancellationReasonTypeAcquisition = Objects.requireNonNull(
                serviceSuspensionCancellationReasonTypeAcquisition,
                "serviceSuspensionCancellationReasonTypeAcquisition");
        this.billingCollectionPlatformApiIntegrationForPreviousContract = Objects.requireNonNull(
                billingCollectionPlatformApiIntegrationForPreviousContract,
                "billingCollectionPlatformApiIntegrationForPreviousContract");
        this.irslkkzDataAccess = Objects.requireNonNull(irslkkzDataAccess, "irslkkzDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService requiredCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.previousContractCategory = requireKaiheisenCode(
                this.kaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.UnopeningCancellationRequestCollectionPlatformCollaborationCategory.TYPE,
                ConstantLookupKeys.Kaiheisen.UnopeningCancellationRequestCollectionPlatformCollaborationCategory
                        .CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.noServiceSuspensionOrCancellationReasonType = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                        .NO_SERVICE_SUSPENSION_OR_CANCELLATION);
        this.supplyOutageReasonType = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType.SUPPLY_OUTAGE);
        this.blankContactType = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ContactType.TYPE,
                ConstantLookupKeys.Cirius.ContactType.BLANK);
        this.noPhoneType = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.PhoneType.TYPE,
                ConstantLookupKeys.Cirius.PhoneType.NO_PHONE);
        this.tokyoGasNetworkOperatorCode = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.NwOperatorCode.TYPE,
                ConstantLookupKeys.Cirius.NwOperatorCode.TOKYO_GAS_CO_LTD);
        this.billablePaymentManagementTransferCode = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .BILLABLE_PAYMENT_MANAGEMENT_TRANSFER);
        this.billableContractManagementTransferCode = requireCiriusCode(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .BILLABLE_CONTRACT_MANAGEMENT_TRANSFER);
    }

    @Override
    public void execute(
            String kastkSeqkRkKbn,
            Integer suguGmtKnsu,
            Boolean ksenFlg,
            Boolean kkytskrKktUmFlg,
            Boolean kkytskrUmFlg,
            Boolean suguKbnUpdFlg,
            Long gasMtstNo,
            SihakyInf sihakyInf,
            SvckyaInf svckyaInf,
            SvckyaOgaInf svckyaOgaInf,
            GmtsetdOgaInf gmtsetdOgaInf,
            SvmstkrInf svmstkrInf) {
        RequestState requestState = requireRequestState();
        InternalState state = new InternalState(blankContactType, noPhoneType);
        applyComprehensiveMeterCountStep(suguGmtKnsu, gasMtstNo, suguKbnUpdFlg, svckyaInf, svmstkrInf, state);
        applyServiceSuspensionReasonStep(suguKbnUpdFlg, svckyaOgaInf, gmtsetdOgaInf, state);
        applyCollaborationCategoryStep(kastkSeqkRkKbn, suguKbnUpdFlg, sihakyInf, svckyaOgaInf, gmtsetdOgaInf, state);
        executeCollaborationStep(
                kastkSeqkRkKbn,
                suguGmtKnsu,
                ksenFlg,
                kkytskrKktUmFlg,
                kkytskrUmFlg,
                sihakyInf,
                svckyaInf,
                svckyaOgaInf,
                requestState,
                state);
    }

    private void applyComprehensiveMeterCountStep(
            Integer suguGmtKnsu,
            Long gasMtstNo,
            Boolean suguKbnUpdFlg,
            SvckyaInf svckyaInf,
            SvmstkrInf svmstkrInf,
            InternalState state) {
        if (Integer.valueOf(0).equals(suguGmtKnsu)) {
            state.gasMeterLocationNo = FIXED_GAS_METER_LOCATION_NO;
            state.noticeCustomerNo = gasMtstNo;
            return;
        }
        state.gasMeterLocationNo = svckyaInf.svkyNo();
        if (Boolean.TRUE.equals(suguKbnUpdFlg)) {
            state.noticeCustomerNo = svmstkrInf.mtstNo();
            return;
        }
        state.noticeCustomerNo = svckyaInf.svkyNo();
    }

    private void applyServiceSuspensionReasonStep(
            Boolean suguKbnUpdFlg,
            SvckyaOgaInf svckyaOgaInf,
            GmtsetdOgaInf gmtsetdOgaInf,
            InternalState state) {
        if (Boolean.TRUE.equals(suguKbnUpdFlg)) {
            state.serviceSuspensionCancellationReasonType = noServiceSuspensionOrCancellationReasonType;
            return;
        }
        state.serviceSuspensionCancellationReasonType =
                serviceSuspensionCancellationReasonTypeAcquisition.execute(
                        svckyaOgaInf.svkyEndJysb(),
                        gmtsetdOgaInf.hesnJysb()).svkjyShbt();
    }

    private void applyCollaborationCategoryStep(
            String kastkSeqkRkKbn,
            Boolean suguKbnUpdFlg,
            SihakyInf sihakyInf,
            SvckyaOgaInf svckyaOgaInf,
            GmtsetdOgaInf gmtsetdOgaInf,
            InternalState state) {
        if (isPreviousContract(kastkSeqkRkKbn)) {
            state.requestSourceCustomerRoleNo = sihakyInf.sihKiyNo();
            state.networkOperatorCode = tokyoGasNetworkOperatorCode;
            state.billingEndDate = svckyaOgaInf.rkkiEndYmd();
            if (!Boolean.TRUE.equals(suguKbnUpdFlg)) {
                state.serviceSuspensionCancellationDate = svckyaOgaInf.rkkiEndYmd();
                state.contractEndDate = svckyaOgaInf.rkkiEndYmd();
            }
            return;
        }
        if (supplyOutageReasonType.equals(state.serviceSuspensionCancellationReasonType)) {
            state.serviceSuspensionCancellationDate = gmtsetdOgaInf.hesnYmd();
            return;
        }
        state.serviceSuspensionCancellationDate = svckyaOgaInf.rkkiEndYmd();
    }

    private void executeCollaborationStep(
            String kastkSeqkRkKbn,
            Integer suguGmtKnsu,
            Boolean ksenFlg,
            Boolean kkytskrKktUmFlg,
            Boolean kkytskrUmFlg,
            SihakyInf sihakyInf,
            SvckyaInf svckyaInf,
            SvckyaOgaInf svckyaOgaInf,
            RequestState requestState,
            InternalState state) {
        if (isPreviousContract(kastkSeqkRkKbn)) {
            if (suguGmtKnsu != null && suguGmtKnsu.intValue() <= PREVIOUS_CONTRACT_PHONE_FETCH_LIMIT) {
                loadPayerPhoneNumbers(sihakyInf.shsyKyoTrno(), state);
                executePreviousContractIntegration(sihakyInf, svckyaInf, svckyaOgaInf, requestState, state);
            }
            return;
        }
        if (!Boolean.TRUE.equals(ksenFlg)
                && (Boolean.TRUE.equals(kkytskrKktUmFlg) || Boolean.TRUE.equals(kkytskrUmFlg))) {
            executePreviousContractIntegration(sihakyInf, svckyaInf, svckyaOgaInf, requestState, state);
        }
    }

    private void loadPayerPhoneNumbers(Long kyoTrkNo, InternalState state) {
        List<KyotelkRecord> phoneRecords = kyotelkDataAccess.findAllByKyoTrkNo(kyoTrkNo);
        if (!phoneRecords.isEmpty() && hasPhoneNumber(phoneRecords.get(0))) {
            KyotelkRecord firstRecord = phoneRecords.get(0);
            state.contactType1 = firstRecord.telRrskShbt();
            state.phoneNumber1 = firstRecord.kyoTelNo();
            state.phoneType1 = firstRecord.telShbt();
            if (phoneRecords.size() > 1 && hasPhoneNumber(phoneRecords.get(1))) {
                KyotelkRecord secondRecord = phoneRecords.get(1);
                state.contactType2 = secondRecord.telRrskShbt();
                state.phoneNumber2 = secondRecord.kyoTelNo();
                state.phoneType2 = secondRecord.telShbt();
            }
        }
    }

    private void executePreviousContractIntegration(
            SihakyInf sihakyInf,
            SvckyaInf svckyaInf,
            SvckyaOgaInf svckyaOgaInf,
            RequestState requestState,
            InternalState state) {
        billingCollectionPlatformApiIntegrationForPreviousContract.execute(
                buildSkkbRkySgjhInf(sihakyInf, svckyaInf, requestState, state),
                buildSktsSihTrirInf(sihakyInf, svckyaOgaInf, state),
                buildSktsSihHkirInf(state),
                buildSktsKykHkirInf(svckyaOgaInf, state),
                new SktsSihDlirInf(billablePaymentManagementTransferCode));
        irslkkzDataAccess.insert(
                buildIrslkkzWriteRecord(sihakyInf, svckyaInf, requestState, state),
                AuditContext.of(SERVICE_USER_ID, requestState.originFunctionId()));
    }

    private SkkbRkySgjhInf buildSkkbRkySgjhInf(
            SihakyInf sihakyInf,
            SvckyaInf svckyaInf,
            RequestState requestState,
            InternalState state) {
        return new SkkbRkySgjhInf(
                null,
                sihakyInf.sikKrno(),
                svckyaInf.sqtiKiyNo(),
                svckyaInf.svkyNo(),
                svckyaInf.skno(),
                sihakyInf.sihKiyNo(),
                sihakyInf.skno(),
                state.gasMeterLocationNo,
                svckyaInf.rkkiSttYmd(),
                svckyaInf.rkkiEndYmd(),
                Boolean.TRUE,
                Boolean.FALSE,
                Boolean.FALSE,
                Boolean.FALSE,
                requestState.systemDate());
    }

    private SktsSihTrirInf buildSktsSihTrirInf(
            SihakyInf sihakyInf,
            SvckyaOgaInf svckyaOgaInf,
            InternalState state) {
        return new SktsSihTrirInf(
                null,
                null,
                state.requestSourceCustomerRoleNo,
                null,
                null,
                state.contactType1,
                state.contactType2,
                state.phoneNumber1,
                state.phoneNumber2,
                state.phoneType1,
                state.phoneType2,
                billablePaymentManagementTransferCode,
                svckyaOgaInf.rkkiEndYmd(),
                sihakyInf.tkysYmd(),
                state.networkOperatorCode);
    }

    private SktsSihHkirInf buildSktsSihHkirInf(InternalState state) {
        return new SktsSihHkirInf(
                billablePaymentManagementTransferCode,
                state.billingEndDate);
    }

    private SktsKykHkirInf buildSktsKykHkirInf(
            SvckyaOgaInf svckyaOgaInf,
            InternalState state) {
        return new SktsKykHkirInf(
                state.serviceSuspensionCancellationReasonType,
                state.serviceSuspensionCancellationDate,
                state.contractEndDate,
                billableContractManagementTransferCode,
                state.noticeCustomerNo,
                svckyaOgaInf.szirHtyoYmd());
    }

    private IrslkkzWriteRecord buildIrslkkzWriteRecord(
            SihakyInf sihakyInf,
            SvckyaInf svckyaInf,
            RequestState requestState,
            InternalState state) {
        return new IrslkkzWriteRecord(
                requestState.systemDate(),
                Integer.valueOf(REQUEST_TIME_FORMAT.format(requestState.systemDatetime())),
                state.gasMeterLocationNo,
                Boolean.TRUE,
                Boolean.FALSE,
                state.gasMeterLocationNo,
                svckyaInf.svkyNo(),
                sihakyInf.sihKiyNo(),
                sihakyInf.sikKrno(),
                svckyaInf.sqtiKiyNo(),
                svckyaInf.rkkiEndYmd(),
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private RequestState requireRequestState() {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        String reportingSystemTypeCode = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return new RequestState(
                systemDatetime.toLocalDate(),
                systemDatetime,
                requireOriginFunctionId(reportingSystemTypeCode));
    }

    private String requireOriginFunctionId(String reportingSystemTypeCode) {
        String type = ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE;
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(type, reportingSystemTypeCode);
        if (originFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + type + ", value=" + reportingSystemTypeCode);
        }
        return originFunctionId;
    }

    private boolean isPreviousContract(String kastkSeqkRkKbn) {
        return previousContractCategory.equals(kastkSeqkRkKbn);
    }

    private static boolean hasPhoneNumber(KyotelkRecord record) {
        return record.kyoTelNo() != null && !record.kyoTelNo().isBlank();
    }

    private static String requireKaiheisenCode(
            KaiheisenConstantService kaiheisenConstantService,
            String type,
            String value) {
        String name = kaiheisenConstantService.findNameByTypeAndValue(type, value);
        if (name == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + type + ", value=" + value);
        }
        return value;
    }

    private static String requireCiriusCode(
            CiriusConstantService ciriusConstantService,
            String typeCode,
            String code) {
        if (ciriusConstantService.findActiveByTypeCodeAndCode(typeCode, code) == null) {
            throw new ConstantDefinitionNotFoundException(
                    "CiriusConstantService",
                    "typeCode=" + typeCode + ", code=" + code);
        }
        return code;
    }

    private record RequestState(
            LocalDate systemDate,
            LocalDateTime systemDatetime,
            String originFunctionId) {
    }

    private static final class InternalState {

        private Long gasMeterLocationNo;

        private Long requestSourceCustomerRoleNo;

        private String contactType1;

        private String contactType2;

        private String phoneNumber1;

        private String phoneNumber2;

        private String phoneType1;

        private String phoneType2;

        private LocalDate billingEndDate;

        private String networkOperatorCode;

        private LocalDate contractEndDate;

        private String serviceSuspensionCancellationReasonType;

        private LocalDate serviceSuspensionCancellationDate;

        private Long noticeCustomerNo;

        private InternalState(String blankContactType, String noPhoneType) {
            this.contactType1 = blankContactType;
            this.contactType2 = blankContactType;
            this.phoneType1 = noPhoneType;
            this.phoneType2 = noPhoneType;
        }
    }
