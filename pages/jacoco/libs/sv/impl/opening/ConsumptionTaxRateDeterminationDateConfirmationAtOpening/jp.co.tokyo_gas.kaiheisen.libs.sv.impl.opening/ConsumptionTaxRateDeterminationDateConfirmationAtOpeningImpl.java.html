package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrRecord;

/**
 * 概要: 開栓時消費税率判定用年月日確定サービス（SV-229）の実装。
 * 主な仕様:
 * - 代替判定結果、開栓事由、供給停止精算状態区分を優先的に評価し、確定日付を早期決定する。
 * - 必要時のみ総合他メーター関連の下位 DA を参照し、総合メーターの状態に応じて返却日付を決定する。
 * - 下位 DA 例外は握りつぶさずにそのまま上位へ伝播する。
 * 既知の制約:
 * - `correlationId`、`hukkSysShbt`、`systemDatetime` は RequestContext/MDC 管理であり、本クラスの公開メソッド引数に含めない。
 * - 総合他メーター判定経路では RequestContext の `systemDatetime` が必須であり、未設定の場合は `IllegalStateException` を送出する。
 */
public class ConsumptionTaxRateDeterminationDateConfirmationAtOpeningImpl
        implements ConsumptionTaxRateDeterminationDateConfirmationAtOpening {

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final RequestContext requestContext;

    private final List<String> relatedStatuses;

    private final Set<String> priorOpeningReasons;

    private final Set<String> priorClosureReasons;

    private final String supplyStopPaymentPending;

    private final String gasMeterOpening;

    private final String gasMeterClosed;

    private final String closureReasonSupplyOutage;

    /**
     * 概要: SV-229 の実行に必要な DA 依存を初期化する。
     * 主な仕様: サービス契約・メーター設置場所関連、ガスメーター設置場所＜導管＞、供給停止管理の各 DA を保持する。
     * 既知の制約: 必須依存が null の場合はコンストラクタ初期化時点で fail-fast する。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DA
     * @param kkytskrDataAccess 供給停止管理 DA
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param requestContext リクエストコンテキスト
     */
    public ConsumptionTaxRateDeterminationDateConfirmationAtOpeningImpl(
            SvmstkrDataAccess svmstkrDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            CiriusConstantService ciriusConstantService,
            RequestContext requestContext) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");

        CiriusConstantService requiredCiriusConstantService =
            Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.relatedStatuses = List.of(
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID),
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03));
        this.priorOpeningReasons = Set.of(
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.SECURITY_RESTORATION),
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                ConstantLookupKeys.Cirius.OpeningReasonType.DISASTER_RECOVERY));
        this.priorClosureReasons = Set.of(
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE),
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER));
        this.supplyStopPaymentPending =
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
        this.gasMeterOpening =
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.gasMeterClosed =
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
        this.closureReasonSupplyOutage =
            RequiredConstantResolver.requireCiriusConstant(
                requiredCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
    }

    /**
     * 概要: 開栓時の確定消費税率判定用年月日を確定する。
         * 主な仕様: 入力優先の短絡判定後、総合他メーターの状態を評価して、開栓作業年月日または消費税率判定用年月日を返却する。
         * 既知の制約: 代替判定結果フラグは null を許容せず、総合他メーター評価経路で RequestContext の `systemDatetime` が未設定の場合は `IllegalStateException` となる。
     */
    @Override
    public ConsumptionTaxRateDeterminationDateConfirmationAtOpeningOutput execute(
            Long gasMtstNo,
            Long kktSvkyNo,
            LocalDate kasgYmd,
            String kasJysb,
            LocalDate szirHtyoYmd,
            Boolean digwrHntFlg,
            String kkysSisnJtkb,
            LocalDate ksenHkkKnuKigenYmd) {
        boolean alternativeJudgmentResult = digwrHntFlg;

        if (alternativeJudgmentResult) {
            return output(kasgYmd);
        }
        if (priorOpeningReasons.contains(kasJysb)) {
            return output(szirHtyoYmd);
        }
        if (Objects.equals(kkysSisnJtkb, supplyStopPaymentPending)) {
            return output(szirHtyoYmd);
        }

        List<SvmstkrRecord> relatedMeters = svmstkrDataAccess
                .findAllBySvkyNoAndRkkiKnrnJtkbAndRkkiSttYmdAndRkkiEndYmdAndMtstNoForR015(
                        kktSvkyNo,
                        relatedStatuses,
                        kasgYmd,
                        kasgYmd,
                        gasMtstNo);
        if (relatedMeters.isEmpty()) {
            return output(kasgYmd);
        }

        LocalDate systemDate = requestContext.getSystemDatetime()
                .map(systemDatetime -> systemDatetime.toLocalDate())
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        boolean hasOpeningMeter = false;
        boolean hasOtherClosingReason = false;

        for (SvmstkrRecord relatedMeter : relatedMeters) {
            Optional<GmtsetdRecord> gasMeter =
                    gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(relatedMeter.mtstNo(), systemDate);
            if (gasMeter.isEmpty()) {
                continue;
            }
            GmtsetdRecord gasMeterRecord = gasMeter.get();

            if (Objects.equals(gasMeterRecord.gmtKhsJtkb(), gasMeterOpening)) {
                hasOpeningMeter = true;
                continue;
            }
            if (!Objects.equals(gasMeterRecord.gmtKhsJtkb(), gasMeterClosed)) {
                continue;
            }

            if (Objects.equals(gasMeterRecord.hesnJysb(), closureReasonSupplyOutage)) {
                Optional<KkytskrRecord> supplyStopRecord =
                        kkytskrDataAccess.findByGasMtstNo(relatedMeter.mtstNo());
                if (supplyStopRecord
                        .map(KkytskrRecord::kkysSisnJtkb)
                        .filter(supplyStopPaymentPending::equals)
                        .isPresent()) {
                    return output(szirHtyoYmd);
                }
                continue;
            }

            if (priorClosureReasons.contains(gasMeterRecord.hesnJysb())) {
                return output(szirHtyoYmd);
            }

            hasOtherClosingReason = true;
        }

        if (!hasOpeningMeter && hasOtherClosingReason) {
            return output(kasgYmd);
        }
        if (!ksenHkkKnuKigenYmd.isBefore(szirHtyoYmd)) {
            return output(szirHtyoYmd);
        }
        if (kasgYmd.isBefore(szirHtyoYmd)) {
            return output(kasgYmd);
        }
        return output(szirHtyoYmd);
    }

    private static ConsumptionTaxRateDeterminationDateConfirmationAtOpeningOutput output(LocalDate date) {
        return new ConsumptionTaxRateDeterminationDateConfirmationAtOpeningOutput(date);
    }
}
