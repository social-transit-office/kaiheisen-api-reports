package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningDateCheckFutureDate;

/** SV-180 の実装。 */
public final class OpeningDateCheckFutureDateImpl implements OpeningDateCheckFutureDate {

    /**
     * 基準日補完に使用するシステム日時プロバイダ。
     */
    private final SystemDateTimeProvider systemDateTimeProvider;

    /**
     * 相関ID・報告システム種別・システム日時を保持するリクエストコンテキスト。
     */
    private final RequestContext requestContext;

    /**
     * 既定のシステム日時プロバイダを利用するコンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    public OpeningDateCheckFutureDateImpl(RequestContext requestContext) {
        this(new DefaultSystemDateTimeProvider(), requestContext);
    }

    /**
     * テスト差し替え用のシステム日時プロバイダを受け取るコンストラクタ。
     *
     * @param systemDateTimeProvider システム日時プロバイダ
     * @param requestContext リクエストコンテキスト
     */
    public OpeningDateCheckFutureDateImpl(
            SystemDateTimeProvider systemDateTimeProvider,
            RequestContext requestContext) {
        this.systemDateTimeProvider = Objects.requireNonNull(systemDateTimeProvider, "systemDateTimeProvider");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
    }

    /**
     * 開栓作業年月日が未来日であれば開栓共通エラーを送出する。
     *
     * <p>入力値の必須性や null バリデーションは上位層責務とし、
     * 本メソッドでは SV レイヤ規約に従って public 引数の null ガードを行わない。</p>
     *
     * @param kasgYmd 開栓作業年月日
     */
    @Override
    public void execute(LocalDate kasgYmd) {
        if (kasgYmd.isAfter(resolveBaseDate())) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
        }
    }

    /**
     * 未来日判定に用いる基準日を解決する。
     *
     * <p>RequestContext に systemDatetime が設定されている場合はその年月日を採用し、
     * 未設定の場合は SystemDateTimeProvider から現在日付を取得する。</p>
     *
     * @return 未来日判定の基準日
     */
    private LocalDate resolveBaseDate() {
        return requestContext.getSystemDatetime()
                .map(LocalDateTime::toLocalDate)
                .orElseGet(systemDateTimeProvider::currentDate);
    }
}
