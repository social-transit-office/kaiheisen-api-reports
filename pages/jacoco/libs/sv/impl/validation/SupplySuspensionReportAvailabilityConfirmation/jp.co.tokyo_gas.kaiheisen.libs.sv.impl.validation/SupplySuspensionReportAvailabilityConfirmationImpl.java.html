package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtksts.GmtkstsDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtksts.GmtkstsRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.taksmtr.TaksmtrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.taksmtr.TaksmtrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.SupplySuspensionReportAvailabilityConfirmation;

/**
 * 概要: SV-277（供給停止報告可否確認）の実装クラス。
 * 主な仕様:
 * - 託送メータ情報の仮登録有無を起点に、スイッチング受付エラーおよび他小売開栓落成待ちエラーを判定する。
 * - スイッチ申込経路では、確定前ガスメーター検針結果件数と定例検針予定年月日を使って 020200048 を判定する。
 * - スイッチ申込以外の経路では、託送開始年月日を使って 020200053 を判定する。
 * 既知の制約:
 * - `correlationId`、`hukkSysShbt`、`systemDatetime` は SV 規約に従い公開引数へ含めない。
 * - DA-019-R002 の取得結果が複数件の場合でも、設計に明示された優先順がないため先頭レコードを評価対象とする。
 * - public 引数の null ガードや入力補正は行わず、下位比較処理に委譲する。
 */
public final class SupplySuspensionReportAvailabilityConfirmationImpl
        implements SupplySuspensionReportAvailabilityConfirmation {

    private final TaksmtrDataAccess taksmtrDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final GmtkstsDataAccess gmtkstsDataAccess;

    private final String temporaryRegistrationStatus;

    private final String switchApplicationCisJudgmentCode;

    private final List<String> readingReasonTypes;

    /**
     * 概要: 判定に必要な依存と CIRIUS 定数を初期化する。
     * 主な仕様:
     * - 託送契約状態コード 0H800 の仮登録コードを必須解決して保持する。
     * - 託送契約ＣＩＳ判定コード 1AP00 のスイッチ申込コードを必須解決して保持する。
     * - 検針結果作成事由種別 07000 の定例・臨時コードを必須解決して保持する。
     *
     * @param taksmtrDataAccess 託送メータ情報データアクセス
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果情報データアクセス
     * @param gmtkstsDataAccess ガスメーター定例検針対象管理情報データアクセス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public SupplySuspensionReportAvailabilityConfirmationImpl(
            TaksmtrDataAccess taksmtrDataAccess,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            GmtkstsDataAccess gmtkstsDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.taksmtrDataAccess = Objects.requireNonNull(taksmtrDataAccess, "taksmtrDataAccess");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.gmtkstsDataAccess = Objects.requireNonNull(gmtkstsDataAccess, "gmtkstsDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.temporaryRegistrationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TEMPORARY_REGISTRATION);
        this.switchApplicationCisJudgmentCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.SWITCH_APPLICATION);
        this.readingReasonTypes = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TEMPORARY));
    }

    /**
     * 概要: 供給停止報告の可否を判定する。
     * 主な仕様:
     * - DA-019-R002 の取得結果が 0 件なら後続判定を行わず正常終了する。
     * - スイッチ申込経路では DA-052-R007 と DA-036-R001 を使って 020200048 を判定する。
     * - スイッチ申込以外の経路では託送開始年月日と閉栓年月日を比較し、020200053 を判定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param hesnYmd 閉栓年月日
     */
    @Override
    public void execute(Long gasMtstNo, LocalDate hesnYmd) {
        List<TaksmtrRecord> taksmtrRecords =
                taksmtrDataAccess.findAllByGmtSetNoAndTaksKyJtCd(
                        gasMtstNo,
                        temporaryRegistrationStatus);
        if (taksmtrRecords.isEmpty()) {
            return;
        }

        TaksmtrRecord targetRecord = taksmtrRecords.getFirst();
        if (switchApplicationCisJudgmentCode.equals(targetRecord.taksCisHntCd())) {
            validateSwitchApplication(gasMtstNo, hesnYmd);
            return;
        }

        if (!targetRecord.taksStartYmd().isAfter(hesnYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200053);
        }
    }

    private void validateSwitchApplication(Long gasMtstNo, LocalDate hesnYmd) {
        long count = ktmgmkkDataAccess.countByGasMtstNoAndKnkkSksiJysb(gasMtstNo, readingReasonTypes);
        if (count != 0L) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200048);
        }

        Optional<GmtkstsRecord> gmtkstsRecord = gmtkstsDataAccess.findByGasMtstNo(gasMtstNo);
        if (gmtkstsRecord.isPresent() && !gmtkstsRecord.orElseThrow().tkytYmd().isAfter(hesnYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200048);
        }
    }
}
