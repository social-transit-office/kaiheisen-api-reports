    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            String trkIsrrGymCd,

            String rirekiMemo,
            OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailTkssvkyInf tkssvkyInf,
            OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailIsrrkrUpdInf isrrkrUpdInf) {
        LocalDateTime systemDatetime = requireSystemDatetime();

        boolean wkSvkyUpdFlg = false;
        boolean wkSuguDelFlg = false;
        boolean wkSuguKbnUpdFlg = false;
        boolean wkKsenFlg = false;
        boolean wkKkytskrKktUmFlg = false;
        String wkSuguHntKbn = null;

        GmtsetdRecord gmtsetdRecord = findGmtsetd(gasMtstNo, trkKnsYmd);
        GmtsetkRecord gmtsetkRecord = findGmtsetk(gasMtstNo, trkKnsYmd);

        previousGenerationJudgment.execute(gasMtstNo, trkKnsYmd);

        String tksuKijsCd = retailerCodeTokyoGas;
        String settlementTksuKijsCd = tkssvkyInf.tksuKijsCd();
        OpeningCancelFeasibilityJudgmentOutput feasibilityOutput = openingCancelFeasibilityJudgment.execute(
                gasMtstNo,
                trkKnsYmd,
                gmtsetdRecord.mtgMtstNo(),
                tksuKijsCd,
                gmtsetdRecord.ckrtGmscJtkb(),
                tkssvkyInf.tksuSvkyNo());

        OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput dedicatedMeterCheckOutput =
                dedicatedMeterCheck.execute(
                        gmtsetdRecord.ckrtGmscJtkb(),
                        gasMtstNo,
                        trkKnsYmd,
                        gmtsetdRecord.szcGkkCd(),
                        gmtsetdRecord.szcMchCd(),
                        gmtsetdRecord.szcCmazCd());

        OpeningCancelFeasibilityJudgmentPaymentContractOutput paymentContractOutput =
                paymentContractJudgment.execute(
                        feasibilityOutput.svmstkrInf().svkyNo(),
                        trkKnsYmd);

        TskgmstRecord previousTskgmst = tskgmstDataAccess
                .findByGasMtstNoAndTkysYmdBeforeAndTksuSvkkJtkb(
                        gasMtstNo,
                        trkKnsYmd,
                        transportationServiceContractStatusInvalid)
                .orElse(null);

        if (previousTskgmst != null && tkssvkyDataAccess
                .findAllByTksuSvkyNoAndTkysYmdAndTksuSvkyJtkbList(
                        previousTskgmst.tksuSvkyNo(),
                        trkKnsYmd,
                                                transportationServiceContractStatusValidOrEnded)
                .isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
        }

        OpeningCancelMeterLocationUpdateOutput meterLocationUpdateOutput =
                openingCancelMeterLocationUpdate.execute(
                        gasMtstNo,
                        trkKnsYmd,
                        toOpeningCancelMeterLocationUpdateGmtsetdInf(gmtsetdRecord),
                        toOpeningCancelMeterLocationUpdateGmtsetkInf(gmtsetkRecord));

        ContractValidityJudgmentAtOpeningOutput contractValidityOutput =
                contractValidityJudgmentAtOpening.execute(
                        previousTskgmst == null ? null : previousTskgmst.tksuSvkyNo(),
                        tkssvkyInf.tksuSvkyNo(),
                        trkKnsYmd,
                        tkssvkyInf.tkysYmd(),
                        gmtsetdRecord.kasJysb(),
                        tkssvkyInf.tksuSvksJysb());

        ComprehensiveJudgmentProcessOutput comprehensiveOutput =
                comprehensiveJudgmentProcess.execute(
                        gasMtstNo,
                        null,
                        comprehensiveJudgmentProcessingNoRelatedEnd);
        List<SuguGmtInf> suguGmtInfList = emptyIfNull(comprehensiveOutput.suguGmtInfList());

        if (Boolean.TRUE.equals(feasibilityOutput.kkytskrUmFlg())) {
            ServiceContractSupplySuspensionUpdateWithNoSettlementOutput noSettlementOutput =
                    noSettlementUpdate.execute(
                            comprehensiveOutput.suguHntFlg(),
                            feasibilityOutput.sny4XUmFlg(),
                            toNoSettlementSvckyaInf(feasibilityOutput.svckyaInf()),
                            toRegisterSuguGmtInfList(suguGmtInfList),
                            trkKnsYmd,
                            toRegisterKkytskrInf(gasMtstNo, feasibilityOutput.kkytskrInf()));
            wkKsenFlg = Boolean.TRUE.equals(noSettlementOutput.ksenFlg());
        } else {
            wkSuguHntKbn = determineSuguHntKbn(suguGmtInfList.size());
            ServiceContractSupplySuspensionUpdateWithSettlementOutput settlementOutput =
                    settlementUpdate.execute(
                            wkSuguHntKbn,
                            contractValidityOutput.ksenYukuFlg(),
                            feasibilityOutput.sny4XUmFlg(),
                            gasMtstNo,
                            trkKnsYmd,
                            gmtsetdRecord.kasYmd(),
                            gmtsetdRecord.ckrtGmscJtkb(),
                            suguGmtInfList.size(),
                            toSettlementSuguGmtInfList(suguGmtInfList),
                            toSettlementKtmgmkkInf(feasibilityOutput.ktmgmkkInf()),
                            new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model
                                    .ServiceContractSupplySuspensionUpdateWithSettlementTkssvkyInf(
                                            settlementTksuKijsCd),
                            toSettlementSvckyaInf(feasibilityOutput.svckyaInf()),
                            toSettlementSvmstkrInf(feasibilityOutput.svmstkrInf()),
                            toSettlementSihakyInf(paymentContractOutput.sihakyInf()),
                            toSettlementSihakyKaTrksInf(paymentContractOutput.sihakyKaTrksInf()),
                            toSettlementSvsihkrInf(paymentContractOutput.svsihkrInf()),
                            toSettlementGmtsetdInfList(dedicatedMeterCheckOutput.gmtsetdInfList()));
            wkSvkyUpdFlg = true;
            wkSuguDelFlg = Boolean.TRUE.equals(settlementOutput.suguDelFlg());
            wkSuguKbnUpdFlg = Boolean.TRUE.equals(settlementOutput.suguKbnUpdFlg());
            wkKsenFlg = Boolean.TRUE.equals(settlementOutput.ksenFlg());
            wkKkytskrKktUmFlg = Boolean.TRUE.equals(settlementOutput.kkytskrKktUmFlg());
        }

        List<ItensakRecord> itensakRecords = itensakDataAccess.findAllByGmtSetNoAndSyokyNo(
                gasMtstNo,
                feasibilityOutput.svckyaInf().svkyNo());
        if (!itensakRecords.isEmpty()) {
            ItensakRecord itensakRecord = itensakRecords.get(0);
            if (Objects.equals(itensakRecord.recUpdYmd(), systemDatetime.toLocalDate())
                    && Objects.equals(itensakRecord.kosinTranNo(), KHSS212)) {
                itensakDataAccess.updateByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
                        toItensakWriteRecord(itensakRecord, transferDestinationCancellationFlagNoConcept),
                        Set.of(ItensakColumn.ITEN_TKS_CD),
                        new ItensakUpdateKey(
                                itensakRecord.gmtSetNo(),
                                itensakRecord.syokyNo(),
                                itensakRecord.itenSakSeqNo(),
                                itensakRecord.recUpdJnjNo()),
                        AuditContext.of(KHSS331));
            }
        }

        customerNumberingRelationUpdateLivalitAdapter.update(new CustomerNumberingRelationUpdateRequest(
                hukkUsrId,
                gasMtstNo,
                trkKnsYmd.format(DateTimeFormatter.BASIC_ISO_DATE),
                null,
                DGWRI_UMU_FLG_NO_CONCEPT,
                SKJ_FLG_CANCEL,
                0L));

        cancelTransferWorkHistoryCreation.execute(
                cancelTransferWorkHistoryCreationProcessingClosing,
                gasMtstNo,
                transferWorkHistoryWorkCodeConduitOpeningCancel,
                transferWorkHistoryWorkDetailsCodeConduitOpeningCancel,
                trkIsrrGymCd,
                transferWorkHistoryWorkDetailCodeRetailOpeningCancel,
                transferWorkHistoryReportingRouteOnline,
                trkKnsYmd,
                hukkUsrId,
                hukkUnmKj,
                rirekiMemo,
                tksuKijsCd,
                wkSvkyUpdFlg,
                wkSuguDelFlg,
                null,
                List.of(),
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.IsrrkrUpdInf(
                        isrrkrUpdInf.duknIsrgCd(),
                        isrrkrUpdInf.duknIsrsCd(),
                        isrrkrUpdInf.isrrGymCd(),
                        isrrkrUpdInf.isrrGmssCd()));

        if (Boolean.FALSE.equals(feasibilityOutput.sny4XUmFlg())) {
            billingCollectionPlatformIntegrationAtOpeningCancel.execute(
                    gasMtstNo,
                    trkKnsYmd,
                    new jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening
                            .billingcollectionplatformintegrationatopeningcancel.model.GmtsetdOgaInf(
                                    meterLocationUpdateOutput.hesnYmd(),
                                    meterLocationUpdateOutput.hesnJysb()),
                    previousTskgmst == null ? null : previousTskgmst.tksuSvkyNo(),
                    tkssvkyInf.tksuSvkyNo(),
                    toBillingSvckyaInf(feasibilityOutput.svckyaInf()),
                    toBillingSihakyInf(paymentContractOutput.sihakyInf()),
                    contractValidityOutput.ksenYukuFlg(),
                    toBillingSuguGmtInfList(suguGmtInfList),
                    wkKsenFlg,
                    wkKkytskrKktUmFlg,
                    feasibilityOutput.kkytskrUmFlg(),
                    wkSuguKbnUpdFlg);
        }
    }

        /**
         * 概要: RequestContext からシステム日時を取得する。
         *
         * 主な仕様:
         * - systemDatetime が未設定の場合は fail-fast で例外を送出する。
         *
         * 既知の制約:
         * - RequestContext への値設定は呼出し元で完了している前提である。
         *
         * @return RequestContext に保持されたシステム日時
         */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
