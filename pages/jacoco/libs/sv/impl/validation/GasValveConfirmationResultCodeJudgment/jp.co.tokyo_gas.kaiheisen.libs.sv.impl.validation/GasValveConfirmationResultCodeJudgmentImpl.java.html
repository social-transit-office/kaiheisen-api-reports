    @Override
    public GasValveConfirmationResultCodeJudgmentOutput execute(
            String cstBrSign, String stkJkCd, String tsyGssnSign, String tnknTsyKbCd) {

        String gassKknKkaCd = null;
        String ulfKknKkaCd = null;
        String flxStzTktCd = null;

        if (isCompleteAndNotFlexibleTarget(stkJkCd, cstBrSign)) {
            gassKknKkaCd = evaluateForPrimaryBranch(cstBrSign, tsyGssnSign);
        } else {
            gassKknKkaCd = evaluateForSecondaryBranch(cstBrSign, stkJkCd, tsyGssnSign, tnknTsyKbCd);
            ulfKknKkaCd = evaluateUlFlexibleBranch(cstBrSign, stkJkCd);
            flxStzTktCd = evaluateFlexiblePipeBranch(cstBrSign, stkJkCd);
        }

        return new GasValveConfirmationResultCodeJudgmentOutput(
                gassKknKkaCd, ulfKknKkaCd, flxStzTktCd);
    }

    /**
     * 概要: 進捗完了かつフレキ関連対象外条件を判定する。 主な仕様: - 進捗状況コードが完了系（1/2/4）か確認する。 -
     * お客さま分類サインがフレキ関連対象（5/6/B/C/D/E/F/G）以外であることを確認する。 既知の制約: - 引数が null の場合は false と判定する。
     *
     * @param stkJkCd 進捗状況コード
     * @param cstBrSign お客さま分類サイン
     * @return 判定結果
     */
    private boolean isCompleteAndNotFlexibleTarget(String stkJkCd, String cstBrSign) {
        return progressComplete.contains(stkJkCd) && !custSignFlexibleTarget.contains(cstBrSign);
    }

    /**
     * 概要: 主分岐（1.a）でガス栓確認作業結果コードを判定する。 主な仕様: - 第1回/第2回PR対象等は不在:用紙投函を返す。 - 追加対策対象は対象ガス栓有無サインで分岐する。
     * 既知の制約: - 条件非該当時は null を返す。
     *
     * @param cstBrSign お客さま分類サイン
     * @param tsyGssnSign 対象ガス栓有無サイン
     * @return 判定結果コード
     */
    private String evaluateForPrimaryBranch(String cstBrSign, String tsyGssnSign) {
        if (custSignAbsentNotice.contains(cstBrSign)) {
            return resultAbsentNotice;
        }
        if (custSignExtraTarget.contains(cstBrSign)) {
            if (valveExists.equals(tsyGssnSign)) {
                return resultExtraWithValve;
            }
            if (valveNotExists.equals(tsyGssnSign)) {
                return resultExtraWithoutValve;
            }
            return resultScrewTarget;
        }
        return null;
    }

    /**
     * 概要: 副分岐（1.b）でガス栓確認作業結果コードを判定する。 主な仕様: - お客さま分類サインがPR対象の場合、点検対象区分と進捗状況で未確認/拒否を返す。 -
     * 追加対策対象の場合、進捗および対象ガス栓有無で条件付/対象/追加対策実施済を返す。 既知の制約: - 条件非該当時は null を返す。
     *
     * @param cstBrSign お客さま分類サイン
     * @param stkJkCd 進捗状況コード
     * @param tsyGssnSign 対象ガス栓有無サイン
     * @param tnknTsyKbCd 点検対象区分コード
     * @return 判定結果コード
     */
    private String evaluateForSecondaryBranch(
            String cstBrSign, String stkJkCd, String tsyGssnSign, String tnknTsyKbCd) {

        String result = null;

        if (custSignPrTarget.contains(cstBrSign) && inspectionTarget.contains(tnknTsyKbCd)) {
            if (progressRejected.contains(stkJkCd)) {
                result = resultRejected;
            } else if (progressUnchecked.contains(stkJkCd)) {
                result = resultUnchecked;
            }
        }

        if (custSignExtraTarget.contains(cstBrSign)) {
            if (progressRejected.contains(stkJkCd)) {
                result = resultScrewConditional;
            } else if (Set.of(
                            ConstantLookupKeys.Cirius.ProgressStatusCode
                                    .NOT_VISITED_INCLUDES_FIRST_ROUND_ABSENCE_ONLY,
                            ConstantLookupKeys.Cirius.ProgressStatusCode
                                    .PARTIALLY_CONFIRMED_PARTIALLY_UNCONFIRMED,
                            ConstantLookupKeys.Cirius.ProgressStatusCode
                                    .CONFIRMATION_REFUSED_THIS_TIME_ONLY,
                            ConstantLookupKeys.Cirius.ProgressStatusCode.ABSENCE_COMPLETED)
                    .contains(stkJkCd)) {
                result = resultScrewTarget;
            } else if (ConstantLookupKeys.Cirius.ProgressStatusCode.CLOSED_SUSPENDED_REMOVED.equals(
                    stkJkCd)) {
                if (valveNotExists.equals(tsyGssnSign)) {
                    result = resultExtraWithoutValve;
                } else {
                    result = resultScrewTarget;
                }
            }
        }

        return result;
    }

    /**
     * 概要: フレキU-L対象分岐で結果コードを判定する。 主な仕様: - 進捗状況コードに応じて不在:用紙投函/未確認/拒否を返す。 既知の制約: - 条件非該当時は null を返す。
     *
     * @param cstBrSign お客さま分類サイン
     * @param stkJkCd 進捗状況コード
     * @return 判定結果コード
     */
    private String evaluateUlFlexibleBranch(String cstBrSign, String stkJkCd) {
        if (!custSignFlexibleUl.contains(cstBrSign)) {
            return null;
        }
        if (progressUlAbsentNotice.contains(stkJkCd)) {
            return resultAbsentNotice;
        }
        if (progressUlUnchecked.contains(stkJkCd)) {
            return resultUnchecked;
        }
        if (ConstantLookupKeys.Cirius.ProgressStatusCode.ABSENCE_COMPLETED.equals(stkJkCd)) {
            return resultRejected;
        }
        return null;
    }

    /**
     * 概要: フレキ管接続対象分岐で点検対象コードを判定する。 主な仕様: - 進捗状況コードに応じて未確認/拒否/不在:用紙投函を返す。 既知の制約: - 条件非該当時は null を返す。

    private String evaluateFlexiblePipeBranch(String cstBrSign, String stkJkCd) {

    }
