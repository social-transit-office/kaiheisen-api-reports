package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.math.BigDecimal;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.TgPipelineReadingJudgment;

/**
 * 概要: SV-279（ＴＧ導管指示数判定）の実装クラス。
 * 主な仕様:
 * - ガスメーター指示数が `指示数回帰判定値数 * 指示数最小値数` 以上かを判定する。
 * - 閾値以上の場合のみ開閉栓処理区分を評価し、開栓処理なら 020100039 の
 *   KaisenRuleException、それ以外なら 020200021 の HeisenRuleException を送出する。
 * - 判定に必要な開閉栓処理区分定数はコンストラクタで解決して保持する。
 * 既知の制約:
 * - 数値系引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 * - 開閉栓処理区分は null を許容し、開栓処理以外として閉栓側エラー分岐で扱う。
 */
public final class TgPipelineReadingJudgmentImpl implements TgPipelineReadingJudgment {

    /** 開閉栓処理区分: 開栓処理。 */
    private final String openingProcess;

    /**
     * 概要: 判定に必要な開閉栓定数を解決して初期化する。
     * 主な仕様:
     * - 開閉栓定数サービスから KH003 の開栓処理値を必須解決する。
     * - 解決済み値は execute の分岐判定で再利用する。
     * 既知の制約:
     * - 定数サービスの null は許容しない。
     *
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public TgPipelineReadingJudgmentImpl(KaiheisenConstantService kaiheisenConstantService) {
        KaiheisenConstantService resolvedService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.openingProcess = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedService,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.OPENING_PROCESS);
    }

    /**
     * 概要: ＴＧ導管指示数判定を実行する。
     * 主な仕様:
     * - ガスメーター指示数が閾値未満の場合は正常終了する。
     * - ガスメーター指示数が閾値以上の場合は開閉栓処理区分を評価し、
      *   開栓処理なら 020100039 の KaisenRuleException、
      *   それ以外なら 020200021 の HeisenRuleException を送出する。
     * 既知の制約:
     * - 引数値の正規化や blank 補完は行わない。
     */
    @Override
    public void execute(
            String khsSyoriKbn,
            Long sjskHticSu,
            BigDecimal sjsuMincSu,
            Long gmtShjSu) {
        if (!isThresholdMet(sjskHticSu, sjsuMincSu, gmtShjSu)) {
            return;
        }

        if (isOpeningProcess(khsSyoriKbn)) {
            throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100039);
        }

        throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200021);
    }

    /**
     * 概要: ガスメーター指示数が判定閾値以上かを判定する。
     * 主な仕様:
     * - `指示数回帰判定値数 * 指示数最小値数` を閾値として比較する。
     * - 閾値ちょうどは成立とみなす。
     * 既知の制約:
     * - 引数が null の場合は NullPointerException となる。
     *
     * @param sjskHticSu 指示数回帰判定値数
     * @param sjsuMincSu 指示数最小値数
     * @param gmtShjSu ガスメーター指示数
     * @return 閾値以上であれば true
     */
    private boolean isThresholdMet(Long sjskHticSu, BigDecimal sjsuMincSu, Long gmtShjSu) {
        BigDecimal threshold = BigDecimal.valueOf(sjskHticSu).multiply(sjsuMincSu);
        BigDecimal meterReading = BigDecimal.valueOf(gmtShjSu);
        return meterReading.compareTo(threshold) >= 0;
    }

    /**
     * 概要: 開閉栓処理区分が開栓処理かを判定する。
     * 主な仕様:
     * - コンストラクタで解決した KH003-1 と一致する場合に true を返す。
     * 既知の制約:
     * - 引数が null の場合は false と同義で扱われ、後続の閉栓側エラー分岐へ進む。
     *
     * @param khsSyoriKbn 開閉栓処理区分
     * @return 開栓処理であれば true
     */
    private boolean isOpeningProcess(String khsSyoriKbn) {
        return openingProcess.equals(khsSyoriKbn);
    }
}
