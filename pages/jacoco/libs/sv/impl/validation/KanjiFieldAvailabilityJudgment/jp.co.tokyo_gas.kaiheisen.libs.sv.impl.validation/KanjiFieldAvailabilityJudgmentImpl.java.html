package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.KanjiFieldAvailabilityJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.KanjiFieldAvailabilityJudgmentOutput;

/**
 * <ul>
 *   <li>概要: 漢字項目1から3の設定有無から、可否フラグ1から3を判定するサービス実装。</li>
 *   <li>主な仕様: 各入力値が null または空白のみ文字列の場合は未設定とみなし false を返し、それ以外は true を返す。</li>
 *   <li>既知の制約: 3つの入力は独立に判定し、RequestContext や他の下位依存は使用しない。</li>
 * </ul>
 */
public class KanjiFieldAvailabilityJudgmentImpl implements KanjiFieldAvailabilityJudgment {

    /**
     * <ul>
     *   <li>概要: 漢字項目1から3の設定有無を判定し、可否フラグを返却する。</li>
 *   <li>主な仕様: null または空白のみ文字列の入力は未設定とみなし false、それ以外は true を設定する。</li>
     *   <li>既知の制約: 入力項目ごとに独立判定を行い、相互依存は持たない。</li>
     * </ul>
     *
     * @param kjKmk1 漢字項目1
     * @param kjKmk2 漢字項目2
     * @param kjKmk3 漢字項目3
     * @return 可否フラグ1から3を持つ出力DTO
     */
    @Override
    public KanjiFieldAvailabilityJudgmentOutput execute(
            String kjKmk1, String kjKmk2, String kjKmk3) {
        return new KanjiFieldAvailabilityJudgmentOutput(
                hasValue(kjKmk1), hasValue(kjKmk2), hasValue(kjKmk3));
    }

    /**
     * <ul>
     *   <li>概要: 漢字項目に値が設定されているかを判定する。</li>
 *   <li>主な仕様: null または空白のみ文字列の場合は未設定とみなし false を返す。</li>
 *   <li>既知の制約: trim は行わず、String#isBlank の判定結果に準拠する。</li>
     * </ul>
     *
     * @param value 判定対象の漢字項目
     * @return 値が設定されている場合 true、未設定の場合 false
     */
    private static Boolean hasValue(String value) {
        return value == null || value.isBlank() ? Boolean.FALSE : Boolean.TRUE;
    }
}
