package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.SeriesDedicatedContractInProgressErrorJudgment;

/**
 * 直列専用契約中エラー判定サービス実装。
 *
 * <p>概要: 契約終了判定フラグと直列専用契約中フラグの組み合わせに基づき、
 * 直列専用契約中エラーの送出要否を判定する。
 *
 * <p>主な仕様:
 * <ul>
 *   <li>{@code kiyEndFlg == true} かつ {@code snyKyktFlg == true} の場合、
 *       {@link jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException} を送出する。</li>
 *   <li>上記以外の場合、正常終了として判定結果を返却する。</li>
 * </ul>
 *
 * <p>既知の制約:
 * <ul>
 *   <li>本実装は引数のnull許容/不許可を設計・仕様化しません（SV規約に準拠）。</li>
 *   <li>条件式で{@link Boolean}型を直接評価しているため、引数がnullの場合はアンボクシングによりNPEとなる可能性があります。</li>
 * </ul>
 */
public final class SeriesDedicatedContractInProgressErrorJudgmentImpl
        implements SeriesDedicatedContractInProgressErrorJudgment {

    /**
     * 契約終了判定と直列専用契約中判定の組み合わせから業務エラー送出要否を判定する。
     *
     * <p>概要: 入力された 2 つの判定フラグを評価し、
     * 直列専用契約中エラーに該当する場合のみ業務エラー
     * （{@link jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException}）を送出する。
     *
     * <p>主な仕様:
     * <ul>
     *     <li>{@code kiyEndFlg} と {@code snyKyktFlg} がともに {@code true} の場合、
     *         {@link jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException} を送出する。</li>
     *     <li>それ以外の組み合わせでは、何も返さずに終了する。</li>
     * </ul>
     */
    @Override
    public void execute(
            final Boolean kiyEndFlg,
            final Boolean snyKyktFlg) {
        if (kiyEndFlg && snyKyktFlg) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200050);
        }
    }
}
