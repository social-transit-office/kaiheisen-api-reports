package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingInfoAcquisitionForOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingInfoAcquisitionForOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningDateCheckForOpening;

/**
 * 開栓日チェック（開栓専用）サービス実装。
 */
public class OpeningDateCheckForOpeningImpl implements OpeningDateCheckForOpening {

    private final String gasMeterEquipmentStatusType;

    private final MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService                 CIRIUS コンスタント取得サービス
     * @param meterReadingInfoAcquisitionForOpening 検針情報取得（開栓）サービス
     */
    public OpeningDateCheckForOpeningImpl(CiriusConstantService ciriusConstantService,
            MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening) {

        this.meterReadingInfoAcquisitionForOpening = meterReadingInfoAcquisitionForOpening;

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(ciriusConstantService,
                "ciriusConstantService");

        this.gasMeterEquipmentStatusType = RequiredConstantResolver.requireCiriusConstant(resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
    }

    /**
     * 概要: 供内管・ガスメーター設備状態区分が取付中以外の場合、開栓年月日をチェックする。 主な仕様: -
     * 供内管・ガスメーター設備状態区分が取付中の場合、処理を終了する。 取付中以外の場合、SV-027で取得した開栓報告可能期限日と引数の開栓年月日を比較する。
     *
     * @param kasYmd      開栓年月日
     * @param kngmStbJtkb 供内管・ガスメーター設備状態区分
     * @param gasMtstNo   ガスメーター設置場所番号
     */
    @Override
    public void execute(LocalDate kasYmd, String kngmStbJtkb, Long gasMtstNo) {

        if (Objects.equals(gasMeterEquipmentStatusType, kngmStbJtkb)) {

            MeterReadingInfoAcquisitionForOpeningOutput meterReadingInfoAcquisitionForOpeningOutput = meterReadingInfoAcquisitionForOpening
                    .execute(gasMtstNo);

            if (kasYmd.isBefore(meterReadingInfoAcquisitionForOpeningOutput.ksenHkkKnuKigenYmd())) {

                throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100018);
            }
        }
        return;
    }

}
