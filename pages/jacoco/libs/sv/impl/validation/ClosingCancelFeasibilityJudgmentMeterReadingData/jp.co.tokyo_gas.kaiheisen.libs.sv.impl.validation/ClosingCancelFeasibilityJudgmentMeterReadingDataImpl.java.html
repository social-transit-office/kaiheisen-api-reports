package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancelFeasibilityJudgmentMeterReadingData;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancelFeasibilityJudgmentMeterReadingDataOutput;

/**
 * 概要: SV-099（閉栓取消実施可否判定：検針データ）のサービス実装。
 * 主な仕様:
 * - `duknIsrsCd=CCJG1301` 以外の場合は KTMGMKK の最新レコードを取得し、報告年月日と
 *   RequestContext.systemDatetime、検針結果作成事由種別、報告年月日と `trkKnsYmd` の順に
 *   閉栓取消可否を判定する。
 * - 正常終了時は KTMGMKK の主要キーを API 出力へ写像する。
 * 既知の制約:
 * - `duknIsrsCd=CCJG1301` の分岐は設計曖昧さがあるため、best-effort で空 output を返し、
 *   この経路では RequestContext.systemDatetime を参照しない。
 * - RequestContext.systemDatetime は KTMGMKK 判定へ進む経路で必須であり、未設定時は
 *   fail-fast する。
 */
public class ClosingCancelFeasibilityJudgmentMeterReadingDataImpl
        implements ClosingCancelFeasibilityJudgmentMeterReadingData {

    private static final ClosingCancelFeasibilityJudgmentMeterReadingDataOutput EMPTY_OUTPUT =
            new ClosingCancelFeasibilityJudgmentMeterReadingDataOutput(null, null, null);

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final RequestContext requestContext;

    private final String supplyOutageWorkDetailCode;

    private final Set<String> supportedReasonTypes;

    /**
     * 概要: 必要な依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - KTMGMKK 参照と RequestContext を保持する。
     * - 判定に必要な J0496 / 07000 の定数を初期化時に検証する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * - 必須 CIRIUS 定数が不足している場合は初期化時点で失敗する。
     *
     * @param ktmgmkkDataAccess KTMGMKK データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public ClosingCancelFeasibilityJudgmentMeterReadingDataImpl(
            KtmgmkkDataAccess ktmgmkkDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.supplyOutageWorkDetailCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit
                        .SUPPLY_OUTAGE_CCJG1301);
        String relocationClosureReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                        .RELOCATION_CLOSURE);
        String unusedClosureReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.UNUSED_CLOSURE);
        this.supportedReasonTypes = Set.of(relocationClosureReasonType, unusedClosureReasonType);
    }

    /**
     * 概要: 検針データから閉栓取消可否を判定する。
     * 主な仕様:
     * - 対象業務詳細コードの場合は KTMGMKK を参照せず空 output を返す。
     * - 対象外の場合は最新 KTMGMKK レコードを検証し、正常時に主要キーを返す。
     * 既知の制約:
     * - `systemDatetime` は RequestContext から取得する。
     *
     * @param duknIsrsCd 異動・作業履歴業務詳細コード（導管）
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 判定結果
     */
    @Override
    public ClosingCancelFeasibilityJudgmentMeterReadingDataOutput execute(
            String duknIsrsCd,
            Long gasMtstNo,
            LocalDate trkKnsYmd) {
        if (shouldSkipEvaluation(duknIsrsCd)) {
            return EMPTY_OUTPUT;
        }
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        KtmgmkkRecord meterReadingResult = findRequiredMeterReadingResult(gasMtstNo);
        validateReportDateMatchesSystemDate(meterReadingResult, systemDate);
        validateSupportedReasonType(meterReadingResult);
        validateReportDateMatchesTrackingDate(meterReadingResult, trkKnsYmd);
        return toOutput(meterReadingResult);
    }

    /**
     * 概要: 判定対象外の業務詳細コードかを判定する。
     * 主な仕様:
     * - `CCJG1301` の場合は spec の 1.b 分岐として扱う。
     * - それ以外は KTMGMKK 判定へ進める。
     *
     * @param duknIsrsCd 異動・作業履歴業務詳細コード（導管）
     * @return 判定対象外なら true
     */
    private boolean shouldSkipEvaluation(String duknIsrsCd) {
        return supplyOutageWorkDetailCode.equals(duknIsrsCd);
    }

    /**
     * 概要: KTMGMKK の必須レコードを取得する。
     * 主な仕様:
     * - ガスメーター設置場所番号で最新レコードを取得する。
     * - 未取得時は業務エラー 020000021 を送出する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 取得した KTMGMKK レコード
     */
    private KtmgmkkRecord findRequiredMeterReadingResult(Long gasMtstNo) {
        return ktmgmkkDataAccess.findByGasMtstNo(gasMtstNo)
                .orElseThrow(() -> new BusinessRuleException(
                        ErrorCodeConstants.ERROR_CODE_020000021));
    }

    /**
     * 概要: 報告年月日とシステム日付の一致を検証する。
     * 主な仕様:
     * - KTMGMKK の検針業務担当者報告年月日と RequestContext.systemDatetime の日付部を比較する。
     * - 不一致時は業務エラー 020000021 を送出する。
     *
     * @param meterReadingResult KTMGMKK レコード
     * @param systemDate システム日付
     */
    private void validateReportDateMatchesSystemDate(
            KtmgmkkRecord meterReadingResult,
            LocalDate systemDate) {
        if (!Objects.equals(meterReadingResult.kgtsHukkYmd(), systemDate)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000021);
        }
    }

    /**
     * 概要: 許容された検針結果作成事由種別かを検証する。
     * 主な仕様:
     * - `07000 132` と `07000 136` のみを許容する。
     * - それ以外は業務エラー 020000022 を送出する。
     *
     * @param meterReadingResult KTMGMKK レコード
     */
    private void validateSupportedReasonType(KtmgmkkRecord meterReadingResult) {
        if (!supportedReasonTypes.contains(meterReadingResult.knkkSksiJysb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000022);
        }
    }

    /**
     * 概要: 検針年月日と取消対象検針年月日の一致を検証する。
     * 主な仕様:
     * - KTMGMKK の検針年月日と引数の取消対象検針年月日を比較する。
     * - 不一致時は業務エラー 020000022 を送出する。
     *
     * @param meterReadingResult KTMGMKK レコード
     * @param trkKnsYmd 取消対象検針年月日
     */
    private void validateReportDateMatchesTrackingDate(
            KtmgmkkRecord meterReadingResult,
            LocalDate trkKnsYmd) {
        if (!Objects.equals(meterReadingResult.knsYmd(), trkKnsYmd)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000022);
        }
    }

    /**
     * 概要: KTMGMKK レコードを API 出力へ変換する。
     * 主な仕様:
     * - 主要キー 3 項目のみを API 出力へ詰め替える。
     * - 下位 DA レコードをそのまま公開しない。
     *
     * @param meterReadingResult KTMGMKK レコード
     * @return API 出力モデル
     */
    private ClosingCancelFeasibilityJudgmentMeterReadingDataOutput toOutput(
            KtmgmkkRecord meterReadingResult) {
        return new ClosingCancelFeasibilityJudgmentMeterReadingDataOutput(
                meterReadingResult.gasMtstNo(),
                meterReadingResult.knsYmd(),
                meterReadingResult.knsKkaTsno());
    }

    /**
     * 概要: RequestContext のシステム日時を取得する。
     * 主な仕様:
     * - `RequestContext.systemDatetime` を取得する。
     * - 未設定時は項目名付きの IllegalStateException を送出する。
     *
     * @return システム日時
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
}
