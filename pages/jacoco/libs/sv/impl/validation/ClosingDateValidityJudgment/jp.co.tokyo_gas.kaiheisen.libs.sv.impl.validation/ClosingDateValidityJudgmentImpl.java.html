package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingDateValidityJudgment;

/**
 * 概要: SV-281（閉栓日妥当性判定）の業務判定を実装するクラス。
 * 主な仕様:
 * - 料金発生年月日がガスメーター取付年月日、開栓年月日、閉栓報告可能期限日のいずれかを遡る場合は閉栓共通エラーとする。
 * - 判定を通過した場合は何も返さずに終了する。
 * 既知の制約:
 * - 入力値の補正や null ガードは行わず、渡された日付をそのまま比較する。
 */
public final class ClosingDateValidityJudgmentImpl implements ClosingDateValidityJudgment {

    /**
     * 概要: 閉栓日妥当性判定を実行する。
     * 主な仕様:
     * - 料金発生年月日がガスメーター取付年月日、開栓年月日、閉栓報告可能期限日のいずれかを遡る場合は閉栓共通エラー 020200015 を送出する。
     * - すべての比較条件を満たした場合は何も返さずに正常終了する。
     * 既知の制約:
     * - correlationId、hukkSysShbt、systemDatetime は SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードや前後空白補正は行わず、渡された日付をそのまま比較する。
     * - 判定順は TSV step 定義に合わせて、ガスメーター取付年月日、開栓年月日、閉栓報告可能期限日の順とする。
     * - いずれの条件にも抵触しない場合は何も返さずに終了する。
     *
     * @param rkhsYmd 料金発生年月日
     * @param kasYmd 開栓年月日
     * @param hsenHkkKnuKigenYmd 閉栓報告可能期限日
     * @param gmtkYmd ガスメーター取付年月日
     */
    @Override
    public void execute(
            LocalDate rkhsYmd,
            LocalDate kasYmd,
            LocalDate hsenHkkKnuKigenYmd,
            LocalDate gmtkYmd) {
        if (rkhsYmd.isBefore(gmtkYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200015);
        }

        if (rkhsYmd.isBefore(kasYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200015);
        }

        if (rkhsYmd.isBefore(hsenHkkKnuKigenYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200015);
        }
    }
}
