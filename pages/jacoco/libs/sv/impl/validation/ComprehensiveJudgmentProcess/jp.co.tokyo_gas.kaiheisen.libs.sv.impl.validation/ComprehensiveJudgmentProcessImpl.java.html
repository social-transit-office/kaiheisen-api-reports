package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ComprehensiveJudgmentProcess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ComprehensiveJudgmentProcessOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.comprehensivejudgmentprocess.model.SuguGmtInf;

/**
 * 概要: SV-049（総合判定処理）の実装クラス。
 * 主な仕様:
 * - 総合判定処理区分に応じて検索基準日と料金計算関連状態区分一覧を切り替える。
 * - DA-041-R013 で主契約を特定し、存在する場合のみ DA-041-R017 で関連メーター一覧を取得する。
 * - 関連メーター一覧が 2 件以上の場合だけ総合判定フラグを true とする。
 * 既知の制約:
 * - `systemDatetime` は `KH029 1` の分岐で RequestContext から取得する。
 * - public 引数の null ガードや入力補正は追加せず、上位バリデーション済み入力を前提とする。
 */
public final class ComprehensiveJudgmentProcessImpl implements ComprehensiveJudgmentProcess {

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final RequestContext requestContext;

    private final String overallJudgmentNoRelatedEnd;

    private final String overallJudgmentRelatedEnd;

    private final String relatedValid;

    private final String relatedEndWait;

    private final String relatedEnd;

    /**
     * 概要: 必須依存と定数を解決して初期化する。
     * 主な仕様:
     * - DA-041 と RequestContext を保持する。
     * - 開閉栓定数 KH029 と CIRIUS 定数 J0611 の必須値を fail-fast で検証する。
     *
     * @param svmstkrDataAccess DA-041 サービス契約・メーター設置場所関連データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public ComprehensiveJudgmentProcessImpl(
            final SvmstkrDataAccess svmstkrDataAccess,
            final RequestContext requestContext,
            final CiriusConstantService ciriusConstantService,
            final KaiheisenConstantService kaiheisenConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.overallJudgmentNoRelatedEnd =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ComprehensiveJudgmentProcessing.TYPE,
                        ConstantLookupKeys.Kaiheisen.ComprehensiveJudgmentProcessing
                                .OVERALL_JUDGMENT_NO_RELATED_END);
        this.overallJudgmentRelatedEnd =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ComprehensiveJudgmentProcessing.TYPE,
                        ConstantLookupKeys.Kaiheisen.ComprehensiveJudgmentProcessing
                                .OVERALL_JUDGMENT_RELATED_END);
        this.relatedValid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.relatedEndWait =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);
        this.relatedEnd =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public ComprehensiveJudgmentProcessOutput execute(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd,
            final String suguHntShrKbn) {
        SearchCriteria criteria = resolveSearchCriteria(knskKjnYmd, suguHntShrKbn);
        Optional<SvmstkrRecord> baseRelation =
                svmstkrDataAccess.findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                        gasMtstNo,
                        criteria.knskKjnYmd(),
                        criteria.rkkiKnrnJtkbList());
        if (baseRelation.isEmpty()) {
            return nonComprehensiveOutput();
        }

        List<SvmstkrRecord> relatedMeterRelations =
                defaultList(
                        svmstkrDataAccess.findAllBySvkyNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkbList(
                                baseRelation.get().svkyNo(),
                                criteria.knskKjnYmd(),
                                criteria.rkkiKnrnJtkbList()));
        if (relatedMeterRelations.size() == 1) {
            return nonComprehensiveOutput();
        }

        return new ComprehensiveJudgmentProcessOutput(
                toSuguGmtInfList(relatedMeterRelations),
                true);
    }

    private SearchCriteria resolveSearchCriteria(
            final LocalDate knskKjnYmd,
            final String suguHntShrKbn) {
        if (overallJudgmentNoRelatedEnd.equals(suguHntShrKbn)) {
            return new SearchCriteria(
                    requireSystemDatetime().toLocalDate(),
                    List.of(relatedValid, relatedEndWait));
        }
        if (overallJudgmentRelatedEnd.equals(suguHntShrKbn)) {
            return new SearchCriteria(
                    knskKjnYmd,
                    List.of(relatedValid, relatedEndWait, relatedEnd));
        }
        return new SearchCriteria(
                knskKjnYmd,
                List.of(relatedValid, relatedEndWait, relatedEnd));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(
                        () -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private static List<SvmstkrRecord> defaultList(final List<SvmstkrRecord> source) {
        return source == null ? List.of() : source;
    }

    private static ArrayList<SuguGmtInf> toSuguGmtInfList(
            final List<SvmstkrRecord> relatedMeterRelations) {
        ArrayList<SuguGmtInf> results = new ArrayList<>();
        for (SvmstkrRecord relatedMeterRelation : relatedMeterRelations) {
            results.add(new SuguGmtInf(relatedMeterRelation.mtstNo()));
        }
        return results;
    }

    private static ComprehensiveJudgmentProcessOutput nonComprehensiveOutput() {
        return new ComprehensiveJudgmentProcessOutput(null, false);
    }

    private record SearchCriteria(LocalDate knskKjnYmd, List<String> rkkiKnrnJtkbList) {
    }
}
