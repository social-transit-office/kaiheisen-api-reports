package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ContractValidityJudgmentAtOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ContractValidityJudgmentAtOpeningOutput;

/**
 * 開栓時の契約有効判定サービス実装。
 * 同一契約かどうか、日付条件、開栓事由種別、託送サービス契約開始事由種別を順に評価し、
 * 開栓有効フラグを返却する。
 */
public final class ContractValidityJudgmentAtOpeningImpl
        implements ContractValidityJudgmentAtOpening {

    private final String continuedUseOpeningReason;

    private final String switchingNewConsignmentServiceContractStartReason;

    /**
     * 判定に必要な CIRIUS 定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ContractValidityJudgmentAtOpeningImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolved =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.continuedUseOpeningReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType.CONTINUED_USE);
        this.switchingNewConsignmentServiceContractStartReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ConsignmentServiceContractStartReasonType
                                .SWITCHING_NEW);
    }

    /**
     * 開栓時の契約有効判定を実行する。
     * 1世代前契約番号が欠落している場合は、現契約と不同一として評価する。
     *
     * @param tksuSvkyNoOga 託送サービス契約番号（1世代前）
     * @param tksuSvkyNo 託送サービス契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param tkysYmd 適用開始年月日
     * @param kasJysb 開栓事由種別
     * @param tksuSvksJysb 託送サービス契約開始事由種別
     * @return 判定結果
     */
    @Override
    public ContractValidityJudgmentAtOpeningOutput execute(
            Long tksuSvkyNoOga,
            Long tksuSvkyNo,
            LocalDate trkKnsYmd,
            LocalDate tkysYmd,
            String kasJysb,
            String tksuSvksJysb) {
        boolean sameContract = Objects.equals(tksuSvkyNoOga, tksuSvkyNo);
        boolean ksenYukuFlg;

        if (sameContract) {
            if (tkysYmd.isBefore(trkKnsYmd)) {
                ksenYukuFlg = true;
            } else {
                ksenYukuFlg = !kasJysb.equals(continuedUseOpeningReason);
            }
        } else {
            ksenYukuFlg =
                    tksuSvksJysb.equals(switchingNewConsignmentServiceContractStartReason);
        }

        return new ContractValidityJudgmentAtOpeningOutput(ksenYukuFlg);
    }
}
