    private TkssvkyRecord findTransportContract(
            TskgmstRecord transportContractRelated,
            LocalDate trkKnsYmd) {
        if (transportContractRelated == null) {
            return null;
        }
        List<TkssvkyRecord> contracts =
                tkssvkyDataAccess.findAllByTksuSvkyNoAndTkysYmdAndTksuSvkyJtkbList(
                        transportContractRelated.tksuSvkyNo(),
                        trkKnsYmd,
                        List.of(underContractStatus, endOfContractStatus));
        if (contracts.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
        }
        return contracts.getFirst();
    }

    private void deleteTemporaryMeterReading(Long gasMtstNo, KtmgmkkInf temporaryMeterReading) {
        if (temporaryMeterReading == null) {
            throw createDa052DeleteInconsistency(gasMtstNo, null, null);
        }
        int deletedCount = ktmgmkkDataAccess.deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                gasMtstNo,
                temporaryMeterReading.knsYmd(),
                temporaryMeterReading.knsKkaTsno(),
                null);
        if (deletedCount == 0) {
            throw createDa052DeleteInconsistency(
                    gasMtstNo,
                    temporaryMeterReading.knsYmd(),
                    temporaryMeterReading.knsKkaTsno());
        }
    }

    private void updateOpeningErrorFollowCase(
            OpeningErrorFollowCancelFeasibilityJudgmentOutput cancelAvailability,
            RequestContextValues requestContextValues) {
        KseflInf kseflInf = Objects.requireNonNull(cancelAvailability.kseflInf(), "SV-052 output.kseflInf");
        String decodedOriginFunctionId = decodeOriginFunctionId(requestContextValues.hukkSysShbt());
        int updatedCount =
                kseflDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndHukkSysShbtAndSgyAnknNoAndRecUpdJnjNo(
                        buildKseflWriteRecord(),
                        KSEFL_TARGET_COLUMNS,
                        new KseflU001UpdateKey(
                                cancelAvailability.gmb(),
                                kseflInf.kasgYmd(),
                                kseflInf.kasHukkYmd(),
                                kseflInf.hukkSysShbt(),
                                kseflInf.sgyAnknNo(),
                                kseflInf.recUpdJnjNo()),
                        AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kseflDataAccess.tableName(),
                    KseflDataAccess.CallSpecs
                            .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_HUKK_SYS_SHBT_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("gmb", cancelAvailability.gmb())
                            .put("kasgYmd", kseflInf.kasgYmd())
                            .put("kasHukkYmd", kseflInf.kasHukkYmd())
                            .put("hukkSysShbt", kseflInf.hukkSysShbt())
                            .put("sgyAnknNo", kseflInf.sgyAnknNo())
                            .put("recUpdJnjNo", kseflInf.recUpdJnjNo())
                            .put("flwStCd", followNotStartedStatus)
                            .put("fltkYmd", null)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }
