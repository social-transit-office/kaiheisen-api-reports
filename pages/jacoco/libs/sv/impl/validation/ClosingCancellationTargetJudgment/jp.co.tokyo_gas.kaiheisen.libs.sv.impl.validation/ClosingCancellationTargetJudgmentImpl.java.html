package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasFeeContractTypeSortingContentAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.GasFeeContractTypeSortingContentAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GrkKsbHnsb;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.GrkKsbHwsb;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancellationTargetJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancellationTargetJudgmentOutput;

/** SV-126 閉栓時解約対象判定サービス実装. */
public final class ClosingCancellationTargetJudgmentImpl
        implements ClosingCancellationTargetJudgment {

    /** SV-259 ガス料金契約種別振分内容取得. */
    private final GasFeeContractTypeSortingContentAcquisition
            gasFeeContractTypeSortingContentAcquisition;

    /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 専用ガスメーター判定値. */
    private final String dedicatedGasMeterStatus;

    /** 閉栓時解約対象判定用の振分種別リスト. */
    private final List<GrkKsbHwsb> closingCancellationTargetList;

    /** 閉栓時解約対象の振分内容種別. */
    private final String closingCancellationTargetDistributionContent;

    /**
     * 必須依存を受け取って初期化する.
     *
     * @param sortingContentAcquisition SV-259 ガス料金契約種別振分内容取得
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param requestContext リクエストコンテキスト
     */
    public ClosingCancellationTargetJudgmentImpl(
            final GasFeeContractTypeSortingContentAcquisition
                    sortingContentAcquisition,
            final CiriusConstantService ciriusConstantService,
            final RequestContext requestContext) {
        this.gasFeeContractTypeSortingContentAcquisition =
                Objects.requireNonNull(
                        sortingContentAcquisition,
                        "gasFeeContractTypeSortingContentAcquisition");
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        String closingCancellationTargetDistributionType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionType.TYPE,
                        ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionType
                                .SUBJECT_TO_CANCELLATION_WHEN_THE_CAP_IS_CLOSED);
        this.closingCancellationTargetDistributionContent =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionContentType.TYPE,
                        ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionContentType
                                .CANCELLATION_WHEN_CLOSED_TARGET);
        this.closingCancellationTargetList = List.of(
                new GrkKsbHwsb(closingCancellationTargetDistributionType, null));
    }

    @Override
    public ClosingCancellationTargetJudgmentOutput execute(
            final String ckrtGmscJtkb,
            final Boolean svmstkrUmFlg,
            final Boolean dgrpKasFlg,
            final String grkKiyShbt) {
        if (isEarlyReturnTarget(ckrtGmscJtkb, svmstkrUmFlg)) {
            return new ClosingCancellationTargetJudgmentOutput(
                    false,
                    false,
                    null);
        }

        GasFeeContractTypeSortingContentAcquisitionOutput sortingResult =
                gasFeeContractTypeSortingContentAcquisition.execute(
                        grkKiyShbt,
                        closingCancellationTargetList,
                        resolveBaseDate());

        boolean hsnjKytgFlg =
                isClosingCancellationTarget(sortingResult.grkKsbHnsbList());
        if (!hsnjKytgFlg) {
            return new ClosingCancellationTargetJudgmentOutput(
                    false,
                    true,
                    false);
        }

        return new ClosingCancellationTargetJudgmentOutput(
                !Boolean.TRUE.equals(dgrpKasFlg),
                true,
                true);
    }

    /**
     * 早期返却対象かどうかを判定する.
     * `Boolean` フラグの null は条件不成立として扱う.
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param svmstkrUmFlg サービス契約・メーター設置場所関連有無フラグ
     * @return 早期返却対象の場合 true
     */
    private boolean isEarlyReturnTarget(
            final String ckrtGmscJtkb,
            final Boolean svmstkrUmFlg) {
        return dedicatedGasMeterStatus.equals(ckrtGmscJtkb)
                && Boolean.FALSE.equals(svmstkrUmFlg);
    }

    /**
     * SV-259 の戻り値先頭要素が対象コードかを判定する.
     *
     * @param grkKsbHnsbList 振分内容種別リスト
     * @return 先頭要素が閉栓時解約対象コードの場合 true
     */
    private boolean isClosingCancellationTarget(
            final List<GrkKsbHnsb> grkKsbHnsbList) {
        return !grkKsbHnsbList.isEmpty()
                && closingCancellationTargetDistributionContent.equals(
                        grkKsbHnsbList.get(0).grkKsbHnsb());
    }

    private LocalDate resolveBaseDate() {
        return requireSystemDatetime().toLocalDate();
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
}
