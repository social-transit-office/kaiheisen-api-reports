package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.InspectionExpiryExceededErrorJudgment;

/**
 * 概要: SV-337（検満超過エラー判定）の実装クラス。
 * 主な仕様:
 * - 検満超過確認フラグが false の場合のみ、検定有効年月と閉栓作業年月日の年月を比較する。
 * - 検定有効年月が閉栓作業年月日の年月より前の場合は閉栓共通エラー 020200051 を送出する。
 * - 条件に該当しない場合は何も返さずに正常終了する。
 * 既知の制約:
 * - public メソッド引数の null ガードは行わず、上位層バリデーション済み入力を前提とする。
 */
public final class InspectionExpiryExceededErrorJudgmentImpl
    implements InspectionExpiryExceededErrorJudgment {

    @Override
    public void execute(Boolean knmaCkaKknnFlg, YearMonth ktyYm, LocalDate hssgYmd) {
        if (!knmaCkaKknnFlg && isInspectionExpiryExceeded(ktyYm, hssgYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200051);
        }
    }

    /**
     * 概要: 検定有効年月が閉栓作業年月日の年月より前かを判定する。
     * 主な仕様: 閉栓作業年月日から年月のみを抽出し、YearMonth の compareTo で大小比較する。
     * 既知の制約: null は呼び出し元前提のため考慮しない。
     *
     * @param ktyYm 検定有効年月
     * @param hssgYmd 閉栓作業年月日
     * @return 検定有効年月が閉栓作業年月日の年月より前の場合 true
     */
    private boolean isInspectionExpiryExceeded(YearMonth ktyYm, LocalDate hssgYmd) {
        YearMonth closingWorkYearMonth = YearMonth.of(hssgYmd.getYear(), hssgYmd.getMonthValue());
        return ktyYm.compareTo(closingWorkYearMonth) < 0;
    }
}
