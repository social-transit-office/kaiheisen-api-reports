/** SV-103 閉栓取消実施可否判定: 小売情報サービス実装. */
public final class ClosingCancelFeasibilityJudgmentRetailInfoImpl
        implements ClosingCancelFeasibilityJudgmentRetailInfo {

        /** 将来サービス契約の返却上限件数. */

    private final SvckyaDataAccess svckyaDataAccess;

    private final String invalidServiceContractStatus;

    @Override
    public ClosingCancelFeasibilityJudgmentRetailInfoOutput execute(
            final LocalDate trkKnsYmd,
            final GmtsetdInf gmtsetdInf,
            final String duknIsrsCd) {
        Long wkServiceContractNo =
                resolveServiceContractNo(gmtsetdInf, trkKnsYmd);
        SvckyaRecord activeServiceContract =
                requireActiveServiceContract(wkServiceContractNo, trkKnsYmd);

        List<SvckyaInf> serviceContracts = buildServiceContracts(
                wkServiceContractNo,
                trkKnsYmd,
                activeServiceContract);

        RelatedMeterAggregation relatedMeters = aggregateRelatedMeters(
                wkServiceContractNo,
                trkKnsYmd,
                gmtsetdInf);
        KkytskrInf supplyStopInfo =
                resolveSupplyStopInfo(
                        gmtsetdInf.gasMtstNo(),
                        trkKnsYmd,
                        duknIsrsCd);

        return new ClosingCancelFeasibilityJudgmentRetailInfoOutput(
                serviceContracts,
                relatedMeters.svmstkrInfList(),
                relatedMeters.gmtsetdInfList(),
                supplyStopInfo);
    }

    private Long resolveServiceContractNo(
            final GmtsetdInf gmtsetdInf,
            final LocalDate trkKnsYmd) {
        Optional<SvmstkrRecord> directRecord =
                findServiceContractMeterRelation(
                        gmtsetdInf.gasMtstNo(),
                        trkKnsYmd);
        if (directRecord.isPresent()) {
            return directRecord.get().svkyNo();
        }

        if (!dedicatedGasMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            throw dataInconsistencyForSvmstkr(
                    gmtsetdInf.gasMtstNo(),
                    trkKnsYmd);
        }

        return findServiceContractMeterRelation(
                        gmtsetdInf.mtgMtstNo(),
                        trkKnsYmd)
                .map(SvmstkrRecord::svkyNo)
                .orElseThrow(
                        () ->
                                dataInconsistencyForSvmstkr(
                                        gmtsetdInf.mtgMtstNo(),
                                        trkKnsYmd));
    }

    private Optional<SvmstkrRecord> findServiceContractMeterRelation(
            final Long mtstNo,
            final LocalDate trkKnsYmd) {
        return svmstkrDataAccess
                .findByMtstNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                        mtstNo,
                        trkKnsYmd,
                        invalidChargeCalculationRelatedStatus);
    }

    private SvckyaRecord requireActiveServiceContract(
            final Long svkyNo,
            final LocalDate trkKnsYmd) {
        List<String> targetStatuses =
                List.of(underContractStatus, endOfContractStatus);
        List<SvckyaRecord> records =
                svckyaDataAccess.findAllBySvkyNoAtKnskKjnYmdAndSvkyJtkbList(
                        svkyNo,
                        trkKnsYmd,
                        targetStatuses);
        if (records.isEmpty()) {
            throw dataInconsistencyForSvckyaActive(
                    svkyNo,
                    trkKnsYmd,
                    targetStatuses);
        }
        return records.get(0);
    }

    private List<SvckyaInf> buildServiceContracts(
            final Long svkyNo,
            final LocalDate trkKnsYmd,
            final SvckyaRecord activeServiceContract) {
        List<SvckyaRecord> records = svckyaDataAccess
                .findAllBySvkyNoAndRkkiSttYmdAfterAndSvkyJtkbNotOrderByRkkiSttYmd(
                        svkyNo,
                        trkKnsYmd,
                        invalidServiceContractStatus);
        List<SvckyaInf> result = new ArrayList<>();
        result.add(mapSvckya(activeServiceContract));
        for (SvckyaRecord record : records) {
            if (!dutyFreeConsumptionTax.equals(activeServiceContract.sziKziKbn())
                    && dutyFreeConsumptionTax.equals(record.sziKziKbn())) {
                throw new BusinessRuleException(
                        ErrorCodeConstants.ERROR_CODE_020000060);
            }
            if (result.size() < FUTURE_SERVICE_CONTRACT_LIMIT) {
                result.add(mapSvckya(record));
            }
        }
        return List.copyOf(result);
    }

    private RelatedMeterAggregation aggregateRelatedMeters(
            final Long svkyNo,
            final LocalDate trkKnsYmd,
            final GmtsetdInf inputMeter) {
        List<SvmstkrRecord> records = svmstkrDataAccess
                .findAllBySvkyNoAndRkkiSttYmdAndRkkiEndYmdAndRkkiKnrnJtkb(
                        svkyNo,
                        trkKnsYmd,
                        invalidChargeCalculationRelatedStatus);
        List<SvmstkrInf> svmstkrInfList = new ArrayList<>();
        List<GmtsetdRelatedInf> gmtsetdInfList = new ArrayList<>();
        Long previousMtstNo = 0L;
        int processed = 0;
        for (SvmstkrRecord record : records) {
            if (processed >= RELATED_METER_PROCESS_LIMIT) {
                break;
            }
            processed++;
            if (!Objects.equals(record.mtstNo(), previousMtstNo)) {
                svmstkrInfList.add(mapSvmstkr(record));
                String seriesStatus = resolveSeriesStatus(
                        record.mtstNo(),
                        trkKnsYmd,
                        inputMeter);
                gmtsetdInfList.add(new GmtsetdRelatedInf(seriesStatus));
            }
            previousMtstNo = record.mtstNo();
        }
        return new RelatedMeterAggregation(
                List.copyOf(svmstkrInfList),
                List.copyOf(gmtsetdInfList));
    }

    private String resolveSeriesStatus(
            final Long mtstNo,
            final LocalDate trkKnsYmd,
            final GmtsetdInf inputMeter) {
        if (Objects.equals(mtstNo, inputMeter.gasMtstNo())) {
            return inputMeter.ckrtGmscJtkb();
        }
        GmtsetdRecord record =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(
                                mtstNo,
                                trkKnsYmd)
                        .orElseThrow(
                                () -> dataInconsistencyForGmtsetd(
                                        mtstNo,
                                        trkKnsYmd));
        return record.ckrtGmscJtkb();
    }

    private KkytskrInf resolveSupplyStopInfo(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd,
            final String duknIsrsCd) {
        Optional<KkytskrRecord> record =
                kkytskrDataAccess
                        .findByGasMtstNoAndHesnYmdAndKkysSisnJtkb(
                                gasMtstNo,
                                trkKnsYmd,
                                supplyStopPaymentPending);
        if (record.isEmpty()) {
            if (supplyOutageWorkDetailCode.equals(duknIsrsCd)) {
                throw new BusinessRuleException(
                        ErrorCodeConstants.ERROR_CODE_020000023);
            }
            return null;
        }
        return new KkytskrInf(record.get().gasMtstNo(), record.get().hesnYmd());
    }

    private SvckyaInf mapSvckya(final SvckyaRecord record) {
        return new SvckyaInf(
                record.svkyNo(),
                record.skno(),
                record.kyttSkno(),
                record.rkkiSttYmd(),
                record.rkkiEndYmd(),
                record.sqtiKiyNo(),
                record.syaKbn(),
                record.kkysSisnJtkb(),
                record.sziKziKbn());
    }

    private SvmstkrInf mapSvmstkr(final SvmstkrRecord record) {
        return new SvmstkrInf(
                record.mtstNo(),
                record.svkyNo(),
                record.skno(),
                record.rkkiEndYmd());
    }

    private DataInconsistencyException dataInconsistencyForSvmstkr(
            final Long mtstNo,
            final LocalDate knskKjnYmd) {
        DataAccessCallSpec callSpec =
                SvmstkrDataAccess.CallSpecs
                        .FIND_BY_MTST_NO_AND_RKKI_STT_YMD_AND_RKKI_END_YMD_AND_RKKI_KNRN_JTKB;
        FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
                .put("mtstNo", mtstNo)
                .put("knskKjnYmd", knskKjnYmd)
                .put(
                        "rkkiKnrnJtkb",
                        invalidChargeCalculationRelatedStatus);
        return dataInconsistency(
                svmstkrDataAccess.tableName(),
                callSpec,
                parameterBuilder);
    }

    private DataInconsistencyException dataInconsistencyForSvckyaActive(
            final Long svkyNo,
            final LocalDate knskKjnYmd,
            final List<String> svkyJtkbList) {
        DataAccessCallSpec callSpec =
                SvckyaDataAccess.CallSpecs
                        .FIND_ALL_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_LIST;
        FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
                .put("svkyNo", svkyNo)
                .put("knskKjnYmd", knskKjnYmd)
                .put(
                        "svkyJtkbList",
                        svkyJtkbList);
        return dataInconsistency(
                svckyaDataAccess.tableName(),
                callSpec,
                parameterBuilder);
    }

    private DataInconsistencyException dataInconsistencyForGmtsetd(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        DataAccessCallSpec callSpec =
                GmtsetdDataAccess.CallSpecs
                        .FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD;
        FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
                .put("gasMtstNo", gasMtstNo)
                .put("knskKjnYmd", knskKjnYmd);
        return dataInconsistency(
                gmtsetdDataAccess.tableName(),
                callSpec,
                parameterBuilder);
    }

    private static String requireConstant(
            final CiriusConstantService ciriusConstantService,
            final String type,
            final String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                type,
                code);
    }

    private DataInconsistencyException dataInconsistency(
            final String tableName,
            final DataAccessCallSpec callSpec,
            final FlatJsonBuilder parameterBuilder) {
        return DataInconsistencySupport.create(
                tableName,
                callSpec.identifier(),
                parameterBuilder);
    }
