package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svckya.SvckyaDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svckya.SvckyaRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingTargetGasMeterCompanyUseCategoryJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingTargetGasMeterCompanyUseCategoryJudgmentOutput;

/**
 * 概要: SV-271（閉栓対象ガスメーターの社用区分判定）の実装。
 * 主な仕様:
 * - RequestContext の systemDatetime を検索基準日に変換し、DA-038-R006 相当の契約一覧検索を行う。
 * - 契約 0 件時は業務エラー 020000027、社用区分が社用時は業務エラー 020000028 を送出する。
 * - 正常時は取得した契約種別を出力 DTO へ格納して返す。
 * 既知の制約:
 * - 契約が存在する場合は取得一覧の先頭レコードを判定対象として扱う。
 */
public final class ClosingTargetGasMeterCompanyUseCategoryJudgmentImpl
        implements ClosingTargetGasMeterCompanyUseCategoryJudgment {

    /** サービス契約情報を取得するデータアクセス。 */
    private final SvckyaDataAccess svckyaDataAccess;

    /** systemDatetime を取得するためのリクエストコンテキスト。 */
    private final RequestContext requestContext;

    /** 社用区分が社用であることを表す CIRIUS 定数値。 */
    private final String companyUseDivision;

    /** サービス契約状態区分が無効であることを表す CIRIUS 定数値。 */
    private final String invalidServiceContractStatus;

    /**
     * 概要: 依存と判定用定数を受け取って初期化する。
     * 主な仕様:
     * - サービス契約 DA と RequestContext を必須依存として保持する。
     * - 社用区分判定に使う CIRIUS 定数を初期化時に解決する。
     * 既知の制約:
     * - 依存または必須定数が不足する場合はコンストラクタで fail-fast する。
     *
     * @param svckyaDataAccess サービス契約情報取得 DA
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ClosingTargetGasMeterCompanyUseCategoryJudgmentImpl(
            SvckyaDataAccess svckyaDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccess, "svckyaDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.companyUseDivision = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Corporate.TYPE,
                ConstantLookupKeys.Cirius.Corporate.FOR_COMPANY_USE);
        this.invalidServiceContractStatus = RequiredConstantResolver.requireCiriusConstant(
            resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
            ConstantLookupKeys.Cirius.ServiceContractStatus.INVALID);
    }

    @Override
    public ClosingTargetGasMeterCompanyUseCategoryJudgmentOutput execute(Long svkyNo) {
        List<SvckyaRecord> contracts = svckyaDataAccess
            .findAllBySvkyNoAtKnskKjnYmdAndSvkyJtkbNot(
                svkyNo,
                getSystemDatetime().toLocalDate(),
                invalidServiceContractStatus);

        if (contracts.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000027);
        }

        SvckyaRecord contract = contracts.getFirst();

        if (companyUseDivision.equals(contract.syaKbn())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000028);
        }

        return new ClosingTargetGasMeterCompanyUseCategoryJudgmentOutput(contract.kiyShbt());
    }

    /**
     * 概要: RequestContext からシステム日時を取得する。
     * 主な仕様:
     * - 設計上の systemDatetime 入力は RequestContext から補う。
     * - 実装規約に合わせて未設定時は IllegalStateException で fail-fast する。
     * 既知の制約:
     * - RequestContext に systemDatetime が設定されていない場合は実行を継続しない。
     *
     * @return リクエストコンテキスト上のシステム日時
     */
    private LocalDateTime getSystemDatetime() {
        return requestContext
                .getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
}
