package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.math.BigDecimal;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.RegressionReverseRegressionMarkCheck;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.RegressionReverseRegressionMarkCheckOutput;

/** 回帰逆回帰マークチェックサービス実装。 */
public final class RegressionReverseRegressionMarkCheckImpl
        implements RegressionReverseRegressionMarkCheck {

    private final String regressionSegment;

    private final String reverseRegressionSegment;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public RegressionReverseRegressionMarkCheckImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolved =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.regressionSegment =
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.REGRESSION);
        this.reverseRegressionSegment =
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment
                                .REVERSE_REGRESSION);
    }

    /**
     * 回帰逆回帰マーク判定を実行する。
     *
     * @param kknnFlg 確認フラグ
     * @param sjskKbn 指示数回帰区分
     * @param konSyry 今回使用量
     * @param sjskHticSu 指示数回帰判定値数
     * @return 判定結果
     */
    @Override
    public RegressionReverseRegressionMarkCheckOutput execute(
            Boolean kknnFlg,
            String sjskKbn,
            BigDecimal konSyry,
            Long sjskHticSu) {
        if (!Boolean.TRUE.equals(kknnFlg)) {
            return new RegressionReverseRegressionMarkCheckOutput(false);
        }

        BigDecimal upperBound = resolveUpperBound(sjskHticSu);
        BigDecimal lowerBound = upperBound.negate();
        boolean kaikiHtkFlg;

        if (konSyry.compareTo(upperBound) < 0) {
            kaikiHtkFlg = evaluateLessThanUpperBound(sjskKbn, konSyry, lowerBound);
        } else {
            kaikiHtkFlg = evaluateUpperBoundOrMore(sjskKbn);
        }

        return new RegressionReverseRegressionMarkCheckOutput(kaikiHtkFlg);
    }

    /**
     * 概要: 指示数回帰判定値数から上限値を算出する。
     * 主な仕様: `Long` を 2 で整数除算し、その結果を BigDecimal へ変換する。
     * 既知の制約: `sjskHticSu` が null の場合は NullPointerException となる。
     *
     * @param sjskHticSu 指示数回帰判定値数
     * @return 上限値
     */
    private BigDecimal resolveUpperBound(Long sjskHticSu) {
        return BigDecimal.valueOf(sjskHticSu / 2L);
    }

    /**
     * 概要: 今回使用量が上限値未満の経路を判定する。
     * 主な仕様: 下限値より大きい場合は中間帯判定、それ以外は下限側境界判定を行う。
     * 既知の制約: `konSyry` が null の場合は NullPointerException となる。
     *
     * @param sjskKbn 指示数回帰区分
     * @param konSyry 今回使用量
     * @param lowerBound 下限値
     * @return 判定結果
     */
    private boolean evaluateLessThanUpperBound(
            String sjskKbn,
            BigDecimal konSyry,
            BigDecimal lowerBound) {
        if (konSyry.compareTo(lowerBound) > 0) {
            return isRegressionOrReverseRegression(sjskKbn);
        }
        return !isRegression(sjskKbn);
    }

    /**
     * 概要: 今回使用量が上限値以上の経路を判定する。
     * 主な仕様: 逆回帰区分のみ false とし、それ以外は true とする。
     * 既知の制約: なし。
     *
     * @param sjskKbn 指示数回帰区分
     * @return 判定結果
     */
    private boolean evaluateUpperBoundOrMore(String sjskKbn) {
        return !isReverseRegression(sjskKbn);
    }

    /**
     * 概要: 回帰または逆回帰区分かを判定する。
     * 主な仕様: 回帰または逆回帰コンスタントと一致した場合に true を返す。
     * 既知の制約: なし。
     *
     * @param sjskKbn 指示数回帰区分
     * @return 判定結果
     */
    private boolean isRegressionOrReverseRegression(String sjskKbn) {
        return isRegression(sjskKbn) || isReverseRegression(sjskKbn);
    }

    /**
     * 概要: 回帰区分かを判定する。
     * 主な仕様: コンストラクタで解決した回帰区分定数と比較する。
     * 既知の制約: なし。
     *
     * @param sjskKbn 指示数回帰区分
     * @return 判定結果
     */
    private boolean isRegression(String sjskKbn) {
        return regressionSegment.equals(sjskKbn);
    }

    /**
     * 概要: 逆回帰区分かを判定する。
     * 主な仕様: コンストラクタで解決した逆回帰区分定数と比較する。
     * 既知の制約: なし。
     *
     * @param sjskKbn 指示数回帰区分
     * @return 判定結果
     */
    private boolean isReverseRegression(String sjskKbn) {
        return reverseRegressionSegment.equals(sjskKbn);
    }
}
