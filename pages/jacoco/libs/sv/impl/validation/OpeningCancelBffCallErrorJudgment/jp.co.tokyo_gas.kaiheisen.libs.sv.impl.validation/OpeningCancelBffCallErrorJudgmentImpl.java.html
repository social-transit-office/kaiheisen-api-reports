package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningCancelBffCallErrorJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningCancelBffCallErrorJudgmentOutput;

/**
 * 概要: SV-045（開栓取消用BFF呼出しエラー判定）のサービス実装。
 * 主な仕様:
 * - 小売・導管の異動・作業履歴を取得し、取得済みの業務コードで BFF 呼出しエラー判定を行う。
 * - 小売履歴 0 件、導管履歴 0 件、不一致判定は業務エラー `020000018` とする。
 * - 小売業務コードが `CCBG47` の場合は業務エラー `020000019` とする。
 */
public final class OpeningCancelBffCallErrorJudgmentImpl
        implements OpeningCancelBffCallErrorJudgment {

    private final IsrkkDataAccess isrkkDataAccess;

    private final IsrkdDataAccess isrkdDataAccess;

    private final RequestContext requestContext;

    private final List<String> retailHistoryDetailCodes;

    private final List<String> conduitHistoryDetailCodes;

    private final String retailCodeOpeningRegistration;

    /**
     * 必須依存を初期化する。
     *
     * @param isrkkDataAccess 小売履歴データアクセス
     * @param isrkdDataAccess 導管履歴データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public OpeningCancelBffCallErrorJudgmentImpl(
            IsrkkDataAccess isrkkDataAccess,
            IsrkdDataAccess isrkdDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.isrkkDataAccess = Objects.requireNonNull(isrkkDataAccess, "isrkkDataAccess");
        this.isrkdDataAccess = Objects.requireNonNull(isrkdDataAccess, "isrkdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.retailHistoryDetailCodes = List.of(
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.NEWLY_OPENED_TAP_CCBG4701),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.ALTERNATIVE_OPENING_CCBG4702),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.CONTINUED_USE_CCBG4704),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.SUPPLY_RESUMED_CCBG4705),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.NEWLY_OPENED_TAP_CCBG5001),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.ALTERNATIVE_OPENING_CCBG5002),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.CONTINUED_USE_CCBG5004),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.SUPPLY_RESUMED_CCBG5005),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.CUSTOMER_OPENING_CCBG4801),
                requireRetailHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.START_SUPPLYING_TO_OTHER_PARTIES));
        this.conduitHistoryDetailCodes = List.of(
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.NEWLY_OPENED_TAP_CCJG0101),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.ALTERNATIVE_OPENING_CCJG0102),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.CONTINUED_USE_CCJG0104),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.SUPPLY_RESUMED_CCJG0105),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.NEWLY_OPENED_TAP_CCJG0301),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.ALTERNATIVE_OPENING_CCJG0302),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.CONTINUED_USE_CCJG0304),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.SUPPLY_RESUMED_CCJG0305),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.CUSTOMER_OPENING_CCJG0501),
                requireConduitHistoryDetailCode(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.START_SUPPLYING_TO_OTHER_PARTIES_CCJG0502));
        this.retailCodeOpeningRegistration = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.OPENING_REGISTRATION);
    }

    /**
     * 開栓取消用 BFF 呼出し可否を判定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkIsrrGymCd 取消対象異動・作業履歴業務コード
     * @return 判定結果
     */
    @Override
    public OpeningCancelBffCallErrorJudgmentOutput execute(
            Long gasMtstNo,
            String trkIsrrGymCd) {
        LocalDate targetDate = requireSystemDatetime().toLocalDate();

        IsrkkRecord retailRecord = findRetailHistory(gasMtstNo, targetDate);
        IsrkdRecord conduitRecord = findConduitHistory(gasMtstNo, targetDate);

        validateRetailBusinessCode(trkIsrrGymCd, retailRecord.isrrGymCd());

        return new OpeningCancelBffCallErrorJudgmentOutput(
                conduitRecord.duknIsrgCd(),
                conduitRecord.duknIsrsCd(),
                retailRecord.isrrGymCd(),
                retailRecord.isrrGmssCd());
    }

    private IsrkkRecord findRetailHistory(Long gasMtstNo, LocalDate targetDate) {
        List<IsrkkRecord> retailRecords = Objects.requireNonNull(
                isrkkDataAccess.findAllByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
                        gasMtstNo,
                        targetDate,
                        retailHistoryDetailCodes),
                "isrkkDataAccess.findAllByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd "
                        + "(DA-102-R001) returned null");
        if (retailRecords.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000018);
        }
        return retailRecords.get(0);
    }

    private IsrkdRecord findConduitHistory(Long gasMtstNo, LocalDate targetDate) {
        return isrkdDataAccess.findByArtyCstNoAndIsrrTriYmdAndDuknIsrsCd(
                gasMtstNo,
                targetDate,
                conduitHistoryDetailCodes).orElseThrow(
                        () -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000018));
    }

    private void validateRetailBusinessCode(String trkIsrrGymCd, String retailBusinessCode) {
        if (retailCodeOpeningRegistration.equals(retailBusinessCode)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000019);
        }
        if (!Objects.equals(trkIsrrGymCd, retailBusinessCode)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000018);
        }
    }

    private static String requireRetailHistoryDetailCode(
            CiriusConstantService ciriusConstantService,
            String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.TYPE,
                code);
    }

    private static String requireConduitHistoryDetailCode(
            CiriusConstantService ciriusConstantService,
            String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.TYPE,
                code);
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
}
