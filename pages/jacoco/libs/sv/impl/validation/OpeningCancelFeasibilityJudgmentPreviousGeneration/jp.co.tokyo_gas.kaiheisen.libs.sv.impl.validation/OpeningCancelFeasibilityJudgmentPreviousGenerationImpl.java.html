package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningCancelFeasibilityJudgmentPreviousGeneration;

/**
 * 概要: SV-051（開栓取消実施可否判定：前世代チェック）の実装クラス。
 *
 * 主な仕様:
 * - 判定対象の前世代レコードを DA-033-R001 で取得する。
 * - 取得レコードが存在し、かつガスメーター開閉栓状態区分が開栓中の場合のみ業務エラーを送出する。
 * - 取得結果が存在しない場合、または開栓中以外の場合は何も返さず正常終了する。
 *
 * 既知の制約:
 * - public 引数の null / blank 補正や追加バリデーションは行わない。
 * - systemDatetime、correlationId、hukkSysShbt は本サービスでは参照しない。
 */
public final class OpeningCancelFeasibilityJudgmentPreviousGenerationImpl
        implements OpeningCancelFeasibilityJudgmentPreviousGeneration {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final String openingStatus;

    /**
     * 概要: 判定に必要な依存コンポーネントと必須定数を初期化する。
     *
     * 主な仕様:
     * - ガスメーター設置場所＜導管＞DA と CIRIUS 定数サービスを必須依存として受け取る。
     * - 開栓中判定に必要な `02100-1` をコンストラクタ初期化時に解決し、未登録なら fail-fast する。
     *
     * 既知の制約:
     * - 必須依存が null の場合は {@link NullPointerException} を送出する。
     *
     * @param gmtsetdDataAccessParam ガスメーター設置場所＜導管＞データアクセス
     * @param ciriusConstantServiceParam CIRIUS 定数サービス
     */
    public OpeningCancelFeasibilityJudgmentPreviousGenerationImpl(
            final GmtsetdDataAccess gmtsetdDataAccessParam,
            final CiriusConstantService ciriusConstantServiceParam) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccessParam, "gmtsetdDataAccess");
        CiriusConstantService ciriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceParam, "ciriusConstantService");
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 概要: 前世代のガスメーター開閉栓状態を判定し、取消不可の場合に業務エラーを送出する。
     *
     * 主な仕様:
     * - DA-033-R001 を 1 回だけ呼び出す。
     * - 取得レコードが存在し、かつ `gmtKhsJtkb` が開栓中定数と一致する場合のみ
     *   `BusinessRuleException(ERROR_CODE_020000015)` を送出する。
     * - 上記以外の場合は何も返さず正常終了する。
     *
     * 既知の制約:
     * - DA 戻り値が `Optional.empty()` の場合はエラー化しない。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     */
    @Override
    public void execute(final Long gasMtstNo, final LocalDate trkKnsYmd) {
        Optional<GmtsetdRecord> previousGeneration =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmd(gasMtstNo, trkKnsYmd);

        if (previousGeneration
                .map(GmtsetdRecord::gmtKhsJtkb)
                .filter(openingStatus::equals)
                .isPresent()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000015);
        }
    }
}
