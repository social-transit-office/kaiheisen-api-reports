package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.math.BigDecimal;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UnitNumberExceededErrorJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UnitNumberExceededErrorJudgmentOutput;

/**
 * 概要: SV-265（号数超過エラー判定）の実装クラス。
 * 主な仕様:
 * - ガスメーター号数の帯ごとに使用量閾値を比較し、超過時に変動使用量エラー有無フラグへ true を設定する。
 * - どの帯にも該当しない場合、または閾値以下の場合は false を返却する。
 * 既知の制約:
 * - public メソッド引数の null ガードは行わず、上位層バリデーション済み入力を前提とする。
 * - 号数と使用量は BigDecimal の compareTo による大小比較で扱う。
 */
public final class UnitNumberExceededErrorJudgmentImpl implements UnitNumberExceededErrorJudgment {

    private static final BigDecimal METER_NUMBER_1 = BigDecimal.valueOf(1L);

    private static final BigDecimal METER_NUMBER_2 = BigDecimal.valueOf(2L);

    private static final BigDecimal METER_NUMBER_3 = BigDecimal.valueOf(3L);

    private static final BigDecimal METER_NUMBER_5 = BigDecimal.valueOf(5L);

    private static final BigDecimal METER_NUMBER_7 = BigDecimal.valueOf(7L);

    private static final BigDecimal METER_NUMBER_16 = BigDecimal.valueOf(16L);

    private static final BigDecimal METER_NUMBER_50 = BigDecimal.valueOf(50L);

    private static final BigDecimal METER_NUMBER_200 = BigDecimal.valueOf(200L);

    private static final BigDecimal USAGE_THRESHOLD_200 = BigDecimal.valueOf(200L);

    private static final BigDecimal USAGE_THRESHOLD_400 = BigDecimal.valueOf(400L);

    private static final BigDecimal USAGE_THRESHOLD_600 = BigDecimal.valueOf(600L);

    private static final BigDecimal USAGE_THRESHOLD_1000 = BigDecimal.valueOf(1000L);

    private static final BigDecimal USAGE_THRESHOLD_1400 = BigDecimal.valueOf(1400L);

    private static final BigDecimal USAGE_THRESHOLD_3500 = BigDecimal.valueOf(3500L);

    private static final BigDecimal USAGE_THRESHOLD_10000 = BigDecimal.valueOf(10000L);

    private static final BigDecimal USAGE_THRESHOLD_35000 = BigDecimal.valueOf(35000L);

    @Override
    public UnitNumberExceededErrorJudgmentOutput execute(BigDecimal syry, BigDecimal gmtGusu) {
        return new UnitNumberExceededErrorJudgmentOutput(isExceeded(syry, gmtGusu));
    }

    /**
     * 概要: 入力値が閾値超過エラーに該当するかを判定する。
     * 主な仕様:
     * - 号数帯に対応する閾値を上からではなく最小帯から順に評価する。
     * - いずれの帯にも該当しない号数は false とする。
     *
     * @param syry 使用量
     * @param gmtGusu ガスメーター号数
     * @return 閾値超過時 true
     */
    private boolean isExceeded(BigDecimal syry, BigDecimal gmtGusu) {
        if (isWithin(gmtGusu, METER_NUMBER_1)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_200);
        }
        if (isWithin(gmtGusu, METER_NUMBER_2)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_400);
        }
        if (isWithin(gmtGusu, METER_NUMBER_3)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_600);
        }
        if (isWithin(gmtGusu, METER_NUMBER_5)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_1000);
        }
        if (isWithin(gmtGusu, METER_NUMBER_7)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_1400);
        }
        if (isWithin(gmtGusu, METER_NUMBER_16)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_3500);
        }
        if (isWithin(gmtGusu, METER_NUMBER_50)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_10000);
        }
        if (isWithin(gmtGusu, METER_NUMBER_200)) {
            return isGreaterThan(syry, USAGE_THRESHOLD_35000);
        }
        return false;
    }

    /**
     * 概要: ガスメーター号数が指定上限以下かを判定する。
     * 主な仕様: BigDecimal の compareTo で大小比較する。
     * 既知の制約: null は呼び出し元前提のため考慮しない。
     *
     * @param gmtGusu ガスメーター号数
     * @param upperBound 上限号数
     * @return 指定上限以下の場合 true
     */
    private boolean isWithin(BigDecimal gmtGusu, BigDecimal upperBound) {
        return gmtGusu.compareTo(upperBound) <= 0;
    }

    /**
     * 概要: 使用量が閾値より大きいかを判定する。
     * 主な仕様: 閾値と等しい場合は false とする。
     * 既知の制約: null は呼び出し元前提のため考慮しない。
     *
     * @param syry 使用量
     * @param threshold 閾値
     * @return 閾値超過の場合 true
     */
    private boolean isGreaterThan(BigDecimal syry, BigDecimal threshold) {
        return syry.compareTo(threshold) > 0;
    }
}
