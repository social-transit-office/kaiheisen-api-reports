package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys.Kaiheisen.OpeningClosingInProgressErrorJudgmentType;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningClosingInProgressErrorJudgment;

/**
 * 概要:
 * 開栓中・閉栓中エラー判定サービスの実装クラス。
 * ガスメーター開閉栓状態区分と開栓中・閉栓中エラー判定処理区分をもとに、
 * 設計書で定義された条件に一致した場合に業務例外を送出する。
 *
 * <p>主な仕様:
 * <ul>
 *   <li>コンストラクタ初期化時に、判定で使用する必須定数を各コンスタントサービスから解決する。</li>
 *   <li>開栓中エラー判定処理区分かつガスメーター開閉栓状態区分が開栓中の場合は、
 *       {@link KaisenRuleException} を送出する。</li>
 *   <li>開栓中エラー判定処理区分以外かつガスメーター開閉栓状態区分が閉栓中の場合は、
 *       {@link HeisenRuleException} を送出する。</li>
 *   <li>上記以外の場合は何もせず終了する。</li>
 * </ul>
 *
 * <p>既知の制約:
 * <ul>
 *   <li>判定に必要な定数がコンスタントサービスから取得できない場合は初期化できない。</li>
 *   <li>判定条件および送出するエラーコードは、実装内に定義された設計書準拠の分岐に依存する。</li>
 * </ul>
 */
public final class OpeningClosingInProgressErrorJudgmentImpl
        implements OpeningClosingInProgressErrorJudgment {

    private final String openingInProgressErrorJudgment;

    private final String openingStatus;

    private final String closedStatus;

    /**
     * 概要:
     * 判定で使用する必須定数を解決し、本サービスのインスタンスを初期化する。
     *
     * 主な仕様:
     * 開栓中・閉栓中エラー判定に必要な判定種別と、ガスメーター開閉栓状態の必須定数を
     * コンストラクタ実行時に解決して保持する。
     * 引数のコンスタント取得サービスは null を許容せず、null の場合は fail-fast で
     * 例外を送出する。
     *
     * 既知の制約:
     * 必須定数が取得できない場合は遅延評価せず、コンストラクタ実行時点で fail-fast として
     * 例外を送出する。
     * また、引数に null が渡された場合は {@link NullPointerException} を送出する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public OpeningClosingInProgressErrorJudgmentImpl(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");

        this.openingInProgressErrorJudgment =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        OpeningClosingInProgressErrorJudgmentType.TYPE,
                        OpeningClosingInProgressErrorJudgmentType.OPENING);
        RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                OpeningClosingInProgressErrorJudgmentType.TYPE,
                OpeningClosingInProgressErrorJudgmentType.CLOSING);
        this.openingStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.closedStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
    }

    /**
     * 設計書の分岐条件に従って業務例外要否を判定する。
     *
     * <p>開栓中エラー判定処理区分の場合は開栓中状態を判定し、それ以外の処理区分の場合は
     * 閉栓中状態の判定へ進む。
     *
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param kascHsncErrHntShrKbn 開栓中・閉栓中エラー判定処理区分
     */
    @Override
    public void execute(String gmtKhsJtkb, String kascHsncErrHntShrKbn) {
        if (openingInProgressErrorJudgment.equals(kascHsncErrHntShrKbn)) {
            if (openingStatus.equals(gmtKhsJtkb)) {
                throw new KaisenRuleException(ErrorCodeConstants.ERROR_CODE_020100016);
            }
        } else if (closedStatus.equals(gmtKhsJtkb)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200014);
        }
    }
}
