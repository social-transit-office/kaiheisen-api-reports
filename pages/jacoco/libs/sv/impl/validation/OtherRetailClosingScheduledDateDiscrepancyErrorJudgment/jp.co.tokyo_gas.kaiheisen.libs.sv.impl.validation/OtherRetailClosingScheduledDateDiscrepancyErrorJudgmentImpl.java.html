package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OtherRetailClosingScheduledDateDiscrepancyErrorJudgment;

/**
 * 他小売閉栓予定日相違エラー判定サービス実装。
 */
public final class OtherRetailClosingScheduledDateDiscrepancyErrorJudgmentImpl
        implements OtherRetailClosingScheduledDateDiscrepancyErrorJudgment {

    private final String relocationClosureReason;

    /**
     * 判定に必要な定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public OtherRetailClosingScheduledDateDiscrepancyErrorJudgmentImpl(
            CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedCiriusConstantService =
            Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.relocationClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE);
    }

    /**
     * 他小売閉栓予定日の相違有無を判定する。
     *
     * @param tkurYteKknFlg 他小売閉栓予定日確認フラグ
     * @param hesnJysb 閉栓事由種別
     * @param tksuEndYmd 託送終了年月日
     * @param hssgYmd 閉栓作業年月日
     */
    @Override
    public void execute(
            Boolean tkurYteKknFlg,
            String hesnJysb,
            LocalDate tksuEndYmd,
            LocalDate hssgYmd) {
        if (Boolean.FALSE.equals(tkurYteKknFlg)
                && isRelocationClosureReason(hesnJysb)
                && hasDateDiscrepancy(tksuEndYmd, hssgYmd)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200049);
        }
    }

    /**
     * 閉栓事由種別が移転閉栓かを判定する。
     *
     * @param hesnJysb 閉栓事由種別
     * @return 移転閉栓の場合は true
     */
    private boolean isRelocationClosureReason(String hesnJysb) {
        return relocationClosureReason.equals(hesnJysb);
    }

    /**
     * 託送終了年月日と閉栓作業年月日に相違があるかを判定する。
     *
     * @param tksuEndYmd 託送終了年月日
     * @param hssgYmd 閉栓作業年月日
     * @return 日付が不一致の場合は true
     */
    private boolean hasDateDiscrepancy(LocalDate tksuEndYmd, LocalDate hssgYmd) {
        return !Objects.equals(tksuEndYmd, hssgYmd);
    }
}
