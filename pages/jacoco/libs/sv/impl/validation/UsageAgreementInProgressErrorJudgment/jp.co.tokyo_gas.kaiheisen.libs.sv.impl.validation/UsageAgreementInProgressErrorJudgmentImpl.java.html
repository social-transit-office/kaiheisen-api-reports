package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleShiyouryoKyoteichuException;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UsageAgreementInProgressErrorJudgment;

/**
 * 概要: SV-144（使用量協定中エラー判定）の実装。
 * 主な仕様:
 * - 使用量協定関連の各フラグを優先順位に従って評価する。
 * - 条件に該当する場合は対象の設置場所番号を埋込み文字列に含む使用量協定中エラーを送出する。
 * 既知の制約:
 * - public メソッド引数の null バリデーションは上位層責務とし、本クラスでは null ガードを行わない。
 */
public final class UsageAgreementInProgressErrorJudgmentImpl
        implements UsageAgreementInProgressErrorJudgment {

    /**
     * 概要: 使用量協定中エラー判定を実行する。
     * 主な仕様:
        * - 最初に一致した条件に対応する設置場所番号で使用量協定中エラーを送出する。
     * - いずれの条件にも一致しない場合は正常終了する。
     * 既知の制約:
        * - Boolean / Integer の null は自動アンボクシング時に NullPointerException となる。
          * - Long の設置場所番号が null のまま使用量協定中エラー生成に使用された場合は、
          *   共通例外契約に従い埋込み文字列へ空文字が設定され、errorGasMtstNo は null となる。
     *
     * @param syryKytiTyuFlg 使用量協定有無フラグ
     * @param mtgSyryKytiTyuFlg 直列元使用量協定中フラグ
     * @param snySyryKytiTyuFlg 直列専用使用量協定中フラグ
     * @param suguGmtKnsu 総合扱いガスメーター件数
     * @param dgrpSyryKytiTyuFlg 同一グループ内使用量協定中フラグ
     * @param gasMtstNo ガスメーター設置場所番号
     * @param mtgSyryKytiTyuMtstno 直列元使用量協定中設置場所番号
     * @param snySyryKytiTyuMtstno 直列専用使用量協定中設置場所番号
     * @param dgrpSyryKytiTyuMtstno 同一グループ内使用量協定中設置場所番号
     */
    @Override
    public void execute(
            Boolean syryKytiTyuFlg,
            Boolean mtgSyryKytiTyuFlg,
            Boolean snySyryKytiTyuFlg,
            Integer suguGmtKnsu,
            Boolean dgrpSyryKytiTyuFlg,
            Long gasMtstNo,
            Long mtgSyryKytiTyuMtstno,
            Long snySyryKytiTyuMtstno,
            Long dgrpSyryKytiTyuMtstno) {
        if (syryKytiTyuFlg) {
            throwUsageAgreementInProgress(gasMtstNo);
        }
        if (mtgSyryKytiTyuFlg) {
            throwUsageAgreementInProgress(mtgSyryKytiTyuMtstno);
        }
        if (snySyryKytiTyuFlg) {
            throwUsageAgreementInProgress(snySyryKytiTyuMtstno);
        }
        if (hasGroupAgreementInProgress(suguGmtKnsu, dgrpSyryKytiTyuFlg)) {
            throwUsageAgreementInProgress(dgrpSyryKytiTyuMtstno);
        }
    }

    /**
     * 概要: 総合扱いメーターの同一グループ判定条件を評価する。
     * 主な仕様:
     * - 総合扱いガスメーター件数が 1 以上、かつ同一グループ内使用量協定中フラグが true の場合に true を返す。
     * 既知の制約:
     * - `suguGmtKnsu` が null の場合は比較時に NullPointerException となる。
     * - `dgrpSyryKytiTyuFlg` が null の場合は、`suguGmtKnsu >= 1` が true のときに限り
     *   自動アンボクシングで NullPointerException となる。
     *
     * @param suguGmtKnsu 総合扱いガスメーター件数
     * @param dgrpSyryKytiTyuFlg 同一グループ内使用量協定中フラグ
     * @return 同一グループ内使用量協定中エラーを送出すべき場合は true
     */
    private boolean hasGroupAgreementInProgress(
            Integer suguGmtKnsu,
            Boolean dgrpSyryKytiTyuFlg) {
        return suguGmtKnsu >= 1 && dgrpSyryKytiTyuFlg;
    }

    /**
     * 概要: 使用量協定中エラーを送出する。
     * 主な仕様:
     * - エラーコード 020200026 を使用する。
     * - エラーメッセージ埋込み文字列には対象の設置場所番号を 1 件設定する。
     *
     * @param gasMtstNo エラー対象のガスメーター設置場所番号
     */
    private void throwUsageAgreementInProgress(Long gasMtstNo) {
        throw new HeisenRuleShiyouryoKyoteichuException(gasMtstNo == null ? null : String.valueOf(gasMtstNo));
    }
}
