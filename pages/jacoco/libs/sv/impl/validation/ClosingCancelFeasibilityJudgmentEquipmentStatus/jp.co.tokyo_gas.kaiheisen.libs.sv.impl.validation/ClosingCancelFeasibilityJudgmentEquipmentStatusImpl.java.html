package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancelFeasibilityJudgmentEquipmentStatus;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancelFeasibilityJudgmentEquipmentStatusOutput;

/** SV-101 閉栓取消実施可否判定: 供内管・ガスメーター設備状態区分サービス実装. */
public final class ClosingCancelFeasibilityJudgmentEquipmentStatusImpl
        implements ClosingCancelFeasibilityJudgmentEquipmentStatus {

    /** ガスメーター設置場所＜導管＞ DA. */
    private final GmtsetdDataAccess gmtsetdDataAccess;

    /** メーター（保有）状態の供内管・ガスメーター設備状態区分. */
    private final String meterPossessionStatus;

    /**
     * 必須依存を受け取って初期化する.
     *
     * @param gmtsetdDataAccessParam ガスメーター設置場所＜導管＞ DA
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ClosingCancelFeasibilityJudgmentEquipmentStatusImpl(
            final GmtsetdDataAccess gmtsetdDataAccessParam,
            final CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(
                gmtsetdDataAccessParam,
                "gmtsetdDataAccess");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(
                        ciriusConstantService,
                        "ciriusConstantService");
        this.meterPossessionStatus = requireConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius
                        .ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus
                        .METER_POSSESSION_STATUS);
    }

    /**
     * 供内管・ガスメーター設備状態区分を判定し、判定対象レコードを返す.
     *
     * <p>主な仕様: DA-033-R006 で対象レコードを 1 件取得し、0 件の場合は
     * {@link DataInconsistencyException} を送出する。取得した
     * 供内管・ガスメーター設備状態区分が 05800 1:メーター（保有）状態と一致しない場合は
     * {@link BusinessRuleException} として業務エラー 020000052 を送出する。
     * 一致する場合は取得したレコードの必要項目を出力 DTO へ移送して返却する。
     *
     * <p>既知の制約: public 引数の null/形式妥当性は上位レイヤで検証済みを前提とし、
     * 本メソッドでは追加バリデーションを行わない。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消検針日
     * @return 判定対象レコードから生成した出力 DTO
     */
    @Override
    public ClosingCancelFeasibilityJudgmentEquipmentStatusOutput execute(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd) {
        GmtsetdRecord record = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(
                        gasMtstNo,
                        trkKnsYmd)
                .orElseThrow(
                        () -> dataInconsistencyForGmtsetd(
                                gasMtstNo,
                                trkKnsYmd));

        if (!meterPossessionStatus.equals(record.kngmStbJtkb())) {
            throw new BusinessRuleException(
                    ErrorCodeConstants.ERROR_CODE_020000052);
        }

        return toOutput(record);
    }

    private ClosingCancelFeasibilityJudgmentEquipmentStatusOutput toOutput(
            final GmtsetdRecord record) {
        return new ClosingCancelFeasibilityJudgmentEquipmentStatusOutput(
                record.gasMtstNo(),
                record.szcGkkCd(),
                record.szcMchCd(),
                record.szcCmazCd(),
                record.kasYmd(),
                record.hesnYmd(),
                record.hesnJysb(),
                record.ckrtGmscJtkb(),
                record.mtgMtstNo());
    }

    private DataInconsistencyException dataInconsistencyForGmtsetd(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        DataAccessCallSpec callSpec =
                GmtsetdDataAccess.CallSpecs
                        .FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD;
        FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
                .put("gasMtstNo", gasMtstNo)
                .put("knskKjnYmd", knskKjnYmd);
        return DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                callSpec.identifier(),
                parameterBuilder);
    }

    private static String requireConstant(
            final CiriusConstantService ciriusConstantService,
            final String type,
            final String code) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                type,
                code);
    }
}
