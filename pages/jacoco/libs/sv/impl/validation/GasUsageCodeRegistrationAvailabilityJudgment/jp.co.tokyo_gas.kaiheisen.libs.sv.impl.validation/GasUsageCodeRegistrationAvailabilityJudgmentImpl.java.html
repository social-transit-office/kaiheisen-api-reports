package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstkism.CstkismDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.GasUsageCodeRegistrationAvailabilityJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.GasUsageCodeRegistrationAvailabilityJudgmentOutput;

/**
 * 概要: SV-274（ガス用途コード登録可否判定）の実装クラス。
 * 主な仕様:
 * - ガス用途コードが `null` または空文字の場合は、業務エラー `020000091` をスローする。
 * - SV-089 の例外時は、業務エラー `020000092` をスローする。
 * - SV-089 で取得したコンスタント名ミクスを `cntnMix` として返却する。
 * - SV-089 が正常終了し、かつ DA-162-R001 が 0 件を返した場合のみ登録可を返却する。
 * - DA-162-R001 の例外は呼出元へ伝播せず、登録不可へ吸収する。
 * 既知の制約:
 * - `hukkSysShbt`、`correlationId`、`systemDatetime` は public 引数で扱わない。
 * - blank 補完や trim は行わず、空判定は空文字との完全一致のみを対象とする。
 */
public final class GasUsageCodeRegistrationAvailabilityJudgmentImpl
        implements GasUsageCodeRegistrationAvailabilityJudgment {

    private final ConstantMasterInfoAcquisition constantMasterInfoAcquisition;

    private final CstkismDataAccess cstkismDataAccess;

    /**
     * 概要: 判定に必要な下位依存を保持して初期化する。
     * 主な仕様:
     * - SV-089 と DA-162-R001 を必須依存として受け取る。
     * - 依存不足時はコンストラクタで fail-fast する。
     *
     * @param constantMasterInfoAcquisition SV-089 コンスタントマスタ情報取得
     * @param cstkismDataAccess DA-162-R001 コンスタント階層構造マスター取得
     */
    public GasUsageCodeRegistrationAvailabilityJudgmentImpl(
            ConstantMasterInfoAcquisition constantMasterInfoAcquisition,
            CstkismDataAccess cstkismDataAccess) {
        this.constantMasterInfoAcquisition = Objects.requireNonNull(
                constantMasterInfoAcquisition,
                "constantMasterInfoAcquisition");
        this.cstkismDataAccess = Objects.requireNonNull(cstkismDataAccess, "cstkismDataAccess");
    }

    /**
     * 概要: ガス用途コードの登録可否を判定する。
     * 主な仕様:
     * - 入力が `null` または空文字の場合は業務エラー `020000091` をスローする。
     * - SV-089 の例外時は業務エラー `020000092` をスローする。
     * - DA-162-R001 の例外、既存階層構造ありのいずれでも `false` を返却する。
     * - SV-089 正常終了かつ DA-162-R001 が 0 件の場合のみ `true` を返却する。
     *
     * @param gasYtCd ガス用途コード
     * @return 判定結果 DTO
     * @throws BusinessRuleException ガス用途コードが `null` または空文字の場合、または SV-089 が例外を送出した場合
     */
    @Override
    public GasUsageCodeRegistrationAvailabilityJudgmentOutput execute(String gasYtCd) {
        if (gasYtCd == null || gasYtCd.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000091);
        }

        ConstantMasterInfoAcquisitionOutput constantMasterOutput;
        try {
            constantMasterOutput = constantMasterInfoAcquisition.execute(
                    ConstantLookupKeys.Cirius.GasUsageCode.TYPE,
                    gasYtCd);
        } catch (RuntimeException ignored) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000092);
        }

        String cntnMix = constantMasterOutput == null ? null : constantMasterOutput.cntn();
        return output(isRegisterable(gasYtCd), cntnMix);
    }

    /**
     * 概要: DA-162-R001 を使って登録可否を判定する。
     * 主な仕様:
     * - `J0526` / 入力ガス用途コード / `17300` で階層構造マスタを検索する。
     * - 取得結果が 0 件なら登録可、1 件以上なら登録不可と判定する。
     * - DA 例外時は登録不可と判定する。
     *
     * @param gasYtCd ガス用途コード
     * @return 登録可能な場合 true
     */
    private boolean isRegisterable(String gasYtCd) {
        try {
            return cstkismDataAccess.findByJiCtshCdAndJiCnstCdAndKaiCtshCd(
                    ConstantLookupKeys.Cirius.GasUsageCodeHierarchyType.TYPE,
                    gasYtCd,
                    ConstantLookupKeys.Cirius.GasUsageCode.TYPE)
                    .isEmpty();
        } catch (RuntimeException exception) {
            return false;
        }
    }

    private static GasUsageCodeRegistrationAvailabilityJudgmentOutput output(
            boolean gasYtCdTkhFlg, String cntnMix) {
        return new GasUsageCodeRegistrationAvailabilityJudgmentOutput(gasYtCdTkhFlg, cntnMix);
    }
}
