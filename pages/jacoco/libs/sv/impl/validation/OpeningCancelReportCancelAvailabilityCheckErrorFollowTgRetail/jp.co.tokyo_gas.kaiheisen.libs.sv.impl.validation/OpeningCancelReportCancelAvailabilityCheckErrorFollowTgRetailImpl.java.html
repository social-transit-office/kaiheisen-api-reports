    private static <T> T mapRecord(Object source, Class<T> targetType) {
        if (source == null) {
            return null;
        }
        try {
            RecordComponent[] recordComponents = targetType.getRecordComponents();
            Object[] arguments = new Object[recordComponents.length];
            for (int index = 0; index < recordComponents.length; index++) {
                arguments[index] = source.getClass()
                        .getMethod(recordComponents[index].getName())
                        .invoke(source);
            }
            Class<?>[] parameterTypes = Arrays.stream(recordComponents)
                    .map(RecordComponent::getType)
                    .toArray(Class<?>[]::new);
            return targetType.getDeclaredConstructor(parameterTypes).newInstance(arguments);
        } catch (ReflectiveOperationException exception) {
            throw new IllegalStateException(
                    "Failed to map record: " + source.getClass().getName() + " -> " + targetType.getName(),
                    exception);
        }
    }
