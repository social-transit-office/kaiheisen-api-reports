/**
 * SV-305 元メーター解約時専用メーター契約チェックの実装。
 */
public final class OriginalMeterCancellationDedicatedContractCheckImpl
        implements OriginalMeterCancellationDedicatedContractCheck {

    private final MtsetkDataAccess mtsetkDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final RequestContext requestContext;

    private final String originalMeterStatus;

    private final List<String> relatedStatusList;

    /**
     * 依存コンポーネントと必須定数を受け取って初期化する。
     *
     * @param mtsetkDataAccess メーター設置場所＜小売＞データアクセス
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public OriginalMeterCancellationDedicatedContractCheckImpl(
            MtsetkDataAccess mtsetkDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            SvmstkrDataAccess svmstkrDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.mtsetkDataAccess = Objects.requireNonNull(mtsetkDataAccess, "mtsetkDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.originalMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.relatedStatusList = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedConstantService,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03));
    }

    /**
     * 概要: SV-305 を実行する。
     * 主な仕様:
     * - `gasMtstNo` と `knskKjnYmd` を条件に小売メーター設置場所情報を取得する。
     * - `ckrtGmscJtkb` が元メーターの場合のみ `RequestContext.systemDatetime` を使って専用メーター候補を探索し、契約有無を判定する。
     * 既知の制約:
     * - public 引数の null 補完や trim は行わず、必須依存や必須コンテキスト値は fail-fast とする。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return 判定結果
     */
    @Override
    public OriginalMeterCancellationDedicatedContractCheckOutput execute(
            String ckrtGmscJtkb,
            Long gasMtstNo,
            LocalDate knskKjnYmd) {
        MtsetkRecord mtsetkRecord = Objects.requireNonNull(
                mtsetkDataAccess.findByMtstNoAndTkysYmd(gasMtstNo, knskKjnYmd),
                "mtsetkDataAccess.findByMtstNoAndTkysYmd returned null")
                .orElseThrow(() -> DataInconsistencySupport.create(
                        mtsetkDataAccess.tableName(),
                        MtsetkDataAccess.CallSpecs.FIND_BY_MTST_NO_AND_TKYS_YMD.identifier(),
                        new FlatJsonBuilder()
                                .put("mtstNo", gasMtstNo)
                                .put("tkysYmd", knskKjnYmd)));

        if (!originalMeterStatus.equals(ckrtGmscJtkb)) {
            return new OriginalMeterCancellationDedicatedContractCheckOutput(
                    toMtsetkInf(mtsetkRecord),
                    List.of());
        }

        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();

        List<GmtsetdRecord> gmtsetdRecords = Objects.requireNonNull(
                gmtsetdDataAccess
                        .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                                mtsetkRecord.szcGkkCd(),
                                mtsetkRecord.szcCmazCd(),
                                mtsetkRecord.szcMchCd(),
                                gasMtstNo,
                                gasMtstNo,
                                systemDate),
                "gmtsetdDataAccess.findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAnd"
                        + "MtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd returned null");

        List<SnyInf> snyInfList = gmtsetdRecords.stream()
                .map(gmtsetdRecord -> new SnyInf(
                        gmtsetdRecord.gasMtstNo()))
                .toList();

        for (GmtsetdRecord gmtsetdRecord : gmtsetdRecords) {
            Optional<SvmstkrRecord> relatedContract = Objects.requireNonNull(
                    svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb(
                            gmtsetdRecord.gasMtstNo(),
                            relatedStatusList),
                    "svmstkrDataAccess.findByMtstNoAndRkkiKnrnJtkb returned null");
            if (relatedContract.isPresent()) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000076);
            }
        }

        return new OriginalMeterCancellationDedicatedContractCheckOutput(
                toMtsetkInf(mtsetkRecord),
                List.copyOf(snyInfList));
    }

    private static MtsetkInf toMtsetkInf(
            MtsetkRecord mtsetkRecord) {
        return new MtsetkInf(
                mtsetkRecord.mtstNo(),
                mtsetkRecord.tkysYmd(),
                mtsetkRecord.tyenYmd(),
                mtsetkRecord.mtrShbt(),
                mtsetkRecord.nwjgCd(),
                mtsetkRecord.szcGkkCd(),
                mtsetkRecord.szcMchCd(),
                mtsetkRecord.szcCmazCd(),
                mtsetkRecord.cbgSuj1(),
                mtsetkRecord.cbgKn2(),
                mtsetkRecord.cbgKn3(),
                mtsetkRecord.cbgKn4(),
                mtsetkRecord.bldNo(),
                mtsetkRecord.tunoKn(),
                mtsetkRecord.hynKn(),
                mtsetkRecord.htuFlg(),
                mtsetkRecord.jutKbn());
    }
}
