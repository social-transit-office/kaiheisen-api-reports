package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ComprehensiveSeriesGRelatedErrorJudgment;

/**
 * 概要: SV-148（総合・直列Ｇ関連エラー判定）の実装クラス。
 * 主な仕様:
 * - 総合扱いガスメーター件数を使う判定系統と、直列元ガスメーター設置状態区分を使う判定系統を順に評価する。
 * - いずれかの判定系統が成立した場合は閉栓共通エラー 020200046（HeisenRuleException）を送出する。
 * - 正常系では何も返さずに終了する。
 * 既知の制約:
 * - 入力値の null 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となり、インスタンス生成に失敗する。
 */
public final class ComprehensiveSeriesGRelatedErrorJudgmentImpl
        implements ComprehensiveSeriesGRelatedErrorJudgment {

    private final String tokyoGasRetailerCode;

    private final String relocationClosureReason;

    private final String unusedClosureReason;

    private final String firedAndLostCapClosureReason;

    private final String supplyOutageClosureReason;

    private final String originalGasMeterStatus;

    /**
     * 概要: 判定に使用する CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 小売事業者コード J0580 の東京ガス、および閉栓事由種別 J0273 の対象値を必須解決して保持する。
     * - 直列ガスメーター設置状態区分 J0244 の元ガスメーターを必須解決して保持する。
     * 既知の制約:
     * - CIRIUS 定数取得サービスの null は許容しない。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ComprehensiveSeriesGRelatedErrorJudgmentImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.tokyoGasRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.relocationClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE);
        this.unusedClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.UNUSED_CLOSURE);
        this.firedAndLostCapClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.FIRED_AND_LOST_CAP);
        this.supplyOutageClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
    }

    /**
     * 概要: 総合・直列Ｇ関連エラー判定を実行する。
     * 主な仕様:
     * - 第1判定系統では東京ガス、小売対象の閉栓事由、総合扱い件数、同一グループ内開栓メータ有無、全部閉栓不可事由フラグを評価する。
     * - 第2判定系統では直列元ガスメーター、閉栓事由、全部閉栓不可事由フラグを評価する。
        * - いずれかが成立した場合は閉栓共通エラー 020200046（HeisenRuleException）を送出する。
     * 既知の制約:
     * - `correlationId`、`hukkSysShbt`、`systemDatetime` は SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードは追加せず、上位バリデーション済み前提で評価する。
     *
     * @param kjgsCd 小売事業者コード
     * @param hesnJysb 閉栓事由種別
     * @param suguGmtKnsu 総合扱いガスメーター件数
     * @param dgrpKasFlg 同一グループ内開栓メータ有フラグ
     * @param dgrpHkhesnJiyuFlg 同一グループ内全部閉栓不可閉栓事由フラグ
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
        * @throws HeisenRuleException いずれかの判定系統が成立し、閉栓共通エラーを送出する場合
     */
    @Override
         public void execute(
            String kjgsCd,
            String hesnJysb,
            Integer suguGmtKnsu,
            Boolean dgrpKasFlg,
            Boolean dgrpHkhesnJiyuFlg,
            String ckrtGmscJtkb) {
        if (shouldRejectForComprehensiveGroup(
                        kjgsCd,
                        hesnJysb,
                        suguGmtKnsu,
                        dgrpKasFlg,
                        dgrpHkhesnJiyuFlg)
                || shouldRejectForSeriesOriginMeter(ckrtGmscJtkb, hesnJysb, dgrpHkhesnJiyuFlg)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200046);
        }
    }

    /**
     * 概要: 総合扱いガスメーター判定系統の条件を評価する。
     * 主な仕様:
     * - 小売事業者コードが東京ガス、閉栓事由が移転閉栓/不使用閉栓/
     *   焼紛失閉栓、総合扱い件数が 0 より大きい、同一グループ内開栓メータ有フラグが
     *   false、全部閉栓不可事由フラグが true の全条件成立時に true を返す。
     * - step `1.a.1.a.1.a.1.a.1.a.1` の例外送出条件を
     *   そのままメソッド化している。
     * 既知の制約:
     * - `Integer` や `Boolean` の null は条件不成立として扱う。
     *
     * @param kjgsCd 小売事業者コード
     * @param hesnJysb 閉栓事由種別
     * @param suguGmtKnsu 総合扱いガスメーター件数
     * @param dgrpKasFlg 同一グループ内開栓メータ有フラグ
     * @param dgrpHkhesnJiyuFlg 同一グループ内全部閉栓不可閉栓事由フラグ
     * @return 第1判定系統が成立する場合 true
     */
    private boolean shouldRejectForComprehensiveGroup(
            String kjgsCd,
            String hesnJysb,
            Integer suguGmtKnsu,
            Boolean dgrpKasFlg,
            Boolean dgrpHkhesnJiyuFlg) {
        return tokyoGasRetailerCode.equals(kjgsCd)
                && isClosureReasonForComprehensiveGroup(hesnJysb)
                && suguGmtKnsu != null
                && suguGmtKnsu > 0
                && Boolean.FALSE.equals(dgrpKasFlg)
                && Boolean.TRUE.equals(dgrpHkhesnJiyuFlg);
    }

    /**
     * 概要: 直列元ガスメーター判定系統の条件を評価する。
     * 主な仕様:
     * - 直列ガスメーター設置状態区分が元ガスメーターであり、
     *   閉栓事由が移転閉栓/不使用閉栓/焼紛失閉栓/供給停止のいずれかで、
     *   全部閉栓不可事由フラグが true の全条件成立時に true を返す。
     * - step `2.a.1.a.1.a.1` の例外送出条件をそのまま
     *   メソッド化している。
     * 既知の制約:
     * - `Boolean` の null は条件不成立として扱う。
     * - 第1判定系統と独立に評価するが、呼び出し順は execute 内の記述順に従う。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param hesnJysb 閉栓事由種別
     * @param dgrpHkhesnJiyuFlg 同一グループ内全部閉栓不可閉栓事由フラグ
     * @return 第2判定系統が成立する場合 true
     */
    private boolean shouldRejectForSeriesOriginMeter(
            String ckrtGmscJtkb,
            String hesnJysb,
            Boolean dgrpHkhesnJiyuFlg) {
        return originalGasMeterStatus.equals(ckrtGmscJtkb)
                && isClosureReasonForSeriesOriginMeter(hesnJysb)
                && Boolean.TRUE.equals(dgrpHkhesnJiyuFlg);
    }

    /**
     * 概要: 第1判定系統で対象となる閉栓事由かを判定する。
     * 主な仕様:
     * - 移転閉栓、不使用閉栓、焼紛失閉栓のいずれかと一致する場合に true を返す。
     * 既知の制約:
     * - 設計に含まれない閉栓事由は一律に false とする。
     *
     * @param hesnJysb 閉栓事由種別
     * @return 第1判定系統の対象閉栓事由である場合 true
     */
    private boolean isClosureReasonForComprehensiveGroup(String hesnJysb) {
        return relocationClosureReason.equals(hesnJysb)
                || unusedClosureReason.equals(hesnJysb)
                || firedAndLostCapClosureReason.equals(hesnJysb);
    }

    /**
     * 概要: 第2判定系統で対象となる閉栓事由かを判定する。
     * 主な仕様:
     * - 移転閉栓、不使用閉栓、焼紛失閉栓、供給停止のいずれかと一致する場合に true を返す。
     * 既知の制約:
     * - 第1判定系統の対象に加えて供給停止を許容する点のみが差分である。
     *
     * @param hesnJysb 閉栓事由種別
     * @return 第2判定系統の対象閉栓事由である場合 true
     */
    private boolean isClosureReasonForSeriesOriginMeter(String hesnJysb) {
        return isClosureReasonForComprehensiveGroup(hesnJysb)
                || supplyOutageClosureReason.equals(hesnJysb);
    }
}
