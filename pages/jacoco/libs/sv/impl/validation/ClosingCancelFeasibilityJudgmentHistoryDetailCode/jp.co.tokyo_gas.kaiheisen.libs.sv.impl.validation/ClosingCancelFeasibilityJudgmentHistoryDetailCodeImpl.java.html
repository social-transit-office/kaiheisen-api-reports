package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;

import java.util.Optional;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit;

/**
 * 概要: SV-097（閉栓取消実施可否判定：異動・作業履歴業務詳細コード）のサービス実装。
 * 主な仕様:
 * - 小売・導管の異動・作業履歴を取得し、取消系業務詳細コードは返却値を null 化する。
 * - 小売履歴が取消系業務詳細コードの場合は全項目 null を返却し、導管履歴取得および業務エラー判定を行わない。
 * - 導管履歴が未取得または取消系業務詳細コードの場合は、導管返却値を null 化して処理を終了し、業務エラー判定を行わない。
 * - 業務エラー判定は、導管履歴が 1 件かつ取消系業務詳細コードではない場合に、小売履歴の業務コードと取消対象業務コードの組み合わせで実施する。
 * 既知の制約:
 * - 業務判定には DA-102-R002 の取得業務コードを用いる。
 * - 小売履歴が未取得の場合は業務コード null として扱う。
 * - systemDatetime は RequestContext から取得し、検索年月日に変換する。
 */
public final class ClosingCancelFeasibilityJudgmentHistoryDetailCodeImpl
        implements ClosingCancelFeasibilityJudgmentHistoryDetailCode {

    private static final List<String> RETAIL_HISTORY_DETAIL_CODES = List.of(
        TransferWorkHistoryWorkDetailCodeRetail.RELOCATION_CLOSURE_CCBG6301,
        TransferWorkHistoryWorkDetailCodeRetail.UNUSED_CLOSURE_CCBG6302,
        TransferWorkHistoryWorkDetailCodeRetail.RELOCATION_CLOSURE_CCBG6801,
        TransferWorkHistoryWorkDetailCodeRetail.UNUSED_CLOSURE_CCBG6802,
        TransferWorkHistoryWorkDetailCodeRetail.RELOCATION_CLOSURE_CCBG6601,
        TransferWorkHistoryWorkDetailCodeRetail.UNUSED_CLOSURE_CCBG6604,
        TransferWorkHistoryWorkDetailCodeRetail.SUPPLY_OUTAGE_CCBG8401,
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6303,
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6804,
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6603,
        TransferWorkHistoryWorkDetailCodeRetail.CANCEL_CCBG8402);

    private static final List<String> CONDUIT_HISTORY_DETAIL_CODES = List.of(
        TransferWorkHistoryWorkDetailsCodeConduit.RELOCATION_CLOSURE_CCJG0701,
        TransferWorkHistoryWorkDetailsCodeConduit.UNUSED_CLOSURE_CCJG0702,
        TransferWorkHistoryWorkDetailsCodeConduit.RELOCATION_CLOSURE_CCJG0801,
        TransferWorkHistoryWorkDetailsCodeConduit.UNUSED_CLOSURE_CCJG0802,
        TransferWorkHistoryWorkDetailsCodeConduit.RELOCATION_CLOSURE_CCJG1001,
        TransferWorkHistoryWorkDetailsCodeConduit.UNUSED_CLOSURE_CCJG1004,
        TransferWorkHistoryWorkDetailsCodeConduit.SUPPLY_OUTAGE_CCJG1301,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG0703,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG0803,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG1003,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG1302);

    private static final Set<String> RETAIL_CANCELLATION_DETAIL_CODES = Set.of(
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6303,
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6804,
        TransferWorkHistoryWorkDetailCodeRetail.CANCELLATION_CLOSED_CCBG6603,
        TransferWorkHistoryWorkDetailCodeRetail.CANCEL_CCBG8402);

    private static final Set<String> CONDUIT_CANCELLATION_DETAIL_CODES = Set.of(
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG0703,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG0803,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG1003,
        TransferWorkHistoryWorkDetailsCodeConduit.CANCEL_CCJG1302);

    private final IsrkkDataAccess isrkkDataAccess;

    private final IsrkdDataAccess isrkdDataAccess;

    private final RequestContext requestContext;

    /**
     * 概要: 必須依存を初期化する。
     *
     * @param isrkkDataAccess 小売履歴データアクセス
     * @param isrkdDataAccess 導管履歴データアクセス
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public ClosingCancelFeasibilityJudgmentHistoryDetailCodeImpl(
            IsrkkDataAccess isrkkDataAccess,
            IsrkdDataAccess isrkdDataAccess,
            RequestContext requestContext) {
        this.isrkkDataAccess = Objects.requireNonNull(isrkkDataAccess, "isrkkDataAccess");
        this.isrkdDataAccess = Objects.requireNonNull(isrkdDataAccess, "isrkdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
    }

    /**
     * 概要: 異動・作業履歴業務詳細コードから閉栓取消可否を判定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkIsrrGymCd 取消対象異動・作業履歴業務コード
     * @return 判定結果
     */
    @Override
    public ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput execute(
            Long gasMtstNo,
            String trkIsrrGymCd) {
        LocalDate resolvedTargetDate = resolveTargetDate();

        Optional<IsrkkRecord> retailRecord = isrkkDataAccess
                .findByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
                gasMtstNo,
                        resolvedTargetDate,
                        RETAIL_HISTORY_DETAIL_CODES);
        HistoryLookupResult retailResult = toRetailResult(retailRecord);
        if (retailResult.isCancellation()) {
            return new ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput(
                null,
                null,
                null,
                null);
        }

        Optional<IsrkdRecord> conduitRecord = isrkdDataAccess
                .findByArtyCstNoAndIsrrTriYmdAndDuknIsrsCd(
                gasMtstNo,
                        resolvedTargetDate,
                        CONDUIT_HISTORY_DETAIL_CODES);
        HistoryLookupResult conduitResult = toConduitResult(conduitRecord);
        if (conduitResult.isEmpty() || conduitResult.isCancellation()) {
            return new ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput(
                retailResult.output().businessCode(),
                retailResult.output().detailCode(),
                conduitResult.output().businessCode(),
                conduitResult.output().detailCode());
        }

        validateClosingCancelTarget(
                trkIsrrGymCd,
                retailResult.sourceBusinessCode());

        return new ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput(
                retailResult.output().businessCode(),
                retailResult.output().detailCode(),
                conduitResult.output().businessCode(),
                conduitResult.output().detailCode());
    }

    private HistoryLookupResult toRetailResult(Optional<IsrkkRecord> retailRecord) {
        if (retailRecord.isEmpty()) {
            return HistoryLookupResult.empty();
        }
        IsrkkRecord record = retailRecord.orElseThrow();
        if (RETAIL_CANCELLATION_DETAIL_CODES.contains(record.isrrGmssCd())) {
            return HistoryLookupResult.cancellation(record.isrrGymCd());
        }
        return HistoryLookupResult.active(record.isrrGymCd(), record.isrrGmssCd());
    }

    private HistoryLookupResult toConduitResult(Optional<IsrkdRecord> conduitRecord) {
        if (conduitRecord.isEmpty()) {
            return HistoryLookupResult.empty();
        }
        IsrkdRecord record = conduitRecord.orElseThrow();
        if (CONDUIT_CANCELLATION_DETAIL_CODES.contains(record.duknIsrsCd())) {
            return HistoryLookupResult.cancellation(record.duknIsrgCd());
        }
        return HistoryLookupResult.active(record.duknIsrgCd(), record.duknIsrsCd());
    }

    private LocalDate resolveTargetDate() {
        return requireSystemDatetime().toLocalDate();
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }

    private void validateClosingCancelTarget(String trkIsrrGymCd, String retailBusinessCode) {
        if (ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION
                .equals(trkIsrrGymCd)) {
            if (ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                    .CLOSURE_ERROR_FOLLOW_UP_CCBG68.equals(retailBusinessCode)) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000064);
            }
            if (!ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION
                    .equals(retailBusinessCode)
                    && !ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                            .SUPPLY_OUTAGE_REPORT_RETAIL.equals(retailBusinessCode)) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000065);
            }
            return;
        }

        if (ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                .CLOSURE_ERROR_FOLLOW_UP_CCBG68.equals(trkIsrrGymCd)) {
            if (ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION
                    .equals(retailBusinessCode)) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000051);
            }
            if (!ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                    .CLOSURE_ERROR_FOLLOW_UP_CCBG68.equals(retailBusinessCode)) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000055);
            }
            return;
        }

        if (ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION
                .equals(retailBusinessCode)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000051);
        }
        if (!ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                .ONLINE_CLOSURE_REPORT_RETAIL.equals(retailBusinessCode)
                && !ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .SUPPLY_OUTAGE_REPORT_RETAIL.equals(retailBusinessCode)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000055);
        }
    }

    private record HistoryCode(String businessCode, String detailCode) {

        private static HistoryCode empty() {
            return new HistoryCode(null, null);
        }
    }

    private record HistoryLookupResult(
            String sourceBusinessCode,
            HistoryCode output,
            HistoryRecordState state) {

        private static HistoryLookupResult empty() {
            return new HistoryLookupResult(null, HistoryCode.empty(), HistoryRecordState.EMPTY);
        }

        private static HistoryLookupResult cancellation(String sourceBusinessCode) {
            return new HistoryLookupResult(
                    sourceBusinessCode,
                    HistoryCode.empty(),
                    HistoryRecordState.CANCELLATION);
        }

        private static HistoryLookupResult active(String businessCode, String detailCode) {
            return new HistoryLookupResult(
                    businessCode,
                    new HistoryCode(businessCode, detailCode),
                    HistoryRecordState.ACTIVE);
        }

        private boolean isEmpty() {
            return state == HistoryRecordState.EMPTY;
        }

        private boolean isCancellation() {
            return state == HistoryRecordState.CANCELLATION;
        }
    }

    private enum HistoryRecordState {
        EMPTY,
        ACTIVE,
        CANCELLATION
    }
}
