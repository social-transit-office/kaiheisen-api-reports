package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.SuspendedRemovedErrorJudgment;

/**
 * 概要: SV-136（休止・撤去済エラー判定）の実装クラス。
 * 主な仕様:
 * - 判定に必要なメーター保有状態コードをコンストラクタ初期化時に必須解決して保持する。
 * - 供内管・ガスメーター設備状態区分がメーター保有状態と一致しない場合は業務例外 020200011 を送出する。
 * - 一致する場合は追加処理を行わず正常終了する。
 * 既知の制約:
 * - public 引数の null 判定や空白補正は上位層責務とし、本クラスでは追加しない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となる。
 */
public final class SuspendedRemovedErrorJudgmentImpl implements SuspendedRemovedErrorJudgment {

    private final String meterPossessionStatus;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する。
     * 主な仕様:
     * - 05800 のメーター保有状態コードを必須解決して保持する。
     * - CIRIUS 定数取得サービスの依存は fail-fast で検証する。
     * 既知の制約:
     * - 必須定数解決失敗時はインスタンス生成を継続しない。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public SuspendedRemovedErrorJudgmentImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.meterPossessionStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus
                                .METER_POSSESSION_STATUS);
    }

    /**
     * 概要: 供内管・ガスメーター設備状態区分を判定する。
     * 主な仕様:
     * - 引数が解決済みメーター保有状態コードと一致しない場合は業務例外 020200011 を送出する。
     * - 一致する場合は処理を継続せず正常終了する。
     * 既知の制約:
     * - 引数が null や空文字でも個別ガードせず、不一致として業務例外を送出する。
     *
     * @param kngmStbJtkb 供内管・ガスメーター設備状態区分
     */
    @Override
    public void execute(String kngmStbJtkb) {
        if (!meterPossessionStatus.equals(kngmStbJtkb)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200011);
        }
    }
}
