package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.GmtsetdInf;

/** SV-074 開栓取消実施可否判定：元メーターに紐づく専用メーターチェックの実装. */
public final class OpeningCancelFeasibilityJudgmentDedicatedMeterCheckImpl
        implements OpeningCancelFeasibilityJudgmentDedicatedMeterCheck {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final RequestContext requestContext;

    private final String originalGasMeterStatus;

    private final String meterPossessionStatus;

    private final String openingStatus;

    /**
     * 依存コンポーネントを受け取ってサービス実装を初期化する.
     *
     * @param gmtsetdDataAccessParam ガスメーター設置場所＜導管＞データアクセス
     * @param ciriusConstantServiceParam CIRIUS 定数サービス
     * @param requestContextParam リクエストコンテキスト
     */
    public OpeningCancelFeasibilityJudgmentDedicatedMeterCheckImpl(
            final GmtsetdDataAccess gmtsetdDataAccessParam,
            final CiriusConstantService ciriusConstantServiceParam,
            final RequestContext requestContextParam) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccessParam, "gmtsetdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContextParam, "requestContext");
        CiriusConstantService ciriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceParam, "ciriusConstantService");
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.meterPossessionStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 元ガスメーターに紐づく専用メーター候補を判定し、正常時は候補リストを返却する.
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param szcGkkCd 所在地行政区画コード
     * @param szcMchCd 所在地町コード
     * @param szcCmazCd 所在地丁目・字コード
     * @return 判定結果
     */
    @Override
    public OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput execute(
            final String ckrtGmscJtkb,
            final Long gasMtstNo,
            final LocalDate trkKnsYmd,
            final String szcGkkCd,
            final String szcMchCd,
            final String szcCmazCd) {
        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return new OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput(new ArrayList<>());
        }

        LocalDate knskKjnYmd = requireSystemDatetime().toLocalDate();
        List<GmtsetdRecord> gmtsetdRecords = Objects.requireNonNull(
                gmtsetdDataAccess
                        .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmdAndKngmStbJtkb(
                                szcGkkCd,
                                szcCmazCd,
                                szcMchCd,
                                gasMtstNo,
                                gasMtstNo,
                                knskKjnYmd,
                                meterPossessionStatus),
                "gmtsetdDataAccess.findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAnd"
                        + "MtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmdAndKngmStbJtkb "
                        + "(DA-033-R008) returned null");

        if (gmtsetdRecords.isEmpty()) {
            return new OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput(new ArrayList<>());
        }

        ArrayList<GmtsetdInf> gmtsetdInfList = new ArrayList<>(gmtsetdRecords.size());
        for (GmtsetdRecord gmtsetdRecord : gmtsetdRecords) {
            validateOpeningStatus(gmtsetdRecord);
            gmtsetdInfList.add(new GmtsetdInf(gmtsetdRecord.gasMtstNo()));
        }
        return new OpeningCancelFeasibilityJudgmentDedicatedMeterCheckOutput(gmtsetdInfList);
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private void validateOpeningStatus(final GmtsetdRecord gmtsetdRecord) {
        if (!openingStatus.equals(gmtsetdRecord.gmtKhsJtkb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000050);
        }
    }
}
