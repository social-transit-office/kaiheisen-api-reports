package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;

/**
 * 概要: SV-096（閉栓取消実施可否判定：フォロー状態コード）のサービス実装。
 * 主な仕様:
 * - 処理区分に応じて HSEFL 検索先を切り替える。
 * - 取得したフォローステータスコードと RequestContext の値から閉栓取消可否を判定する。
 * - 業務条件に応じて業務エラー 020000041 / 020000064 / 020000066 を送出する。
 * 既知の制約:
 * - 実行時には RequestContext の hukkSysShbt と systemDatetime が必須である。
 * - HSEFL 検索結果は先頭 1 件を判定対象として扱う。
 */
public class ClosingCancelFeasibilityJudgmentFollowStatusImpl
        implements ClosingCancelFeasibilityJudgmentFollowStatus {

    private static final String NOT_YET_STARTED_KBN = "1";

    private static final String OTHER_KBN = "0";

    private final HseflDataAccess hseflDataAccess;

    private final RequestContext requestContext;

    private final String reportingSystemTypeLivalit;

    private final String followStatusNotYetStarted;

    private final String followStatusCompletedNoFollowRequired;

    private final String followStatusCompletedFollowCompleted;

    /**
     * 概要: 必要な依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - HSEFL 参照、RequestContext、定数サービスを保持する。
     * - 判定に必要なフォローステータスコード定数を初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * - 必須 CIRIUS 定数が不足している場合は初期化時点で失敗する。
     *
     * @param hseflDataAccess HSEFL データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public ClosingCancelFeasibilityJudgmentFollowStatusImpl(
            HseflDataAccess hseflDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
            kaiheisenConstantService,
            "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.reportingSystemTypeLivalit = RequiredConstantResolver.requireKaiheisenConstant(
            resolvedKaiheisenConstantService,
            ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
            ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT);
        this.followStatusNotYetStarted = RequiredConstantResolver.requireCiriusConstant(
            resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);
        this.followStatusCompletedNoFollowRequired = RequiredConstantResolver.requireCiriusConstant(
            resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_NO_FOLLOW_REQUIRED);
        this.followStatusCompletedFollowCompleted = RequiredConstantResolver.requireCiriusConstant(
            resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
            ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_FOLLOW_COMPLETED);
    }

    /**
     * 概要: 閉栓エラーフォロー件名の状態から閉栓取消可否を判定する。
     * 主な仕様:
     * - 処理区分に応じて HSEFL を検索する。
     * - 取消対象業務コードと報告システム種別に応じて 4.a 系 / 4.b 系の判定を実施する。
     * - 戻り値には派生した未着手区分と必要時のみ HSEFL 情報を設定する。
     *
     * @param trkTishGmCd 取消対象業務コード
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param shrKbn 処理区分
     * @return 判定結果
     */
    @Override
    public ClosingCancelFeasibilityJudgmentFollowStatusOutput execute(
            String trkTishGmCd,
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            String shrKbn) {
        LocalDateTime systemDatetime = requireRequestContext();
        Optional<HseflRecord> targetRecordOpt = findTargetRecord(gasMtstNo, trkKnsYmd, shrKbn);
        String hseflFlwMcksKbn = deriveNotStartedFlag(targetRecordOpt);
        if (isReportDateValidationTarget(trkTishGmCd)) {
            return evaluateClosingReportDateValidation(
            targetRecordOpt,
                hseflFlwMcksKbn,
                systemDatetime.toLocalDate());
        }
        return evaluateFollowCompletionDateValidation(
            targetRecordOpt,
            hseflFlwMcksKbn,
            systemDatetime.toLocalDate());
    }

    /**
     * 概要: 処理区分に応じて HSEFL を検索する。
     * 主な仕様:
     * - `shrKbn=1` は DA-141-R005、`shrKbn=2` は DA-141-R006 を使用する。
     * - それ以外の処理区分は検索せず未取得扱いとする。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param shrKbn 処理区分
     * @return 判定対象レコード
     */
    private Optional<HseflRecord> findTargetRecord(Long gasMtstNo, LocalDate trkKnsYmd, String shrKbn) {
        List<HseflRecord> records;
        if ("1".equals(shrKbn)) {
            records = hseflDataAccess.findAllByGasMtstNoAndTsigHesnSgyYmd(gasMtstNo, trkKnsYmd);
        } else if ("2".equals(shrKbn)) {
            records = hseflDataAccess.findAllByGasMtstNoAndHssgYmd(gasMtstNo, trkKnsYmd);
        } else {
            records = List.of();
        }
        if (records.isEmpty()) {
            return Optional.empty();
        }
        return Optional.of(records.getFirst());
    }

    /**
     * 概要: 未着手区分の派生値を求める。
     * 主な仕様:
     * - HSEFL 未取得または未着手以外のフォローステータスでは `0` を返す。
     * - HSEFL 取得済みかつ未着手の場合のみ `1` を返す。
     *
     * @param targetRecordOpt 判定対象レコード
     * @return 派生した未着手区分
     */
    private String deriveNotStartedFlag(Optional<HseflRecord> targetRecordOpt) {
        if (targetRecordOpt.map(HseflRecord::flwStCd)
                .filter(followStatusNotYetStarted::equals)
                .isPresent()) {
            return NOT_YET_STARTED_KBN;
        }
        return OTHER_KBN;
    }

    /**
     * 概要: 報告年月日判定系の対象業務かを判定する。
     * 主な仕様:
     * - `CCBG63` は常に 4.a 系とする。
     * - `CCBG84` は RequestContext の報告システム種別が LIVALIT の場合のみ 4.a 系とする。
     *
     * @param trkTishGmCd 取消対象業務コード
     * @return 4.a 系なら true
     */
    private boolean isReportDateValidationTarget(String trkTishGmCd) {
        boolean isClosingRegistration =
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION
                        .equals(trkTishGmCd);
        if (isClosingRegistration) {
            return true;
        }
        boolean isRetailSupplyOutage =
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.SUPPLY_OUTAGE_REPORT_RETAIL
                        .equals(trkTishGmCd);
        if (!isRetailSupplyOutage) {
            return false;
        }
        return reportingSystemTypeLivalit.equals(requireHukkSysShbt());
    }

    /**
    * 概要: 閉栓報告年月日ベースの判定を行う。
     * 主な仕様:
     * - 未着手かつ当日報告なら正常終了し HSEFL 情報を返す。
     * - 完了系ステータスは業務エラーへ変換する。
     * - 上記以外は HSEFL 情報を返さず派生区分のみ返す。
     *
     * @param targetRecordOpt 判定対象レコード
     * @param hseflFlwMcksKbn 派生した未着手区分
     * @param systemDate システム日付
     * @return 判定結果
     */
    private ClosingCancelFeasibilityJudgmentFollowStatusOutput evaluateClosingReportDateValidation(
            Optional<HseflRecord> targetRecordOpt,
            String hseflFlwMcksKbn,
            LocalDate systemDate) {
        if (targetRecordOpt.isEmpty()) {
            return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(hseflFlwMcksKbn, null);
        }
        HseflRecord targetRecord = targetRecordOpt.orElseThrow();
        if (followStatusNotYetStarted.equals(targetRecord.flwStCd())) {
            if (Objects.equals(targetRecord.hesnHukkYmd(), systemDate)) {
                return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(
                        hseflFlwMcksKbn,
                        toHseflInf(targetRecord));
            }
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000041);
        }
        if (followStatusCompletedFollowCompleted.equals(targetRecord.flwStCd())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000064);
        }
        if (followStatusCompletedNoFollowRequired.equals(targetRecord.flwStCd())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000066);
        }
        return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(hseflFlwMcksKbn, null);
    }

    /**
    * 概要: フォロー対応完了年月日ベースの判定を行う。
     * 主な仕様:
     * - フォロー完了かつ当日報告なら正常終了し HSEFL 情報を返す。
     * - フォロー完了で当日報告以外なら業務エラー 020000041 を送出する。
     * - 上記以外は取得した HSEFL 情報をそのまま返す。
     *
     * @param targetRecordOpt 判定対象レコード
     * @param hseflFlwMcksKbn 派生した未着手区分
     * @param systemDate システム日付
     * @return 判定結果
     */
    private ClosingCancelFeasibilityJudgmentFollowStatusOutput evaluateFollowCompletionDateValidation(
            Optional<HseflRecord> targetRecordOpt,
            String hseflFlwMcksKbn,
            LocalDate systemDate) {
        if (targetRecordOpt.isEmpty()) {
            return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(hseflFlwMcksKbn, null);
        }
        HseflRecord targetRecord = targetRecordOpt.orElseThrow();
        if (followStatusCompletedFollowCompleted.equals(targetRecord.flwStCd())) {
            if (Objects.equals(targetRecord.fltkYmd(), systemDate)) {
                return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(
                        hseflFlwMcksKbn,
                        toHseflInf(targetRecord));
            }
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000041);
        }
        return new ClosingCancelFeasibilityJudgmentFollowStatusOutput(
                hseflFlwMcksKbn,
                toHseflInf(targetRecord));
    }

    /**
     * 概要: HSEFL レコードを API 出力モデルへ変換する。
        * 主な仕様:
        * - SV の戻り値設計で定義された 20 項目へ HSEFL レコードを写像する。
        * - 下位 DA レコードをそのまま公開せず API モデルへ詰め替える。
        * 既知の制約:
        * - SV-096 の出力設計で未使用の HSEFL 項目は写像対象外とする。
     *
     * @param record HSEFL レコード
     * @return API 出力モデル
     */
    private HseflInf toHseflInf(HseflRecord record) {
        return new HseflInf(
                record.flttCktkCd(),
                record.gasMtstNo(),
                record.hssgYmd(),
                record.hesnHukkYmd(),
                toApiRegTsno(record.regTsno()),
                record.sgyAnknNo(),
                record.syoKyoTrno(),
                record.hesnJysb(),
                record.hssgNwkCd(),
                record.sgytSinNo(),
                record.sgytSinnKj(),
                record.hsnjGmsSu(),
                record.hsnjGasSyry(),
                record.sjskKbn(),
                record.rshhSjsb(),
                record.hkshHshhShbt(),
                record.utidxNo(),
                record.sihKiyNo(),
                record.svkyNo(),
                record.recUpdJnjNo());
    }

    private Long toApiRegTsno(Integer regTsno) {
        return regTsno == null ? null : regTsno.longValue();
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
        * 主な仕様:
        * - `hukkSysShbt` と `systemDatetime` の存在を fail-fast で検証する。
        * - 戻り値として検証済みのシステム日時を返却する。
        * 既知の制約:
        * - 未設定時は項目名付きの IllegalStateException を送出する。
     *
     * @return システム日時
     */
    private LocalDateTime requireRequestContext() {
        requireHukkSysShbt();
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: RequestContext の報告システム種別を取得する。
        * 主な仕様:
        * - `RequestContext.hukkSysShbt` を取得する。
        * - 未設定時は fail-fast で異常終了する。
        * 既知の制約:
        * - null や空値の補完は行わない。
     *
     * @return 報告システム種別
     */
    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

}
