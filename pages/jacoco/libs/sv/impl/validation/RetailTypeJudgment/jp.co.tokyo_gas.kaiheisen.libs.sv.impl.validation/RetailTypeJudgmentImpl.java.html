package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;

/**
 * 小売種別判定サービス実装.
 */
public final class RetailTypeJudgmentImpl implements RetailTypeJudgment {

    /** DA-099-R005 の TKSSVKY 参照 CallSpec. */
    private static final DataAccessCallSpec FIND_TKSSVKY_BY_DATE_STATUS =
            TkssvkyDataAccess.CallSpecs
                    .FIND_BY_TKSU_SVKY_NO_AND_TKYS_YMD_AND_TKSU_SVKY_JTKB;

    /** 処理区分: 適用中. */
    private static final String SHR_KBN_ACTIVE = "1";

    /** 処理区分: 直近. */
    private static final String SHR_KBN_NEAREST = "2";

    /** 処理区分: 適用中または終了. */
    private static final String SHR_KBN_ACTIVE_OR_END = "3";

    /** TSKGMST データアクセス. */
    private final TskgmstDataAccess tskgmstDataAccess;

    /** TKSSVKY データアクセス. */
    private final TkssvkyDataAccess tkssvkyDataAccess;

    /** TSKGMST の適用中状態区分. */
    private final String tksSvkkActive;

    /** TSKGMST の終了状態区分. */
    private final String tksSvkkEnd;

    /** TSKGMST の無効状態区分. */
    private final String tksSvkkInvalid;

    /** TKSSVKY の契約適用中状態区分. */
    private final String tksSvkyActive;

    /** TKSSVKY の契約終了状態区分. */
    private final String tksSvkyEnd;

    /** TKSSVKY の無効状態区分. */
    private final String tksSvkyInvalid;

    /** 東京ガスの小売事業者コード. */
    private final String tokyoGasSupplierCode;

    /** TG 小売の判定コード. */
    private final String kuHntKbnTokyoGas;

    /** 他小売の判定コード. */
    private final String kuHntKbnOther;

    /** 対象データなしの判定コード. */
    private final String kuHntKbnNoData;

    /** TSKGMST の適用中または終了状態一覧. */
    private final List<String> activeOrEndSvkk;

    /** TKSSVKY の適用中または終了状態一覧. */
    private final List<String> activeOrEndSvky;

    /**
     * コンストラクタ.
     *
     * @param tskgmstDataAccessParam 託送サービス契約・ガスメーター設置場所関連
     *     データアクセス
     * @param tkssvkyDataAccessParam 託送サービス契約データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public RetailTypeJudgmentImpl(
            final TskgmstDataAccess tskgmstDataAccessParam,
            final TkssvkyDataAccess tkssvkyDataAccessParam,
            final CiriusConstantService ciriusConstantService) {
        this.tskgmstDataAccess =
                Objects.requireNonNull(tskgmstDataAccessParam, "tskgmstDataAccess");
        this.tkssvkyDataAccess =
                Objects.requireNonNull(tkssvkyDataAccessParam, "tkssvkyDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        String contractRelatedStatusType =
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE;
        String contractStatusType =
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE;
        String contractRelatedStatusApplying =
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING;
        String contractRelatedStatusApplicableEnds =
                ConstantLookupKeys.Cirius
                        .TransportationServiceContractRelatedStatus
                        .APPLICABLE_ENDS;
        String contractRelatedStatusInvalid =
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.INVALID;
        String contractStatusUnderContract =
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT;
        String contractStatusEndOfContract =
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.END_OF_CONTRACT;
        String contractStatusInvalid =
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.INVALID;

        this.tksSvkkActive =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractRelatedStatusType,
                        contractRelatedStatusApplying);
        this.tksSvkkEnd =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractRelatedStatusType,
                        contractRelatedStatusApplicableEnds);
        this.tksSvkkInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractRelatedStatusType,
                        contractRelatedStatusInvalid);
        this.tksSvkyActive =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractStatusType,
                        contractStatusUnderContract);
        this.tksSvkyEnd =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractStatusType,
                        contractStatusEndOfContract);
        this.tksSvkyInvalid =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        contractStatusType,
                        contractStatusInvalid);
        this.tokyoGasSupplierCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);

        this.kuHntKbnTokyoGas = RetailTypeJudgmentCode.TOKYO_GAS.code();
        this.kuHntKbnOther = RetailTypeJudgmentCode.OTHER.code();
        this.kuHntKbnNoData = RetailTypeJudgmentCode.NO_DATA.code();

        this.activeOrEndSvkk = List.of(tksSvkkActive, tksSvkkEnd);
        this.activeOrEndSvky = List.of(tksSvkyActive, tksSvkyEnd);
    }

    /**
     * 小売種別の判定を実行する.
     *
     * @param shrKbn 処理区分
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return 小売判定結果
     */
    @Override
    public RetailTypeJudgmentOutput execute(
            final String shrKbn,
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        return switch (shrKbn) {
            case SHR_KBN_ACTIVE -> handleActive(gasMtstNo, knskKjnYmd);
            case SHR_KBN_NEAREST -> handleNearest(gasMtstNo, knskKjnYmd);
            case SHR_KBN_ACTIVE_OR_END -> handleActiveOrEnd(gasMtstNo);
            default ->
                    throw new KaiheisenException(
                            "処理区分の値が不正のため、処理を終了しました。shrKbn=" + shrKbn);
        };
    }

    /**
     * 処理区分1（適用中）で小売判定情報を取得する.
     *
     * <p>TSKGMST を適用中ステータスで検索し、0件の場合は対象データなしを返却する。
     * 取得した最初の TSKGMST をキーに TKSSVKY を適用中ステータスで検索する。
     * TKSSVKY が0件の場合は DA-099-R001 を識別子に
     * DataInconsistencyException を送出する。</p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param baseDate 検索基準日
     * @return 小売判定結果
     */
    private RetailTypeJudgmentOutput handleActive(
            final Long gasMtstNo,
            final LocalDate baseDate) {
        List<TskgmstRecord> tskgmstRecords =
                tskgmstDataAccess.findAllByGasMtstNoAndTkysYmdAndTyenYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        baseDate,
                        tksSvkkActive);
        if (tskgmstRecords.isEmpty()) {
            return noData();
        }
        TskgmstRecord tskgmstRecord = tskgmstRecords.getFirst();

        List<TkssvkyRecord> tkssvkyRecords =
                tkssvkyDataAccess.findAllByTksuSvkyNoAndTksuSvkyJtkb(
                        tskgmstRecord.tksuSvkyNo(),
                        tksSvkyActive);
        if (tkssvkyRecords.isEmpty()) {
            throw dataInconsistency(
                    TkssvkyDataAccess.CallSpecs.FIND_ALL_BY_TKSU_SVKY_NO_AND_TKSU_SVKY_JTKB,
                    new FlatJsonBuilder()
                            .put("tksuSvkyNo", tskgmstRecord.tksuSvkyNo())
                            .put("tksuSvkyJtkb", tksSvkyActive));
        }
        TkssvkyRecord tkssvkyRecord = tkssvkyRecords.getFirst();
        return toOutput(tskgmstRecord, tkssvkyRecord);
    }

    /**
     * 処理区分2（直近）で小売判定情報を取得する.
     *
     * <p>TSKGMST を無効ステータスで検索し、未取得の場合は対象データなしを返却する。
     * 取得した TSKGMST をキーに TKSSVKY を基準日付き無効ステータスで検索する。
     * TKSSVKY が0件の場合は DA-099-R005 を識別子に
     * DataInconsistencyException を送出する。</p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param baseDate 検索基準日
     * @return 小売判定結果
     */
    private RetailTypeJudgmentOutput handleNearest(
            final Long gasMtstNo,
            final LocalDate baseDate) {
        Optional<TskgmstRecord> tskgmstRecord =
                tskgmstDataAccess.findByGasMtstNoAndTkysYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        baseDate,
                        tksSvkkInvalid);
        if (tskgmstRecord.isEmpty()) {
            return noData();
        }
        TskgmstRecord resolvedTskgmstRecord = tskgmstRecord.orElseThrow();

        Optional<TkssvkyRecord> tkssvkyRecord =
                tkssvkyDataAccess.findByTksuSvkyNoAndTkysYmdAndTksuSvkyJtkb(
                        resolvedTskgmstRecord.tksuSvkyNo(),
                        baseDate,
                        tksSvkyInvalid);
        if (tkssvkyRecord.isEmpty()) {
            throw dataInconsistency(
                    FIND_TKSSVKY_BY_DATE_STATUS,
                    new FlatJsonBuilder()
                            .put("tksuSvkyNo", resolvedTskgmstRecord.tksuSvkyNo())
                            .put("tkysYmd", baseDate)
                            .put("tksuSvkyJtkb", tksSvkyInvalid));
        }
        return toOutput(resolvedTskgmstRecord, tkssvkyRecord.orElseThrow());
    }

    /**
     * 処理区分3（適用中または終了）で小売判定情報を取得する.
     *
     * <p>TSKGMST を適用中または終了ステータス一覧で検索し、未取得の場合は
     * 対象データなしを返却する。取得した TSKGMST をキーに TKSSVKY を
     * 適用中または終了ステータス一覧で検索する。TKSSVKY が0件の場合は
     * DA-099-R003 を識別子に DataInconsistencyException を送出する。</p>
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 小売判定結果
     */
    private RetailTypeJudgmentOutput handleActiveOrEnd(final Long gasMtstNo) {
        Optional<TskgmstRecord> tskgmstRecord =
                tskgmstDataAccess.findByGasMtstNoAndTksuSvkkJtkbList(
                        gasMtstNo,
                        activeOrEndSvkk);
        if (tskgmstRecord.isEmpty()) {
            return noData();
        }
        TskgmstRecord resolvedTskgmstRecord = tskgmstRecord.orElseThrow();

        Optional<TkssvkyRecord> tkssvkyRecord =
                tkssvkyDataAccess.findByTksuSvkyNoAndTksuSvkyJtkbList(
                        resolvedTskgmstRecord.tksuSvkyNo(),
                        activeOrEndSvky);
        if (tkssvkyRecord.isEmpty()) {
            throw dataInconsistency(
                    TkssvkyDataAccess.CallSpecs.FIND_BY_TKSU_SVKY_NO_AND_TKSU_SVKY_JTKB_LIST,
                    new FlatJsonBuilder()
                            .put("tksuSvkyNo", resolvedTskgmstRecord.tksuSvkyNo())
                            .put("tksuSvkyJtkbList", activeOrEndSvky));
        }
        return toOutput(resolvedTskgmstRecord, tkssvkyRecord.orElseThrow());
    }

    /**
     * 対象データなしの判定結果を生成する.
     *
     * @return 対象データなしの小売判定結果
     */
    private RetailTypeJudgmentOutput noData() {
        return new RetailTypeJudgmentOutput(
                kuHntKbnNoData,
                noDataTskgmstInfo(),
                noDataTkssvkyInfo());
    }

    private static TskgmstInfo noDataTskgmstInfo() {
        return new TskgmstInfo(null, null, null, null, null, null, null);
    }

    private static TkssvkyInfo noDataTkssvkyInfo() {
        return new TkssvkyInfo(
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    /**
     * TSKGMST と TKSSVKY から小売判定の出力 DTO を構築する.
     *
     * @param tskgmstRecord TSKGMST 取得結果
     * @param tkssvkyRecord TKSSVKY 取得結果
     * @return 小売判定結果
     */
    private RetailTypeJudgmentOutput toOutput(
            final TskgmstRecord tskgmstRecord,
            final TkssvkyRecord tkssvkyRecord) {
        String kuHntKbn =
                tokyoGasSupplierCode.equals(tkssvkyRecord.tksuKijsCd())
                        ? kuHntKbnTokyoGas
                        : kuHntKbnOther;
        return new RetailTypeJudgmentOutput(
                kuHntKbn,
                toTskgmstInfo(tskgmstRecord),
                toTkssvkyInfo(tkssvkyRecord));
    }

    /**
     * TSKGMST の DA record を API DTO に変換する.
     *
     * @param record TSKGMST の DA record
     * @return TSKGMST 情報 DTO
     */
    private static TskgmstInfo toTskgmstInfo(final TskgmstRecord record) {
        return new TskgmstInfo(
                record.gasMtstNo(),
                record.tksuSvkyNo(),
                record.skno(),
                record.tksuSvkkJtkb(),
                record.tkysYmd(),
                record.tyenYmd(),
                record.recUpdJnjNo());
    }

    /**
     * TKSSVKY の DA record を API DTO に変換する.
     *
     * @param record TKSSVKY の DA record
     * @return TKSSVKY 情報 DTO
     */
    private static TkssvkyInfo toTkssvkyInfo(final TkssvkyRecord record) {
        return new TkssvkyInfo(
                record.tksuSvkyNo(),
                record.skno(),
                record.tksuSvkyJtkb(),
                record.tksuSvksJysb(),
                record.tksuSvkeJysb(),
                record.tkysYmd(),
                record.tyenYmd(),
                record.hesnYtiYmd(),
                record.tkykShbt(),
                record.tksuKijsCd(),
                record.sutiKskhFlg(),
                record.syoKyoTrno(),
                record.recUpdJnjNo());
    }

    /**
     * DA-099 の返却件数不整合を表す例外を生成する.
     *
     * @param callSpec 不整合判定元の DA CallSpec
     * @param parameterBuilder DA 呼出しパラメータを構築した JSON builder
     * @return DataInconsistencyException インスタンス
     */
    private DataInconsistencyException dataInconsistency(
            final DataAccessCallSpec callSpec,
            final FlatJsonBuilder parameterBuilder) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                callSpec.identifier(),
                parameterBuilder);
    }
