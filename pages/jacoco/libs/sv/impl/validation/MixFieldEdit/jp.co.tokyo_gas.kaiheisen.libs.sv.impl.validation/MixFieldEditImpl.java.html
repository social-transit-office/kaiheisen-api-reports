package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.MixFieldEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.MixFieldEditOutput;

/**
 * 概要: SV-017（ミクス項目編集）として、お客さま名と建物名の返却値を漢字またはカナへ編集するサービスの実装。
 * 主な仕様:
 * - おなまえ漢字表示可否フラグが true の場合はお客さま名漢字1とお客さま名漢字2を優先し、各漢字値が未設定または空文字なら対応するカナへ切り替える。
 * - 建物漢字表示可否フラグが true の場合は建物名漢字を優先し、建物名漢字が未設定または空文字なら建物名カナへ切り替える。
 * - 各表示可否フラグが false の場合は対応するカナ値を返却する。
 * 既知の制約:
 * - 各表示可否フラグが未設定の場合は対応する出力項目を未設定のまま返却する。
 * - リクエストコンテキストや下位サービス、下位データアクセスには依存せず、このクラス単体で入力値のみを判定する。
 */
public class MixFieldEditImpl implements MixFieldEdit {

    /**
     * 概要: 表示可否フラグと名称入力値に基づいて、お客さま名1/2と建物名の返却値を組み立てる。
     * 主な仕様:
     * - おなまえ漢字表示可否フラグが true の場合はお客さま名漢字1とお客さま名漢字2を優先し、未設定時はお客さま名カナ1とお客さま名カナ2を採用する。
     * - 建物漢字表示可否フラグが true の場合は建物名漢字を優先し、未設定時は建物名カナを採用する。
     * - 各表示可否フラグが false の場合は対応するカナ値を返し、未設定の場合は対応する出力項目を設定しない。
     * 既知の制約:
     * - 漢字値の未設定判定は未設定または空文字のみを対象とし、空白のみの文字列は値ありとして扱う。
     * - 入力値の正規化、trim、文字種妥当性判定は行わない。
     */
    @Override
    public MixFieldEditOutput execute(
            Boolean onmeKjhkFlg,
            String ctnmKj1,
            String ctnmKj2,
            String ctnmKn1,
            String ctnmKn2,
            Boolean bldKjhkFlg,
            String bldnKj,
            String bldnKn) {
        String ctnm1 = null;
        String ctnm2 = null;
        String bldName = null;

        if (Boolean.TRUE.equals(onmeKjhkFlg)) {
            ctnm1 = hasText(ctnmKj1) ? ctnmKj1 : ctnmKn1;
            ctnm2 = hasText(ctnmKj2) ? ctnmKj2 : ctnmKn2;
        } else if (Boolean.FALSE.equals(onmeKjhkFlg)) {
            ctnm1 = ctnmKn1;
            ctnm2 = ctnmKn2;
        }

        if (Boolean.TRUE.equals(bldKjhkFlg)) {
            bldName = hasText(bldnKj) ? bldnKj : bldnKn;
        } else if (Boolean.FALSE.equals(bldKjhkFlg)) {
            bldName = bldnKn;
        }

        return new MixFieldEditOutput(ctnm1, ctnm2, bldName);
    }

    /**
     * 概要: 文字列を漢字候補として採用できるかを判定する。
     * 主な仕様:
     * - 未設定ではなく、かつ空文字でない場合に true を返す。
     * - 実行処理から呼ばれ、漢字値を採用するかカナへ切り替えるかの判定に利用する。
     * 既知の制約:
     * - 前後空白の trim や blank 判定は行わない。
     * - 文字種判定は行わず、非空であれば内容を問わず有効とみなす。
     *
     * @param value 判定対象の文字列
     * @return 漢字候補として有効な場合は true
     */
    private static boolean hasText(String value) {
        return value != null && !value.isEmpty();
    }
}
