package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageMultiplierCategoryAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageMultiplierCategoryAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageJudgmentOutput;

/**
 * 概要: 変動使用量判定サービス実装。
 * 主な仕様:
 * - SV-198 から取得した倍率分類を用いて、前回使用量および前年同月使用量との比較で変動使用量エラー有無を判定する。
 * 既知の制約:
 * - エラー理由コード自体は返却せず、Boolean フラグのみを出力する。
 */
public final class VariableUsageJudgmentImpl implements VariableUsageJudgment {

    private static final VariableUsageJudgmentOutput TRUE_OUTPUT =
            new VariableUsageJudgmentOutput(Boolean.TRUE);

    private static final VariableUsageJudgmentOutput FALSE_OUTPUT =
            new VariableUsageJudgmentOutput(Boolean.FALSE);

    private final UsageMultiplierCategoryAcquisition usageMultiplierCategoryAcquisition;

    /**
     * 概要: 変動使用量判定サービス実装を生成する。
     *
     * @param usageMultiplierCategoryAcquisition 使用量倍率分類取得サービス
     */
    public VariableUsageJudgmentImpl(UsageMultiplierCategoryAcquisition usageMultiplierCategoryAcquisition) {
        this.usageMultiplierCategoryAcquisition =
                Objects.requireNonNull(
                        usageMultiplierCategoryAcquisition,
                        "usageMultiplierCategoryAcquisition");
    }

    /**
     * 概要: 変動使用量エラー有無を判定する。
     * 主な仕様:
     * - 前回使用日数が 0 の場合は判定処理を行わず false を返す。
     * - 下限倍率条件と上限倍率条件のいずれかでエラー条件が成立した場合に true を返す。
     *
     * @param konSyry 今回使用量
     * @param znkiSyry 前回使用量
     * @param zndmSyry 前年同月使用量
     * @param znkiShyDdsu 前回使用日数
     * @param zndmShyDdsu 前年同月使用日数
     * @param knsYmd 検針年月日
     * @return 判定結果
     */
    @Override
    public VariableUsageJudgmentOutput execute(
            BigDecimal konSyry,
            BigDecimal znkiSyry,
            BigDecimal zndmSyry,
            Integer znkiShyDdsu,
            Integer zndmShyDdsu,
            LocalDate knsYmd) {
        if (!isNonZero(znkiShyDdsu)) {
            return FALSE_OUTPUT;
        }

        UsageMultiplierCategoryAcquisitionOutput multiplierOutput =
            usageMultiplierCategoryAcquisition.execute(knsYmd, konSyry);

        if (isLessThanOrEqual(multiply(konSyry, multiplierOutput.tiZnkiKgnBirt()), znkiSyry)) {
            return judgePreviousYearPath(
                konSyry,
                zndmSyry,
                zndmShyDdsu,
                multiplierOutput.tiZndmKgnBirt(),
                multiplierOutput.tiZndmJgnBirt());
        }

        if (!isMinusOne(multiplierOutput.tiZnkiJgnBirt())
                && isLessThanOrEqual(
                        multiply(znkiSyry, multiplierOutput.tiZnkiJgnBirt()),
                        konSyry)) {
            return judgePreviousYearPath(
                konSyry,
                zndmSyry,
                zndmShyDdsu,
                multiplierOutput.tiZndmKgnBirt(),
                multiplierOutput.tiZndmJgnBirt());
        }

        return FALSE_OUTPUT;
    }

    /**
     * 概要: 前年同月側のエラー判定を共通化。
     *
     * @param konSyry 今回使用量
     * @param zndmSyry 前年同月使用量
     * @param zndmShyDdsu 前年同月使用日数
     * @param tiZndmKgnBirt 対前年同月下限倍率
     * @param tiZndmJgnBirt 対前年同月上限倍率
     * @return 判定結果
     */
    private VariableUsageJudgmentOutput judgePreviousYearPath(
            BigDecimal konSyry,
            BigDecimal zndmSyry,
            Integer zndmShyDdsu,
            Integer tiZndmKgnBirt,
            Integer tiZndmJgnBirt) {
        if (isZero(zndmShyDdsu)) {
            return TRUE_OUTPUT;
        }
        if (isLessThanOrEqual(multiply(konSyry, tiZndmKgnBirt), zndmSyry)) {
            return TRUE_OUTPUT;
        }
        if (isLessThanOrEqual(multiply(zndmSyry, tiZndmJgnBirt), konSyry)) {
            return TRUE_OUTPUT;
        }
        return FALSE_OUTPUT;
    }

    /**
     * 概要: 使用日数が 0 であることを判定する。
     *
     * @param value 使用日数
     * @return 0 の場合 true
     */
    private boolean isZero(Integer value) {
        return value.intValue() == 0;
    }

    /**
     * 概要: 使用日数が 0 以外であることを判定する。
     *
     * @param value 使用日数
     * @return 0 以外の場合 true
     */
    private boolean isNonZero(Integer value) {
        return value.intValue() != 0;
    }

    /**
     * 概要: 倍率が -1 であることを判定する。
     *
     * @param multiplier 倍率
     * @return -1 の場合 true
     */
    private boolean isMinusOne(Integer multiplier) {
        return multiplier.intValue() == -1;
    }

    /**
     * 概要: 使用量と倍率を乗算する。
     *
     * @param value 使用量
     * @param multiplier 倍率
     * @return 乗算結果
     */
    private BigDecimal multiply(BigDecimal value, Integer multiplier) {
        return value.multiply(BigDecimal.valueOf(multiplier.longValue()));
    }

    /**
     * 概要: 左辺が右辺以下かを判定する。
     *
     * @param left 左辺
     * @param right 右辺
     * @return 左辺が右辺以下の場合 true
     */
    private boolean isLessThanOrEqual(BigDecimal left, BigDecimal right) {
        return left.compareTo(right) <= 0;
    }
}
