package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.TransportContractRelatedErrorJudgment;

/** 託送契約関連エラー判定サービス実装。 */
public final class TransportContractRelatedErrorJudgmentImpl
        implements TransportContractRelatedErrorJudgment {

    /** 仮解約の託送契約状態コード。 */
    private final String provisionalCancellationStatus;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public TransportContractRelatedErrorJudgmentImpl(
            CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.provisionalCancellationStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                        ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.PROVISIONAL_CANCELLATION);
    }

    /**
     * 託送契約関連エラー条件を判定し、該当時は閉栓共通エラーを送出する。
     *
     * <p>SV レイヤ規約に従い、public 引数の Null 許可/不許可は
     * 本メソッド実装で独自判断せず、null ガードは行わない。</p>
     *
     * @param taksmtrUmFlg 託送メータ情報有無フラグ
     * @param tkykJtCd 託送契約状態コード
     */
    @Override
    public void execute(Boolean taksmtrUmFlg, String tkykJtCd) {
        if (!Boolean.TRUE.equals(taksmtrUmFlg)) {
            throw heisenRule();
        }

        if (!provisionalCancellationStatus.equals(tkykJtCd)) {
            throw heisenRule();
        }
    }

    private static HeisenRuleException heisenRule() {
        return new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200034);
    }
}
