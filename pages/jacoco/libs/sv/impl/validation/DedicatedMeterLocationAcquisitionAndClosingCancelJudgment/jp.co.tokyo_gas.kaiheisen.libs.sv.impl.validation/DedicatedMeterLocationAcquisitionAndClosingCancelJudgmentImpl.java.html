package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.DedicatedMeterLocationAcquisitionAndClosingCancelJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentGmtsetdInf;

/**
 * SV-102 閉栓取消時元メーター情報取得および開閉栓状態チェックの実装クラス。
 */
public final class DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentImpl
        implements DedicatedMeterLocationAcquisitionAndClosingCancelJudgment {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final String dedicatedGasMeterStatus;

    private final String openingValveStatus;

    /**
     * 判定に必要な依存性と CIRIUS 定数を解決して初期化する。
     *
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DataAccess
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentImpl(
            final GmtsetdDataAccess gmtsetdDataAccess,
            final CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.openingValveStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 元メーターの開閉栓状態を取得し、閉栓取消対象かを判定する。
     *
     * @param trkKnsYmd 取消対象検針年月日
     * @param gmtsetdInf ガスメーター設置場所＜導管＞情報
     */
    @Override
    public void execute(
            final LocalDate trkKnsYmd,
            final DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentGmtsetdInf gmtsetdInf) {
        if (!dedicatedGasMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            return;
        }

        GmtsetdRecord originalMeter =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(gmtsetdInf.mtgMtstNo(), trkKnsYmd)
                        .orElseThrow(
                                () ->
                                        DataInconsistencySupport.create(
                                                gmtsetdDataAccess.tableName(),
                                                GmtsetdDataAccess.CallSpecs
                                                        .FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD
                                                        .identifier(),
                                                new FlatJsonBuilder()
                                                        .put("gasMtstNo", gmtsetdInf.mtgMtstNo())
                                                        .put("knskKjnYmd", trkKnsYmd)));

        if (!openingValveStatus.equals(originalMeter.gmtKhsJtkb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000053);
        }
    }
}
