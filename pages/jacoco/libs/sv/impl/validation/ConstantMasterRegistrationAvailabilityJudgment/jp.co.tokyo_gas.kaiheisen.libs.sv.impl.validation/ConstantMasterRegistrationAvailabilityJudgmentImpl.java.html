package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ConstantMasterInfoAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ConstantMasterRegistrationAvailabilityJudgment;

/**
 * 概要: SV-273（コンスタントマスタ登録利用可否判定）の実装クラス。
 * 主な仕様:
 * - SV-089 を呼び出してコンスタントマスタ情報を取得する。
 * - 取得した登録時利用可否フラグが false の場合のみ業務エラー 020000016 を送出する。
 * - 上記条件に一致しない場合は何も返さず正常終了する。
 * 既知の制約:
 * - correlationId、hukkSysShbt、systemDatetime は SV 規約に従い公開引数へ含めない。
 * - public 引数の null ガードや前後空白補正は行わず、上位バリデーション済み入力を前提とする。
 */
public final class ConstantMasterRegistrationAvailabilityJudgmentImpl
        implements ConstantMasterRegistrationAvailabilityJudgment {

    private final ConstantMasterInfoAcquisition constantMasterInfoAcquisition;

    /**
     * 概要: 判定に必要な下位 SV を保持してインスタンスを初期化する。
     * 主な仕様:
     * - SV-089 コンスタントマスタ情報取得を保持する。
     * - 必須依存が不足している場合はコンストラクタで fail-fast する。
     *
     * @param constantMasterInfoAcquisition SV-089 コンスタントマスタ情報取得
     */
    public ConstantMasterRegistrationAvailabilityJudgmentImpl(
            final ConstantMasterInfoAcquisition constantMasterInfoAcquisition) {
        this.constantMasterInfoAcquisition = Objects.requireNonNull(
                constantMasterInfoAcquisition,
                "constantMasterInfoAcquisition");
    }

    /**
     * 概要: コンスタントマスタ登録利用可否判定を実行する。
     * 主な仕様:
     * - SV-089 の取得結果に含まれる登録時利用可否フラグが false の場合は業務エラー 020000016 を送出する。
     * - 埋め込み文字列 {0} には入力のコンスタント種別コードを設定する。
     * - 登録時利用可否フラグが true または null の場合は正常終了する。
     *
     * @param cnstShbtCd コンスタント種別コード
     * @param cnstCd コンスタントコード
     */
    @Override
    public void execute(final String cnstShbtCd, final String cnstCd) {
        ConstantMasterInfoAcquisitionOutput output =
                constantMasterInfoAcquisition.execute(cnstShbtCd, cnstCd);

        if (Boolean.FALSE.equals(output.trkjRykhFlg())) {
            throw new BusinessRuleException(
                    ErrorCodeConstants.ERROR_CODE_020000016,
                    List.of(cnstShbtCd));
        }
    }
}
