package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.List;

/**
 * 概要: SV-046 開栓取消実施可否判定: 支払契約サービスの実装です。
 *
 * 主な仕様:
 * サービス契約番号から適用中のサービス契約・支払契約関連と支払契約を取得し、
 * 請求サービス種別が設定されている場合は業務エラーを返します。
 * 請求サービス種別が未設定の場合は取消対象検針年月日の支払契約を取得し、
 * 戻り値へ支払契約情報と取消時点の適用開始年月日を設定します。
 *
 * 既知の制約:
 * DA-044-R002 が複数件返る場合は source TSV の記載に従い先頭 1 件を採用します。
 */
public final class OpeningCancelFeasibilityJudgmentPaymentContractImpl
        implements OpeningCancelFeasibilityJudgmentPaymentContract {

    /** サービス契約・支払契約関連を参照するデータアクセスです。 */
    private final SvsihkrDataAccess svsihkrDataAccess;

    /** 支払契約を参照するデータアクセスです。 */
    private final SihakyDataAccess sihakyDataAccess;

    /** 支払契約状態区分の「適用中」を表す CIRIUS 定数値です。 */
    private final String applyingStatus;

    /** 支払契約状態区分の「無効」を表す CIRIUS 定数値です。 */
    private final String invalidStatus;

    /**
     * 概要: 依存コンポーネントを受け取ってサービス実装を初期化します。
     *
     * 主な仕様:
     * サービス契約・支払契約関連 DA、支払契約 DA、CIRIUS 定数サービスは必須依存とし、
     * 生成時に null を禁止します。
     * 支払契約状態区分の必須定数は初期化時点で解決し、未登録なら fail-fast します。
     *
     * 既知の制約:
     * @param svsihkrDataAccessParam サービス契約・支払契約関連 DA
     * @param sihakyDataAccessParam 支払契約 DA
     * @param ciriusConstantServiceParam CIRIUS 定数サービス
     */
    public OpeningCancelFeasibilityJudgmentPaymentContractImpl(
            final SvsihkrDataAccess svsihkrDataAccessParam,
            final SihakyDataAccess sihakyDataAccessParam,
            final CiriusConstantService ciriusConstantServiceParam) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccessParam, "svsihkrDataAccess");
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccessParam, "sihakyDataAccess");
        CiriusConstantService ciriusConstantService =
                Objects.requireNonNull(ciriusConstantServiceParam, "ciriusConstantService");
        this.applyingStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.invalidStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
    }

    /**
     * 概要: 開栓取消可否を支払契約観点で判定します。
     *
     * 主な仕様:
     * 適用中のサービス契約・支払契約関連と支払契約を順に取得し、
     * 請求サービス種別が設定されている場合は業務エラー 020000049 を送出します。
     * 請求サービス種別が未設定の場合は取消対象検針年月日の支払契約を取得し、
     * 支払契約情報と取消時点の適用開始年月日を戻り値へ設定します。
     *
     * 既知の制約:
     * 入力値の null / blank 補正や妥当性検証は行わず、呼び出し元から渡された値をそのまま判定します。
     *
     * @param svkyNo サービス契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 支払契約観点の判定結果
     */
    @Override
    public OpeningCancelFeasibilityJudgmentPaymentContractOutput execute(
            final Long svkyNo,
            final LocalDate trkKnsYmd) {
        SvsihkrRecord svsihkrRecord = requireServicePaymentRelation(svkyNo);
        SihakyRecord sihakyRecord = requireActivePaymentContract(svsihkrRecord.sihKiyNo());

        if (hasBillingServiceType(sihakyRecord.sikSvcShbt())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000049);
        }

        SihakyRecord canceledPointRecord = requirePaymentContractAtCancelPoint(
                svsihkrRecord.sihKiyNo(),
                trkKnsYmd);

        return new OpeningCancelFeasibilityJudgmentPaymentContractOutput(
                mapSihaky(sihakyRecord),
                mapSvsihkr(svsihkrRecord),
                new SihakyKaTrksInf(canceledPointRecord.tkysYmd()));
    }

    /**
     * 概要: サービス契約番号に対応する適用中のサービス契約・支払契約関連を取得します。
     *
     * 主な仕様:
     * DA-044-R002 を呼び出し、取得結果が空の場合はデータ不整合例外を送出します。
     * 複数件取得時は先頭レコードを採用します。
     *
     * 既知の制約:
     * 先頭 1 件の採用順は DA の返却順に依存し、業務上の優先順位はこのメソッドでは補正しません。
     *
     * @param svkyNo サービス契約番号
     * @return 適用中のサービス契約・支払契約関連レコード
     */
    private SvsihkrRecord requireServicePaymentRelation(final Long svkyNo) {
        List<SvsihkrRecord> records = svsihkrDataAccess.findAllBySvkyNoAndSihKiyJtkb(svkyNo, applyingStatus);
        if (records.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svsihkrDataAccess.tableName(),
                    SvsihkrDataAccess.CallSpecs.FIND_ALL_BY_SVKY_NO_AND_SIH_KIY_JTKB.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svkyNo)
                            .put("sihKiyJtkb", applyingStatus));
        }
        return records.get(0);
    }

    /**
     * 概要: 支払契約番号に対応する適用中の支払契約を取得します。
     *
     * 主な仕様:
     * DA-075-R001 を呼び出し、取得できない場合はデータ不整合例外を送出します。
     *
     * 既知の制約:
     * 取得条件は適用中の固定値のみで、呼出元から状態区分を変更することはできません。
     *
     * @param sihKiyNo 支払契約番号
     * @return 適用中の支払契約レコード
     */
    private SihakyRecord requireActivePaymentContract(final Long sihKiyNo) {
        return sihakyDataAccess.findBySihKiyNoAndSihKiyJtkb(sihKiyNo, applyingStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sihakyDataAccess.tableName(),
                        SihakyDataAccess.CallSpecs.FIND_BY_SIH_KIY_NO_AND_SIH_KIY_JTKB.identifier(),
                        new FlatJsonBuilder()
                                .put("sihKiyNo", sihKiyNo)
                                .put("sihKiyJtkb", applyingStatus)));
    }

    /**
     * 概要: 取消対象検針年月日を基準に有効な支払契約を取得します。
     *
     * 主な仕様:
     * DA-075-R006 を呼び出し、適用開始年月日降順・世代管理番号降順で選ばれた
     * 1 件を取得します。取得できない場合はデータ不整合例外を送出します。
     *
     * 既知の制約:
     * 現行の戻り値利用は適用開始年月日のみで、他項目はこのサービスでは参照しません。
     *
     * @param sihKiyNo 支払契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 取消対象検針年月日に対応する支払契約レコード
     */
    private SihakyRecord requirePaymentContractAtCancelPoint(
            final Long sihKiyNo,
            final LocalDate trkKnsYmd) {
        return sihakyDataAccess
                .findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDescAndSknoDesc(
                        sihKiyNo,
                        trkKnsYmd,
                        invalidStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sihakyDataAccess.tableName(),
                        SihakyDataAccess.CallSpecs
                                .FIND_BY_SIH_KIY_NO_AND_KNSK_KJN_YMD_AND_NOT_SIH_KIY_JTKB_ORDER_BY_TKYS_YMD_DESC_AND_SKNO_DESC
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("sihKiyNo", sihKiyNo)
                                .put("knskKjnYmd", trkKnsYmd)
                                .put("sihKiyJtkb", invalidStatus)));
    }

    /**
     * 概要: 請求サービス種別が設定済みかどうかを判定します。
     *
     * 主な仕様:
     * null でも空文字でもない値を「設定あり」とみなします。
     *
     * 既知の制約:
     * source TSV の条件記述は曖昧なため、この実装では best-effort として空白文字列も未設定扱いにしています。
     *
     * @param billingServiceType 請求サービス種別
     * @return 請求サービス種別が設定されている場合は true
     */
    private boolean hasBillingServiceType(final String billingServiceType) {
        return billingServiceType != null && !billingServiceType.isBlank();
    }

    /**
     * 概要: 支払契約レコードを出力モデルへ変換します。
     *
     * 主な仕様:
     * SV-046 の output.tsv で定義された支払契約情報の項目へ必要な値を写像します。
     *
     * 既知の制約:
     * DA レコードに存在する監査系項目のうち、SV-046 出力で未定義の項目は変換対象に含めません。
     *
     * @param record 支払契約レコード
     * @return 支払契約情報モデル
     */
    private SihakyInf mapSihaky(final SihakyRecord record) {
        return new SihakyInf(
                record.sihKiyNo(),
                record.skno(),
                record.tkysYmd(),
                record.tyenYmd(),
                record.sihKiyJtkb(),
                record.sihHuhuShbt(),
                record.sikSvcShbt(),
                record.sikKibNo(),
                record.yggsTkyShbt(),
                record.thsCdNo(),
                record.sikCycsNo(),
                record.sikKrno(),
                record.shtiSqddKbn(),
                record.jduKykShbt(),
                record.jduKykYmd(),
                record.shsyKyoTrno(),
                record.byssUmFlg(),
                record.skshHyjsKrno(),
                record.recUpdJnjNo());
    }

    /**
     * 概要: サービス契約・支払契約関連レコードを出力モデルへ変換します。
     *
     * 主な仕様:
     * SV-046 の output.tsv で必要な最小項目のみをサービス出力向けモデルへ写像します。
     *
     * 既知の制約:
     * 元レコードの請求開始年月や商材区分など、SV-046 の出力定義にない項目は返却しません。
     *
     * @param record サービス契約・支払契約関連レコード
     * @return サービス契約・支払契約関連情報モデル
     */
    private SvsihkrInf mapSvsihkr(final SvsihkrRecord record) {
        return new SvsihkrInf(
                record.svkyNo(),
                record.sihKiyNo(),
                record.skno(),
                record.tkysYmd());
    }
}
