/**
 * 概要: SV-149（直列G使用量・変動使用量エラー判定）の実装クラス。
 * 主な仕様:
 * - 未確認のときだけ SV-291 の判定を実行し、元ガスメーター時は直列グループ使用量集計も加えて評価する。
 * - 集計使用量不足、マイナス使用量、号数超過の業務エラー条件を順に評価する。
 * - 条件に一致した子エラーを収集し、判定完了後に報告システム種別に応じた業務エラーを送出する。
 */
public final class SeriesGUsageVariableUsageErrorJudgmentImpl
        implements SeriesGUsageVariableUsageErrorJudgment {

    private static final BigDecimal ZERO = BigDecimal.ZERO;

    private static final String ERROR_CODE_USAGE_TOTAL_EXCEEDED = ErrorCodeConstants.ERROR_CODE_020200041;

    private static final String ERROR_CODE_MINUS_USAGE = ErrorCodeConstants.ERROR_CODE_020200044;

    private static final String ERROR_CODE_METER_CAPACITY_EXCEEDED = ErrorCodeConstants.ERROR_CODE_020200045;

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final VariableUsageCheckPipeline variableUsageCheckPipeline;

    private final RequestContext requestContext;

    private final String reportingSystemTypeCirius;

    private final String originalGasMeterStatus;

    private final String supplyStopPaymentPending;

    private final String negativeUsageAmountErrorType;

    /**
     * 判定に必要な依存を初期化する。
     *
     * @param gmkskkdDataAccess ガスメーター検針結果＜導管＞取得 DA
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果取得 DA
     * @param kkytskrDataAccess 供給停止管理取得 DA
     * @param variableUsageCheckPipeline SV-291
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓API独自コンスタント取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public SeriesGUsageVariableUsageErrorJudgmentImpl(
            GmkskkdDataAccess gmkskkdDataAccess,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            VariableUsageCheckPipeline variableUsageCheckPipeline,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.variableUsageCheckPipeline = Objects.requireNonNull(
                variableUsageCheckPipeline,
                "variableUsageCheckPipeline");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        KaiheisenConstantService nonNullKaiheisenConstantService =
            Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService nonNullCiriusConstantService =
            Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.reportingSystemTypeCirius =
            RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        this.originalGasMeterStatus =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.supplyStopPaymentPending =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
        this.negativeUsageAmountErrorType =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.GhpSerialGroupUsageErrorType.TYPE,
                ConstantLookupKeys.Cirius.GhpSerialGroupUsageErrorType.THERE_IS_A_NEGATIVE_USAGE_AMOUNT);
    }

    @Override
    public void execute(
            Boolean kknnFlg,
            String ckrtGmscJtkb,
            LocalDate znkiKnsYmd,
            LocalDate hssgYmd,
            Long gasMtstNo,
            BigDecimal gasSyry,
            Integer snyMtrKnsu,
            String gmtKhsJtkb,
            BigDecimal gmtGusu,
            Long tksuSvkyNo,
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.GasRknKsnInf gasRknKsnInf) {
        if (!Boolean.FALSE.equals(kknnFlg)) {
            return;
        }

        List<HeisenRuleException> childErrorList = new ArrayList<>();

        List<GmkskkdRecord> gmkskkdRecords = List.of();
        if (originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            gmkskkdRecords = gmkskkdDataAccess.findAllByKnsYmdAndMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
                    znkiKnsYmd,
                    hssgYmd,
                    gasMtstNo,
                    null,
                    0);
            BigDecimal totalUsage = sumGmkskkdUsage(gmkskkdRecords)
                    .add(sumKtmgmkkUsage(ktmgmkkDataAccess.findAllByMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
                            gasMtstNo,
                            null,
                            0)))
                    .add(sumSupplyStopUsage(snyMtrKnsu, gmkskkdRecords));
            if (gasSyry.compareTo(totalUsage) < 0) {
                childErrorList.add(new HeisenRuleException(ERROR_CODE_USAGE_TOTAL_EXCEEDED));
            }
        }

        VariableUsageCheckPipelineOutput sv291Output = variableUsageCheckPipeline.execute(
                gasMtstNo,
                gmtKhsJtkb,
                gmtGusu,
                tksuSvkyNo,
                hssgYmd,
                gasSyry,
                resolveUsageDays(gasRknKsnInf));
        if (hasMinusUsageWarning(sv291Output)) {
            childErrorList.add(new HeisenRuleException(ERROR_CODE_MINUS_USAGE));
        }
        if (Boolean.TRUE.equals(sv291Output.gusuCkaErrUmFlg())) {
            childErrorList.add(new HeisenRuleException(ERROR_CODE_METER_CAPACITY_EXCEEDED));
        }
        if (!childErrorList.isEmpty()) {
            throwCommonBusinessError(childErrorList);
        }
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private int resolveUsageDays(
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.GasRknKsnInf gasRknKsnInf) {
        return gasRknKsnInf.shyDdsu() - gasRknKsnInf.kkyCushDdsu();
    }

    private boolean hasMinusUsageWarning(VariableUsageCheckPipelineOutput output) {
        return negativeUsageAmountErrorType.equals(output.gcgnSyreShbt())
                || Boolean.TRUE.equals(output.mnsSyryErrUmFlg());
    }

    private BigDecimal sumGmkskkdUsage(List<GmkskkdRecord> records) {
        BigDecimal total = ZERO;
        for (GmkskkdRecord record : records) {
            total = total.add(defaultValue(record.gasSyry()));
        }
        return total;
    }

    private BigDecimal sumKtmgmkkUsage(List<KtmgmkkRecord> records) {
        BigDecimal total = ZERO;
        for (KtmgmkkRecord record : records) {
            total = total.add(defaultValue(record.gasSyry()));
        }
        return total;
    }

    private BigDecimal sumSupplyStopUsage(Integer snyMtrKnsu, List<GmkskkdRecord> gmkskkdRecords) {
        BigDecimal total = ZERO;
        int meterCount = snyMtrKnsu == null ? 0 : snyMtrKnsu;
        for (int index = 0; index < meterCount; index++) {
            GmkskkdRecord gmkskkdRecord = gmkskkdRecords.get(index);
            Optional<KkytskrRecord> supplyStopRecord = kkytskrDataAccess.findByGasMtstNoAndHesnYmdAndKkysSisnJtkb(
                    gmkskkdRecord.gasMtstNo(),
                    gmkskkdRecord.knsYmd(),
                    supplyStopPaymentPending);
            if (supplyStopRecord.isPresent()) {
                total = total.add(defaultValue(supplyStopRecord.get().gasSyry()));
            }
        }
        return total;
    }

    private void throwCommonBusinessError(List<HeisenRuleException> childErrorList) {
        if (reportingSystemTypeCirius.equals(requireHukkSysShbt())) {
            throw new HeisenRuleListException(List.copyOf(childErrorList));
        }
        throw childErrorList.get(0);
    }

    private BigDecimal defaultValue(BigDecimal value) {
        return value == null ? ZERO : value;
    }
}
