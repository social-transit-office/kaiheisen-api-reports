package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ThirtyDayConvertedUsageCalculation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.ThirtyDayConvertedUsageCalculationOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UnitNumberExceededErrorJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UnitNumberExceededErrorJudgmentOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheck;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheckOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageJudgmentOutput;

/**
 * 概要: SV-304（変動使用量チェック）の実装クラス。
 * 主な仕様:
 * - 入力ガード、開栓中判定、３０日換算、号数超過エラー判定、変動使用量判定の順で処理する。
 * - 途中でエラー条件が成立した場合は、変動使用量エラー有無フラグを true に設定して早期終了する。
 * 既知の制約:
 * - RequestContext 項目は下位 SV 側で参照する前提とし、本クラスの public メソッド引数へは露出しない。
 */
public final class VariableUsageCheckImpl implements VariableUsageCheck {

    private static final BigDecimal ZERO = BigDecimal.ZERO;

    private final ThirtyDayConvertedUsageCalculation thirtyDayConvertedUsageCalculation;

    private final UnitNumberExceededErrorJudgment unitNumberExceededErrorJudgment;

    private final VariableUsageJudgment variableUsageJudgment;

    private final RequestContext requestContext;

    private final String openingStatus;

    private final String initialMeterReadingRequiredCode;

    /**
     * 概要: 必須依存と定数を注入するコンストラクタ。
     * 主な仕様:
     * - ガスメーター開閉栓状態区分の「開栓中」と初回検針適用コードの「初回検針適用要」を定数ライブラリから解決する。
     * - DI 依存はインスタンス生成時に fail-fast する。
     *
     * @param thirtyDayConvertedUsageCalculation SV-130
     * @param unitNumberExceededErrorJudgment SV-265
     * @param variableUsageJudgment SV-134
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public VariableUsageCheckImpl(
            ThirtyDayConvertedUsageCalculation thirtyDayConvertedUsageCalculation,
            UnitNumberExceededErrorJudgment unitNumberExceededErrorJudgment,
            VariableUsageJudgment variableUsageJudgment,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this(
                thirtyDayConvertedUsageCalculation,
                unitNumberExceededErrorJudgment,
                variableUsageJudgment,
            requestContext,
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING),
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.TYPE,
                        ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode
                                .INITIAL_METER_READING_REQUIRED));
    }

    /**
     * 概要: テスト用に依存と定数を直接受け取るコンストラクタ。
     * 主な仕様:
     * - 条件判定に使用する定数値を直接与える。
     * - DI 依存と定数文字列の null を fail-fast する。
     *
     * @param thirtyDayConvertedUsageCalculation SV-130
     * @param unitNumberExceededErrorJudgment SV-265
     * @param variableUsageJudgment SV-134
     * @param requestContext リクエストコンテキスト
     * @param openingStatus 開栓中状態コード
     * @param initialMeterReadingRequiredCode 初回検針適用要コード
     */
    VariableUsageCheckImpl(
            ThirtyDayConvertedUsageCalculation thirtyDayConvertedUsageCalculation,
            UnitNumberExceededErrorJudgment unitNumberExceededErrorJudgment,
            VariableUsageJudgment variableUsageJudgment,
            RequestContext requestContext,
            String openingStatus,
            String initialMeterReadingRequiredCode) {
        this.thirtyDayConvertedUsageCalculation =
                Objects.requireNonNull(
                        thirtyDayConvertedUsageCalculation, "thirtyDayConvertedUsageCalculation");
        this.unitNumberExceededErrorJudgment =
                Objects.requireNonNull(
                        unitNumberExceededErrorJudgment, "unitNumberExceededErrorJudgment");
        this.variableUsageJudgment =
                Objects.requireNonNull(variableUsageJudgment, "variableUsageJudgment");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.openingStatus = Objects.requireNonNull(openingStatus, "openingStatus");
        this.initialMeterReadingRequiredCode =
                Objects.requireNonNull(
                        initialMeterReadingRequiredCode, "initialMeterReadingRequiredCode");
    }

    @Override
    public VariableUsageCheckOutput execute(
            String gmtKhsJtkb,
            BigDecimal gmtGusu,
            Integer kokiShyDdsu,
            BigDecimal konSyry,
            Integer znkiShyDdsu,
            BigDecimal znkiSyry,
            Integer zndmShyDdsu,
            BigDecimal zndmSyry,
            String skknTkyCd,
            LocalDate knsYmd) {
        requireRequestContext();
        if (isBlank(gmtKhsJtkb) || isBlank(skknTkyCd)) {
            return errorOutput();
        }
        if (isInvalidNumericInput(gmtGusu, kokiShyDdsu, znkiShyDdsu, zndmShyDdsu)) {
            return errorOutput();
        }
        if (!openingStatus.equals(gmtKhsJtkb)) {
            return normalOutput();
        }
        if (konSyry.compareTo(ZERO) < 0) {
            return errorOutput();
        }

        BigDecimal currentUsage = convertUsage(konSyry, kokiShyDdsu);
        if (isUnitNumberExceeded(currentUsage, gmtGusu)) {
            return errorOutput();
        }
        if (initialMeterReadingRequiredCode.equals(skknTkyCd)) {
            return normalOutput();
        }

        BigDecimal previousUsage = convertUsage(znkiSyry, znkiShyDdsu);
        BigDecimal yearAgoUsage = convertUsage(zndmSyry, zndmShyDdsu);
        VariableUsageJudgmentOutput variableUsageJudgmentOutput =
                variableUsageJudgment.execute(
                        currentUsage,
                        previousUsage,
                        yearAgoUsage,
                        znkiShyDdsu,
                        zndmShyDdsu,
                        knsYmd);
        return new VariableUsageCheckOutput(variableUsageJudgmentOutput.hdsrErrUmFlg());
    }

    /**
     * 概要: 使用量を３０日換算する。
     * 主な仕様:
     * - SV-130 を呼び出し、戻り値の３０日換算後使用量を返す。
     * - 下位例外は変換せず上位へ伝播する。
     *
     * @param usage 使用量
     * @param usageDays 使用日数
     * @return ３０日換算後使用量
     */
    private BigDecimal convertUsage(BigDecimal usage, Integer usageDays) {
        ThirtyDayConvertedUsageCalculationOutput output =
                thirtyDayConvertedUsageCalculation.execute(usage, usageDays);
        return output.d30KnsnSyryZndm();
    }

    /**
     * 概要: 号数超過エラー判定を行う。
     * 主な仕様:
     * - SV-265 の戻り値に含まれる変動使用量エラー有無フラグをそのまま採用する。
     * - true の場合は後続の SV-134 判定へ進まない。
     *
     * @param usage 今回使用量
     * @param gmtGusu ガスメーター号数
     * @return 号数超過エラーに該当する場合 true
     */
    private boolean isUnitNumberExceeded(BigDecimal usage, BigDecimal gmtGusu) {
        UnitNumberExceededErrorJudgmentOutput output =
                unitNumberExceededErrorJudgment.execute(usage, gmtGusu);
        return Boolean.TRUE.equals(output.hdsrErrUmFlg());
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     * 主な仕様:
     * - correlationId、hukkSysShbt、systemDatetime の存在を確認する。
     * - 欠落がある場合は最初に検出した項目で処理を中断する。
     */
    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 文字列入力が null または空文字かを判定する。
     * 主な仕様:
     * - trim は行わず、空文字のみを不正値とみなす。
     * - null も不正値として扱う。
     *
     * @param value 判定対象
     * @return null または空文字の場合 true
     */
    private boolean isBlank(String value) {
        return value == null || value.isEmpty();
    }

    /**
     * 概要: 数値入力の事前ガード条件を判定する。
     * 主な仕様:
     * - ガスメーター号数が 0 以下、またはいずれかの使用日数が負値の場合をエラーとする。
     * - BigDecimal / Integer の null は呼び出し元前提のため compare 時にそのまま例外化される。
     *
     * @param gmtGusu ガスメーター号数
     * @param kokiShyDdsu 今回使用日数
     * @param znkiShyDdsu 前回使用日数
     * @param zndmShyDdsu 前年同月使用日数
     * @return 事前ガード条件に該当する場合 true
     */
    private boolean isInvalidNumericInput(
            BigDecimal gmtGusu,
            Integer kokiShyDdsu,
            Integer znkiShyDdsu,
            Integer zndmShyDdsu) {
        return gmtGusu.compareTo(ZERO) <= 0
                || kokiShyDdsu < 0
                || znkiShyDdsu < 0
                || zndmShyDdsu < 0;
    }

    /**
     * 概要: エラー結果を返却する。
     * 主な仕様:
     * - 変動使用量エラー有無フラグに true を設定する。
     * - 早期終了分岐の共通返却値として使用する。
     *
     * @return エラー結果
     */
    private VariableUsageCheckOutput errorOutput() {
        return new VariableUsageCheckOutput(Boolean.TRUE);
    }

    /**
     * 概要: 正常結果を返却する。
     * 主な仕様:
     * - 変動使用量エラー有無フラグに false を設定する。
     * - フォールスルー終了分岐の共通返却値として使用する。
     *
     * @return 正常結果
     */
    private VariableUsageCheckOutput normalOutput() {
        return new VariableUsageCheckOutput(Boolean.FALSE);
    }
}
