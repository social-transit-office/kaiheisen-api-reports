package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 概要: SV-052（開栓エラーフォロー取消実施可否判定）のサービス実装。
 * 主な仕様:
 * - DA-110-R002 でガスメーター社番を取得し、0件時は業務エラー 020000039 を送出する。
 * - DA-139-R003 でフォロー完了データを取得し、0件時は業務エラー 020000040 を送出する。
 * - フォロー対応完了年月日とシステム日付が不一致の場合は業務エラー 020000041 を送出する。
 * 既知の制約:
 * - 検索結果が複数件の場合は先頭1件を判定対象とする。
 * - システム日時は RequestContext から取得し、未設定時は fail-fast する。
 */
public class OpeningErrorFollowCancelFeasibilityJudgmentImpl
        implements OpeningErrorFollowCancelFeasibilityJudgment {

    private final GasmtrdDataAccess gasmtrdDataAccess;

    private final KseflDataAccess kseflDataAccess;

    private final RequestContext requestContext;

    private final String followStatusCompleted;

    /**
     * 概要: 必須依存と定数を初期化する。
     *
     * @param gasmtrdDataAccess ガスメーター＜導管＞データアクセス
     * @param kseflDataAccess 開栓エラーフォロー件名データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public OpeningErrorFollowCancelFeasibilityJudgmentImpl(
            GasmtrdDataAccess gasmtrdDataAccess,
            KseflDataAccess kseflDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gasmtrdDataAccess = Objects.requireNonNull(gasmtrdDataAccess, "gasmtrdDataAccess");
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.followStatusCompleted = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_FOLLOW_COMPLETED);
    }

    /**
     * 概要: 開栓エラーフォロー取消実施可否を判定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 判定結果
     */
    @Override
    public OpeningErrorFollowCancelFeasibilityJudgmentOutput execute(Long gasMtstNo, LocalDate trkKnsYmd) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        List<GasmtrdRecord> gasmtrdRecords =
                gasmtrdDataAccess.findAllByTtkGmsbNoAndGmtThkgYmd(
                        gasMtstNo,
                null);
        if (gasmtrdRecords.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000039);
        }

        String gmb = gasmtrdRecords.getFirst().gmb();
        List<KseflRecord> kseflRecords = kseflDataAccess.findAllByGmbAndFlwStCdAndKasgYmdAndTsigKasgYmd(
                gmb,
                followStatusCompleted,
                trkKnsYmd,
                trkKnsYmd);
        if (kseflRecords.isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000040);
        }

        KseflRecord target = kseflRecords.getFirst();
        if (!Objects.equals(target.fltkYmd(), systemDate)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000041);
        }

        return new OpeningErrorFollowCancelFeasibilityJudgmentOutput(
                gmb,
                toKseflInf(target));
    }

    private KseflInf toKseflInf(KseflRecord record) {
        return new KseflInf(
                record.kasgYmd(),
                record.kasHukkYmd(),
                record.hukkSysShbt(),
                record.sgyAnknNo(),
                record.kasJysb(),
                record.kasgNwkCd(),
                record.sgytSinNo(),
                record.sgytSinnKj(),
                record.utidxNo(),
                record.recUpdJnjNo());
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
}
