package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingDateErrorJudgment;

/**
 * 概要: SV-135（閉栓日エラー判定）の実装クラス。
 * 主な仕様:
 * - 閉栓作業年月日が基準日より後、同一グループ内設置場所未登録フラグが true、直列ガスメーター設置状態区分が不一致のいずれかで業務エラーを送出する。
 * - 基準日は RequestContext の systemDatetime を優先し、未設定時は SystemDateTimeProvider から補完する。
 * - 条件に該当しない場合は何も返さずに正常終了する。
 * 既知の制約:
 * - public メソッド引数の null ガードは行わず、上位層バリデーション済み入力を前提とする。
 * - 文字列の trim やコード値補正は行わない。
 */
public final class ClosingDateErrorJudgmentImpl implements ClosingDateErrorJudgment {

    private final SystemDateTimeProvider systemDateTimeProvider;

    private final RequestContext requestContext;

    /**
     * 概要: 既定のシステム日時プロバイダを利用するコンストラクタ。
     * 主な仕様: RequestContext を受け取り、現在日付補完には DefaultSystemDateTimeProvider を用いる。
     * 既知の制約: RequestContext の null は許容しない。
     *
     * @param requestContext リクエストコンテキスト
     */
    public ClosingDateErrorJudgmentImpl(RequestContext requestContext) {
        this(new DefaultSystemDateTimeProvider(), requestContext);
    }

    /**
     * 概要: テスト差し替え用のシステム日時プロバイダを受け取るコンストラクタ。
     * 主な仕様: provider と RequestContext を保持し、基準日の解決に使用する。
     * 既知の制約: provider と RequestContext の null は許容しない。
     *
     * @param systemDateTimeProvider システム日時プロバイダ
     * @param requestContext リクエストコンテキスト
     */
    public ClosingDateErrorJudgmentImpl(
            SystemDateTimeProvider systemDateTimeProvider,
            RequestContext requestContext) {
        this.systemDateTimeProvider = Objects.requireNonNull(systemDateTimeProvider, "systemDateTimeProvider");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
    }

    @Override
    public void execute(
            LocalDate hssgYmd,
            Boolean dgrpMiRegFlg,
            String systemDateCkrtGmscJtkb,
            String ckrtGmscJtkb) {
        if (isClosingDateAfterBaseDate(hssgYmd)
                || dgrpMiRegFlg
                || isSeriesMeterStatusDifferent(systemDateCkrtGmscJtkb, ckrtGmscJtkb)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200015);
        }
    }

    /**
     * 概要: 閉栓作業年月日が基準日より後かを判定する。
     * 主な仕様: RequestContext / provider から解決した基準日と比較し、後日の場合に true を返す。
     * 既知の制約: hssgYmd が null の場合は NullPointerException となる。
     *
     * @param hssgYmd 閉栓作業年月日
     * @return 閉栓作業年月日が基準日より後の場合 true
     */
    private boolean isClosingDateAfterBaseDate(LocalDate hssgYmd) {
        return hssgYmd.isAfter(resolveBaseDate());
    }

    /**
     * 概要: 直列ガスメーター設置状態区分が不一致かを判定する。
     * 主な仕様: システム日時点の区分と現在区分を完全一致で比較し、不一致なら true を返す。
     * 既知の制約: systemDateCkrtGmscJtkb が null の場合は NullPointerException となる。
     * ckrtGmscJtkb が null の場合は NullPointerException とはならず、不一致として true を返す。
     *
     * @param systemDateCkrtGmscJtkb システム日時点直列ガスメーター設置状態区分
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @return 区分が不一致の場合 true
     */
    private boolean isSeriesMeterStatusDifferent(
            String systemDateCkrtGmscJtkb,
            String ckrtGmscJtkb) {
        return !systemDateCkrtGmscJtkb.equals(ckrtGmscJtkb);
    }

    /**
     * 概要: 閉栓日判定に用いる基準日を解決する。
     * 主な仕様: RequestContext に systemDatetime がある場合はその年月日を使い、未設定時は provider の現在日付を使う。
     * 既知の制約: provider 自体の時刻取得失敗はそのまま上位へ伝播する。
     *
     * @return 閉栓日判定の基準日
     */
    private LocalDate resolveBaseDate() {
        return requestContext.getSystemDatetime()
                .map(LocalDateTime::toLocalDate)
                .orElseGet(systemDateTimeProvider::currentDate);
    }
}
