package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tkssvky.TkssvkyDataAccess;

/**
 * 概要: 閉栓対象ガスメーターの小売分類判定サービス実装。
 * 主な仕様:
 * - DA-108-R006 と DA-099-R008 を順に呼び出し、閉栓対象ガスメーターに紐づく託送契約情報を取得する。
 * - 導管ユーザーは東京ガス契約時に、小売ユーザーは東京ガス以外契約時に業務エラー 020000027 を送出する。
 * 既知の制約:
 * - DA-108-R006 と DA-099-R008 は仕様上 1 件のみ返却される前提で、取得結果の先頭 1 件を判定対象とする。
 * - correlationId、hukkSysShbt、systemDatetime は public 引数へ含めない。
 */
public final class ClosingTargetGasMeterRetailCategoryJudgmentImpl
        implements ClosingTargetGasMeterRetailCategoryJudgment {

    /** 託送サービス契約・ガスメーター設置場所関連データアクセス。 */
    private final TskgmstDataAccess tskgmstDataAccess;

    /** 託送サービス契約データアクセス。 */
    private final TkssvkyDataAccess tkssvkyDataAccess;

    /** リクエストコンテキスト。 */
    private final RequestContext requestContext;

    /** 導管ユーザーを表す選択メニュー区分。 */
    private final String conduitMenu;

    /** 託送サービス契約関連状態区分: 適用中。 */
    private final String applicableRelatedStatus;

    /** 託送サービス契約状態区分: 契約適用中。 */
    private final String underContractStatus;

    /** 東京ガスの小売事業者コード。 */
    private final String tokyoGasRetailerCode;

    /**
     * 概要: 判定に必要な依存と CIRIUS 定数を初期化する。
     * 主な仕様:
     * - DA-108-R006 / DA-099-R008 呼び出しに必要なデータアクセスを保持する。
     * - 基準日判定に必要な systemDatetime は RequestContext から取得する。
     * - 託送契約関連状態、託送契約状態、東京ガス小売事業者コードをコンストラクタで fail-fast に解決する。
     * 既知の制約:
     * - 必須依存または必須定数が不足している場合はコンストラクタ初期化時点で例外となる。
     * @param tskgmstDataAccessParam 託送サービス契約・ガスメーター設置場所関連データアクセス
     * @param tkssvkyDataAccessParam 託送サービス契約データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public ClosingTargetGasMeterRetailCategoryJudgmentImpl(
            TskgmstDataAccess tskgmstDataAccessParam,
            TkssvkyDataAccess tkssvkyDataAccessParam,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccessParam, "tskgmstDataAccess");
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccessParam, "tkssvkyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.conduitMenu =
                RequiredConstantResolver.requireKaiheisenConstant(
                        resolvedKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ConduitRetail.TYPE,
                        ConstantLookupKeys.Kaiheisen.ConduitRetail.CONDUIT);
        this.applicableRelatedStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.underContractStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.tokyoGasRetailerCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
    }

    /**
     * 概要: 閉栓対象ガスメーターの小売分類を判定する。
     * 主な仕様:
     * - RequestContext の systemDatetime の年月日を基準日に変換し、DA-108-R006 で託送サービス契約番号を取得する。
     * - 取得した託送サービス契約番号で DA-099-R008 を検索し、stkMnKbn と小売事業者コードの組み合わせを判定する。
     * - 正常時は託送契約種別、託送供給依頼事業者コード、使用者契約お客さま登録番号を返却する。
     * 既知の制約:
     * - gasMtstNo を含む public メソッド引数の null 許容可否は SV では自己判断せず、上位層バリデーション結果を前提とする。
     * - DA-108-R006 と DA-099-R008 は仕様上 1 件取得を前提とし、取得結果の先頭 1 件を使用する。
     * - RequestContext に systemDatetime が無い場合は IllegalStateException を送出する。
     * @param gasMtstNo ガスメーター設置場所番号
     * @param stkMnKbn 選択メニュー区分
     * @return 判定結果
     */
    @Override
    public ClosingTargetGasMeterRetailCategoryJudgmentOutput execute(
            Long gasMtstNo,
            String stkMnKbn) {
        LocalDate baseDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();

        List<TskgmstRecord> tskgmstRecords =
                tskgmstDataAccess.findAllByGasMtstNoAndTkysYmdAndTyenYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        baseDate,
                        applicableRelatedStatus);
        if (tskgmstRecords.isEmpty()) {
            throw businessRule();
        }

        TskgmstRecord tskgmstRecord = tskgmstRecords.get(0);
        List<TkssvkyRecord> tkssvkyRecords =
                tkssvkyDataAccess.findAllByTksuSvkyNoAndKnskKjnYmdAndTksuSvkyJtkb(
                        tskgmstRecord.tksuSvkyNo(),
                        baseDate,
                        underContractStatus);
        if (tkssvkyRecords.isEmpty()) {
            throw businessRule();
        }

        TkssvkyRecord tkssvkyRecord = tkssvkyRecords.get(0);
        if (shouldRaiseBusinessError(stkMnKbn, tkssvkyRecord.tksuKijsCd())) {
            throw businessRule();
        }

        return new ClosingTargetGasMeterRetailCategoryJudgmentOutput(
                tkssvkyRecord.tkykShbt(),
                tkssvkyRecord.tksuKijsCd(),
                tkssvkyRecord.syoKyoTrno());
    }

    /**
     * 概要: メニュー区分と小売事業者コードの組み合わせが業務エラー条件かを判定する。
     * 主な仕様:
     * - 導管ユーザーは KH004 の導管区分と一致する場合に東京ガス契約時エラーとする。
     * - 小売ユーザーは東京ガス以外契約時にエラーとする。
     * 既知の制約:
     * - stkMnKbn が KH004 の導管区分以外であればすべて小売ユーザーとして扱う。
     * @param stkMnKbn 選択メニュー区分
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @return 業務エラー条件に該当する場合は true
     */
    private boolean shouldRaiseBusinessError(String stkMnKbn, String tksuKijsCd) {
        boolean pipelineMenu = conduitMenu.equals(stkMnKbn);
        boolean tokyoGasRetailer = tokyoGasRetailerCode.equals(tksuKijsCd);
        return pipelineMenu == tokyoGasRetailer;
    }

    /**
     * 概要: SV-267 の業務エラー例外を生成する。
     * 主な仕様:
     * - 有効な契約が存在しない、または契約分類条件に一致しない場合の共通業務エラー 020000027 を返す。
     * 既知の制約:
     * - 例外の補足情報は本メソッドでは付加しない。
     * @return 業務エラー 020000027
     */
    private static BusinessRuleException businessRule() {
        return new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000027);
    }
}
