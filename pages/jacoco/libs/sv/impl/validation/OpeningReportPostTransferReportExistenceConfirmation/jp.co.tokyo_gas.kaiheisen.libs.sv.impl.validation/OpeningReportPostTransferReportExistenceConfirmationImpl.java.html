package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svckya.SvckyaDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svckya.SvckyaRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningReportPostTransferReportExistenceConfirmation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OpeningReportPostTransferReportExistenceConfirmationOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.OpeningReportPostTransferReportExistenceConfirmationSvckyaInf;

/** 開栓報告後異動報告有無確認サービス実装。 */
public final class OpeningReportPostTransferReportExistenceConfirmationImpl
        implements OpeningReportPostTransferReportExistenceConfirmation {

    private final SvckyaDataAccess svckyaDataAccess;

    private final String underContractStatus;

    private final String endOfContractStatus;

    private final String generalContractType;

    /**
     * コンストラクタ。
     *
     * @param svckyaDataAccess サービス契約データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public OpeningReportPostTransferReportExistenceConfirmationImpl(
            SvckyaDataAccess svckyaDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccess, "svckyaDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.endOfContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.END_OF_CONTRACT);
        this.generalContractType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ContractType.TYPE,
                ConstantLookupKeys.Cirius.ContractType.GENERAL_CONTRACT);
    }

    @Override
    public OpeningReportPostTransferReportExistenceConfirmationOutput execute(
            Long svkyNo,
            LocalDate knskKjnYmd) {
        List<String> targetStatuses = List.of(underContractStatus, endOfContractStatus);
        List<SvckyaRecord> records = svckyaDataAccess.findAllBySvkyNoAtKnskKjnYmdAndSvkyJtkbList(
                svkyNo,
                knskKjnYmd,
                targetStatuses);

        if (records.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs.FIND_ALL_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_LIST.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svkyNo)
                            .put("knskKjnYmd", knskKjnYmd)
                            .put("svkyJtkbList", targetStatuses));
        }

        SvckyaRecord record = records.getFirst();
        if (isTransferReportExisting(record, knskKjnYmd)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000048);
        }

        return new OpeningReportPostTransferReportExistenceConfirmationOutput(toOutputSvckyaInf(record));
    }

    private boolean isTransferReportExisting(SvckyaRecord record, LocalDate knskKjnYmd) {
        return Objects.equals(record.rkkiSttYmd(), knskKjnYmd)
                && Objects.equals(record.svkyJtkb(), underContractStatus)
                && !generalContractType.equals(record.kiyShbt());
    }

    private static OpeningReportPostTransferReportExistenceConfirmationSvckyaInf toOutputSvckyaInf(
            SvckyaRecord record) {
        return new OpeningReportPostTransferReportExistenceConfirmationSvckyaInf(
                record.svkySttJysb(),
                record.svkyEndJysb(),
                record.rkkiEndYmd(),
                record.kiyShbt());
    }
}
