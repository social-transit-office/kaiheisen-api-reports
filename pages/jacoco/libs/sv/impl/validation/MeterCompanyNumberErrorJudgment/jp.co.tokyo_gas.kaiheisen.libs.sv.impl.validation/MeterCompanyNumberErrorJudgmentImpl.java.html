package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.MeterCompanyNumberErrorJudgment;

/** メータ社番エラー判定サービス実装。 */
public final class MeterCompanyNumberErrorJudgmentImpl
        implements MeterCompanyNumberErrorJudgment {

    private final Set<String> meterPresenceCheckTargetWorkCodes;

    private final Set<String> meterCompanyNumberAlwaysCheckWorkCodes;

    private final Set<String> meterCompanyNumberRetailerCheckWorkCodes;

    private final String tokyoGasRetailerCode;

    /**
     * 判定に必要な CIRIUS 定数を解決して初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public MeterCompanyNumberErrorJudgmentImpl(CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        String transferWorkHistoryWorkCodeConduitType =
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE;
        String closingRegistration =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        transferWorkHistoryWorkCodeConduitType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .CLOSING_REGISTRATION);
        String closureErrorFollowUp =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        transferWorkHistoryWorkCodeConduitType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .CLOSURE_ERROR_FOLLOW_UP_CCJG08);
        String onlineClosureReportConduit =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        transferWorkHistoryWorkCodeConduitType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .ONLINE_CLOSURE_REPORT_CONDUIT);
        String supplyOutageReportConduit =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        transferWorkHistoryWorkCodeConduitType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .SUPPLY_OUTAGE_REPORT_CONDUIT);
        String supplyOutageReportRetail =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        transferWorkHistoryWorkCodeConduitType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit
                                .SUPPLY_OUTAGE_REPORT_RETAIL);

        this.meterPresenceCheckTargetWorkCodes =
                Set.of(closingRegistration, closureErrorFollowUp, supplyOutageReportRetail);
        this.meterCompanyNumberAlwaysCheckWorkCodes =
                Set.of(closingRegistration, closureErrorFollowUp, supplyOutageReportRetail);
        this.meterCompanyNumberRetailerCheckWorkCodes =
                Set.of(onlineClosureReportConduit, supplyOutageReportConduit);
        this.tokyoGasRetailerCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                        ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
    }

    /**
     * メータ社番エラー判定を実行する。
     *
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     * @param gasmtrdUmFlg ガスメーター＜導管＞有無フラグ
     * @param kjgsCd 小売事業者コード
     * @param gmb ガスメーター社番
     * @param heisenKihonArGmb 閉栓報告エリアガスメーター社番
     */
    @Override
    public void execute(
            String duknIsrgCd,
            Boolean gasmtrdUmFlg,
            String kjgsCd,
            String gmb,
            String heisenKihonArGmb) {
        if (shouldRejectForMeterPresence(duknIsrgCd, gasmtrdUmFlg)) {
            throw heisenRuleException();
        }
        if (!Objects.equals(gmb, heisenKihonArGmb)
                && shouldRejectForMeterCompanyNumberMismatch(duknIsrgCd, kjgsCd)) {
            throw heisenRuleException();
        }
    }

    /**
     * step 1 のガスメーター有無判定対象かを判定する。
     *
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     * @param gasmtrdUmFlg ガスメーター＜導管＞有無フラグ
     * @return 業務エラー対象の場合 true
     */
    private boolean shouldRejectForMeterPresence(String duknIsrgCd, Boolean gasmtrdUmFlg) {
        return meterPresenceCheckTargetWorkCodes.contains(duknIsrgCd)
                && Boolean.FALSE.equals(gasmtrdUmFlg);
    }

    /**
     * step 2 の社番一致判定を行う対象かを判定する。
     *
     * @param duknIsrgCd 異動・作業履歴業務コード（導管）
     * @param kjgsCd 小売事業者コード
     * @return 社番一致判定を行う場合 true
     */
    private boolean shouldRejectForMeterCompanyNumberMismatch(String duknIsrgCd, String kjgsCd) {
        if (meterCompanyNumberAlwaysCheckWorkCodes.contains(duknIsrgCd)) {
            return true;
        }
        return meterCompanyNumberRetailerCheckWorkCodes.contains(duknIsrgCd)
                && !tokyoGasRetailerCode.equals(kjgsCd);
    }

    /**
     * SV-137 で送出する閉栓共通エラーを生成する。
     *
     * @return 閉栓共通エラー
     */
    private HeisenRuleException heisenRuleException() {
        return new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200012);
    }
}
