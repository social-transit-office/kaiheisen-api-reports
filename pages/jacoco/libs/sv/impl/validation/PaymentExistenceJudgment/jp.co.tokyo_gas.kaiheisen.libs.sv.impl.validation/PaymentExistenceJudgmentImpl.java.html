package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.HashSet;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;

import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.uncollected_billings_convenience_deposits.UncollectedBillingsConvenienceDepositsRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.uncollected_billings_convenience_deposits.UncollectedBillingsConvenienceDepositsResponse;

/** SV-018 入金有無判定の実装。 */
public final class PaymentExistenceJudgmentImpl implements PaymentExistenceJudgment {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final String STATUS_PREFIX = "status=";

    private static final String BODY_PREFIX = ", body=";

    private final BillingCiriusAdapter billingCiriusAdapter;

    private final UncollectedBillingsConvenienceDepositsCiriusAdapter
            uncollectedBillingsConvenienceDepositsCiriusAdapter;

    private final RequestContext requestContext;

    private final String latestBillingStatusBilling;

    private final Set<String> targetDebtCodes;

    private final Set<String> allowedConvenienceRecordKinds;

    /**
     * コンストラクタ。
     *
     * @param billingCiriusAdapter 請求情報取得 adapter
     * @param uncollectedBillingsConvenienceDepositsCiriusAdapter 未収請求一覧取得 adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PaymentExistenceJudgmentImpl(
            BillingCiriusAdapter billingCiriusAdapter,
            UncollectedBillingsConvenienceDepositsCiriusAdapter
                    uncollectedBillingsConvenienceDepositsCiriusAdapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.billingCiriusAdapter = Objects.requireNonNull(billingCiriusAdapter, "billingCiriusAdapter");
        this.uncollectedBillingsConvenienceDepositsCiriusAdapter = Objects.requireNonNull(
                uncollectedBillingsConvenienceDepositsCiriusAdapter,
                "uncollectedBillingsConvenienceDepositsCiriusAdapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolved = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.latestBillingStatusBilling = RequiredConstantResolver.requireCiriusConstant(
                resolved,
                ConstantLookupKeys.Cirius.LatestBillingStatus.TYPE,
                ConstantLookupKeys.Cirius.LatestBillingStatus.BILLING);
        this.targetDebtCodes = Set.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.DebtCode.TYPE,
                        ConstantLookupKeys.Cirius.DebtCode.GAS_FEE),
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.DebtCode.TYPE,
                        ConstantLookupKeys.Cirius.DebtCode.LATE_GAS_BILL_INTEREST));
        this.allowedConvenienceRecordKinds = Set.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord.TYPE,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord
                                .POS_BREAKING_NEWS_RECEPTION),
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord.TYPE,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord
                                .CONFIRMATION_OF_THE_SAME_ROUTE_HAS_BEEN_RECEIVED),
                RequiredConstantResolver.requireCiriusConstant(
                        resolved,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord.TYPE,
                        ConstantLookupKeys.Cirius.ConvenienceStoreDepositReportRecord
                                .INVESTIGATION_IS_BEING_CONFIRMED_AFTER_RECEIVING_THE_BREAKING_NEWS));
    }

    @Override
    public PaymentExistenceJudgmentOutput execute(Long sikKrno) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        BillingResponse billingResponse;
        try {
            billingResponse = billingCiriusAdapter.getByBillingManagementNumber(
                    formatYmd(systemDate.minusYears(2)),
                    formatYmd(systemDate),
                    String.valueOf(sikKrno));
        } catch (CiriusApiException exception) {
            if (isNotFound(exception)) {
                return new PaymentExistenceJudgmentOutput(true);
            }
            throw exception;
        }
        List<BillingItem> billingItems = billingResponse == null ? null : billingResponse.sikList();
        if (billingItems == null) {
            return new PaymentExistenceJudgmentOutput(true);
        }

        Set<Integer> targetBillingSerialNumbers = collectTargetBillingSerialNumbers(
                billingItems,
                systemDate);
        if (targetBillingSerialNumbers.isEmpty()) {
            return new PaymentExistenceJudgmentOutput(true);
        }

        UncollectedBillingsConvenienceDepositsResponse response =
                uncollectedBillingsConvenienceDepositsCiriusAdapter.search(
                        UncollectedBillingsConvenienceDepositsRequest.of(
                                String.valueOf(sikKrno),
                                billingItems.size()));

        List<UncollectedBillingItem> uncollectedItems =
                response == null || response.response() == null
                        ? null
                        : response.response().mishuuSeikyuuIchiranList();
        if (uncollectedItems == null) {
            return new PaymentExistenceJudgmentOutput(true);
        }

        boolean existsDisallowedRecord = uncollectedItems.stream()
                .filter(Objects::nonNull)
                .anyMatch(item -> targetBillingSerialNumbers.contains(item.sikTsno())
                        && (item.cnsRecKbn() == null
                                || !allowedConvenienceRecordKinds.contains(item.cnsRecKbn())));
        return new PaymentExistenceJudgmentOutput(!existsDisallowedRecord);
    }

    private Set<Integer> collectTargetBillingSerialNumbers(
            List<BillingItem> billingItems,
            LocalDate systemDate) {
        Set<Integer> targetSerialNumbers = new HashSet<>();
        for (BillingItem billingItem : billingItems) {
            if (!isTargetBilling(billingItem, systemDate)) {
                continue;
            }
            if (containsTargetDebtCode(billingItem.squwSiknList())) {
                targetSerialNumbers.add(billingItem.sikTsno());
            }
        }
        return targetSerialNumbers;
    }

    private boolean isTargetBilling(BillingItem billingItem, LocalDate systemDate) {
        if (billingItem == null
                || billingItem.sikTsno() == null
                || billingItem.shkgYmd() == null
                || !latestBillingStatusBilling.equals(billingItem.sshnSqjkKbn())) {
            return false;
        }
        return toLocalDate(billingItem.shkgYmd()).isBefore(systemDate);
    }

    private boolean containsTargetDebtCode(List<BillingBreakdownDebtItem> breakdownDebtItems) {
        if (breakdownDebtItems == null) {
            return false;
        }
        for (BillingBreakdownDebtItem breakdownDebtItem : breakdownDebtItems) {
            if (breakdownDebtItem == null || breakdownDebtItem.squwDto() == null) {
                continue;
            }
            if (targetDebtCodes.contains(breakdownDebtItem.squwDto().siknCd())) {
                return true;
            }
        }
        return false;
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }

    private static String formatYmd(LocalDate date) {
        return date.format(BASIC_DATE);
    }

    private static boolean isNotFound(CiriusApiException exception) {
        Integer status = extractStatus(exception.getResponseInfo());
        return status != null && status == 404;
    }

    private static Integer extractStatus(String responseInfo) {
        if (responseInfo == null || !responseInfo.startsWith(STATUS_PREFIX)) {
            return null;
        }

        int bodyPrefixIndex = responseInfo.indexOf(BODY_PREFIX);
        String statusText = bodyPrefixIndex >= 0
                ? responseInfo.substring(STATUS_PREFIX.length(), bodyPrefixIndex)
                : responseInfo.substring(STATUS_PREFIX.length());
        try {
            return Integer.valueOf(statusText.trim());
        } catch (NumberFormatException ignored) {
            return null;
        }
    }

    private static LocalDate toLocalDate(Integer ymd) {
        String normalized = String.format("%08d", ymd);
        return LocalDate.parse(normalized, BASIC_DATE);
    }
}
