package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.DedicatedMeterOpeningInProgressErrorJudgment;

/**
 * 概要: SV-139（専用メータ開栓中エラー判定）の実装クラス.
 * 主な仕様:
 * - 直列ガスメーター設置状態区分が元ガスメーターであることを CIRIUS 定数で判定する。
 * - 直列専用開栓メータ有フラグが true の場合のみ業務エラー 020200020 を送出する。
 * - 上記条件に一致しない場合は何も返さず正常終了する。
 * 既知の制約:
 * - 入力値の null 許可/不許可や前後空白補正は上位層責務とし、本クラスでは追加の正規化を行わない。
 * - 必須定数が未登録の場合はコンストラクタ初期化時点で例外送出となり、インスタンス生成に失敗する。
 */
public final class DedicatedMeterOpeningInProgressErrorJudgmentImpl
        implements DedicatedMeterOpeningInProgressErrorJudgment {

    /** 元ガスメーターを表す直列ガスメーター設置状態区分. */
    private final String originalGasMeterStatus;

    /**
     * 概要: 判定に必要な CIRIUS 定数を解決してインスタンスを初期化する.
     * 主な仕様:
     * - 直列ガスメーター設置状態区分 J0244 の「元ガスメーター」を必須解決して保持する。
     * - CIRIUS 定数取得サービスの null は許容せず、fail-fast で検知する。
     * 既知の制約:
     * - 必須定数が未登録の場合は遅延評価せず、コンストラクタ実行時点で fail-fast として例外を送出する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public DedicatedMeterOpeningInProgressErrorJudgmentImpl(
            final CiriusConstantService ciriusConstantService) {
        CiriusConstantService resolvedService =
                Objects.requireNonNull(
                        ciriusConstantService,
                        "ciriusConstantService");
        String seriesGasMeterInstallationStatusType =
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE;
        String originalGasMeter =
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                        .ORIGINAL_GAS_METER;
        this.originalGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedService,
                        seriesGasMeterInstallationStatusType,
                        originalGasMeter);
    }

    /**
     * 概要: 専用メータ開栓中エラー判定を実行する.
     * 主な仕様:
     * - 直列ガスメーター設置状態区分が元ガスメーターで、
     *   かつ直列専用開栓メータ有フラグが true のとき業務例外を送出する。
     * - それ以外の組み合わせでは正常終了する。
     * 既知の制約:
     * - correlationId、hukkSysShbt、systemDatetime は
     *   SV 規約に従い公開引数へ含めない。
     * - public 引数の null ガードは追加せず、上位バリデーション済み前提で評価する。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param snyKasFlg 直列専用開栓メータ有フラグ
     */
    @Override
    public void execute(final String ckrtGmscJtkb, final Boolean snyKasFlg) {
        if (originalGasMeterStatus.equals(ckrtGmscJtkb)
                && Boolean.TRUE.equals(snyKasFlg)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200020);
        }
    }
}
