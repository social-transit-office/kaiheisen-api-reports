package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.RegressionReverseRegressionMarkCheck;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.UsageAmountErrorJudgment;

/**
 * 概要: SV-143（使用量エラー判定）の実装クラス。
 * 主な仕様:
 * - `SV-011` の回帰逆回帰マークチェック結果を利用して使用量エラー送出要否を判定する。
 * - 回帰判定結果フラグが true の場合に閉栓共通エラー `020200016` を送出する。
 * 既知の制約:
 * - `SV-011` への受渡しは、現行設計書および API 契約に合わせて 4 業務引数のみとする。
 */
public final class UsageAmountErrorJudgmentImpl implements UsageAmountErrorJudgment {

    private final RegressionReverseRegressionMarkCheck regressionReverseRegressionMarkCheck;

    /**
     * 概要: CDI から依存サービスを受け取って初期化する。
     * 主な仕様: 回帰逆回帰マークチェックサービスを保持する。
     * 既知の制約: 依存サービスの null は許容しない。
     *
     * @param regressionReverseRegressionMarkCheck 回帰逆回帰マークチェックサービス
     */
    @Inject
    public UsageAmountErrorJudgmentImpl(
            RegressionReverseRegressionMarkCheck regressionReverseRegressionMarkCheck) {
        this.regressionReverseRegressionMarkCheck = Objects.requireNonNull(
                regressionReverseRegressionMarkCheck,
                "regressionReverseRegressionMarkCheck");
    }

    /**
     * 概要: 使用量エラー判定を実行する。
     * 主な仕様:
     * - `SV-011` の回帰判定結果フラグが true の場合に閉栓共通エラー `020200016` を送出する。
     * - 回帰判定結果フラグが false の場合は正常終了する。
     * 既知の制約:
     * - public メソッド引数の null ガードは行わず、下位サービス契約に委譲する。
     *
     * @param kknnFlg 確認フラグ
     * @param sjskKbn 指示数回帰区分
     * @param konSyry 今回使用量
     * @param sjskHticSu 指示数回帰判定値数
     */
    @Override
    public void execute(
            Boolean kknnFlg,
            String sjskKbn,
            BigDecimal konSyry,
            Long sjskHticSu) {
        Boolean kaikiHtkFlg = regressionReverseRegressionMarkCheck.execute(
                kknnFlg,
                sjskKbn,
                konSyry,
                sjskHticSu).kaikiHtkFlg();
        if (Boolean.TRUE.equals(kaikiHtkFlg)) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200016);
        }
    }
}
