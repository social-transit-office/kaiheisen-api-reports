/**
 * 概要: SV-142（起算日特定エラー判定）の実装クラス。
 * 主な仕様:
 * - 既存の SV-312 を 1 回呼び出し、その戻り値と入力値を使って閉栓時のエラー条件を順に評価する。
 * - 指示数回帰区分が未設定の場合も後続判定を実施し、最後に未設定用の戻り値を返す。
 * 既知の制約:
 * - SV-312 の公開 API で受け取れる引数だけを利用するため、補助資料上の全入力値を直接転送はしない。
 * - public メソッド引数の null ガードや文字列正規化は追加しない。
 */
public final class CalculationStartDateIdentificationErrorJudgmentImpl
        implements CalculationStartDateIdentificationErrorJudgment {

    private static final int USAGE_DAYS_LIMIT = 51;

    private final GasUsageForBillingDerivation gasUsageForBillingDerivation;

    private final String relocationClosureForMeterReadingResult;

    private final String disasterClosureReason;

    private final String originalGasMeterStatus;

    private final String meterPossessionStatus;

    /**
     * 概要: 判定に必要な依存と定数を解決して初期化する。
     * 主な仕様:
    * - SV-312 を保持する。
     * - CIRIUS 定数から災害時閉栓、元ガスメーター、メーター保有状態、移転閉栓のコードを必須解決する。
     * 既知の制約:
    * - SV-312 と定数取得サービスの null は許容しない。
     *
     * @param gasUsageForBillingDerivation TG導管ガス使用量（料金計算用）取得サービス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public CalculationStartDateIdentificationErrorJudgmentImpl(
            GasUsageForBillingDerivation gasUsageForBillingDerivation,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gasUsageForBillingDerivation =
                Objects.requireNonNull(gasUsageForBillingDerivation, "gasUsageForBillingDerivation");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.relocationClosureForMeterReadingResult =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .RELOCATION_CLOSURE);
        this.disasterClosureReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
        this.originalGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.meterPossessionStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                        ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus
                                .METER_POSSESSION_STATUS);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public CalculationStartDateIdentificationErrorJudgmentOutput execute(
            Long gasMtstNo,
            LocalDate hssgYmd,
            Long gmtShjSu,
            String sjskKbn,
            String gmtKhsJtkb,
            Integer jkarSu,
            BigDecimal sjsuMincSu,
            Long sjskHticSu,
            String hesnJysb,
            BigDecimal gasSyry,
            LocalDate kokiRkksYmd,
            String ckrtGmscJtkb,
            ArrayList<SnyInfList> snyInfList,
            ArrayList<SuguGmtInfList> suguGmtInfList,
            ArrayList<SnySkinKnsInfList> snySkinKnsInfList,
            ArrayList<SuguGmtSkinKnsInfList> suguGmtSkinKnsInfList) {
        GasUsageForBillingDerivationState sv312State =
                executeGasUsageForBillingDerivation(
                        gasMtstNo,
                        hssgYmd,
                        gmtShjSu,
                        sjskKbn,
                        gmtKhsJtkb,
                        jkarSu,
                        sjsuMincSu,
                        sjskHticSu);
        GasUsageForBillingDerivationOutput sv312Output = sv312State.output();

        if (sv312State.targetMeterReadingResultError()) {
            return createInitialOutput();
        }

        if (sv312Output == null) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200040);
        }

        if (!disasterClosureReason.equals(hesnJysb) && isRegressionMarkSet(sjskKbn)) {
            validateReportedGasUsage(gasSyry, sv312Output);
        }
        validateUsageDays(sv312Output);

        validateDedicatedMeterRecentReadingDates(ckrtGmscJtkb, snyInfList, snySkinKnsInfList);
        validateComprehensiveMeterRecentReadingDates(suguGmtInfList, suguGmtSkinKnsInfList);

        return isRegressionMarkSet(sjskKbn)
                ? toSetRegressionOutput(gasSyry, sjskKbn, sv312Output)
                : toUnsetRegressionOutput(sv312Output);
    }

    private GasUsageForBillingDerivationState executeGasUsageForBillingDerivation(
            Long gasMtstNo,
            LocalDate hssgYmd,
            Long gmtShjSu,
            String sjskKbn,
            String gmtKhsJtkb,
            Integer jkarSu,
            BigDecimal sjsuMincSu,
            Long sjskHticSu) {
        try {
            return GasUsageForBillingDerivationState.success(
                    gasUsageForBillingDerivation.execute(
                            gasMtstNo,
                            hssgYmd,
                            gmtShjSu,
                            relocationClosureForMeterReadingResult,
                            sjskKbn,
                            gmtKhsJtkb,
                            jkarSu,
                            sjsuMincSu,
                            sjskHticSu,
                            BigDecimal.ZERO,
                            new KnsKkaTseiMaeJh(null, 0, Boolean.FALSE)));
        } catch (KaiheisenException exception) {
            if (ErrorCodeConstants.ERROR_CODE_020000074.equals(exception.getErrorCode())) {
                throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200040);
            }
            if (ErrorCodeConstants.ERROR_CODE_020000077.equals(exception.getErrorCode())) {
                return GasUsageForBillingDerivationState.withTargetMeterReadingResultError();
            }
            throw exception;
        }
    }

    private CalculationStartDateIdentificationErrorJudgmentOutput createInitialOutput() {
        return new CalculationStartDateIdentificationErrorJudgmentOutput(null, null, null, null, null, null);
    }

    /**
     * 概要: 指示数回帰区分が評価対象かを判定する。
     * 主な仕様: null でも空文字でもない場合のみ true を返す。
     * 既知の制約: 前後空白 trim は行わず、空文字との完全一致だけを未設定扱いとする。
     *
     * @param sjskKbn 指示数回帰区分
     * @return 評価対象の場合 true
     */
    private boolean isRegressionMarkSet(String sjskKbn) {
        return sjskKbn != null && !sjskKbn.isEmpty();
    }

    /**
     * 概要: 報告使用量の不一致を判定する。
     * 主な仕様:
     * - SV-312 の実供給圧力数換算前ガス使用量と入力のガス使用量を compareTo で比較する。
     * - 不一致時は算出使用量を保持した専用例外を送出する。
     * 既知の制約:
     * - いずれかが null の場合も不一致として扱う。
     *
     * @param gasSyry ガス使用量
     * @param sv312Output SV-312 の戻り値
     */
    private void validateReportedGasUsage(
            BigDecimal gasSyry,
            GasUsageForBillingDerivationOutput sv312Output) {
        BigDecimal calculatedUsage = resolveCalculatedGasUsage(sv312Output);
        if (gasSyry == null || calculatedUsage == null || gasSyry.compareTo(calculatedUsage) != 0) {
            throw new HaisenRuleHokokuShiyouryoHuicchiException(
                    calculatedUsage == null ? null : calculatedUsage.toPlainString());
        }
    }

    /**
     * 概要: 使用日数上限を判定する。
     * 主な仕様: 使用日数が 51 日以上の場合に業務エラー 020200043 を送出する。
     * 既知の制約: ガス料金計算用情報または使用日数が欠損している場合は業務エラー 020200040 を送出する。
     *
     * @param sv312Output SV-312 の戻り値
     */
    private void validateUsageDays(GasUsageForBillingDerivationOutput sv312Output) {
        if (resolveUsageDays(sv312Output) >= USAGE_DAYS_LIMIT) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200043);
        }
    }

    /**
     * 概要: 専用メーターの最近検針年月日を判定する。
     * 主な仕様:
     * - 直列元ガスメーターの場合だけ専用メーター情報を走査する。
     * - 保有状態の専用メーターに対応する最近検針年月日が null の場合は業務エラー 020200040 を送出する。
     * 既知の制約:
     * - 最近検針情報リストが不足している場合は対応要素なしとして null 扱いする。
     *
     * @param ckrtGmscJtkb 直列ガスメーター設置状態区分
     * @param snyInfList 専用メーター情報リスト
     * @param snySkinKnsInfList 専用メーター最近検針情報リスト
     */
    private void validateDedicatedMeterRecentReadingDates(
            String ckrtGmscJtkb,
            ArrayList<SnyInfList> snyInfList,
            ArrayList<SnySkinKnsInfList> snySkinKnsInfList) {
        if (!originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return;
        }

        ArrayList<SnyInfList> resolvedMeterInfoList = defaultList(snyInfList);
        ArrayList<SnySkinKnsInfList> resolvedRecentReadingList = defaultList(snySkinKnsInfList);
        for (int index = 0; index < resolvedMeterInfoList.size(); index++) {
            SnyInfList meterInfo = resolvedMeterInfoList.get(index);
            if (meterPossessionStatus.equals(meterInfo.kngmStbJtkb())
                    && resolveDedicatedRecentReadingDate(resolvedRecentReadingList, index) == null) {
                throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200040);
            }
        }
    }

    /**
     * 概要: 総合扱いガスメーターの最近検針年月日を判定する。
     * 主な仕様:
     * - 総合扱いガスメーター情報を走査し、保有状態の要素に対する最近検針年月日 null を業務エラーとする。
     * - リストが 0 件の場合は何もせず正常終了に進む。
     * 既知の制約:
     * - 最近検針情報リストが不足している場合は対応要素なしとして null 扱いする。
     *
     * @param suguGmtInfList 総合扱いガスメーター情報リスト
     * @param suguGmtSkinKnsInfList 総合他ガスメーター最近検針情報リスト
     */
    private void validateComprehensiveMeterRecentReadingDates(
            ArrayList<SuguGmtInfList> suguGmtInfList,
            ArrayList<SuguGmtSkinKnsInfList> suguGmtSkinKnsInfList) {
        ArrayList<SuguGmtInfList> resolvedMeterInfoList = defaultList(suguGmtInfList);
        ArrayList<SuguGmtSkinKnsInfList> resolvedRecentReadingList = defaultList(suguGmtSkinKnsInfList);
        for (int index = 0; index < resolvedMeterInfoList.size(); index++) {
            SuguGmtInfList meterInfo = resolvedMeterInfoList.get(index);
            if (meterPossessionStatus.equals(meterInfo.kngmStbJtkb())
                    && resolveComprehensiveRecentReadingDate(resolvedRecentReadingList, index) == null) {
                throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200040);
            }
        }
    }

    /**
     * 概要: 判定結果 DTO を生成する。
     * 主な仕様: SV-312 の戻り値から validation レイヤ専用 DTO を組み立てる。
     * 既知の制約: SV-312 の子 DTO が null の場合は対応する子 DTO を null とする。
     *
     * @param sv312Output SV-312 の戻り値
     * @return SV-142 の出力 DTO
     */
    private CalculationStartDateIdentificationErrorJudgmentOutput toOutput(
            BigDecimal outputUsage,
            String outputRegressionMark,
            GasUsageForBillingDerivationOutput sv312Output) {
        var sourcePreviousReading = sv312Output.znkiKnkkInf();
        var sourceRecentReading = sv312Output.skinKnkkInf();
        var sourceGasUsageInfo = sv312Output.gasRknKsnInf();
        var sourceCurrentReading = sv312Output.konKnkkInf();

        return new CalculationStartDateIdentificationErrorJudgmentOutput(
                outputUsage,
                sourcePreviousReading == null
                        ? null
                        : new ZnkiKnkkInf(sourcePreviousReading.knsYmd(), sourcePreviousReading.gmtShjSu()),
                sourceRecentReading == null
                        ? null
                : new SkinKnkkInf(
                    sourceRecentReading.knsYmd(),
                    sourceRecentReading.gmtShjSu(),
                    sourceRecentReading.sjskKbn()),
                sourceGasUsageInfo == null
                        ? null
                        : new GasRknKsnInf(
                        sourceGasUsageInfo.kokiRkksYmd(),
                                sourceGasUsageInfo.jkarSuKnzmGasSyry(),
                                sourceGasUsageInfo.shyDdsu(),
                                sourceGasUsageInfo.kkyCushDdsu()),
                sourceCurrentReading == null
                        ? null
                        : new KonKnkkInf(
                                sourceCurrentReading.jkarSuKnzmGasSyry(),
                                sourceCurrentReading.shyDdsu()),
                outputRegressionMark);
    }

    /**
     * 概要: 指示数回帰区分設定済み時の出力 DTO を生成する。
     * 主な仕様:
     * - 戻り値の算出使用量と指示数回帰区分には引数の値を設定する。
     * - それ以外の項目は SV-312 の戻り値を写像する。
     *
     * @param gasSyry 引数のガス使用量
     * @param sjskKbn 引数の指示数回帰区分
     * @param sv312Output SV-312 の戻り値
     * @return 指示数回帰区分設定済み時の出力 DTO
     */
    private CalculationStartDateIdentificationErrorJudgmentOutput toSetRegressionOutput(
            BigDecimal gasSyry,
            String sjskKbn,
            GasUsageForBillingDerivationOutput sv312Output) {
        return toOutput(gasSyry, sjskKbn, sv312Output);
    }

    /**
     * 概要: 指示数回帰区分未設定時の早期 return 用 DTO を生成する。
     * 主な仕様:
     * - 戻り値の算出使用量と指示数回帰区分には SV-312 で取得した値を設定する。
     * - それ以外の項目は SV-312 の戻り値を写像する。
     * 既知の制約: SV-312 の指示数回帰区分が null の場合は null をそのまま返す。
     *
     * @param sv312Output SV-312 の戻り値
     * @return 指示数回帰区分未設定時の出力 DTO
     */
    private CalculationStartDateIdentificationErrorJudgmentOutput toUnsetRegressionOutput(
            GasUsageForBillingDerivationOutput sv312Output) {
        return toOutput(resolveCalculatedGasUsage(sv312Output), sv312Output.sjskKbn(), sv312Output);
    }

    /**
     * 概要: SV-312 の実供給圧力数換算前ガス使用量を取得する。
     * 主な仕様: ガス料金計算用情報がない場合は null を返す。
     * 既知の制約: なし。
     *
     * @param sv312Output SV-312 の戻り値
     * @return 実供給圧力数換算前ガス使用量
     */
    private BigDecimal resolveCalculatedGasUsage(GasUsageForBillingDerivationOutput sv312Output) {
        var gasUsageInfo = sv312Output.gasRknKsnInf();
        return gasUsageInfo == null ? null : gasUsageInfo.jkarSuKnzmGasSyry();
    }

    private record GasUsageForBillingDerivationState(
            GasUsageForBillingDerivationOutput output,
            boolean targetMeterReadingResultError) {

        private static GasUsageForBillingDerivationState success(
                GasUsageForBillingDerivationOutput output) {
            return new GasUsageForBillingDerivationState(output, false);
        }

        private static GasUsageForBillingDerivationState withTargetMeterReadingResultError() {
            return new GasUsageForBillingDerivationState(
                    new GasUsageForBillingDerivationOutput(null, null, null, null, null, null),
                    true);
        }
    }

    /**
     * 概要: SV-312 戻り値から使用日数を取得する。
     * 主な仕様: 使用日数判定に必要な情報が欠損している場合は業務エラー 020200040 を送出する。
     * 既知の制約: 欠損値の補完は行わない。
     *
     * @param sv312Output SV-312 の戻り値
     * @return 使用日数
     */
    private Integer resolveUsageDays(
            GasUsageForBillingDerivationOutput sv312Output) {
        var gasUsageInfo = sv312Output.gasRknKsnInf();
        if (gasUsageInfo == null || gasUsageInfo.shyDdsu() == null) {
            throw new HeisenRuleException(ErrorCodeConstants.ERROR_CODE_020200040);
        }
        return gasUsageInfo.shyDdsu();
    }

    /**
     * 概要: 専用メーターの最近検針年月日を取り出す。
     * 主な仕様: 対応要素が存在しない場合は null を返す。
     * 既知の制約: なし。
     *
     * @param recentReadingList 専用メーター最近検針情報リスト
     * @param index 対象インデックス
     * @return 最近検針年月日
     */
    private LocalDate resolveDedicatedRecentReadingDate(
            ArrayList<SnySkinKnsInfList> recentReadingList,
            int index) {
        return index < recentReadingList.size() ? recentReadingList.get(index).snySkinKnsYmd() : null;
    }

    /**
     * 概要: 総合扱い他メーターの最近検針年月日を取り出す。
     * 主な仕様: 対応要素が存在しない場合は null を返す。
     * 既知の制約: なし。
     *
     * @param recentReadingList 総合他ガスメーター最近検針情報リスト
     * @param index 対象インデックス
     * @return 最近検針年月日
     */
    private LocalDate resolveComprehensiveRecentReadingDate(
            ArrayList<SuguGmtSkinKnsInfList> recentReadingList,
            int index) {
        return index < recentReadingList.size()
                ? recentReadingList.get(index).suguTmtrSkinKnsYmd()
                : null;
    }

    /**
     * 概要: null 許容の ArrayList を空リストへ正規化する。
     * 主な仕様: null の場合だけ新しい空の ArrayList を返す。
     * 既知の制約: 要素の defensive copy は行わない。
     *
     * @param list 対象リスト
     * @param <T> 要素型
     * @return 空リストまたは元のリスト
     */
    private <T> ArrayList<T> defaultList(ArrayList<T> list) {
        return list == null ? new ArrayList<>() : list;
    }
