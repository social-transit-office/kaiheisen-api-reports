package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingIncompleteMeteterErrorJudgment;

/**
 * 閉栓未済メーターエラー判定サービス実装。
 */
public class ClosingIncompleteMeteterErrorJudgmentImpl implements ClosingIncompleteMeteterErrorJudgment {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final String feePaymentMethodInstructionCategoryNoConcept;

    private final String chargeCalculationRelatedStatusRelatedValid;

    private final String chargeCalculationRelatedStatusBlank03;

    private final String gasMeterOpenCloseValveStatusOpening;

    private final List<String> targetchargeCalculationRelated;

    private final RequestContext requestContext;

    /**
     * コンストラクタ
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param SystemDateTimeProvider システム日付・日時を提供するインターフェース
     * @param GmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param SvmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     */
    public ClosingIncompleteMeteterErrorJudgmentImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            SvmstkrDataAccess svmstkrDataAccess,
            CiriusConstantService ciriusConstantService,
            RequestContext requestContext) {

        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                 ciriusConstantService,
                 "ciriusConstantService");

        //料金支払方法指示種別：概念なし
        this.feePaymentMethodInstructionCategoryNoConcept = RequiredConstantResolver.requireCiriusConstant(
                 resolvedCiriusConstantService,
                 ConstantLookupKeys.Cirius.FeePaymentMethodInstructionCategory.TYPE,
                 ConstantLookupKeys.Cirius.FeePaymentMethodInstructionCategory.NO_CONCEPT);

        //料金計算関連状態区分：関連有効
        this.chargeCalculationRelatedStatusRelatedValid = RequiredConstantResolver.requireCiriusConstant(
                 resolvedCiriusConstantService,
                 ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                 ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);

        //料金計算関連状態区分：関連終了待ち
        this.chargeCalculationRelatedStatusBlank03 = RequiredConstantResolver.requireCiriusConstant(
                 resolvedCiriusConstantService,
                 ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                 ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03);

        //料金関連状態区分リスト化
        this.targetchargeCalculationRelated = List.of(chargeCalculationRelatedStatusRelatedValid, chargeCalculationRelatedStatusBlank03);

        //ガスメーター開閉栓状態区分：開栓中
        this.gasMeterOpenCloseValveStatusOpening = RequiredConstantResolver.requireCiriusConstant(
                 resolvedCiriusConstantService,
                 ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                 ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);

    }

    /**
     * 概要: 引数の料金支払指示種別によって判定を行う
     * 主な仕様:
     * - 引数の料金支払方法指示種別が【50500 0：概念なし】ではない場合　GMTSETD取得処理を行う
     * - 引数の料金支払方法指示種別が【50500 0：概念なし】の場合　戻り値なしで処理終了
     * - GMTSETDの取得件数が0件の場合、DataInconsistencyExceptionを送出する
     * - GMTSETD で取得したガスメーター開閉栓状態区分が【02100 1：開栓中】の場合 BusinessRuleException を送出する。
     * - GMTSETD で取得したガスメーター開閉栓状態区分が【02100 1：開栓中】ではない場合 戻り値なしで処理終了
     * @param rshhSjsb 料金支払方法指示種別
     * @param svkyNo サービス契約番号
     * @param gmtSetNo ガスメーター設置場所番号
     */
    @Override
    public void execute(String rshhSjsb, Long svkyNo, Long gmtSetNo ) {

        if (!feePaymentMethodInstructionCategoryNoConcept.equals(rshhSjsb)) {
            List<SvmstkrRecord> svmstkrRecords =
                    svmstkrDataAccess.findAllBySvkyNoAndMtstNoAndRkkiKnrnJtkb(svkyNo, gmtSetNo, targetchargeCalculationRelated);

            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();

            svmstkrRecords.forEach(svmstkrRecord -> {
                Optional<GmtsetdRecord> gmtsetdRecords =
                        gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(svmstkrRecord.mtstNo(), knskKjnYmd);

                if (gmtsetdRecords.isEmpty()) {
                    throw DataInconsistencySupport.create(
                            gmtsetdDataAccess.tableName(),
                            GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                            new FlatJsonBuilder()
                                    .put("gasMtstNo", svmstkrRecord.mtstNo())
                                    .put("knskKjnYmd", knskKjnYmd));
                }
                gmtsetdRecords.ifPresent(gmtsetdRecord -> {
                    if (gasMeterOpenCloseValveStatusOpening.equals(gmtsetdRecord.gmtKhsJtkb())) {
                        throw businessRuleException(ErrorCodeConstants.ERROR_CODE_020000044);
                    }
                });
            });
        }
    }

    /**
     * 概要:  BusinessRuleException を生成する。
     * 主な仕様:
     * - executeメソッド から errorCode を取得する。
     * - 生成した例外は呼び出し元で直ちに送出することを想定する。
     * @param errorCode executeメソッドから受け取るエラーコード
     * @return BusinessRuleException インスタンス
     */
    private static BusinessRuleException businessRuleException(String errorCode) {
        return new BusinessRuleException(errorCode);
    }
}
