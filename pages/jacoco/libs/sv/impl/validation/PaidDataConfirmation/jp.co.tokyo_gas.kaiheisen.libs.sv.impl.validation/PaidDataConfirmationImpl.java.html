package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sihaky.SihakyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sihaky.SihakyRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.PaidDataConfirmation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.PaidDataConfirmationOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.PaymentExistenceJudgment;

/** SV-021 入金済みデータ確認の実装。 */
public final class PaidDataConfirmationImpl implements PaidDataConfirmation {

    private final SihakyDataAccess sihakyDataAccess;

    private final PaymentExistenceJudgment paymentExistenceJudgment;

    private final RequestContext requestContext;

    private final String invalidPaymentContractStatus;

    /**
     * コンストラクタ。
     *
     * @param sihakyDataAccess 支払契約 DA
     * @param paymentExistenceJudgment 入金有無判定 SV
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PaidDataConfirmationImpl(
            SihakyDataAccess sihakyDataAccess,
            PaymentExistenceJudgment paymentExistenceJudgment,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccess, "sihakyDataAccess");
        this.paymentExistenceJudgment = Objects.requireNonNull(
                paymentExistenceJudgment,
                "paymentExistenceJudgment");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.invalidPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
    }

    @Override
    public PaidDataConfirmationOutput execute(Long shrkyNo1, Long shrkyNo2) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        boolean firstResult = judgePaymentStatus(shrkyNo1, systemDate);
        if (!firstResult) {
            return new PaidDataConfirmationOutput(false);
        }
        if (shrkyNo2 == null) {
            return new PaidDataConfirmationOutput(true);
        }
        return new PaidDataConfirmationOutput(judgePaymentStatus(shrkyNo2, systemDate));
    }

    private boolean judgePaymentStatus(Long sihKiyNo, LocalDate systemDate) {
        SihakyRecord sihakyRecord = sihakyDataAccess
                .findBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
                        sihKiyNo,
                        systemDate,
                        invalidPaymentContractStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sihakyDataAccess.tableName(),
                        SihakyDataAccess.CallSpecs
                                .FIND_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_NOT_SIH_KIY_JTKB
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("sihKiyNo", sihKiyNo)
                                .put("tkysYmd", systemDate)
                                .put("sihKiyJtkb", invalidPaymentContractStatus)));
        return paymentExistenceJudgment.execute(sihakyRecord.sikKrno()).nyukinZmFlg();
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"));
    }
}
