package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageAmountUsageDaysCalculation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.UsageAmountUsageDaysCalculationOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheck;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheckOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheckPipeline;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.VariableUsageCheckPipelineOutput;

/**
 * 概要: SV-291（変動使用量チェック（導管））の実装クラス.
 * 主な仕様:
 * - 報告システム種別が CIRIUS の場合のみ、SV-285 を今回検針日と前年同月日で実行し、
 *   取得した使用量/使用日数を SV-304 へ連携する。
 * - 報告システム種別が CIRIUS 以外の場合は、号数超過エラー有無フラグへ false を設定する。
 * - ＧＨＰ直列グループ内使用量エラー種別は J0631 のエラーなしを返却する。
 * - マイナス使用量エラー有無フラグは、SV-304 の判定結果とは独立に
 *   入力使用量の符号で判定する。
 */
public final class VariableUsageCheckPipelineImpl
        implements VariableUsageCheckPipeline {

    /** 数値比較用のゼロ. */
    private static final BigDecimal ZERO = BigDecimal.ZERO;

    /** 使用量・使用日数算出サービス. */
    private final UsageAmountUsageDaysCalculation usageAmountUsageDaysCalculation;

    /** 変動使用量チェックサービス. */
    private final VariableUsageCheck variableUsageCheck;

    /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 報告システム種別: CIRIUS. */
    private final String reportingSystemTypeCirius;

    /** ＧＨＰ直列グループ内使用量エラー種別: エラーなし. */
    private final String ghpSerialGroupUsageNoErrors;

    /** 初回検針適用コード: 初回検針適用済. */
    private final String initialMeterReadingAppliedCode;

    /**
     * 概要: 必須依存と定数を注入するコンストラクタ.
     *
     * @param usageAmountUsageDaysCalculationService SV-285
     * @param variableUsageCheckService SV-304
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓API独自コンスタント取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    @Inject
    public VariableUsageCheckPipelineImpl(
            final UsageAmountUsageDaysCalculation usageAmountUsageDaysCalculationService,
            final VariableUsageCheck variableUsageCheckService,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.usageAmountUsageDaysCalculation = Objects.requireNonNull(
                usageAmountUsageDaysCalculationService,
                "usageAmountUsageDaysCalculationService");
        this.variableUsageCheck = Objects.requireNonNull(
                variableUsageCheckService,
                "variableUsageCheckService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        final KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(
                        kaiheisenConstantService,
                        "kaiheisenConstantService");
        final CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(
                        ciriusConstantService,
                        "ciriusConstantService");
        this.reportingSystemTypeCirius = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        this.ghpSerialGroupUsageNoErrors = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.GhpSerialGroupUsageErrorType.TYPE,
                ConstantLookupKeys.Cirius.GhpSerialGroupUsageErrorType.NO_ERRORS);
        this.initialMeterReadingAppliedCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.TYPE,
                ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.INITIAL_METER_READING_APPLIED);
    }

    @Override
    public VariableUsageCheckPipelineOutput execute(
            final Long gasMtstNo,
            final String gmtKhsJtkb,
            final BigDecimal gmtGusu,
            final Long tksuSvkyNo,
            final LocalDate kokiKnsYmd,
            final BigDecimal gasSyry,
            final Integer shyDdsu) {
        final boolean negativeUsage = isNegative(gasSyry);
        if (!reportingSystemTypeCirius.equals(requireHukkSysShbt())) {
            return new VariableUsageCheckPipelineOutput(
                    ghpSerialGroupUsageNoErrors,
                    false,
                    negativeUsage);
        }

        final UsageAmountUsageDaysCalculationOutput previousUsage =
                usageAmountUsageDaysCalculation.execute(
                        gasMtstNo,
                        kokiKnsYmd,
                        tksuSvkyNo);
        final UsageAmountUsageDaysCalculationOutput yearAgoUsage =
                usageAmountUsageDaysCalculation.execute(
                        gasMtstNo,
                        kokiKnsYmd.minusYears(1),
                        tksuSvkyNo);
        final VariableUsageCheckOutput variableUsageCheckOutput =
                variableUsageCheck.execute(
                        gmtKhsJtkb,
                        gmtGusu,
                        shyDdsu,
                        gasSyry,
                        previousUsage.shyDdsu(),
                        previousUsage.syry(),
                        yearAgoUsage.shyDdsu(),
                        yearAgoUsage.syry(),
                        initialMeterReadingAppliedCode,
                        kokiKnsYmd);

        return new VariableUsageCheckPipelineOutput(
                ghpSerialGroupUsageNoErrors,
                variableUsageCheckOutput.hdsrErrUmFlg(),
                negativeUsage);
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    /**
     * 概要: 入力使用量が負値か判定する.
     * 主な仕様:
     * - 入力使用量が 0 未満の場合に true を返却する。
     * - 入力使用量が 0 以上の場合に false を返却する。
     * 既知の制約:
     * - null は許容せず、compareTo 呼出し時に例外化される。
     *
     * @param usageAmount 判定対象の使用量
     * @return 入力使用量が負値の場合は true、それ以外は false
     */
    private boolean isNegative(final BigDecimal usageAmount) {
        return usageAmount.compareTo(ZERO) < 0;
    }
}
