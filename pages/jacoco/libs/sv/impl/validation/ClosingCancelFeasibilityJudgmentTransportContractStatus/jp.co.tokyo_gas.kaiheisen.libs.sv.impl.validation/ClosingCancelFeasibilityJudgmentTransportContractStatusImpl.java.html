package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;

/**
 * 概要: SV-104 閉栓取消実施可否判定: 託送契約状態コードサービス実装.
 *
 * <p>主な仕様: 託送メータ情報を取得し、託送契約状態コードが解約済かを判定して、
 * 判定対象レコードを SV API の出力 DTO へ変換して返却する。
 *
 * <p>既知の制約: public 引数の入力妥当性は上位レイヤで保証済みを前提とし、
 * 本クラスでは追加バリデーションや HTTP 変換を行わない。
 */
public final class ClosingCancelFeasibilityJudgmentTransportContractStatusImpl
        implements ClosingCancelFeasibilityJudgmentTransportContractStatus {

    /** 託送メータ情報 DA. */
    private final TaksmtrDataAccess taksmtrDataAccess;

    /** 解約済の託送契約状態コード. */
    private final String canceledStatus;

    /**
        * 概要: 必須依存を受け取って初期化する.
        *
        * <p>主な仕様: 依存 DA と CIRIUS 定数サービスを fail-fast で検証し、
        * 解約済の託送契約状態コードを初期化時に解決する。
        *
        * <p>既知の制約: 必須定数が定義されていない場合はインスタンス生成時に例外で失敗する。
     *
     * @param taksmtrDataAccessParam 託送メータ情報 DA
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ClosingCancelFeasibilityJudgmentTransportContractStatusImpl(
            final TaksmtrDataAccess taksmtrDataAccessParam,
            final CiriusConstantService ciriusConstantService) {
        this.taksmtrDataAccess = Objects.requireNonNull(
                taksmtrDataAccessParam,
                "taksmtrDataAccess");
        CiriusConstantService resolvedConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.canceledStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.CANCELED);
    }

        /**
         * 概要: 託送契約状態コードを判定し、判定対象レコードを返す.
     *
     * <p>主な仕様: DA-019-R001 で対象レコード一覧を取得し、0 件の場合は
     * {@link DataInconsistencyException} を送出する。先頭レコードの託送契約状態コードが
     * 0H800 4:解約済と一致しない場合は {@link BusinessRuleException} として業務エラー
     * 020000054 を送出する。一致する場合は取得した託送契約番号、引込位置番号、
     * ガスメータ設置場所番号を出力 DTO へ移送して返却する。
     *
     * <p>既知の制約: public 引数の null/形式妥当性は上位レイヤで検証済みを前提とし、
     * 本メソッドでは追加バリデーションを行わない。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 判定対象レコードから生成した出力 DTO
     */
    @Override
    public ClosingCancelFeasibilityJudgmentTransportContractStatusOutput execute(
            final Long gasMtstNo) {
        List<TaksmtrRecord> records = taksmtrDataAccess.findAllByGmtSetNo(gasMtstNo);
        if (records.isEmpty()) {
            throw dataInconsistencyForTaksmtr(gasMtstNo);
        }

        TaksmtrRecord record = records.get(0);
        if (!canceledStatus.equals(record.taksKyJtCd())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000054);
        }

        return toOutput(record);
    }

    /**
     * 概要: 取得レコードを出力 DTO へ変換する.
     *
     * <p>主な仕様: 託送契約番号、引込位置番号、ガスメータ設置場所番号を
     * SV API の出力 DTO へ移送する。
     *
     * <p>既知の制約: 変換対象は SV-104 が利用する項目に限定し、それ以外の DA 項目は返却しない。
     *
     * @param record 託送メータ情報
     * @return 出力 DTO
     */
    private ClosingCancelFeasibilityJudgmentTransportContractStatusOutput toOutput(
            final TaksmtrRecord record) {
        return new ClosingCancelFeasibilityJudgmentTransportContractStatusOutput(
                new TaksmtrInf(
                        record.takskyNo(),
                        record.hkkmItiNo(),
                        record.gmtSetNo()));
    }

        /**
         * 概要: DA-019-R001 の 0 件結果をデータ不整合として生成する.
         *
         * <p>主な仕様: 対象 DA の tableName、CallSpecs の identifier、gmtSetNo を含む
         * parameterJsonString を用いて DataInconsistencyException を構築する。
         *
         * <p>既知の制約: parameterJsonString は gmtSetNo のみを出力し、キー順はこのメソッドの
         * put 呼び出し順に従う。
         *
         * @param gasMtstNo ガスメーター設置場所番号
         * @return データ不整合例外
         */
    private DataInconsistencyException dataInconsistencyForTaksmtr(final Long gasMtstNo) {
        DataAccessCallSpec callSpec = TaksmtrDataAccess.CallSpecs.FIND_ALL_BY_GMT_SET_NO;
        FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
            .put("gmtSetNo", gasMtstNo);
        return DataInconsistencySupport.create(
                taksmtrDataAccess.tableName(),
                callSpec.identifier(),
                parameterBuilder);
    }
}
