/**
 * 支払契約関連情報更新（契約終了）のサービス実装。
 */
public class PaymentContractInfoUpdateForTerminationImpl
        implements PaymentContractInfoUpdateForTermination {

    static final String SERVICE_USER_ID = "KHSS032";

    private final SihakyDataAccess sihakyDataAccess;

    private final KuzfrkDataAccess kuzfrkDataAccess;

    private final HrikmiDataAccess hrikmiDataAccess;

    private final CrdhriDataAccess crdhriDataAccess;

    private final DsqkjhDataAccess dsqkjhDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String paymentContractApplicableEnds;

    private final String accountTransfer;

    private final String payment;

    private final String cardPayment;

    private final String electronicBilling;

    /**
     * 依存オブジェクトと必須定数を初期化する。
     *
     * @param sihakyDataAccess 支払契約 DA
     * @param kuzfrkDataAccess 口座振替関連情報 DA
     * @param hrikmiDataAccess 払込関連情報 DA
     * @param crdhriDataAccess カード払い関連情報 DA
     * @param dsqkjhDataAccess 電子請求関連情報 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PaymentContractInfoUpdateForTerminationImpl(
            SihakyDataAccess sihakyDataAccess,
            KuzfrkDataAccess kuzfrkDataAccess,
            HrikmiDataAccess hrikmiDataAccess,
            CrdhriDataAccess crdhriDataAccess,
            DsqkjhDataAccess dsqkjhDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccess, "sihakyDataAccess");
        this.kuzfrkDataAccess = Objects.requireNonNull(kuzfrkDataAccess, "kuzfrkDataAccess");
        this.hrikmiDataAccess = Objects.requireNonNull(hrikmiDataAccess, "hrikmiDataAccess");
        this.crdhriDataAccess = Objects.requireNonNull(crdhriDataAccess, "crdhriDataAccess");
        this.dsqkjhDataAccess = Objects.requireNonNull(dsqkjhDataAccess, "dsqkjhDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.paymentContractApplicableEnds = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.accountTransfer = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ACCOUNT_TRANSFER);
        this.payment = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.PAYMENT);
        this.cardPayment = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.CARD_PAYMENT);
        this.electronicBilling = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ELECTRONIC_BILLING);
    }

    /**
     * 支払契約関連情報を契約終了状態へ更新する。
     *
         * @param sihKiyNo 支払契約番号
         * @param skno 世代管理番号
         * @param sihHuhuShbt 支払方法種別
         * @param recUpdJnjNo レコード更新排他制御順序番号
     */
    @Override
        public void execute(
                        Long sihKiyNo,
                        Integer skno,
                        String sihHuhuShbt,
                        Integer recUpdJnjNo) {
        requireRequestContext();
        LocalDate tyenYmd = resolveSystemDate();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        updatePaymentContract(sihKiyNo, skno, recUpdJnjNo, tyenYmd, decodedOriginFunctionId, auditContext);

        if (accountTransfer.equals(sihHuhuShbt)) {
            updateAccountTransferInfo(sihKiyNo, skno, tyenYmd, decodedOriginFunctionId, auditContext);
        } else if (payment.equals(sihHuhuShbt)) {
            updatePaymentSlipInfo(sihKiyNo, skno, tyenYmd, decodedOriginFunctionId, auditContext);
        } else if (cardPayment.equals(sihHuhuShbt)) {
            updateCardPaymentInfo(sihKiyNo, skno, tyenYmd, decodedOriginFunctionId, auditContext);
        } else if (electronicBilling.equals(sihHuhuShbt)) {
            updateElectronicBillingInfo(sihKiyNo, skno, tyenYmd, decodedOriginFunctionId, auditContext);
        }

    }

    /**
     * RequestContext の必須項目を検証する。
     */
    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * RequestContext から更新年月日を取得する。
     *
     * @return 更新年月日
     */
    private LocalDate resolveSystemDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    /**
     * RequestContext の報告システム種別を監査用機能 ID に変換する。
     *
     * @return 監査用機能 ID
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    /**
     * 支払契約を適用終了へ更新する。
     */
    private void updatePaymentContract(
            Long sihKiyNo,
            Integer skno,
            Integer recUpdJnjNo,
            LocalDate tyenYmd,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = sihakyDataAccess.updateBySihKiyNoAndSknoAndRecUpdJnjNo(
                new SihakyWriteRecord(
                        sihKiyNo,
                        skno,
                        null,
                        tyenYmd,
                        paymentContractApplicableEnds,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(SihakyColumn.TYEN_YMD, SihakyColumn.SIH_KIY_JTKB),
                new SihakySknoJnjUpdateKey(sihKiyNo, skno, recUpdJnjNo),
                auditContext);
        if (updatedCount == 0) {
            throw createSihakyUpdateException(
                    sihKiyNo,
                    skno,
                    recUpdJnjNo,
                    tyenYmd,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 口座振替関連情報を更新する。
     */
    private void updateAccountTransferInfo(
            Long sihKiyNo,
            Integer skno,
            LocalDate tyenYmd,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = kuzfrkDataAccess.updateBySihKiyNoAndSihKiySkno(
                new KuzfrkWriteRecord(
                        sihKiyNo,
                        skno,
                        null,
                        tyenYmd,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(KuzfrkColumn.TYEN_YMD),
                new KuzfrkSihKiyUpdateKey(sihKiyNo, skno),
                auditContext);
        if (updatedCount == 0) {
            throw createPaymentMethodUpdateException(
                    kuzfrkDataAccess.tableName(),
                    KuzfrkDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                    sihKiyNo,
                    skno,
                    tyenYmd,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 払込関連情報を更新する。
     */
    private void updatePaymentSlipInfo(
            Long sihKiyNo,
            Integer skno,
            LocalDate tyenYmd,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = hrikmiDataAccess.updateBySihKiyNoAndSihKiySkno(
                new HrikmiWriteRecord(
                        sihKiyNo,
                        skno,
                        null,
                        tyenYmd,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(HrikmiColumn.TYEN_YMD),
                new HrikmiSihKiyUpdateKey(sihKiyNo, skno),
                auditContext);
        if (updatedCount == 0) {
            throw createPaymentMethodUpdateException(
                    hrikmiDataAccess.tableName(),
                    HrikmiDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                    sihKiyNo,
                    skno,
                    tyenYmd,
                    decodedOriginFunctionId);
        }
    }

    /**
     * カード払い関連情報を更新する。
     */
    private void updateCardPaymentInfo(
            Long sihKiyNo,
            Integer skno,
            LocalDate tyenYmd,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = crdhriDataAccess.updateBySihKiyNoAndSihKiySkno(
                new CrdhriWriteRecord(sihKiyNo, skno, null, tyenYmd, null, null, null),
                EnumSet.of(CrdhriColumn.TYEN_YMD),
                new CrdhriSihKiyUpdateKey(sihKiyNo, skno),
                auditContext);
        if (updatedCount == 0) {
            throw createPaymentMethodUpdateException(
                    crdhriDataAccess.tableName(),
                    CrdhriDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                    sihKiyNo,
                    skno,
                    tyenYmd,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 電子請求関連情報を更新する。
     */
    private void updateElectronicBillingInfo(
            Long sihKiyNo,
            Integer skno,
            LocalDate tyenYmd,
            String decodedOriginFunctionId,
            AuditContext auditContext) {
        int updatedCount = dsqkjhDataAccess.updateBySihKiyNoAndSihKiySkno(
                new DsqkjhWriteRecord(sihKiyNo, skno, null, tyenYmd, null, null, null, null),
                EnumSet.of(DsqkjhColumn.TYEN_YMD),
                new DsqkjhSihKiyUpdateKey(sihKiyNo, skno),
                auditContext);
        if (updatedCount == 0) {
            throw createPaymentMethodUpdateException(
                    dsqkjhDataAccess.tableName(),
                    DsqkjhDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                    sihKiyNo,
                    skno,
                    tyenYmd,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 支払契約更新失敗時のデータ不整合例外を生成する。
     */
    private RuntimeException createSihakyUpdateException(
            Long sihKiyNo,
            Integer skno,
            Integer recUpdJnjNo,
            LocalDate tyenYmd,
            String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                sihakyDataAccess.tableName(),
                SihakyDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                new FlatJsonBuilder()
                        .put("sihKiyNo", sihKiyNo)
                        .put("skno", skno)
                        .put("recUpdJnjNo", recUpdJnjNo)
                        .put("tyenYmd", tyenYmd)
                        .put("sihKiyJtkb", paymentContractApplicableEnds)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    /**
     * 支払方法別更新失敗時のデータ不整合例外を生成する。
     */
    private RuntimeException createPaymentMethodUpdateException(
            String tableName,
            String daIdentifier,
            Long sihKiyNo,
            Integer skno,
            LocalDate tyenYmd,
            String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tableName,
                daIdentifier,
                new FlatJsonBuilder()
                        .put("sihKiyNo", sihKiyNo)
                        .put("sihKiySkno", skno)
                        .put("tyenYmd", tyenYmd)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }
