/**
 * 概要: 契約復活時に託送サービス契約と関連情報を更新する SV 実装.
 * 主な仕様: 取消対象検針年月日時点で有効な関連情報と契約情報を順に取得し、
 * 契約復活状態へ更新したうえで使用者契約お客さま登録番号を返却する.
 * 既知の制約: recUpdJnjNo を指定しない非排他更新として扱うため、
 * 更新件数 0 件時のデータ不整合判定は SV 側で実施する.
 */
public final class TransportServiceContractMeterLocationUpdateForRevivalImpl
        implements TransportServiceContractMeterLocationUpdateForRevival {

    /** 監査項目に設定する更新ユーザーID. */
    static final String SERVICE_USER_ID = "KHSS110";

    /** 契約復活時に設定する終了年月日. */
    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    /** 関連更新用 DA 識別子. */
    private static final String RELATION_UPDATE_DA_IDENTIFIER =
            UPDATE_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO
                    .identifier();

    /** 託送サービス契約・ガスメーター設置場所関連 DA. */
    private final TskgmstDataAccess tskgmstDataAccess;

    /** 託送サービス契約 DA. */
    private final TkssvkyDataAccess tkssvkyDataAccess;

    /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 開閉栓定数取得サービス. */
    private final KaiheisenConstantService kaiheisenConstantService;

    /** 契約適用中の関連状態区分. */
    private final String applyingRelationStatus;

    /** 契約適用終了の関連状態区分. */
    private final String applicableEndRelationStatus;

    /** 契約中の契約状態区分. */
    private final String underContractStatus;

    /** 契約終了の契約状態区分. */
    private final String endOfContractStatus;

    /**
     * 必須依存と定数を初期化する.
     *
     * @param relationDataAccess TSKGMST データアクセス
     * @param contractDataAccess TKSSVKY データアクセス
     * @param context リクエストコンテキスト
     * @param constantService 開閉栓コンスタント取得サービス
     * @param ciriusService CIRIUS コンスタント取得サービス
     */
    public TransportServiceContractMeterLocationUpdateForRevivalImpl(
            final TskgmstDataAccess relationDataAccess,
            final TkssvkyDataAccess contractDataAccess,
            final RequestContext context,
            final KaiheisenConstantService constantService,
            final CiriusConstantService ciriusService) {
        String relationStatusType = ConstantLookupKeys.Cirius
                .TransportationServiceContractRelatedStatus.TYPE;
        String contractStatusType = ConstantLookupKeys.Cirius
                .TransportationServiceContractStatus.TYPE;
        this.tskgmstDataAccess = Objects.requireNonNull(
                relationDataAccess,
                "tskgmstDataAccess");
        this.tkssvkyDataAccess = Objects.requireNonNull(
                contractDataAccess,
                "tkssvkyDataAccess");
        this.requestContext = Objects.requireNonNull(context, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                constantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusService =
                Objects.requireNonNull(ciriusService, "ciriusConstantService");
        this.applyingRelationStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusService,
                        relationStatusType,
                        ConstantLookupKeys.Cirius
                                .TransportationServiceContractRelatedStatus
                                .APPLYING);
        this.applicableEndRelationStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusService,
                        relationStatusType,
                        ConstantLookupKeys.Cirius
                                .TransportationServiceContractRelatedStatus
                                .APPLICABLE_ENDS);
        this.underContractStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusService,
                        contractStatusType,
                        ConstantLookupKeys.Cirius
                                .TransportationServiceContractStatus
                                .UNDER_CONTRACT);
        this.endOfContractStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusService,
                        contractStatusType,
                        ConstantLookupKeys.Cirius
                                .TransportationServiceContractStatus
                                .END_OF_CONTRACT);
    }

    @Override
    public TransportServiceContractMeterLocationUpdateForRevivalOutput execute(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(
                SERVICE_USER_ID,
                decodedOriginFunctionId);

        TskgmstRecord relationRecord = tskgmstDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        trkKnsYmd,
                        relationStatuses())
                .orElseThrow(
                        () -> createRelationMissingException(
                                gasMtstNo,
                                trkKnsYmd));

        TkssvkyRecord contractRecord = tkssvkyDataAccess
                .findByTksuSvkyNoAndKnskKjnYmdAndTksuSvkyJtkbList(
                        relationRecord.tksuSvkyNo(),
                        trkKnsYmd,
                        contractStatuses())
                .orElseThrow(() -> createContractMissingException(
                        relationRecord.tksuSvkyNo(),
                        trkKnsYmd));

        int updatedContractCount =
                tkssvkyDataAccess.updateByTksuSvkyNoAndSknoAndRecUpdJnjNo(
                        buildTkssvkyWriteRecord(contractRecord),
                        Set.of(
                                TkssvkyColumn.TKSU_SVKY_JTKB,
                                TkssvkyColumn.TKSU_SVKE_JYSB,
                                TkssvkyColumn.TYEN_YMD),
                        new TkssvkySknoJnjUpdateKey(
                                contractRecord.tksuSvkyNo(),
                                contractRecord.skno()),
                        auditContext);
        if (updatedContractCount == 0) {
            throw createContractUpdateException(
                    contractRecord,
                    decodedOriginFunctionId);
        }

        int updatedRelationCount =
                tskgmstDataAccess
                        .updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                                buildTskgmstWriteRecord(relationRecord),
                                Set.of(
                                        TskgmstColumn.TKSU_SVKK_JTKB,
                                        TskgmstColumn.TYEN_YMD),
                                new TskgmstUpdateKey(
                                        relationRecord.gasMtstNo(),
                                        relationRecord.tksuSvkyNo(),
                                        relationRecord.skno()),
                                auditContext);
        if (updatedRelationCount == 0) {
            throw createRelationUpdateException(
                    relationRecord,
                    decodedOriginFunctionId);
        }

        return new TransportServiceContractMeterLocationUpdateForRevivalOutput(
                contractRecord.syoKyoTrno());
    }

    /**
     * 復活対象として許容する関連状態区分一覧を返す.
     *
     * @return 復活対象の関連状態区分一覧
     */
    private List<String> relationStatuses() {
        return List.of(applyingRelationStatus, applicableEndRelationStatus);
    }

    /**
     * 復活対象として許容する契約状態区分一覧を返す.
     *
     * @return 復活対象の契約状態区分一覧
     */
    private List<String> contractStatuses() {
        return List.of(underContractStatus, endOfContractStatus);
    }

    /**
     * リクエストコンテキストから更新機能IDを解決する.
     *
     * @return 更新機能ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt().orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.hukkSysShbt is required")));
    }

    /**
     * 関連情報が存在しない場合のデータ不整合例外を生成する.
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 生成したデータ不整合例外
     */
    private RuntimeException createRelationMissingException(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                relationFindDaIdentifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", trkKnsYmd)
                        .put(
                                "tksuSvkkJtkbList",
                                relationStatuses()));
    }

    /**
     * 契約情報が存在しない場合のデータ不整合例外を生成する.
     *
     * @param tksuSvkyNo 託送サービス契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 生成したデータ不整合例外
     */
    private RuntimeException createContractMissingException(
            final Long tksuSvkyNo,
            final LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                contractFindDaIdentifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", tksuSvkyNo)
                        .put("knskKjnYmd", trkKnsYmd)
                        .put(
                                "tksuSvkyJtkbList",
                                contractStatuses()));
    }

    /**
     * 契約更新件数が 0 件だった場合のデータ不整合例外を生成する.
     *
     * @param contractRecord 契約レコード
     * @param decodedOriginFunctionId 更新機能ID
     * @return 生成したデータ不整合例外
     */
    private RuntimeException createContractUpdateException(
            final TkssvkyRecord contractRecord,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                contractUpdateDaIdentifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", contractRecord.tksuSvkyNo())
                        .put("skno", contractRecord.skno())
                        .put("tksuSvkyJtkb", underContractStatus)
                        .put("tksuSvkeJysb", null)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put(
                                "updKtnKinoId",
                                decodedOriginFunctionId));
    }

    /**
     * 関連更新件数が 0 件だった場合のデータ不整合例外を生成する.
     *
     * @param relationRecord 関連レコード
     * @param decodedOriginFunctionId 更新機能ID
     * @return 生成したデータ不整合例外
     */
    private RuntimeException createRelationUpdateException(
            final TskgmstRecord relationRecord,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                RELATION_UPDATE_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("gasMtstNo", relationRecord.gasMtstNo())
                        .put("tksuSvkyNo", relationRecord.tksuSvkyNo())
                        .put("skno", relationRecord.skno())
                        .put("tksuSvkkJtkb", applyingRelationStatus)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put(
                                "updKtnKinoId",
                                decodedOriginFunctionId));
    }

    /**
     * 関連検索の DA 識別子を返す.
     *
     * @return 関連検索の DA 識別子
     */
    private String relationFindDaIdentifier() {
        return FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_TKSU_SVKK_JTKB
                .identifier();
    }

    /**
     * 契約検索の DA 識別子を返す.
     *
     * @return 契約検索の DA 識別子
     */
    private String contractFindDaIdentifier() {
        return FIND_BY_TKSU_SVKY_NO_AND_KNSK_KJN_YMD_AND_TKSU_SVKY_JTKB_LIST
                .identifier();
    }

    /**
     * 契約更新の DA 識別子を返す.
     *
     * @return 契約更新の DA 識別子
     */
    private String contractUpdateDaIdentifier() {
        return UPDATE_BY_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO
                .identifier();
    }

    /**
     * 契約復活用の契約更新レコードを生成する.
     *
     * @param contractRecord 契約レコード
     * @return 契約更新レコード
     */
    private TkssvkyWriteRecord buildTkssvkyWriteRecord(
            final TkssvkyRecord contractRecord) {
        return new TkssvkyWriteRecord(
                contractRecord.tksuSvkyNo(),
                contractRecord.skno(),
                underContractStatus,
                contractRecord.tksuSvksJysb(),
                null,
                contractRecord.tkysYmd(),
                MAX_DATE,
                contractRecord.hesnYtiYmd(),
                contractRecord.tkykShbt(),
                contractRecord.tksuKijsCd(),
                contractRecord.sutiKskhFlg(),
                contractRecord.syoKyoTrno());
    }

    /**
     * 契約復活用の関連更新レコードを生成する.
     *
     * @param relationRecord 関連レコード
     * @return 関連更新レコード
     */
    private TskgmstWriteRecord buildTskgmstWriteRecord(
            final TskgmstRecord relationRecord) {
        return new TskgmstWriteRecord(
                relationRecord.gasMtstNo(),
                relationRecord.tksuSvkyNo(),
                relationRecord.skno(),
                applyingRelationStatus,
                relationRecord.tkysYmd(),
                MAX_DATE);
    }
