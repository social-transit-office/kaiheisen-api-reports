/**
 * 概要: SV-106（サービス契約復活（TG小売））のサービス実装。
 * 主な仕様:
 * - 契約復活対象の場合は SV-108 を実行し、対象のサービス契約・メーター設置場所関連を更新する。
 * - 直列ガスメーター設置状態区分に応じて SV-110、SV-111、DA-089-D001 を分岐実行する。
 * - DA-041-U002 および DA-033-R006 の 0 件は DataInconsistencyException に変換する。
 */
public final class ServiceContractRevivalTgRetailImpl
        implements ServiceContractRevivalTgRetail {

    static final String SERVICE_USER_ID = "KHSS106";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final DktksryDataAccess dktksryDataAccess;

    private final ServiceContractUpdateForRevival serviceContractUpdateForRevival;

    private final TransportServiceContractMeterLocationUpdateForRevival
            transportServiceContractMeterLocationUpdateForRevival;

    private final DedicatedMeterTransportServiceContractUpdateForRevival
            dedicatedMeterTransportServiceContractUpdateForRevival;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String relatedValidStatus;

    private final String originalGasMeterStatus;

    private final String dedicatedGasMeterStatus;

    /**
     * 必須依存と定数を初期化する。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DA
     * @param dktksryDataAccess 導管通知確定使用量 DA
     * @param serviceContractUpdateForRevival サービス契約更新（契約復活）SV
     * @param transportServiceContractMeterLocationUpdateForRevival 託送サービス契約更新（契約復活）SV
     * @param dedicatedMeterTransportServiceContractUpdateForRevival 専用メーター託送サービス契約更新（契約復活）SV
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ServiceContractRevivalTgRetailImpl(
            final SvmstkrDataAccess svmstkrDataAccess,
            final GmtsetdDataAccess gmtsetdDataAccess,
            final DktksryDataAccess dktksryDataAccess,
            final ServiceContractUpdateForRevival serviceContractUpdateForRevival,
            final TransportServiceContractMeterLocationUpdateForRevival transportServiceContractMeterLocationUpdateForRevival,
            final DedicatedMeterTransportServiceContractUpdateForRevival dedicatedMeterTransportServiceContractUpdateForRevival,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.dktksryDataAccess = Objects.requireNonNull(dktksryDataAccess, "dktksryDataAccess");
        this.serviceContractUpdateForRevival = Objects.requireNonNull(
                serviceContractUpdateForRevival,
                "serviceContractUpdateForRevival");
        this.transportServiceContractMeterLocationUpdateForRevival = Objects.requireNonNull(
                transportServiceContractMeterLocationUpdateForRevival,
                "transportServiceContractMeterLocationUpdateForRevival");
        this.dedicatedMeterTransportServiceContractUpdateForRevival = Objects.requireNonNull(
                dedicatedMeterTransportServiceContractUpdateForRevival,
                "dedicatedMeterTransportServiceContractUpdateForRevival");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.relatedValidStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
    }

    /**
     * TG 小売のサービス契約復活処理を実行する。
     *
     * @param svckyaInf サービス契約情報
     * @param trkKnsYmd 取消対象検針年月日
     * @param svmstkrInfList サービス契約・メーター設置場所関連情報リスト
     * @param gmtsetdInf ガスメーター設置場所＜導管＞情報
     * @return サービス契約復活処理結果
     */
    @Override
    public ServiceContractRevivalTgRetailOutput execute(
            final SvckyaInf svckyaInf,
            final LocalDate trkKnsYmd,
            final List<SvmstkrInf> svmstkrInfList,
            final GmtsetdInf gmtsetdInf) {
        if (!isRevivalTarget(svckyaInf.rkkiEndYmd(), trkKnsYmd)) {
            return new ServiceContractRevivalTgRetailOutput(false, false, null);
        }

        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        serviceContractUpdateForRevival.execute(
                svckyaInf.svkyNo(),
                trkKnsYmd,
                svckyaInf.sziKziKbn(),
                svckyaInf.skno(),
                svckyaInf.kyttSkno());

        Long firstCustomerRegistrationNumber = null;
        boolean capturedCustomerRegistrationNumber = false;
        LocalDate nextMeterReadingDate = trkKnsYmd.plusDays(1);

        for (SvmstkrInf svmstkrInf : svmstkrInfList) {
            if (!Objects.equals(svmstkrInf.rkkiEndYmd(), svckyaInf.rkkiEndYmd())) {
                continue;
            }

            updateRelatedServiceContractMeterLocation(
                    svmstkrInf,
                    auditContext,
                    decodedOriginFunctionId);

            GmtsetdRecord gasMeterLocation = gmtsetdDataAccess
                    .findByGasMtstNoAndTkysYmdAndTyenYmd(svmstkrInf.mtstNo(), trkKnsYmd)
                    .orElseThrow(() -> createGasMeterLocationMissingException(
                            svmstkrInf.mtstNo(),
                            trkKnsYmd));

            String seriesGasMeterStatus = gasMeterLocation.ckrtGmscJtkb();
            if (!dedicatedGasMeterStatus.equals(seriesGasMeterStatus)) {
                TransportServiceContractMeterLocationUpdateForRevivalOutput output =
                        transportServiceContractMeterLocationUpdateForRevival.execute(
                                svmstkrInf.mtstNo(),
                                trkKnsYmd);
                if (!capturedCustomerRegistrationNumber) {
                    firstCustomerRegistrationNumber = output.syoKyoTrno();
                    capturedCustomerRegistrationNumber = true;
                }
            }

            if (originalGasMeterStatus.equals(seriesGasMeterStatus)) {
                dedicatedMeterTransportServiceContractUpdateForRevival.execute(
                        svmstkrInf.mtstNo(),
                        trkKnsYmd,
                        gmtsetdInf);
            } else if (dedicatedGasMeterStatus.equals(seriesGasMeterStatus)) {
                dktksryDataAccess.deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                        svmstkrInf.mtstNo(),
                        nextMeterReadingDate,
                        0,
                        null);
            }
        }

        return new ServiceContractRevivalTgRetailOutput(
                true,
                true,
                firstCustomerRegistrationNumber);
    }

    private boolean isRevivalTarget(
            final LocalDate serviceContractEndDate,
            final LocalDate cancellationMeterReadingDate) {
        return !MAX_DATE.equals(serviceContractEndDate)
                && !serviceContractEndDate.isBefore(cancellationMeterReadingDate);
    }

    private void updateRelatedServiceContractMeterLocation(
            final SvmstkrInf svmstkrInf,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        int updatedCount = svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                buildSvmstkrWriteRecord(svmstkrInf),
                Set.of(
                        SvmstkrColumn.RKKI_KNRN_JTKB,
                        SvmstkrColumn.RKKI_END_YMD),
                new SvmstkrSknoUpdateKey(
                        svmstkrInf.mtstNo(),
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        svmstkrInf.recUpdJnjNo()),
                auditContext);
        if (updatedCount == 0) {
            throw createRelatedServiceContractMeterLocationUpdateException(
                    svmstkrInf,
                    decodedOriginFunctionId);
        }
    }

    private SvmstkrWriteRecord buildSvmstkrWriteRecord(final SvmstkrInf svmstkrInf) {
        return new SvmstkrWriteRecord(
                svmstkrInf.mtstNo(),
                svmstkrInf.svkyNo(),
                svmstkrInf.skno(),
                null,
                null,
                relatedValidStatus,
                null,
                MAX_DATE,
                null);
    }

    private RuntimeException createRelatedServiceContractMeterLocationUpdateException(
            final SvmstkrInf svmstkrInf,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                UPDATE_BY_MTST_NO_AND_SVKY_NO_AND_SKNO_KEY_ONLY.identifier(),
                new FlatJsonBuilder()
                        .put("mtstNo", svmstkrInf.mtstNo())
                        .put("svkyNo", svmstkrInf.svkyNo())
                        .put("skno", svmstkrInf.skno())
                        .put("rkkiKnrnJtkb", relatedValidStatus)
                        .put("rkkiEndYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createGasMeterLocationMissingException(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", trkKnsYmd));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt().orElseThrow(
                () -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }
}
