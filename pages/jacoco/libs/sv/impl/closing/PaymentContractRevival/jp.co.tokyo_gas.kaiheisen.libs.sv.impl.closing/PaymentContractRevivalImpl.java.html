import static jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svsihkr.SvsihkrDataAccess.CallSpecs.FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_SIH_KIY_JTKB;

/**
 * 概要: SV-112（支払契約復活）のサービス実装。
 * 主な仕様:
 * - DA-044-R006、DA-075-R006 で契約復活対象の支払契約関連と支払契約を取得する。
 * - 契約復活日に応じて SV-095、SV-030、SV-113 を分岐実行する。
 * - ガス最終精算支払方法指示削除と他商材更新を最後に実行し、支払契約情報を返却する。
 */
public final class PaymentContractRevivalImpl implements PaymentContractRevival {

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final SihakyDataAccess sihakyDataAccess;

    private final GssihhhDataAccess gssihhhDataAccess;

    private final ServicePaymentContractUpdateForRevival servicePaymentContractUpdateForRevival;

    private final PaymentContractInfoUpdateForRevival paymentContractInfoUpdateForRevival;

    private final CompanyTransferUpdate companyTransferUpdate;

    private final InvoiceDisplayControlUpdateForRevival invoiceDisplayControlUpdateForRevival;

    private final OtherProductServicePaymentContractUpdateForRevival
            otherProductServicePaymentContractUpdateForRevival;

    private final RequestContext requestContext;

    private final String applicableEndsPaymentContractStatus;

    private final String invalidPaymentContractStatus;

    /**
     * 概要: 必須依存と検索用定数を初期化する。
     *
     * @param svsihkrDataAccess サービス契約・支払契約関連 DA
     * @param sihakyDataAccess 支払契約 DA
     * @param gssihhhDataAccess ガス最終精算支払方法指示 DA
     * @param servicePaymentContractUpdateForRevival SV-114 API
     * @param paymentContractInfoUpdateForRevival SV-095 API
     * @param companyTransferUpdate SV-030 API
     * @param invoiceDisplayControlUpdateForRevival SV-113 API
     * @param otherProductServicePaymentContractUpdateForRevival SV-116 API
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PaymentContractRevivalImpl(
            SvsihkrDataAccess svsihkrDataAccess,
            SihakyDataAccess sihakyDataAccess,
            GssihhhDataAccess gssihhhDataAccess,
            ServicePaymentContractUpdateForRevival servicePaymentContractUpdateForRevival,
            PaymentContractInfoUpdateForRevival paymentContractInfoUpdateForRevival,
            CompanyTransferUpdate companyTransferUpdate,
            InvoiceDisplayControlUpdateForRevival invoiceDisplayControlUpdateForRevival,
            OtherProductServicePaymentContractUpdateForRevival otherProductServicePaymentContractUpdateForRevival,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccess, "svsihkrDataAccess");
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccess, "sihakyDataAccess");
        this.gssihhhDataAccess = Objects.requireNonNull(gssihhhDataAccess, "gssihhhDataAccess");
        this.servicePaymentContractUpdateForRevival = Objects.requireNonNull(
                servicePaymentContractUpdateForRevival,
                "servicePaymentContractUpdateForRevival");
        this.paymentContractInfoUpdateForRevival = Objects.requireNonNull(
                paymentContractInfoUpdateForRevival,
                "paymentContractInfoUpdateForRevival");
        this.companyTransferUpdate = Objects.requireNonNull(
                companyTransferUpdate,
                "companyTransferUpdate");
        this.invoiceDisplayControlUpdateForRevival = Objects.requireNonNull(
                invoiceDisplayControlUpdateForRevival,
                "invoiceDisplayControlUpdateForRevival");
        this.otherProductServicePaymentContractUpdateForRevival = Objects.requireNonNull(
                otherProductServicePaymentContractUpdateForRevival,
                "otherProductServicePaymentContractUpdateForRevival");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applicableEndsPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.invalidPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
    }

    @Override
    public PaymentContractRevivalOutput execute(Long svkyNo, String syaKbn) {
        LocalDate businessDate = requireBusinessDate();
        SvsihkrRecord servicePaymentRelation = requireServicePaymentRelation(svkyNo, businessDate);
        SihakyRecord paymentContract = requirePaymentContract(
                servicePaymentRelation.sihKiyNo(),
                businessDate);

        servicePaymentContractUpdateForRevival.execute(
                servicePaymentRelation.sihKiyNo(),
                svkyNo,
                servicePaymentRelation.skno(),
                paymentContract.sikSvcShbt());

        boolean sameDayRevival = businessDate.equals(paymentContract.tyenYmd());
        if (sameDayRevival) {
            paymentContractInfoUpdateForRevival.execute(
                    paymentContract.sihHuhuShbt(),
                    paymentContract.sihKiyNo(),
                    paymentContract.skno());
            companyTransferUpdate.execute(
                    syaKbn,
                    servicePaymentRelation.sihKiyNo(),
                    MAX_DATE,
                    businessDate);
        }

        if (sameDayRevival && paymentContract.skshHyjsKrno() != null) {
            invoiceDisplayControlUpdateForRevival.execute(paymentContract.skshHyjsKrno());
        }

        gssihhhDataAccess.deleteBySvkyNo(svkyNo, null);
        otherProductServicePaymentContractUpdateForRevival.execute(
                svkyNo,
                servicePaymentRelation.tyenYmd(),
                paymentContract.sikSvcShbt());

        return new PaymentContractRevivalOutput(new PaymentContractRevivalSihakyInf(
                paymentContract.sihKiyNo(),
                paymentContract.sikSvcShbt(),
                paymentContract.sikKrno()));
    }

    private LocalDate requireBusinessDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    private SvsihkrRecord requireServicePaymentRelation(Long svkyNo, LocalDate businessDate) {
        return svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndTyenYmdAndSihKiyJtkb(
                        svkyNo,
                        businessDate,
                        applicableEndsPaymentContractStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svsihkrDataAccess.tableName(),
                        FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_SIH_KIY_JTKB.identifier(),
                        new FlatJsonBuilder()
                                .put("svkyNo", svkyNo)
                                .put("knskKjnYmd", businessDate)
                                .put("sihKiyJtkb", applicableEndsPaymentContractStatus)));
    }

    private SihakyRecord requirePaymentContract(Long sihKiyNo, LocalDate businessDate) {
        return sihakyDataAccess
                .findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDescAndSknoDesc(
                        sihKiyNo,
                        businessDate,
                        invalidPaymentContractStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sihakyDataAccess.tableName(),
                        FIND_BY_SIH_KIY_NO_AND_KNSK_KJN_YMD_AND_NOT_SIH_KIY_JTKB_ORDER_BY_TKYS_YMD_DESC_AND_SKNO_DESC.identifier(),
                        new FlatJsonBuilder()
                                .put("sihKiyNo", sihKiyNo)
                                .put("knskKjnYmd", businessDate)
                                .put("sihKiyJtkb", invalidPaymentContractStatus)));
    }
}
