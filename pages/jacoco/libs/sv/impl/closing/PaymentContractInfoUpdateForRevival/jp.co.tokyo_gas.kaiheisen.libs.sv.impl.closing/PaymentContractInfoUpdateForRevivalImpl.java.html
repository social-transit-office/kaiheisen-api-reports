/**
 * SV-095（支払契約関連情報更新（契約復活））のサービス実装.
 * 主な仕様:
 * - 支払契約を契約復活状態へ更新した後、支払方法種別に応じて
 *   関連情報を 1 系統のみ更新する.
 * - 共通更新および分岐先更新の更新件数が 0 件の場合は
 *   データ不整合例外を送出する.
 * - 監査情報には固定更新ユーザー ID と
 *   RequestContext 由来の報告元機能 ID を設定する.
 */
public class PaymentContractInfoUpdateForRevivalImpl
    implements PaymentContractInfoUpdateForRevival {

    /** 固定の更新ユーザー ID. */
    static final String SERVICE_USER_ID = "KHSS095";

    /** 契約復活時に設定する最大日付. */
    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    /** 支払契約更新を行う DA. */
    private final SihakyDataAccess sihakyDataAccess;

    /** 払込関連情報更新を行う DA. */
    private final HrikmiDataAccess hrikmiDataAccess;

    /** 口座振替関連情報更新を行う DA. */
    private final KuzfrkDataAccess kuzfrkDataAccess;

    /** カード払い関連情報更新を行う DA. */
    private final CrdhriDataAccess crdhriDataAccess;

    /** 電子請求関連情報更新を行う DA. */
    private final DsqkjhDataAccess dsqkjhDataAccess;

    /** 監査値とコンテキスト値を参照するリクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 報告システム種別のデコードに使用する定数サービス. */
    private final KaiheisenConstantService kaiheisenConstantService;

    /** 契約復活時に設定する支払契約状態の定数値. */
    private final String applyingPaymentContractStatus;

    /** 払込みを表す検証済み支払方法種別コード. */
    private final String transferSlipPaymentMethodType;

    /** 口座振替を表す検証済み支払方法種別コード. */
    private final String accountTransferPaymentMethodType;

    /** カード払いを表す検証済み支払方法種別コード. */
    private final String cardPaymentMethodType;

    /** 電子請求を表す検証済み支払方法種別コード. */
    private final String electronicBillingPaymentMethodType;

    /**
     * SV-095 実装を初期化する.
     *
     * @param paymentContractDataAccess 支払契約更新用 DA
     * @param transferSlipDataAccess 払込関連情報更新用 DA
     * @param accountTransferDataAccess 口座振替関連情報更新用 DA
     * @param cardPaymentDataAccess カード払い関連情報更新用 DA
     * @param electronicBillingDataAccess 電子請求関連情報更新用 DA
     * @param requestContextHolder リクエストコンテキスト
     * @param reportingConstantService Kaiheisen 定数サービス
     * @param paymentContractConstantService Cirius 定数サービス
     */
    public PaymentContractInfoUpdateForRevivalImpl(
            final SihakyDataAccess paymentContractDataAccess,
            final HrikmiDataAccess transferSlipDataAccess,
            final KuzfrkDataAccess accountTransferDataAccess,
            final CrdhriDataAccess cardPaymentDataAccess,
            final DsqkjhDataAccess electronicBillingDataAccess,
            final RequestContext requestContextHolder,
            final KaiheisenConstantService reportingConstantService,
            final CiriusConstantService paymentContractConstantService) {
        this.sihakyDataAccess = Objects.requireNonNull(
            paymentContractDataAccess,
            "paymentContractDataAccess");
        this.hrikmiDataAccess = Objects.requireNonNull(
            transferSlipDataAccess,
            "transferSlipDataAccess");
        this.kuzfrkDataAccess = Objects.requireNonNull(
            accountTransferDataAccess,
            "accountTransferDataAccess");
        this.crdhriDataAccess = Objects.requireNonNull(
            cardPaymentDataAccess,
            "cardPaymentDataAccess");
        this.dsqkjhDataAccess = Objects.requireNonNull(
            electronicBillingDataAccess,
            "electronicBillingDataAccess");
        this.requestContext = Objects.requireNonNull(
                requestContextHolder,
                "requestContextHolder");
        this.kaiheisenConstantService = Objects.requireNonNull(
                reportingConstantService,
                "reportingConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(
                        paymentContractConstantService,
                        "paymentContractConstantService");
        this.applyingPaymentContractStatus = RequiredConstantResolver
            .requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.transferSlipPaymentMethodType = RequiredConstantResolver
            .requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.PAYMENT);
        this.accountTransferPaymentMethodType = RequiredConstantResolver
            .requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ACCOUNT_TRANSFER);
        this.cardPaymentMethodType = RequiredConstantResolver
            .requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.CARD_PAYMENT);
        this.electronicBillingPaymentMethodType = RequiredConstantResolver
            .requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ELECTRONIC_BILLING);
    }

    /**
     * 支払契約と支払方法別の関連情報を契約復活状態へ更新する.
     *
     * @param sihHuhuShbt 支払方法種別
     * @param sihKiyNo 支払契約番号
     * @param skno 世代管理番号
     */
    @Override
    public void execute(
            final String sihHuhuShbt,
            final Long sihKiyNo,
            final Integer skno) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(
                SERVICE_USER_ID,
                decodedOriginFunctionId);

        updatePaymentContract(
                sihKiyNo,
                skno,
                auditContext,
                decodedOriginFunctionId);

        if (transferSlipPaymentMethodType.equals(sihHuhuShbt)) {
            updateTransferSlipInfo(
                    sihKiyNo,
                    skno,
                    auditContext,
                    decodedOriginFunctionId);
            return;
        }
        if (accountTransferPaymentMethodType.equals(sihHuhuShbt)) {
            updateAccountTransferInfo(
                    sihKiyNo,
                    skno,
                    auditContext,
                    decodedOriginFunctionId);
            return;
        }
        if (cardPaymentMethodType.equals(sihHuhuShbt)) {
            updateCardInfo(
                    sihKiyNo,
                    skno,
                    auditContext,
                    decodedOriginFunctionId);
            return;
        }
        if (electronicBillingPaymentMethodType.equals(sihHuhuShbt)) {
            updateElectronicBillingInfo(
                    sihKiyNo,
                    skno,
                    auditContext,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 支払契約を契約復活状態へ更新する.
     *
     * @param sihKiyNo 支払契約番号
     * @param skno 世代管理番号
     * @param auditContext 更新監査情報
     * @param decodedOriginFunctionId デコード済み報告元機能 ID
     */
    private void updatePaymentContract(
            final Long sihKiyNo,
            final Integer skno,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        int updatedCount = sihakyDataAccess.updateBySihKiyNoAndSknoAndRecUpdJnjNo(
            new SihakyWriteRecord(
                sihKiyNo,
                skno,
                null,
                MAX_DATE,
                applyingPaymentContractStatus,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null),
            Set.of(
                SihakyColumn.TYEN_YMD,
                SihakyColumn.SIH_KIY_JTKB),
            new SihakySknoJnjUpdateKey(
                sihKiyNo,
                skno,
                null),
            auditContext);
        ensureUpdatedCount(updatedCount,
                sihakyDataAccess.tableName(),
                SihakyDataAccess.CallSpecs
                    .UPDATE_BY_SIH_KIY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO
                    .identifier(),
                new FlatJsonBuilder()
                    .put("sihKiyNo", sihKiyNo)
                    .put("skno", skno)
                    .put("recUpdJnjNo", null)
                    .put("tyenYmd", MAX_DATE)
                    .put("sihKiyJtkb", applyingPaymentContractStatus)
                    .put("updUsrId", SERVICE_USER_ID)
                    .put("updKtnKinoId", decodedOriginFunctionId));
    }

    /**
     * 払込み向けの関連情報を更新する.

    /**
     * 更新件数が 0 件の場合にデータ不整合例外を送出する.
     *
     * @param updatedCount 更新件数
     * @param tableName 対象テーブル名
     * @param daIdentifier DA 識別子
     * @param parameterJsonBuilder パラメータ JSON 構築子
     */
    private void ensureUpdatedCount(
            final int updatedCount,
            final String tableName,
            final String daIdentifier,
            final FlatJsonBuilder parameterJsonBuilder) {
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    tableName,
                    daIdentifier,
                    parameterJsonBuilder);
        }
    }

    /**
     * RequestContext の報告システム種別を報告元機能 ID へデコードする.
     *
     * @return デコード済み報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                .orElseThrow(
                    () -> new IllegalStateException(
                        "RequestContext.hukkSysShbt is required")));
    }
