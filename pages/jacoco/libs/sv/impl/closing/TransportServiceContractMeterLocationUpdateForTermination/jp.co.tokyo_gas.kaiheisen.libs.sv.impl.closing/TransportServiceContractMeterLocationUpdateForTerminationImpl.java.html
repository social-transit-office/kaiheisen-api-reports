/**
 * 概要: 契約終了時に託送サービス契約と託送サービス契約関連を終了更新する SV 実装。
 * 主な仕様:
 * - 初期入力で指定された託送サービス契約と関連情報を契約終了状態へ更新する。
 * - 専用メーター情報リストごとに適用中の関連情報を取得し、取得結果があれば追加の契約終了更新を行う。
 * - ループ内の契約更新件数が 0 件の場合は DataInconsistencyException を送出する。
 * 既知の制約:
 * - hukkSysShbt、correlationId、systemDatetime は RequestContext から参照する。
 */
public final class TransportServiceContractMeterLocationUpdateForTerminationImpl
        implements TransportServiceContractMeterLocationUpdateForTermination {

    /** 監査項目に設定する更新ユーザー ID。 */
    static final String SERVICE_USER_ID = "KHSS043";

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applyingRelationStatus;

    private final String applicableEndsRelationStatus;

    private final String underContractStatus;

    private final String endOfContractStatus;

    /**
     * 必須依存と定数を初期化する。
     *
     * @param tkssvkyDataAccess 託送サービス契約 DA
     * @param tskgmstDataAccess 託送サービス契約・ガスメーター設置場所関連 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public TransportServiceContractMeterLocationUpdateForTerminationImpl(
            final TkssvkyDataAccess tkssvkyDataAccess,
            final TskgmstDataAccess tskgmstDataAccess,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applyingRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.applicableEndsRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLICABLE_ENDS);
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.endOfContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.END_OF_CONTRACT);
    }

    @Override
    public void execute(
            final Long gasMtstNo,
            final LocalDate tyenYmd,
            final String tksuSvkeJysb,
            final TskgmstInf tskgmstInf,
            final TkssvkyInf tkssvkyInf,
            final List<SnyInf> snyInfList) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(
                SERVICE_USER_ID,
                decodedOriginFunctionId);

        updateInitialTransportContract(
                tkssvkyInf,
                tyenYmd,
                tksuSvkeJysb,
                auditContext);
        updateInitialRelation(
                gasMtstNo,
                tskgmstInf,
                tyenYmd,
                auditContext);

        for (SnyInf snyInf : snyInfList) {
            updateDedicatedMeterRelations(
                    snyInf,
                    tyenYmd,
                    tksuSvkeJysb,
                    auditContext,
                    decodedOriginFunctionId);
        }
    }

    /**
     * 概要: 初期入力の託送サービス契約を契約終了状態へ更新する。
     */
    private void updateInitialTransportContract(
            final TkssvkyInf tkssvkyInf,
            final LocalDate tyenYmd,
            final String tksuSvkeJysb,
            final AuditContext auditContext) {
        tkssvkyDataAccess.updateByTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TkssvkyWriteRecord(
                        tkssvkyInf.tksuSvkyNo(),
                        null,
                        endOfContractStatus,
                        null,
                        tksuSvkeJysb,
                        null,
                        tyenYmd,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(
                        TkssvkyColumn.TKSU_SVKY_JTKB,
                        TkssvkyColumn.TKSU_SVKE_JYSB,
                        TkssvkyColumn.TYEN_YMD),
                new TkssvkySknoJnjUpdateKey(
                        tkssvkyInf.tksuSvkyNo(),
                        toIntegerSkno(tkssvkyInf.skno()),
                        tkssvkyInf.recUpdJnjNo()),
                auditContext);
    }

    /**
     * 概要: 初期入力の託送サービス契約関連を適用終了状態へ更新する。
     */
    private void updateInitialRelation(
            final Long gasMtstNo,
            final TskgmstInf tskgmstInf,
            final LocalDate tyenYmd,
            final AuditContext auditContext) {
        tskgmstDataAccess.updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TskgmstWriteRecord(
                        gasMtstNo,
                        tskgmstInf.tksuSvkyNo(),
                        tskgmstInf.skno(),
                        applicableEndsRelationStatus,
                        null,
                        tyenYmd),
                EnumSet.of(
                        TskgmstColumn.TKSU_SVKK_JTKB,
                        TskgmstColumn.TYEN_YMD),
                new TskgmstUpdateKey(
                        gasMtstNo,
                        tskgmstInf.tksuSvkyNo(),
                        tskgmstInf.skno(),
                        tskgmstInf.recUpdJnjNo()),
                auditContext);
    }

    /**
     * 概要: 専用メーター情報に紐づく適用中関連を追加で終了更新する。
     */
    private void updateDedicatedMeterRelations(
            final SnyInf snyInf,
            final LocalDate tyenYmd,
            final String tksuSvkeJysb,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        Optional<TskgmstRecord> applyingRelation = tskgmstDataAccess
                .findByGasMtstNoAndTksuSvkkJtkb(
                        snyInf.gmtsetdInf().gasMtstNo(),
                        applyingRelationStatus);
        if (applyingRelation.isPresent()) {
            updateDedicatedTransportContract(
                    applyingRelation.get(),
                    tyenYmd,
                    tksuSvkeJysb,
                    auditContext,
                    decodedOriginFunctionId);
            updateDedicatedRelation(
                    applyingRelation.get(),
                    tyenYmd,
                    auditContext);
        }
    }

    /**
     * 概要: ループ内で取得した託送サービス契約を契約終了状態へ更新する。
     */
    private void updateDedicatedTransportContract(
            final TskgmstRecord applyingRelation,
            final LocalDate tyenYmd,
            final String tksuSvkeJysb,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        int updatedCount = tkssvkyDataAccess.updateByTksuSvkyNoAndTksuSvkyJtkb(
                new TkssvkyWriteRecord(
                        applyingRelation.tksuSvkyNo(),
                        null,
                        endOfContractStatus,
                        null,
                        tksuSvkeJysb,
                        null,
                        tyenYmd,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(
                        TkssvkyColumn.TKSU_SVKY_JTKB,
                        TkssvkyColumn.TKSU_SVKE_JYSB,
                        TkssvkyColumn.TYEN_YMD),
                new TkssvkyJtkbUpdateKey(
                        applyingRelation.tksuSvkyNo(),
                        underContractStatus),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    tkssvkyDataAccess.tableName(),
                    TkssvkyDataAccess.CallSpecs.UPDATE_BY_TKSU_SVKY_NO_AND_TKSU_SVKY_JTKB
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("tksuSvkyNo", applyingRelation.tksuSvkyNo())
                            .put("tksuSvkyJtkb", underContractStatus)
                            .put("newTksuSvkyJtkb", endOfContractStatus)
                            .put("tksuSvkeJysb", tksuSvkeJysb)
                            .put("tyenYmd", tyenYmd)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    /**
     * 概要: ループ内で取得した託送サービス契約関連を適用終了状態へ更新する。
     */
    private void updateDedicatedRelation(
            final TskgmstRecord applyingRelation,
            final LocalDate tyenYmd,
            final AuditContext auditContext) {
        tskgmstDataAccess.updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TskgmstWriteRecord(
                        applyingRelation.gasMtstNo(),
                        applyingRelation.tksuSvkyNo(),
                        applyingRelation.skno(),
                        applicableEndsRelationStatus,
                        applyingRelation.tkysYmd(),
                        tyenYmd),
                EnumSet.of(
                        TskgmstColumn.TKSU_SVKK_JTKB,
                        TskgmstColumn.TYEN_YMD),
                new TskgmstUpdateKey(
                        applyingRelation.gasMtstNo(),
                        applyingRelation.tksuSvkyNo(),
                        applyingRelation.skno(),
                        applyingRelation.recUpdJnjNo()),
                auditContext);
    }

    /**
     * 概要: RequestContext から起点機能 ID を解決する。
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt().orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.hukkSysShbt is required")));
    }

    /**
     * 概要: Long の世代管理番号を DA update key 用の Integer へ変換する。
     */
    private Integer toIntegerSkno(final Long skno) {
        return Math.toIntExact(skno);
    }
}
