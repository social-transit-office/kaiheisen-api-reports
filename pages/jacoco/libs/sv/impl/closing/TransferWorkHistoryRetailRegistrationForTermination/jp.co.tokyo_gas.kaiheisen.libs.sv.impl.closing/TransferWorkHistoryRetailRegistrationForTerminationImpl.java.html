package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.closing;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.TransferWorkHistoryRetailRegistrationForTermination;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransferWorkHistorySogoTmtrInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.ClosingCancellationHistoryDetailCodeRetailAcquisition;

/**
 * 概要: SV-036（異動・作業履歴（共通）＜小売＞登録（契約終了））の実装。
 * 主な仕様:
 * - SV-037 で異動・作業履歴業務詳細コード（小売）を取得し、他-004 の登録リクエストを編集して実行する。
 * - 元メーター分の初回登録に加え、総合他ガスメーター情報リスト件数分の繰返し登録を行う。
 * - 異動・作業履歴報告ユーザーIDは、入力値が非 null かつ異動・作業履歴業務コード（小売）が `CCBG63` 以外の場合のみ他-004 へ設定する。
 * - 繰返し登録対象の `svmstkrInf.mtstNo` が未設定の場合、他-004 の登録依頼用お客さま番号は空文字で登録する。
 * 既知の制約:
 * - public メソッド引数の null / blank 判定は追加せず、登録リクエスト生成に必要な RequestContext.systemDatetime を fail-fast する。
 * - 詳細情報は設計どおり空配列で登録する。
 */
public final class TransferWorkHistoryRetailRegistrationForTerminationImpl
        implements TransferWorkHistoryRetailRegistrationForTermination {

    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private static final DateTimeFormatter TIME_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    private final ClosingCancellationHistoryDetailCodeRetailAcquisition detailCodeAcquisition;

    private final IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter;

    private final RequestContext requestContext;

    private final String caseHistoryResultTransfer;

    private final String itemHistoryRegistrationDestinationCustomer;

    private final String endOfContract;

    private final String closingRegistration;

    private final String transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable;

    private final String customerInfoAcquisitionPossible;

    /**
     * 概要: SV-036 実行に必要な依存と定数を初期化する。
     * 主な仕様:
     * - 下位 SV、RequestContext、他-004 adapter を保持する。
     * - 判定・固定値設定に必要な CIRIUS 定数はコンストラクタ初期化時に fail-fast で解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * - 必須定数が未定義の場合はコンストラクタ初期化時点で失敗する。
     *
     * @param detailCodeAcquisition SV-037
     * @param ciriusAdapter 他-004 adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public TransferWorkHistoryRetailRegistrationForTerminationImpl(
            ClosingCancellationHistoryDetailCodeRetailAcquisition detailCodeAcquisition,
            IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.detailCodeAcquisition = Objects.requireNonNull(detailCodeAcquisition, "detailCodeAcquisition");
        this.ciriusAdapter = Objects.requireNonNull(ciriusAdapter, "ciriusAdapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService nonNullCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.caseHistoryResultTransfer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.CaseHistory.TYPE,
                ConstantLookupKeys.Cirius.CaseHistory.RESULT_TRANSFER);
        this.itemHistoryRegistrationDestinationCustomer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.TYPE,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.CUSTOMER);
        this.endOfContract = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.END_OF_CONTRACT);
        this.closingRegistration = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION);
        this.transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.NOT_APPLICABLE_TO_DETAILS_ONLY);
        this.customerInfoAcquisitionPossible = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PossibilityFlag.TYPE,
                ConstantLookupKeys.Cirius.PossibilityFlag.POSSIBLE);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate tyenYmd,
            String svkyEndJysb,
            List<TransferWorkHistorySogoTmtrInf> sogoTmtrInfList,
            String isrrGymCd,
            String isrrHusrId,
            String isrrHkunKj,
            String isrrHkkrKbn) {
        LocalDateTime systemDatetime = requireSystemDatetime();
        String detailCode = detailCodeAcquisition.execute(svkyEndJysb).isrrGmssCd();
        String reportUserId = resolveReportUserId(isrrGymCd, isrrHusrId);
        ciriusAdapter.register(toRequest(
                gasMtstNo,
                tyenYmd,
                detailCode,
                reportUserId,
                isrrHkunKj,
                isrrHkkrKbn,
                systemDatetime));
        for (TransferWorkHistorySogoTmtrInf sogoTmtrInf : normalizeSogoTmtrInfList(sogoTmtrInfList)) {
            ciriusAdapter.register(toRequest(
                    resolveRepeatGasMtstNo(sogoTmtrInf),
                    tyenYmd,
                    detailCode,
                    reportUserId,
                    isrrHkunKj,
                    isrrHkkrKbn,
                    systemDatetime));
        }
    }

    /**
     * 概要: 他-004 の異動・作業履歴報告ユーザーＩＤへ設定する値を判定する。
     *
     * @param isrrGymCd 異動・作業履歴業務コード（小売）
     * @param reportUserId 異動・作業履歴報告ユーザーＩＤ
     * @return 条件成立時は入力ユーザーＩＤ、非成立時は未設定扱いの null
     */
    private String resolveReportUserId(String isrrGymCd, String reportUserId) {
        if (reportUserId == null || closingRegistration.equals(isrrGymCd)) {
            return null;
        }
        return reportUserId;
    }

    /**
     * 概要: SV 入力を他-004 リクエストへ変換する。
     *
     * @param gasMtstNo メーター設置場所番号
     * @param tyenYmd 適用終了年月日
     * @param detailCode 異動・作業履歴業務詳細コード（小売）
     * @param reportUserId 報告ユーザーID。設定対象外の場合は null
     * @param reportUserName 報告ユーザー名漢字
     * @param reportRouteCode 報告経路区分
     * @param systemDatetime システム日時
     * @return 他-004 リクエスト
     */
    private IdousagyourirekiKouriJouhouRequest toRequest(
            Long gasMtstNo,
            LocalDate tyenYmd,
            String detailCode,
            String reportUserId,
            String reportUserName,
            String reportRouteCode,
            LocalDateTime systemDatetime) {
        return new IdousagyourirekiKouriJouhouRequest(
                new IdousagyourirekiKouriJouhouRequest.IdousagyourirekiKouriJouhouRenkei(
                        new IdousagyourirekiKouriJouhouRequest.IdousagyourirekigyoumuKyoutsuuKouri(
                                toCustomerNumber(gasMtstNo),
                                caseHistoryResultTransfer,
                                itemHistoryRegistrationDestinationCustomer,
                                nullToEmpty(endOfContract),
                                nullToEmpty(detailCode),
                                nullToEmpty(reportRouteCode),
                                transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable,
                                toDateText(systemDatetime.toLocalDate()),
                                toTimeText(systemDatetime),
                                toDateText(tyenYmd),
                                nullToEmpty(reportUserId),
                                nullToEmpty(reportUserName),
                                ""),
                        List.of(),
                        customerInfoAcquisitionPossible,
                        "",
                        toDateText(tyenYmd)));
    }

    /**
     * 概要: RequestContext から必須のシステム日時を取得する。
     *
     * @return RequestContext の必須システム日時
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 総合他ガスメーター情報リストを走査可能な値へ正規化する。
     *
     * @param sogoTmtrInfList 総合他ガスメーター情報リスト
     * @return null の場合は空リスト
     */
    private List<TransferWorkHistorySogoTmtrInf> normalizeSogoTmtrInfList(
            List<TransferWorkHistorySogoTmtrInf> sogoTmtrInfList) {
        return sogoTmtrInfList == null ? List.of() : sogoTmtrInfList;
    }

    /**
     * 概要: 繰返し登録対象のメーター設置場所番号を取得する。
     *
     * @param sogoTmtrInf 総合他ガスメーター情報
     * @return メーター設置場所番号。未設定時は null
     */
    private Long resolveRepeatGasMtstNo(TransferWorkHistorySogoTmtrInf sogoTmtrInf) {
        if (sogoTmtrInf == null || sogoTmtrInf.svmstkrInf() == null) {
            return null;
        }
        return sogoTmtrInf.svmstkrInf().mtstNo();
    }

    /**
     * 概要: メーター設置場所番号を 11 桁文字列へ変換する。
     *
     * @param gasMtstNo メーター設置場所番号
     * @return 11 桁文字列
     */
    private String toCustomerNumber(Long gasMtstNo) {
        if (gasMtstNo == null) {
            return "";
        }
        return String.format("%011d", gasMtstNo);
    }

    /**
     * 概要: 日付を yyyyMMdd 文字列へ変換する。
     *
     * @param value 日付
     * @return yyyyMMdd 文字列
     */
    private String toDateText(LocalDate value) {
        if (value == null) {
            return "";
        }
        return value.format(DATE_FORMATTER);
    }

    /**
     * 概要: 文字列の null を空文字へ変換する。
     *
     * @param value 変換対象
     * @return 非 null の文字列
     */
    private String nullToEmpty(String value) {
        return value == null ? "" : value;
    }

    /**
     * 概要: 時刻を HHmmss 文字列へ変換する。
     *
     * @param value システム日時
     * @return HHmmss 文字列
     */
    private String toTimeText(LocalDateTime value) {
        return value.format(TIME_FORMATTER);
    }

}
