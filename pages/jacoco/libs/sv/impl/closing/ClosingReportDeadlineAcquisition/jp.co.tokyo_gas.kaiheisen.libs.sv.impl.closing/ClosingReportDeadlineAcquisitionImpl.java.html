package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.closing;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dktksry.DktksryDataAccess;

/**
 * 概要: 閉栓報告可能期限日取得サービス（SV-068）の実装.
 * 主な仕様:
 * - 基準日、該当年月日、各DA取得日を比較し、優先順位に従って閉栓報告可能期限日を決定する。
 * - 検針結果作成事由種別の判定およびDA固定値は CIRIUS 共通定数から解決する。
 * - 基準日は execute 引数の検針年月日を使用し、DA検索上限日は RequestContext の systemDatetime を使用する。
 * 既知の制約:
 * - 入力妥当性検証と例外変換は本クラスでは行わず、上位層責務とする。
 * - 下位層で発生した例外は握りつぶさずに上位へ伝播する。
 */
public class ClosingReportDeadlineAcquisitionImpl
    implements ClosingReportDeadlineAcquisition {

    /** 導管通知確定使用量データアクセス. */
    private final DktksryDataAccess dktksryDataAccess;

    /** 確定前ガスメーター検針結果データアクセス. */
    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    /** ガスメーター検針結果＜導管＞データアクセス. */
    private final GmkskkdDataAccess gmkskkdDataAccess;

    /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 定例のSW開始を表す検針結果作成事由種別. */
    private final String reasonRegularSwStart;

    /** 臨時のSW開始を表す検針結果作成事由種別. */
    private final String reasonTemporarySwStart;

    /** 異動受付みなし検針を表す検針結果作成事由種別. */
    private final String reasonMeterReadingWithoutTransferAcceptance;

    /** 災害時閉栓関連の検針結果作成事由種別一覧. */
    private final List<String> disasterClosingReasons;

    /**
     * 概要: 依存を注入するコンストラクタ.
     * 主な仕様:
     * - 3件のDA、定数サービス、RequestContext を保持する。
     * - CIRIUS定数サービスから期限日判定に必要な検針結果作成事由種別を解決して保持する。
     * 既知の制約:
     * - 必須定数が未登録の場合は初期化時に例外が発生する。
     *
     * @param dktksryDataAccessArg 導管通知確定使用量データアクセス
     * @param ktmgmkkDataAccessArg 確定前ガスメーター検針結果データアクセス
     * @param gmkskkdDataAccessArg ガスメーター検針結果＜導管＞
     *     データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param requestContextArg リクエストコンテキスト
     */
    public ClosingReportDeadlineAcquisitionImpl(
            final DktksryDataAccess dktksryDataAccessArg,
            final KtmgmkkDataAccess ktmgmkkDataAccessArg,
            final GmkskkdDataAccess gmkskkdDataAccessArg,
            final CiriusConstantService ciriusConstantService,
            final RequestContext requestContextArg) {
        CiriusConstantService nonNullCiriusConstantService =
            Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dktksryDataAccess =
            Objects.requireNonNull(
                dktksryDataAccessArg,
                "dktksryDataAccessArg");
        this.ktmgmkkDataAccess =
            Objects.requireNonNull(
                ktmgmkkDataAccessArg,
                "ktmgmkkDataAccessArg");
        this.gmkskkdDataAccess =
            Objects.requireNonNull(
                gmkskkdDataAccessArg,
                "gmkskkdDataAccessArg");
        this.requestContext =
            Objects.requireNonNull(requestContextArg, "requestContextArg");
        this.reasonRegularSwStart =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults
                    .REGULAR_SW_START);
        this.reasonTemporarySwStart =
                RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults
                    .TEMPORARY_SW_START);
        this.reasonMeterReadingWithoutTransferAcceptance =
                RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius
                    .ReasonTypeForCreatingMeterReadingResults
                    .METER_READING_WITHOUT_TRANSFER_ACCEPTANCE);
        this.disasterClosingReasons =
                List.of(
                        RequiredConstantResolver.requireCiriusConstant(
                    nonNullCiriusConstantService,
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .TYPE,
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .RESTORATION_OF_SHUTOFF_IN_CASE_OF_DISASTER),
                        RequiredConstantResolver.requireCiriusConstant(
                            nonNullCiriusConstantService,
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .TYPE,
                    ConstantLookupKeys.Cirius
                        .ReasonTypeForCreatingMeterReadingResults
                        .CLOSED_IN_CASE_OF_DISASTER));
    }

    /**
     * 概要: SV-068を実行し、閉栓報告可能期限日を返却する.
     * 主な仕様:
     * - 基準日、該当年月日、3件のDA取得日を比較して優先順位順に返却値を決定する。
     * - 判定順序は設計書 step 6 の 6.a → 6.b → 6.c → 6.d → 6.e に一致させる。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @param knkkSsjyShbt 検針結果作成事由種別
     * @return 閉栓報告可能期限日
     */
    @Override
    public ClosingReportDeadlineAcquisitionOutput execute(
            final Long gasMtstNo,
            final LocalDate knsYmd,
            final String knkkSsjyShbt) {
        LocalDate baseDate = knsYmd;
        LocalDate currentDate = resolveCurrentDateForDaRangeTo();
        LocalDate dktksryDate =
                findDktksryDate(gasMtstNo, baseDate, currentDate);
        if (dktksryDate != null && dktksryDate.isAfter(baseDate)) {
            return new ClosingReportDeadlineAcquisitionOutput(dktksryDate);
        }
        LocalDate deemedDate = resolveApplicableDate(baseDate, knkkSsjyShbt);
        if (deemedDate != null && deemedDate.isAfter(baseDate)) {
            return new ClosingReportDeadlineAcquisitionOutput(deemedDate);
        }
        LocalDate ktmgmkkDate =
                findKtmgmkkDate(gasMtstNo, baseDate, currentDate);
        if (ktmgmkkDate != null && ktmgmkkDate.isAfter(baseDate)) {
            return new ClosingReportDeadlineAcquisitionOutput(ktmgmkkDate);
        }
        LocalDate gmkskkdDate =
                findGmkskkdDate(gasMtstNo, baseDate, currentDate);
        if (gmkskkdDate != null && gmkskkdDate.isAfter(baseDate)) {
            return new ClosingReportDeadlineAcquisitionOutput(gmkskkdDate);
        }
        return new ClosingReportDeadlineAcquisitionOutput(baseDate);
    }

    /**
     * 概要: DA検索上限日に用いる実日付を解決する.
     * 主な仕様:
     * - RequestContext の systemDatetime の年月日を返却する。
        * - systemDatetime の設定と妥当性確認は上位層責務とし、本メソッドでは補完を行わない。
     *
     * @return DA検索上限日に用いる実日付
     */
    private LocalDate resolveCurrentDateForDaRangeTo() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    /**
     * 概要: 導管通知確定使用量の取得結果から検針年月日を取得する.
     * 主な仕様:
     * - DA-089-R005 を固定事由で呼び出す。
     * - 取得結果が存在する場合は knsYmd を返却し、未取得時は null を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmdFrom 検針年月日（開始）
     * @param knsYmdTo 検針年月日（終了）
     * @return 取得した検針年月日
     */
    private LocalDate findDktksryDate(
            final Long gasMtstNo,
            final LocalDate knsYmdFrom,
            final LocalDate knsYmdTo) {
        Optional<DktksryRecord> result =
                dktksryDataAccess.findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
                        gasMtstNo,
                        knsYmdFrom,
                        knsYmdTo,
                        reasonMeterReadingWithoutTransferAcceptance);
        return result.map(DktksryRecord::knsYmd).orElse(null);
    }

    /**
     * 概要: 検針結果作成事由種別に応じて該当年月日を算出する.
     * 主な仕様:
     * - 定例または臨時の SW 開始は基準日の翌日を返却する。
     * - 上記以外は null を返却する。
     *
     * @param baseDate 基準日
     * @param knkkSsjyShbt 検針結果作成事由種別
     * @return 該当年月日
     */
    private LocalDate resolveApplicableDate(
            final LocalDate baseDate,
            final String knkkSsjyShbt) {
        if (reasonRegularSwStart.equals(knkkSsjyShbt)
                || reasonTemporarySwStart.equals(knkkSsjyShbt)) {
            return baseDate.plusDays(1);
        }
        return null;
    }

    /**
     * 概要: 確定前ガスメーター検針結果から検針年月日を取得する.
     * 主な仕様:
     * - DA-052-R005 を災害時閉栓事由一覧で呼び出す。
     * - 取得結果が存在する場合は knsYmd を返却し、未取得時は null を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmdFrom 検針年月日（開始）
     * @param knsYmdTo 検針年月日（終了）
     * @return 取得した検針年月日
     */
    private LocalDate findKtmgmkkDate(
            final Long gasMtstNo,
            final LocalDate knsYmdFrom,
            final LocalDate knsYmdTo) {
        Optional<KtmgmkkRecord> result =
                ktmgmkkDataAccess.findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
                        gasMtstNo,
                        knsYmdFrom,
                        knsYmdTo,
                        disasterClosingReasons);
        return result.map(KtmgmkkRecord::knsYmd).orElse(null);
    }

    /**
     * 概要: ガスメーター検針結果＜導管＞から検針年月日を取得する.
     * 主な仕様:
     * - DA-027-R006 を災害時閉栓事由一覧で呼び出す。
     * - 取得結果が存在する場合は knsYmd を返却し、未取得時は null を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmdFrom 検針年月日（開始）
     * @param knsYmdTo 検針年月日（終了）
     * @return 取得した検針年月日
     */
    private LocalDate findGmkskkdDate(
            final Long gasMtstNo,
            final LocalDate knsYmdFrom,
            final LocalDate knsYmdTo) {
        Optional<GmkskkdRecord> result =
                gmkskkdDataAccess.findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
                        gasMtstNo,
                        knsYmdFrom,
                        knsYmdTo,
                        disasterClosingReasons);
        return result.map(GmkskkdRecord::knsYmd).orElse(null);
    }

}
