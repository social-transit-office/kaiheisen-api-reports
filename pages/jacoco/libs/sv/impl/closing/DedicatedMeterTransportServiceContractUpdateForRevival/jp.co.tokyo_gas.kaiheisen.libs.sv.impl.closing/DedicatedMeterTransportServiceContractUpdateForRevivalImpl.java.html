/**
 * 概要: SV-111（専用メーターの託送サービス契約、託送サービス契約・ガスメーター設置場所関連更新（契約復活））の実装。
 * 主な仕様:
 * - 指定住所配下の専用メーター候補を検索し、ガスメーター設置場所番号ごとに重複更新を抑止する。
 * - 対象状態の専用メーターだけを関連情報、契約情報の順に契約復活状態へ更新する。
 */
public final class DedicatedMeterTransportServiceContractUpdateForRevivalImpl
        implements DedicatedMeterTransportServiceContractUpdateForRevival {

    static final String SERVICE_USER_ID = "KHSS111";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applyingRelationStatus;

    private final String applicableEndRelationStatus;

    private final String underContractStatus;

    private final String endOfContractStatus;

    private final String meterPossessionStatus;

    /**
     * 必須依存と定数を初期化する。
     *
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DA
     * @param tskgmstDataAccess 託送サービス契約・ガスメーター設置場所関連 DA
     * @param tkssvkyDataAccess 託送サービス契約 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public DedicatedMeterTransportServiceContractUpdateForRevivalImpl(
            final GmtsetdDataAccess gmtsetdDataAccess,
            final TskgmstDataAccess tskgmstDataAccess,
            final TkssvkyDataAccess tkssvkyDataAccess,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applyingRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.applicableEndRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLICABLE_ENDS);
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.endOfContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.END_OF_CONTRACT);
        this.meterPossessionStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
    }

    @Override
    public void execute(
            final Long mtgMtstNo,
            final LocalDate trkKnsYmd,
            final GmtsetdInf gmtsetdInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        List<GmtsetdRecord> dedicatedMeters = gmtsetdDataAccess
                .findAllBySzcGkkCdAndSzcMchCdAndSzcCmazCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                        gmtsetdInf.szcGkkCd(),
                        gmtsetdInf.szcMchCd(),
                        gmtsetdInf.szcCmazCd(),
                        mtgMtstNo,
                        gmtsetdInf.gasMtstNo(),
                        trkKnsYmd);
        if (dedicatedMeters.isEmpty()) {
            throw createGmtsetdMissingException(mtgMtstNo, trkKnsYmd, gmtsetdInf);
        }

        Long lastGasMtstNo = null;
        for (GmtsetdRecord dedicatedMeter : dedicatedMeters) {
            if (Objects.equals(lastGasMtstNo, dedicatedMeter.gasMtstNo())) {
                continue;
            }
            lastGasMtstNo = dedicatedMeter.gasMtstNo();

            if (!meterPossessionStatus.equals(dedicatedMeter.kngmStbJtkb())) {
                continue;
            }

            TskgmstRecord relationRecord = tskgmstDataAccess
                    .findByGasMtstNoAndTyenYmdAndTksuSvkkJtkb(
                            dedicatedMeter.gasMtstNo(),
                            trkKnsYmd,
                            applicableEndRelationStatus)
                    .orElseThrow(() -> createRelationMissingException(dedicatedMeter.gasMtstNo(), trkKnsYmd));

            int updatedRelationCount = tskgmstDataAccess
                    .updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                            buildTskgmstWriteRecord(relationRecord),
                            Set.of(TskgmstColumn.TKSU_SVKK_JTKB, TskgmstColumn.TYEN_YMD),
                            new TskgmstUpdateKey(
                                    relationRecord.gasMtstNo(),
                                    relationRecord.tksuSvkyNo(),
                                    relationRecord.skno(),
                                    relationRecord.recUpdJnjNo()),
                            auditContext);
            if (updatedRelationCount == 0) {
                throw createRelationUpdateException(relationRecord, decodedOriginFunctionId);
            }

            TkssvkyRecord contractRecord = tkssvkyDataAccess
                    .findByTksuSvkyNoAndKnskKjnYmdAndTksuSvkyJtkb(
                            relationRecord.tksuSvkyNo(),
                            trkKnsYmd,
                            endOfContractStatus)
                    .orElseThrow(() -> createContractMissingException(relationRecord.tksuSvkyNo(), trkKnsYmd));

            int updatedContractCount = tkssvkyDataAccess
                    .updateByTksuSvkyNoAndSknoAndRecUpdJnjNo(
                            buildTkssvkyWriteRecord(contractRecord),
                            Set.of(
                                    TkssvkyColumn.TKSU_SVKY_JTKB,
                                    TkssvkyColumn.TKSU_SVKE_JYSB,
                                    TkssvkyColumn.TYEN_YMD),
                            new TkssvkySknoJnjUpdateKey(
                                    contractRecord.tksuSvkyNo(),
                                    contractRecord.skno(),
                                    contractRecord.recUpdJnjNo()),
                            auditContext);
            if (updatedContractCount == 0) {
                throw createContractUpdateException(contractRecord, decodedOriginFunctionId);
            }
        }
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt().orElseThrow(
                () -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private RuntimeException createGmtsetdMissingException(
            final Long mtgMtstNo,
            final LocalDate trkKnsYmd,
            final GmtsetdInf gmtsetdInf) {
        return DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                gmtsetdFindDaIdentifier(),
                new FlatJsonBuilder()
                        .put("szcGkkCd", gmtsetdInf.szcGkkCd())
                        .put("szcMchCd", gmtsetdInf.szcMchCd())
                        .put("szcCmazCd", gmtsetdInf.szcCmazCd())
                        .put("mtgMtstNo", mtgMtstNo)
                        .put("gasMtstNo", gmtsetdInf.gasMtstNo())
                        .put("knskKjnYmd", trkKnsYmd));
    }

    private RuntimeException createRelationMissingException(
            final Long gasMtstNo,
            final LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                relationFindDaIdentifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("tyenYmd", trkKnsYmd)
                        .put("tksuSvkkJtkb", applicableEndRelationStatus));
    }

    private RuntimeException createRelationUpdateException(
            final TskgmstRecord relationRecord,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                relationUpdateDaIdentifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", relationRecord.gasMtstNo())
                        .put("tksuSvkyNo", relationRecord.tksuSvkyNo())
                        .put("skno", relationRecord.skno())
                        .put("recUpdJnjNo", relationRecord.recUpdJnjNo())
                        .put("tksuSvkkJtkb", applyingRelationStatus)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createContractMissingException(
            final Long tksuSvkyNo,
            final LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                contractFindDaIdentifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", tksuSvkyNo)
                        .put("knskKjnYmd", trkKnsYmd)
                        .put("tksuSvkyJtkb", endOfContractStatus));
    }

    private RuntimeException createContractUpdateException(
            final TkssvkyRecord contractRecord,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                contractUpdateDaIdentifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", contractRecord.tksuSvkyNo())
                        .put("skno", contractRecord.skno())
                        .put("recUpdJnjNo", contractRecord.recUpdJnjNo())
                        .put("tksuSvkyJtkb", underContractStatus)
                        .put("tksuSvkeJysb", null)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private String gmtsetdFindDaIdentifier() {
        return FIND_ALL_BY_LOCATION_CODES_ALT_ORDER_AND_MTG_MTST_NO_AND_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD
                .identifier();
    }

    private String relationFindDaIdentifier() {
        return FIND_BY_GAS_MTST_NO_AND_TYEN_YMD_AND_TKSU_SVKK_JTKB.identifier();
    }

    private String relationUpdateDaIdentifier() {
        return UPDATE_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier();
    }

    private String contractFindDaIdentifier() {
        return FIND_BY_TKSU_SVKY_NO_AND_KNSK_KJN_YMD_AND_TKSU_SVKY_JTKB.identifier();
    }

    private String contractUpdateDaIdentifier() {
        return UPDATE_BY_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier();
    }

    private TskgmstWriteRecord buildTskgmstWriteRecord(final TskgmstRecord relationRecord) {
        return new TskgmstWriteRecord(
                relationRecord.gasMtstNo(),
                relationRecord.tksuSvkyNo(),
                relationRecord.skno(),
                applyingRelationStatus,
                relationRecord.tkysYmd(),
                MAX_DATE);
    }

    private TkssvkyWriteRecord buildTkssvkyWriteRecord(final TkssvkyRecord contractRecord) {
        return new TkssvkyWriteRecord(
                contractRecord.tksuSvkyNo(),
                contractRecord.skno(),
                underContractStatus,
                contractRecord.tksuSvksJysb(),
                null,
                contractRecord.tkysYmd(),
                MAX_DATE,
                contractRecord.hesnYtiYmd(),
                contractRecord.tkykShbt(),
                contractRecord.tksuKijsCd(),
                contractRecord.sutiKskhFlg(),
                contractRecord.syoKyoTrno());
    }
