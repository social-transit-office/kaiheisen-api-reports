        /**
         * 閉栓前確認に必要な各種情報を取得し、戻り値 DTO に集約する。
         *
         * <p>ガスメーター設置場所情報、検針関連情報、ガスメーター情報を取得したうえで、
         * TG小売の場合はサービス契約情報と総合他メーター情報を取得し、
         * 他小売の場合は託送メータ情報を取得する。取得件数に応じた有無フラグや、
         * 複数下位 SV の結果をまとめる集約フラグも併せて設定する。
         *
         * @param tgTkurKbn TG他小売区分
         * @param gasMtstNo ガスメーター設置場所番号
         * @param hssgYmd 閉栓作業年月日
         * @return 閉栓前確認に必要な各種取得結果
         */
    @Override
    public PreClosingInfoAcquisitionOutput execute(String tgTkurKbn, Long gasMtstNo, LocalDate hssgYmd) {
        LocalDateTime systemDatetime = requireSystemDatetime();
        LocalDate systemDate = systemDatetime.toLocalDate();
        YearMonth targetYearMonth = YearMonth.of(hssgYmd.getYear(), hssgYmd.getMonthValue());

        GmtsetdRecord systemDateGmtsetd = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemDate)
                .orElseThrow(() -> systemDateGmtsetdNotFound(gasMtstNo, systemDate));
        GmtsetdRecord hsenDateGmtsetd = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, hssgYmd)
                .orElseThrow(() -> hsenDateGmtsetdNotFound(gasMtstNo, hssgYmd));
        Optional<GmtsetkRecord> gmtsetk = gmtsetkDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, hssgYmd);
        Optional<KnsytiRecord> knsyti = knsytiDataAccess
                .findAllByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, targetYearMonth)
                .stream()
                .findFirst();
        KbagmstRecord kbagmst = kbagmstDataAccess
                .findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, targetYearMonth)
                .orElseThrow(() -> kbagmstNotFound(gasMtstNo, targetYearMonth));
        MeterReadingResultAndRelatedInfoAcquisitionOutput meterReadingResult =
                meterReadingResultAndRelatedInfoAcquisition.execute(
                        systemDateGmtsetd.kngmStbJtkb(),
                        gasMtstNo);
        ClosingTargetMeterReadingAndRelatedInfoAcquisitionOutput closingTargetInfo =
                closingTargetMeterReadingAndRelatedInfoAcquisition.execute(
                        hsenDateGmtsetd.ckrtGmscJtkb(),
                        hsenDateGmtsetd.szcGkkCd(),
                        hsenDateGmtsetd.szcCmazCd(),
                        hsenDateGmtsetd.szcMchCd(),
                        hsenDateGmtsetd.mtgMtstNo(),
                        gasMtstNo,
                        systemDateGmtsetd.gmtKhsJtkb(),
                        hsenDateGmtsetd.hesnJysb(),
                        systemDateGmtsetd.kngmStbJtkb(),
                        hssgYmd);

        List<GasmtrdRecord> gasmtrdRecords =
                gasmtrdDataAccess.findAllByTtkGmsbNoAndGmtThkgYmd(gasMtstNo, null);
        boolean gasmtrdUmFlg = !gasmtrdRecords.isEmpty();
        GasmtrdRecord gasmtrdRecord = gasmtrdRecords.isEmpty() ? null : gasmtrdRecords.getFirst();

        List<String> switchReservationStatusList = List.of(
                temporaryRegistrationStatus,
                provisionalCancellationStatus);
        List<TaksmtrRecord> switchReservations = taksmtrDataAccess
                .findAllByGmtSetNoAndTaksKyJtCdListAndTaksCisHntCdOrderByTaksStartYmdDesc(
                        gasMtstNo,
                        switchReservationStatusList,
                        switchApplication);
        boolean swYykUmuFlg = !switchReservations.isEmpty();
        TaksmtrRecord switchReservation = switchReservations.isEmpty() ? null : switchReservations.getFirst();

        boolean tgRetailTarget = Objects.equals("", tgTkurKbn) || Objects.equals(tgRetail, tgTkurKbn);
        ServiceContractNumberByMeterAcquisitionOutput serviceContractNumber = null;
        SvckyaRecord svckya = null;
        SvckyaRecord waitingContract = null;
        List<SvckyaRecord> waitingContractList = List.of();
        ComprehensiveOtherMeterReadingAndRelatedInfoAcquisitionOutput comprehensiveOtherMeter = null;
        List<TaksmtrRecord> otherRetailTaksmtrList = List.of();
        TaksmtrRecord transportMeterInfo = switchReservation;

        if (tgRetailTarget) {
            serviceContractNumber = serviceContractNumberByMeterAcquisition.execute(
                    gasMtstNo,
                    hsenDateGmtsetd.ckrtGmscJtkb(),
                    hsenDateGmtsetd.mtgMtstNo(),
                    hssgYmd);
            List<String> svkyStatusList = List.of(
                    underContractStatus,
                    endOfContractStatus);
            ServiceContractNumberByMeterAcquisitionOutput finalServiceContractNumber = serviceContractNumber;
            svckya = svckyaDataAccess
                    .findLatestBySvkyNoAndSvkyJtkbList(serviceContractNumber.svkyNo(), svkyStatusList)
                    .orElseThrow(() -> svckyaNotFound(finalServiceContractNumber.svkyNo(), svkyStatusList));
            waitingContractList = svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(
                    serviceContractNumber.svkyNo(),
                    waitingForContractApplicationStatus);
            waitingContract = waitingContractList.isEmpty() ? null : waitingContractList.getFirst();
            comprehensiveOtherMeter = comprehensiveOtherMeterReadingAndRelatedInfoAcquisition.execute(
                    Boolean.TRUE,
                    hsenDateGmtsetd.ckrtGmscJtkb(),
                    serviceContractNumber.svkyNo(),
                    gasMtstNo,
                    hssgYmd,
                    hssgYmd);
        } else {
            List<String> otherRetailContractStatusList = List.of(
                    bookRegistrationStatus,
                    provisionalCancellationStatus);
            otherRetailTaksmtrList = taksmtrDataAccess.findAllByGmtSetNoAndTaksKyJtCdList(
                    gasMtstNo,
                    otherRetailContractStatusList);
            transportMeterInfo = selectOtherRetailTaksmtrRecord(otherRetailTaksmtrList);
        }

        return new PreClosingInfoAcquisitionOutput(
                toGmtsetdSystemDateInf(systemDateGmtsetd),
                toGmtsetdHsenSaymdInf(hsenDateGmtsetd),
                gmtsetk.isPresent() ? toGmtsetkInf(gmtsetk.get()) : null,
                knsyti.isPresent() ? toKnsytiInf(knsyti.get()) : null,
                toKbagmstInf(kbagmst),
                svckya != null ? toSvckyaInf(svckya) : null,
                toSvckyaKiyKtymInf(waitingContract),
                serviceContractNumber != null ? toSvmstkrInf(serviceContractNumber) : null,
                toClosingSkinKnkkInf(
                        meterReadingResult.skinKnkkInf(),
                        meterReadingResult.hsenHkkKnuKigenYmd()),
                meterReadingResult.hsenHkkKnuKigenYmd(),
                toClosingZnkiKnkkInf(meterReadingResult.znkiKnkkInf()),
                gasmtrdRecord != null ? toGasmtrdInf(gasmtrdRecord) : null,
                closingTargetInfo.knsKkaSnyInf().skinKnkkInfList().size(),
                toClosingGmtsetdSnyInf(closingTargetInfo.gmtsetdSnyInf()),
                toClosingKnsKkaMtInf(
                        closingTargetInfo.knsKkaMtInf(),
                        meterReadingResult.hsenHkkKnuKigenYmd()),
                toClosingKnsKkaSnyInf(closingTargetInfo.knsKkaSnyInf()),
                toClosingComprehensiveOtherMeterInfoList(comprehensiveOtherMeter),
                transportMeterInfo != null ? toTaksmtrInf(transportMeterInfo) : null,
                gasmtrdUmFlg,
                serviceContractNumber != null,
                !otherRetailTaksmtrList.isEmpty(),
                !waitingContractList.isEmpty(),
                closingTargetInfo.knsKkaSnyInf().snyHsngKnsFlg(),
                closingTargetInfo.dgrpHesnMturFlg(),
                comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpKasFlg() : null,
                comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpMiRegFlg() : Boolean.FALSE,
                closingTargetInfo.snyKasFlg(),
                meterReadingResult.gmsrKtumFlg(),
                closingTargetInfo.knsKkaMtInf().mtgSyryKytiTyuFlg(),
                closingTargetInfo.knsKkaSnyInf().snySyryKytiTyuFlg(),
                comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpSyryKytiTyuFlg() : null,
                closingTargetInfo.knsKkaMtInf().mtgSyryKytiTyuMtstno(),
                closingTargetInfo.knsKkaSnyInf().snySyryKytiTyuMtstno(),
                comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpSyryKytiTyuMtstno() : null,
                aggregateTrue(
                        closingTargetInfo.knsKkaSnyInf().snyKyktFlg(),
                        comprehensiveOtherMeter != null ? comprehensiveOtherMeter.snyKyktFlg() : null),
                swYykUmuFlg,
                aggregateTrue(
                        closingTargetInfo.dgrpHkhesnJiyuFlg(),
                        comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpHkhesnJiyuFlg() : null),
                comprehensiveOtherMeter != null ? comprehensiveOtherMeter.dgrpHsenYmd() : null);
    }

         * それ以外はどちらか一方でも {@code true} であれば {@code true} を返す。
         *
         * @param left 集約対象の左値
         * @param right 集約対象の右値
         * @return OR 集約結果

    private Boolean aggregateTrue(Boolean left, Boolean right) {
        if (left == null && right == null) {
            return null;
        }
        return Boolean.TRUE.equals(left) || Boolean.TRUE.equals(right);
    }
