/**
 * SV-031（サービス契約、サービス契約・メーター設置場所関連更新（契約終了））のサービス実装。
 */
public class ServiceContractMeterLocationUpdateForTerminationImpl
        implements ServiceContractMeterLocationUpdateForTermination {

    static final String SERVICE_USER_ID = "KHSS031";

    private final SvckyaDataAccess svckyaDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String waitingForContractApplication;

    private final String invalidServiceContract;

    private final String endOfContract;

    private final String invalidChargeCalculationRelatedStatus;

    private final String relatedValidWaiting;

    private final String relatedEnd;

    private final String supplyStopPaymentPending;

    private final String supplyOutage;

    /**
     * コンストラクタ。
     *
     * @param svckyaDataAccess サービス契約 DA
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ServiceContractMeterLocationUpdateForTerminationImpl(
            SvckyaDataAccess svckyaDataAccess,
            SvmstkrDataAccess svmstkrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccess, "svckyaDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.waitingForContractApplication = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.WAITING_FOR_CONTRACT_APPLICATION);
        this.invalidServiceContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.INVALID);
        this.endOfContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.END_OF_CONTRACT);
        this.invalidChargeCalculationRelatedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.INVALID);
        this.relatedValidWaiting = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_01);
        this.relatedEnd = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.supplyStopPaymentPending = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
        this.supplyOutage = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_OUTAGE);
    }

    @Override
    public ServiceContractMeterLocationUpdateForTerminationOutput execute(
            Long gasMtstNo,
            LocalDate tyenYmd,
            String svkyEndJysb,
            SvmstkrInf svmstkrInf,
            SvckyaInf svckyaInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        invalidateWaitingContractsIfPresent(svckyaInf.svkyNo(), auditContext, decodedOriginFunctionId);

        SvckyaRecord currentServiceContract = svckyaDataAccess
                .findBySvkyNoAndSkno(svckyaInf.svkyNo(), svckyaInf.skno())
                .orElseThrow(() -> createCurrentContractMissingException(svckyaInf));

        if (!currentServiceContract.rkkiSttYmd().isAfter(tyenYmd)) {
            updateCurrentContractToEnd(
                    currentServiceContract,
                    tyenYmd,
                    svkyEndJysb,
                    svckyaInf.recUpdJnjNo(),
                    auditContext,
                    decodedOriginFunctionId);
            updateSameContractGeneration(currentServiceContract, tyenYmd, auditContext);
            updateSupplyStopStatusIfNeeded(currentServiceContract, auditContext);
        } else {
            invalidateCurrentContract(
                    currentServiceContract,
                    svckyaInf.recUpdJnjNo(),
                    auditContext,
                    decodedOriginFunctionId);
            updatePreviousContractIfPresent(
                    currentServiceContract,
                    tyenYmd,
                    svkyEndJysb,
                    auditContext,
                    decodedOriginFunctionId);
        }

        updateMeterLocationRelation(
                gasMtstNo,
                tyenYmd,
                svmstkrInf,
                auditContext,
                decodedOriginFunctionId);

        return new ServiceContractMeterLocationUpdateForTerminationOutput(
                currentServiceContract.rkkiSttYmd());
    }

    private void invalidateWaitingContractsIfPresent(
            Long svkyNo,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        List<SvckyaRecord> waitingContracts =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(svkyNo, waitingForContractApplication);
        if (waitingContracts.isEmpty()) {
            return;
        }

        for (SvckyaRecord waitingContract : waitingContracts) {
            int updatedCount = svckyaDataAccess.updateBySvkyNoAndSknoAndRecUpdJnjNo(
                    buildSvckyaWriteRecord(
                            waitingContract,
                            invalidServiceContract,
                            waitingContract.svkyEndJysb(),
                            waitingContract.rkkiEndYmd(),
                            waitingContract.ksbEndYmd(),
                            waitingContract.ksbEnytYmd(),
                            waitingContract.kkysSisnJtkb()),
                    EnumSet.of(SvckyaColumn.SVKY_JTKB),
                    new SvckyaSknoJnjUpdateKey(
                            waitingContract.svkyNo(),
                            waitingContract.skno(),
                            waitingContract.recUpdJnjNo()),
                    auditContext);
            if (updatedCount == 0) {
                throw createSvckyaUpdateException(
                        waitingContract.svkyNo(),
                        waitingContract.skno(),
                        waitingContract.recUpdJnjNo(),
                        invalidServiceContract,
                        null,
                        null,
                        null,
                        null,
                        SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                        decodedOriginFunctionId);
            }
        }

        svmstkrDataAccess.updateBySvkyNoAndRkkiKnrnJtkb(
                new SvmstkrWriteRecord(
                        null,
                        svkyNo,
                        null,
                        null,
                        null,
                        invalidChargeCalculationRelatedStatus,
                        null,
                        null,
                        null),
                EnumSet.of(SvmstkrColumn.RKKI_KNRN_JTKB),
                new SvmstkrRkkiKnrnJtkbUpdateKey(svkyNo, relatedValidWaiting, null),
                auditContext);
    }

    private void updateCurrentContractToEnd(
            SvckyaRecord currentServiceContract,
            LocalDate tyenYmd,
            String svkyEndJysb,
            Integer recUpdJnjNo,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svckyaDataAccess.updateBySvkyNoAndSknoAndRecUpdJnjNo(
                buildSvckyaWriteRecord(
                        currentServiceContract,
                        endOfContract,
                        svkyEndJysb,
                        tyenYmd,
                        tyenYmd,
                        tyenYmd,
                        currentServiceContract.kkysSisnJtkb()),
                EnumSet.of(
                        SvckyaColumn.SVKY_JTKB,
                        SvckyaColumn.SVKY_END_JYSB,
                        SvckyaColumn.RKKI_END_YMD,
                        SvckyaColumn.KSB_END_YMD,
                        SvckyaColumn.KSB_ENYT_YMD),
                new SvckyaSknoJnjUpdateKey(
                        currentServiceContract.svkyNo(),
                        currentServiceContract.skno(),
                        recUpdJnjNo),
                auditContext);
        if (updatedCount == 0) {
            throw createSvckyaUpdateException(
                    currentServiceContract.svkyNo(),
                    currentServiceContract.skno(),
                    recUpdJnjNo,
                    endOfContract,
                    svkyEndJysb,
                    tyenYmd,
                    tyenYmd,
                    tyenYmd,
                    SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                    decodedOriginFunctionId);
        }
    }

    private void invalidateCurrentContract(
            SvckyaRecord currentServiceContract,
            Integer recUpdJnjNo,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svckyaDataAccess.updateBySvkyNoAndSknoAndRecUpdJnjNo(
                buildSvckyaWriteRecord(
                        currentServiceContract,
                        invalidServiceContract,
                        currentServiceContract.svkyEndJysb(),
                        currentServiceContract.rkkiEndYmd(),
                        currentServiceContract.ksbEndYmd(),
                        currentServiceContract.ksbEnytYmd(),
                        currentServiceContract.kkysSisnJtkb()),
                EnumSet.of(SvckyaColumn.SVKY_JTKB),
                new SvckyaSknoJnjUpdateKey(
                        currentServiceContract.svkyNo(),
                        currentServiceContract.skno(),
                        recUpdJnjNo),
                auditContext);
        if (updatedCount == 0) {
            throw createSvckyaUpdateException(
                    currentServiceContract.svkyNo(),
                    currentServiceContract.skno(),
                    recUpdJnjNo,
                    invalidServiceContract,
                    null,
                    null,
                    null,
                    null,
                    SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                    decodedOriginFunctionId);
        }
    }

    private void updatePreviousContractIfPresent(
            SvckyaRecord currentServiceContract,
            LocalDate tyenYmd,
            String svkyEndJysb,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        svckyaDataAccess.findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNotOrderByRkkiSttYmd(
                        currentServiceContract.svkyNo(),
                        currentServiceContract.rkkiSttYmd().minusDays(1),
                        invalidServiceContract)
                .ifPresent(previousServiceContract -> {
                    int updatedCount = svckyaDataAccess.updateBySvkyNoAndSknoAndRecUpdJnjNo(
                            buildSvckyaWriteRecord(
                                    previousServiceContract,
                                    endOfContract,
                                    svkyEndJysb,
                                    previousServiceContract.rkkiEndYmd(),
                                    tyenYmd,
                                    tyenYmd,
                                    previousServiceContract.kkysSisnJtkb()),
                            EnumSet.of(
                                    SvckyaColumn.SVKY_JTKB,
                                    SvckyaColumn.SVKY_END_JYSB,
                                    SvckyaColumn.RKKI_END_YMD,
                                    SvckyaColumn.KSB_END_YMD,
                                    SvckyaColumn.KSB_ENYT_YMD),
                            new SvckyaSknoJnjUpdateKey(
                                    previousServiceContract.svkyNo(),
                                    previousServiceContract.skno(),
                                    previousServiceContract.recUpdJnjNo()),
                            auditContext);
                    if (updatedCount == 0) {
                        throw createSvckyaUpdateException(
                                previousServiceContract.svkyNo(),
                                previousServiceContract.skno(),
                                previousServiceContract.recUpdJnjNo(),
                                endOfContract,
                                svkyEndJysb,
                                previousServiceContract.rkkiEndYmd(),
                                tyenYmd,
                                tyenYmd,
                                SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                                decodedOriginFunctionId);
                    }
                    updateSameContractGeneration(previousServiceContract, tyenYmd, auditContext);
                });
    }

    private void updateSameContractGeneration(
            SvckyaRecord baseServiceContract,
            LocalDate tyenYmd,
            AuditContext auditContext) {
        svckyaDataAccess.updateBySvkyNoAndKyttSknoAndSknoNot(
                buildSvckyaWriteRecord(
                        baseServiceContract,
                        baseServiceContract.svkyJtkb(),
                        baseServiceContract.svkyEndJysb(),
                        baseServiceContract.rkkiEndYmd(),
                        tyenYmd,
                        tyenYmd,
                        baseServiceContract.kkysSisnJtkb()),
                EnumSet.of(SvckyaColumn.KSB_END_YMD, SvckyaColumn.KSB_ENYT_YMD),
                new SvckyaKyttSknoUpdateKey(
                        baseServiceContract.svkyNo(),
                        baseServiceContract.kyttSkno(),
                        baseServiceContract.skno()),
                auditContext);
    }

    private void updateSupplyStopStatusIfNeeded(
            SvckyaRecord currentServiceContract,
            AuditContext auditContext) {
        if (!supplyStopPaymentPending.equals(currentServiceContract.kkysSisnJtkb())) {
            return;
        }
        svckyaDataAccess.updateBySvkyNoAndSkno(
                buildSvckyaWriteRecord(
                        currentServiceContract,
                        currentServiceContract.svkyJtkb(),
                        currentServiceContract.svkyEndJysb(),
                        currentServiceContract.rkkiEndYmd(),
                        currentServiceContract.ksbEndYmd(),
                        currentServiceContract.ksbEnytYmd(),
                        supplyOutage),
                EnumSet.of(SvckyaColumn.KKYS_SISN_JTKB),
                new SvckyaSknoUpdateKey(
                        currentServiceContract.svkyNo(),
                        currentServiceContract.skno()),
                auditContext);
    }

    private void updateMeterLocationRelation(
            Long gasMtstNo,
            LocalDate tyenYmd,
            SvmstkrInf svmstkrInf,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                new SvmstkrWriteRecord(
                        gasMtstNo,
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        null,
                        null,
                        relatedEnd,
                        null,
                        tyenYmd,
                        null),
                EnumSet.of(SvmstkrColumn.RKKI_END_YMD, SvmstkrColumn.RKKI_KNRN_JTKB),
                new SvmstkrSknoOptionalUpdateKey(
                        gasMtstNo,
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        svmstkrInf.recUpdJnjNo()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    svmstkrDataAccess.tableName(),
                    SvmstkrDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_SVKY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()
                            .put("mtstNo", gasMtstNo)
                            .put("svkyNo", svmstkrInf.svkyNo())
                            .put("skno", svmstkrInf.skno())
                            .put("recUpdJnjNo", svmstkrInf.recUpdJnjNo())
                            .put("rkkiEndYmd", tyenYmd)
                            .put("rkkiKnrnJtkb", relatedEnd)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                                requestContext.getHukkSysShbt()
                                                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }

    private RuntimeException createCurrentContractMissingException(SvckyaInf svckyaInf) {
        return DataInconsistencySupport.create(
                svckyaDataAccess.tableName(),
                SvckyaDataAccess.CallSpecs.FIND_BY_SVKY_NO_AND_SKNO.identifier(),
                new FlatJsonBuilder()
                        .put("svkyNo", svckyaInf.svkyNo())
                        .put("skno", svckyaInf.skno()));
    }

    private RuntimeException createSvckyaUpdateException(
            Long svkyNo,
            Integer skno,
            Integer recUpdJnjNo,
            String svkyJtkb,
            String svkyEndJysb,
            LocalDate rkkiEndYmd,
            LocalDate ksbEndYmd,
            LocalDate ksbEnytYmd,
            String daIdentifier,
            String decodedOriginFunctionId) {
        FlatJsonBuilder builder = new FlatJsonBuilder()
                .put("svkyNo", svkyNo)
                .put("skno", skno)
                .put("recUpdJnjNo", recUpdJnjNo)
                .put("svkyJtkb", svkyJtkb);
        if (svkyEndJysb != null) {
            builder.put("svkyEndJysb", svkyEndJysb);
        }
        if (rkkiEndYmd != null) {
            builder.put("rkkiEndYmd", rkkiEndYmd);
        }
        if (ksbEndYmd != null) {
            builder.put("ksbEndYmd", ksbEndYmd);
        }
        if (ksbEnytYmd != null) {
            builder.put("ksbEnytYmd", ksbEnytYmd);
        }
        builder.put("updUsrId", SERVICE_USER_ID)
                .put("updKtnKinoId", decodedOriginFunctionId);
        return DataInconsistencySupport.create(
                svckyaDataAccess.tableName(),
                daIdentifier,
                builder);
    }

    private SvckyaWriteRecord buildSvckyaWriteRecord(
            SvckyaRecord baseRecord,
            String svkyJtkb,
            String svkyEndJysb,
            LocalDate rkkiEndYmd,
            LocalDate ksbEndYmd,
            LocalDate ksbEnytYmd,
            String kkysSisnJtkb) {
        return new SvckyaWriteRecord(
                baseRecord.svkyNo(),
                baseRecord.skno(),
                baseRecord.kyttSkno(),
                baseRecord.kyttTsno(),
                svkyJtkb,
                baseRecord.svkySttJysb(),
                baseRecord.ksbTkyJysb(),
                svkyEndJysb,
                baseRecord.rkkiSttYmd(),
                rkkiEndYmd,
                baseRecord.ksbSttYmd(),
                ksbEndYmd,
                baseRecord.ksbStytYmd(),
                ksbEnytYmd,
                baseRecord.kykYtiYmd(),
                baseRecord.svcRyksYmd(),
                baseRecord.svkyTiktYmd(),
                baseRecord.ksbTiktYmd(),
                baseRecord.szirHtyoYmd(),
                baseRecord.sqtiKiyNo(),
                baseRecord.syoKyoTrno(),
                baseRecord.kysKyoTrno(),
                baseRecord.byssUmFlg(),
                baseRecord.shziKbn(),
                baseRecord.kiyShbt(),
                baseRecord.gasYtCd(),
                baseRecord.sziKziKbn(),
                baseRecord.rknKhumFlg(),
                kkysSisnJtkb,
                baseRecord.kiyKnrkCd(),
                baseRecord.syaKbn(),
                baseRecord.tsziKysbNo(),
                baseRecord.recUpdJnjNo());
    }
