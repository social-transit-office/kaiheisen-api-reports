/**
 * 概要: SV-107（託送サービス契約復活（他小売））の実装。
 * 主な仕様:
 * - 専用ガスメーター以外では、託送サービス契約と関連情報を順に契約復活状態へ更新する。
 * - 元ガスメーターの場合は、更新成功後に補助資料 `SV-111.tsv` に従って SV-111 を追加実行する。
 */
public final class TransportServiceContractRevivalOtherRetailImpl
        implements TransportServiceContractRevivalOtherRetail {

    static final String SERVICE_USER_ID = "KHSS107";

    static final String EMPTY_CONTRACT_END_REASON = "";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final DedicatedMeterTransportServiceContractUpdateForRevival
            dedicatedMeterTransportServiceContractUpdateForRevival;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applyingRelationStatus;

    private final String underContractStatus;

    private final String originalGasMeterStatus;

    private final String dedicatedGasMeterStatus;

    /**
     * 必須依存と定数を初期化する。
     *
     * @param tkssvkyDataAccess 託送サービス契約 DA
     * @param tskgmstDataAccess 託送サービス契約・ガスメーター設置場所関連 DA
     * @param dedicatedMeterTransportServiceContractUpdateForRevival SV-111
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public TransportServiceContractRevivalOtherRetailImpl(
            final TkssvkyDataAccess tkssvkyDataAccess,
            final TskgmstDataAccess tskgmstDataAccess,
            final DedicatedMeterTransportServiceContractUpdateForRevival dedicatedMeterTransportServiceContractUpdateForRevival,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.dedicatedMeterTransportServiceContractUpdateForRevival = Objects.requireNonNull(
                dedicatedMeterTransportServiceContractUpdateForRevival,
                "dedicatedMeterTransportServiceContractUpdateForRevival");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applyingRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
    }

    @Override
    public void execute(
            final GmtsetdInf gmtsetdInf,
            final TkssvkyInf tkssvkyInf,
            final TskgmstInf tskgmstInf,
            final LocalDate trkKnsYmd) {
        if (dedicatedGasMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            return;
        }

        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        int updatedContractCount = tkssvkyDataAccess.updateByTksuSvkyNoAndSknoAndRecUpdJnjNo(
                buildTkssvkyWriteRecord(tkssvkyInf),
                Set.of(
                        TkssvkyColumn.TKSU_SVKY_JTKB,
                        TkssvkyColumn.TKSU_SVKE_JYSB,
                        TkssvkyColumn.TYEN_YMD),
                new TkssvkySknoJnjUpdateKey(tkssvkyInf.tksuSvkyNo(), tkssvkyInf.skno()),
                auditContext);
        if (updatedContractCount == 0) {
            throw createContractUpdateException(tkssvkyInf, decodedOriginFunctionId);
        }

        int updatedRelationCount = tskgmstDataAccess.updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                buildTskgmstWriteRecord(tskgmstInf),
                Set.of(TskgmstColumn.TKSU_SVKK_JTKB, TskgmstColumn.TYEN_YMD),
                new TskgmstUpdateKey(tskgmstInf.gasMtstNo(), tskgmstInf.tksuSvkyNo(), tskgmstInf.skno()),
                auditContext);
        if (updatedRelationCount == 0) {
            throw createRelationUpdateException(tskgmstInf, decodedOriginFunctionId);
        }

        if (originalGasMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            dedicatedMeterTransportServiceContractUpdateForRevival.execute(
                    gmtsetdInf.gasMtstNo(),
                    trkKnsYmd,
                    toSv111GmtsetdInf(gmtsetdInf));
        }
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf toSv111GmtsetdInf(
            final GmtsetdInf gmtsetdInf) {
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf(
                gmtsetdInf.gasMtstNo(),
                gmtsetdInf.szcGkkCd(),
                gmtsetdInf.szcMchCd(),
                gmtsetdInf.szcCmazCd());
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt().orElseThrow(
                () -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private TkssvkyWriteRecord buildTkssvkyWriteRecord(final TkssvkyInf tkssvkyInf) {
        return new TkssvkyWriteRecord(
                tkssvkyInf.tksuSvkyNo(),
                tkssvkyInf.skno(),
                underContractStatus,
                null,
                EMPTY_CONTRACT_END_REASON,
                null,
                MAX_DATE,
                null,
                null,
                null,
                null,
                null);
    }

    private TskgmstWriteRecord buildTskgmstWriteRecord(final TskgmstInf tskgmstInf) {
        return new TskgmstWriteRecord(
                tskgmstInf.gasMtstNo(),
                tskgmstInf.tksuSvkyNo(),
                tskgmstInf.skno(),
                applyingRelationStatus,
                null,
                MAX_DATE);
    }

    private RuntimeException createContractUpdateException(
            final TkssvkyInf tkssvkyInf,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                UPDATE_BY_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", tkssvkyInf.tksuSvkyNo())
                        .put("skno", tkssvkyInf.skno())
                        .put("tksuSvkyJtkb", underContractStatus)
                        .put("tksuSvkeJysb", EMPTY_CONTRACT_END_REASON)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createRelationUpdateException(
            final TskgmstInf tskgmstInf,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                UPDATE_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", tskgmstInf.gasMtstNo())
                        .put("tksuSvkyNo", tskgmstInf.tksuSvkyNo())
                        .put("skno", tskgmstInf.skno())
                        .put("tksuSvkkJtkb", applyingRelationStatus)
                        .put("tyenYmd", MAX_DATE)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }
}
