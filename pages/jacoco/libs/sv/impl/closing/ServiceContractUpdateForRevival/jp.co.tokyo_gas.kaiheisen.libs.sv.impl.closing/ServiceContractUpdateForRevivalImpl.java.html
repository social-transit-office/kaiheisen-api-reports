/**
 * 概要: SV-108（サービス契約更新（契約復活））のサービス実装。
 * 主な仕様:
 * - 取消対象検針年月日の翌日時点で無効のサービス契約を参照し、契約復活状態へ更新する。
 * - 免税条件に一致する場合は先に SV-109 を呼び出す。
 * - 非排他更新の更新件数が 0 件の場合は DataInconsistencyException を送出する。
 */
public final class ServiceContractUpdateForRevivalImpl
        implements ServiceContractUpdateForRevival {

    /** 監査項目に設定する更新ユーザー ID。 */
    static final String SERVICE_USER_ID = "KHSS108";

    /** 契約復活時に設定する最大日付。 */
    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final SvckyaDataAccess svckyaDataAccess;

    private final TaxExemptContractUpdateForRevival taxExemptContractUpdateForRevival;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String invalidServiceContractStatus;

    private final String underContractStatus;

    private final String dutyFreeConsumptionTax;

    /**
     * 必須依存と定数を初期化する。
     *
     * @param serviceContractDataAccess サービス契約 DA
     * @param taxExemptUpdateService 免税契約更新 SV
     * @param context リクエストコンテキスト
     * @param reportingConstantService 開閉栓定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public ServiceContractUpdateForRevivalImpl(
            final SvckyaDataAccess serviceContractDataAccess,
            final TaxExemptContractUpdateForRevival taxExemptUpdateService,
            final RequestContext context,
            final KaiheisenConstantService reportingConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.svckyaDataAccess = Objects.requireNonNull(
                serviceContractDataAccess,
                "serviceContractDataAccess");
        this.taxExemptContractUpdateForRevival = Objects.requireNonNull(
                taxExemptUpdateService,
                "taxExemptUpdateService");
        this.requestContext = Objects.requireNonNull(context, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                reportingConstantService,
                "reportingConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.invalidServiceContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.INVALID);
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.dutyFreeConsumptionTax = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsumptionTax.TYPE,
                ConstantLookupKeys.Cirius.ConsumptionTax.DUTY_FREE);
    }

    /**
     * サービス契約の契約復活更新を実行する。
     *
     * @param svkyNo サービス契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @param sziKziKbn 消費税課税区分
     * @param skno 世代管理番号
     * @param kyttSkno 契約特定世代管理番号
     */
    @Override
    public void execute(
            final Long svkyNo,
            final LocalDate trkKnsYmd,
            final String sziKziKbn,
            final Integer skno,
            final Integer kyttSkno) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(
                SERVICE_USER_ID,
                decodedOriginFunctionId);

        Optional<SvckyaRecord> serviceContract = svckyaDataAccess
                .findLatestBySvkyNoAndRkkiSttYmdAndSvkyJtkb(
                        svkyNo,
                        trkKnsYmd.plusDays(1),
                        invalidServiceContractStatus);

        if (shouldInvokeTaxExemptUpdate(serviceContract.orElse(null), sziKziKbn)) {
            taxExemptContractUpdateForRevival.execute(svkyNo, trkKnsYmd);
        }

        if (serviceContract.isEmpty()) {
            updateWhenNoContractExists(
                    svkyNo,
                    skno,
                    kyttSkno,
                    auditContext,
                    decodedOriginFunctionId);
            return;
        }

        updateWhenContractExists(
                serviceContract.get(),
                svkyNo,
                skno,
                kyttSkno,
                sziKziKbn,
                auditContext,
                decodedOriginFunctionId);
    }

    private boolean shouldInvokeTaxExemptUpdate(
            final SvckyaRecord serviceContract,
            final String inputConsumptionTax) {
        return dutyFreeConsumptionTax.equals(inputConsumptionTax)
                || (serviceContract != null
                && dutyFreeConsumptionTax.equals(serviceContract.sziKziKbn()));
    }

    private void updateWhenNoContractExists(
            final Long svkyNo,
            final Integer skno,
            final Integer kyttSkno,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        int updatedPrimaryCount = svckyaDataAccess.updateBySvkyNoAndSkno(
                buildRevivalWriteRecord(
                        underContractStatus,
                        null,
                        MAX_DATE,
                        MAX_DATE,
                        MAX_DATE,
                        null),
                EnumSet.of(
                        SvckyaColumn.SVKY_JTKB,
                        SvckyaColumn.SVKY_END_JYSB,
                        SvckyaColumn.RKKI_END_YMD,
                        SvckyaColumn.KSB_END_YMD,
                        SvckyaColumn.KSB_ENYT_YMD),
                new SvckyaSknoUpdateKey(svkyNo, skno),
                auditContext);
        ensureUpdated(
                updatedPrimaryCount,
                createPrimaryUpdateException(
                        svkyNo,
                        skno,
                        underContractStatus,
                        null,
                        MAX_DATE,
                        MAX_DATE,
                        MAX_DATE,
                        null,
                        decodedOriginFunctionId));

        svckyaDataAccess.updateBySvkyNoAndKyttSknoAndSknoNot(
                buildRelatedUpdateWriteRecord(),
                EnumSet.of(
                        SvckyaColumn.KSB_END_YMD,
                        SvckyaColumn.KSB_ENYT_YMD),
                new SvckyaKyttSknoUpdateKey(svkyNo, kyttSkno, skno),
                auditContext);
    }

    private void updateWhenContractExists(
            final SvckyaRecord serviceContract,
            final Long svkyNo,
            final Integer skno,
            final Integer kyttSkno,
            final String sziKziKbn,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        int updatedRevivalCount = svckyaDataAccess.updateBySvkyNoAndSkno(
                buildRevivalWriteRecord(
                        underContractStatus,
                        null,
                        MAX_DATE,
                        MAX_DATE,
                        MAX_DATE,
                        sziKziKbn),
                EnumSet.of(
                        SvckyaColumn.SVKY_JTKB,
                        SvckyaColumn.SVKY_END_JYSB,
                        SvckyaColumn.RKKI_END_YMD,
                        SvckyaColumn.KSB_END_YMD,
                        SvckyaColumn.KSB_ENYT_YMD,
                        SvckyaColumn.SZI_KZI_KBN),
                new SvckyaSknoUpdateKey(svkyNo, serviceContract.skno()),
                auditContext);
        ensureUpdated(
                updatedRevivalCount,
                createPrimaryUpdateException(
                        svkyNo,
                        serviceContract.skno(),
                        underContractStatus,
                        null,
                        MAX_DATE,
                        MAX_DATE,
                        MAX_DATE,
                        sziKziKbn,
                        decodedOriginFunctionId));

        int updatedEndReasonCount = svckyaDataAccess.updateBySvkyNoAndSkno(
                buildEndReasonClearWriteRecord(),
                EnumSet.of(SvckyaColumn.SVKY_END_JYSB),
                new SvckyaSknoUpdateKey(svkyNo, skno),
                auditContext);
        ensureUpdated(
                updatedEndReasonCount,
                createEndReasonClearException(
                        svkyNo,
                        skno,
                        decodedOriginFunctionId));

        svckyaDataAccess.updateBySvkyNoAndKyttSknoAndSknoNot(
                buildRelatedUpdateWriteRecord(),
                EnumSet.of(
                        SvckyaColumn.KSB_END_YMD,
                        SvckyaColumn.KSB_ENYT_YMD),
                new SvckyaKyttSknoUpdateKey(svkyNo, kyttSkno, skno),
                auditContext);
    }

    private void ensureUpdated(
            final int updatedCount,
            final RuntimeException exception) {
        if (updatedCount == 0) {
            throw exception;
        }
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt().orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.hukkSysShbt is required")));
    }

    private RuntimeException createPrimaryUpdateException(
            final Long svkyNo,
            final Integer skno,
            final String svkyJtkb,
            final String svkyEndJysb,
            final LocalDate rkkiEndYmd,
            final LocalDate ksbEndYmd,
            final LocalDate ksbEnytYmd,
            final String sziKziKbn,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                svckyaDataAccess.tableName(),
                UPDATE_BY_SVKY_NO_AND_SKNO.identifier(),
                new FlatJsonBuilder()
                        .put("svkyNo", svkyNo)
                        .put("skno", skno)
                        .put("svkyJtkb", svkyJtkb)
                        .put("svkyEndJysb", svkyEndJysb)
                        .put("rkkiEndYmd", rkkiEndYmd)
                        .put("ksbEndYmd", ksbEndYmd)
                        .put("ksbEnytYmd", ksbEnytYmd)
                        .putIfNonNull("sziKziKbn", sziKziKbn)
                        .put("regUsrId", SERVICE_USER_ID)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createEndReasonClearException(
            final Long svkyNo,
            final Integer skno,
            final String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                svckyaDataAccess.tableName(),
                UPDATE_BY_SVKY_NO_AND_SKNO.identifier(),
                new FlatJsonBuilder()
                        .put("svkyNo", svkyNo)
                        .put("skno", skno)
                        .put("svkyEndJysb", null)
                        .put("regUsrId", SERVICE_USER_ID)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private static SvckyaWriteRecord buildRevivalWriteRecord(
            final String svkyJtkb,
            final String svkyEndJysb,
            final LocalDate rkkiEndYmd,
            final LocalDate ksbEndYmd,
            final LocalDate ksbEnytYmd,
            final String sziKziKbn) {
        return new SvckyaWriteRecord(
                null,
                null,
                null,
                null,
                svkyJtkb,
                null,
                null,
                svkyEndJysb,
                null,
                rkkiEndYmd,
                null,
                ksbEndYmd,
                null,
                ksbEnytYmd,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                sziKziKbn,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private static SvckyaWriteRecord buildEndReasonClearWriteRecord() {
        return buildRevivalWriteRecord(null, null, null, null, null, null);
    }

    private static SvckyaWriteRecord buildRelatedUpdateWriteRecord() {
        return new SvckyaWriteRecord(
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                MAX_DATE,
                null,
                MAX_DATE,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }
