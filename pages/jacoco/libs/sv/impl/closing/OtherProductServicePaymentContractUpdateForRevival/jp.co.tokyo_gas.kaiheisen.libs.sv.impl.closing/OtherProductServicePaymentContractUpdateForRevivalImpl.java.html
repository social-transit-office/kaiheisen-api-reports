package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.closing;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svsihkr.SvsihkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svsihkr.SvsihkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.OtherProductServicePaymentContractUpdateForRevival;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.ServicePaymentContractUpdateForRevival;

/**
 * 概要: SV-116（他商材のサービス契約・支払契約関連更新（契約復活））のサービス実装。
 * 主な仕様:
 * - `DA-044-R009` で他商材のサービス契約・支払契約関連を取得する。
 * - 取得した各レコードについて `SV-114` を順に呼び出す。
 * - 下位 DA / 下位 SV の例外は変換せず、そのまま上位へ伝播する。
 */
public final class OtherProductServicePaymentContractUpdateForRevivalImpl
        implements OtherProductServicePaymentContractUpdateForRevival {

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final ServicePaymentContractUpdateForRevival servicePaymentContractUpdateForRevival;

    private final String applicableEndsPaymentContractStatus;

    private final List<String> targetSingleBillingPartyIdentificationTypes;

    /**
     * 概要: 必須依存と検索用定数を初期化する。
     *
     * @param svsihkrDataAccess サービス契約・支払契約関連 DA
     * @param servicePaymentContractUpdateForRevival `SV-114` の API
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public OtherProductServicePaymentContractUpdateForRevivalImpl(
            SvsihkrDataAccess svsihkrDataAccess,
            ServicePaymentContractUpdateForRevival servicePaymentContractUpdateForRevival,
            CiriusConstantService ciriusConstantService) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccess, "svsihkrDataAccess");
        this.servicePaymentContractUpdateForRevival = Objects.requireNonNull(
                servicePaymentContractUpdateForRevival,
                "servicePaymentContractUpdateForRevival");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applicableEndsPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.targetSingleBillingPartyIdentificationTypes = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TYPE,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TOTAL),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TYPE,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType
                                .SINGLE_SAME_NAME_SAME_PAYMENT_METHOD));
    }

    /**
     * 概要: 他商材のサービス契約・支払契約関連を取得し、契約復活更新を実行する。
     *
     * @param svkyNo サービス契約番号
     * @param tyenYmd 適用終了年月日
     * @param sikSvcShbt 請求サービス種別
     */
    @Override
    public void execute(
            Long svkyNo,
            LocalDate tyenYmd,
            String sikSvcShbt) {
        for (SvsihkrRecord relatedRecord : svsihkrDataAccess
                .findAllBySnskSvkyNoAndTyenYmdAndSihKiyJtkbAndTdssSbtShbt(
                        svkyNo,
                        tyenYmd,
                        applicableEndsPaymentContractStatus,
                        targetSingleBillingPartyIdentificationTypes)) {
            servicePaymentContractUpdateForRevival.execute(
                    relatedRecord.sihKiyNo(),
                    relatedRecord.svkyNo(),
                    relatedRecord.skno(),
                    sikSvcShbt);
        }
    }
}
