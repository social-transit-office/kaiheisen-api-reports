/**
 * 概要: SV-044（サービス契約・支払契約関連情報更新（契約終了））のサービス実装。
 * 主な仕様:
 * - 入力レコードを契約終了状態へ更新した後、他商材情報を取得して同様の更新を繰り返す。
 * - 下位 DA が DataInconsistencyException をスローした場合は、必要な入力値を補って送出する。
 * 既知の制約:
 * - hukkSysShbt と systemDatetime は RequestContext から取得する。
 */
public class ServicePaymentContractInfoUpdateForTerminationImpl
        implements ServicePaymentContractInfoUpdateForTermination {

    static final String SERVICE_USER_ID = "KHSS044";

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String paymentContractStatusApplying;

    private final String paymentContractStatusApplicableEnds;

    private final List<String> targetSingleBillingPartyIdentificationTypes;

    /**
     * 概要: SV-044 実装を初期化する。
     * 主な仕様:
     * - 必須依存を fail-fast で検証する。
     * - 契約終了更新と他商材検索に必要な定数値を事前解決する。
     * 既知の制約:
     * - RequestContext の実値は execute 呼出し時に評価する。
     *
     * @param svsihkrDataAccess サービス契約・支払契約関連情報 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ServicePaymentContractInfoUpdateForTerminationImpl(
            final SvsihkrDataAccess svsihkrDataAccess,
            final RequestContext requestContext,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccess, "svsihkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.paymentContractStatusApplying = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.paymentContractStatusApplicableEnds = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.targetSingleBillingPartyIdentificationTypes = List.of(
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TYPE,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TOTAL),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.TYPE,
                        ConstantLookupKeys.Cirius.SingleBillingPartyIdentificationType.SINGLE_SAME_NAME_SAME_PAYMENT_METHOD));
    }

    /**
     * 概要: 支払契約番号に紐づくサービス契約・支払契約関連情報を契約終了状態へ更新する。
     * 主な仕様:
        * - 初回更新を行い、下位 DA が業務例外をスローした場合は後続処理へ進めない。
        * - 他商材ごとに同じ契約終了更新を繰り返す。
     * 既知の制約:
     * - public 引数の null ガードは追加しない。
     *
     * @param svckyaInf サービス契約情報
     * @param svsihkrInf サービス契約・支払契約関連情報
     */
    @Override
    public void execute(final SvckyaInf svckyaInf, final SvsihkrInf svsihkrInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        updateServicePaymentRelation(
                svsihkrInf.sihKiyNo(),
                svsihkrInf.svkyNo(),
                svsihkrInf.skno(),
                svsihkrInf.recUpdJnjNo(),
                svsihkrInf.ikskEndYm(),
                systemDate,
                auditContext);

        List<SvsihkrRecord> relatedRecords = svsihkrDataAccess
                .findAllBySnskSvkyNoAndSihKiyJtkbAndTdssSbtShbt(
                        svckyaInf.svkyNo(),
                        paymentContractStatusApplying,
                        targetSingleBillingPartyIdentificationTypes);
        for (SvsihkrRecord relatedRecord : relatedRecords) {
            updateServicePaymentRelation(
                    relatedRecord.sihKiyNo(),
                    relatedRecord.svkyNo(),
                    relatedRecord.skno(),
                    relatedRecord.recUpdJnjNo(),
                    svsihkrInf.ikskEndYm(),
                    systemDate,
                    auditContext);
        }
    }

    /**
     * 概要: サービス契約・支払契約関連情報を契約終了状態へ更新する。
     * 主な仕様:
     * - 適用終了年月日、支払契約状態区分、一括請求終了年月を更新する。
        * - 下位 DA が generic な DataInconsistencyException をスローした場合は、SV 入力値を付与して再構築する。
     * 既知の制約:
     * - recUpdJnjNo は null の場合でも updateKey に保持する。
     *
     * @param sihKiyNo 支払契約番号
     * @param svkyNo サービス契約番号
     * @param skno 世代管理番号
     * @param recUpdJnjNo レコード更新排他制御順序番号
     * @param ikskEndYm 一括請求終了年月
     * @param systemDate システム日付
     * @param auditContext 更新監査情報
     */
    private void updateServicePaymentRelation(
            final Long sihKiyNo,
            final Long svkyNo,
            final Integer skno,
            final Integer recUpdJnjNo,
            final YearMonth ikskEndYm,
            final LocalDate systemDate,
            final AuditContext auditContext) {
        SvsihkrWriteRecord updateRecord =
                buildUpdateRecord(sihKiyNo, svkyNo, skno, ikskEndYm, systemDate);
        SvsihkrSknoJnjUpdateKey updateKey =
                new SvsihkrSknoJnjUpdateKey(sihKiyNo, svkyNo, skno, recUpdJnjNo);
        try {
            svsihkrDataAccess.updateBySihKiyNoAndSvkyNoAndSknoAndRecUpdJnjNo(
                    updateRecord,
                    EnumSet.of(
                            SvsihkrColumn.TYEN_YMD,
                            SvsihkrColumn.SIH_KIY_JTKB,
                            SvsihkrColumn.IKSK_END_YM),
                    updateKey,
                    auditContext);
        } catch (DataInconsistencyException ex) {
            throw createUpdateDataInconsistency(
                    sihKiyNo,
                    svkyNo,
                    skno,
                    recUpdJnjNo,
                    ikskEndYm,
                    systemDate,
                    auditContext,
                    ex);
        }
    }

    private DataInconsistencyException createUpdateDataInconsistency(
            final Long sihKiyNo,
            final Long svkyNo,
            final Integer skno,
            final Integer recUpdJnjNo,
            final YearMonth ikskEndYm,
            final LocalDate systemDate,
            final AuditContext auditContext,
            final DataInconsistencyException cause) {
        FlatJsonBuilder parameterJsonBuilder = new FlatJsonBuilder()
                .put("sihKiyNo", sihKiyNo)
                .put("svkyNo", svkyNo)
                .put("skno", skno)
                .putIfNonNull("recUpdJnjNo", recUpdJnjNo)
                .put("tyenYmd", systemDate)
                .put("sihKiyJtkb", paymentContractStatusApplicableEnds)
                .put("ikskEndYm", formatYearMonth(ikskEndYm))
                .put("updUsrId", SERVICE_USER_ID)
                .put("updKtnKinoId", auditContext.kinoId());
        return DataInconsistencySupport.create(
                svsihkrDataAccess.tableName(),
                SvsihkrDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier(),
                parameterJsonBuilder,
                cause);
    }

    /**
     * 概要: SVSIHKR 更新用の write record を組み立てる。
     * 主な仕様:
     * - 更新対象列以外は null のまま保持する。
     * - 更新対象列は設計書の SET 句に対応する。
     * 既知の制約:
     * - targetColumns で指定しない列は DA で更新されない前提とする。
     *
     * @param sihKiyNo 支払契約番号
     * @param svkyNo サービス契約番号
     * @param skno 世代管理番号
     * @param ikskEndYm 一括請求終了年月
     * @param systemDate システム日付
     * @return 更新用 write record
     */
    private SvsihkrWriteRecord buildUpdateRecord(
            final Long sihKiyNo,
            final Long svkyNo,
            final Integer skno,
            final YearMonth ikskEndYm,
            final LocalDate systemDate) {
        return new SvsihkrWriteRecord(
                sihKiyNo,
                svkyNo,
                skno,
                null,
                systemDate,
                paymentContractStatusApplicableEnds,
                null,
                null,
                ikskEndYm,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    /**
     * 概要: RequestContext の報告システム種別を報告元機能 ID にデコードする。
     * 主な仕様:
     * - RequestContext から hukkSysShbt を取得し、開閉栓定数サービスで名称へ変換する。
     * - 返却値は AuditContext の kinoId として使用する。
     * - KH001 の name は監査列へ格納する機能 ID と同じ値を返す前提とする。
     * 既知の制約:
     * - hukkSysShbt の補完は行わない。
     *
     * @return デコード済み報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private String formatYearMonth(final YearMonth value) {
        return value == null ? null : value.toString();
    }
