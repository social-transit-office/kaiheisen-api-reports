package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.closing;

import static jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svsihkr.SvsihkrDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO;

/**
 * 概要: SV-114（サービス契約・支払契約関連更新（契約復活））のサービス実装。
 * 支払契約番号、サービス契約番号、および世代管理番号に紐づくサービス契約・支払契約関連を更新する。
 */
public final class ServicePaymentContractUpdateForRevivalImpl
        implements ServicePaymentContractUpdateForRevival {

    static final String SERVICE_USER_ID = "KHSS114";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    static final YearMonth MAX_YEAR_MONTH = YearMonth.of(9999, 12);

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applyingPaymentContractStatus;

    private final String allAtOnceFinalInstallmentBillingServiceType;

    private final String allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType;

    /**
     * 概要: 必須依存と定数を初期化する。
     *
     * @param svsihkrDataAccess サービス契約・支払契約関連 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ServicePaymentContractUpdateForRevivalImpl(
            SvsihkrDataAccess svsihkrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccess, "svsihkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applyingPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.allAtOnceFinalInstallmentBillingServiceType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                        ConstantLookupKeys.Cirius.BillingServiceType.ALL_AT_ONCE_FINAL_INSTALLMENT);
        this.allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                        ConstantLookupKeys.Cirius.BillingServiceType.ALL_AT_ONCE_NEXT_3RD_BUSINESS_YEAR_CLOSING_AVAILABLE_AT_SHORT_NOTICE);
    }

    /**
     * 概要: 支払契約番号、サービス契約番号、世代管理番号に紐づくサービス契約・支払契約関連を契約復活状態へ更新する。
     * 仕様:
     * - 更新対象キーは支払契約番号、サービス契約番号、世代管理番号
     * - 請求サービス種別が一括対象の場合は一括請求終了年月に最大年月を設定する
     *
     * @param sihKiyNo 支払契約番号
     * @param svkyNo サービス契約番号
     * @param skno 世代管理番号
     * @param sikSvcShbt 請求サービス種別
     */
    @Override
    public void execute(
            Long sihKiyNo,
            Long svkyNo,
            Integer skno,
            String sikSvcShbt) {
        YearMonth ikskEndYm = resolveIkskEndYm(sikSvcShbt);
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        int updatedCount = svsihkrDataAccess.updateBySihKiyNoAndSvkyNoAndSkno(
                new SvsihkrWriteRecord(
                        sihKiyNo,
                        svkyNo,
                        skno,
                        null,
                        MAX_DATE,
                        applyingPaymentContractStatus,
                        null,
                        null,
                        ikskEndYm,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(
                        SvsihkrColumn.TYEN_YMD,
                        SvsihkrColumn.SIH_KIY_JTKB,
                        SvsihkrColumn.IKSK_END_YM),
                new SvsihkrSknoOptionalUpdateKey(sihKiyNo, svkyNo, skno, null),
                auditContext);
        if (updatedCount == 0) {
            String ikskEndYmJsonValue = ikskEndYm == null ? null : ikskEndYm.toString();
            throw DataInconsistencySupport.create(
                    svsihkrDataAccess.tableName(),
                    UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()
                            .put("sihKiyNo", sihKiyNo)
                            .put("svkyNo", svkyNo)
                            .put("skno", skno)
                            .put("tyenYmd", MAX_DATE)
                            .put("sihKiyJtkb", applyingPaymentContractStatus)
                            .put("ikskEndYm", ikskEndYmJsonValue)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private YearMonth resolveIkskEndYm(String sikSvcShbt) {
        if (allAtOnceFinalInstallmentBillingServiceType.equals(sikSvcShbt)
                || allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType.equals(
                        sikSvcShbt)) {
            return MAX_YEAR_MONTH;
        }
        return null;
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt().orElseThrow(
                () -> new IllegalStateException(
                        "RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }
}
