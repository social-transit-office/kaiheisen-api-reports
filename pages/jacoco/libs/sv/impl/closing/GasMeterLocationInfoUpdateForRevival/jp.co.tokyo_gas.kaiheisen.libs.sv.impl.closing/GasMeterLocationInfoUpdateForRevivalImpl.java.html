package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.closing;

/**
 * SV-105（ガスメーター設置場所情報更新（契約復活））のサービス実装。
 * 主な仕様:
 * - 導管側の設置場所情報を条件検索し、取得件数分だけ小売側の設置場所情報を確認する。
 * - 小売側情報が存在する場合は SV-119 を呼び出して開栓状態への更新を委譲する。
 * - 小売側情報が見つからない場合は DataInconsistencyException を送出する。
 * 既知の制約:
 * - public API は RequestContext 項目を引数に持たない。
 */
public class GasMeterLocationInfoUpdateForRevivalImpl
        implements GasMeterLocationInfoUpdateForRevival {

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final GasMeterLocationInfoUpdate gasMeterLocationInfoUpdate;

    private final String withoutExclusionDivision;

    private final String openingGasMeterStatus;

    /**
     * SV-105 実装に必要な依存オブジェクトと必須定数を初期化する。
     *
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param gmtsetkDataAccess ガスメーター設置場所＜小売＞データアクセス
     * @param gasMeterLocationInfoUpdate ガスメーター設置場所情報更新サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public GasMeterLocationInfoUpdateForRevivalImpl(
            final GmtsetdDataAccess gmtsetdDataAccess,
            final GmtsetkDataAccess gmtsetkDataAccess,
            final GasMeterLocationInfoUpdate gasMeterLocationInfoUpdate,
            final KaiheisenConstantService kaiheisenConstantService,
            final CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(
                gmtsetdDataAccess,
                "gmtsetdDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(
                gmtsetkDataAccess,
                "gmtsetkDataAccess");
        this.gasMeterLocationInfoUpdate = Objects.requireNonNull(
                gasMeterLocationInfoUpdate,
                "gasMeterLocationInfoUpdate");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.withoutExclusionDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.Exclusive.TYPE,
                ConstantLookupKeys.Kaiheisen.Exclusive.NO_EXCLUSION);
        this.openingGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    /**
     * 契約復活向けのガスメーター設置場所情報更新を実行する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kasYmd 開栓年月日
     * @param hesnYmd 閉栓年月日
     */
    @Override
    public void execute(
            final Long gasMtstNo,
            final LocalDate kasYmd,
            final LocalDate hesnYmd) {
        List<GmtsetdRecord> conduitRecords = gmtsetdDataAccess
                .findAllByGasMtstNoAndKasYmdAndHesnYmd(gasMtstNo, kasYmd, hesnYmd);

        for (GmtsetdRecord conduitRecord : conduitRecords) {
            GmtsetkRecord retailRecord = gmtsetkDataAccess
                    .findByGasMtstNoAndTkysYmd(gasMtstNo, conduitRecord.tkysYmd())
                    .orElseThrow(() -> createRetailLocationDataInconsistency(
                            gasMtstNo,
                            conduitRecord.tkysYmd()));

            gasMeterLocationInfoUpdate.execute(
                    withoutExclusionDivision,
                    new GmtsetdInf(
                            gasMtstNo,
                            conduitRecord.tkysYmd(),
                            conduitRecord.tyenYmd(),
                            null),
                    new GmtsetkInf(
                            retailRecord.gasMtstNo(),
                            retailRecord.tkysYmd(),
                            null),
                    new MtsetkInf(null, null, null),
                    new GmtsetdUpdInf(
                            openingGasMeterStatus,
                            conduitRecord.kasYmd(),
                            null,
                            conduitRecord.kasJysb(),
                            null,
                            null),
                    new MtsetkUpdInf(null, null, null, null, null));
        }
    }

    private DataInconsistencyException createRetailLocationDataInconsistency(
            final Long gasMtstNo,
            final LocalDate tkysYmd) {
        return DataInconsistencySupport.create(
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("tkysYmd", tkysYmd));
    }
}
