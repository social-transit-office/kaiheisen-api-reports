/**
 * SV-039（総合契約の託送サービス契約、託送サービス契約・ガスメーター設置場所関連、サービス契約・メーター設置場所関連更新（契約終了））のサービス実装。
 */
public class ComprehensiveContractTransportServiceUpdateForTerminationImpl
        implements ComprehensiveContractTransportServiceUpdateForTermination {

    static final String SERVICE_USER_ID = "KHSS039";

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final TskgmstDataAccess tskgmstDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applyingTransportContractRelatedStatus;

    private final String applicableEndsTransportContractRelatedStatus;

    private final String relatedEnd;

    private final String underContract;

    private final String endOfContract;

    private final String originalGasMeter;

    private final String dedicatedGasMeter;

    /**
     * コンストラクタ。
     *
     * @param tkssvkyDataAccess 託送サービス契約 DA
     * @param tskgmstDataAccess 託送サービス契約・ガスメーター設置場所関連 DA
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public ComprehensiveContractTransportServiceUpdateForTerminationImpl(
            TkssvkyDataAccess tkssvkyDataAccess,
            TskgmstDataAccess tskgmstDataAccess,
            SvmstkrDataAccess svmstkrDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.applyingTransportContractRelatedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.applicableEndsTransportContractRelatedStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLICABLE_ENDS);
        this.relatedEnd = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_END);
        this.underContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.endOfContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.END_OF_CONTRACT);
        this.originalGasMeter = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedGasMeter = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
    }

    @Override
    public void execute(
            LocalDate tyenYmd,
            String tksuSvkeJysb,
            GmtsetdInf gmtsetdInf,
            List<SogoTmtrInf> sogoTmtrInfList) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        for (SogoTmtrInf sogoTmtrInf : normalizeSogoTmtrInfList(sogoTmtrInfList)) {
            updateTransportContractsIfNeeded(
                    sogoTmtrInf,
                    tyenYmd,
                    tksuSvkeJysb,
                    auditContext,
                    decodedOriginFunctionId);
            updateMeterLocationRelation(
                    sogoTmtrInf.svmstkrInf(),
                    tyenYmd,
                    auditContext);
            updateSpecialOverallTransportContractsIfNeeded(
                    gmtsetdInf,
                    sogoTmtrInf,
                    systemDate,
                    tyenYmd,
                    tksuSvkeJysb,
                    auditContext,
                    decodedOriginFunctionId);
        }
    }

    private List<SogoTmtrInf> normalizeSogoTmtrInfList(List<SogoTmtrInf> sogoTmtrInfList) {
        return sogoTmtrInfList == null ? List.of() : sogoTmtrInfList;
    }

    private void updateTransportContractsIfNeeded(
            SogoTmtrInf sogoTmtrInf,
            LocalDate tyenYmd,
            String tksuSvkeJysb,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        if (dedicatedGasMeter.equals(sogoTmtrInf.gmtsetdInf().ckrtGmscJtkb())) {
            return;
        }
        resolveSingleTransportContractRelation(
                        sogoTmtrInf.svmstkrInf().mtstNo(),
                        applyingTransportContractRelatedStatus)
                .ifPresent(record -> updateTransportContractPair(
                        record,
                        tyenYmd,
                        tksuSvkeJysb,
                        auditContext,
                        decodedOriginFunctionId));
    }

    private void updateSpecialOverallTransportContractsIfNeeded(
            GmtsetdInf rootGmtsetdInf,
            SogoTmtrInf sogoTmtrInf,
            LocalDate systemDate,
            LocalDate tyenYmd,
            String tksuSvkeJysb,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        if (!originalGasMeter.equals(sogoTmtrInf.gmtsetdInf().ckrtGmscJtkb())) {
            return;
        }

        List<GmtsetdRecord> specialMeters =
                gmtsetdDataAccess.findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                        rootGmtsetdInf.szcGkkCd(),
                        rootGmtsetdInf.szcCmazCd(),
                        rootGmtsetdInf.szcMchCd(),
                        sogoTmtrInf.svmstkrInf().mtstNo(),
                        sogoTmtrInf.svmstkrInf().mtstNo(),
                        systemDate);

        for (GmtsetdRecord specialMeter : specialMeters) {
            resolveSingleTransportContractRelation(
                            specialMeter.gasMtstNo(),
                            applyingTransportContractRelatedStatus)
                    .ifPresent(record -> updateTransportContractPair(
                            record,
                            tyenYmd,
                            tksuSvkeJysb,
                            auditContext,
                            decodedOriginFunctionId));
        }
    }

    private java.util.Optional<TskgmstRecord> resolveSingleTransportContractRelation(
            Long gasMtstNo,
            String tksuSvkkJtkb) {
        List<TskgmstRecord> transportContractRelations =
                tskgmstDataAccess.findAllByGasMtstNoAndTksuSvkkJtkb(gasMtstNo, tksuSvkkJtkb);
        if (transportContractRelations.size() != 1) {
            return java.util.Optional.empty();
        }
        return java.util.Optional.of(transportContractRelations.get(0));
    }

    private void updateTransportContractPair(
            TskgmstRecord transportContractRelation,
            LocalDate tyenYmd,
            String tksuSvkeJysb,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        updateTransportServiceContract(
                transportContractRelation,
                tyenYmd,
                tksuSvkeJysb,
                auditContext,
                decodedOriginFunctionId);
        updateTransportContractRelation(
                transportContractRelation,
                tyenYmd,
                auditContext);
    }

    private void updateTransportServiceContract(
            TskgmstRecord transportContractRelation,
            LocalDate tyenYmd,
            String tksuSvkeJysb,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = tkssvkyDataAccess.updateByTksuSvkyNoAndTksuSvkyJtkb(
                new TkssvkyWriteRecord(
                        transportContractRelation.tksuSvkyNo(),
                        null,
                        endOfContract,
                        null,
                        tksuSvkeJysb,
                        null,
                        tyenYmd,
                        null,
                        null,
                        null,
                        null,
                        null),
                EnumSet.of(
                        TkssvkyColumn.TKSU_SVKY_JTKB,
                        TkssvkyColumn.TKSU_SVKE_JYSB,
                        TkssvkyColumn.TYEN_YMD),
                new TkssvkyJtkbUpdateKey(
                        transportContractRelation.tksuSvkyNo(),
                        underContract),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    tkssvkyDataAccess.tableName(),
                    TkssvkyDataAccess.CallSpecs.UPDATE_BY_TKSU_SVKY_NO_AND_TKSU_SVKY_JTKB.identifier(),
                    new FlatJsonBuilder()
                            .put("tksuSvkyNo", transportContractRelation.tksuSvkyNo())
                            .put("tksuSvkyJtkb", underContract)
                            .put("newTksuSvkyJtkb", endOfContract)
                            .put("tksuSvkeJysb", tksuSvkeJysb)
                            .put("tyenYmd", tyenYmd)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private void updateTransportContractRelation(
            TskgmstRecord transportContractRelation,
            LocalDate tyenYmd,
            AuditContext auditContext) {
        tskgmstDataAccess.updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TskgmstWriteRecord(
                        transportContractRelation.gasMtstNo(),
                        transportContractRelation.tksuSvkyNo(),
                        transportContractRelation.skno(),
                        applicableEndsTransportContractRelatedStatus,
                        transportContractRelation.tkysYmd(),
                        tyenYmd),
                EnumSet.of(TskgmstColumn.TKSU_SVKK_JTKB, TskgmstColumn.TYEN_YMD),
                new TskgmstUpdateKey(
                        transportContractRelation.gasMtstNo(),
                        transportContractRelation.tksuSvkyNo(),
                        transportContractRelation.skno(),
                        transportContractRelation.recUpdJnjNo()),
                auditContext);
    }

    private void updateMeterLocationRelation(
            ComprehensiveContractSvmstkrInf svmstkrInf,
            LocalDate tyenYmd,
            AuditContext auditContext) {
        svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                new SvmstkrWriteRecord(
                        svmstkrInf.mtstNo(),
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        null,
                        null,
                        relatedEnd,
                        null,
                        tyenYmd,
                        null),
                EnumSet.of(SvmstkrColumn.RKKI_END_YMD, SvmstkrColumn.RKKI_KNRN_JTKB),
                new SvmstkrSknoOptionalUpdateKey(
                        svmstkrInf.mtstNo(),
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        svmstkrInf.recUpdJnjNo()),
                auditContext);
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requireHukkSysShbt());
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
}
