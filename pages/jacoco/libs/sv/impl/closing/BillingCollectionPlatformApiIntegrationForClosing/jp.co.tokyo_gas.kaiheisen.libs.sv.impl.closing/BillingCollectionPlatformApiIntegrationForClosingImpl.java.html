/**
 * 概要: SV-034（請求回収基盤API連携（閉栓））の実装。
 * 主な仕様:
 * - step 1 で請求管理番号と請求対象契約番号の整合を判定し、業務エラーを送出する。
 * - step 2〜4 で他-045 向け連携値を編集する。
 * - step 5 で SV-035、step 6 で 他-045 を連鎖呼び出しする。
 * 既知の制約:
 * - RequestContext から参照する値のうち、本機能では systemDatetime のみを必須値として扱う。
 */
public final class BillingCollectionPlatformApiIntegrationForClosingImpl
        implements BillingCollectionPlatformApiIntegrationForClosing {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final long FIXED_GAS_METER_LOCATION_NUMBER = 11111111111L;

    private static final int ZERO_DATE = 0;

    private final ServiceSuspensionCancellationReasonTypeAcquisition reasonTypeAcquisition;

    private final HeisenkannrenjiIdoujouhouTourokuCiriusAdapter other045Adapter;

    private final RequestContext requestContext;

    private final String presenceFlagNothing;

    private final String presenceFlagYes;

    private final String paymentContractStatusApplying;

    private final String paymentContractStatusApplicableEnds;

    private final String onlineCancellationReport;

    private final String billablePaymentManagementTransfer;

    private final String billableContractManagementTransfer;

    /**
     * 概要: 必須依存と定数を解決して初期化する。
     *
     * @param reasonTypeAcquisition SV-035 API
     * @param other045Adapter 他-045 adapter
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param requestContext RequestContext
     */
    @Inject
    public BillingCollectionPlatformApiIntegrationForClosingImpl(
            ServiceSuspensionCancellationReasonTypeAcquisition reasonTypeAcquisition,
            HeisenkannrenjiIdoujouhouTourokuCiriusAdapter other045Adapter,
            CiriusConstantService ciriusConstantService,
            RequestContext requestContext) {
        this.reasonTypeAcquisition = Objects.requireNonNull(reasonTypeAcquisition, "reasonTypeAcquisition");
        this.other045Adapter = Objects.requireNonNull(other045Adapter, "other045Adapter");
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.presenceFlagNothing = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.NOTHING);
        this.presenceFlagYes = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.YES);
        this.paymentContractStatusApplying = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.paymentContractStatusApplicableEnds = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.onlineCancellationReport = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.ONLINE_CANCELLATION_REPORT);
        this.billablePaymentManagementTransfer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_PAYMENT_MANAGEMENT_TRANSFER);
        this.billableContractManagementTransfer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.BILLABLE_CONTRACT_MANAGEMENT_TRANSFER);
    }

    /** 概要: 閉栓時の請求回収基盤連携を実行する。 */
    @Override
    public BillingCollectionPlatformApiIntegrationForClosingOutput execute(
            LocalDate tyenYmd,
            String svkyEndJysb,
            BillingCollectionPlatformApiIntegrationForClosingSvckyaInf svckyaInf,
            SihakyInf sihakyInf,
            int suguTgmtKnsu,
            String isrrGymCd,
            LocalDate rkkiSttYmd,
            boolean sihakyEndFlg,
            String hesnJysb,
            YearMonth ikskEndYm) {
        LocalDateTime systemDatetime = requireRequestContext();

        Long originalBillingManagementNumberValue = sihakyInf.sikKrno();
        String targetContractNumber = svckyaInf.sqtiKiyNo();
        if (originalBillingManagementNumberValue == null && targetContractNumber == null) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000078);
        }

        boolean paymentContractEndTarget = sihakyEndFlg;
        Long billingManagementNumber = paymentContractEndTarget
                ? originalBillingManagementNumberValue
                : Long.valueOf(0L);

        String paymentContractStatus = paymentContractEndTarget
                ? paymentContractStatusApplicableEnds
                : paymentContractStatusApplying;
        LocalDate billingEndDate = paymentContractEndTarget ? systemDatetime.toLocalDate() : null;

        long gasMeterLocationNumber = suguTgmtKnsu > 0
                ? svckyaInf.svkyNo()
                : FIXED_GAS_METER_LOCATION_NUMBER;
        boolean onlineCancellation = onlineCancellationReport.equals(isrrGymCd);

        ServiceSuspensionCancellationReasonTypeAcquisitionOutput reasonTypeOutput =
                reasonTypeAcquisition.execute(svkyEndJysb, hesnJysb);

        HeisenkannrenjiIdoujouhou other045Payload = buildOther045Request(
                svckyaInf,
                billingManagementNumber,
                paymentContractStatus,
                gasMeterLocationNumber,
                tyenYmd,
                rkkiSttYmd,
                billingEndDate,
                onlineCancellation,
                systemDatetime,
                ikskEndYm,
                reasonTypeOutput.svkjyShbt());
        validateSequentialControlKeys(other045Payload);
        other045Adapter.register(new HeisenkannrenjiIdoujouhouTourokuRequest(other045Payload));

        return new BillingCollectionPlatformApiIntegrationForClosingOutput(
                gasMeterLocationNumber,
                false,
                onlineCancellation,
                svckyaInf.svkyNo(),
                tyenYmd,
                billingManagementNumber,
                targetContractNumber);
    }

    /** 概要: 他-045 連携 DTO を組み立てる。 */
    private HeisenkannrenjiIdoujouhou buildOther045Request(
            BillingCollectionPlatformApiIntegrationForClosingSvckyaInf svckyaInf,
            Long billingManagementNumber,
            String paymentContractStatus,
            long gasMeterLocationNumber,
            LocalDate tyenYmd,
            LocalDate rkkiSttYmd,
            LocalDate billingEndDate,
            boolean onlineCancellation,
            LocalDateTime systemDatetime,
            YearMonth ikskEndYm,
            String svkjyShbt) {
        String skrkGrpkJhu = "";
        Long skrkSqkrJhu = billingManagementNumber;
        String skrkSqtkJhu = svckyaInf.sqtiKiyNo();
        Long svkyNo = svckyaInf.svkyNo();
        Integer svkySkno = svckyaInf.skno();
        String sihKiyJtkb = paymentContractStatus;
        Long gasMtstNo = gasMeterLocationNumber;
        Integer tkysYmd = toYmd(rkkiSttYmd);
        Integer khsYmd = toYmd(tyenYmd);
        String trksUmFlg = presenceFlagNothing;
        String sikeKykUmFlg = presenceFlagNothing;
        String tishSikeGrpUmFlg = presenceFlagNothing;
        String tishSikeGrpTishUmFlg = presenceFlagNothing;
        String olkyUmFlg = onlineCancellation ? presenceFlagYes : presenceFlagNothing;
        Integer shrYmd = toYmd(systemDatetime.toLocalDate());
        String stskHkirIsrrGymCd = billablePaymentManagementTransfer;
        Integer sikEndYmd = toYmdOrZero(billingEndDate);
        Integer svtkYmd = toYmd(tyenYmd);
        Integer kiyEndYmd = toYmdOrZero(tyenYmd);
        String stkrHkirIsrrGymCd = billableContractManagementTransfer;
        Integer convertedIkskEndYm = toYmOrZero(ikskEndYm);
        String stsiHkirIrimSksgNo = "";
        String stsiHkirIsrrGymCd = "";
        Long stsiHkirSqtsIriNo = 0L;
        Integer sqtsEndYmd = ZERO_DATE;
        String sqttHkirIrimSksgNo = "";
        String sqttHkirIsrrGymCd = "";
        Long sqttHkirSqtsIriNo = 0L;
        Integer sqttSqsrYmd = ZERO_DATE;

        return new HeisenkannrenjiIdoujouhou(
                skrkGrpkJhu,
                skrkSqkrJhu,
                skrkSqtkJhu,
                svkyNo,
                svkySkno,
                sihKiyJtkb,
                gasMtstNo,
                tkysYmd,
                khsYmd,
                trksUmFlg,
                sikeKykUmFlg,
                tishSikeGrpUmFlg,
                tishSikeGrpTishUmFlg,
                olkyUmFlg,
                shrYmd,
                stskHkirIsrrGymCd,
                sikEndYmd,
                svkjyShbt,
                svtkYmd,
                kiyEndYmd,
                stkrHkirIsrrGymCd,
                convertedIkskEndYm,
                stsiHkirIrimSksgNo,
                stsiHkirIsrrGymCd,
                stsiHkirSqtsIriNo,
                sqtsEndYmd,
                sqttHkirIrimSksgNo,
                sqttHkirIsrrGymCd,
                sqttHkirSqtsIriNo,
                sqttSqsrYmd);
    }

    /** 概要: 他-045 送信時に Gx・9x・Sx のいずれかが設定されていることを確認する。 */
    private void validateSequentialControlKeys(HeisenkannrenjiIdoujouhou payload) {
        boolean groupKeyMissing = payload.skrkGrpkJhu() == null || payload.skrkGrpkJhu().isBlank();
        boolean billingKeyMissing = payload.skrkSqkrJhu() == null || payload.skrkSqkrJhu() == 0L;
        boolean contractKeyMissing = payload.skrkSqtkJhu() == null || payload.skrkSqtkJhu().isBlank();
        if (groupKeyMissing && billingKeyMissing && contractKeyMissing) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000078);
        }
    }

    /** 概要: 実行に必要な RequestContext.systemDatetime を取得する。 */
    private LocalDateTime requireRequestContext() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /** 概要: LocalDate を yyyyMMdd 形式の数値へ変換する。 */
    private Integer toYmd(LocalDate value) {
        return value == null ? null : Integer.valueOf(value.format(BASIC_DATE));
    }

    /** 概要: LocalDate の null を 0、それ以外を yyyyMMdd 形式の数値へ変換する。 */
    private Integer toYmdOrZero(LocalDate value) {
        return value == null ? ZERO_DATE : Integer.valueOf(value.format(BASIC_DATE));
    }

    /** 概要: YearMonth を yyyyMM 形式の数値へ変換する。 */
    private Integer toYmOrZero(YearMonth value) {
        return value == null ? ZERO_DATE : value.toYmInt();
    }
}
