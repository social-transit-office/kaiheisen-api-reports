package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Comparator;
import java.util.List;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DateCalculationUtil;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmkskkd.GmkskkdRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ktmgmkk.KtmgmkkRecord;

/** 直近検針情報取得サービス実装。 */
public class RecentMeterReadingInfoAcquisitionImpl implements RecentMeterReadingInfoAcquisition {

    private static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final RequestContext requestContext;

    private final String excludedRegularReasonType;

    private final String supplyOutageReasonType;

    private final String supplyStopPaymentPendingStatus;

    /**
     * コンストラクタ。
     *
     * @param gmkskkdDataAccess ガスメーター検針結果データアクセス
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果データアクセス
     * @param kkytskrDataAccess 供給停止管理データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public RecentMeterReadingInfoAcquisitionImpl(
            GmkskkdDataAccess gmkskkdDataAccess,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gmkskkdDataAccess = gmkskkdDataAccess;
        this.ktmgmkkDataAccess = ktmgmkkDataAccess;
        this.kkytskrDataAccess = kkytskrDataAccess;
        this.requestContext = requestContext;
        this.excludedRegularReasonType =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR);
        this.supplyOutageReasonType =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .SUPPLY_OUTAGE);
        this.supplyStopPaymentPendingStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                        ConstantLookupKeys.Cirius.SupplyStopPaymentStatus
                                .SUPPLY_STOP_PAYMENT_PENDING);
    }

    @Override
    public RecentMeterReadingInfoAcquisitionOutput execute(Long gasMtstNo) {
        LocalDateTime systemDatetime =
                requestContext
                        .getSystemDatetime()
                .orElseThrow(
                    () ->
                        new IllegalStateException(
                            "RequestContext.systemDatetime is required"));
        LocalDate systemDate = systemDatetime.toLocalDate();
        LocalDate fromDate = DateCalculationUtil.firstDayMonthsAgo(systemDate, 6);

        Optional<KtmgmkkRecord> preConfirmation = findLatestPreConfirmation(gasMtstNo);
        Optional<GmkskkdRecord> pipeline =
                findLatestDoukanInSearchWindows(gasMtstNo, fromDate, systemDate);
        String pipelineTableName = gmkskkdDataAccess.tableName();

        LocalDate knsYmd = null;
        Long gmtShjSu = null;
        String knkkSksiJysb = null;

        if (preConfirmation.isPresent()) {
            KtmgmkkRecord preConfirmationRecord = preConfirmation.get();
            if (pipeline.isPresent()) {
                GmkskkdRecord pipelineRecord = pipeline.get();
                if (preConfirmationRecord.knsYmd().isBefore(pipelineRecord.knsYmd())) {
                    knsYmd = pipelineRecord.knsYmd();
                    gmtShjSu = pipelineRecord.gmtShjSu();
                    knkkSksiJysb = pipelineRecord.knkkSksiJysb();
                } else {
                    knsYmd = preConfirmationRecord.knsYmd();
                    gmtShjSu = preConfirmationRecord.gmtShjSu();
                    knkkSksiJysb = preConfirmationRecord.knkkSksiJysb();
                }
            } else {
                knsYmd = preConfirmationRecord.knsYmd();
                gmtShjSu = preConfirmationRecord.gmtShjSu();
                knkkSksiJysb = preConfirmationRecord.knkkSksiJysb();
            }
        } else if (pipeline.isPresent()) {
            GmkskkdRecord pipelineRecord = pipeline.get();
            knsYmd = pipelineRecord.knsYmd();
            gmtShjSu = pipelineRecord.gmtShjSu();
            knkkSksiJysb = pipelineRecord.knkkSksiJysb();
        } else {
            throw new DataNotFoundException(pipelineTableName);
        }

        Optional<KkytskrRecord> supplyStopData = findSupplyStopData(gasMtstNo);
        if (supplyStopData.isPresent() && supplyStopData.get().hesnYmd() != null) {
            KkytskrRecord supplyStopRecord = supplyStopData.get();
            if (knsYmd == null || !knsYmd.isAfter(supplyStopRecord.hesnYmd())) {
                knsYmd = supplyStopRecord.hesnYmd();
                gmtShjSu = supplyStopRecord.gmtShjSu();
                knkkSksiJysb = supplyOutageReasonType;
            }
        }

        return new RecentMeterReadingInfoAcquisitionOutput(knsYmd, gmtShjSu, knkkSksiJysb);
    }

    /**
     * 概要: 確定前ガスメーター検針結果を取得する。 主な仕様: - DA-052-R006 を呼び出し、定例以外の最新 1 件を取得する。 既知の制約: - 検索期間の絞り込みは
     * DA-052-R006 の契約外のため行わない。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 確定前最新候補
     */
    private Optional<KtmgmkkRecord> findLatestPreConfirmation(Long gasMtstNo) {
        return ktmgmkkDataAccess.findByGasMtstNoAndKnkkSksiJysb(
                gasMtstNo, excludedRegularReasonType);
    }

    /**
     * 概要: 導管データの探索ウィンドウ（6か月、17か月、27か月）で最新候補を取得する。 主な仕様: - 6か月範囲で導管データが見つかれば返却する。 -
     * 見つからない場合は17か月範囲、さらに見つからない場合は27か月範囲で探索する。 既知の制約: - 探索開始日は systemDatetime 基準の月初日を利用する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param fromDate 初回探索開始日
     * @param systemDate システム日付
     * @return 導管最新候補
     */
    private Optional<GmkskkdRecord> findLatestDoukanInSearchWindows(
            Long gasMtstNo, LocalDate fromDate, LocalDate systemDate) {
        Optional<GmkskkdRecord> firstWindow = findLatest(gasMtstNo, fromDate, MAX_DATE);
        if (firstWindow.isPresent()) {
            return firstWindow;
        }

        LocalDate secondWindowStart = DateCalculationUtil.firstDayMonthsAgo(systemDate, 17);
        LocalDate secondWindowEnd =
                DateCalculationUtil.firstDayMonthsAgo(systemDate, 6).minusDays(1);
        Optional<GmkskkdRecord> secondWindow =
                findLatest(gasMtstNo, secondWindowStart, secondWindowEnd);
        if (secondWindow.isPresent()) {
            return secondWindow;
        }

        LocalDate thirdWindowStart = DateCalculationUtil.firstDayMonthsAgo(systemDate, 27);
        LocalDate thirdWindowEnd =
                DateCalculationUtil.firstDayMonthsAgo(systemDate, 17).minusDays(1);
        return findLatest(gasMtstNo, thirdWindowStart, thirdWindowEnd);
    }

    /**
     * 概要: 指定範囲の検針結果から最新候補を取得する。 主な仕様: - DA-027-R003 を呼び出す。 - 検針年月日、検針結果通し番号の降順で最新候補を選択する。 既知の制約: -
     * 取得対象は gmkskkdDataAccess の提供範囲に依存する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param fromDate 検索開始日
     * @param toDate 検索終了日
     * @return 最新候補
     */
    private Optional<GmkskkdRecord> findLatest(
            Long gasMtstNo, LocalDate fromDate, LocalDate toDate) {
        List<GmkskkdRecord> records =
                gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmd(gasMtstNo, fromDate, toDate);
        return records.stream()
                .max(
                        Comparator.comparing(GmkskkdRecord::knsYmd)
                                .thenComparing(GmkskkdRecord::knsKkaTsno));
    }

    /**
     * 概要: 供給停止情報を取得する。 主な仕様: - DA-055-R003 を呼び出す。 - `kasYmd` は {@code null} を渡し、DA 側で 0
     * 相当へ補完して検索する。 既知の制約: - 供給停止精算状態区分は定数マスタに登録された値を前提とする。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 供給停止情報
     */
    private Optional<KkytskrRecord> findSupplyStopData(Long gasMtstNo) {
        return kkytskrDataAccess.findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
                gasMtstNo, null, supplyStopPaymentPendingStatus);
    }
}
