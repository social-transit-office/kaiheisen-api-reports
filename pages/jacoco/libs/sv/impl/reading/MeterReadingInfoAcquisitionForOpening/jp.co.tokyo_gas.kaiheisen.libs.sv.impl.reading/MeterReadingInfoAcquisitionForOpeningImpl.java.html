package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReportDeadlineAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.OpeningReportDeadlineAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingInfoAcquisitionForOpening;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingInfoAcquisitionForOpeningOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisitionOutput;

/**
 * 概要: 検針情報取得（開栓）サービス実装。
 * 主な仕様:
 * - SV-026 と SV-025 を順番に呼び出し、開栓用の検針情報を組み立てて返却する。
 * - SV 規約に従い、報告システム種別、相関ID、システム日時は下位 SV 側で RequestContext から扱う。
 * 既知の制約:
 * - 本クラスは入力妥当性検証を追加せず、下位 SV の例外も変換せずに上位へ伝播する。
 */
public class MeterReadingInfoAcquisitionForOpeningImpl implements MeterReadingInfoAcquisitionForOpening {

    private final RecentMeterReadingInfoAcquisition recentMeterReadingInfoAcquisition;

    private final OpeningReportDeadlineAcquisition openingReportDeadlineAcquisition;

    /**
     * 概要: 依存 SV を受け取り、実装を初期化するコンストラクタ。
     * 主な仕様:
     * - 直近検針情報取得と開栓報告可能期限日取得の 2 サービスを保持する。
     * - DI 構成不備は fail-fast で検知する。
     * 既知の制約:
     * - public メソッド引数の妥当性検証は本コンストラクタでは扱わない。
     *
     * @param recentMeterReadingInfoAcquisition SV-026
     * @param openingReportDeadlineAcquisition SV-025
     */
    @Inject
    public MeterReadingInfoAcquisitionForOpeningImpl(
            RecentMeterReadingInfoAcquisition recentMeterReadingInfoAcquisition,
            OpeningReportDeadlineAcquisition openingReportDeadlineAcquisition) {
        this.recentMeterReadingInfoAcquisition =
            Objects.requireNonNull(
                recentMeterReadingInfoAcquisition, "recentMeterReadingInfoAcquisition");
        this.openingReportDeadlineAcquisition =
            Objects.requireNonNull(
                openingReportDeadlineAcquisition, "openingReportDeadlineAcquisition");
    }

    @Override
    public MeterReadingInfoAcquisitionForOpeningOutput execute(Long gasMtstNo) {
        RecentMeterReadingInfoAcquisitionOutput recentOutput = recentMeterReadingInfoAcquisition.execute(gasMtstNo);
        OpeningReportDeadlineAcquisitionOutput deadlineOutput = openingReportDeadlineAcquisition.execute(
                gasMtstNo,
                recentOutput.knsYmd(),
                recentOutput.knkkSksiJysb());
        return new MeterReadingInfoAcquisitionForOpeningOutput(
                recentOutput.knsYmd(),
                recentOutput.gmtShjSu(),
                recentOutput.knkkSksiJysb(),
                deadlineOutput.ksenHkkKnuKigenYmd());
    }
}
