package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import java.time.LocalDate;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsyti.KnsytiDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsyti.KnsytiRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.NextMeterReadingScheduledDateConfirmation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.NextMeterReadingScheduledDateAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.NextMeterReadingScheduledDateAcquisitionOutput;

/** 次回検針予定年月日取得サービス実装。 */
public class NextMeterReadingScheduledDateAcquisitionImpl
        implements NextMeterReadingScheduledDateAcquisition {

    private final KnsytiDataAccess knsytiDataAccess;

    private final NextMeterReadingScheduledDateConfirmation
            nextMeterReadingScheduledDateConfirmation;

    private final RequestContext requestContext;

    /**
     * コンストラクタ。
     *
     * @param knsytiDataAccess 検針予定データアクセス
     * @param nextMeterReadingScheduledDateConfirmation 次回検針予定日確定処理サービス
     * @param requestContext リクエストコンテキスト
     */
    public NextMeterReadingScheduledDateAcquisitionImpl(
            KnsytiDataAccess knsytiDataAccess,
            NextMeterReadingScheduledDateConfirmation nextMeterReadingScheduledDateConfirmation,
            RequestContext requestContext) {
        this.knsytiDataAccess = knsytiDataAccess;
        this.nextMeterReadingScheduledDateConfirmation = nextMeterReadingScheduledDateConfirmation;
        this.requestContext = requestContext;
    }

    /**
     * 概要: 検索基準年月日とシステム日時の年月で検針予定を探索し、次回検針予定年月日を返却する。 主な仕様: - まず検索基準年月日の年月で DA-067-R001 を検索する。 -
        * 0件の場合は RequestContext の systemDatetime の年月で再検索する。 - SV-047 で定例検針予定年月日を確定し、基準日未満なら翌月で再確定する。
        * 既知の制約: - RequestContext に systemDatetime が未設定の場合は IllegalStateException を送出する。 -
     * 取得候補が複数ある場合は先頭レコードを採用する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kskKjnYmd 検索基準年月日
     * @return 次回検針予定年月日取得結果
     */
    @Override
    public NextMeterReadingScheduledDateAcquisitionOutput execute(
            Long gasMtstNo, LocalDate kskKjnYmd) {
        KnsytiRecord scheduledRecord = findScheduledRecord(gasMtstNo, kskKjnYmd);
        LocalDate tkytYmd = resolveScheduledDate(scheduledRecord, kskKjnYmd);
        return toOutput(scheduledRecord, tkytYmd);
    }

    /**
     * 概要: 検針予定を検索し、検索基準年月日でヒットしない場合はシステム日時年月で再検索する。 主な仕様: - 1回目の検索は検索基準年月日の年月を使用する。 - 0件時のみ
     * RequestContext の systemDatetime の年月を使用して再検索する。 - 再検索でも0件の場合は処理対象データ無しエラーを送出する。 既知の制約: -
     * 複数件ヒット時は先頭レコードを採用する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param kskKjnYmd 検索基準年月日
     * @return 採用した検針予定レコード
     */
    private KnsytiRecord findScheduledRecord(Long gasMtstNo, LocalDate kskKjnYmd) {
        List<KnsytiRecord> records =
                knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(
                        gasMtstNo, toYearMonth(kskKjnYmd));
        if (!records.isEmpty()) {
            return records.getFirst();
        }

        LocalDate systemDate =
                requestContext
                        .getSystemDatetime()
                        .orElseThrow(
                                () ->
                                        new IllegalStateException(
                                                "RequestContext.systemDatetime is required"))
                        .toLocalDate();
        List<KnsytiRecord> fallbackRecords =
                knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(
                        gasMtstNo, toYearMonth(systemDate));
        if (!fallbackRecords.isEmpty()) {
            return fallbackRecords.getFirst();
        }
        throw new DataNotFoundException(knsytiDataAccess.tableName());
    }

    /**
     * 概要: KNSYTI と基準日から定例検針予定年月日を確定する。 主な仕様: - 最初に基準日で SV-047 を呼び出す。 - 確定した年月日が基準日未満の場合は翌月基準で再度
     * SV-047 を呼び出す。 既知の制約: - SV-047 の例外は補足せずそのまま上位へ送出する。
     *
     * @param scheduledRecord 採用した検針予定レコード
     * @param kskKjnYmd 検索基準年月日
     * @return 確定した定例検針予定年月日
     */
    private LocalDate resolveScheduledDate(KnsytiRecord scheduledRecord, LocalDate kskKjnYmd) {
        LocalDate tkytYmd =
                nextMeterReadingScheduledDateConfirmation
                        .execute(
                                kskKjnYmd,
                                scheduledRecord.stkbKbn(),
                                scheduledRecord.knsHuhuKbn(),
                                scheduledRecord.shtiKskbNo(),
                                scheduledRecord.tkkbNo())
                        .tkytYmd();
        if (!tkytYmd.isBefore(kskKjnYmd)) {
            return tkytYmd;
        }
        return nextMeterReadingScheduledDateConfirmation
                .execute(
                        kskKjnYmd.plusMonths(1),
                        scheduledRecord.stkbKbn(),
                        scheduledRecord.knsHuhuKbn(),
                        scheduledRecord.shtiKskbNo(),
                        scheduledRecord.tkkbNo())
                .tkytYmd();
    }

    /**
     * 概要: KNSYTI レコードと確定日から出力 DTO を構築する。 主な仕様: - KNSYTI 由来の項目はそのまま出力 DTO へ転記する。 - 定例検針予定年月日は SV-047
     * の確定値を使用する。
     *
     * @param scheduledRecord 採用した検針予定レコード
     * @param tkytYmd 確定した定例検針予定年月日
     * @return 次回検針予定年月日取得結果
     */
    private static NextMeterReadingScheduledDateAcquisitionOutput toOutput(
            KnsytiRecord scheduledRecord, LocalDate tkytYmd) {
        return new NextMeterReadingScheduledDateAcquisitionOutput(
                tkytYmd,
                scheduledRecord.tkkbNo(),
                scheduledRecord.knsHuhuKbn(),
                scheduledRecord.stkbKbn(),
                scheduledRecord.shtiKskbNo(),
                scheduledRecord.tkysYm(),
                scheduledRecord.tyenYm(),
                scheduledRecord.recUpdJnjNo());
    }

    /**
     * 概要: LocalDate を YearMonth へ変換する。 主な仕様: - 年月日の年と月のみを使用する。
     *
     * @param date 変換元日付
     * @return YearMonth
     */
    private static jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth toYearMonth(
            LocalDate date) {
        return jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth.of(
                date.getYear(), date.getMonthValue());
    }
}
