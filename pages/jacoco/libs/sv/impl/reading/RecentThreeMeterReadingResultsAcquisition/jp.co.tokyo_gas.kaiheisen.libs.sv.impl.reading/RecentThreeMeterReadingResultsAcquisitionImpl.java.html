/** 検針結果情報取得（直近3回）サービス実装。 */
public class RecentThreeMeterReadingResultsAcquisitionImpl
        implements RecentThreeMeterReadingResultsAcquisition {

    private static final StructuredLogger LOGGER =
            new JbossStructuredLogger(RecentThreeMeterReadingResultsAcquisitionImpl.class);

    private static final Long DEFAULT_GAS_METER_INDICATION = 0L;

    private static final Integer DEFAULT_READING_SEQUENCE = 0;

    private static final Integer DEFAULT_USAGE_DAYS = 0;

    private static final BigDecimal DEFAULT_GAS_USAGE = BigDecimal.ZERO;

    private final GasUsageForBillingDerivation gasUsageForBillingDerivation;

    private final RequestContext requestContext;

    private final String relocationClosureReason;

    private final String openingReason;

    private final String supplyResumedReason;

    private final String temporarySwCancellationReason;

    private final String noRegressionSegment;

    private final String closedValveStatus;

    private final String initialMeterReadingRequiredCode;

    /**
     * 検針結果情報取得（直近3回）サービスの実行に必要な依存関係と業務定数を初期化する。
     *
     * @param gasUsageForBillingDerivation TG導管ガス使用量（料金計算用）取得サービス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUSコンスタント取得サービス
     */
    @Inject
    public RecentThreeMeterReadingResultsAcquisitionImpl(
            GasUsageForBillingDerivation gasUsageForBillingDerivation,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        GasUsageForBillingDerivation nonNullGasUsageForBillingDerivation =
                Objects.requireNonNull(
                        gasUsageForBillingDerivation, "gasUsageForBillingDerivation");
        RequestContext nonNullRequestContext =
                Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.gasUsageForBillingDerivation = nonNullGasUsageForBillingDerivation;
        this.requestContext = nonNullRequestContext;
        this.relocationClosureReason =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .RELOCATION_CLOSURE);
        this.openingReason =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .OPENING);
        this.supplyResumedReason =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .SUPPLY_RESUMED);
        this.regularSwCancellationReason =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .REGULAR_SW_CANCELLATION);
        this.temporarySwCancellationReason =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .TEMPORARY_SW_CANCELLATION);
        this.noRegressionSegment =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.NONE);
        this.closedValveStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
        this.initialMeterReadingRequiredCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.TYPE,
                        ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode
                                .INITIAL_METER_READING_REQUIRED);
    }

    @Override
    public RecentThreeMeterReadingResultsAcquisitionOutput execute(Long gasMtstNo, LocalDate knsYmd) {
        MeterReadingResultState first =
                invokeBestEffort(
                        gasMtstNo,
                        knsYmd,
                        DEFAULT_GAS_METER_INDICATION,
                        relocationClosureReason,
                        noRegressionSegment,
                        closedValveStatus,
                        DEFAULT_GAS_USAGE,
                        new KnsKkaTseiMaeJh(null, null, null));

        MeterReadingResultState second = MeterReadingResultState.defaultState();
        if (shouldInvokeFollowUp(first.znkiKnkkInf().knkkSsjyShbt())) {
            second =
                    invokeBestEffort(
                            gasMtstNo,
                            first.znkiKnkkInf().knsYmd(),
                            first.znkiKnkkInf().gmtShjSu(),
                            first.znkiKnkkInf().knkkSsjyShbt(),
                            first.znkiKnkkInf().sjskKbn(),
                            closedValveStatus,
                            DEFAULT_GAS_USAGE,
                            toKnsKkaTseiMaeJh(first.znkiKnkkInf()));
        }

        MeterReadingResultState third = MeterReadingResultState.defaultState();
        if (shouldInvokeFollowUp(second.znkiKnkkInf().knkkSsjyShbt())) {
            third =
                    invokeBestEffort(
                            gasMtstNo,
                            second.znkiKnkkInf().knsYmd(),
                            second.znkiKnkkInf().gmtShjSu(),
                            second.znkiKnkkInf().knkkSsjyShbt(),
                            second.znkiKnkkInf().sjskKbn(),
                            closedValveStatus,
                            DEFAULT_GAS_USAGE,
                            toKnsKkaTseiMaeJh(second.znkiKnkkInf()));
        }

        if (isSwCancellation(first.znkiKnkkInf().knkkSsjyShbt())) {
            second = second.withZeroGasInfo();
        }
        if (isSwCancellation(second.znkiKnkkInf().knkkSsjyShbt())) {
            third = third.withZeroGasInfo();
        }

        ArrayList<KnsnInf> knsnInf = new ArrayList<>();
        knsnInf.add(createFirstKnsnInf(knsYmd, first));
        knsnInf.add(createSecondKnsnInf(first, second));
        knsnInf.add(createThirdKnsnInf(second, third));
        return new RecentThreeMeterReadingResultsAcquisitionOutput(knsnInf);
    }

    /**
     * ガス使用量取得サービスを best-effort で呼び出し、失敗時は既定値を返す。
     *
     * @param gasMtstNo ガスメーター設置番号
     * @param knsYmd 検針日
     * @param gmtShjSu ガスメーター指示数
     * @param knkkSksiJysb 検針結果作成事由
     * @param sjskKbn 指示数回帰区分
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param sgyjKytiSyry 作業時協定使用量
     * @param knsKkaTseiMaeJh 検針結果訂正前情報
     * @return 取得結果、または既定値で構成した状態
     */
    private MeterReadingResultState invokeBestEffort(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long gmtShjSu,
            String knkkSksiJysb,
            String sjskKbn,
            String gmtKhsJtkb,
            BigDecimal sgyjKytiSyry,
            KnsKkaTseiMaeJh knsKkaTseiMaeJh) {
        try {
            GasUsageForBillingDerivationOutput output =
                    gasUsageForBillingDerivation.execute(
                            gasMtstNo,
                            knsYmd,
                            gmtShjSu,
                            knkkSksiJysb,
                            sjskKbn,
                            gmtKhsJtkb,
                            sgyjKytiSyry,
                            knsKkaTseiMaeJh);
            return MeterReadingResultState.from(output);
        } catch (RuntimeException exception) {
            logInvokeBestEffortFailure(
                    gasMtstNo,
                    knsYmd,
                    gmtShjSu,
                    knkkSksiJysb,
                    sjskKbn,
                    gmtKhsJtkb,
                    exception);
            return MeterReadingResultState.defaultState();
        }
    }

    /**
     * best-effort 呼び出し失敗時に、フォールバック内容を追跡できる例外ログを出力する。
     *
     * @param gasMtstNo ガスメーター設置番号
     * @param knsYmd 検針日
     * @param gmtShjSu ガスメーター指示数
     * @param knkkSksiJysb 検針結果作成事由
     * @param sjskKbn 指示数回帰区分
     * @param gmtKhsJtkb ガスメーター開閉栓状態区分
     * @param exception 発生した実行時例外
     */
    private void logInvokeBestEffortFailure(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long gmtShjSu,
            String knkkSksiJysb,
            String sjskKbn,
            String gmtKhsJtkb,
            RuntimeException exception) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        requestContext.getCorrelationId().ifPresent(v -> attributes.put(LogField.CORRELATION_ID.key(), v));
        requestContext.getHukkSysShbt().ifPresent(v -> attributes.put("hukkSysShbt", v));
        requestContext.getSystemDatetime().ifPresent(v -> attributes.put("systemDatetime", v));
        attributes.put(LogField.EXCEPTION_CLASS.key(), exception.getClass().getName());
        attributes.put("operation", "GasUsageForBillingDerivation.execute");
        attributes.put("gasMtstNo", gasMtstNo);
        attributes.put("knsYmd", knsYmd);
        attributes.put("gmtShjSu", gmtShjSu);
        attributes.put("knkkSksiJysb", knkkSksiJysb);
        attributes.put("sjskKbn", sjskKbn);
        attributes.put("gmtKhsJtkb", gmtKhsJtkb);

        LOGGER.error(
                new LogEvent(
                        "sv.bestEffortFallback",
                        "GasUsageForBillingDerivation.execute failed and default state was used.",
                        LogLayer.SV,
                        500,
                        ErrorCodeConstants.ERROR_CODE_990000001,
                        attributes),
                exception);
    }

    /**
     * 直後の検針結果を遡って取得すべき事由かを判定する。
     *
     * @param reasonType 検針結果作成事由
     * @return 後続取得を行う場合は true
     */
    private boolean shouldInvokeFollowUp(String reasonType) {
        return reasonType != null
                && !openingReason.equals(reasonType)
                && !regularSwCancellationReason.equals(reasonType)
                && !temporarySwCancellationReason.equals(reasonType);
    }

    /**
     * 開閉栓中止系の事由かを判定する。
     *
     * @param reasonType 検針結果作成事由
     * @return 開閉栓中止事由の場合は true
     */
    private boolean isSwCancellation(String reasonType) {
        return regularSwCancellationReason.equals(reasonType)
                || temporarySwCancellationReason.equals(reasonType);
    }

    /**
     * 初回検針申込コードを付与すべき事由に応じてコードを解決する。
     *
     * @param reasonType 検針結果作成事由
     * @return 初回検針申込コード、不要な場合は null
     */
    private String resolveInitialMeterReadingCode(String reasonType) {
        if (openingReason.equals(reasonType) || supplyResumedReason.equals(reasonType)) {
            return initialMeterReadingRequiredCode;
        }
        return null;
    }

    /**
     * 前回検針結果から次回呼び出し用の検針結果訂正前情報を生成する。
     *
     * @param znkiKnkkInf 前回検針結果情報
     * @return 検針結果訂正前情報
     */
    private KnsKkaTseiMaeJh toKnsKkaTseiMaeJh(ZnkiKnkkInf znkiKnkkInf) {
        return new KnsKkaTseiMaeJh(
                znkiKnkkInf.knsYmd(),
                znkiKnkkInf.knsKkaTsno(),
                znkiKnkkInf.knkkKtmeFlg());
    }

    /**
     * 直近1回目の検針結果情報を組み立てる。
     *
     * @param knsYmd 依頼時点の検針日
     * @param first 1回目の取得結果
     * @return 直近1回目の検針結果情報
     */
    private KnsnInf createFirstKnsnInf(LocalDate knsYmd, MeterReadingResultState first) {
        return new KnsnInf(
                knsYmd,
                null,
                DEFAULT_GAS_METER_INDICATION,
                normalizeGasUsage(first.kyuMtrInf().jkarSuKnzmGasSyry()),
                first.gasRknKsnInf().shyDdsu(),
                resolveInitialMeterReadingCode(first.znkiKnkkInf().knkkSsjyShbt()));
    }

    /**
     * 直近2回目の検針結果情報を組み立てる。
     *
     * @param first 1回目の取得結果
     * @param second 2回目の取得結果
     * @return 直近2回目の検針結果情報
     */
    private KnsnInf createSecondKnsnInf(
            MeterReadingResultState first, MeterReadingResultState second) {
        return new KnsnInf(
                first.znkiKnkkInf().knsYmd(),
                first.znkiKnkkInf().knsJsjkCd(),
                first.skinKnkkInf().gmtShjSu(),
                normalizeGasUsage(second.gasRknKsnInf().jkarSuKnzmGasSyry()),
                second.gasRknKsnInf().shyDdsu(),
                null);
    }

    /**
     * 直近3回目の検針結果情報を組み立てる。
     *
     * @param second 2回目の取得結果
     * @param third 3回目の取得結果
     * @return 直近3回目の検針結果情報
     */
    private KnsnInf createThirdKnsnInf(
            MeterReadingResultState second, MeterReadingResultState third) {
        return new KnsnInf(
                second.znkiKnkkInf().knsYmd(),
                second.znkiKnkkInf().knsJsjkCd(),
                second.skinKnkkInf().gmtShjSu(),
                normalizeGasUsage(third.gasRknKsnInf().jkarSuKnzmGasSyry()),
                third.gasRknKsnInf().shyDdsu(),
                null);
    }

    /**
     * ガス使用量を null 非許容の BigDecimal に正規化する。
     *
     * @param value 変換元のガス使用量
     * @return 変換後の値。入力が null の場合は BigDecimal.ZERO
     */
    private BigDecimal normalizeGasUsage(BigDecimal value) {
        if (value == null) {
            return BigDecimal.ZERO;
        }
        if (BigDecimal.ZERO.compareTo(value) == 0) {
            return BigDecimal.ZERO;
        }
        return value;
    }

    /**
     * 前回検針結果情報の既定値を生成する。
     *
     * <p>検針回数とガスメーター指示数のみ 0 系既定値を設定し、日付や事由など未取得項目は null のまま保持する。
     *
     * @return 0/null 方針で初期化した前回検針結果情報
     */
    private static ZnkiKnkkInf defaultZnkiKnkkInf() {
        return new ZnkiKnkkInf(
                null,
                DEFAULT_READING_SEQUENCE,
                DEFAULT_GAS_METER_INDICATION,
                null,
                null,
                null,
                null,
                null);
    }

    private static SkinKnkkInf defaultSkinKnkkInf() {
        return new SkinKnkkInf(

                null,

                null,
                null,
                null);
    }

    /**
     * ガス料金計算結果の既定値を生成する。
     *
     * <p>ガス使用量と使用日数は 0 を設定し、他の未取得項目は null とする。
     *
     * @return 0/null 方針で初期化したガス料金計算結果

    private static GasRknKsnInf defaultGasRknKsnInf() {
        return new GasRknKsnInf(
                null,
                null,

                null);

    /**
     * 休止メーター情報の既定値を生成する。
     *
     * <p>ガス使用量のみ 0 を設定し、他の未取得項目は null とする。
     *
     * @return 0/null 方針で初期化した休止メーター情報
     */
    private static KyuMtrInf defaultKyuMtrInf() {
        return new KyuMtrInf(null, DEFAULT_GAS_USAGE, null, null, null);
    }

    /**
     * 検針結果組み立て中の中間状態を保持する。
     *
     * <p>前回検針結果・今回検針結果・ガス料金計算結果・休止メーター情報をひとまとめに扱い、null 補完や SW 解約時の補正結果もこの record に集約する。
     */
    private record MeterReadingResultState(
            ZnkiKnkkInf znkiKnkkInf,
            SkinKnkkInf skinKnkkInf,
            GasRknKsnInf gasRknKsnInf,
            KyuMtrInf kyuMtrInf) {

        /**
         * 取得失敗時に返す既定状態を生成する。
         *
         * <p>前回検針結果・今回検針結果・ガス料金計算結果・休止メーター情報をそれぞれ 0/null 方針の helper で初期化する。
         *
         * @return 各 subDTO を既定値で保持した状態
         */
        private static MeterReadingResultState defaultState() {
            return new MeterReadingResultState(
                    defaultZnkiKnkkInf(),
                    defaultSkinKnkkInf(),
                    defaultGasRknKsnInf(),
                    defaultKyuMtrInf());
        }

        /**
         * サービス出力を状態オブジェクトへ変換する。
         *
         * <p>出力全体が null の場合は既定状態を返し、各 subDTO が null の場合も対応する helper の既定値へフォールバックする。
         *
         * @param output ガス使用量取得サービスの出力
         * @return null を既定値へ補完した状態
         */
        private static MeterReadingResultState from(GasUsageForBillingDerivationOutput output) {
            if (output == null) {
                return defaultState();
            }
            return new MeterReadingResultState(
                    output.znkiKnkkInf() == null ? defaultZnkiKnkkInf() : output.znkiKnkkInf(),
                    output.skinKnkkInf() == null ? defaultSkinKnkkInf() : output.skinKnkkInf(),
                    output.gasRknKsnInf() == null ? defaultGasRknKsnInf() : output.gasRknKsnInf(),
                    output.kyuMtrInf() == null ? defaultKyuMtrInf() : output.kyuMtrInf());
        }

        /**
         * SW 解約系事由向けにガス情報を 0 へ補正した状態を生成する。
         *
         * <p>ガス料金計算結果のガス使用量と使用日数のみ 0 に置換し、それ以外の項目と前回検針結果・今回検針結果・休止メーター情報は元の値を保持する。
         *
         * @return ガス情報のみ 0 に補正した状態
         */
        private MeterReadingResultState withZeroGasInfo() {
            return new MeterReadingResultState(
                    znkiKnkkInf,
                    skinKnkkInf,
                    new GasRknKsnInf(
                            gasRknKsnInf.kokiRkksYmd(),
                            gasRknKsnInf.jkarSuKnzgGasSyry(),
                            DEFAULT_GAS_USAGE,
                            gasRknKsnInf.tsmJkarSuKnzgGasSyry(),
                            DEFAULT_USAGE_DAYS,
                            gasRknKsnInf.kkyCushDdsu(),
                            gasRknKsnInf.tksuSvkyNo(),
                            gasRknKsnInf.sghsFkumFlg()),
                    kyuMtrInf);
        }
    }
