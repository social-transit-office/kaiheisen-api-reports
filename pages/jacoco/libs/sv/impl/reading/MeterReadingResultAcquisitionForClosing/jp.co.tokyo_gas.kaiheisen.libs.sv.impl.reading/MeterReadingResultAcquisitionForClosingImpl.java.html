package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.ClosingReportDeadlineAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.ClosingReportDeadlineAcquisitionOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.SkinKnkkInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForClosing;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForClosingOutput;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisitionForClosing;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisitionForClosingOutput;

/**
 * 概要: 検針結果取得（閉栓）サービス実装。
 * 主な仕様:
 * - SV-069 と SV-068 を順番に呼び出し、閉栓用の検針結果情報を組み立てて返却する。
 * - SV 規約に従い、報告システム種別、相関ID、システム日時は下位 SV 側で RequestContext から扱う。
 * 既知の制約:
 * - 本クラスは入力妥当性検証を追加せず、下位 SV の例外も変換せずに上位へ伝播する。
 */
public class MeterReadingResultAcquisitionForClosingImpl
        implements MeterReadingResultAcquisitionForClosing {

    private final RecentMeterReadingInfoAcquisitionForClosing recentMeterReadingInfoAcquisitionForClosing;

    private final ClosingReportDeadlineAcquisition closingReportDeadlineAcquisition;

    /**
     * 概要: 依存 SV を受け取り、実装を初期化するコンストラクタ。
     * 主な仕様:
     * - 直近検針情報取得（閉栓）と閉栓報告可能期限日取得の 2 サービスを保持する。
     * - DI 構成不備は fail-fast で検知する。
     * 既知の制約:
     * - public メソッド引数の妥当性検証は本コンストラクタでは扱わない。
     *
     * @param recentMeterReadingInfoAcquisitionForClosing SV-069
     * @param closingReportDeadlineAcquisition SV-068
     */
    @Inject
    public MeterReadingResultAcquisitionForClosingImpl(
            RecentMeterReadingInfoAcquisitionForClosing recentMeterReadingInfoAcquisitionForClosing,
            ClosingReportDeadlineAcquisition closingReportDeadlineAcquisition) {
        this.recentMeterReadingInfoAcquisitionForClosing =
                Objects.requireNonNull(
                        recentMeterReadingInfoAcquisitionForClosing,
                        "recentMeterReadingInfoAcquisitionForClosing");
        this.closingReportDeadlineAcquisition =
                Objects.requireNonNull(
                        closingReportDeadlineAcquisition,
                        "closingReportDeadlineAcquisition");
    }

    /**
     * 概要: SV-162 を実行し、閉栓用の検針結果情報を返却する。
     * 主な仕様:
     * - SV-069 の最近検針結果情報から SV-068 の入力値を組み立てる。
     * - SV-069 の戻り値は閉栓向け DTO へ移送し、SV-068 の戻り値を期限日として補完する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 閉栓用の検針結果情報
     */
    @Override
    public MeterReadingResultAcquisitionForClosingOutput execute(Long gasMtstNo) {
        RecentMeterReadingInfoAcquisitionForClosingOutput recentOutput =
                requireRecentOutput(recentMeterReadingInfoAcquisitionForClosing.execute(gasMtstNo));
        SkinKnkkInf recentMeterReading = requireRecentMeterReading(recentOutput);
        ClosingReportDeadlineAcquisitionOutput deadlineOutput = requireDeadlineOutput(
                closingReportDeadlineAcquisition.execute(
                        gasMtstNo,
                        requireRecentMeterReadingDate(recentMeterReading),
                        requireRecentMeterReadingReason(recentMeterReading)));
        return new MeterReadingResultAcquisitionForClosingOutput(
                mapZnkiKnkkInf(recentOutput.znkiKnkkInf()),
                mapSkinKnkkInf(recentMeterReading),
                resolveDeadline(deadlineOutput));
    }

    /**
     * 概要: SV-069 の戻り値 DTO を検証する。
     * 主な仕様:
     * - 想定外に null が返却された場合は契約違反として fail-fast する。
     *
     * @param recentOutput SV-069 の戻り値
     * @return 非 null の戻り値 DTO
     */
    private RecentMeterReadingInfoAcquisitionForClosingOutput requireRecentOutput(
            RecentMeterReadingInfoAcquisitionForClosingOutput recentOutput) {
        if (recentOutput == null) {
            throw new IllegalStateException("SV-069 returned null output");
        }
        return recentOutput;
    }

    /**
     * 概要: SV-069 の最近検針結果情報を検証する。
     * 主な仕様:
     * - SV-068 呼出しに必要な最近検針結果情報が欠けている場合は契約違反として fail-fast する。
     *
     * @param recentOutput SV-069 の戻り値
     * @return 非 null の最近検針結果情報
     */
    private SkinKnkkInf requireRecentMeterReading(
            RecentMeterReadingInfoAcquisitionForClosingOutput recentOutput) {
        if (recentOutput.skinKnkkInf() == null) {
            throw new IllegalStateException("SV-069 returned null skinKnkkInf");
        }
        return recentOutput.skinKnkkInf();
    }

    /**
     * 概要: SV-068 呼出しに必要な検針年月日を検証する。
     * 主な仕様:
     * - 最近検針結果情報の検針年月日が欠けている場合は契約違反として fail-fast する。
     *
     * @param recentMeterReading 最近検針結果情報
     * @return 非 null の検針年月日
     */
    private LocalDate requireRecentMeterReadingDate(SkinKnkkInf recentMeterReading) {
        if (recentMeterReading.knsYmd() == null) {
            throw new IllegalStateException("SV-069 returned null skinKnkkInf.knsYmd");
        }
        return recentMeterReading.knsYmd();
    }

    /**
     * 概要: SV-068 呼出しに必要な検針結果作成事由種別を検証する。
     * 主な仕様:
     * - 最近検針結果情報の検針結果作成事由種別が欠けている場合は契約違反として fail-fast する。
     *
     * @param recentMeterReading 最近検針結果情報
     * @return 非 null の検針結果作成事由種別
     */
    private String requireRecentMeterReadingReason(SkinKnkkInf recentMeterReading) {
        if (recentMeterReading.knkkSsjyShbt() == null) {
            throw new IllegalStateException("SV-069 returned null skinKnkkInf.knkkSsjyShbt");
        }
        return recentMeterReading.knkkSsjyShbt();
    }

    /**
     * 概要: SV-068 の戻り値 DTO を検証する。
     * 主な仕様:
     * - 想定外に null が返却された場合は契約違反として fail-fast する。
     *
     * @param deadlineOutput SV-068 の戻り値
     * @return 非 null の戻り値 DTO
     */
    private ClosingReportDeadlineAcquisitionOutput requireDeadlineOutput(
            ClosingReportDeadlineAcquisitionOutput deadlineOutput) {
        if (deadlineOutput == null) {
            throw new IllegalStateException("SV-068 returned null output");
        }
        return deadlineOutput;
    }

    /**
     * 概要: 前回検針結果情報を閉栓向け DTO へ変換する。
     * 主な仕様:
     * - 子 SV の戻り値を閉栓向け API モデルへ項目移送する。
     * - 入力が null の場合は null を返却する。
     *
     * @param childValue 子 SV の前回検針結果情報
     * @return 閉栓向け前回検針結果情報
     */
    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.ZnkiKnkkInf mapZnkiKnkkInf(
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.ZnkiKnkkInf childValue) {
        if (childValue == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.ZnkiKnkkInf(
                childValue.knsYmd(),
                childValue.knsKkaTsno(),
                childValue.gmtShjSu(),
                childValue.knkkSsjyShbt(),
                childValue.knkkKtmeFlg(),
                childValue.sjskKbn(),
                childValue.knsJsjkCd());
    }

    /**
     * 概要: 最近検針結果情報を閉栓向け DTO へ変換する。
     * 主な仕様:
     * - 子 SV の戻り値を閉栓向け API モデルへ項目移送する。
     * - 入力が null の場合は null を返却する。
     *
     * @param childValue 子 SV の最近検針結果情報
     * @return 閉栓向け最近検針結果情報
     */
    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf mapSkinKnkkInf(
            SkinKnkkInf childValue) {
        if (childValue == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SkinKnkkInf(
                childValue.knsYmd(),
                childValue.knsKkaTsno(),
                childValue.gmtShjSu(),
                childValue.knkkSsjyShbt(),
                childValue.knkkKtmeFlg(),
                childValue.sjskKbn(),
                childValue.knsJsjkCd());
    }

    /**
     * 概要: 閉栓報告可能期限日を解決する。
     * 主な仕様:
      * - 事前検証済みの子 SV 戻り値 DTO から期限日を返却する。
     *
     * @param deadlineOutput 子 SV の戻り値
     * @return 閉栓報告可能期限日
     */
    private LocalDate resolveDeadline(ClosingReportDeadlineAcquisitionOutput deadlineOutput) {
        return deadlineOutput.hsenHkkKnuKigenYmd();
    }
}
