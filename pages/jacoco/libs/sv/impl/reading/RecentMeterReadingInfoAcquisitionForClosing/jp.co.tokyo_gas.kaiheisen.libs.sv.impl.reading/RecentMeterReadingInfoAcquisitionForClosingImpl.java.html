package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.temporal.TemporalAdjusters;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetk.GmtsetkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetk.GmtsetkRecord;

import jp.co.tokyo_gas.kaiheisen.libs.sv.api.get.model.KnsKkaTseiMaeJh;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisitionForClosing;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.RecentMeterReadingInfoAcquisitionForClosingOutput;

/** 直近検針情報取得（閉栓）サービス実装。 */
public class RecentMeterReadingInfoAcquisitionForClosingImpl
        implements RecentMeterReadingInfoAcquisitionForClosing {

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final GasmtrkDataAccess gasmtrkDataAccess;

    private final GasUsageForBillingDerivation gasUsageForBillingDerivation;

    private final RequestContext requestContext;

    private final String relocationClosureReason;

    private final String noRegressionSegment;

    /**
     * コンストラクタ。
     *
     * @param gmtsetkDataAccess ガスメーター設置場所＜小売＞データアクセス
     * @param gasmtrkDataAccess ガスメーター＜小売＞データアクセス
     * @param gasUsageForBillingDerivation TG導管ガス使用量（料金計算用）取得サービス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public RecentMeterReadingInfoAcquisitionForClosingImpl(
            GmtsetkDataAccess gmtsetkDataAccess,
            GasmtrkDataAccess gasmtrkDataAccess,
            GasUsageForBillingDerivation gasUsageForBillingDerivation,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.gasmtrkDataAccess = Objects.requireNonNull(gasmtrkDataAccess, "gasmtrkDataAccess");
        this.gasUsageForBillingDerivation =
                Objects.requireNonNull(gasUsageForBillingDerivation, "gasUsageForBillingDerivation");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.relocationClosureReason =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.RELOCATION_CLOSURE);
        this.noRegressionSegment =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                        ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.NONE);
    }

    /**
     * 閉栓向けの直近検針情報を取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return 閉栓向け直近検針情報
     */
    @Override
    public RecentMeterReadingInfoAcquisitionForClosingOutput execute(Long gasMtstNo) {
        LocalDateTime systemDatetime =
                requestContext
                        .getSystemDatetime()
                        .orElseThrow(
                                () ->
                                        new IllegalStateException(
                                                "RequestContext.systemDatetime is required"));

        LocalDate monthEndDate = systemDatetime.toLocalDate().with(TemporalAdjusters.lastDayOfMonth());

        GmtsetkRecord gmtsetkRecord =
                gmtsetkDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemDatetime.toLocalDate())
                        .orElseThrow(() -> dataInconsistencyForGmtsetk(gasMtstNo, systemDatetime.toLocalDate()));

        GasmtrkRecord gasmtrkRecord = requireGasmtrk(gasMtstNo);

        GasUsageForBillingDerivationOutput output =
                gasUsageForBillingDerivation.execute(
                        gasMtstNo,
                        monthEndDate,
                        gasmtrkRecord.sjskHticSu() == null ? null : gasmtrkRecord.sjskHticSu() - 1L,
                        relocationClosureReason,
                        noRegressionSegment,
                        gmtsetkRecord.gmtKhsJtkb(),
                        0,
                        gasmtrkRecord.sjsuMincSu(),
                        gasmtrkRecord.sjskHticSu(),
                        BigDecimal.ZERO,
                        new KnsKkaTseiMaeJh(null, null, Boolean.FALSE));

        return new RecentMeterReadingInfoAcquisitionForClosingOutput(
                output.znkiKnkkInf(), output.skinKnkkInf());
    }

    /**
     * ガスメーター＜小売＞を必須で取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return ガスメーター＜小売＞
     */
    private GasmtrkRecord requireGasmtrk(Long gasMtstNo) {
        List<GasmtrkRecord> gasmtrkRecords =
                gasmtrkDataAccess.findAllByTtkGmsbNoAndGmtThkgYmd(gasMtstNo, null);
        if (!gasmtrkRecords.isEmpty()) {
            return gasmtrkRecords.getFirst();
        }
        throw dataInconsistencyForGasmtrk(gasMtstNo);
    }

    /**
     * GMTSETK 0件時のデータ不整合例外を生成する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準日
     * @return データ不整合例外
     */
    private DataInconsistencyException dataInconsistencyForGmtsetk(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        return DataInconsistencySupport.create(
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", knskKjnYmd));
    }

    /**
     * GASMTRK 0件時のデータ不整合例外を生成する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @return データ不整合例外
     */
    private DataInconsistencyException dataInconsistencyForGasmtrk(Long gasMtstNo) {
        return DataInconsistencySupport.create(
                gasmtrkDataAccess.tableName(),
                GasmtrkDataAccess.CallSpecs.FIND_ALL_BY_TTK_GMSB_NO_AND_GMT_THKG_YMD
                        .identifier(),
                new FlatJsonBuilder()
                        .put("ttkGmsbNo", gasMtstNo)
                        .put("gmtThkgYmd", null));
    }
}
