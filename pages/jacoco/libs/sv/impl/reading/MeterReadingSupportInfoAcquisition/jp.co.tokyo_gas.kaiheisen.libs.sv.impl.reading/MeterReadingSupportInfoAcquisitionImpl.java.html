package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingSupportInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingSupportInfoAcquisitionOutput;

/** 検針支援情報取得サービス実装。 */
public class MeterReadingSupportInfoAcquisitionImpl implements MeterReadingSupportInfoAcquisition {

    private final KenssieDataAccess kenssieDataAccess;

    private final String meterInstallationCode;

    private final String readingOnlyCode;

    private final String readingContactCode;

    private final List<String> targetCodes;

    /**
     * コンストラクタ。
     *
     * @param kenssieDataAccess 検針支援 データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public MeterReadingSupportInfoAcquisitionImpl(
            KenssieDataAccess kenssieDataAccess, CiriusConstantService ciriusConstantService) {
        this.kenssieDataAccess = Objects.requireNonNull(kenssieDataAccess, "kenssieDataAccess");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.meterInstallationCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.MeterReadingSupport.TYPE,
                        ConstantLookupKeys.Cirius.MeterReadingSupport
                                .SPECIAL_GAS_METER_MOUNTING_POSITION);
        this.readingOnlyCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.MeterReadingSupport.TYPE,
                        ConstantLookupKeys.Cirius.MeterReadingSupport
                                .INFORMATION_EXCLUSIVELY_FOR_METER_READING_OPERATIONS);
        this.readingContactCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.MeterReadingSupport.TYPE,
                        ConstantLookupKeys.Cirius.MeterReadingSupport
                                .METER_READING_BUSINESS_CONTACT_INFORMATION);
        this.targetCodes = List.of(meterInstallationCode, readingOnlyCode, readingContactCode);
    }

    @Override
    public MeterReadingSupportInfoAcquisitionOutput execute(Long gasMtstNo) {
        List<KenssieRecord> records =
                kenssieDataAccess.findAllByGmtSetNoAndKnsnSienBrCd(gasMtstNo, targetCodes);
        return toOutput(records);
    }

    /**
     * 概要: KENSSIE レコード一覧から検針支援情報の出力DTOを組み立てる。 主な仕様: -
     * 各支援分類コード（メーター設置位置/検針業務専用情報/業務連絡先）ごとに先頭一致レコードを採用する。 -
     * 業務連絡先情報が存在する場合は第2項目へ業務連絡先、第3項目へ検針業務専用情報を設定する。 - 業務連絡先情報が存在しない場合は第2項目へ検針業務専用情報を設定し、第3項目は null
     * とする。 既知の制約: - 引数 records は null を許容しない。 - レコード要素が null、または分類コードが null/空白のみの場合は当該要素を無視する。
     *
     * @param records KENSSIE の取得結果一覧
     * @return 検針支援情報取得の出力DTO
     */
    private MeterReadingSupportInfoAcquisitionOutput toOutput(List<KenssieRecord> records) {
        Objects.requireNonNull(records, "records");

        String gmtItiInf = null;
        String knsnSienInf2Candidate = null;
        String knsnSienInf1Candidate = null;

        for (KenssieRecord record : records) {
            if (record == null) {
                continue;
            }

            String code = normalize(record.knsnSienBrCd());
            if (code == null) {
                continue;
            }

            if (gmtItiInf == null && code.equals(meterInstallationCode)) {
                gmtItiInf = record.torokuNaiyoMx();
                continue;
            }
            if (knsnSienInf2Candidate == null && code.equals(readingOnlyCode)) {
                knsnSienInf2Candidate = record.torokuNaiyoMx();
                continue;
            }
            if (knsnSienInf1Candidate == null && code.equals(readingContactCode)) {
                knsnSienInf1Candidate = record.torokuNaiyoMx();
            }
        }

        if (knsnSienInf1Candidate != null) {
            return new MeterReadingSupportInfoAcquisitionOutput(
                    gmtItiInf, knsnSienInf1Candidate, knsnSienInf2Candidate);
        }

        return new MeterReadingSupportInfoAcquisitionOutput(gmtItiInf, knsnSienInf2Candidate, null);
    }

    /**
     * 概要: 文字列を trim して空値を null に正規化する。 主な仕様: - 入力値が null の場合は null を返却する。 - trim 後の値が空文字の場合は null
     * を返却する。 - それ以外の場合は trim 後の文字列を返却する。 既知の制約: - 先頭/末尾空白の除去は Java 標準の trim の仕様に従う。
     *
     * @param value 正規化対象の文字列
     * @return 正規化後の文字列（空値は null）
     */
    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }
}
