package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;

/** 検針結果情報取得（１年前）サービス実装。 */
public class OneYearAgoMeterReadingResultAcquisitionImpl
        implements OneYearAgoMeterReadingResultAcquisition {

    private final GmkskkdDataAccess gmkskkdDataAccess;

    private final GasmtrkDataAccess gasmtrkDataAccess;

    private final GasUsageForBillingDerivation gasUsageForBillingDerivation;

    private final RequestContext requestContext;

    private final String regularReasonType;

    private final String temporaryReasonType;

    private final String closedStatusCode;

    /**
     * コンストラクタ。
     *
     * @param gmkskkdDataAccess ガスメーター検針結果データアクセス
     * @param gasmtrkDataAccess ガスメーター小売データアクセス
     * @param gasUsageForBillingDerivation TG導管ガス使用量（料金計算用）取得サービス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public OneYearAgoMeterReadingResultAcquisitionImpl(
            GmkskkdDataAccess gmkskkdDataAccess,
            GasmtrkDataAccess gasmtrkDataAccess,
            GasUsageForBillingDerivation gasUsageForBillingDerivation,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.gmkskkdDataAccess = Objects.requireNonNull(gmkskkdDataAccess, "gmkskkdDataAccess");
        this.gasmtrkDataAccess = Objects.requireNonNull(gasmtrkDataAccess, "gasmtrkDataAccess");
        this.gasUsageForBillingDerivation =
                Objects.requireNonNull(gasUsageForBillingDerivation, "gasUsageForBillingDerivation");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.regularReasonType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.REGULAR);
        this.temporaryReasonType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TEMPORARY);
        this.closedStatusCode =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
    }

    /**
     * 概要: 指定された条件で1年前同回の検針結果情報を取得する。
     * 主な仕様:
     * - 検針年月日が未指定の場合は RequestContext のシステム日時を基準日に使用する。
     * - 基準日の前年同月の月初と月末を算出し、GMKSKKD を期間検索する。
     * - 対象レコードのうち、託送サービス契約番号一致かつ対象事由種別のものだけを処理し、
     *   GASMTRK 存在確認後に SV-312 を呼び出す。
     * - 条件に合致しない場合は初期値を返却する。
     * 既知の制約:
     * - GMKSKKD の走査は取得順に依存し、先頭レコードの託送サービス契約番号が不一致なら
     *   即座に初期値を返却する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knsYmd 検針年月日
     * @param tksuSvkyNo 託送サービス契約番号
     * @return 1年前同回の検針結果情報
     */
    @Override
    public OneYearAgoMeterReadingResultAcquisitionOutput execute(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long tksuSvkyNo) {
        LocalDate baseDate = resolveBaseDate(knsYmd);
        LocalDate previousYearDate = baseDate.minusYears(1);
        LocalDate knsYmdFrom = previousYearDate.withDayOfMonth(1);
        LocalDate knsYmdTo = previousYearDate.withDayOfMonth(previousYearDate.lengthOfMonth());

        List<GmkskkdRecord> records =
                gmkskkdDataAccess.findAllByGasMtstNoAndKnsYmd(gasMtstNo, knsYmdFrom, knsYmdTo);

        for (GmkskkdRecord record : records) {
            if (!Objects.equals(record.tksuSvkyNo(), tksuSvkyNo)) {
                return defaultOutput();
            }
            if (!isTargetReasonType(record.knkkSksiJysb())) {
                continue;
            }

            requireGasmtrk(record, gasMtstNo);

            GasUsageForBillingDerivationOutput sv312Output =
                    gasUsageForBillingDerivation.execute(
                            gasMtstNo,
                            record.knsYmd(),
                            record.gmtShjSu(),
                            record.knkkSksiJysb(),
                            record.sjskKbn(),
                            closedStatusCode,
                            BigDecimal.ZERO,
                            new KnsKkaTseiMaeJh(
                                    record.knsYmd(),
                                    record.knsKkaTsno(),
                                    Boolean.FALSE));
            return toOutput(sv312Output, record);
        }

        return defaultOutput();
    }

    /**
     * 概要: 基準日を解決する。
        * 主な仕様: - 引数の検針年月日が指定されていればその値を返却する。 - 未指定時は RequestContext の systemDatetime を参照し、日付部分を返却する。
        * 既知の制約: - RequestContext に systemDatetime がない場合は IllegalStateException を送出する。
     *
     * @param knsYmd 検針年月日
     * @return 基準日
     */
    private LocalDate resolveBaseDate(LocalDate knsYmd) {
        if (knsYmd != null) {
            return knsYmd;
        }
        LocalDateTime systemDatetime =
                requestContext
                        .getSystemDatetime()
                        .orElseThrow(
                                () ->
                                        new IllegalStateException(
                                                "RequestContext.systemDatetime is required"));
        return systemDatetime.toLocalDate();
    }

    /**
     * 概要: 検針結果作成事由種別が SV-006 の対象か判定する。
     * 主な仕様: - 定例または臨時の事由種別コードのみを true とする。 - 対象外コードは false とする。
     * 既知の制約: - 判定に使う事由種別コードはコンストラクタで解決した定数値に依存する。
     *
     * @param reasonType 検針結果作成事由種別
     * @return 対象の場合 true
     */
    private boolean isTargetReasonType(String reasonType) {
        return regularReasonType.equals(reasonType) || temporaryReasonType.equals(reasonType);
    }

    /**
     * 概要: SV-312 呼出し前に GASMTRK の存在を確認する。
     * 主な仕様: - 指定した設置場所番号と検針年月日を基準に DA-026-R002 を検索する。 - 取得結果が1件以上なら何もせず処理を継続する。
     * - 0件の場合は DataInconsistencyException を送出する。
     * 既知の制約: - 返却件数の妥当性は1件以上かどうかだけを確認し、複数件時の内容精査は行わない。
     *
     * @param record GMKSKKD の取得結果
     * @param gasMtstNo ガスメーター設置場所番号
     */
    private void requireGasmtrk(GmkskkdRecord record, Long gasMtstNo) {
        List<GasmtrkRecord> gasmtrkRecords =
                gasmtrkDataAccess.findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(gasMtstNo, record.knsYmd());
        if (!gasmtrkRecords.isEmpty()) {
            return;
        }
        throw dataInconsistency(
                GasmtrkDataAccess.CallSpecs.FIND_ALL_BY_TTK_GMSB_NO_AND_GMTK_YMD_AND_GMT_THKG_YMD,
                gasMtstNo,
                record.knsYmd());
    }

    /**
     * 概要: SV-312 と GMKSKKD の結果から SV-006 の出力DTOを構築する。
     * 主な仕様: - 使用日数は SV-312 の戻り値をそのまま返す。 - ガス使用量は SV-312 の実供給圧力数換算前ガス使用量を BigDecimal へ変換して返す。
     * - 検針実施状況コードは GMKSKKD の値を返す。
     * 既知の制約: - SV-312 の戻り値が null の場合は使用日数とガス使用量に null を設定する。
     *
     * @param sv312Output SV-312 の戻り値
     * @param record GMKSKKD の取得結果
     * @return SV-006 の出力DTO
     */
    private OneYearAgoMeterReadingResultAcquisitionOutput toOutput(
            GasUsageForBillingDerivationOutput sv312Output,
            GmkskkdRecord record) {
        GasRknKsnInf gasRknKsnInf = sv312Output == null ? null : sv312Output.gasRknKsnInf();
        return new OneYearAgoMeterReadingResultAcquisitionOutput(
                gasRknKsnInf == null ? null : gasRknKsnInf.shyDdsu(),
                gasRknKsnInf == null || gasRknKsnInf.jkarSuKnzmGasSyry() == null
                        ? null
                        : gasRknKsnInf.jkarSuKnzmGasSyry(),
                record.knsJsjkCd());
    }

    /**
     * 概要: GASMTRK の返却件数不整合を表す DataInconsistencyException を生成する。
     * 主な仕様: - DA の CallSpec から識別子を取得する。 - テーブル名は GasmtrkDataAccess から、キー情報は JSON 文字列へ詰めて例外に保持する。
     * 既知の制約: - 受け取った引数値の妥当性検証は行わず、そのまま例外情報へ埋め込む。
     *
     * @param callSpec 不整合判定元の DA CallSpec
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準日
     * @return DataInconsistencyException
     */
    private DataInconsistencyException dataInconsistency(
            DataAccessCallSpec callSpec, Long gasMtstNo, LocalDate knskKjnYmd) {
        return DataInconsistencySupport.create(
                gasmtrkDataAccess.tableName(),
                callSpec.identifier(),
                new FlatJsonBuilder()
                        .put("ttkGmsbNo", gasMtstNo)
                        .put("knskKjnYmd", knskKjnYmd));
    }

    /**
     * 概要: 対象データなしを表す初期値を返却する。
     * 主な仕様: - 使用日数に 0、使用量に 0、検針実施状況コードに空文字を設定する。
     * 既知の制約: - 初期値の内容は SV-006 の業務仕様に固定されている。
     *
     * @return 初期値の出力DTO
     */
    private static OneYearAgoMeterReadingResultAcquisitionOutput defaultOutput() {
        return new OneYearAgoMeterReadingResultAcquisitionOutput(0, BigDecimal.ZERO, "");
    }
}
