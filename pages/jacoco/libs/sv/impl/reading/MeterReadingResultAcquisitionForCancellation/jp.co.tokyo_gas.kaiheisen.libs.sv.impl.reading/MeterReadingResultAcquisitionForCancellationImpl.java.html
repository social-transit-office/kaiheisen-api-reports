package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dktksry.DktksryDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dktksry.DktksryRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.KkytskrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForCancellation;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.MeterReadingResultAcquisitionForCancellationOutput;

/** 検針結果取得（解約）サービス実装。 */
public class MeterReadingResultAcquisitionForCancellationImpl
        implements MeterReadingResultAcquisitionForCancellation {

    private final DktksryDataAccess dktksryDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final RequestContext requestContext;

    private final String supplyOutageReasonType;

    private final String closedStatus;

    private final String supplyStopPaymentPending;

    /**
     * コンストラクタ。
     *
     * @param dktksryDataAccess 導管通知確定使用量データアクセス
     * @param kkytskrDataAccess 供給停止管理データアクセス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public MeterReadingResultAcquisitionForCancellationImpl(
            DktksryDataAccess dktksryDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService) {
        this.dktksryDataAccess = Objects.requireNonNull(dktksryDataAccess, "dktksryDataAccess");
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.supplyOutageReasonType =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                        ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults
                                .SUPPLY_OUTAGE);
        this.closedStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
        this.supplyStopPaymentPending =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                        ConstantLookupKeys.Cirius.SupplyStopPaymentStatus
                                .SUPPLY_STOP_PAYMENT_PENDING);
    }

    @Override
    public MeterReadingResultAcquisitionForCancellationOutput execute(Long gasMtstNo) {
        LocalDateTime systemDatetime =
                requestContext
                        .getSystemDatetime()
                        .orElseThrow(
                                () ->
                                        new IllegalStateException(
                                                "RequestContext.systemDatetime is required"));
        LocalDate systemDate = systemDatetime.toLocalDate();
        LocalDate knsYmdFrom = systemDate.minusMonths(17).withDayOfMonth(1);

        Optional<DktksryRecord> dktksryRecord =
                dktksryDataAccess.findByGasMtstNoAndKnsYmdRange(gasMtstNo, knsYmdFrom, systemDate);
        Optional<KkytskrRecord> kkytskrRecord =
                kkytskrDataAccess.findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
                        gasMtstNo, null, supplyStopPaymentPending);

        if (dktksryRecord.isPresent() && kkytskrRecord.isEmpty()) {
            return fromDktksry(dktksryRecord.orElseThrow());
        }
        if (dktksryRecord.isEmpty() && kkytskrRecord.isPresent()) {
            return fromKkytskr(kkytskrRecord.orElseThrow());
        }
        if (dktksryRecord.isPresent() && kkytskrRecord.isPresent()) {
            DktksryRecord dktksry = dktksryRecord.orElseThrow();
            KkytskrRecord kkytskr = kkytskrRecord.orElseThrow();
            if (dktksry.knsYmd().isAfter(kkytskr.hesnYmd())) {
                return fromDktksry(dktksry);
            }
            return fromKkytskr(kkytskr);
        }
        throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_990000001);
    }

    private MeterReadingResultAcquisitionForCancellationOutput fromDktksry(DktksryRecord record) {
        return new MeterReadingResultAcquisitionForCancellationOutput(
                record.knsYmd(),
                record.gmtShjSu(),
                record.knkkSksiJysb(),
                record.gmtKhsJtkb());
    }

    private MeterReadingResultAcquisitionForCancellationOutput fromKkytskr(KkytskrRecord record) {
        return new MeterReadingResultAcquisitionForCancellationOutput(
                record.hesnYmd(),
                record.gmtShjSu(),
                supplyOutageReasonType,
                closedStatus);
    }
}
