package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.reading;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;

/**
 * 解約用みなし検針登録サービス実装。
 */
public class DeemedMeterReadingRegistrationForCancellationImpl
        implements DeemedMeterReadingRegistrationForCancellation {

    private static final String DEEMED_METER_READING_REQUIRED = "1";

    private static final Integer DEEMED_METER_READING_SEQUENCE = 0;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final DeemedMeterReadingRegistration deemedMeterReadingRegistration;

    private final RequestContext requestContext;

    private final String dedicatedGasMeterStatus;

    private final String ghpEquipmentGasMeter;

    private final String closedGasMeterStatus;

    private final List<String> rkkiKnrnJtkbList;

    /**
     * コンストラクタ。
     *
     * @param gmtsetkDataAccess ガスメーター設置場所＜小売＞データアクセス
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連データアクセス
     * @param deemedMeterReadingRegistration みなし検針登録サービス
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public DeemedMeterReadingRegistrationForCancellationImpl(
            final GmtsetkDataAccess gmtsetkDataAccess,
            final GmtsetdDataAccess gmtsetdDataAccess,
            final SvmstkrDataAccess svmstkrDataAccess,
            final DeemedMeterReadingRegistration deemedMeterReadingRegistration,
            final RequestContext requestContext,
            final CiriusConstantService ciriusConstantService) {
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.deemedMeterReadingRegistration =
                Objects.requireNonNull(deemedMeterReadingRegistration, "deemedMeterReadingRegistration");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.ghpEquipmentGasMeter =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.EquipmentGasMeter.TYPE,
                        ConstantLookupKeys.Cirius.EquipmentGasMeter.FOR_GHP_EQUIPMENT);
        this.rkkiKnrnJtkbList =
                List.of(
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID),
                        RequiredConstantResolver.requireCiriusConstant(
                                resolvedCiriusConstantService,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03));
        this.closedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                        ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
    }

    /**
        * ガスメーター設置場所番号に紐づく解約用みなし検針を登録する。
        * 直列専用メーターの場合は当該メーターに紐づく直列元契約を用いて登録し、
        * GHP 機器用メーターの場合は入力された総合他ガスメーター情報リストを走査して総合他ガスメーター分も登録する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param tkenYmdYoku 適用終了年月日翌日
        * @param sogoTmtrInfList GHP 機器用メーターの場合に走査して登録対象とする総合他ガスメーター情報リスト
     */
    @Override
        public void execute(
            final Long gasMtstNo,
            final LocalDate tkenYmdYoku,
            final List<SogoTmtrInf> sogoTmtrInfList) {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));

        GmtsetkRecord retailMeter = requireRetailMeter(gasMtstNo, systemDatetime.toLocalDate());

        if (isDedicatedMeter(retailMeter.ckrtGmscJtkb())) {
            executeDeemedMeterReading(gasMtstNo, tkenYmdYoku, requireSeriesOriginContract(retailMeter.mtgMtstNo()));
        }

        if (isGhpMeter(retailMeter.kkyuGmtKbn())) {
            for (SogoTmtrInf sogoTmtrInf : normalizeSogoTmtrInfList(sogoTmtrInfList)) {
                executeForComprehensiveMeter(sogoTmtrInf, tkenYmdYoku);
            }
        }
    }

    private void executeForComprehensiveMeter(
            final SogoTmtrInf sogoTmtrInf,
            final LocalDate tkenYmdYoku) {
        Long sogoGasMtstNo = requireSogoGasMtstNo(sogoTmtrInf);
        GmtsetdRecord pipelineMeter = requirePipelineMeter(sogoGasMtstNo, tkenYmdYoku);
        if (!isDedicatedMeter(pipelineMeter.ckrtGmscJtkb())) {
            return;
        }
        executeDeemedMeterReading(
                sogoGasMtstNo,
                tkenYmdYoku,
                requireSeriesOriginContract(pipelineMeter.mtgMtstNo()));
    }

    private void executeDeemedMeterReading(
            final Long gasMtstNo,
            final LocalDate tkenYmdYoku,
            final SvmstkrRecord svmstkrRecord) {
        deemedMeterReadingRegistration.execute(
                DEEMED_METER_READING_REQUIRED,
                tkenYmdYoku,
                gasMtstNo,
                DEEMED_METER_READING_SEQUENCE,
                svmstkrRecord.svkyNo(),
                closedGasMeterStatus);
    }

    private GmtsetkRecord requireRetailMeter(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        return gmtsetkDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> dataInconsistencyForGmtsetk(gasMtstNo, knskKjnYmd));
    }

    private GmtsetdRecord requirePipelineMeter(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        return gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> dataInconsistencyForGmtsetd(gasMtstNo, knskKjnYmd));
    }

    private SvmstkrRecord requireSeriesOriginContract(final Long mtstNo) {
        return svmstkrDataAccess
                                .findByMtstNoAndRkkiKnrnJtkb(mtstNo, rkkiKnrnJtkbList)
                .orElseThrow(() -> dataInconsistencyForSvmstkr(mtstNo));
    }

    private Long requireSogoGasMtstNo(final SogoTmtrInf sogoTmtrInf) {
        if (sogoTmtrInf == null || sogoTmtrInf.svmstkrInf() == null || sogoTmtrInf.svmstkrInf().mtstNo() == null) {
            throw new IllegalArgumentException("sogoTmtrInf.svmstkrInf.mtstNo is required");
        }
        return sogoTmtrInf.svmstkrInf().mtstNo();
    }

    private List<SogoTmtrInf> normalizeSogoTmtrInfList(final List<SogoTmtrInf> sogoTmtrInfList) {
        return sogoTmtrInfList == null ? List.of() : sogoTmtrInfList;
    }

    private boolean isDedicatedMeter(final String ckrtGmscJtkb) {
        return dedicatedGasMeterStatus.equals(ckrtGmscJtkb);
    }

    private boolean isGhpMeter(final String kkyuGmtKbn) {
        return ghpEquipmentGasMeter.equals(kkyuGmtKbn);
    }

    private DataInconsistencyException dataInconsistencyForGmtsetk(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        return DataInconsistencySupport.create(
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", knskKjnYmd));
    }

    private DataInconsistencyException dataInconsistencyForGmtsetd(
            final Long gasMtstNo,
            final LocalDate knskKjnYmd) {
        return DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", knskKjnYmd));
    }

    private DataInconsistencyException dataInconsistencyForSvmstkr(final Long mtstNo) {
        return DataInconsistencySupport.create(
                svmstkrDataAccess.tableName(),
                SvmstkrDataAccess.CallSpecs.FIND_BY_MTST_NO_AND_RKKI_KNRN_JTKB_LIST.identifier(),
                new FlatJsonBuilder()
                        .put("mtstNo", mtstNo)
                        .put("rkkiKnrnJtkbList", rkkiKnrnJtkbList));
    }
}
