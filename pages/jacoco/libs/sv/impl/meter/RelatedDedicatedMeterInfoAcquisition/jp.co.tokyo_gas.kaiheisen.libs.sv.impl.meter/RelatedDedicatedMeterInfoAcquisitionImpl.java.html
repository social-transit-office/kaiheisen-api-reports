package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.meter;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.meter.RelatedDedicatedMeterInfoAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.meter.RelatedDedicatedMeterInfoAcquisitionOutput;

/**
 * 概要: 関連する専用メーター情報取得サービス実装。
 * 主な仕様:
 * - 直列元ガスメーターまたは専用ガスメーターの場合のみ、関連する専用メーター情報を返却する。
 * - 専用ガスメーターの場合は元ガスメーターを先頭へ追加し、その後 DA 取得結果を最大10件まで返却する。
 * - 判定に利用する J0244 / 05800 のコード値は CIRIUS コンスタントとして起動時に検証する。
 * 既知の制約:
 * - ガスメーター社番が取得できない場合は戻り値の該当要素へ null を設定する。
 */
public class RelatedDedicatedMeterInfoAcquisitionImpl implements RelatedDedicatedMeterInfoAcquisition {

    private final String directOriginMeterStatus;

    private final String dedicatedMeterStatus;

    private final String meterPossessionStatus;

    private final GasmtrkDataAccess gasmtrkDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    /**
     * コンストラクタ。
     *
     * @param gasmtrkDataAccess ガスメーター小売データアクセス
     * @param gmtsetdDataAccess ガスメーター設置場所導管データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public RelatedDedicatedMeterInfoAcquisitionImpl(
            GasmtrkDataAccess gasmtrkDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.gasmtrkDataAccess = Objects.requireNonNull(gasmtrkDataAccess, "gasmtrkDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.directOriginMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.meterPossessionStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.TYPE,
                ConstantLookupKeys.Cirius.ServicePipeGasMeterEquipmentStatus.METER_POSSESSION_STATUS);
    }

    @Override
    public RelatedDedicatedMeterInfoAcquisitionOutput execute(
            Long gasMtstNo,
            Long mtgMtstNo,
            String ckrtGmscJtkb,
            LocalDate kskKjnYmd,
            String szcGkkCd,
            String szcCmazCd,
            String szcMchCd) {
        ArrayList<Long> gasMtstNoList = new ArrayList<>();
        ArrayList<String> gmbList = new ArrayList<>();

        if (!directOriginMeterStatus.equals(ckrtGmscJtkb) && !dedicatedMeterStatus.equals(ckrtGmscJtkb)) {
            return new RelatedDedicatedMeterInfoAcquisitionOutput(List.copyOf(gasMtstNoList), List.copyOf(gmbList));
        }

        int count = 0;
        Long zenGasMtstNo = null;
        Long originGasMtstNo = gasMtstNo;

        if (dedicatedMeterStatus.equals(ckrtGmscJtkb)) {
            appendResult(gasMtstNoList, gmbList, mtgMtstNo, resolveGmb(mtgMtstNo, kskKjnYmd));
            count = 1;
            zenGasMtstNo = mtgMtstNo;
            originGasMtstNo = mtgMtstNo;
        }

        List<GmtsetdRecord> candidates = gmtsetdDataAccess
                .findAllBySzcGkkCdAndSzcCmazCdAndSzcMchCdAndMtgMtstNoAndGasMtstNoAndTkysYmdAndTyenYmd(
                        szcGkkCd,
                        szcCmazCd,
                        szcMchCd,
                        originGasMtstNo,
                        gasMtstNo,
                        kskKjnYmd);

        for (GmtsetdRecord candidate : candidates) {
            if (count >= 10) {
                break;
            }
            Long candidateGasMtstNo = candidate.gasMtstNo();
            if (Objects.equals(zenGasMtstNo, candidateGasMtstNo)) {
                continue;
            }
            if (!meterPossessionStatus.equals(candidate.kngmStbJtkb())) {
                continue;
            }

            appendResult(gasMtstNoList, gmbList, candidateGasMtstNo, resolveGmb(candidateGasMtstNo, kskKjnYmd));
            count++;
            zenGasMtstNo = candidateGasMtstNo;
        }

        return new RelatedDedicatedMeterInfoAcquisitionOutput(List.copyOf(gasMtstNoList), List.copyOf(gmbList));
    }

    /**
     * ガスメーター社番を取得する。
     *
     * @param targetGasMtstNo ガスメーター設置場所番号
     * @param kskKjnYmd 検索基準年月日
     * @return ガスメーター社番
     * @throws DataInconsistencyException 検索結果が0件の場合
     */
    private String resolveGmb(Long targetGasMtstNo, LocalDate kskKjnYmd) {
        List<GasmtrkRecord> gasmtrkRecords =
                gasmtrkDataAccess.findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(targetGasMtstNo, kskKjnYmd);
        if (gasmtrkRecords.isEmpty()) {
            throw createMissingGasmtrkException(targetGasMtstNo, kskKjnYmd);
        }
        return gasmtrkRecords.getFirst().gmb();
    }

    private DataInconsistencyException createMissingGasmtrkException(Long targetGasMtstNo, LocalDate kskKjnYmd) {
        FlatJsonBuilder parameterJsonBuilder = new FlatJsonBuilder()
                .put("ttkGmsbNo", targetGasMtstNo)
                .put("knskKjnYmd", kskKjnYmd);
        return DataInconsistencySupport.create(
                gasmtrkDataAccess.tableName(),
                GasmtrkDataAccess.CallSpecs.FIND_ALL_BY_TTK_GMSB_NO_AND_GMTK_YMD_AND_GMT_THKG_YMD.identifier(),
                parameterJsonBuilder);
    }

    /**
     * 戻り値へ1件追加する。
     *
     * @param gasMtstNoList ガスメーター設置場所番号一覧
     * @param gmbList ガスメーター社番一覧
     * @param gasMtstNo ガスメーター設置場所番号
     * @param gmb ガスメーター社番
     */
    private static void appendResult(List<Long> gasMtstNoList, List<String> gmbList, Long gasMtstNo, String gmb) {
        gasMtstNoList.add(gasMtstNo);
        gmbList.add(gmb);
    }
}
