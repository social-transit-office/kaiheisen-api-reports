package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transferconfig;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer.EditCustomerResponse;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transferconfig.CustomerNameEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transferconfig.CustomerNameEditOutput;

/**
 * 概要: SV-133（おなまえ編集）サービス実装。
 * 主な仕様:
 * - 処理区分に応じて 他-003 のリクエスト項目を切り替える。
 * - 一括編集は名前１・２の漢字とカナを返し、単一名の互換呼び出しでは従来の漢字優先を維持する。
 * - 対象外の処理区分や編集パターン三が未設定の場合は未設定出力を返却する。
 */
public class CustomerNameEditImpl implements CustomerNameEdit {

    private final EditCustomerCiriusAdapter editCustomerCiriusAdapter;

    private final String nameEditCategoryOne;

    private final String nameEditCategoryTwo;

    private final String contractorNameEditCategory;

    /**
     * コンストラクタ。
     *
     * @param editCustomerCiriusAdapter 他-003 adapter
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public CustomerNameEditImpl(
            EditCustomerCiriusAdapter editCustomerCiriusAdapter,
            KaiheisenConstantService kaiheisenConstantService) {
        this.editCustomerCiriusAdapter =
                Objects.requireNonNull(editCustomerCiriusAdapter, "editCustomerCiriusAdapter");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.nameEditCategoryOne = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.EDIT_NAME_1);
        this.nameEditCategoryTwo = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.EDIT_NAME_2);
        this.contractorNameEditCategory = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.NameEditingProcessing.EDIT_CONTRACTOR_NAME);
    }

    @Override
    public CustomerNameEditOutput execute(
            String cmnCtnmKn1,
            String cmnCtnmKj1,
            String cmnCtnmHujnShbt1,
            String cmnCtnmHshiShbt1,
            String cmnCtnmKn2,
            String cmnCtnmKj2,
            String cmnCtnmHujnShbt2,
            String cmnCtnmHshiShbt2,
            Boolean ctnmKjhkFlg,
            String ctnmHnsSyoriKbn) {
        if (nameEditCategoryOne.equals(ctnmHnsSyoriKbn)
                || nameEditCategoryTwo.equals(ctnmHnsSyoriKbn)) {
            EditCustomerRequest request = new EditCustomerRequest(new EditCustomerRequest.BeforeCustomerInfo(
                    new EditCustomerRequest.BeforeNameInfo(
                            cmnCtnmKj1, cmnCtnmKn1, cmnCtnmHujnShbt1, cmnCtnmHshiShbt1,
                            cmnCtnmKj2, cmnCtnmKn2, cmnCtnmHujnShbt2, cmnCtnmHshiShbt2,
                            toFlag(ctnmKjhkFlg), null, null, null, null, null),
                    null));
            EditCustomerResponse response = editCustomerCiriusAdapter.editCustomer(request);
            EditCustomerResponse.Pattern3 pattern3 = pattern3OrNull(response);
            if (pattern3 == null) {
                return new CustomerNameEditOutput(null, null, null, null);
            }
            if (cmnCtnmKn2 == null || cmnCtnmKn2.isEmpty()) {
                return new CustomerNameEditOutput(
                        pattern3.ctnmKj2PHjsb(), pattern3.ctnmKn2PHjsb(), null, null);
            }
            return new CustomerNameEditOutput(
                    pattern3.ctnmKj1PHjsb(), pattern3.ctnmKn1PHjsb(),
                    pattern3.ctnmKj2PHjsb(), pattern3.ctnmKn2PHjsb());
        }
        if (contractorNameEditCategory.equals(ctnmHnsSyoriKbn)) {
            return executeForContractorName(
                    cmnCtnmKn1, cmnCtnmKj1, cmnCtnmHujnShbt1, cmnCtnmHshiShbt1, ctnmKjhkFlg);
        }
        return new CustomerNameEditOutput(null, null, null, null);
    }

    @Override
    public CustomerNameEditOutput execute(
            String cmnCtnmKn,
            String cmnCtnmKj,
            String cmnCtnmHujnShbt,
            String cmnCtnmHshiShbt,
            Boolean ctnmKjhkFlg,
            String ctnmHnsSyoriKbn) {
        if (nameEditCategoryOne.equals(ctnmHnsSyoriKbn)) {
            return executeForNameOneOrTwo(
                    cmnCtnmKn,
                    cmnCtnmKj,
                    cmnCtnmHujnShbt,
                    cmnCtnmHshiShbt,
                    ctnmKjhkFlg);
        }

        if (nameEditCategoryTwo.equals(ctnmHnsSyoriKbn)) {
            return executeForNameOneOrTwo(
                    cmnCtnmKn,
                    cmnCtnmKj,
                    cmnCtnmHujnShbt,
                    cmnCtnmHshiShbt,
                    ctnmKjhkFlg);
        }

        if (contractorNameEditCategory.equals(ctnmHnsSyoriKbn)) {
            return executeForContractorName(
                    cmnCtnmKn,
                    cmnCtnmKj,
                    cmnCtnmHujnShbt,
                    cmnCtnmHshiShbt,
                    ctnmKjhkFlg);
        }

        return new CustomerNameEditOutput(null, null);
    }

    /**
     * おなまえ1編集・おなまえ2編集を実行する。
     */
    private CustomerNameEditOutput executeForNameOneOrTwo(
            String cmnCtnmKn,
            String cmnCtnmKj,
            String cmnCtnmHujnShbt,
            String cmnCtnmHshiShbt,
            Boolean ctnmKjhkFlg) {
        EditCustomerResponse response = editCustomerCiriusAdapter.editCustomer(
                buildSecondNameRequest(
                        cmnCtnmKn,
                        cmnCtnmKj,
                        cmnCtnmHujnShbt,
                        cmnCtnmHshiShbt,
                        ctnmKjhkFlg));

        EditCustomerResponse.Pattern3 pattern3 = pattern3OrNull(response);

        return toOutput(
                pattern3 == null ? null : pattern3.ctnmKj2PHjsb(),
                pattern3 == null ? null : pattern3.ctnmKn2PHjsb());
    }

    /**
     * 契約者名編集を実行する。
     */
    private CustomerNameEditOutput executeForContractorName(
            String cmnCtnmKn,
            String cmnCtnmKj,
            String cmnCtnmHujnShbt,
            String cmnCtnmHshiShbt,
            Boolean ctnmKjhkFlg) {
        EditCustomerResponse response = editCustomerCiriusAdapter.editCustomer(
                buildContractorNameRequest(
                        cmnCtnmKn,
                        cmnCtnmKj,
                        cmnCtnmHujnShbt,
                        cmnCtnmHshiShbt,
                        ctnmKjhkFlg));

        EditCustomerResponse.Pattern3 pattern3 = pattern3OrNull(response);

        return toOutput(
                pattern3 == null ? null : pattern3.kynmKjPHjsb(),
                pattern3 == null ? null : pattern3.kynmKnPHjsb());
    }

    /**
     * 他-003 レスポンスから編集パターン三を取得する。
     */
    private EditCustomerResponse.Pattern3 pattern3OrNull(EditCustomerResponse response) {
        if (response == null || response.henshuugoOnamaeJouhou() == null) {
            return null;
        }

        return response.henshuugoOnamaeJouhou().henshuuPattern3();
    }

    /**
     * おなまえ二編集向けリクエストを組み立てる。
     */
    private EditCustomerRequest buildSecondNameRequest(
            String cmnCtnmKn,
            String cmnCtnmKj,
            String cmnCtnmHujnShbt,
            String cmnCtnmHshiShbt,
            Boolean ctnmKjhkFlg) {
        return new EditCustomerRequest(
                new EditCustomerRequest.BeforeCustomerInfo(
                        new EditCustomerRequest.BeforeNameInfo(
                                null,
                                null,
                                null,
                                null,
                                cmnCtnmKj,
                                cmnCtnmKn,
                                cmnCtnmHujnShbt,
                                cmnCtnmHshiShbt,
                                toFlag(ctnmKjhkFlg),
                                null,
                                null,
                                null,
                                null,
                                null),
                        null));
    }

    /**
     * 契約者名編集向けリクエストを組み立てる。
     */
    private EditCustomerRequest buildContractorNameRequest(
            String cmnCtnmKn,
            String cmnCtnmKj,
            String cmnCtnmHujnShbt,
            String cmnCtnmHshiShbt,
            Boolean ctnmKjhkFlg) {
        return new EditCustomerRequest(
                new EditCustomerRequest.BeforeCustomerInfo(
                        new EditCustomerRequest.BeforeNameInfo(
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                cmnCtnmKj,
                                cmnCtnmKn,
                                cmnCtnmHujnShbt,
                                cmnCtnmHshiShbt,
                                toFlag(ctnmKjhkFlg)),
                        null));
    }

    /**
     * 編集済み名称から返却 DTO を組み立てる。
     */
    private CustomerNameEditOutput toOutput(String editedKanji, String editedKana) {
        if (editedKanji == null || editedKanji.isEmpty()) {
            return new CustomerNameEditOutput(null, editedKana);
        }

        return new CustomerNameEditOutput(editedKanji, null);
    }

    /**
     * 真偽値フラグを CIRIUS 向け文字列へ変換する。
     */
    private String toFlag(Boolean ctnmKjhkFlg) {
        if (ctnmKjhkFlg == null) {
            return null;
        }
        return ctnmKjhkFlg ? "1" : "0";
    }
}
