    @Override
    public ServiceContractSupplySuspensionUpdateWithSettlementOutput execute(
            String suguHntKbn,
            Boolean ksenYukuFlg,
            Boolean sny4XUmFlg,
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            LocalDate kasYmd,
            String ckrtGmscJtkb,
            Integer suguGmtKnsu,
            List<ServiceContractSupplySuspensionUpdateWithSettlementSuguGmtInf> suguGmtInfList,
            ServiceContractSupplySuspensionUpdateWithSettlementKtmgmkkInf ktmgmkkInf,
            ServiceContractSupplySuspensionUpdateWithSettlementTkssvkyInf tkssvkyInf,
            ServiceContractSupplySuspensionUpdateWithSettlementSvckyaInf svckyaInf,
            ServiceContractSupplySuspensionUpdateWithSettlementSvmstkrInf svmstkrInf,
            ServiceContractSupplySuspensionUpdateWithSettlementSihakyInf sihakyInf,
            ServiceContractSupplySuspensionUpdateWithSettlementSihakyKaTrksInf sihakyKaTrksInf,
            ServiceContractSupplySuspensionUpdateWithSettlementSvsihkrInf svsihkrInf,
            List<ServiceContractSupplySuspensionUpdateWithSettlementGmtsetdInf> gmtsetdInfList) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        boolean kkytskrKktUmFlg = false;
        boolean suguDelFlg = false;
        boolean suguKbnUpdFlg = false;
        boolean ksenFlg = false;

        if (isNonComprehensive(suguHntKbn)) {
            if (Boolean.TRUE.equals(ksenYukuFlg)) {
                if (findSupplySuspension(gasMtstNo, trkKnsYmd).isPresent()
                        && Boolean.FALSE.equals(sny4XUmFlg)) {
                    updateServiceContractSupplyOutage(svckyaInf, auditContext, decodedOriginFunctionId);
                }
            } else {
                callSv063(gasMtstNo, trkKnsYmd, tkssvkyInf, ckrtGmscJtkb, gmtsetdInfList);
                if (Boolean.FALSE.equals(sny4XUmFlg)) {
                    updateServiceContractInvalid(svckyaInf, auditContext, decodedOriginFunctionId);
                    List<SvckyaRecord> waitingContracts = svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(
                            svckyaInf.svkyNo(),
                            waitingForContractApplicationStatus);
                    if (!waitingContracts.isEmpty()) {
                        for (SvckyaRecord waitingContract : waitingContracts) {
                            updateWaitingServiceContract(
                                    waitingContract,
                                    auditContext,
                                    decodedOriginFunctionId);
                        }
                    }
                    callSv060(svckyaInf, sihakyInf, sihakyKaTrksInf, svmstkrInf, svsihkrInf, kasYmd);
                }
            }
        } else {
            if (Boolean.TRUE.equals(ksenYukuFlg)) {
                if (findSupplySuspension(gasMtstNo, trkKnsYmd).isPresent()) {
                    ksenFlg = existsOpeningGasMeter(
                            gasMtstNo,
                            trkKnsYmd,
                            suguGmtKnsu,
                            suguGmtInfList);
                    if (!ksenFlg) {
                        updateServiceContractSupplyOutage(
                                svckyaInf,
                                auditContext,
                                decodedOriginFunctionId);
                    }
                }
            } else {
                if (Integer.valueOf(2).equals(suguGmtKnsu)) {
                    suguKbnUpdFlg = updateRelatedSvmstkrSuguKbn(
                            gasMtstNo,
                            suguGmtKnsu,
                            suguGmtInfList,
                            svmstkrInf,
                            auditContext,
                            decodedOriginFunctionId);
                }
                callSv063(gasMtstNo, trkKnsYmd, tkssvkyInf, ckrtGmscJtkb, gmtsetdInfList);
                openingCancelServiceContractMeterLocationUpdate.execute(
                        svmstkrInf.mtstNo(),
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        svmstkrInf.sknrFlg(),
                        svckyaInf.svkySttJysb());
                if (previousContractContinuation.equals(svckyaInf.svkySttJysb())
                        || Boolean.FALSE.equals(svmstkrInf.sknrFlg())) {
                    suguDelFlg = true;
                }
            }
        }

        var supplySuspension = findSupplySuspension(gasMtstNo, trkKnsYmd);
        if (supplySuspension.isPresent()) {
            updateSupplySuspensionManagement(
                    supplySuspension.orElseThrow(), auditContext, decodedOriginFunctionId);
            kkytskrKktUmFlg = true;
        }

        deletePreConfirmedMeterReading(gasMtstNo, ktmgmkkInf);

        return new ServiceContractSupplySuspensionUpdateWithSettlementOutput(
                kkytskrKktUmFlg,
                suguDelFlg,
                suguKbnUpdFlg,
                ksenFlg);
    }

    private boolean isNonComprehensive(String suguHntKbn) {
        return nonComprehensiveJudgmentCategory.equals(suguHntKbn);
    }

    private java.util.Optional<KkytskrRecord> findSupplySuspension(Long gasMtstNo, LocalDate trkKnsYmd) {
        return kkytskrDataAccess.findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
                gasMtstNo,
                trkKnsYmd,
                supplyOutageStatus);
    }

    private boolean existsOpeningGasMeter(
            Long inputGasMtstNo,
            LocalDate trkKnsYmd,
            Integer suguGmtKnsu,
            List<ServiceContractSupplySuspensionUpdateWithSettlementSuguGmtInf> suguGmtInfList) {
        int count = Math.max(0, suguGmtKnsu == null ? 0 : suguGmtKnsu);
        List<ServiceContractSupplySuspensionUpdateWithSettlementSuguGmtInf> gasMeters =
                Objects.requireNonNull(suguGmtInfList, "suguGmtInfList");
        int loopCount = Math.min(count, gasMeters.size());
        for (int index = 0; index < loopCount; index++) {
            Long gasMtstNo = gasMeters.get(index).gasMtstNo();
            if (Objects.equals(gasMtstNo, inputGasMtstNo)) {
                continue;
            }
            GmtsetdRecord gasMeter = gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, trkKnsYmd)
                    .orElseThrow(() -> DataInconsistencySupport.create(
                            gmtsetdDataAccess.tableName(),
                            GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                            new FlatJsonBuilder()
                                    .put("gasMtstNo", gasMtstNo)
                                    .put("knskKjnYmd", trkKnsYmd)));
            if (openingGasMeterStatus.equals(gasMeter.gmtKhsJtkb())) {
                return true;
            }
        }
        return false;
    }

    private void updateServiceContractSupplyOutage(
            ServiceContractSupplySuspensionUpdateWithSettlementSvckyaInf svckyaInf,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svckyaDataAccess.updateBySvkyNoAndSkno(
                buildSvckyaWriteRecordWithSupplyOutage(),
                EnumSet.of(SvckyaColumn.KKYS_SISN_JTKB),
                new SvckyaSknoUpdateKey(svckyaInf.svkyNo(), svckyaInf.skno()),

        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()

                            .put("skno", svckyaInf.skno())
                            .put("kkysSisnJtkb", supplyOutageStatus)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private boolean updateRelatedSvmstkrSuguKbn(
            Long gasMtstNo,
            Integer suguGmtKnsu,
            List<ServiceContractSupplySuspensionUpdateWithSettlementSuguGmtInf> suguGmtInfList,
            ServiceContractSupplySuspensionUpdateWithSettlementSvmstkrInf svmstkrInf,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int count = Math.max(0, suguGmtKnsu == null ? 0 : suguGmtKnsu);
        List<ServiceContractSupplySuspensionUpdateWithSettlementSuguGmtInf> gasMeters =
                Objects.requireNonNull(suguGmtInfList, "suguGmtInfList");
        int loopCount = Math.min(count, gasMeters.size());
        boolean updated = false;
        for (int index = 0; index < loopCount; index++) {
            if (Objects.equals(gasMeters.get(index).gasMtstNo(), gasMtstNo)) {
                continue;
            }
            List<SvmstkrRecord> records = findRelatedSvmstkr(svmstkrInf.svkyNo(), gasMtstNo);
            if (records.isEmpty()) {
                throw DataInconsistencySupport.create(
                        svmstkrDataAccess.tableName(),
                        relatedSvmstkrFindIdentifier(),
                        new FlatJsonBuilder()
                                .put("svkyNo", svmstkrInf.svkyNo())
                                .put("rkkiKnrnJtkbList", chargeCalculationRelatedStatuses())
                                .put("knskKjnYmd", requestSystemDate())
                                .put("mtstNo", gasMtstNo));
            }
            for (SvmstkrRecord record : records) {
                updateSvmstkrSuguKbn(record, auditContext, decodedOriginFunctionId);
                updated = true;
            }
        }
        return updated;
    }

    private List<SvmstkrRecord> findRelatedSvmstkr(Long svkyNo, Long gasMtstNo) {
        return svmstkrDataAccess.findAllBySvkyNoAndRkkiKnrnJtkbAndRkkiSttYmdAndRkkiEndYmdAndMtstNo(
                svkyNo,
                chargeCalculationRelatedStatuses(),
                requestSystemDate(),
                gasMtstNo);
    }

    private void updateSvmstkrSuguKbn(
            SvmstkrRecord svmstkrRecord,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                new SvmstkrWriteRecord(null, null, null, nonComprehensiveGeneral, null, null, null, null, null),
                EnumSet.of(SvmstkrColumn.SUGU_KBN),
                new SvmstkrSknoOptionalUpdateKey(
                        svmstkrRecord.mtstNo(),
                        svmstkrRecord.svkyNo(),
                        svmstkrRecord.skno(),
                        svmstkrRecord.recUpdJnjNo()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    svmstkrDataAccess.tableName(),
                    SvmstkrDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_SVKY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()
                            .put("mtstNo", svmstkrRecord.mtstNo())
                            .put("svkyNo", svmstkrRecord.svkyNo())
                            .put("skno", svmstkrRecord.skno())
                            .put("recUpdJnjNo", svmstkrRecord.recUpdJnjNo())
