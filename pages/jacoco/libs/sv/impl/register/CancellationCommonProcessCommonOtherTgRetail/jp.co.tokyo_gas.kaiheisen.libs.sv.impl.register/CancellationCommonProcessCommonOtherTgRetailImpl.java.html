    /**
     * TG小売の解約共通処理を実行する。
     * 専用ガスメーター以外の場合は SV-043 を実行し、専用ガスメーターの場合は SV-033 を実行する。
     * あわせて契約終了関連の下位 SV 群を順次呼び出し、最後に請求回収基盤連携ログ登録を行う。
     * RequestContext には correlationId、hukkSysShbt、systemDatetime が設定されていることを前提とする。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param tyenYmd 適用終了年月日
     * @param hesnJysb 閉栓事由種別
     * @param tksuSvkeJysb 託送サービス契約終了事由種別
     * @param svkyEndJysb サービス契約終了事由種別
     * @param gmtsetdInf ガスメーター設置場所＜導管＞情報
     * @param tskgmstInf 託送サービス契約・ガスメーター設置場所関連情報
     * @param tkssvkyInf 託送サービス契約情報
     * @param svmstkrInf サービス契約・メーター設置場所関連情報
     * @param svckyaInf サービス契約情報
     * @param svsihkrInf サービス契約・支払契約関連情報
     * @param sihakyInf 支払契約情報
     * @param snyInfList 専用メーター情報リスト
     * @param sogoTmtrInfList 総合他ガスメーター情報リスト
     * @param isrrGymCd 異動・作業履歴業務コード（小売）
     * @param isrrHusrId 異動・作業履歴報告ユーザーID
     * @param isrrHkunKj 異動・作業履歴報告ユーザー名漢字
     * @param isrrHkkrKbn 異動・作業履歴報告経路区分
     */
    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate tyenYmd,
            String hesnJysb,
            String tksuSvkeJysb,
            String svkyEndJysb,
            CancellationGmtsetdInf gmtsetdInf,
            TskgmstInf tskgmstInf,
            TkssvkyInf tkssvkyInf,
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SvmstkrInf svmstkrInf,
            jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SvckyaInf svckyaInf,
            SvsihkrInf svsihkrInf,
            SihakyInf sihakyInf,
            List<SnyInf> snyInfList,
            List<SogoTmtrInf> sogoTmtrInfList,
            String isrrGymCd,
            String isrrHusrId,
            String isrrHkunKj,
            String isrrHkkrKbn) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        LocalDateTime systemDatetime = requireSystemDatetime();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        LocalDate tkenYmdYoku = tyenYmd.plusDays(1);

        if (!isDedicatedGasMeter(gmtsetdInf)) {
            sv043.execute(
                gasMtstNo,
                tyenYmd,
                tksuSvkeJysb,
                tskgmstInf == null
                    ? null
                    : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TskgmstInf(
                        tskgmstInf.tksuSvkyNo(),
                        tskgmstInf.skno(),
                        tskgmstInf.recUpdJnjNo()),
                tkssvkyInf == null
                    ? null
                    : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TkssvkyInf(
                        tkssvkyInf.tksuSvkyNo(),
                        tkssvkyInf.skno() == null ? null : Long.valueOf(tkssvkyInf.skno()),
                        tkssvkyInf.recUpdJnjNo()),
                mapDedicatedMeters(snyInfList));
        }

        ServiceContractMeterLocationUpdateForTerminationOutput sv031Output = sv031.execute(
                gasMtstNo,
                tyenYmd,
                svkyEndJysb,
            new SvmstkrInf(
                        svmstkrInf.svkyNo(),
                        svmstkrInf.skno(),
                        svmstkrInf.recUpdJnjNo()),
            new SvckyaInf(
                        svckyaInf.svkyNo(),
                        svckyaInf.skno(),
                        svckyaInf.recUpdJnjNo()));

        sv039.execute(
            tyenYmd,
            tksuSvkeJysb,
            gmtsetdInf == null
                ? null
                : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf(
                    gmtsetdInf.szcGkkCd(),
                    gmtsetdInf.szcCmazCd(),
                    gmtsetdInf.szcMchCd(),
                    gmtsetdInf.ckrtGmscJtkb()),
            mapComprehensiveMeters(sogoTmtrInfList));
        sv029.execute(tyenYmd, new TaxExemptContractSvckyaInf(svckyaInf.svkyNo()), svckyaInf.sziKziKbn());
        sv041.execute(sihakyInf.skshHyjsKrno());
        sv044.execute(
            new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.common.model.SvckyaInf(
                svckyaInf.svkyNo(),
                svckyaInf.skno(),
                svckyaInf.recUpdJnjNo()),
            new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SvsihkrInf(
                svsihkrInf.sihKiyNo(),
                svsihkrInf.svkyNo(),
                svsihkrInf.skno(),
                svsihkrInf.ikskEndYm(),
                svsihkrInf.recUpdJnjNo()));
        sv032.execute(
            sihakyInf.sihKiyNo(),
            sihakyInf.skno(),
            sihakyInf.sihHuhuShbt(),
            sihakyInf.recUpdJnjNo());
        sv030.execute(svckyaInf.syaKbn(), sihakyInf.sihKiyNo(), systemDatetime.toLocalDate(), tyenYmd);

        if (isDedicatedGasMeter(gmtsetdInf)) {
            sv033.execute(gasMtstNo, tkenYmdYoku, mapSogoTmtrInfList(sogoTmtrInfList));
        }

        sv036.execute(
                gasMtstNo,
                tyenYmd,
                svkyEndJysb,
            mapTransferWorkHistoryMeters(sogoTmtrInfList),
                isrrGymCd,
                isrrHusrId,
                isrrHkunKj,
                isrrHkkrKbn);

        BillingCollectionPlatformIntegrationForClosingResult sv034Result =
            toBillingCollectionPlatformIntegrationForClosingResult(
                sv034.execute(
                    tyenYmd,
                    svkyEndJysb,
                    new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.
                        BillingCollectionPlatformApiIntegrationForClosingSvckyaInf(
                        svckyaInf.svkyNo(),
                        svckyaInf.skno(),
                        null,
                        svckyaInf.sqtiKiyNo()),
                    new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SihakyInf(
                        sihakyInf.sikSvcShbt(),
                        sihakyInf.sikKrno()),
                    countSogoTmtrInfList(sogoTmtrInfList),
                    isrrGymCd,
                    sv031Output.rkkiSttYmd(),
                    true,
                    null,
                    svsihkrInf.ikskEndYm()));

        irslhsDataAccess.insert(buildIrslhsWriteRecord(systemDatetime, sv034Result), auditContext);
    }

    /**
     * 概要: IRSLHS 登録用レコードを組み立てる。
     * 主な仕様:
     * - システム日時と SV-034 結果から他システム連携履歴の登録値を生成する。
     * - 時刻は HHmmss 形式の整数へ変換して設定する。
     * 既知の制約:
     * - 未使用項目は本処理では null のまま設定する。
     */
    private IrslhsWriteRecord buildIrslhsWriteRecord(
            LocalDateTime systemDatetime,
            BillingCollectionPlatformIntegrationForClosingResult sv034Result) {
        return new IrslhsWriteRecord(
                systemDatetime.toLocalDate(),
                toHms(systemDatetime),
                sv034Result.gasMtstNo(),
                sv034Result.trksUmFlg(),
                sv034Result.olkyUmFlg(),
                sv034Result.gasMtstNo(),
                sv034Result.svkyNo(),
                0L,
                sv034Result.sikKrno(),
                sv034Result.sqtiKiyNo(),
                sv034Result.idsgYmd(),
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    /**
     * 概要: 専用メーター一覧を閉栓連携向け DTO へ変換する。
     * 主な仕様:
     * - 各要素からガスメーター設置場所番号のみを抽出して DTO を生成する。
     * - 入力が null の場合は空一覧を返す。
     * 既知の制約:
     * - 変換後 DTO の詳細項目は本処理では設定しない。
     */
    private static List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf> mapDedicatedMeters(
            List<SnyInf> snyInfList) {
        if (snyInfList == null) {
            return List.of();
        }
        return snyInfList.stream()
                .map(value -> new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf(
                        new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransportServiceContractGmtsetdInf(
                                value.gmtsetdInf() == null ? null : value.gmtsetdInf().gasMtstNo())))
                .toList();
    }

    /**
     * 概要: 総合メーター一覧を閉栓連携向け DTO へ変換する。
     * 主な仕様:
     * - 小売契約情報と設置場所情報の必要項目だけを抽出して DTO を生成する。
     * - 入力が null の場合は空一覧を返す。
     * 既知の制約:
     * - 元 DTO の一部項目は連携不要のため変換対象外とする。
     */
    private static List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SogoTmtrInf> mapComprehensiveMeters(
            List<SogoTmtrInf> sogoTmtrInfList) {
        if (sogoTmtrInfList == null) {
            return List.of();
        }
        return sogoTmtrInfList.stream()
                .map(value -> new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SogoTmtrInf(
                        value.svmstkrInf() == null
                                ? null
                                : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.ComprehensiveContractSvmstkrInf(
                                        value.svmstkrInf().mtstNo(),
                                        value.svmstkrInf().svkyNo(),
                                        value.svmstkrInf().skno(),
                                        value.svmstkrInf().recUpdJnjNo()),
                        value.gmtsetdInf() == null
                                ? null
                                : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf(
                                        value.gmtsetdInf().szcGkkCd(),
                                        value.gmtsetdInf().szcCmazCd(),
                                        value.gmtsetdInf().szcMchCd(),
                                        value.gmtsetdInf().ckrtGmscJtkb())))
                .toList();
    }

                .map(value -> new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransferWorkHistorySogoTmtrInf(

                                ? null

    /**

     * 主な仕様:
     * - 出力がある場合は必要項目をそのまま転記する。
     * - 出力が null の場合は全項目 null の DTO を返す。

    private static BillingCollectionPlatformIntegrationForClosingResult toBillingCollectionPlatformIntegrationForClosingResult(

            return new BillingCollectionPlatformIntegrationForClosingResult(null, null, null, null, null, null, null);
        }
        return new BillingCollectionPlatformIntegrationForClosingResult(
                output.gasMtstNo(),
                output.trksUmFlg(),
                output.olkyUmFlg(),
                output.svkyNo(),
                output.sikKrno(),

                output.khsYmd());
    }

    /**
     * 概要: 総合メーター一覧を検針関連 SV 向け DTO 一覧へ変換する。
     * 主な仕様:
     * - 各要素からメーター設置場所番号だけを抽出して DTO を生成する。
     * - 入力が null の場合は null を返す。
     * 既知の制約:
     * - 呼び出し先 SV が null と空一覧を区別する前提で null を保持する。

    private List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.reading.model.SogoTmtrInf> mapSogoTmtrInfList(

                .toList();
    }

    /**
     * 概要: 総合メーター一覧の件数を取得する。
     * 主な仕様:
     * - 入力が null の場合は 0 を返す。
     * - それ以外は一覧サイズを返す。
     * 既知の制約:
     * - 要素内容の妥当性は確認しない。
     */
    private int countSogoTmtrInfList(List<SogoTmtrInf> sogoTmtrInfList) {
        return sogoTmtrInfList == null ? 0 : sogoTmtrInfList.size();
    }

    /**
     * 概要: ガスメーターが専用メーターかどうかを判定する。
     * 主な仕様:
     * - 関係メーター設置状態区分が専用メーター定数と一致するかを判定する。
     * - 比較には ConstantLookupKeys の定数値を使用する。
     * 既知の制約:
     * - gmtsetdInf 自体の null チェックは呼び出し元で担保する。
     */
    private boolean isDedicatedGasMeter(CancellationGmtsetdInf gmtsetdInf) {
        return ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER
                .equals(gmtsetdInf.ckrtGmscJtkb());
    }

    /**
     * 概要: 報告システム種別から起点機能 ID を解決する。
     * 主な仕様:
     * - RequestContext の hukkSysShbt を開閉栓定数マスタ名へ変換して返す。
     * - 未設定時は IllegalStateException を送出する。
     * 既知の制約:
     * - 定数マスタに対応値が存在することを前提とする。
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    /**
     * 概要: RequestContext からシステム日時を必須取得する。
     * 主な仕様:
     * - RequestContext に保持された systemDatetime を返す。
     * - 未設定時は IllegalStateException を送出する。
     * 既知の制約:
     * - SV 呼び出し前に systemDatetime が設定済みであることを前提とする。
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: 日時を HHmmss 形式の整数へ変換する。
     * 主な仕様:
     * - 時、分、秒を連結した 6 桁整数を返す。
     * - 日付部分は計算に含めない。
     * 既知の制約:
     * - null 入力は呼び出し元で排除する。
     */
    private int toHms(LocalDateTime systemDatetime) {
        return systemDatetime.getHour() * 10000
                + systemDatetime.getMinute() * 100
                + systemDatetime.getSecond();
    }
