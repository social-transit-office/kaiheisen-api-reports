/**
 * 概要: SV-158（閉栓エラーフォロー件名更新）のサービス実装。
 * 主な仕様:
 * - 託送供給依頼事業者コードに応じて、HSEFL 更新に使用するサービス契約番号を切り替える。
 * - HSEFL のフォロー状態コードを完了に更新し、フォロー対応年月日には RequestContext のシステム日付を設定する。
 * - 更新件数が 0 件の場合はデータ不整合例外を送出する。
 * 既知の制約:
 * - 実行時には RequestContext の correlationId、hukkSysShbt、systemDatetime が必須である。
 * - regTsno は SV 入力では Long だが、DA 呼び出し時に Integer へ変換する。
 */
public class ClosingErrorFollowCaseNameUpdateImpl implements ClosingErrorFollowCaseNameUpdate {

    static final String SERVICE_USER_ID = "KHSS158";

    private final HseflDataAccess hseflDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String followCompletedStatus;

    /**
     * 概要: 閉栓エラーフォロー件名更新に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - HSEFL 更新、RequestContext、開閉栓定数サービスを保持する。
          * - フォロー完了コードを初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
          * - 必須 CIRIUS 定数が不足している場合はコンストラクタ初期化時点で失敗する。
     *
     * @param hseflDataAccess HSEFL データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public ClosingErrorFollowCaseNameUpdateImpl(
            HseflDataAccess hseflDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.followCompletedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.COMPLETED_FOLLOW_COMPLETED);
    }

    @Override
        public void execute(
            String tksuKijsCd,
            Long gasMtstNo,
            LocalDate hssgYmd,
            Long regTsno,
            LocalDate hesnHukkYmd,
            Integer recUpdJnjNo,
            LocalDate tsigHesnSgyYmd,
            String tsigGmb,
            Long sihKiyNo,
            Long svkyNo,
            Long tksuSvkyNo,
            Long hsnjGmsSu,
            BigDecimal hsnjGasSyry,
            String sjskKbn) {
        LocalDateTime systemDatetime = requireRequestContext();

        Integer safeRegTsno = toRegTsno(regTsno);
        Long resolvedServiceContractNo = resolveServiceContractNumber(tksuKijsCd, svkyNo, tksuSvkyNo);
        LocalDate followDate = systemDatetime.toLocalDate();
        String decodedOriginFunctionId = decodeOriginFunctionId();

        HseflWriteRecord record = buildWriteRecord(
                gasMtstNo,
                hssgYmd,
                hesnHukkYmd,
                safeRegTsno,
                tsigHesnSgyYmd,
                tsigGmb,
                sihKiyNo,
                resolvedServiceContractNo,
                hsnjGmsSu,
                hsnjGasSyry,
                sjskKbn,
                followDate);

        int updatedCount = hseflDataAccess.updateByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsnoAndRecUpdJnjNo(
                record,
                Set.of(
                        HseflColumn.TSIG_HESN_SGY_YMD,
                        HseflColumn.TSIG_GMB,
                        HseflColumn.FLW_ST_CD,
                        HseflColumn.FLTK_YMD,
                        HseflColumn.SIH_KIY_NO,
                        HseflColumn.SVKY_NO,
                        HseflColumn.HSNJ_GMS_SU,
                        HseflColumn.HSNJ_GAS_SYRY,
                        HseflColumn.SJSK_KBN),
                new HseflU001UpdateKey(gasMtstNo, hssgYmd, hesnHukkYmd, safeRegTsno, recUpdJnjNo),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));

        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    hseflDataAccess.tableName(),
                    HseflDataAccess.CallSpecs
                            .UPDATE_BY_GAS_MTST_NO_AND_HSSG_YMD_AND_HESN_HUKK_YMD_AND_REG_TSNO_AND_REC_UPD_JNJ_NO
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", gasMtstNo)
                            .put("hssgYmd", hssgYmd)
                            .put("hesnHukkYmd", hesnHukkYmd)
                            .put("regTsno", safeRegTsno)
                            .put("recUpdJnjNo", recUpdJnjNo)
                            .put("tsigHesnSgyYmd", tsigHesnSgyYmd)
                            .put("tsigGmb", tsigGmb)
                            .put("flwStCd", followCompletedStatus)
                            .put("fltkYmd", followDate)
                            .put("sihKiyNo", sihKiyNo)
                            .put("svkyNo", resolvedServiceContractNo)
                            .put("hsnjGmsSu", hsnjGmsSu)
                            .put("hsnjGasSyry", hsnjGasSyry)
                            .put("sjskKbn", sjskKbn)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    /**
     * 概要: HSEFL 更新用の write record を組み立てる。
     * 主な仕様:
     * - 更新対象列に対応する値のみを設定し、その他は null のまま渡す。
     * - フォロー状態コードにはコンストラクタで解決した完了コードを設定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param hssgYmd 閉栓作業年月日
     * @param hesnHukkYmd 閉栓報告年月日
     * @param regTsno 登録通し番号
     * @param tsigHesnSgyYmd 訂正後閉栓作業年月日
     * @param tsigGmb 訂正後ガスメーター社番
     * @param sihKiyNo 支払契約番号
     * @param resolvedServiceContractNo 更新対象のサービス契約番号
     * @param hsnjGmsSu 閉栓時ガスメーター指示数
     * @param hsnjGasSyry 閉栓時ガス使用量
     * @param sjskKbn 指示数回帰区分
     * @param followDate フォロー対応年月日
     * @return HSEFL 更新用レコード
     */
    private HseflWriteRecord buildWriteRecord(
            Long gasMtstNo,
            LocalDate hssgYmd,
            LocalDate hesnHukkYmd,
            Integer regTsno,
            LocalDate tsigHesnSgyYmd,
            String tsigGmb,
            Long sihKiyNo,
            Long resolvedServiceContractNo,
            Long hsnjGmsSu,
            BigDecimal hsnjGasSyry,
            String sjskKbn,
            LocalDate followDate) {
        return new HseflWriteRecord(
                                /* gasMtstNo */ gasMtstNo,
                                /* hssgYmd */ hssgYmd,
                                /* hesnHukkYmd */ hesnHukkYmd,
                                /* regTsno */ regTsno,
                                /* hukkSysShbt */ null,
                                /* sgyAnknNo */ null,
                                /* flwStCd */ followCompletedStatus,
                                /* fltkYmd */ followDate,
                                /* syoKyoTrno */ null,
                                /* gmb */ null,
                                /* hesnJysb */ null,
                                /* hesnErryCd */ null,
                                /* flttCktkCd */ null,
                                /* flttKtnCd */ null,
                                /* hssgNwkCd */ null,
                                /* sgytSinNo */ null,
                                /* sgytSinnKj */ null,
                                /* hsnjGmsSu */ hsnjGmsSu,
                                /* hsnjGasSyry */ hsnjGasSyry,
                                /* sjskKbn */ sjskKbn,
                                /* rshhSjsb */ null,
                                /* hkshHshhShbt */ null,
                                /* hesnSdnShbt */ null,
                                /* utidxNo */ null,
                                /* ctnmKn1 */ null,
                                /* ctnmKj1 */ null,
                                /* ctnmKjhkFlg */ null,
                                /* ctnm1HujnShbt */ null,
                                /* ctnm1HshiShbt */ null,
                                /* tsigHesnSgyYmd */ tsigHesnSgyYmd,
                                /* tsigGmb */ tsigGmb,
                                /* sihKiyNo */ sihKiyNo,
                                /* svkyNo */ resolvedServiceContractNo,
                                /* tkjMem */ null);
    }

    /**
     * 概要: 託送供給依頼事業者コードに応じて更新対象のサービス契約番号を解決する。
     * 主な仕様:
     * - 東京ガス株式会社の場合は `svkyNo` を返す。
     * - それ以外の場合は `tksuSvkyNo` を返す。
     *
     * @param tksuKijsCd 託送供給依頼事業者コード
     * @param svkyNo サービス契約番号
     * @param tksuSvkyNo 託送サービス契約番号
     * @return 更新対象のサービス契約番号
     */
    private Long resolveServiceContractNumber(String tksuKijsCd, Long svkyNo, Long tksuSvkyNo) {
        if (ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD.equals(tksuKijsCd)) {
            return svkyNo;
        }
        return tksuSvkyNo;
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     * 主な仕様:
     * - correlationId、hukkSysShbt、systemDatetime の存在を確認する。
     * - 欠落がある場合は最初に検出した項目で処理を中断する。
     */
    private LocalDateTime requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: RequestContext の報告システム種別から監査用の機能 ID を取得する。
     * 主な仕様:
     * - RequestContext から hukkSysShbt を取得し、開閉栓定数サービスで名称へ変換する。
     * - 返却値は監査情報の updKtnKinoId に設定する。
     *
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    /**
     * 概要: SV 入力の登録通し番号を DA 用の Integer へ変換する。
     * 主な仕様:
     * - Long から Integer へ `Math.toIntExact` で変換する。
     * - 範囲外値は ArithmeticException とする。
     *
     * @param regTsno 登録通し番号
     * @return DA 用の登録通し番号
     */
    private Integer toRegTsno(Long regTsno) {
        return Math.toIntExact(regTsno);
    }
}
