package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.KaiheisenConstantItem;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.SqshyjsDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.SqshyjsRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsTkysYmdUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsWriteRecord;

/**
 * SV-113（請求書表示制御情報更新（契約復活））のサービス実装.
 * 主な仕様:
 * - RequestContext のシステム日時を基準年月日に使用して請求書表示制御を取得する。
 * - 取得した適用開始年月日を更新キーに使用し、適用終了年月日を最大日付へ更新する。
 * - 更新件数が 0 件の場合はデータ不整合例外へ変換する。
 * 既知の制約:
 * - public メソッド引数の null 検証は追加しない。
 * - 報告システム種別定数の存在確認はコンストラクタ初期化時に行う。
 */
public class InvoiceDisplayControlUpdateForRevivalImpl
        implements InvoiceDisplayControlUpdateForRevival {

        /** 監査更新で使用する固定の更新ユーザー ID. */
    static final String SERVICE_USER_ID = "KHSS113";

        /** 契約復活で設定する適用終了年月日の最大日付. */
    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

        /** SQSHYJS 更新を担当するデータアクセス. */
    private final SqshyjsDataAccess sqshyjsDataAccess;

        /** 実行時の監査・システム情報を保持するリクエストコンテキスト. */
    private final RequestContext requestContext;

        /** 報告元機能 ID の復号に使用する開閉栓定数サービス. */
    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * SV-113 の依存オブジェクトを初期化する.
     * 主な仕様:
     * - SQSHYJS データアクセス、RequestContext、開閉栓定数サービスを保持する。
     * - 報告システム種別定数が未定義の場合は初期化時点で失敗する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * @param inputSqshyjsDataAccess SQSHYJS データアクセス
     * @param inputRequestContext リクエストコンテキスト
     * @param inputKaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public InvoiceDisplayControlUpdateForRevivalImpl(
            final SqshyjsDataAccess inputSqshyjsDataAccess,
            final RequestContext inputRequestContext,
            final KaiheisenConstantService inputKaiheisenConstantService) {
        this.sqshyjsDataAccess = Objects.requireNonNull(
                inputSqshyjsDataAccess,
                "sqshyjsDataAccess");
        this.requestContext = Objects.requireNonNull(
                inputRequestContext,
                "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                inputKaiheisenConstantService,
                "kaiheisenConstantService");
        requireReportingSystemTypeConstants();
    }

    /**
     * 契約復活対象の請求書表示制御を更新する.
     * 主な仕様:
     * - RequestContext の systemDatetime と hukkSysShbt を fail-fast で検証する。
     * - 取得結果が 0 件の場合は処理対象データ無しエラーを送出する。
     * - 更新件数が 0 件の場合はデータ不整合例外を送出する。
     * @param skshHyjsKrno 請求書表示制御管理番号
     */
    @Override
    public void execute(final Long skshHyjsKrno) {
        final LocalDate knskKjnYmd = requireSystemDate();
        final String decodedOriginFunctionId = decodeOriginFunctionId();
        final SqshyjsRecord existingRecord = findExistingRecord(
                skshHyjsKrno,
                knskKjnYmd);
        updateExistingRecord(existingRecord, decodedOriginFunctionId);
    }

    /**
     * 指定条件で有効な請求書表示制御を取得する.
     * 主な仕様:
     * - 請求書表示制御管理番号と基準年月日で有効レコードを 1 件取得する。
     * - 取得結果が 0 件の場合は処理対象データ無しエラーを送出する。
     * @param skshHyjsKrno 請求書表示制御管理番号
     * @param knskKjnYmd 基準年月日
     * @return 取得済み請求書表示制御
     */
    private SqshyjsRecord findExistingRecord(
            final Long skshHyjsKrno,
            final LocalDate knskKjnYmd) {
        return sqshyjsDataAccess
                .findBySkshHyjsKrnoAndKnskKjnYmd(skshHyjsKrno, knskKjnYmd)
                .orElseThrow(() -> new DataNotFoundException(
                        sqshyjsDataAccess.tableName()));
    }

    /**
     * 取得済み請求書表示制御を契約復活状態へ更新する.
     * 主な仕様:
     * - 適用終了年月日を最大日付へ更新する。
     * - 更新件数が 0 件の場合はデータ不整合例外へ変換する。
     * @param existingRecord 取得済み請求書表示制御
     * @param decodedOriginFunctionId 報告元機能 ID
     */
    private void updateExistingRecord(
            final SqshyjsRecord existingRecord,
            final String decodedOriginFunctionId) {
        final int updatedCount =
                sqshyjsDataAccess.updateBySkshHyjsKrnoAndTkysYmd(
                        new SqshyjsWriteRecord(
                        existingRecord.skshHyjsKrno(),
                        existingRecord.tkysYmd(),
                        MAX_DATE,
                        SERVICE_USER_ID,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                        Set.of(
                                SqshyjsColumn.TYEN_YMD,
                                SqshyjsColumn.REG_USR_ID),
                        new SqshyjsTkysYmdUpdateKey(
                        existingRecord.skshHyjsKrno(),
                        existingRecord.tkysYmd()),
                        AuditContext.of(
                                SERVICE_USER_ID,
                                decodedOriginFunctionId));
        if (updatedCount == 0) {
            final String daIdentifier =
                    SqshyjsDataAccess.CallSpecs
                            .UPDATE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD
                            .identifier();
            throw DataInconsistencySupport.create(
                    sqshyjsDataAccess.tableName(),
                    daIdentifier,
                    buildUpdateParameterJson(existingRecord,
                            decodedOriginFunctionId));
        }
    }

    /**
     * データ不整合例外に含める更新パラメータ JSON を構築する.
     * 主な仕様:
     * - 更新キー、更新値、監査情報を FlatJsonBuilder へ定義順で格納する。
     * @param existingRecord 取得済み請求書表示制御
     * @param decodedOriginFunctionId 報告元機能 ID
     * @return 更新パラメータ JSON ビルダー
     */
    private FlatJsonBuilder buildUpdateParameterJson(
            final SqshyjsRecord existingRecord,
            final String decodedOriginFunctionId) {
        return new FlatJsonBuilder()
                .put("skshHyjsKrno", existingRecord.skshHyjsKrno())
                .put("tkysYmd", existingRecord.tkysYmd())
                .put("tyenYmd", MAX_DATE)
                .put("regUsrId", SERVICE_USER_ID)
                .put("updUsrId", SERVICE_USER_ID)
                .put("updKtnKinoId", decodedOriginFunctionId);
    }

    /**
     * RequestContext のシステム日時を業務日付へ変換する.
     * 主な仕様:
     * - systemDatetime が未設定の場合は fail-fast する。
     * @return 業務日付
     */
    private LocalDate requireSystemDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException(
                        "RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    /**
     * RequestContext の報告システム種別を監査用機能 ID へ変換する.
     * 主な仕様:
     * - hukkSysShbt が未設定の場合は fail-fast する。
     * - 変換には開閉栓コンスタント取得サービスを使用する。
     * @return 監査用機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException(
                                "RequestContext.hukkSysShbt is required")));
    }

    /**
     * 報告システム種別定数の存在を初期化時に検証する.
     * 主な仕様:
     * - 報告システム種別一覧が空の場合は fail-fast する。
     */
    private void requireReportingSystemTypeConstants() {
        final List<KaiheisenConstantItem> reportingSystemTypes =
                kaiheisenConstantService.listActiveByType(
                        ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE);
        if (reportingSystemTypes.isEmpty()) {
            throw new IllegalStateException(
                    "Kaiheisen reporting system type constants are required");
        }
    }
}
