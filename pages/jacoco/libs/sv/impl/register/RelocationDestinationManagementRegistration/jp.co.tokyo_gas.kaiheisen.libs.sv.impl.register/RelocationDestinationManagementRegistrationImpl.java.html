package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import jakarta.inject.Inject;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.RelocationDestinationManagementAddition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.RelocationDestinationManagementRegistration;

/**
 * 概要: SV-129（移転先管理登録）のサービス実装。
 * 主な仕様:
 * - 移転前管轄支社コードから移転前管轄基地コードを決定する。
 * - 計算した基地コードを付与して SV-262 へ処理を委譲する。
 * 既知の制約:
 * - SV-129 自身は RequestContext を参照せず、群馬支社判定と下位 SV 委譲のみを責務とする。
 */
public class RelocationDestinationManagementRegistrationImpl
        implements RelocationDestinationManagementRegistration {

    static final String GUNMA_BRANCH_BASE_CODE = "002";

    static final String DEFAULT_BRANCH_BASE_CODE = "001";

    private final RelocationDestinationManagementAddition relocationDestinationManagementAddition;

    private final String gunmaBranchCode;

    /**
     * 概要: SV-129 に必要な依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - 下位 SV 依存と開閉栓定数サービス依存を fail-fast で検証する。
     * - 群馬支社コード定数の存在をコンストラクタ初期化時点で確認する。
     * 既知の制約:
     * - 基地コード `001` / `002` 自体は source 正本に従う固定値として保持する。
     *
     * @param relocationDestinationManagementAddition 移転先管理追加サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    @Inject
    public RelocationDestinationManagementRegistrationImpl(
            RelocationDestinationManagementAddition relocationDestinationManagementAddition,
            KaiheisenConstantService kaiheisenConstantService) {
        this.relocationDestinationManagementAddition = Objects.requireNonNull(
                relocationDestinationManagementAddition,
                "relocationDestinationManagementAddition");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.gunmaBranchCode = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.BranchCode.TYPE,
                ConstantLookupKeys.Kaiheisen.BranchCode.GUNMA_BRANCH);
    }

    /**
     * 概要: 移転先管理登録を実行する。
     * 主な仕様:
     * - 群馬支社コードの場合は基地コード `002`、それ以外は `001` を設定する。
     * - 計算結果を付与して SV-262 に処理を委譲する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param svkyNo サービス契約番号
     * @param itmSihKiyNo 移転前支払契約番号
     * @param itmKktSsyaCd 移転前管轄支社コード
     * @param itmKktEgsyCd 移転前管轄営業所コード
     */
    @Override
    public void execute(
            Long gasMtstNo,
            Long svkyNo,
            Long itmSihKiyNo,
            String itmKktSsyaCd,
            String itmKktEgsyCd) {
        relocationDestinationManagementAddition.execute(
                gasMtstNo,
                svkyNo,
                itmSihKiyNo,
                itmKktSsyaCd,
                itmKktEgsyCd,
                resolveBaseCode(itmKktSsyaCd));
    }

    /**
     * 概要: 支社コードから移転前管轄基地コードを決定する。
     * 主な仕様:
     * - 群馬支社コードの場合は `002` を返す。
     * - それ以外の場合は `001` を返す。
     * 既知の制約:
     * - 基地コードは source 正本に記載された固定値をそのまま返す。
     *
     * @param itmKktSsyaCd 移転前管轄支社コード
     * @return 移転前管轄基地コード
     */
    private String resolveBaseCode(String itmKktSsyaCd) {
        if (gunmaBranchCode.equals(itmKktSsyaCd)) {
            return GUNMA_BRANCH_BASE_CODE;
        }
        return DEFAULT_BRANCH_BASE_CODE;
    }
}
