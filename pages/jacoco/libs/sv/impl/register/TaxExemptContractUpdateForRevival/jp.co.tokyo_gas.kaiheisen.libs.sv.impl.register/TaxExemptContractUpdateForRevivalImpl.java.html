package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.MnzkiyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.MnzkiyRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.write.MnzkiyColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.write.MnzkiySvkyTkysUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.write.MnzkiyWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.TaxExemptContractUpdateForRevival;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.TaxExemptContractUpdateForRevivalOutput;

/**
 * 概要: SV-109（免税契約更新（契約復活））のサービス実装。
 * 主な仕様:
 * - 取消対象検針年月日時点で有効な免税契約を 1 件検索し、存在する場合のみ契約復活状態へ更新する。
 * - 更新時は免税適用終了年月日を最大日付へ戻し、免税適用終了報告年月日を null にする。
 * 既知の制約:
 * - public メソッド引数の null 検証は追加しない。
 * - 報告システム種別は RequestContext から取得し、監査情報の起点機能 ID に利用する。
 */
public class TaxExemptContractUpdateForRevivalImpl implements TaxExemptContractUpdateForRevival {

    static final String SERVICE_USER_ID = "KHSS109";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private final MnzkiyDataAccess mnzkiyDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * 概要: SV-109 の依存を初期化する。
     * 主な仕様:
     * - MNZKIY データアクセス、RequestContext、開閉栓定数サービスを保持する。
     * - 依存が不足している場合は初期化時点で失敗する。
     *
     * @param mnzkiyDataAccess MNZKIY データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public TaxExemptContractUpdateForRevivalImpl(
            MnzkiyDataAccess mnzkiyDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.mnzkiyDataAccess = Objects.requireNonNull(mnzkiyDataAccess, "mnzkiyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
    }

    /**
     * 概要: 契約復活対象の免税契約を更新する。
     * 主な仕様:
     * - 免税契約が存在しない場合は更新せず false を返却する。
     * - 免税契約が存在する場合は更新し true を返却する。
     *
     * @param svkyNo サービス契約番号
     * @param trkKnsYmd 取消対象検針年月日
     * @return 免税契約有無フラグ
     */
    @Override
    public TaxExemptContractUpdateForRevivalOutput execute(Long svkyNo, LocalDate trkKnsYmd) {
        return mnzkiyDataAccess.findBySvkyNoAndMnziTkysYmdAndMnziTyenYmd(svkyNo, trkKnsYmd)
                .map(this::updateExistingRecord)
                .orElseGet(() -> new TaxExemptContractUpdateForRevivalOutput(false));
    }

    /**
     * 概要: 取得済みの免税契約情報を契約復活状態へ更新する。
     * 主な仕様:
     * - 免税適用終了年月日を最大日付へ更新する。
     * - 免税適用終了報告年月日を null に更新する。
     *
     * @param record 取得済み免税契約情報
     * @return 免税契約有無フラグ
     */
    private TaxExemptContractUpdateForRevivalOutput updateExistingRecord(MnzkiyRecord record) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        mnzkiyDataAccess.updateBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
                new MnzkiyWriteRecord(
                        record.svkyNo(),
                        record.mnziTkysYmd(),
                        MAX_DATE,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        record.recUpdJnjNo()),
                Set.of(MnzkiyColumn.MNZI_TYEN_YMD, MnzkiyColumn.MNZI_TYEH_YMD),
                new MnzkiySvkyTkysUpdateKey(record.svkyNo(), record.mnziTkysYmd(), record.recUpdJnjNo()),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        return new TaxExemptContractUpdateForRevivalOutput(true);
    }

    /**
     * 概要: 報告システム種別から監査情報用の起点機能 ID を解決する。
     * 主な仕様:
     * - RequestContext から報告システム種別を取得する。
     * - 報告システム種別が未設定の場合は fail-fast する。
     *
     * @return 起点機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
