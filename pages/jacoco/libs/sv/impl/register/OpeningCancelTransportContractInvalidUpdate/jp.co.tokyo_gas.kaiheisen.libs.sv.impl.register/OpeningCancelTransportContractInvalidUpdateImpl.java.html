/**
 * 概要: SV-063（開栓取消託送サービス契約無効更新）のサービス実装。
 * 主な仕様:
 * - 専用ガスメーター以外では top-level のガスメーター設置場所番号に対して無効更新を行う。
 * - 元ガスメーターでは導管側ガスメーター設置場所情報リストを追加で反復し、東京ガスの場合は導管通知確定使用量も削除する。
 * - 更新件数または削除件数が 0 件の場合は DataInconsistencyException を送出する。
 */
public final class OpeningCancelTransportContractInvalidUpdateImpl
        implements OpeningCancelTransportContractInvalidUpdate {

    static final String SERVICE_USER_ID = "KHSS063";

    static final int DELETE_KNS_KKA_TSNO = 0;

    private static final String RELATION_READ_DA_IDENTIFIER =
            TskgmstDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_TKSU_SVKK_JTKB
                    .identifier();

    private static final String CONTRACT_READ_DA_IDENTIFIER =
            TkssvkyDataAccess.CallSpecs.FIND_ALL_BY_TKSU_SVKY_NO_AND_KNSK_KJN_YMD_AND_TKSU_SVKY_JTKB.identifier();

    private static final String CONTRACT_UPDATE_DA_IDENTIFIER =
            TkssvkyDataAccess.CallSpecs.UPDATE_BY_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO.identifier();

    private static final String RELATION_UPDATE_DA_IDENTIFIER =
            TskgmstDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO
                    .identifier();

    private static final String DELETE_DA_IDENTIFIER =
            DktksryDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO.identifier();

    private final TskgmstDataAccess tskgmstDataAccess;

    private final TkssvkyDataAccess tkssvkyDataAccess;

    private final DktksryDataAccess dktksryDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String originalGasMeterStatus;

    private final String dedicatedGasMeterStatus;

    private final String tokyoGasRetailerCode;

    private final String applyingRelationStatus;

    private final String invalidRelationStatus;

    private final String underContractStatus;

    private final String invalidContractStatus;

    /**
     * 必須依存と定数を初期化する。
     */
    public OpeningCancelTransportContractInvalidUpdateImpl(
            TskgmstDataAccess tskgmstDataAccess,
            TkssvkyDataAccess tkssvkyDataAccess,
            DktksryDataAccess dktksryDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.tskgmstDataAccess = Objects.requireNonNull(tskgmstDataAccess, "tskgmstDataAccess");
        this.tkssvkyDataAccess = Objects.requireNonNull(tkssvkyDataAccess, "tkssvkyDataAccess");
        this.dktksryDataAccess = Objects.requireNonNull(dktksryDataAccess, "dktksryDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.originalGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.ORIGINAL_GAS_METER);
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.tokyoGasRetailerCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);
        this.applyingRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING);
        this.invalidRelationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.INVALID);
        this.underContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT);
        this.invalidContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.TransportationServiceContractStatus.INVALID);
    }

    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            String tksuKijsCd,
            String ckrtGmscJtkb,
            List<OpeningCancelTransportContractInvalidUpdateGmtsetdInf> gmtsetdInfList) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        if (!dedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            invalidateTransportContract(gasMtstNo, trkKnsYmd, false, auditContext, decodedOriginFunctionId);
        }

        if (originalGasMeterStatus.equals(ckrtGmscJtkb)) {
            boolean shouldDeleteDktksry = tokyoGasRetailerCode.equals(tksuKijsCd);
            for (OpeningCancelTransportContractInvalidUpdateGmtsetdInf gmtsetdInf : gmtsetdInfList) {
                invalidateTransportContract(
                        gmtsetdInf.gasMtstNo(),
                        trkKnsYmd,
                        shouldDeleteDktksry,
                        auditContext,
                        decodedOriginFunctionId);
            }
        }
    }

    private void invalidateTransportContract(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            boolean shouldDeleteDktksry,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        TskgmstRecord relationRecord = findRelationRecord(gasMtstNo, trkKnsYmd);
        TkssvkyRecord contractRecord = findContractRecord(relationRecord.tksuSvkyNo(), trkKnsYmd);

        int updatedContractCount = tkssvkyDataAccess.updateByTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TkssvkyWriteRecord(
                        contractRecord.tksuSvkyNo(),
                        contractRecord.skno(),
                        invalidContractStatus,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                Set.of(TkssvkyColumn.TKSU_SVKY_JTKB),
                new TkssvkySknoJnjUpdateKey(contractRecord.tksuSvkyNo(), contractRecord.skno()),
                auditContext);
        if (updatedContractCount == 0) {
            throw createContractUpdateException(contractRecord, decodedOriginFunctionId);
        }

        int updatedRelationCount = tskgmstDataAccess.updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(
                new TskgmstWriteRecord(
                        relationRecord.gasMtstNo(),
                        relationRecord.tksuSvkyNo(),
                        relationRecord.skno(),
                        invalidRelationStatus,
                        null,
                        null),
                Set.of(TskgmstColumn.TKSU_SVKK_JTKB),
                new TskgmstUpdateKey(
                        relationRecord.gasMtstNo(),
                        relationRecord.tksuSvkyNo(),
                        relationRecord.skno()),
                auditContext);
        if (updatedRelationCount == 0) {
            throw createRelationUpdateException(relationRecord, decodedOriginFunctionId);
        }

        if (shouldDeleteDktksry) {
            int deletedCount = dktksryDataAccess.deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                    relationRecord.gasMtstNo(),
                    trkKnsYmd,
                    DELETE_KNS_KKA_TSNO,
                    null);
            if (deletedCount == 0) {
                throw createDktksryDeleteException(relationRecord.gasMtstNo(), trkKnsYmd);
            }
        }
    }

    private TskgmstRecord findRelationRecord(Long gasMtstNo, LocalDate trkKnsYmd) {
        List<TskgmstRecord> relationRecords =
                tskgmstDataAccess.findAllByGasMtstNoAndTkysYmdAndTyenYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        trkKnsYmd,
                        applyingRelationStatus);
        if (relationRecords.isEmpty()) {
            throw createRelationReadException(gasMtstNo, trkKnsYmd);
        }
        return relationRecords.getFirst();
    }

    private TkssvkyRecord findContractRecord(Long tksuSvkyNo, LocalDate trkKnsYmd) {
        List<TkssvkyRecord> contractRecords =
                tkssvkyDataAccess.findAllByTksuSvkyNoAndKnskKjnYmdAndTksuSvkyJtkb(
                        tksuSvkyNo,
                        trkKnsYmd,
                        underContractStatus);
        if (contractRecords.isEmpty()) {
            throw createContractReadException(tksuSvkyNo, trkKnsYmd);
        }
        return contractRecords.getFirst();
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private RuntimeException createRelationReadException(Long gasMtstNo, LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                RELATION_READ_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", trkKnsYmd)
                        .put("tksuSvkkJtkb", applyingRelationStatus));
    }

    private RuntimeException createContractReadException(Long tksuSvkyNo, LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                CONTRACT_READ_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", tksuSvkyNo)
                        .put("knskKjnYmd", trkKnsYmd)
                        .put("tksuSvkyJtkb", underContractStatus));
    }

    private RuntimeException createContractUpdateException(
            TkssvkyRecord contractRecord,
            String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tkssvkyDataAccess.tableName(),
                CONTRACT_UPDATE_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("tksuSvkyNo", contractRecord.tksuSvkyNo())
                        .put("skno", contractRecord.skno())
                        .put("tksuSvkyJtkb", invalidContractStatus)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createRelationUpdateException(
            TskgmstRecord relationRecord,
            String decodedOriginFunctionId) {
        return DataInconsistencySupport.create(
                tskgmstDataAccess.tableName(),
                RELATION_UPDATE_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("gasMtstNo", relationRecord.gasMtstNo())
                        .put("tksuSvkyNo", relationRecord.tksuSvkyNo())
                        .put("skno", relationRecord.skno())
                        .put("tksuSvkkJtkb", invalidRelationStatus)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private RuntimeException createDktksryDeleteException(Long gasMtstNo, LocalDate trkKnsYmd) {
        return DataInconsistencySupport.create(
                dktksryDataAccess.tableName(),
                DELETE_DA_IDENTIFIER,
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knsYmd", trkKnsYmd)
                        .put("knsKkaTsno", DELETE_KNS_KKA_TSNO));
    }
}
