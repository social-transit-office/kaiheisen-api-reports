package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import jakarta.inject.Inject;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaiheisen_change_notification.CustomerServiceHistoryRegistrationRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaiheisen_change_notification.KaiheisenChangeNotificationCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.CustomerServiceHistoryApiIntegration;

/**
 * 概要: SV-093（接客履歴登録API連携）のサービス実装。
 * 主な仕様:
 * - RequestContext の systemDatetime と入力引数から他-043 request を組み立てる。
 * - KH034=1 は開栓、KH034=2 は閉栓の業務詳細区分コードで request を生成する。
 * - KH034 が 1/2 以外の場合は外部呼出しを行わず正常終了する。
 * 既知の制約:
 * - 外部 API 呼出し失敗時の再試行や例外変換は行わない。
 * - 本実装は他-043 への request 編集と委譲に限定し、レスポンス処理は行わない。
 */
public final class CustomerServiceHistoryApiIntegrationImpl implements CustomerServiceHistoryApiIntegration {

    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private static final DateTimeFormatter TIME_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    private final RequestContext requestContext;

    private final KaiheisenChangeNotificationCiriusAdapter adapter;

    private final String otherRetailFlag;

    private final String visitMethodCode;

    private final String openingProcessingDivision;

    private final String closingProcessingDivision;

    private final String openingDetailCode;

    private final String closingDetailCode;

    /**
     * 概要: CDI から他-043 adapter を受け取り初期化する。
     *
     * @param requestContext リクエストコンテキスト
     * @param adapter CIRIUS 接客履歴登録 API adapter
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public CustomerServiceHistoryApiIntegrationImpl(
            RequestContext requestContext,
            KaiheisenChangeNotificationCiriusAdapter adapter,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.adapter = Objects.requireNonNull(adapter, "adapter");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.otherRetailFlag = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.TYPE,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.OTHER_RETAIL);
        this.visitMethodCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CustomerServiceType.TYPE,
                ConstantLookupKeys.Cirius.CustomerServiceType.VISIT);
        this.openingProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.CustomerServiceHistoryApiIntegrationProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.CustomerServiceHistoryApiIntegrationProcessingDivision.OPENING_PROCESS);
        this.closingProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.CustomerServiceHistoryApiIntegrationProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.CustomerServiceHistoryApiIntegrationProcessingDivision.CLOSING_PROCESS);
        this.openingDetailCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkCodeConduit.OPENING);
        this.closingDetailCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkCodeConduit.CLOSING);
    }

    /**
     * 概要: 接客履歴登録 API 連携を実行する。
     */
    @Override
    public void execute(
            Long kyoTrkNo,
            String sagyoSyaEmpNo,
            String sagyoSyaName,
            String sagyoKasyoCd,
            Long gasMtstNo,
            String gymNyoSybtCd,
            String kasrksiIgiSkrrTrkShrKbn) {
        String gymShsiKbnCd = resolveDetailCode(kasrksiIgiSkrrTrkShrKbn);
        if (gymShsiKbnCd == null) {
            return;
        }

        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));

        CustomerServiceHistoryApiIntegrationRequest.CallParameters callParameters = buildCallParameters(
                kyoTrkNo,
                sagyoSyaEmpNo,
                sagyoSyaName,
                sagyoKasyoCd,
                gasMtstNo,
                gymNyoSybtCd,
                gymShsiKbnCd,
                systemDatetime);
        CustomerServiceHistoryApiIntegrationRequest integrationRequest =
                new CustomerServiceHistoryApiIntegrationRequest(callParameters);
        CustomerServiceHistoryRegistrationRequest.CustomerServiceHistoryInfo registrationRequest =
                buildRegistrationRequest(integrationRequest);

        adapter.registerCustomerServiceHistory(registrationRequest);
    }

    private String resolveDetailCode(String kasrksiIgiSkrrTrkShrKbn) {
        if (openingProcessingDivision.equals(kasrksiIgiSkrrTrkShrKbn)) {
            return openingDetailCode;
        }
        if (closingProcessingDivision.equals(kasrksiIgiSkrrTrkShrKbn)) {
            return closingDetailCode;
        }
        return null;
    }

    private CustomerServiceHistoryApiIntegrationRequest.CallParameters buildCallParameters(
            Long kyoTrkNo,
            String sagyoSyaEmpNo,
            String sagyoSyaName,
            String sagyoKasyoCd,
            Long gasMtstNo,
            String gymNyoSybtCd,
            String gymShsiKbnCd,
            LocalDateTime systemDatetime) {
        Integer ymd = toYmd(systemDatetime);
        Integer hm = toHm(systemDatetime);
        return new CustomerServiceHistoryApiIntegrationRequest.CallParameters(
                otherRetailFlag,
                null,
                kyoTrkNo,
                blankIfNull(sagyoSyaEmpNo),
                visitMethodCode,
                gymShsiKbnCd,
                ymd,
                hm,
                null,
                null,
                hm,
                blankIfNull(sagyoSyaName),
                blankIfNull(sagyoKasyoCd),
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                gasMtstNo,
                null,
                null,
                null,
                gymNyoSybtCd,
                null,
                null,
                null);
    }

    private Integer toYmd(LocalDateTime value) {
        if (value == null) {
            return null;
        }
        return Integer.valueOf(DATE_FORMATTER.format(value.toLocalDate()));
    }

    private Integer toHm(LocalDateTime value) {
        if (value == null) {
            return null;
        }
        return Integer.valueOf(TIME_FORMATTER.format(value.toLocalTime()));
    }

    private String blankIfNull(String value) {
        return value == null ? "" : value;
    }

    private CustomerServiceHistoryRegistrationRequest.CustomerServiceHistoryInfo buildRegistrationRequest(
            CustomerServiceHistoryApiIntegrationRequest request) {
        return new CustomerServiceHistoryRegistrationRequest.CustomerServiceHistoryInfo(
                request.parameters().tgkuHtkFlg(),
                request.parameters().ybdssyId(),
                request.parameters().cstRegNo(),
                request.parameters().skkTnsNo(),
                request.parameters().skkHuhuShbtCd(),
                request.parameters().gymShsiKbnCd(),
                request.parameters().skkYmd(),
                request.parameters().skkJkk(),
                request.parameters().sysHukkSinNo(),
                request.parameters().sysHukkKshCd(),
                request.parameters().skkSttJkk(),
                request.parameters().skkTnsSnmKj(),
                request.parameters().skkTnKshCd(),
                request.parameters().sgySntkJkyCd(),
                request.parameters().sgyYtjsYmd(),
                request.parameters().sgyYtiJkkCd(),
                request.parameters().sgyYtiJktCd(),
                request.parameters().jkiHmysYmd(),
                request.parameters().jkiHmysJktCd(),
                request.parameters().jkiHmysJkkCd(),
                request.parameters().sgytKshCd(),
                request.parameters().sgttNo(),
                request.parameters().sgttSnmKj(),
                request.parameters().gmtSetNo(),
                request.parameters().sihKiyNo(),
                request.parameters().gasShyKiyNo(),
                request.parameters().tkjCmnt(),
                request.parameters().gymNiyuShbtCd(),
                request.parameters().utcdShbtCd(),
                request.parameters().utkNo(),
                request.parameters().utidxNo2());
    }
}
