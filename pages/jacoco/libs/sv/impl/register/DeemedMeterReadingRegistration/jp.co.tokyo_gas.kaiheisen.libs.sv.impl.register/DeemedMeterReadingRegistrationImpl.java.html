package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;

/**
 * みなし検針登録サービス実装。
 */
public class DeemedMeterReadingRegistrationImpl implements DeemedMeterReadingRegistration {

    static final String SERVICE_USER_ID = "KHSS015";

    private static final String DEEMED_METER_READING_REQUIRED = "1";

    private static final Integer DEEMED_METER_READING_LOOKUP_SEQUENCE = null;

    private static final Integer DEEMED_METER_READING_INSERT_SEQUENCE = 0;

    private final DktksryDataAccess dktksryDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String noAdjustmentRequired;

    private final String meterReadingWithoutTransferAcceptance;

    private final String indicatedNumberRegressionNone;

    private final String firstMeterReadingRequired;

    /**
     * コンストラクタ。
     *
     * @param dktksryDataAccess 導管通知確定使用量データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public DeemedMeterReadingRegistrationImpl(
            DktksryDataAccess dktksryDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.dktksryDataAccess = Objects.requireNonNull(dktksryDataAccess, "dktksryDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.noAdjustmentRequired = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.AdjustedUsageCalculationRequestStatus.TYPE,
                ConstantLookupKeys.Cirius.AdjustedUsageCalculationRequestStatus.NO_ADJUSTMENT_REQUIRED);
        this.meterReadingWithoutTransferAcceptance = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.METER_READING_WITHOUT_TRANSFER_ACCEPTANCE);
        this.indicatedNumberRegressionNone = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.TYPE,
                ConstantLookupKeys.Cirius.IndicatedNumberRegressionSegment.NONE);
        this.firstMeterReadingRequired = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.TYPE,
                ConstantLookupKeys.Cirius.FirstMeterReadingApplicationCode.INITIAL_METER_READING_REQUIRED);
    }

    @Override
        public void execute(
            String mnsKnsnYhCd,
            LocalDate knsYmd,
            Long gasMtstNo,
            Integer knsKkaTsno,
            Long svkyNo,
            String gmtKhsJtkb) {
        LocalDate resolvedKnsYmd = resolveKnsYmd(knsYmd);
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        if (DEEMED_METER_READING_REQUIRED.equals(mnsKnsnYhCd)) {
            List<DktksryRecord> currentRecords =
                    dktksryDataAccess.findAll(gasMtstNo, resolvedKnsYmd, DEEMED_METER_READING_LOOKUP_SEQUENCE);
            if (currentRecords.isEmpty()) {
                dktksryDataAccess.insert(
                        buildInsertRecord(gasMtstNo, resolvedKnsYmd, svkyNo, gmtKhsJtkb),
                        auditContext);
            } else {
                update(
                        gasMtstNo,
                        resolvedKnsYmd,
                        DEEMED_METER_READING_LOOKUP_SEQUENCE,
                        svkyNo,
                        auditContext);
            }
            return;
        }

        update(gasMtstNo, resolvedKnsYmd, knsKkaTsno, svkyNo, auditContext);
    }

    private void update(
            Long gasMtstNo,
            LocalDate knsYmd,
            Integer knsKkaTsno,
            Long svkyNo,
            AuditContext auditContext) {
        int updatedCount = dktksryDataAccess.updateByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                buildUpdateRecord(gasMtstNo, knsYmd, knsKkaTsno, svkyNo),
                Set.of(DktksryColumn.SVKY_NO),
                new DktksryUpdateKey(gasMtstNo, knsYmd, knsKkaTsno),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    dktksryDataAccess.tableName(),
                    DktksryDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", gasMtstNo)
                            .put("knsYmd", knsYmd)
                            .put("knsKkaTsno", knsKkaTsno)
                            .put("svkyNo", svkyNo)
                            .put("updUsrId", auditContext.userId())
                            .put("updKtnKinoId", auditContext.kinoId()));
        }
    }

    private DktksryWriteRecord buildInsertRecord(
            Long gasMtstNo,
            LocalDate knsYmd,
            Long svkyNo,
            String gmtKhsJtkb) {
        return new DktksryWriteRecord(
                gasMtstNo,
                knsYmd,
                DEEMED_METER_READING_INSERT_SEQUENCE,
                svkyNo,
                null,
                null,
                null,
                noAdjustmentRequired,
                null,
                null,
                null,
                meterReadingWithoutTransferAcceptance,
                null,
                null,
                null,
                indicatedNumberRegressionNone,
                firstMeterReadingRequired,
                gmtKhsJtkb,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private DktksryWriteRecord buildUpdateRecord(
            Long gasMtstNo,
            LocalDate knsYmd,
            Integer knsKkaTsno,
            Long svkyNo) {
        return new DktksryWriteRecord(
                gasMtstNo,
                knsYmd,
                knsKkaTsno,
                svkyNo,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private LocalDate resolveKnsYmd(LocalDate knsYmd) {
        return knsYmd != null ? knsYmd : requireSystemDatetime().toLocalDate();
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                                requestContext.getHukkSysShbt()
                                                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
