package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import jakarta.inject.Inject;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisen_torikeshiji_idoujouhou_touroku.KaisenTorikeshijiIdoujouhou;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisen_torikeshiji_idoujouhou_touroku.KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisen_torikeshiji_idoujouhou_touroku.KaisenTorikeshijiIdoujouhouTourokuRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.BillingCollectionPlatformApiIntegrationForOpeningCancel;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SkkbRkySgjhInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsKykDlirInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsKykHkirInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.SktsSihDlirInf;

/**
 * 概要: SV-196（請求回収基盤API連携（開栓取消））の実装。
 * 主な仕様:
 * - 4 つの入力モデルを `他-047` の 17 項目へ編集し、請求回収基盤（開栓取消）-API を 1 回呼び出す。
 * - `呼出しパラメータ` 補助資料の null 補正規則に従い、文字列は空文字、変更依頼の日付は 0 を設定する。
 * 既知の制約:
 * - `hukkSysShbt`、`correlationId`、`systemDatetime` は公開 API と実装のどちらでも利用しない。
 */
public final class BillingCollectionPlatformApiIntegrationForOpeningCancelImpl
        implements BillingCollectionPlatformApiIntegrationForOpeningCancel {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final int ZERO_DATE = 0;

    private final KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter adapter;

    /**
     * 概要: CDI から依存を受け取り初期化する。
     *
     * @param adapter 他-047 adapter
     */
    @Inject
    public BillingCollectionPlatformApiIntegrationForOpeningCancelImpl(
            KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter adapter) {
        this.adapter = Objects.requireNonNull(adapter, "adapter");
    }

    /**
     * 概要: 開栓取消時の請求回収基盤連携を実行する。
     */
    @Override
    public void execute(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihDlirInf sktsSihDlirInf,
            SktsKykDlirInf sktsKykDlirInf,
            SktsKykHkirInf sktsKykHkirInf) {
        adapter.register(buildRequest(skkbRkySgjhInf, sktsSihDlirInf, sktsKykDlirInf, sktsKykHkirInf));
    }

    private KaisenTorikeshijiIdoujouhouTourokuRequest buildRequest(
            SkkbRkySgjhInf skkbRkySgjhInf,
            SktsSihDlirInf sktsSihDlirInf,
            SktsKykDlirInf sktsKykDlirInf,
            SktsKykHkirInf sktsKykHkirInf) {
        return new KaisenTorikeshijiIdoujouhouTourokuRequest(
                new KaisenTorikeshijiIdoujouhou(
                        blankIfNull(skkbRkySgjhInf.skrkGrpKrno()),
                        skkbRkySgjhInf.sikKrno(),
                        skkbRkySgjhInf.sqtiKiyNo(),
                        skkbRkySgjhInf.svkyNo(),
                        skkbRkySgjhInf.svkySkno(),
                        skkbRkySgjhInf.sihKiyNo(),
                        skkbRkySgjhInf.sihKiySkno(),
                        skkbRkySgjhInf.gasMtstNo(),
                        toYmd(skkbRkySgjhInf.tkysYmd()),
                        toYmd(skkbRkySgjhInf.khsYmd()),
                        sktsSihDlirInf.kytuIsrrGymCd(),
                        sktsKykDlirInf.kytuIsrrGymCd(),
                        blankIfNull(sktsKykHkirInf.kytuIsrrGymCd()),
                        blankIfNull(sktsKykHkirInf.svkjyShbt()),
                        toYmdOrZero(sktsKykHkirInf.svtkYmd()),
                        toYmdOrZero(sktsKykHkirInf.kiyEndYmd()),
                        toYmdOrZero(sktsKykHkirInf.szirHtyoYmd())));
    }

    private String blankIfNull(String value) {
        return value == null ? "" : value;
    }

    private Integer toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        return Integer.valueOf(BASIC_DATE.format(value));
    }

    private Integer toYmdOrZero(LocalDate value) {
        if (value == null) {
            return ZERO_DATE;
        }
        return toYmd(value);
    }
}
