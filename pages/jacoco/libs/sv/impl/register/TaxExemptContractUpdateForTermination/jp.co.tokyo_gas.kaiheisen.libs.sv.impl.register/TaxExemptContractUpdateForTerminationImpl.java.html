package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.MnzkiyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mnzkiy.MnzkiyRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.TaxExemptContractAddUpdateDelete;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.TaxExemptContractUpdateForTermination;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.MnzkiyInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.TaxExemptContractSvckyaInf;

/**
 * 概要: SV-029（免税契約更新（契約終了））サービス実装。
 * 主な仕様:
 * - 消費税課税区分が免税の場合に、解約日時点で有効な免税契約情報を更新し、先日付分の免税契約情報を削除する。
 * - 更新と削除本体は SV-071 へ委譲する。
 * - 適用終了年月日が未指定の場合は RequestContext.systemDatetime から補完する。
 * 既知の制約:
 * - public メソッド引数の null 検証は追加しない。
 */
public class TaxExemptContractUpdateForTerminationImpl
        implements TaxExemptContractUpdateForTermination {

    private static final MnzkiyInf EMPTY_MNZKIY_INF = new MnzkiyInf(
            null,
            null,
            null,
            null,
            null,
            null,
            null,
            null);

    private final MnzkiyDataAccess mnzkiyDataAccess;

    private final TaxExemptContractAddUpdateDelete taxExemptContractAddUpdateDelete;

    private final RequestContext requestContext;

    private final String dutyFreeConsumptionTax;

    private final String alternativeEndProcessingDivision;

    private final String deleteProcessingDivision;

    /**
     * 概要: SV-029 の依存と必須定数を初期化する。
     * 主な仕様:
     * - MNZKIY データアクセス、SV-071、RequestContext、定数サービスを保持する。
     * - 免税判定と SV-071 呼出しに必要な定数の存在を初期化時に検証する。
     * 既知の制約:
     * - 依存が null の場合は初期化時点で NullPointerException を送出する。
     * - 必須定数が不足している場合はコンストラクタ初期化時点で失敗する。
     *
     * @param mnzkiyDataAccess MNZKIY データアクセス
     * @param taxExemptContractAddUpdateDelete SV-071
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public TaxExemptContractUpdateForTerminationImpl(
            MnzkiyDataAccess mnzkiyDataAccess,
            TaxExemptContractAddUpdateDelete taxExemptContractAddUpdateDelete,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.mnzkiyDataAccess = Objects.requireNonNull(mnzkiyDataAccess, "mnzkiyDataAccess");
        this.taxExemptContractAddUpdateDelete = Objects.requireNonNull(
                taxExemptContractAddUpdateDelete,
                "taxExemptContractAddUpdateDelete");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.dutyFreeConsumptionTax = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsumptionTax.TYPE,
                ConstantLookupKeys.Cirius.ConsumptionTax.DUTY_FREE);
        this.alternativeEndProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.ALTERNATIVE_END);
        this.deleteProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.DELETE);
    }

    /**
     * 概要: 免税契約更新（契約終了）を実行する。
     * 主な仕様:
     * - 消費税課税区分が免税以外の場合は何もせず正常終了する。
     * - 解約日時点で有効な免税契約情報が存在する場合は代替終了更新を実行する。
     * - 先日付分の免税契約情報が存在する場合は削除を実行する。
     * - 適用終了年月日が未指定の場合は RequestContext.systemDatetime から補完する。
     *
     * @param tyenYmd 適用終了年月日
     * @param svckyaInf サービス契約情報
     * @param sziKziKbn 消費税課税区分
     */
    @Override
    public void execute(
            LocalDate tyenYmd,
            TaxExemptContractSvckyaInf svckyaInf,
            String sziKziKbn) {
        if (!dutyFreeConsumptionTax.equals(sziKziKbn)) {
            return;
        }

        LocalDate effectiveEndDate = resolveEffectiveEndDate(tyenYmd);
        Long svkyNo = svckyaInf.svkyNo();

        mnzkiyDataAccess.findBySvkyNoAndMnziTkysYmdAndMnziTyenYmd(svkyNo, effectiveEndDate)
                .ifPresent(record -> executeUpdate(svkyNo, record, effectiveEndDate));
        mnzkiyDataAccess.findBySvkyNoAndMnziTkysYmdGreaterThan(svkyNo, effectiveEndDate)
                .ifPresent(record -> executeDelete(record));
    }

    private void executeUpdate(Long svkyNo, MnzkiyRecord record, LocalDate effectiveEndDate) {
        taxExemptContractAddUpdateDelete.execute(
                alternativeEndProcessingDivision,
                svkyNo,
                record.mnziTkysYmd(),
                effectiveEndDate,
                EMPTY_MNZKIY_INF);
    }

    private void executeDelete(MnzkiyRecord record) {
        taxExemptContractAddUpdateDelete.execute(
                deleteProcessingDivision,
                record.svkyNo(),
                record.mnziTkysYmd(),
                null,
                EMPTY_MNZKIY_INF);
    }

    /**
     * 概要: 実際に使用する適用終了年月日を解決する。
     * 主な仕様:
     * - 引数の適用終了年月日が指定されている場合はその値を返却する。
     * - 未指定の場合は RequestContext.systemDatetime の年月日を返却する。
     * - systemDatetime が未設定の場合は IllegalStateException を送出する。
     *
     * @param tyenYmd 適用終了年月日
     * @return 解決した適用終了年月日
     */
    private LocalDate resolveEffectiveEndDate(LocalDate tyenYmd) {
        if (tyenYmd != null) {
            return tyenYmd;
        }
        return requireSystemDatetime().toLocalDate();
    }

    private java.time.LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
}
