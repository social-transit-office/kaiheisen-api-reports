package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Set;

/**
 * 概要: SV-071（免税契約の追加・更新・削除）のサービス実装。
 * 主な仕様:
 * - 免税契約情報処理区分に応じて MNZKIY の登録、更新、削除を切り替える。
 * - 更新と削除では事前に対象レコードを取得し、0 件の場合は処理対象データ無しエラーを送出する。
 * - 更新結果または削除結果が 0 件の場合はデータ不整合エラーへ変換する。
 * 既知の制約:
 * - RequestContext の値設定と妥当性確認は上位層責務とし、本クラスでは補完や追加検証を行わない。
 * - public メソッド引数の null 検証は追加しない。
 */
public class TaxExemptContractAddUpdateDeleteImpl implements TaxExemptContractAddUpdateDelete {

    static final String SERVICE_USER_ID = "KHSS071";

    private final MnzkiyDataAccess mnzkiyDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String processingDivisionGenerate;

    private final String processingDivisionAlternativeEnd;

    private final String processingDivisionDelete;

    /**
     * 概要: 免税契約の追加・更新・削除サービスの依存と必須定数を初期化する。
     * 主な仕様:
     * - MNZKIY データアクセス、RequestContext、開閉栓定数サービスを保持する。
     * - 処理区分判定に必要な KH002 定数の存在を初期化時に検証する。
     * 既知の制約:
     * - 依存が null の場合は初期化時点で NullPointerException を送出する。
     * - 必須定数が不足している場合はコンストラクタ初期化時点で失敗する。
     * @param mnzkiyDataAccess MNZKIY データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public TaxExemptContractAddUpdateDeleteImpl(
            MnzkiyDataAccess mnzkiyDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.mnzkiyDataAccess = Objects.requireNonNull(mnzkiyDataAccess, "mnzkiyDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        this.processingDivisionGenerate = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.GENERATE);
        this.processingDivisionAlternativeEnd = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.ALTERNATIVE_END);
        this.processingDivisionDelete = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.TaxExemptionContractInformationProcessing.DELETE);
    }

    /**
     * 概要: 免税契約の追加・更新・削除を実行する。
     * 主な仕様:
        * - RequestContext の値をそのまま利用して更新日付と監査情報を組み立てる。
     * - 生成は登録、代替終了は更新、削除は物理削除を実行する。
     * - 対応外の処理区分では何もせず正常終了する。
     * 既知の制約:
     * - 免税契約情報処理区分は KH002 定数値との一致で判定する。
     * - 生成分岐では入力の免税契約情報を登録レコードへそのまま移送する。
     * @param mnzkiyInfSyoriKbn 免税契約情報処理区分
     * @param svkyNo サービス契約番号
     * @param tkysYmd 適用開始年月日
     * @param tyenYmd 適用終了年月日
     * @param mnzkiyInf 免税契約情報
     */
    @Override
    public void execute(
            String mnzkiyInfSyoriKbn,
            Long svkyNo,
            LocalDate tkysYmd,
            LocalDate tyenYmd,
            MnzkiyInf mnzkiyInf) {
        if (processingDivisionGenerate.equals(mnzkiyInfSyoriKbn)) {
            insertRecord(svkyNo, tkysYmd, mnzkiyInf);
            return;
        }
        if (processingDivisionAlternativeEnd.equals(mnzkiyInfSyoriKbn)) {
            updateRecord(svkyNo, tkysYmd, tyenYmd);
            return;
        }
        if (processingDivisionDelete.equals(mnzkiyInfSyoriKbn)) {
            deleteRecord(svkyNo, tkysYmd);
            return;
        }
    }

    /**
     * 概要: 免税契約を新規登録する。
     * 主な仕様:
     * - 入力値を MNZKIY の write レコードへ移送する。
     * - 監査項目は固定更新ユーザー ID と報告元機能 ID で補完する。
     * @param svkyNo サービス契約番号
     * @param tkysYmd 適用開始年月日
     * @param mnzkiyInf 免税契約情報
     */
    private void insertRecord(
            Long svkyNo,
            LocalDate tkysYmd,
            MnzkiyInf mnzkiyInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        mnzkiyDataAccess.insert(
                new MnzkiyWriteRecord(
                        svkyNo,
                        tkysYmd,
                        mnzkiyInf.mnziTyenYmd(),
                        mnzkiyInf.mnziTyehYmd(),
                        mnzkiyInf.sseTiskNo(),
                        mnzkiyInf.mnziKysjYmd(),
                        mnzkiyInf.mnziTysjYmd(),
                        mnzkiyInf.mnziHksjYmd(),
                        mnzkiyInf.mzkhHjonKj(),
                        mnzkiyInf.mzkhHjonKn(),
                        null),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    /**
     * 概要: 免税契約の適用終了情報を更新する。
     * 主な仕様:
     * - 事前に対象レコードを取得し、0 件なら処理対象データ無しエラーを送出する。
     * - 更新対象は免税適用終了年月日と免税適用終了報告年月日に限定する。
     * - 更新件数 0 件はデータ不整合エラーへ変換する。
     * @param svkyNo サービス契約番号
     * @param tkysYmd 適用開始年月日
     * @param tyenYmd 適用終了年月日
     */
    private void updateRecord(
            Long svkyNo,
            LocalDate tkysYmd,
            LocalDate tyenYmd) {
        MnzkiyRecord existingRecord = findExistingRecord(svkyNo, tkysYmd);
        String decodedOriginFunctionId = decodeOriginFunctionId();
        LocalDate currentDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
        int updatedCount = mnzkiyDataAccess.updateBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
                new MnzkiyWriteRecord(
                        svkyNo,
                        tkysYmd,
                        tyenYmd,
                        currentDate,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        existingRecord.recUpdJnjNo()),
                Set.of(MnzkiyColumn.MNZI_TYEN_YMD, MnzkiyColumn.MNZI_TYEH_YMD),
                new MnzkiySvkyTkysUpdateKey(svkyNo, tkysYmd, existingRecord.recUpdJnjNo()),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    mnzkiyDataAccess.tableName(),
                    MnzkiyDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO_U001
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svkyNo)
                            .put("mnziTkysYmd", tkysYmd)
                            .put("recUpdJnjNo", existingRecord.recUpdJnjNo())
                            .put("mnziTyenYmd", tyenYmd)
                            .put("mnziTyehYmd", currentDate)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    /**
     * 概要: 免税契約を物理削除する。
     * 主な仕様:
     * - 事前に対象レコードを取得し、0 件なら処理対象データ無しエラーを送出する。
     * - 削除件数 0 件はデータ不整合エラーへ変換する。
     * @param svkyNo サービス契約番号
     * @param tkysYmd 適用開始年月日
     */
    private void deleteRecord(
            Long svkyNo,
            LocalDate tkysYmd) {
        MnzkiyRecord existingRecord = findExistingRecord(svkyNo, tkysYmd);
        int deletedCount = mnzkiyDataAccess.deleteBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
                svkyNo,
                tkysYmd,
                existingRecord.recUpdJnjNo());
        if (deletedCount == 0) {
            throw DataInconsistencySupport.create(
                    mnzkiyDataAccess.tableName(),
                    MnzkiyDataAccess.CallSpecs.DELETE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svkyNo)
                            .put("mnziTkysYmd", tkysYmd)
                            .put("recUpdJnjNo", existingRecord.recUpdJnjNo()));
        }
    }

    /**
     * 概要: 更新・削除対象の免税契約を取得する。
     * 主な仕様:
     * - サービス契約番号と適用開始年月日で対象レコードを取得する。
     * - 取得結果が 0 件の場合は処理対象データ無しエラーを送出する。
     * @param svkyNo サービス契約番号
     * @param tkysYmd 適用開始年月日
     * @return 取得した免税契約レコード
     */
    private MnzkiyRecord findExistingRecord(Long svkyNo, LocalDate tkysYmd) {
        return mnzkiyDataAccess.findBySvkyNoAndMnziTkysYmd(svkyNo, tkysYmd)
                .orElseThrow(() -> new DataNotFoundException(mnzkiyDataAccess.tableName()));
    }

    /**
     * 概要: RequestContext の報告システム種別から監査用の機能 ID を取得する。
     * 主な仕様:
     * - RequestContext から hukkSysShbt を取得し、開閉栓定数サービスで名称へ変換する。
     * - 返却値は監査情報の updKtnKinoId に設定する。
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                                requestContext.getHukkSysShbt()
                                                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
