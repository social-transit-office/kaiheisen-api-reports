package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaiheisen_change_notification.KaiheisenChangeNotificationCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaiheisen_change_notification.KaiheisenChangeNotificationRequest;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.ReceptionInfoChangeNotificationApiIntegration;

/**
 * 概要: SV-092（受付情報変更通知API連携）のサービス実装。
 * 主な仕様:
 * - SV 入力を CIRIUS 開閉栓変更通知登録 API の request へマッピングして連携する。
 * - 設計書に記載された必須定数の存在をコンストラクタ初期化時点で検証する。
 * 既知の制約:
 * - CIRIUS 連携結果は戻り値へ返さず、異常時は下位例外をそのまま上位へ伝播する。
 */
public class ReceptionInfoChangeNotificationApiIntegrationImpl
        implements ReceptionInfoChangeNotificationApiIntegration {

    static final LocalDate CIRIUS_ZERO_DATE = LocalDate.of(1900, 1, 1);

    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private final KaiheisenChangeNotificationCiriusAdapter adapter;

    private final String tgRetailFlag;

    private final String gasMerchandiseWorkType;

    private final String presenceFlagYes;

    private final String presenceFlagNothing;

    private final String customerRegistrationNumberKeyType;

    private final String gasServiceContractType;

    private final String gasMeterInstallationLocationNumberKeyType;

    /**
     * 概要: SV-092 実行に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - adapter を保持する。
     * - 受付情報変更通知 API 連携で使用する必須定数を解決する。
     *
     * @param adapter 開閉栓変更通知登録 API adapter
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public ReceptionInfoChangeNotificationApiIntegrationImpl(
            KaiheisenChangeNotificationCiriusAdapter adapter,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.adapter = Objects.requireNonNull(adapter, "adapter");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.tgRetailFlag = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.TYPE,
                ConstantLookupKeys.Cirius.TgRetailJudgmentResultFlag.TG_RETAIL);
        this.gasMerchandiseWorkType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MerchandiseWorkType.TYPE,
                ConstantLookupKeys.Cirius.MerchandiseWorkType.GAS);
        this.presenceFlagYes = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.YES);
        this.presenceFlagNothing = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.NOTHING);
        this.customerRegistrationNumberKeyType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE_OF_CUSTOMER_REGISTRATION_NUMBER);
        this.gasServiceContractType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ServiceContractType.TYPE,
                ConstantLookupKeys.Kaiheisen.ServiceContractType.GAS);
        this.gasMeterInstallationLocationNumberKeyType = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE,
                ConstantLookupKeys.Kaiheisen.KeyType.TYPE_OF_GAS_METER_INSTALLATION_LOCATION_NUMBER);
    }

    /**
     * 概要: 受付情報変更通知 API 連携を実行する。
     * 主な仕様:
     * - SV 入力を CIRIUS request へ詰め替えて adapter に引き渡す。
     * - CIRIUS のレスポンスは業務上返却せず、正常終了のみを扱う。
     */
    @Override
    public void execute(
            String utidxNo,
            String uketkSeiriNo,
            Long svkyNo,
            String ttzh,
            Long gasMtstNo,
            Long cstRegNo,
            String gymNyoSybtCd,
            String nwknKj,
            String sagyoSyaName,
            String mkSbt,
            LocalDate hssgYmd,
            LocalDate kytktYmd,
            Boolean mghkFlg,
            String kkkSbt1) {
        adapter.registerNotification(buildRequest(
                utidxNo,
                uketkSeiriNo,
                svkyNo,
                ttzh,
                gasMtstNo,
                cstRegNo,
                gymNyoSybtCd,
                nwknKj,
                sagyoSyaName,
                mkSbt,
                hssgYmd,
                kytktYmd,
                mghkFlg,
                kkkSbt1));
    }

    private KaiheisenChangeNotificationRequest.NotificationInfo buildRequest(
            String utidxNo,
            String uketkSeiriNo,
            Long svkyNo,
            String ttzh,
            Long gasMtstNo,
            Long cstRegNo,
            String gymNyoSybtCd,
            String nwknKj,
            String sagyoSyaName,
            String mkSbt,
            LocalDate hssgYmd,
            LocalDate kytktYmd,
            Boolean mghkFlg,
            String kkkSbt1) {
        return new KaiheisenChangeNotificationRequest.NotificationInfo(
                tgRetailFlag,
                utidxNo,
                gasMerchandiseWorkType,
                mkSbt,
                uketkSeiriNo,
                toBigDecimal(svkyNo),
                ttzh,
                toBigDecimal(gasMtstNo),
                toBigDecimal(cstRegNo),
                toBigDecimal(svkyNo),
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                gymNyoSybtCd,
                null,
                null,
                nwknKj,
                sagyoSyaName,
                toYmd(kytktYmd),
                toPresenceFlag(mghkFlg),
                customerRegistrationNumberKeyType,
                toBigDecimal(cstRegNo),
                gasServiceContractType,
                toYmd(hssgYmd),
                kkkSbt1,
                toBigDecimal(svkyNo),
                gasMeterInstallationLocationNumberKeyType,
                toBigDecimal(gasMtstNo));
    }

    private String toPresenceFlag(Boolean value) {
        return Boolean.TRUE.equals(value) ? presenceFlagYes : presenceFlagNothing;
    }

    private static BigDecimal toBigDecimal(Long value) {
        return value == null ? null : BigDecimal.valueOf(value);
    }

    private Integer toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        if (CIRIUS_ZERO_DATE.equals(value)) {
            return 0;
        }
        return Integer.valueOf(DATE_FORMATTER.format(value));
    }
}
