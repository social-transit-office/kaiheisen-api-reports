/**
 * 概要: SV-156（閉栓用解約共通処理）のサービス実装。
 * 主な仕様:
 * - 閉栓事由種別を下位 SV 呼出し用の終了事由種別へ変換する。
 * - 異動・作業履歴業務コードは引数値をそのまま下位 SV へ引き渡す。
 * - 託送供給依頼事業者コードと請求サービス種別に応じて、SV-326 / SV-328 / SV-329 相当の処理を切り替える。
 * - 下位呼出しに必要な `correlationId`、`hukkSysShbt`、`systemDatetime` は RequestContext で保持される前提とする。
 * 既知の制約:
 * - public メソッド引数の null / blank 妥当性は上位層責務とし、本実装では追加検証しない。
 */

    /**
     * 概要: 閉栓用解約共通処理の依存オブジェクトを初期化する。
     * 主な仕様:
     * - SV-326 / SV-328 / SV-329 の API interface と RequestContext を保持する。
     * - 設計書の `【】` 囲いで表現された CIRIUS 定数は CiriusConstantService で解決する。
     * - 依存オブジェクトの null はコンストラクタ初期化時点で fail-fast とする。
     *
     * @param sv326Process SV-326 API
     * @param sv328Process SV-328 API
     * @param sv329Process SV-329 API
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param requestContext リクエストコンテキスト
     */
    @SuppressWarnings("checkstyle:HiddenField")
    public ClosingCancellationCommonProcessImpl(
            final CancellationCommonProcessBatchCommonTgRetail sv326Process,
            final CancellationCommonProcessCommonOtherTgRetail sv328Process,
            final CancellationCommonProcessCommonOtherRetail sv329Process,
            final CiriusConstantService ciriusConstantService,
            final RequestContext requestContext) {
        this.sv326Process = Objects.requireNonNull(sv326Process, "sv326Process");
        this.sv328Process = Objects.requireNonNull(sv328Process, "sv328Process");
        this.sv329Process = Objects.requireNonNull(sv329Process, "sv329Process");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.transferClosingReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE);
        this.burnLostClosingReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.FIRED_AND_LOST_CAP);
        this.unusedClosingReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.UNUSED_CLOSURE);
        this.transportTransferCancellationReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius
                        .TypeOfReasonForTerminationOfTransportationServiceContract.TYPE,
                ConstantLookupKeys.Cirius
                        .TypeOfReasonForTerminationOfTransportationServiceContract
                        .TRANSFER_CANCELLATION);
        this.transportUnusedCancellationReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius
                        .TypeOfReasonForTerminationOfTransportationServiceContract.TYPE,
                ConstantLookupKeys.Cirius
                        .TypeOfReasonForTerminationOfTransportationServiceContract
                        .NON_USE_CANCELLATION);
        this.serviceTransferCancellationReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                        .TRANSFER_CANCELLATION);
        this.serviceUnusedCancellationReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                        .NON_USE_CANCELLATION);
        this.tokyoGasCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RetailerCode.TYPE,
                ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD);

    /**
     * 閉栓用解約共通処理の呼出し先を選択して必要な入力を引き渡す。
     *
     * <p>主な仕様:
     * - 閉栓事由種別から終了事由種別を導出する。
     * - 異動・作業履歴業務コードは引数値をそのまま下位 SV へ引き渡す。
     * - 託送供給依頼事業者コードと請求サービス種別に応じて 1 つの dispatch 先のみを呼び出す。
     */
    @Override
    @SuppressWarnings("checkstyle:ParameterNumber")
    public void execute(
            final Long gasMtstNo,
            final LocalDate dgrpHsenYmd,
            final String hesnJysb,
            final CancellationGmtsetdInf gmtsetdInf,
            final TskgmstInf tskgmstInf,
            final TkssvkyInf tkssvkyInf,
            final SvmstkrInf svmstkrInf,
            final SvckyaInf svckyaInf,
            final SvsihkrInf svsihkrInf,
            final SihakyInf sihakyInf,
            final String tkykJtCd,
            final TaksmtrInf taksmtrInf,
            final List<SnyInf> snyInfList,
            final List<SogoTmtrInf> sgoTmtrInfList,
            final String isrrGymCd,
            final String hukkUsrId,
            final String hukkUnmKj,
            final String isrrHkkrKbn) {
        requireRequestContext();
        TerminationReasonMapping terminationReasonMapping = mapTerminationReasons(hesnJysb);
        TaksmtrInf normalizedTaksmtrInf = normalizeTaksmtrInf(tkykJtCd, taksmtrInf);
        List<SnyInf> copiedSnyInfList = copyList(snyInfList);
        List<SogoTmtrInf> normalizedSgoTmtrInfList = normalizeSgoTmtrInfList(
                sgoTmtrInfList);

        if (tokyoGasCode.equals(tkssvkyInf.tksuKijsCd())) {
            if (isBatchServiceType(sihakyInf.sikSvcShbt())) {
                executeSv326(
                        gasMtstNo,
                        dgrpHsenYmd,
                        hesnJysb,
                        terminationReasonMapping,
                        gmtsetdInf,
                        tskgmstInf,
                        tkssvkyInf,
                        svmstkrInf,
                        svckyaInf,
                        svsihkrInf,
                        sihakyInf,
                        copiedSnyInfList,
                        normalizedSgoTmtrInfList,
                        isrrGymCd,
                        hukkUsrId,
                        hukkUnmKj,
                        isrrHkkrKbn);
                return;
            }

            sv328Process.execute(
                    gasMtstNo,
                    dgrpHsenYmd,
                    hesnJysb,
                    terminationReasonMapping.tksuSvkeJysb(),
                    terminationReasonMapping.svkyEndJysb(),
                    gmtsetdInf,
                    tskgmstInf,
                    tkssvkyInf,
                    svmstkrInf,
                    svckyaInf,
                    svsihkrInf,
                    sihakyInf,
                    copiedSnyInfList,
                    normalizedSgoTmtrInfList,
                    isrrGymCd,
                    hukkUsrId,
                    hukkUnmKj,
                    isrrHkkrKbn);
            return;
        }

        sv329Process.execute(
                gasMtstNo,
                dgrpHsenYmd,
                terminationReasonMapping.tksuSvkeJysb(),
                toSv329GmtsetdInf(gmtsetdInf),
                toSv329TskgmstInf(tskgmstInf),
                toSv329TkssvkyInf(tkssvkyInf),
                toSv329SnyInfList(copiedSnyInfList),
                toSv329TaksmtrInf(normalizedTaksmtrInf));
    }

    private TerminationReasonMapping mapTerminationReasons(final String hesnJysb) {
        if (transferClosingReason.equals(hesnJysb)
                || burnLostClosingReason.equals(hesnJysb)) {
            return new TerminationReasonMapping(
                    transportTransferCancellationReason,
                    serviceTransferCancellationReason);
        }
        if (unusedClosingReason.equals(hesnJysb)) {
            return new TerminationReasonMapping(
                    transportUnusedCancellationReason,
                    serviceUnusedCancellationReason);
        }
        return new TerminationReasonMapping(null, null);
    }

    private boolean isBatchServiceType(final String sikSvcShbt) {
        return batchFinalCloseServiceType.equals(sikSvcShbt)
                || batchNextThreeTemporaryServiceType.equals(sikSvcShbt);
    }

    private TaksmtrInf normalizeTaksmtrInf(
            final String tkykJtCd,
            final TaksmtrInf taksmtrInf) {
        if (temporaryCancellationStatus.equals(tkykJtCd)) {
            return taksmtrInf;
        }
        return new TaksmtrInf(null, null, null);
    }

    private List<SogoTmtrInf> normalizeSgoTmtrInfList(
            final List<SogoTmtrInf> sgoTmtrInfList) {
        if (sgoTmtrInfList == null) {
            return null;
        }
        return sgoTmtrInfList.stream()
                .map(this::normalizeSgoTmtrInf)
                .toList();
    }

    private SogoTmtrInf normalizeSgoTmtrInf(final SogoTmtrInf sgoTmtrInf) {
        if (relatedEndStatus.equals(sgoTmtrInf.rkkiKnrnJtkb())) {
            return new SogoTmtrInf(
                    null,
                    null,
                    sgoTmtrInf.rkkiKnrnJtkb(),
                    null,
                    null,
                    null);
        }
        return sgoTmtrInf;
    }

    private <T> List<T> copyList(final List<T> values) {
        if (values == null) {
            return null;
        }
        return List.copyOf(values);
    }

    private void requireRequestContext() {
        requestContext.getCorrelationId()
                .orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(
                        () -> new IllegalStateException(
                                "RequestContext.systemDatetime is required"));
    }
