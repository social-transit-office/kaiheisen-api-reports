/**
 * 概要: SV-072（サービス契約、供給停止管理更新（供給停止精算不要データ有））のサービス実装。
 * 主な仕様:
 * - 総合判定フラグと GHP専用4x有無フラグの組み合わせに応じて、サービス契約更新、総合扱いガスメーター走査、または処理なしを切り替える。
 * - 総合扱いガスメーター走査中に開栓中メーターを見つけた場合は `ksenFlg=true` を返す。
 * - `DA-033-R006` 0 件、`DA-038-U001` 0 件、`DA-055-U001` 0 件は `DataInconsistencyException` を送出する。
 */
public final class ServiceContractSupplySuspensionUpdateWithNoSettlementImpl
        implements ServiceContractSupplySuspensionUpdateWithNoSettlement {

    static final String SERVICE_USER_ID = "KHSS072";

    private final SvckyaDataAccess svckyaDataAccess;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final KkytskrDataAccess kkytskrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String supplyStopPaymentPendingStatus;

    private final String openingGasMeterStatus;

    /**
     * 必須依存と必須定数を初期化する。
     *
     * @param svckyaDataAccess サービス契約 DA
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞ DA
     * @param kkytskrDataAccess 供給停止管理 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public ServiceContractSupplySuspensionUpdateWithNoSettlementImpl(
            SvckyaDataAccess svckyaDataAccess,
            GmtsetdDataAccess gmtsetdDataAccess,
            KkytskrDataAccess kkytskrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccess, "svckyaDataAccess");
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.supplyStopPaymentPendingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
        this.openingGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
    }

    @Override
    public ServiceContractSupplySuspensionUpdateWithNoSettlementOutput execute(
            Boolean suguHntFlg,
            Boolean sny4XUmFlg,
            SvckyaInf svckyaInf,
            List<SuguGmtInf> suguGmtInfList,
            LocalDate trkKnsYmd,
            KkytskrInf kkytskrInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        boolean ksenFlg = false;
        if (Boolean.FALSE.equals(suguHntFlg)) {
            if (shouldExecuteFirstServiceContractUpdate(sny4XUmFlg)) {
                updateServiceContractSupplyStopPaymentStatus(
                        svckyaInf,
                        auditContext,
                        decodedOriginFunctionId);
            }
        } else {
            for (SuguGmtInf suguGmtInf : suguGmtInfList) {
                GmtsetdRecord gasMeter = findGasMeterOrThrow(suguGmtInf, trkKnsYmd);
                if (openingGasMeterStatus.equals(gasMeter.gmtKhsJtkb())) {
                    ksenFlg = true;
                    break;
                }
            }
            if (!ksenFlg) {
                updateServiceContractSupplyStopPaymentStatus(
                        svckyaInf,
                        auditContext,
                        decodedOriginFunctionId);
            }
        }

        updateSupplySuspensionManagement(kkytskrInf, auditContext, decodedOriginFunctionId);
        return new ServiceContractSupplySuspensionUpdateWithNoSettlementOutput(ksenFlg);
    }

    private boolean shouldExecuteFirstServiceContractUpdate(Boolean sny4XUmFlg) {
        return Boolean.FALSE.equals(sny4XUmFlg);
    }

    private GmtsetdRecord findGasMeterOrThrow(SuguGmtInf suguGmtInf, LocalDate trkKnsYmd) {
        Long gasMtstNo = suguGmtInf.gasMtstNo();
        return gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, trkKnsYmd)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        gmtsetdDataAccess.tableName(),
                        GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYmd", trkKnsYmd)));
    }

    private void updateServiceContractSupplyStopPaymentStatus(
            SvckyaInf svckyaInf,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = svckyaDataAccess.updateBySvkyNoAndSkno(
                buildSvckyaWriteRecord(),
                EnumSet.of(SvckyaColumn.KKYS_SISN_JTKB),
                new SvckyaSknoUpdateKey(svckyaInf.svkyNo(), svckyaInf.skno()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svckyaInf.svkyNo())
                            .put("skno", svckyaInf.skno())
                            .put("kkysSisnJtkb", supplyStopPaymentPendingStatus)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private void updateSupplySuspensionManagement(
            KkytskrInf kkytskrInf,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = kkytskrDataAccess.updateByGasMtstNoAndHesnYmd(
                buildKkytskrWriteRecord(),
                EnumSet.of(KkytskrColumn.KKYS_SISN_JTKB, KkytskrColumn.KAS_YMD),
                new KkytskrHesnUpdateKey(kkytskrInf.gasMtstNo(), kkytskrInf.hesnYmd()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kkytskrDataAccess.tableName(),
                    KkytskrDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", kkytskrInf.gasMtstNo())
                            .put("hesnYmd", kkytskrInf.hesnYmd())
                            .put("kkysSisnJtkb", supplyStopPaymentPendingStatus)
                            .put("kasYmd", null)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

                supplyStopPaymentPendingStatus,
                null,
                null,
                null,
                null);
    }

    private KkytskrWriteRecord buildKkytskrWriteRecord() {
        return new KkytskrWriteRecord(
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                supplyStopPaymentPendingStatus,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }
