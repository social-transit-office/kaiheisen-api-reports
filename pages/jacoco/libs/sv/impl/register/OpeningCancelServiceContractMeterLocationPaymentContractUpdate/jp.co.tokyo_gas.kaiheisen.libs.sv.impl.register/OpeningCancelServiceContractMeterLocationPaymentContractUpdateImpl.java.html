/**
 * 概要: SV-060（開栓取消時サービス契約・メーター設置場所、サービス契約・支払契約関連、支払契約関連更新）のサービス実装。
 */
public final class OpeningCancelServiceContractMeterLocationPaymentContractUpdateImpl
        implements OpeningCancelServiceContractMeterLocationPaymentContractUpdate {

    static final String SERVICE_USER_ID = "KHSS060";

    private final OpeningCancelServiceContractMeterLocationUpdate sv059;

    private final PaymentContractInfoUpdateForTermination sv032;

    private final InvoiceDisplayControlUpdateForTermination sv041;

    private final PaymentContractInvalidUpdate sv064;

    private final SvsihkrDataAccess svsihkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String applicableEndsPaymentContractStatus;

    private final String invalidPaymentContractStatus;

    private final String continuationOfPreviousContractStartReason;

    /**
     * 必須依存と必須定数を初期化する。
     *
     * @param sv059Dependency SV-059
     * @param sv032Dependency SV-032
     * @param sv041Dependency SV-041
     * @param sv064Dependency SV-064
     * @param svsihkrDataAccessDependency サービス契約・支払契約関連 DA
     * @param requestContextDependency リクエストコンテキスト
     * @param kaiheisenConstantServiceDependency 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public OpeningCancelServiceContractMeterLocationPaymentContractUpdateImpl(
            final OpeningCancelServiceContractMeterLocationUpdate sv059Dependency,
            final PaymentContractInfoUpdateForTermination sv032Dependency,
            final InvoiceDisplayControlUpdateForTermination sv041Dependency,
            final PaymentContractInvalidUpdate sv064Dependency,
            final SvsihkrDataAccess svsihkrDataAccessDependency,
            final RequestContext requestContextDependency,
            final KaiheisenConstantService kaiheisenConstantServiceDependency,
            final CiriusConstantService ciriusConstantService) {
        this.sv059 = Objects.requireNonNull(sv059Dependency, "sv059");
        this.sv032 = Objects.requireNonNull(sv032Dependency, "sv032");
        this.sv041 = Objects.requireNonNull(sv041Dependency, "sv041");
        this.sv064 = Objects.requireNonNull(sv064Dependency, "sv064");
        this.svsihkrDataAccess = Objects.requireNonNull(svsihkrDataAccessDependency, "svsihkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContextDependency, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantServiceDependency,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.applicableEndsPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLICABLE_ENDS);
        this.invalidPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
        this.continuationOfPreviousContractStartReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.CONTINUATION_OF_PREVIOUS_CONTRACT);
    }

    @Override
    public void execute(
            final SvckyaInf svckyaInf,
            final SihakyInf sihakyInf,
            final SihakyKaTrksInf sihakyKaTrksInf,
            final SvmstkrInf svmstkrInf,
            final SvsihkrInf svsihkrInf,
            final LocalDate kasYmd) {
        LocalDate systemDate = requireSystemDatetime().toLocalDate();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        sv059.execute(
                svmstkrInf.mtstNo(),
                svmstkrInf.svkyNo(),
                svmstkrInf.skno(),
                svmstkrInf.sknrFlg(),
                svckyaInf.svkySttJysb());

        boolean shouldApplyEndingUpdate = isPreviousContractContinuation(svckyaInf)
                && !Objects.equals(svckyaInf.rkkiSttYmd(), svsihkrInf.tkysYmd());
        updateServicePaymentContract(
                svsihkrInf,
                shouldApplyEndingUpdate ? applicableEndsPaymentContractStatus : invalidPaymentContractStatus,
                shouldApplyEndingUpdate ? systemDate : null,
                auditContext,
                decodedOriginFunctionId);

        boolean shouldTerminatePaymentContract = isPreviousContractContinuation(svckyaInf)
                && !Objects.equals(svckyaInf.rkkiSttYmd(), sihakyKaTrksInf.tkysYmd());
        if (shouldTerminatePaymentContract) {
            sv032.execute(sihakyInf.sihKiyNo(), sihakyInf.skno(), sihakyInf.sihHuhuShbt(), null);
            sv041.execute(sihakyInf.skshHyjsKrno());
            return;
        }
        sv064.execute(sihakyInf.sihKiyNo(), kasYmd);
    }

    private boolean isPreviousContractContinuation(final SvckyaInf svckyaInf) {
        return continuationOfPreviousContractStartReason.equals(svckyaInf.svkySttJysb());
    }

    private void updateServicePaymentContract(
            final SvsihkrInf svsihkrInf,
            final String paymentContractStatus,
            final LocalDate tyenYmd,
            final AuditContext auditContext,
            final String decodedOriginFunctionId) {
        Set<SvsihkrColumn> targetColumns = tyenYmd == null
                ? Set.of(SvsihkrColumn.SIH_KIY_JTKB)
                : Set.of(SvsihkrColumn.TYEN_YMD, SvsihkrColumn.SIH_KIY_JTKB);
        int updatedCount = svsihkrDataAccess.updateBySihKiyNoAndSvkyNoAndSkno(
                new SvsihkrWriteRecord(
                        svsihkrInf.sihKiyNo(),
                        svsihkrInf.svkyNo(),
                        svsihkrInf.skno(),
                        null,
                        tyenYmd,
                        paymentContractStatus,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                targetColumns,
                new SvsihkrSknoOptionalUpdateKey(
                        svsihkrInf.sihKiyNo(),
                        svsihkrInf.svkyNo(),
                        svsihkrInf.skno(),
                        null),
                auditContext);
        if (updatedCount == 0) {
            FlatJsonBuilder parameterBuilder = new FlatJsonBuilder()
                    .put("sihKiyNo", svsihkrInf.sihKiyNo())
                    .put("svkyNo", svsihkrInf.svkyNo())
                    .put("skno", svsihkrInf.skno())
                    .put("sihKiyJtkb", paymentContractStatus)
                    .put("updUsrId", SERVICE_USER_ID)
                    .put("updKtnKinoId", decodedOriginFunctionId);
            if (tyenYmd != null) {
                parameterBuilder.put("tyenYmd", tyenYmd);
            }
            throw DataInconsistencySupport.create(
                    svsihkrDataAccess.tableName(),
                    UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO.identifier(),
                    parameterBuilder);
        }
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }
}
