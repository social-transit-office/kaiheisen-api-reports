/**
 * 概要: SV-307（移転解約時の移転先管理追加）のサービス実装。
 * 主な仕様:
 * - サービス契約終了事由種別が移転解約の場合のみ下位 DA を実行する。
 * - `DA-066-R001` で検針分担エリア情報を取得し、未取得時はデータ不整合例外を送出する。
 * - `DA-004-R001` が 0 件、または先頭レコードの移転先取消コードが取消の場合のみ `DA-004-C001` を実行する。
 * 既知の制約:
 * - `DA-004-R001` が複数件を返す場合は `ITEN_SAK_SEQ_NO DESC` の先頭レコードを最新レコードとして扱う。
 * - `hukkSysShbt`、`correlationId`、`systemDatetime` は public 引数へ含めず、本実装でも参照しない。
 */

        implements RelocationDestinationManagementAdditionAtRelocationCancellation {

    private final String transferDestinationCancellationCancel;

    /**
     * 概要: 移転解約時の移転先管理追加を実行する。
     * 主な仕様:
     * - 移転解約以外の場合は下位 DA を呼ばず正常終了する。
     * - `DA-066-R001` 0 件は `DataInconsistencyException` として送出する。
     * - `DA-004-R001` の先頭レコードが `取消` の場合は通し番号 `+1` で登録する。
     */
    @Override
    public void execute(
            String svkyEndJysb,
            Long gasMtstNo,
            YearMonth knskKjnYm,
            Long svkyNo,
            Long sihKiyNo) {
        if (!transferCancellationReason.equals(svkyEndJysb)) {
            return;
        }

        KbagmstRecord kbagmstRecord = kbagmstDataAccess.findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, knskKjnYm)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kbagmstDataAccess.tableName(),
                        KbagmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM.identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYm", knskKjnYm == null ? null : knskKjnYm.toString())));

        List<ItensakRecord> itensakRecords = itensakDataAccess.findAllByGmtSetNoAndSyokyNo(gasMtstNo, svkyNo);
        if (!shouldInsert(itensakRecords)) {
            return;
        }

        int nextSequenceNo = resolveNextSequenceNo(itensakRecords);
        itensakDataAccess.insert(
                buildWriteRecord(
                        gasMtstNo,
                        svkyNo,
                        nextSequenceNo,
                        sihKiyNo,
                        kbagmstRecord.kbabCktkCd(),
                        kbagmstRecord.kbabKtnCd(),
                        resolveBaseCode(kbagmstRecord.kbabCktkCd())),
                AuditContext.of(SERVICE_USER_ID));
    }

    private boolean shouldInsert(List<ItensakRecord> itensakRecords) {
        if (itensakRecords.isEmpty()) {
            return true;
        }
        ItensakRecord latestRecord = itensakRecords.get(0);
        return transferDestinationCancellationCancel.equals(latestRecord.itenTksCd());
    }

    private int resolveNextSequenceNo(List<ItensakRecord> itensakRecords) {
        return itensakRecords.stream()
                .map(ItensakRecord::itenSakSeqNo)
                .filter(Objects::nonNull)
                .max(Integer::compareTo)
                .orElse(0) + 1;
    }

    private String resolveBaseCode(String itmKktSsyaCd) {
        if (gunmaBranchCode.equals(itmKktSsyaCd)) {
            return GUNMA_BRANCH_BASE_CODE;
        }
        return DEFAULT_BRANCH_BASE_CODE;
    }

            Long gmtSetNo,
            Long syokyNo,
            int nextSequenceNo,
            Long itmSihKiyNo,
            String itmKktSsyaCd,
            String itmKktEgsyCd,
            String itmKktKitiCd) {
        return new ItensakWriteRecord(
                gmtSetNo,
                syokyNo,
                nextSequenceNo,
                null,
                null,
                null,
                null,
                null,
                null,
                noPhone,
                null,
                relocationAddressUnderInvestigation,
                noRoomNumberRegistration,
                null,
                blankContactType,
                null,
                null,
                null,
                null,
                null,
                blankContactType,
                null,
                itmSihKiyNo,
                noInformation,
                noInformation,
                noOtherPaymentCollectionRequest,
                branchOfficeSwitchingNoConcept,
                itmKktSsyaCd,
                itmKktEgsyCd,
                itmKktKitiCd,
                branchNoConcept,
                officeNoConcept,
                baseNoConcept,
                relocationDestinationUnknown,
                relocationInformationNotRegistered,
                null,
                ownerEtcRelocationDestinationInvestigationNotInvestigated,
                null,
                relocationDestinationGovernmentOfficeInvestigationNotInvestigated,
                null,
                relocationTelephoneContactNotInvestigated,
                null,
                relocationSiteVisitNotInvestigated,
                null,
                relocationNameSurveyNotInvestigated,
                null,
                null,
                transferSourceReturnReasonBlank,
                transferDestinationCancellationNoConcept,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                                null,
                                null);
    }
