    /**
     * 概要: SV-326 を実行し、契約終了に伴う共通更新と連携履歴登録を順次実施する。
     * 主な仕様:
     * - step 3 として専用ガスメーター以外の場合のみ SV-043 を呼び出す。
     * - step 7 として DA-044 の照会結果が 0 件の場合のみ SV-041 と SV-032 を順に呼び出す。
     * - step 10 として専用ガスメーターの場合のみ SV-033 を呼び出し、その後 SV-034、SV-036 の順で後続処理を実行する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param tyenYmd 適用終了年月日
     * @param hesnJysb 閉栓事由種別
     * @param tksuSvkeJysb 託送サービス契約終了事由種別
     * @param svkyEndJysb サービス契約終了事由種別
     * @param gmtsetdInf ガスメーター設置場所情報
     * @param tskgmstInf 託送契約情報
     * @param tkssvkyInf 託送サービス契約情報
     * @param svmstkrInf サービス契約・メーター設置場所関連情報
     * @param svckyaInf サービス契約情報
     * @param svsihkrInf サービス支払契約関連情報
     * @param sihakyInf 支払契約情報
     * @param snyInfList 専用メーター情報リスト（SnyInf のリスト）
     * @param sogoTmtrInfList 総合他ガスメーター情報リスト
     * @param isrrGymCd 異動・作業履歴業務コード
     * @param isrrHusrId 異動・作業履歴報告ユーザーID
     * @param isrrHkunKj 異動・作業履歴報告ユーザー名漢字
     * @param isrrHkkrKbn 異動・作業履歴報告経路区分
     */
    @Override
    @SuppressWarnings("checkstyle:ParameterNumber")
    public void execute(
            final Long gasMtstNo,
            final LocalDate tyenYmd,
            final String hesnJysb,
            final String tksuSvkeJysb,
            final String svkyEndJysb,
            final GmtsetdInf gmtsetdInf,
            final TskgmstInf tskgmstInf,
            final TkssvkyInf tkssvkyInf,
            final SvmstkrInf svmstkrInf,
            final SvckyaInf svckyaInf,
            final SvsihkrInf svsihkrInf,
            final SihakyInf sihakyInf,
            final List<SnyInf> snyInfList,
            final List<SogoTmtrInf> sogoTmtrInfList,
            final String isrrGymCd,
            final String isrrHusrId,
            final String isrrHkunKj,
            final String isrrHkkrKbn) {
        LocalDateTime systemDatetime = requireSystemDatetime();
        String hukkSysShbt = requireHukkSysShbt();
        LocalDate nextDay = tyenYmd.plusDays(1);

        if (!dedicatedMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            sv043.execute(
                    gasMtstNo,
                    tyenYmd,
                    tksuSvkeJysb,
                    tskgmstInf,
                    tkssvkyInf,
                    snyInfList);
        }

        ServiceContractMeterLocationUpdateForTerminationOutput sv031Output =
                sv031.execute(
                        gasMtstNo,
                        tyenYmd,
                        svkyEndJysb,
                        svmstkrInf,
                        toClosingSvckyaInf(svckyaInf));

        sv039.execute(tyenYmd, tksuSvkeJysb, gmtsetdInf, sogoTmtrInfList);
        sv029.execute(
                tyenYmd,
                new TaxExemptContractSvckyaInf(svckyaInf.svkyNo()),
                svckyaInf.sziKziKbn());

        List<SvsihkrRecord> relatedContracts = svsihkrDataAccess.findAllBySihKiyNoAndSvkyNoAndSihKiyJtkbAndShziKbn(
                svsihkrInf.sihKiyNo(),
                svsihkrInf.svkyNo(),
                activePaymentStatus,
                gasCommodity);
        if (relatedContracts.isEmpty()) {
            sv041.execute(sihakyInf.skshHyjsKrno());
            sv032.execute(
                    sihakyInf.sihKiyNo(),
                    sihakyInf.skno(),
                    sihakyInf.sihHuhuShbt(),
                    sihakyInf.recUpdJnjNo());
        }

        sv044.execute(toClosingSvckyaInf(svckyaInf), svsihkrInf);
        sv030.execute(
                svckyaInf.syaKbn(),
                svsihkrInf.sihKiyNo(),
                systemDatetime.toLocalDate(),
                tyenYmd);

        if (dedicatedMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb())) {
            sv033.execute(
                    gasMtstNo,
                    nextDay,
                    toReadingSogoTmtrInfList(sogoTmtrInfList));
        }

        BillingCollectionPlatformApiIntegrationForClosingOutput integrationOutput =
                sv034.execute(
                        tyenYmd,
                        svkyEndJysb,
                        toBillingSvckyaInf(svckyaInf),
                        toBillingSihakyInf(sihakyInf),
                        sogoTmtrInfList == null ? 0 : sogoTmtrInfList.size(),
                        isrrGymCd,
                        sv031Output.rkkiSttYmd(),
                        true,
                        null,
                        svsihkrInf.ikskEndYm());

        insertIrslhs(integrationOutput, systemDatetime, hukkSysShbt);
        sv036.execute(
                gasMtstNo,
                tyenYmd,
                svkyEndJysb,
                toTransferWorkHistorySogoTmtrInfList(sogoTmtrInfList),
                isrrGymCd,
                isrrHusrId,
                isrrHkunKj,
                isrrHkkrKbn);
    }

    private void insertIrslhs(
            BillingCollectionPlatformApiIntegrationForClosingOutput output,
            LocalDateTime systemDatetime,
            String hukkSysShbt) {
        irslhsDataAccess.insert(
                new IrslhsWriteRecord(
                        systemDatetime.toLocalDate(),
                        toHms(systemDatetime),
                        output.gasMtstNo(),
                        output.trksUmFlg(),
                        output.olkyUmFlg(),
                        output.gasMtstNo(),
                        output.svkyNo(),
                        0L,
                        output.sikKrno(),
                        output.sqtiKiyNo(),
                        output.khsYmd(),
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId(hukkSysShbt)));
    }

    private String decodeOriginFunctionId(String hukkSysShbt) {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    private int toHms(LocalDateTime systemDatetime) {
        return systemDatetime.getHour() * 10000
                + systemDatetime.getMinute() * 100
                + systemDatetime.getSecond();
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }
