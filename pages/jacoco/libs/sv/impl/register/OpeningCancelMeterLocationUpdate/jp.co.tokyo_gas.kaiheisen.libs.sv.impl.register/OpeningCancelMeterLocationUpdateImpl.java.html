package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;

/**
 * SV-066 開栓取消時ガスメーター設置場所更新処理。
 */
public class OpeningCancelMeterLocationUpdateImpl implements OpeningCancelMeterLocationUpdate {

    private static final MtsetkInf EMPTY_MTSETK_INF = new MtsetkInf(null, null, null);

    private static final MtsetkUpdInf EMPTY_MTSETK_UPD_INF =
            new MtsetkUpdInf(null, null, null, null, null);

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final GasMeterLocationInfoUpdate gasMeterLocationInfoUpdate;

    private final String noExclusionDivision;

    private final String closedStatus;

    private final String newlyInstalledStopperReason;

    /**
     * 必須依存と定数を初期化する。
     */
    public OpeningCancelMeterLocationUpdateImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            GmtsetkDataAccess gmtsetkDataAccess,
            GasMeterLocationInfoUpdate gasMeterLocationInfoUpdate,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.gasMeterLocationInfoUpdate =
                Objects.requireNonNull(gasMeterLocationInfoUpdate, "gasMeterLocationInfoUpdate");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.noExclusionDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.Exclusive.TYPE,
                ConstantLookupKeys.Kaiheisen.Exclusive.NO_EXCLUSION);
        this.closedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.CLOSED);
        this.newlyInstalledStopperReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.NEWLY_INSTALLED_STOPPER);
    }

    @Override
    public OpeningCancelMeterLocationUpdateOutput execute(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            OpeningCancelMeterLocationUpdateGmtsetdInf gmtsetdInf,
            OpeningCancelMeterLocationUpdateGmtsetkInf gmtsetkInf) {
        WorkingState workingState = determineWorkingState(gasMtstNo, trkKnsYmd, gmtsetdInf);
        GmtsetdUpdInf gmtsetdUpdInf = buildGmtsetdUpdInf(workingState);

        gasMeterLocationInfoUpdate.execute(
                noExclusionDivision,
                toSv119GmtsetdInf(gmtsetdInf),
                toSv119GmtsetkInf(gmtsetkInf),
                EMPTY_MTSETK_INF,
                gmtsetdUpdInf,
                EMPTY_MTSETK_UPD_INF);

        List<GmtsetdRecord> pipelineRecords =
                gmtsetdDataAccess.findAllByGasMtstNoAndTkysYmd(gasMtstNo, trkKnsYmd);
        for (GmtsetdRecord pipelineRecord : pipelineRecords) {
            GmtsetkRecord retailRecord =
                    gmtsetkDataAccess.findByGasMtstNoAndTkysYmd(
                                    pipelineRecord.gasMtstNo(),
                                    pipelineRecord.tkysYmd())
                            .orElseThrow(() -> createRetailRecordNotFoundException(
                                    pipelineRecord.gasMtstNo(),
                                    pipelineRecord.tkysYmd()));

            gasMeterLocationInfoUpdate.execute(
                    noExclusionDivision,
                    toSv119GmtsetdInf(pipelineRecord),
                    toSv119GmtsetkInf(retailRecord),
                    EMPTY_MTSETK_INF,
                    gmtsetdUpdInf,
                    EMPTY_MTSETK_UPD_INF);
        }

        return new OpeningCancelMeterLocationUpdateOutput(
                workingState.hesnJysb(),
                workingState.hesnYmd());
    }

    private WorkingState determineWorkingState(
            Long gasMtstNo,
            LocalDate trkKnsYmd,
            OpeningCancelMeterLocationUpdateGmtsetdInf gmtsetdInf) {
        return gmtsetdDataAccess.findByGasMtstNoAndTkysYmdForR003(gasMtstNo, trkKnsYmd)
                .map(record -> new WorkingState(
                        record.kasYmd(),
                        record.hesnYmd(),
                        record.kasJysb(),
                        record.hesnJysb(),
                        record.hesnSdnShbt()))
                .orElseGet(() -> new WorkingState(
                        null,
                        gmtsetdInf.tusoGmtkYmd(),
                        null,
                        newlyInstalledStopperReason,
                        gmtsetdInf.hesnSdnShbt()));
    }

    private GmtsetdUpdInf buildGmtsetdUpdInf(WorkingState workingState) {
        return new GmtsetdUpdInf(
                closedStatus,
                workingState.kasYmd(),
                workingState.hesnYmd(),
                workingState.kasJysb(),
                workingState.hesnJysb(),
                workingState.hesnSdnShbt());
    }

    private GmtsetdInf toSv119GmtsetdInf(OpeningCancelMeterLocationUpdateGmtsetdInf input) {
        return new GmtsetdInf(
                input.gasMtstNo(),
                input.tkysYmd(),
                input.tyenYmd(),
                input.recUpdJnjNo());
    }

    private GmtsetdInf toSv119GmtsetdInf(GmtsetdRecord record) {
        return new GmtsetdInf(
                record.gasMtstNo(),
                record.tkysYmd(),
                record.tyenYmd(),
                record.recUpdJnjNo());
    }

    private GmtsetkInf toSv119GmtsetkInf(OpeningCancelMeterLocationUpdateGmtsetkInf input) {
        return new GmtsetkInf(
                input.gasMtstNo(),
                input.tkysYmd(),
                input.recUpdJnjNo());
    }

    private GmtsetkInf toSv119GmtsetkInf(GmtsetkRecord record) {
        return new GmtsetkInf(
                record.gasMtstNo(),
                record.tkysYmd(),
                record.recUpdJnjNo());
    }

    private RuntimeException createRetailRecordNotFoundException(Long gasMtstNo, LocalDate tkysYmd) {
        return DataInconsistencySupport.create(
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD.identifier(),
                new FlatJsonBuilder()
                        .putIfNonNull("gasMtstNo", gasMtstNo)
                        .putIfNonNull("tkysYmd", tkysYmd));
    }

    private record WorkingState(
            LocalDate kasYmd,
            LocalDate hesnYmd,
            String kasJysb,
            String hesnJysb,
            String hesnSdnShbt) {
    }
}
