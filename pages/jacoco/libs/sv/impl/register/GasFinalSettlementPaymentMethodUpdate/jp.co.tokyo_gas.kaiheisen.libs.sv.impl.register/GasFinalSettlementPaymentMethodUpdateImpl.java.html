package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.GssihhhDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.write.GssihhhWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.GasFinalSettlementPaymentMethodUpdate;

/**
 * 概要: SV-157（ガス最終精算支払方法指示更新）のサービス実装。
 * 主な仕様:
 * - 料金支払方法コードが null または設定なしの場合は何もせず正常終了する。
 * - 上記以外の料金支払方法コードの場合のみ GSSIHHH へ登録する。
 * - 監査用の報告元機能 ID は RequestContext の報告システム種別を復号して設定する。
 * 既知の制約:
 * - 実行時には RequestContext の hukkSysShbt が必須である。
 * - 料金支払方法コード「設定なし」はコンストラクタ初期化時に CIRIUS 定数から fail-fast で解決する。
 */
public class GasFinalSettlementPaymentMethodUpdateImpl implements GasFinalSettlementPaymentMethodUpdate {

    static final String SERVICE_USER_ID = "KHSS157";

    private final GssihhhDataAccess gssihhhDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String noSettingPaymentMethodCode;

    /**
     * 概要: ガス最終精算支払方法指示更新に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - GSSIHHH 登録、RequestContext、開閉栓定数サービスを保持する。
     * - 料金支払方法コード「設定なし」をコンストラクタ初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * - 必須 CIRIUS 定数が不足している場合はコンストラクタ初期化時点で失敗する。
     *
     * @param gssihhhDataAccess GSSIHHH データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public GasFinalSettlementPaymentMethodUpdateImpl(
            GssihhhDataAccess gssihhhDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gssihhhDataAccess = Objects.requireNonNull(gssihhhDataAccess, "gssihhhDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.noSettingPaymentMethodCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodCode.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodCode.NO_SETTING);
    }

    /**
     * 概要: ガス最終精算支払方法指示の登録要否を判定し、必要時に登録する。
     * 主な仕様:
     * - 料金支払方法コードが null または設定なしの場合は何もせず正常終了する。
     * - 登録対象の場合のみ RequestContext の必須値を検証する。
     * - 上記以外の場合は GSSIHHH を 1 件登録する。
     * - 登録件数が 0 件の場合はデータ不整合例外を送出する。
     *
     * @param rshhCd 料金支払方法コード
     * @param svkyNo サービス契約番号
     */
    @Override
    public void execute(String rshhCd, Long svkyNo) {

        if (!shouldInsert(rshhCd)) {
            return;
        }
        requireRequestContext();

        String decodedOriginFunctionId = decodeOriginFunctionId();
        int insertedCount = gssihhhDataAccess.insert(
                new GssihhhWriteRecord(svkyNo, rshhCd, null),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        if (insertedCount == 0) {
            throw DataInconsistencySupport.create(
                    gssihhhDataAccess.tableName(),
                    GssihhhDataAccess.CallSpecs.INSERT.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svkyNo)
                            .put("rshhSjsb", rshhCd)
                            .put("regUsrId", SERVICE_USER_ID)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    /**
     * 概要: 登録対象の料金支払方法コードかどうかを判定する。
     * 主な仕様:
     * - null の場合は登録対象としない。
     * - CIRIUS 定数の設定なしコードと一致する場合は登録対象としない。
     *
     * @param rshhCd 料金支払方法コード
     * @return 登録対象なら true
     */
    private boolean shouldInsert(String rshhCd) {
        return rshhCd != null && !noSettingPaymentMethodCode.equals(rshhCd);
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     * 主な仕様:
     * - hukkSysShbt の存在を確認する。
     * - 欠落がある場合は最初に検出した項目で処理を中断する。
     * 既知の制約:
     * - 欠落時の例外は IllegalStateException に固定される。
     */
    private void requireRequestContext() {
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    /**
     * 概要: RequestContext の報告元システム種別から監査用の機能 ID を取得する。
     * 主な仕様:
     * - RequestContext から hukkSysShbt を取得し、開閉栓定数サービスで名称へ変換する。
     * - 返却値は監査情報の updKtnKinoId に設定する。
     *
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }
}
