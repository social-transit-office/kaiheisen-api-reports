package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.ClosingCancelRelocationDestinationManagementDeletion;

/**
 * 概要: SV-098（閉栓取消時移転先管理削除）のサービス実装。
 * 主な仕様:
 * - 移転閉栓取消対象の業務詳細コードだけを処理対象とする。
 * - 移転先管理が 1 件取得でき、かつ当日登録レコードの場合のみ削除する。
 * 既知の制約:
 * - `correlationId` は本機能では参照しない。
 */
public class ClosingCancelRelocationDestinationManagementDeletionImpl
        implements ClosingCancelRelocationDestinationManagementDeletion {

    static final int ITEN_SAK_SEQ_NO = 1;

    private final ItensakDataAccess itensakDataAccess;

    private final RequestContext requestContext;

    /**
     * 概要: SV-098 の依存オブジェクトを初期化する。
     * 主な仕様:
     * - ITENSAK データアクセスと RequestContext を保持する。
     *
     * @param itensakDataAccess ITENSAK データアクセス
     * @param requestContext リクエストコンテキスト
     */
    public ClosingCancelRelocationDestinationManagementDeletionImpl(
            ItensakDataAccess itensakDataAccess,
            RequestContext requestContext) {
        this.itensakDataAccess = Objects.requireNonNull(itensakDataAccess, "itensakDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
    }

    @Override
    public void execute(String isrrGmssCd, Long gasMtstNo, Long svkyNo) {
        if (!TransferWorkHistoryWorkDetailCodeRetail.RELOCATION_CLOSURE_CCBG6601.equals(isrrGmssCd)) {
            return;
        }

        Optional<ItensakRecord> currentRecord = itensakDataAccess
                .findByGmtSetNoAndSyokyNoAndItenSakSeqNo(gasMtstNo, svkyNo, ITEN_SAK_SEQ_NO);
        if (currentRecord.isEmpty()) {
            return;
        }

        LocalDate systemDate = requireSystemDate();
        ItensakRecord record = currentRecord.orElseThrow();
        if (!Objects.equals(record.recRegYmd(), systemDate)) {
            return;
        }

        itensakDataAccess.deleteByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
                gasMtstNo,
                svkyNo,
                ITEN_SAK_SEQ_NO,
                record.recUpdJnjNo());
    }

    private LocalDate requireSystemDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }
}
