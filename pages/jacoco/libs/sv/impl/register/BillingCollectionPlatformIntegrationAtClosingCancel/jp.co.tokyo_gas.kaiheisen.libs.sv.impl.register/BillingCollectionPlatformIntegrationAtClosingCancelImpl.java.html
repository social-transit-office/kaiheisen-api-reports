/**
 * 概要: SV-100（閉栓取消時請求回収基盤連携）の実装。
 * 主な仕様:
 * - `svkyFkktFlg=true` の場合は helper TSV `他-045_4x復活.tsv` に従って `他-045` を実行し、
 *   続けて `DA-031-C001` へ IRSLHS を登録する。
 * - `svkyFkktFlg=false` の場合は `isrrGymCd=CCBG84` かつ `kkysSisnJtkb` 設定済みのときだけ
 *   helper TSV `他-045_供給停止取消.tsv` に従って `他-045` を実行し、続けて IRSLHS を登録する。
 * - 上記以外の条件では何も連携せずに終了する。
 * 既知の制約:
 * - `step 1.b.1.a.1.a` は best-effort で `nullでも空文字でもない` と解釈する。
 * - `DA-031-C001.tsv` の 7 行目は `sihKiyNo` の設定指示と解釈する。
 */
public final class BillingCollectionPlatformIntegrationAtClosingCancelImpl
        implements BillingCollectionPlatformIntegrationAtClosingCancel {

    static final String SERVICE_USER_ID = "KHSS100";

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final long DEFAULT_GAS_METER_LOCATION_NUMBER = 11111111111L;

    private static final Integer OPEN_END_DATE_VALUE = 99991231;

    private static final Integer OPEN_END_YEAR_MONTH_VALUE = 999912;

    private static final Integer ZERO_VALUE = 0;

    private static final String EMPTY_STRING = "";

    private final HeisenkannrenjiIdoujouhouTourokuCiriusAdapter other045Adapter;

    private final IrslhsDataAccess irslhsDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String presenceFlagNothing;

    private final String presenceFlagYes;

    private final String paymentContractStatusApplying;

    private final String allAtOnceFinalInstallmentBillingServiceType;

    private final String allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType;

    /**
     * 概要: SV-100 の実行に必要な依存と必須定数を初期化する。
     *
     * @param other045Adapter 他-045 adapter
     * @param irslhsDataAccess DA-031-C001
     * @param requestContext RequestContext
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    @Inject
    public BillingCollectionPlatformIntegrationAtClosingCancelImpl(
            HeisenkannrenjiIdoujouhouTourokuCiriusAdapter other045Adapter,
            IrslhsDataAccess irslhsDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.other045Adapter = Objects.requireNonNull(other045Adapter, "other045Adapter");
        this.irslhsDataAccess = Objects.requireNonNull(irslhsDataAccess, "irslhsDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.presenceFlagNothing = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.NOTHING);
        this.presenceFlagYes = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.YES);
        this.paymentContractStatusApplying = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING);
        this.allAtOnceFinalInstallmentBillingServiceType =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                        ConstantLookupKeys.Cirius.BillingServiceType.ALL_AT_ONCE_FINAL_INSTALLMENT);
        this.allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                        ConstantLookupKeys.Cirius.BillingServiceType.ALL_AT_ONCE_NEXT_3RD_BUSINESS_YEAR_CLOSING_AVAILABLE_AT_SHORT_NOTICE);
    }

    /** 概要: 閉栓取消時の請求回収基盤連携を実行する。 */
    @Override
    public void execute(
            Boolean svkyFkktFlg,
            Boolean sihakyFkktFlg,
            BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf sihakyInf,
            BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf svckyaInf,
            Boolean suguFlg,
            LocalDate trkTishYmd,
            String isrrGymCd) {
        if (Boolean.TRUE.equals(svkyFkktFlg)) {
            executeIntegration(
                    buildRevivalPayload(
                            sihakyInf,
                            svckyaInf,
                            Boolean.TRUE.equals(sihakyFkktFlg),
                            Boolean.TRUE.equals(suguFlg),
                            requireSystemDatetime()),
                    sihakyInf,
                    trkTishYmd);
            return;
        }

        if (!ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.SUPPLY_OUTAGE_REPORT_RETAIL
                .equals(isrrGymCd)) {
            return;
        }

        if (svckyaInf.kkysSisnJtkb() == null || svckyaInf.kkysSisnJtkb().isEmpty()) {
            return;
        }

        executeIntegration(
                buildSupplyStopCancelPayload(
                        svckyaInf,
                        Boolean.TRUE.equals(suguFlg),
                        requireSystemDatetime()),
                sihakyInf,
                trkTishYmd);
    }

    private void executeIntegration(
            HeisenkannrenjiIdoujouhou payload,
            BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf sihakyInf,
            LocalDate trkTishYmd) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        LocalDateTime systemDatetime = requireSystemDatetime();
        other045Adapter.register(new HeisenkannrenjiIdoujouhouTourokuRequest(payload));
        irslhsDataAccess.insert(
                buildIrslhsWriteRecord(systemDatetime, payload, sihakyInf, trkTishYmd),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    private HeisenkannrenjiIdoujouhou buildRevivalPayload(
            BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf sihakyInf,
            BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf svckyaInf,
            boolean sihakyFkktFlg,
            boolean suguFlg,
            LocalDateTime systemDatetime) {
        return new HeisenkannrenjiIdoujouhou(
                EMPTY_STRING,
                sihakyInf == null ? 0L : sihakyInf.sikKrno(),
                svckyaInf.sqtiKiyNo(),
                svckyaInf.svkyNo(),
                svckyaInf.skno(),
                paymentContractStatusApplying,
                resolveGasMeterLocationNo(suguFlg, svckyaInf),
                signedDateOf(svckyaInf.rkkiSttYmd()),
                signedDateOf(svckyaInf.rkkiEndYmd()),
                presenceFlagYes,
                presenceFlagNothing,
                presenceFlagNothing,
                presenceFlagNothing,
                presenceFlagNothing,
                signedDateOf(systemDatetime.toLocalDate()),
                sihakyFkktFlg
                        ? ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                                .BILLABLE_PAYMENT_MANAGEMENT_TRANSFER
                        : EMPTY_STRING,
                sihakyFkktFlg ? OPEN_END_DATE_VALUE : ZERO_VALUE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                        .NO_SERVICE_SUSPENSION_OR_CANCELLATION,
                OPEN_END_DATE_VALUE,
                OPEN_END_DATE_VALUE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .BILLABLE_CONTRACT_MANAGEMENT_TRANSFER,
                sihakyInf == null ? ZERO_VALUE : resolveRevivalIkskEndYm(sihakyInf.sikSvcShbt()),
                EMPTY_STRING,
                EMPTY_STRING,
                0L,
                ZERO_VALUE,
                EMPTY_STRING,
                EMPTY_STRING,
                0L,
                ZERO_VALUE);
    }

    private HeisenkannrenjiIdoujouhou buildSupplyStopCancelPayload(
            BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf svckyaInf,
            boolean suguFlg,
            LocalDateTime systemDatetime) {
        return new HeisenkannrenjiIdoujouhou(
                EMPTY_STRING,
                0L,
                svckyaInf.sqtiKiyNo(),
                svckyaInf.svkyNo(),
                svckyaInf.skno(),
                paymentContractStatusApplying,
                resolveGasMeterLocationNo(suguFlg, svckyaInf),
                signedDateOf(svckyaInf.rkkiSttYmd()),
                signedDateOf(svckyaInf.rkkiEndYmd()),
                presenceFlagYes,
                presenceFlagNothing,
                presenceFlagNothing,
                presenceFlagNothing,
                presenceFlagNothing,
                signedDateOf(systemDatetime.toLocalDate()),
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .BILLABLE_PAYMENT_MANAGEMENT_TRANSFER,
                ZERO_VALUE,
                ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                        .NO_SERVICE_SUSPENSION_OR_CANCELLATION,
                OPEN_END_DATE_VALUE,
                ZERO_VALUE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .BILLABLE_CONTRACT_MANAGEMENT_TRANSFER,
                ZERO_VALUE,
                EMPTY_STRING,
                EMPTY_STRING,
                0L,
                ZERO_VALUE,
                EMPTY_STRING,
                EMPTY_STRING,
                0L,
                ZERO_VALUE);
    }

    private IrslhsWriteRecord buildIrslhsWriteRecord(
            LocalDateTime systemDatetime,
            HeisenkannrenjiIdoujouhou payload,
            BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf sihakyInf,
            LocalDate trkTishYmd) {
        Long sihKiyNo;
        if (sihakyInf != null && hasTargetContractNumber(payload.skrkSqtkJhu())) {
            sihKiyNo = sihakyInf.sihKiyNo();
        } else {
            sihKiyNo = Long.valueOf(0L);
        }
        return new IrslhsWriteRecord(
                systemDatetime.toLocalDate(),
                toHms(systemDatetime),
                payload.gasMtstNo(),
                presenceFlagYes.equals(payload.trksUmFlg()),
                presenceFlagYes.equals(payload.olkyUmFlg()),
                payload.gasMtstNo(),
                payload.svkyNo(),
                sihKiyNo,
                payload.skrkSqkrJhu(),
                payload.skrkSqtkJhu(),
                trkTishYmd,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private long resolveGasMeterLocationNo(
            boolean suguFlg,
            BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf svckyaInf) {
        return suguFlg ? svckyaInf.svkyNo() : DEFAULT_GAS_METER_LOCATION_NUMBER;
    }

    private boolean hasTargetContractNumber(String value) {
        if (value == null || value.isEmpty()) {
            return false;
        }
        if (isSxTargetContractNumber(value)) {
            return true;
        }
        try {
            return Long.parseLong(value) > 0L;
        } catch (NumberFormatException ignored) {
            return false;
        }
    }

    private boolean isSxTargetContractNumber(String value) {
        return value.startsWith("S");
    }

    private Integer resolveRevivalIkskEndYm(String sikSvcShbt) {
        if (allAtOnceFinalInstallmentBillingServiceType.equals(sikSvcShbt)
                || allAtOnceNext3rdBusinessYearClosingAvailableAtShortNoticeBillingServiceType
                        .equals(sikSvcShbt)) {
            return OPEN_END_YEAR_MONTH_VALUE;
        }
        return ZERO_VALUE;
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private Integer signedDateOf(LocalDate value) {
        return value == null ? null : Integer.valueOf(value.format(BASIC_DATE));
    }

    private int toHms(LocalDateTime value) {
        return value.getHour() * 10000 + value.getMinute() * 100 + value.getSecond();
    }
}
