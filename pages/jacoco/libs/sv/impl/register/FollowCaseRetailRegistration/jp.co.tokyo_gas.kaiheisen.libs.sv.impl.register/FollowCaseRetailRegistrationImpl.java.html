/**
 * 概要: SV-128（フォロー件名＜小売＞登録）のサービス実装。

 * - 報告システム種別、相関ID、システム日時は RequestContext から取得する。
 * 既知の制約:
 * - ガスメーター設置場所番号の全角化は ASCII 半角文字の全角変換で実施する。

     * 概要: SV-128 に必要な依存オブジェクトと必須定数を初期化する。
     *
     * @param gasFeeContractTypeSortingContentAcquisition ガス料金契約種別振分内容取得サービス
     * @param gymykiyDataAccess 業務用契約データアクセス
     * @param kiycstkDataAccess 契約お客さま＜小売＞データアクセス
     * @param flknmeiDataAccess フォロー件名＜小売＞データアクセス

     * @param kaiheisenConstantService 開閉栓定数サービス

            GasFeeContractTypeSortingContentAcquisition gasFeeContractTypeSortingContentAcquisition,

            KiycstkDataAccess kiycstkDataAccess,
            FlknmeiDataAccess flknmeiDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gasFeeContractTypeSortingContentAcquisition = Objects.requireNonNull(
                gasFeeContractTypeSortingContentAcquisition,
                "gasFeeContractTypeSortingContentAcquisition");
        this.gymykiyDataAccess = Objects.requireNonNull(gymykiyDataAccess, "gymykiyDataAccess");
        this.kiycstkDataAccess = Objects.requireNonNull(kiycstkDataAccess, "kiycstkDataAccess");
        this.flknmeiDataAccess = Objects.requireNonNull(flknmeiDataAccess, "flknmeiDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");

    /**
     * 概要: フォロー件名＜小売＞登録を実行する。
     *
     * @param kiyShbt 契約種別
     * @param svkyNo サービス契約番号
     * @param kyttSkno 契約特定世代管理番号
     * @param kyttTsno 契約特定通し番号
     * @param kysKyoTrno 契約者契約お客さま登録番号
     * @param flttCktkCd フォロー担当地域統括コード
     * @param flttKtnCd フォロー担当拠点コード
     * @param gasMtstNo ガスメーター設置場所番号
     */
    @Override
    public void execute(
            String kiyShbt,
            Long svkyNo,
            Integer kyttSkno,
            Integer kyttTsno,
            Long kysKyoTrno,
            String flttCktkCd,
            String flttKtnCd,
            Long gasMtstNo) {
        LocalDateTime systemDatetime = requireRequestContext();
        String decodedOriginFunctionId = decodeOriginFunctionId();

        GasFeeContractTypeSortingContentAcquisitionOutput sortingOutput =
                gasFeeContractTypeSortingContentAcquisition.execute(
                        kiyShbt,
                        List.of(
                                new GrkKsbHwsb(j0528ClosingTarget, null),
                                new GrkKsbHwsb(j0528FollowTeamType, null)),
                        systemDatetime.toLocalDate());

        Optional<String> targetResult = findSortingCode(sortingOutput, 0);
        if (targetResult.isEmpty() || !j0529TargetCode.equals(targetResult.get())) {
            return;
        }

        GymykiyRecord gymykiyRecord = gymykiyDataAccess.findBySvkyNoAndKyttSknoAndKyttTsno(svkyNo, kyttSkno, kyttTsno)
                .orElse(null);
        KiycstkRecord kiycstkRecord = kiycstkDataAccess.findByKyoTrkNo(kysKyoTrno)
                .orElse(null);

        String teamCode = resolveFollowTeamCode(findSortingCode(sortingOutput, 1).orElse(null));
        String fullWidthGasMeterNo = toFullWidth(String.valueOf(gasMtstNo)) + "閉栓";
        FlknmeiWriteRecord insertRecord = new FlknmeiWriteRecord(
                followTaskType,
                systemDatetime.toLocalDate(),
                toHhmmss(systemDatetime),
                svkyNo,
                svkyNo,
                flttCktkCd,
                flttKtnCd,
                getNullable(gymykiyRecord, GymykiyRecord::kiyKnrkCd),
                getNullable(gymykiyRecord, GymykiyRecord::kiyKtnCd),
                getNullable(gymykiyRecord, GymykiyRecord::kiySsyCd),
                null,
                teamCode,
                followReason,
                fullWidthGasMeterNo,
                null,
                null,
                null,
                null,
                null,
                null,
                "CBGG1357",
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                followStatusNotStarted,
                getNullable(kiycstkRecord, KiycstkRecord::ctnmKn1),
                getNullable(kiycstkRecord, KiycstkRecord::ctnmKj1),
                getNullable(kiycstkRecord, KiycstkRecord::ctnm1HujnShbt),
                getNullable(kiycstkRecord, KiycstkRecord::ctnm1HshiShbt),
                getNullable(kiycstkRecord, KiycstkRecord::ctnmKn2),
                getNullable(kiycstkRecord, KiycstkRecord::ctnmKj2),
                getNullable(kiycstkRecord, KiycstkRecord::ctnm2HujnShbt),
                getNullable(kiycstkRecord, KiycstkRecord::ctnm2HshiShbt),
                getNullable(kiycstkRecord, KiycstkRecord::ctnmKjhkFlg),
                getNullable(kiycstkRecord, KiycstkRecord::yubnNo),
                getNullable(kiycstkRecord, KiycstkRecord::gkkCd),
                getNullable(kiycstkRecord, KiycstkRecord::mchCd),
                getNullable(kiycstkRecord, KiycstkRecord::cmazCd),
                getNullable(kiycstkRecord, KiycstkRecord::cbgSuj1),
                getNullable(kiycstkRecord, KiycstkRecord::cbgKn2),
                getNullable(kiycstkRecord, KiycstkRecord::cbgKn3),
                getNullable(kiycstkRecord, KiycstkRecord::cbgKn4),
                getNullable(kiycstkRecord, KiycstkRecord::bldnKj),
                getNullable(kiycstkRecord, KiycstkRecord::bldnKn),
                getNullable(kiycstkRecord, KiycstkRecord::tunoKn),
                getNullable(kiycstkRecord, KiycstkRecord::hynKn),
                getNullable(kiycstkRecord, KiycstkRecord::htuFlg),
                getNullable(kiycstkRecord, KiycstkRecord::bldnKjhkFlg),
                getNullable(kiycstkRecord, KiycstkRecord::jutKbn),
                null);
        flknmeiDataAccess.insert(
                insertRecord,
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    private <T, R> R getNullable(T record, Function<T, R> accessor) {
        if (record == null) {
            return null;
        }
        return accessor.apply(record);
    }

    private Optional<String> findSortingCode(
            GasFeeContractTypeSortingContentAcquisitionOutput output,
            int index) {
        if (output == null || output.grkKsbHnsbList() == null || output.grkKsbHnsbList().size() <= index) {
            return Optional.empty();
        }
        GrkKsbHnsb item = output.grkKsbHnsbList().get(index);
        if (item == null || item.grkKsbHnsb() == null) {
            return Optional.empty();
        }
        return Optional.of(item.grkKsbHnsb());
    }

    private String resolveFollowTeamCode(String followTeamType) {
        if (j0529LargeContract.equals(followTeamType)) {
            return teamLarge;
        }
        if (j0529LargeGeneral.equals(followTeamType)) {
            return teamLargeGeneral;
        }
        if (j0529OptionalClause.equals(followTeamType)) {
            return teamOptionalClause;
        }
        throw new KaiheisenException(
                "ガス料金契約種別振分内容種別の値が不正のため、処理を終了しました。");
    }

    private LocalDateTime requireRequestContext() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }

    private int toHhmmss(LocalDateTime datetime) {
        return datetime.getHour() * 10000
                + datetime.getMinute() * 100
                + datetime.getSecond();
    }

    private String toFullWidth(String value) {
        StringBuilder builder = new StringBuilder(value.length());
        for (int index = 0; index < value.length(); index++) {
            char ch = value.charAt(index);
            if (ch == ' ') {
                builder.append('　');
            } else if (ch >= '!' && ch <= '~') {
                builder.append((char) (ch + 0xFEE0));
            } else {
                builder.append(ch);
            }
        }
        return builder.toString();
    }
