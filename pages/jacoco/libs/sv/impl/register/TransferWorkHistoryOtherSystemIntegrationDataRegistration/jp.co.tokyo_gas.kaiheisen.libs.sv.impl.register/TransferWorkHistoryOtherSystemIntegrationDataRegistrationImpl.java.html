/**
 * 概要: SV-272（異動・作業履歴他システム連携データ登録）の実装。
 * 主な仕様:
 * - 性能計測のため、コンスタントマスタの存在・登録利用可否チェック計20項目を一時停止する。
 * - DA-158-R002 で通し番号を採番し、DA-158-C001 の主キー重複時は 1 回だけ再試行する。
 * - 監査情報は RequestContext と AuditContext から組み立てる。
 * 既知の制約:
 * - SV-273 の公開契約が未配置のため、登録利用可否判定は既存のコンスタントマスタ取得 SV で代替する。
 * - チェック用の補助メソッドと依存は復元用に保持し、一時停止中は呼び出さない。
 */

    /**
     * 概要: SV-272 実行に必要な依存を初期化する。
     * 主な仕様:
     * - コンスタント確認 SV、DA-158、RequestContext、開閉栓定数サービスを保持する。
     * - 依存が不足している場合はコンストラクタ初期化時点で fail-fast する。
     *
     * @param constantMasterInfoAcquisition コンスタントマスタ情報取得 SV
     * @param isrrkrDataAccess 異動・作業履歴連携管理 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public TransferWorkHistoryOtherSystemIntegrationDataRegistrationImpl(
            ConstantMasterInfoAcquisition constantMasterInfoAcquisition,
            IsrrkrDataAccess isrrkrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.constantMasterInfoAcquisition = Objects.requireNonNull(
                constantMasterInfoAcquisition,
                "constantMasterInfoAcquisition");
        this.isrrkrDataAccess = Objects.requireNonNull(isrrkrDataAccess, "isrrkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
    }

    /**
     * 概要: 異動・作業履歴連携管理情報を登録する。
     * 主な仕様:
     * - 必須項目、所有事業者区分、RequestContext を確認し、コンスタント確認は一時省略する。
     * - 最大通し番号を採番し、重複時は 1 回だけ通し番号を加算して再登録する。
     * - 正常終了時は戻り値を持たずに完了する。
     *
     * @param isrrkrInf 異動・作業履歴連携管理情報
     */
    @Override
    public void execute(IsrrkrInf isrrkrInf) {
        validateRequiredFields(isrrkrInf);
        validateOwnerDivision(isrrkrInf.shyuJgysKbn());
        String decodedOriginFunctionId = requireDecodedOriginFunctionId();
        // 性能計測のため、コンスタントマスタの存在・登録利用可否チェック計20項目を一時停止する。

        int nextSequence = isrrkrDataAccess
                .findMaxIsrrRkkrTsnoByCstNoAndIsrrTriYmd(isrrkrInf.cstNo(), isrrkrInf.isrrTriYmd())
                .map(value -> value + 1)
                .orElse(1);

        try {
            insertIsrrkr(isrrkrInf, nextSequence, decodedOriginFunctionId);
        } catch (DuplicateKeyException firstDuplicateKeyException) {
            insertIsrrkr(isrrkrInf, nextSequence + 1, decodedOriginFunctionId);
        }
    }

    private void validateRequiredFields(IsrrkrInf isrrkrInf) {
        List<String> missingFieldNames = new ArrayList<>();
        if (isrrkrInf.cstNo() == null) {
            missingFieldNames.add("お客さま番号");
        }
        if (isrrkrInf.isrrTriHms() == null) {
            missingFieldNames.add("異動・作業履歴登録依頼時分秒");
        }
        if (isBlank(isrrkrInf.shyuJgysKbn())) {
            missingFieldNames.add("所有事業者区分");
        }
        if (isBlank(isrrkrInf.kytuIsrrGymCd())) {
            missingFieldNames.add("異動・作業履歴業務コード");
        }
        if (isrrkrInf.idsgYmd() == null) {
            missingFieldNames.add("異動・作業年月日");
        }
        if (!missingFieldNames.isEmpty()) {
            throw new ValidationException(
                    List.of(ValidationError.ofField(
                            ErrorCodeConstants.ERROR_CODE_010000001,
                            String.join("、", missingFieldNames))),
                    true);
        }
    }

    private void validateOwnerDivision(String ownerDivision) {
        if (!OWNER_DIVISION_CONDUIT.equals(ownerDivision) && !OWNER_DIVISION_RETAIL.equals(ownerDivision)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000010);
        }
    }

    private List<ConstantCheckDefinition> resolveHelperGroupOneDefinitions(String ownerDivision) {
        if (OWNER_DIVISION_CONDUIT.equals(ownerDivision)) {
            return HELPER_GROUP_ONE_CONDUIT;
        }
        return HELPER_GROUP_ONE_RETAIL;
    }

    private void executeExistenceChecks(IsrrkrInf isrrkrInf, List<ConstantCheckDefinition> definitions) {
        for (ConstantCheckDefinition definition : definitions) {
            String constantCode = normalizeConstantCode(definition.codeResolver().apply(isrrkrInf));
            if (constantCode == null) {
                continue;
            }
            try {
                constantMasterInfoAcquisition.execute(definition.cnstShbtCd(), constantCode);
            } catch (DataNotFoundException dataNotFoundException) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000010);
            }
        }
    }

    private void executeAvailabilityChecks(IsrrkrInf isrrkrInf, List<ConstantCheckDefinition> definitions) {
        for (ConstantCheckDefinition definition : definitions) {
            String constantCode = normalizeConstantCode(definition.codeResolver().apply(isrrkrInf));
            if (constantCode == null) {
                continue;
            }
            try {
                ConstantMasterInfoAcquisitionOutput output =
                        constantMasterInfoAcquisition.execute(definition.cnstShbtCd(), constantCode);
                if (Boolean.FALSE.equals(output.trkjRykhFlg())) {
                    throw new IllegalStateException(
                            "Constant registration unavailable: cnstShbtCd="
                                    + definition.cnstShbtCd()
                                    + ", cnstCd="
                                    + constantCode);
                }
            } catch (DataNotFoundException dataNotFoundException) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000010);
            }
        }
    }

    private String requireDecodedOriginFunctionId() {
        String hukkSysShbt = requestContext
                .getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    private static String normalizeConstantCode(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }

    private static String toConstantCode(Boolean value) {
        if (value == null) {
            return null;
        }
        return value
                ? ConstantLookupKeys.Cirius.MeterReadingAvailabilityFlag.METER_READING_POSSIBLE
                : ConstantLookupKeys.Cirius.MeterReadingAvailabilityFlag.METER_READING_NOT_POSSIBLE;
    }

    private static boolean isBlank(String value) {
        return value == null || value.isBlank();
    }

    private record ConstantCheckDefinition(String cnstShbtCd, Function<IsrrkrInf, String> codeResolver) {
    }
