package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.SqshyjsDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.SqshyjsRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsTkysYmdJnjUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.sqshyjs.write.SqshyjsWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.InvoiceDisplayControlUpdateForTermination;

/**
 * 概要: SV-041（請求書表示制御情報更新（契約終了））のサービス実装。
 * 主な仕様:
 * - 請求書表示制御管理番号が null または 0 の場合は下位 DA を呼ばずに終了する。
 * - それ以外の場合のみ RequestContext の必須値を検証し、SQSHYJS を取得して契約終了状態へ更新する。
 * - 取得 0 件は DataInconsistencyException へ変換する。
 * 既知の制約:
 * - correlationId は存在確認のみを行い、監査や下位引数には直接使用しない。
 */
public class InvoiceDisplayControlUpdateForTerminationImpl
        implements InvoiceDisplayControlUpdateForTermination {

    static final String SERVICE_USER_ID = "KHSS041";

    private final SqshyjsDataAccess sqshyjsDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * 概要: SV-041 に必要な依存を初期化する。
     * 主な仕様:
     * - SQSHYJS データアクセス、RequestContext、開閉栓定数サービスを保持する。
     * - 依存不足はインスタンス生成時に fail-fast で検知する。
     *
     * @param sqshyjsDataAccess SQSHYJS データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public InvoiceDisplayControlUpdateForTerminationImpl(
            SqshyjsDataAccess sqshyjsDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.sqshyjsDataAccess = Objects.requireNonNull(sqshyjsDataAccess, "sqshyjsDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
    }

    /**
     * 概要: 請求書表示制御情報を契約終了状態へ更新する。
     * 主な仕様:
     * - 請求書表示制御管理番号が null または 0 の場合は処理せず終了する。
     * - 通常値の場合のみ RequestContext の correlationId、hukkSysShbt、systemDatetime を必須とする。
     * - 通常値の場合は R002 取得結果を U002 更新条件へ引き継ぐ。
          * - U002 の更新結果は下位 DA の戻り値に委ね、SV では追加の 0 件判定を行わない。
     *
     * @param skshHyjsKrno 請求書表示制御管理番号
     */
    @Override
    public void execute(Long skshHyjsKrno) {
        if (skshHyjsKrno == null || Long.valueOf(0L).equals(skshHyjsKrno)) {
            return;
        }

        RequestContextValues requestContextValues = requireRequestContextValues();
        LocalDate currentDate = requestContextValues.systemDatetime().toLocalDate();
        SqshyjsRecord existingRecord = sqshyjsDataAccess.findBySkshHyjsKrnoAndKnskKjnYmd(
                        skshHyjsKrno,
                        currentDate)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        sqshyjsDataAccess.tableName(),
                        SqshyjsDataAccess.CallSpecs.FIND_BY_SKSH_HYJS_KRNO_AND_KNSK_KJN_YMD.identifier(),
                        new FlatJsonBuilder()
                                .put("skshHyjsKrno", skshHyjsKrno)
                                .put("knskKjnYmd", currentDate)));
        String decodedOriginFunctionId = decodeOriginFunctionId(requestContextValues.hukkSysShbt());
        sqshyjsDataAccess.updateBySkshHyjsKrnoAndTkysYmdAndRecUpdJnjNo(
                new SqshyjsWriteRecord(
                        null,
                        null,
                        currentDate,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                Set.of(SqshyjsColumn.TYEN_YMD),
                new SqshyjsTkysYmdJnjUpdateKey(
                        existingRecord.skshHyjsKrno(),
                        existingRecord.tkysYmd(),
                        existingRecord.recUpdJnjNo()),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     *
     * @return 検証済みの RequestContext 値
     */
    private RequestContextValues requireRequestContextValues() {
        requireCorrelationId();
        return new RequestContextValues(requireHukkSysShbt(), requireSystemDatetime());
    }

    /**
     * 概要: 相関IDを取得する。
     *
     * @return 相関ID
     */
    private String requireCorrelationId() {
        return requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
    }

    /**
     * 概要: 報告システム種別を取得する。
     *
     * @return 報告システム種別
     */
    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    /**
     * 概要: システム日時を取得する。
     *
     * @return システム日時
     */
    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    /**
     * 概要: RequestContext の報告システム種別から監査用の起点機能IDを復号する。
     *
     * @param hukkSysShbt 報告システム種別
     * @return 監査用起点機能ID
     */
    private String decodeOriginFunctionId(String hukkSysShbt) {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    private record RequestContextValues(String hukkSysShbt, LocalDateTime systemDatetime) {
    }
}
