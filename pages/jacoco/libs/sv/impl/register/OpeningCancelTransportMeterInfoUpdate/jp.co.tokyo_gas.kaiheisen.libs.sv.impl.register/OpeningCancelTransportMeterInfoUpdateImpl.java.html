package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;

/**
 * 概要: SV-067（開栓取消託送メータ情報更新）のサービス実装。
 * 主な仕様:
 * - DA-019-R001 で託送メータ情報を取得し、先頭レコードだけを更新判定対象とする。
 * - 先頭レコードが当日更新かつスイッチ申込以外かつ本登録の場合のみ DA-019-U002 を実行する。
 * - 非排他更新のため、更新件数 0 件は DataInconsistencyException へ変換する。
 */
public final class OpeningCancelTransportMeterInfoUpdateImpl
        implements OpeningCancelTransportMeterInfoUpdate {

    static final String SERVICE_USER_ID = "KHSS067";

    static final Set<TaksmtrColumn> TARGET_COLUMNS = Set.of(TaksmtrColumn.TAKS_KY_JT_CD);

    private final TaksmtrDataAccess taksmtrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String bookRegistrationStatus;

    private final String temporaryRegistrationStatus;

    private final String switchApplicationCode;

    /**
     * 必須依存と必須定数を初期化する。
     *
     * @param taksmtrDataAccess 託送メータ情報 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public OpeningCancelTransportMeterInfoUpdateImpl(
            TaksmtrDataAccess taksmtrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.taksmtrDataAccess = Objects.requireNonNull(taksmtrDataAccess, "taksmtrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.bookRegistrationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.BOOK_REGISTRATION);
        this.temporaryRegistrationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TEMPORARY_REGISTRATION);
        this.switchApplicationCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.SWITCH_APPLICATION);
    }

    @Override
    public void execute(Long gasMtstNo) {
        List<TaksmtrRecord> records = taksmtrDataAccess.findAllByGmtSetNo(gasMtstNo);
        if (records.isEmpty()) {
            return;
        }

        TaksmtrRecord record = records.getFirst();
        LocalDate systemDate = requireSystemDate();
        if (!shouldUpdate(record, systemDate)) {
            return;
        }

        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodeOriginFunctionId());
        int updatedCount = taksmtrDataAccess.updateByTakskyNoAndHkkmItiNoAndGmtSetNo(
                buildWriteRecord(),
                TARGET_COLUMNS,
                new TaksmtrUpdateKey(
                        record.takskyNo(),
                        record.hkkmItiNo(),
                        record.gmtSetNo(),
                        null),
                auditContext);
        if (updatedCount == 0) {
            throw createDataInconsistency(record, auditContext);
        }
    }

    private boolean shouldUpdate(TaksmtrRecord record, LocalDate systemDate) {
        return Objects.equals(record.recUpdYmd(), systemDate)
                && !switchApplicationCode.equals(record.taksCisHntCd())
                && bookRegistrationStatus.equals(record.taksKyJtCd());
    }

    private LocalDate requireSystemDate() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .filter(value -> !value.isBlank())
                .orElse(null);
        if (hukkSysShbt == null) {
            return null;
        }
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private TaksmtrWriteRecord buildWriteRecord() {
        return new TaksmtrWriteRecord(
                temporaryRegistrationStatus,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private DataInconsistencyException createDataInconsistency(
            TaksmtrRecord record,
            AuditContext auditContext) {
        return DataInconsistencySupport.create(
                taksmtrDataAccess.tableName(),
                TaksmtrDataAccess.CallSpecs.UPDATE_BY_TAKSKY_NO_AND_HKKM_ITI_NO_AND_GMT_SET_NO.identifier(),
                new FlatJsonBuilder()
                        .put("takskyNo", record.takskyNo())
                        .put("hkkmItiNo", record.hkkmItiNo())
                        .put("gmtSetNo", record.gmtSetNo())
                        .put("taksKyJtCd", temporaryRegistrationStatus)
                        .put("updUsrId", auditContext.userId())
                        .put("updKtnKinoId", auditContext.kinoId()));
    }
}
