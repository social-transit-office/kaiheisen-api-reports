/**
 * 概要: SV-333（用途コード更新機能-導管）の実装クラス。
 * 主な仕様:
 * - DA-033-R006、SV-001、SV-274、DA-033-U003、SV-089、他-011/他-012 を直列実行し、用途コード更新を完了する。
 * - `hukkSysShbt` と `systemDatetime` は RequestContext から取得し、不足時は fail-fast する。
 * - `correlationId` は本 SV では直接参照しない。
 * - step 13 は helper TSV `他-011、他-012.tsv` を正とし、導管履歴登録 request を直接組み立てる。
 */
public final class UsageCodeUpdateFunctionPipelineImpl implements UsageCodeUpdateFunctionPipeline {

    private static final DateTimeFormatter BASIC_DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private static final DateTimeFormatter TIME_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    static final String SERVICE_USER_ID = "KHSS333";

    private static final String SHR_KBN_OTHER = RetailDivisionCode.OTHER.code();

    private static final String GAS_USAGE_CODE_TYPE = ConstantLookupKeys.Cirius.GasUsageCode.TYPE;

    private static final String HISTORY_ITEM_NAME = "用途コード";

    private static final Set<GmtsetdColumn> GMTSETD_TARGET_COLUMNS = Set.of(GmtsetdColumn.GAS_YT_CD);

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final RetailTypeJudgment retailTypeJudgment;

    private final GasUsageCodeRegistrationAvailabilityJudgment gasUsageCodeRegistrationAvailabilityJudgment;

    private final ConstantMasterInfoAcquisition constantMasterInfoAcquisition;

    private final IdousagyourirekiDoukanJouhouCiriusAdapter doukanAdapter;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String openingStatus;

    private final String customerInfoAcquisitionPossible;

    /**
     * 概要: SV-333 実行に必要な依存と必須定数を初期化する。
     * 主な仕様:
     * - 下位 DA/SV、RequestContext、定数サービスを保持する。
     * - 開栓中判定と導管履歴登録に必要な必須定数はコンストラクタ初期化時点で fail-fast に解決する。
     *
     * @param gmtsetdDataAccess DA-033 データアクセス
     * @param retailTypeJudgment SV-001 小売判定
     * @param gasUsageCodeRegistrationAvailabilityJudgment SV-274 ガス用途コード登録可否判定
     * @param constantMasterInfoAcquisition SV-089 コンスタントマスタ情報取得
     * @param doukanAdapter 導管向け異動・作業履歴登録 adapter
     * @param requestContext リクエストコンテキスト
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public UsageCodeUpdateFunctionPipelineImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            RetailTypeJudgment retailTypeJudgment,
            GasUsageCodeRegistrationAvailabilityJudgment gasUsageCodeRegistrationAvailabilityJudgment,
            ConstantMasterInfoAcquisition constantMasterInfoAcquisition,
            IdousagyourirekiDoukanJouhouCiriusAdapter doukanAdapter,
            RequestContext requestContext,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.retailTypeJudgment = Objects.requireNonNull(retailTypeJudgment, "retailTypeJudgment");
        this.gasUsageCodeRegistrationAvailabilityJudgment = Objects.requireNonNull(
                gasUsageCodeRegistrationAvailabilityJudgment,
                "gasUsageCodeRegistrationAvailabilityJudgment");
        this.constantMasterInfoAcquisition = Objects.requireNonNull(
                constantMasterInfoAcquisition,
                "constantMasterInfoAcquisition");
        this.doukanAdapter = Objects.requireNonNull(doukanAdapter, "doukanAdapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.customerInfoAcquisitionPossible = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PossibilityFlag.TYPE,
                ConstantLookupKeys.Cirius.PossibilityFlag.POSSIBLE);
    }

    /**
     * {@inheritDoc}
     */
    @Override
    public void execute(
            Long gasMtstNo,
            LocalDate shrYmd,
            String gasYtCd,
            String akrkKbn,
            String akrkTrskShbt,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrHkkrKbn,
            String isrrSktCd) {
        RequestContextValues requestContextValues = requireRequestContextValues();
        LocalDate knskKjnYmd = requestContextValues.systemDatetime().toLocalDate();

        GmtsetdRecord gmtsetdRecord = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000056));

        if (!openingStatus.equals(gmtsetdRecord.gmtKhsJtkb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000090);
        }

        RetailTypeJudgmentOutput retailTypeJudgmentOutput = executeRetailTypeJudgment(gasMtstNo, knskKjnYmd);
        if (!isSupportedRetailType(retailTypeJudgmentOutput)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000057);
        }

        GasUsageCodeRegistrationAvailabilityJudgmentOutput registrationAvailabilityOutput =
                gasUsageCodeRegistrationAvailabilityJudgment.execute(gasYtCd);
        if (!Boolean.TRUE.equals(registrationAvailabilityOutput.gasYtCdTkhFlg())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000058);
        }

        if (Objects.equals(gmtsetdRecord.gasYtCd(), gasYtCd)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000059);
        }

        int updatedCount = gmtsetdDataAccess.updateByGasMtstNoAndTyenYmd(
                buildGmtsetdWriteRecord(gmtsetdRecord, gasYtCd),
                GMTSETD_TARGET_COLUMNS,
                new GmtsetdTyenUpdateKey(gasMtstNo, gmtsetdRecord.tyenYmd()),
                AuditContext.of(SERVICE_USER_ID, requestContextValues.decodedOriginFunctionId()));
        ensureUpdatedCount(
                updatedCount,
                gasMtstNo,
                gmtsetdRecord.tyenYmd(),
                gasYtCd,
                requestContextValues.decodedOriginFunctionId());

        ConstantMasterInfoAcquisitionOutput beforeUsageCodeInfo =
                constantMasterInfoAcquisition.execute(GAS_USAGE_CODE_TYPE, gmtsetdRecord.gasYtCd());
        String reportBeforeInfo = normalizeText(beforeUsageCodeInfo == null ? null : beforeUsageCodeInfo.cntn());
        String reportAfterInfo = normalizeText(registrationAvailabilityOutput.cntnMix());

        doukanAdapter.register(buildDoukanRequest(
                gasMtstNo,
                shrYmd,
                akrkKbn,
                akrkTrskShbt,
                duknIsrgCd,
                duknIsrsCd,
                isrrHkkrKbn,
                isrrSktCd,
                requestContextValues.systemDatetime(),
                reportBeforeInfo,
                reportAfterInfo));
    }

    private RetailTypeJudgmentOutput executeRetailTypeJudgment(Long gasMtstNo, LocalDate knskKjnYmd) {
        try {
            return retailTypeJudgment.execute(SHR_KBN_OTHER, gasMtstNo, knskKjnYmd);
        } catch (RuntimeException ignored) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000057);
        }
    }

    private boolean isSupportedRetailType(RetailTypeJudgmentOutput output) {
        if (output == null) {
            return false;
        }
        String retailType = output.kuHntKbn();
        return RetailTypeJudgmentCode.TOKYO_GAS.code().equals(retailType)
                || RetailTypeJudgmentCode.OTHER.code().equals(retailType);
    }

    private RequestContextValues requireRequestContextValues() {
        String hukkSysShbt = requestContext
                .getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        LocalDateTime systemDatetime = requestContext
                .getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        String decodedOriginFunctionId = Objects.requireNonNull(
                kaiheisenConstantService.findNameByTypeAndValue(
                        ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                        hukkSysShbt),
                "decodedOriginFunctionId");
        return new RequestContextValues(systemDatetime, decodedOriginFunctionId);
    }

    private void ensureUpdatedCount(
            int updatedCount,
            Long gasMtstNo,
            LocalDate tyenYmd,
            String gasYtCd,
            String decodedOriginFunctionId) {
        if (updatedCount == 1) {
            return;
        }
        throw DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("tyenYmd", tyenYmd)
                        .put("gasYtCd", gasYtCd)
                        .put("updUsrId", SERVICE_USER_ID)
                        .put("updKtnKinoId", decodedOriginFunctionId));
    }

    private IdousagyourirekiDoukanJouhouRequest buildDoukanRequest(
            Long gasMtstNo,
            LocalDate shrYmd,
            String akrkKbn,
            String akrkTrskShbt,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrHkkrKbn,
            String isrrSktCd,
            LocalDateTime systemDatetime,
            String reportBeforeInfo,
            String reportAfterInfo) {
        Objects.requireNonNull(systemDatetime, "systemDatetime");
        Objects.requireNonNull(reportBeforeInfo, "reportBeforeInfo");
        Objects.requireNonNull(reportAfterInfo, "reportAfterInfo");
        return new IdousagyourirekiDoukanJouhouRequest(
                new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekiDoukanJouhouRenkei(
                        new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekigyoumuKyoutsuuDoukan(
                                formatCustomerNumber(gasMtstNo),
                                akrkKbn,
                                akrkTrskShbt,
                                duknIsrgCd,
                                duknIsrsCd,
                                isrrHkkrKbn,
                                isrrSktCd,
                                formatBasicDate(systemDatetime.toLocalDate()),
                                formatTime(systemDatetime),
                                formatBasicDate(shrYmd),
                                "",
                                "",
                                ""),
                        List.of(new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekiShousaiDoukan(
                                BigDecimal.ONE,
                                HISTORY_ITEM_NAME,
                                normalizeText(reportBeforeInfo),
                                normalizeText(reportAfterInfo))),
                        customerInfoAcquisitionPossible,
                        normalizeText(formatBasicDate(shrYmd))));
    }

    private String normalizeText(String value) {
        return value == null ? "" : value;
    }

    private String formatCustomerNumber(Long value) {
        return value == null ? null : "%011d".formatted(value);
    }

    private String formatBasicDate(LocalDate value) {
        return value == null ? null : BASIC_DATE_FORMATTER.format(value);
    }

    private String formatTime(LocalDateTime value) {
        return value == null ? null : TIME_FORMATTER.format(value.toLocalTime());
    }

    private record RequestContextValues(
            LocalDateTime systemDatetime,
            String decodedOriginFunctionId) {
    }
