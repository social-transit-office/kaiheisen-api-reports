/**
 * 概要: SV-082 ガスメーター設置場所情報追加のサービス実装。
 * 主な仕様:
 * - GMTSETD、MTSETK、GMTSETK の 3 つの DA をこの順で逐次呼び出して登録する。
 * - 正常終了時は返却値を持たずに完了する。
 * - RequestContext から監査情報の前提値を取得し、hukkSysShbt を起点に起動機能 ID を解決する。
 * - CIRIUS 定数で既定値を補完し、入力が null または blank の場合のみ差し替える。
 * - 下位 DA や定数解決で発生した例外は握りつぶさずそのまま伝播する。
 * 既知の制約:
 * - correlationId、hukkSysShbt、systemDatetime が RequestContext に設定されていない場合は失敗する。
 * - 監査列や更新列の実際の設定は DA 実装側の責務であり、本実装では write record の該当項目に null を渡す。
 */
public class GasMeterLocationInfoAdditionImpl implements GasMeterLocationInfoAddition {

    static final String SERVICE_USER_ID = "KHSS082";

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final MtsetkDataAccess mtsetkDataAccess;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String unspecifiedNewGasConstructionType;

    private final String supplyCanBeStopped;

    private final String noCutting;

    private final String nonSampleConsumer;

    private final String blankGasUsageCode;

    private final String tokyoGasOperatorCode;

    private final String undecidedHousing;

    /**
     * 依存オブジェクトと必須定数を初期化する。
     *
     * @param gmtsetdDataAccess GMTSETD データアクセス
     * @param mtsetkDataAccess MTSETK データアクセス
     * @param gmtsetkDataAccess GMTSETK データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public GasMeterLocationInfoAdditionImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            MtsetkDataAccess mtsetkDataAccess,
            GmtsetkDataAccess gmtsetkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.mtsetkDataAccess = Objects.requireNonNull(mtsetkDataAccess, "mtsetkDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.unspecifiedNewGasConstructionType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.NewGasConstructionType.TYPE,
                ConstantLookupKeys.Cirius.NewGasConstructionType.UNSPECIFIED);
        this.supplyCanBeStopped = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.WhetherOrNotSupplyStoppageWorkIsPossible.TYPE,
                ConstantLookupKeys.Cirius.WhetherOrNotSupplyStoppageWorkIsPossible.SUPPLY_CAN_BE_STOPPED);
        this.noCutting = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CuttingPointType.TYPE,
                ConstantLookupKeys.Cirius.CuttingPointType.NO_CUTTING);
        this.nonSampleConsumer = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SampleCode.TYPE,
                ConstantLookupKeys.Cirius.SampleCode.NON_SAMPLE_CONSUMER);
        this.blankGasUsageCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasUsageCode.TYPE,
                ConstantLookupKeys.Cirius.GasUsageCode.BLANK);
        this.tokyoGasOperatorCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.NwOperatorCode.TYPE,
                ConstantLookupKeys.Cirius.NwOperatorCode.TOKYO_GAS_CO_LTD);
        this.undecidedHousing = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Housing.TYPE,
                ConstantLookupKeys.Cirius.Housing.UNDECIDED);
    }

    /**
     * 開栓時のガスメーター設置場所情報を登録する。
     *
     * @param gmtsetdInsInf ガスメーター設置場所＜導管＞情報
     * @param gmtsetkInsInf ガスメーター設置場所＜小売＞付帯情報
     * @param mtsetkInsInf メーター設置場所＜小売＞情報
     */
    @Override
        public void execute(
            GmtsetdInsInf gmtsetdInsInf,
            GmtsetkInsInf gmtsetkInsInf,
            MtsetkInsInf mtsetkInsInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        gmtsetdDataAccess.insert(buildGmtsetdWriteRecord(gmtsetdInsInf), auditContext);
        mtsetkDataAccess.insert(buildMtsetkWriteRecord(mtsetkInsInf), auditContext);
        gmtsetkDataAccess.insert(buildGmtsetkWriteRecord(gmtsetdInsInf, gmtsetkInsInf), auditContext);
    }

        /**
         * 概要: GMTSETD 登録用の write record を組み立てる。
         * 主な仕様:
         * - 導管向け入力を GMTSETD の登録項目へそのまま写像する。
         * - 新設ガス工事種別、供給停止作業可否区分、抜栓切断点種別、サンプルコードは既定値補完する。
         * - 監査列に相当する末尾の更新項目は本実装では設定せず null を渡す。
         * 既知の制約:
         * - 入力値の妥当性検証や桁チェックは呼び出し元および下位 DA の責務とする。
         *
         * @param input ガスメーター設置場所＜導管＞情報

        return new GmtsetdWriteRecord(

                input.szcMchCd(),
                input.szcCmazCd(),
                input.cbgSuj1(),
                input.cbgKn2(),
                input.cbgKn3(),
                input.cbgKn4(),
                input.bldNo(),
                input.tunoKn(),
                input.hynKn(),
                input.htuFlg(),
                input.ohmKj(),
                input.kasYmd(),
                input.hesnYmd(),
                input.kasJysb(),
                input.hesnJysb(),
                input.tkshGmsbShbt(),
                input.gtyMtscKbn(),
                input.tksuMtscCd(),
                input.kkyuGmtKbn(),
                input.ckrtGmscJtkb(),
                input.mtgMtstNo(),
                input.snstGkjYm(),
                defaultIfBlank(input.snstGkjShbt(), unspecifiedNewGasConstructionType),

    }

        /**
         * 概要: MTSETK 登録用の write record を組み立てる。
         * 主な仕様:
         * - 小売向けメーター設置場所入力を MTSETK の登録項目へ写像する。
         * - NW 事業者コードと住宅区分は null または blank の場合に既定値補完する。
         * - 監査列に相当する末尾の更新項目は本実装では設定せず null を渡す。
         * 既知の制約:

        /**
         * 概要: RequestContext の報告システム種別から起動機能 ID を解決する。
         * 主な仕様:
         * - hukkSysShbt を取得し、開閉栓定数サービスで機能 ID 名称へ変換する。
         * - 監査情報作成時に利用する起動機能 ID を返却する。
         * 既知の制約:
         * - hukkSysShbt の値設定と妥当性確認は上位層責務とし、本メソッドでは補完しない。
         *
         * @return 起動機能 ID
         */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                                requestContext.getHukkSysShbt()
                                                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }

        /**
         * 概要: 文字列が null または blank の場合に既定値へ補完する。
         * 主な仕様:
         * - value が null または blank の場合のみ defaultValue を返す。
         * - value が非 blank の場合は入力値をそのまま返す。
         * 既知の制約:
         * - trim は行わないため、blank 判定は String.isBlank の結果に依存する。
         *
         * @param value 判定対象の値
         * @param defaultValue 既定値
         * @return 補完後の値
         */
    private String defaultIfBlank(String value, String defaultValue) {
        if (value == null || value.isBlank()) {
            return defaultValue;
        }
        return value;
    }
