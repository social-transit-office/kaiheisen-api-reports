/**
 * 概要: SV-256（異動・作業履歴連携管理当日取消）のサービス実装。
 * 主な仕様:
 * - `DA-158-R001` で未当日取消の異動・作業履歴連携管理を最新 1 件取得する。
 * - 対象が見つかった場合は `DA-158-U002` で当日取消フラグを更新する。
 * - 対象が見つからない場合は処理対象データ無しとしてデータ不整合例外を送出する。
 * 既知の制約:
 * - 実行時には RequestContext の `hukkSysShbt` が必須である。
 * - `J0253 1:有` はコンストラクタ初期化時に必須定数として検証する。
 */
public class TransferWorkHistoryIntegrationManagementSameDayCancelImpl
        implements TransferWorkHistoryIntegrationManagementSameDayCancel {

    static final String SERVICE_USER_ID = "KHSS256";

    private final IsrrkrDataAccess isrrkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String sameDayCancelFlag;

    /**
     * 概要: 依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - ISRRKR データアクセスと RequestContext を保持する。
     * - `J0253 1:有` をコンストラクタ初期化時に解決する。
     *
     * @param isrrkrDataAccess ISRRKR データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public TransferWorkHistoryIntegrationManagementSameDayCancelImpl(
            IsrrkrDataAccess isrrkrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.isrrkrDataAccess = Objects.requireNonNull(isrrkrDataAccess, "isrrkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.sameDayCancelFlag = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PresenceFlag.TYPE,
                ConstantLookupKeys.Cirius.PresenceFlag.YES);
    }

    /**
     * 概要: 指定された異動・作業履歴連携管理を当日取消する。
     * 主な仕様:
     * - 対象が見つからない場合は `DA-158-R001` 起点のデータ不整合例外を送出する。
     * - 対象が見つかった場合は `DA-158-U002` で当日取消フラグを更新する。
     */
    @Override
    public void execute(Long cstNo, LocalDate isrrTriYmd, String kytuIsrrGymCd, String kytuIsrrGymShsiCd) {
        requireRequestContext();

        Optional<IsrrkrRecord> foundRecord = isrrkrDataAccess
                .findByCstNoAndIsrrTriYmdAndIsrrGymCdAndIsrrGmssCdAndNotTjtTrksFlg(
                        cstNo,
                        isrrTriYmd,
                        kytuIsrrGymCd,
                        kytuIsrrGymShsiCd,
                        sameDayCancelFlagAsBoolean());
        if (foundRecord.isEmpty()) {
            throw DataInconsistencySupport.create(
                    isrrkrDataAccess.tableName(),
                    IsrrkrDataAccess.CallSpecs.R001.identifier(),
                    new FlatJsonBuilder()
                            .put("cstNo", cstNo)
                            .put("isrrTriYmd", isrrTriYmd)
                            .put("isrrGymCd", kytuIsrrGymCd)
                            .put("isrrGmssCd", kytuIsrrGymShsiCd)
                            .put("tjtTrksFlg", sameDayCancelFlagAsBoolean()));
        }

        IsrrkrRecord currentRecord = foundRecord.orElseThrow();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        isrrkrDataAccess.updateByCstNoAndIsrrTriYmdAndIsrrRkkrTsnoAndRecUpdJnjNo(
                buildUpdateRecord(),
                Set.of(IsrrkrUpdatableColumn.TJT_TRKS_FLG),
                new IsrrkrU002UpdateKey(
                        currentRecord.cstNo(),
                        currentRecord.isrrTriYmd(),
                        currentRecord.isrrRkkrTsno(),
                        currentRecord.recUpdJnjNo()),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    /**

     * 主な仕様:
     * - 更新対象は当日取消フラグのみとする。

    /**
     * 概要: 実行に必要な RequestContext の必須値を検証する。
     * 主な仕様:
     * - `hukkSysShbt` の存在を確認する。
     */
    private void requireRequestContext() {
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    /**
     * 概要: RequestContext の報告システム種別から監査用の起点機能 ID を取得する。
     * 主な仕様:
     * - `KH001` の報告システム種別を名称へデコードする。
     * - デコード結果が存在しない場合は定数未定義例外を送出する。
     *
     * @return 監査用の起点機能 ID
     */
    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decodedOriginFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decodedOriginFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE
                            + ", value=" + hukkSysShbt);
        }
        return decodedOriginFunctionId;
    }

    private boolean sameDayCancelFlagAsBoolean() {
        return ConstantLookupKeys.Cirius.PresenceFlag.YES.equals(sameDayCancelFlag);
    }
