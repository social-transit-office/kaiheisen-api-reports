/**
 * 概要: SV-262（移転先管理追加）のサービス実装。
 * 主な仕様:
 * - 未取消の移転先管理が既に存在する場合は登録を行わず終了する。
 * - 未登録時は fallback 検索で移転先管理通し番号を採番し、ITENSAK へ登録する。
 * - 主キー重複は握りつぶし、それ以外の DB エラーは `030000001` の DatabaseException として送出する。
 * 既知の制約:
 * - 補助 TSV に定義された固定値はコンストラクタ初期化時に定数存在確認を行う。
 */

     * - 補助 TSV に記載された固定値定数をコンストラクタ初期化時に解決する。
     *
     * @param itensakDataAccess 移転先管理データアクセス
     * @param kaiheisenConstantService 開閉栓定数サービス

    /**
     * 概要: 移転先管理の追加要否を判定して必要な場合のみ新規登録する。
     * 主な仕様:
     * - `DA-004-R003` で既存レコードが見つかった場合は何もせず終了する。
     * - `DA-004-R001` の先頭レコードから次通し番号を決定する。
     * - `DA-004-C001` の主キー重複は正常継続、それ以外の DB エラーは `030000001` に正規化して送出する。
     */
    @Override
    public void execute(
            Long gmtSetNo,
            Long syokyNo,
            Long itmSihKiyNo,
            String itmKktSsyaCd,
            String itmKktEgsyCd,
            String itmKktKitiCd) {
        List<ItensakRecord> existingRecords = itensakDataAccess.findAllByGmtSetNoAndSyokyNoAndItenTksCd(
                gmtSetNo,
                syokyNo,
                transferDestinationCancellationNoConcept);
        if (!existingRecords.isEmpty()) {
            return;
        }

        int nextSequenceNo = resolveNextSequenceNo(gmtSetNo, syokyNo);
        ItensakWriteRecord writeRecord = buildWriteRecord(
                gmtSetNo,
                syokyNo,
                nextSequenceNo,
                itmSihKiyNo,
                itmKktSsyaCd,
                itmKktEgsyCd,
                itmKktKitiCd);
        try {
            itensakDataAccess.insert(writeRecord, AuditContext.of(SERVICE_USER_ID));
        } catch (DatabaseException databaseException) {
            if (databaseException instanceof DuplicateKeyException) {
                return;
            } else {
                throw databaseException;
            }
        }
    }

    /**
     * 概要: fallback 検索結果から次の移転先管理通し番号を決定する。
     * 主な仕様:
     * - `DA-004-R001` が 0 件の場合は 1 を返す。
     * - 1 件以上ある場合は最大の `itenSakSeqNo + 1` を返す。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     * @param syokyNo 使用契約番号
     * @return 次に使用する移転先管理通し番号
     */
    private int resolveNextSequenceNo(Long gmtSetNo, Long syokyNo) {
        List<ItensakRecord> fallbackRecords = itensakDataAccess.findAllByGmtSetNoAndSyokyNo(gmtSetNo, syokyNo);
        return fallbackRecords.stream()
                .map(ItensakRecord::itenSakSeqNo)
                .filter(Objects::nonNull)
                .max(Integer::compareTo)
                .orElse(0) + 1;
    }

    /**
     * 概要: DA-004-C001 へ渡す write record を組み立てる。
     * 主な仕様:
     * - 補助 TSV に記載された固定値と入力値を ITENSAK の登録 DTO へ写像する。
     * - 補助 TSV に未記載の項目は null を維持する。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     * @param syokyNo 使用契約番号
     * @param nextSequenceNo 次の移転先管理通し番号
     * @param itmSihKiyNo 移転前支払契約番号
     * @param itmKktSsyaCd 移転前管轄支社コード
     * @param itmKktEgsyCd 移転前管轄営業所コード
     * @param itmKktKitiCd 移転前管轄基地コード
     * @return ITENSAK 登録用 write record
     */

            Long gmtSetNo,
            Long syokyNo,
            int nextSequenceNo,
            Long itmSihKiyNo,
            String itmKktSsyaCd,
            String itmKktEgsyCd,
            String itmKktKitiCd) {
        return new ItensakWriteRecord(
                gmtSetNo,
                syokyNo,
                nextSequenceNo,
                null,
                null,
                null,
                null,
                null,
                null,
                noPhone,
                null,
                relocationAddressUnderInvestigation,
                noRoomNumberRegistration,
                null,
                blankContactType,
                null,
                null,
                null,
                null,
                null,
                blankContactType,
                null,
                itmSihKiyNo,
                noInformation,
                noInformation,
                noOtherPaymentCollectionRequest,
                branchOfficeSwitchingNoConcept,
                itmKktSsyaCd,
                itmKktEgsyCd,
                itmKktKitiCd,
                branchNoConcept,
                officeNoConcept,
                baseNoConcept,
                relocationDestinationUnknown,
                relocationInformationNotRegistered,
                null,
                ownerEtcRelocationDestinationInvestigationNotInvestigated,
                null,
                relocationDestinationGovernmentOfficeInvestigationNotInvestigated,
                null,
                relocationTelephoneContactNotInvestigated,
                null,
                relocationSiteVisitNotInvestigated,
                null,

                null,
                null,
                null);
    }
