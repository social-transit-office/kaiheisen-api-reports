/**
 * 概要: SV-309（解約処理時の供給停止管理更新および検針結果登録）のサービス実装。
 * 主な仕様:
 * - 供給停止精算状態区分が供給停止精算保留の場合のみ、供給停止管理取得、確定前検針結果登録、供給停止管理更新を実施する。
 * - `DA-055-R002` 0 件と `DA-055-U001` 更新 0 件は `DataInconsistencyException` を送出する。
 * - `DA-052-R002` 0 件時は検針結果通し番号 `1`、取得あり時は `既存 + 1` を採用する。
 */
public final class SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellationImpl
        implements SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellation {

    static final String SERVICE_USER_ID = "KHSS309";

    private final KkytskrDataAccess kkytskrDataAccess;

    private final KtmgmkkDataAccess ktmgmkkDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String supplyStopPaymentPendingStatus;

    private final String supplyOutageStatus;

    private final String supplyOutageReasonType;

    private final String noChangeCode;

    private final String onlineReportingMethodCode;

    private final String irregularMeterReadingCode;

    private final String actualInspectionStatusCode;

    /**
     * 必須依存と定数を受け取ってサービスを初期化する。
     *
     * @param kkytskrDataAccess 供給停止管理 DA
     * @param ktmgmkkDataAccess 確定前ガスメーター検針結果 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellationImpl(
            KkytskrDataAccess kkytskrDataAccess,
            KtmgmkkDataAccess ktmgmkkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.kkytskrDataAccess = Objects.requireNonNull(kkytskrDataAccess, "kkytskrDataAccess");
        this.ktmgmkkDataAccess = Objects.requireNonNull(ktmgmkkDataAccess, "ktmgmkkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.supplyStopPaymentPendingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_STOP_PAYMENT_PENDING);
        this.supplyOutageStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.TYPE,
                ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_OUTAGE);
        this.supplyOutageReasonType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.TYPE,
                ConstantLookupKeys.Cirius.ReasonTypeForCreatingMeterReadingResults.SUPPLY_OUTAGE);
        this.noChangeCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChangeDetection.TYPE,
                ConstantLookupKeys.Cirius.ChangeDetection.NO_CHANGE);
        this.onlineReportingMethodCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MeterReadingResultReportingMethod.TYPE,
                ConstantLookupKeys.Cirius.MeterReadingResultReportingMethod.ONLINE);
        this.irregularMeterReadingCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.RegularMeterReading.TYPE,
                ConstantLookupKeys.Cirius.RegularMeterReading.IRREGULAR);
        this.actualInspectionStatusCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.MeterReadingImplementationStatusCode.TYPE,
                ConstantLookupKeys.Cirius.MeterReadingImplementationStatusCode
                        .ACTUAL_INSPECTION_NORMAL_METER_READING);
    }

    @Override
    public void execute(
            Long gasMtstNo,
            String kkysSisnJtkb,
            SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellationGmtsetkInf gmtsetkInf,
            String knsHuhuKbn,
            Long tksuSvkyNo) {
        if (!supplyStopPaymentPendingStatus.equals(kkysSisnJtkb)) {
            return;
        }

        RequestState requestState = requireRequestState();
        KkytskrRecord supplyStopManagement = kkytskrDataAccess.findByGasMtstNoAndHesnYmd(
                        gasMtstNo,
                        gmtsetkInf.hesnYmd())
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kkytskrDataAccess.tableName(),
                        KkytskrDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_HESN_YMD.identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("hesnYmd", gmtsetkInf.hesnYmd())));

        int nextSequenceNo = ktmgmkkDataAccess.findByGasMtstNoAndKnsYmd(gasMtstNo, gmtsetkInf.hesnYmd())
                .map(record -> record.knsKkaTsno() + 1)
                .orElse(1);

        ktmgmkkDataAccess.insert(
                buildKtmgmkkWriteRecord(
                        supplyStopManagement,
                        nextSequenceNo,
                        gmtsetkInf,
                        knsHuhuKbn,
                        tksuSvkyNo,
                        requestState),
                AuditContext.of(SERVICE_USER_ID, requestState.originFunctionId()));

        int updatedCount = kkytskrDataAccess.updateByGasMtstNoAndHesnYmd(
                buildKkytskrWriteRecord(),
                EnumSet.of(KkytskrColumn.KKYS_SISN_JTKB),
                new KkytskrHesnUpdateKey(supplyStopManagement.gasMtstNo(), gmtsetkInf.hesnYmd()),
                AuditContext.of(SERVICE_USER_ID, requestState.originFunctionId()));
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    kkytskrDataAccess.tableName(),
                    KkytskrDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", supplyStopManagement.gasMtstNo())
                            .put("hesnYmd", gmtsetkInf.hesnYmd())
                            .put("kkysSisnJtkb", supplyOutageStatus)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", requestState.originFunctionId()));
        }
    }

    private KtmgmkkWriteRecord buildKtmgmkkWriteRecord(
            KkytskrRecord supplyStopManagement,
            int nextSequenceNo,
            SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellationGmtsetkInf gmtsetkInf,
            String knsHuhuKbn,
            Long tksuSvkyNo,
            RequestState requestState) {
        return new KtmgmkkWriteRecord(
                supplyStopManagement.gasMtstNo(),
                gmtsetkInf.hesnYmd(),
                nextSequenceNo,
                null,
                supplyOutageReasonType,
                null,
                supplyStopManagement.gmtShjSu(),
                supplyStopManagement.gasSyry(),
                null,
                supplyStopManagement.shyDdsu(),
                supplyStopManagement.sjskKbn(),
                noChangeCode,
                onlineReportingMethodCode,
                supplyStopManagement.sgttKknnFlg(),
                irregularMeterReadingCode,
                actualInspectionStatusCode,
                gmtsetkInf.gmtKhsJtkb(),
                knsHuhuKbn,
                gmtsetkInf.ckrtGmscJtkb(),
                gmtsetkInf.mtgMtstNo(),
                null,
                null,
                supplyStopManagement.gmb(),
                supplyStopManagement.gmbTsno(),
                tksuSvkyNo,
                supplyStopManagement.sgttUsrId(),
                null,
                null,
                null,
                null,
                null,
                null,
                requestState.systemDate(),
                false,
                null);
    }

    private KkytskrWriteRecord buildKkytskrWriteRecord() {
        return new KkytskrWriteRecord(
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                supplyOutageStatus,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private RequestState requireRequestState() {
        var systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"))
                .toLocalDate();
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String originFunctionId = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (originFunctionId == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return new RequestState(systemDate, originFunctionId);
    }

    private record RequestState(
            java.time.LocalDate systemDate,
            String originFunctionId) {
    }
}
