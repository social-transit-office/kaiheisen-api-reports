/**
 * オンライン解約報告用解約共通処理の実装。
 */
public final class CancellationCommonProcessCommonImpl
    implements CancellationCommonProcessCommon {

    private final CancellationCommonProcessBatchCommonTgRetail sv326Process;

    private final CancellationCommonProcessCommonOtherTgRetail sv328Process;

    private final String batchFinalCloseServiceType;

    private final String batchNextThreeTempServiceType;

    /**
     * コンストラクタ。
     *
     * @param sv326ProcessDependency SV-326
     * @param sv328ProcessDependency SV-328
     * @param ciriusConstantServiceDependency CIRIUS コンスタント取得サービス
     */
    public CancellationCommonProcessCommonImpl(
            final CancellationCommonProcessBatchCommonTgRetail sv326ProcessDependency,
            final CancellationCommonProcessCommonOtherTgRetail sv328ProcessDependency,
            final CiriusConstantService ciriusConstantServiceDependency) {
        this.sv326Process = Objects.requireNonNull(
                sv326ProcessDependency,
                "sv326Process");
        this.sv328Process = Objects.requireNonNull(
                sv328ProcessDependency,
                "sv328Process");
        CiriusConstantService ciriusConstantService = Objects.requireNonNull(
                ciriusConstantServiceDependency,
                "ciriusConstantService");
        this.batchFinalCloseServiceType = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                ConstantLookupKeys.Cirius.BillingServiceType.ALL_AT_ONCE_FINAL_INSTALLMENT);
        this.batchNextThreeTempServiceType = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.BillingServiceType.TYPE,
                ConstantLookupKeys.Cirius.BillingServiceType
                        .ALL_AT_ONCE_NEXT_3RD_BUSINESS_YEAR_CLOSING_AVAILABLE_AT_SHORT_NOTICE);
    }

    /**
     * 請求サービス種別に応じて SV-326 / SV-328 を dispatch する。
     */
    @Override
    @SuppressWarnings("checkstyle:ParameterNumber")
    public void execute(
            final Long gasMtstNo,
            final LocalDate tyenYmd,
            final String hesnJysb,
            final String tksuSvkeJysb,
            final String svkyEndJysb,
            final CancellationGmtsetdInf gmtsetdInf,
            final TskgmstInf tskgmstInf,
            final TkssvkyInf tkssvkyInf,
            final SvmstkrInf svmstkrInf,
            final SvckyaInf svckyaInf,
            final SvsihkrInf svsihkrInf,
            final SihakyInf sihakyInf,
            final List<SnyInf> snyInfList,
            final List<SogoTmtrInf> sogoTmtrInfList,
            final String isrrGymCd,
            final String isrrHusrId,
            final String isrrHkunKj,
            final String isrrHkkrKbn) {
        SihakyInf requiredSihakyInf = Objects.requireNonNull(sihakyInf, "sihakyInf");
        if (isBatchServiceType(requiredSihakyInf.sikSvcShbt())) {
            sv326Process.execute(
                    gasMtstNo,
                    tyenYmd,
                    hesnJysb,
                    tksuSvkeJysb,
                    svkyEndJysb,
                    toSv326GmtsetdInf(gmtsetdInf),
                    toSv326TskgmstInf(tskgmstInf),
                    toSv326TkssvkyInf(tkssvkyInf),
                    toSv326SvmstkrInf(svmstkrInf),
                    svckyaInf,
                    toSv326SvsihkrInf(svsihkrInf),
                    requiredSihakyInf,
                    toSv326SnyInfList(snyInfList),
                    toSv326SogoTmtrInfList(sogoTmtrInfList),
                    isrrGymCd,
                    isrrHusrId,
                    isrrHkunKj,
                    isrrHkkrKbn);
            return;
        }

        sv328Process.execute(
                gasMtstNo,
                tyenYmd,
                hesnJysb,
                tksuSvkeJysb,
                svkyEndJysb,
                gmtsetdInf,
                tskgmstInf,
                tkssvkyInf,
                svmstkrInf,
                svckyaInf,
                svsihkrInf,
                requiredSihakyInf,
                snyInfList,
                sogoTmtrInfList,
                isrrGymCd,
                isrrHusrId,
                isrrHkunKj,
                isrrHkkrKbn);
    }

    private boolean isBatchServiceType(final String sikSvcShbt) {
        return batchFinalCloseServiceType.equals(sikSvcShbt)
                || batchNextThreeTempServiceType.equals(sikSvcShbt);
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf toSv326GmtsetdInf(
            final CancellationGmtsetdInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.GmtsetdInf(
                input.szcGkkCd(),
                input.szcCmazCd(),
                input.szcMchCd(),
                input.ckrtGmscJtkb());
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TskgmstInf toSv326TskgmstInf(
            final TskgmstInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TskgmstInf(
                input.tksuSvkyNo(),
                input.skno(),
                input.recUpdJnjNo());
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TkssvkyInf toSv326TkssvkyInf(
            final TkssvkyInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TkssvkyInf(
                input.tksuSvkyNo(),
                input.skno() == null ? null : Long.valueOf(input.skno()),
                input.recUpdJnjNo());
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SvmstkrInf toSv326SvmstkrInf(
            final SvmstkrInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SvmstkrInf(
                input.svkyNo(),
                input.skno(),
                input.recUpdJnjNo());
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SvsihkrInf toSv326SvsihkrInf(
            final SvsihkrInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SvsihkrInf(
                input.sihKiyNo(),
                input.svkyNo(),
                input.skno(),
                input.ikskEndYm(),
                input.recUpdJnjNo());
    }

    private List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf> toSv326SnyInfList(
            final List<SnyInf> input) {
        if (input == null) {
            return null;
        }
        return input.stream()
                .map(this::toSv326SnyInf)
                .toList();
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf toSv326SnyInf(
            final SnyInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf(
                toSv326DedicatedGmtsetdInf(input.gmtsetdInf()));
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransportServiceContractGmtsetdInf
            toSv326DedicatedGmtsetdInf(final CancellationGmtsetdInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransportServiceContractGmtsetdInf(
                input.gasMtstNo());
    }

    private List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SogoTmtrInf> toSv326SogoTmtrInfList(
            final List<SogoTmtrInf> input) {
        if (input == null) {
            return null;
        }
        return input.stream()
                .map(this::toSv326SogoTmtrInf)
                .toList();
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SogoTmtrInf toSv326SogoTmtrInf(
            final SogoTmtrInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SogoTmtrInf(
                toSv326SogoSvmstkrInf(input.svmstkrInf()),
                toSv326GmtsetdInf(input.gmtsetdInf()));
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.ComprehensiveContractSvmstkrInf
            toSv326SogoSvmstkrInf(final SogoSvmstkrInf input) {
        if (input == null) {
            return null;
        }
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.ComprehensiveContractSvmstkrInf(
                input.mtstNo(),
                input.svkyNo(),
                input.skno(),
                input.recUpdJnjNo());
    }
