package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.KyoteldDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.KyoteldRecord;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.PhoneNumberAdditionalUpdatePipeline;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.PhoneNumberUpdateAddDeleteFunctionPipeline;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.PipelinePhoneNumberRegistration;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.NewTelInf;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.TelnoUpdInf;

/**
 * 概要: 導管用電話番号登録サービスの実装です。
 *
 * 主な仕様:
 * 旧電話番号と新電話番号の組み合わせを正規化し、既存電話番号の有無と属性差分に応じて
 * 電話番号削除機能または電話番号追加更新機能を呼び分けます。
 *
 * 既知の制約:
 * 削除処理では下位 SV の仕様に合わせ、電話番号以外の属性は引き渡していません。
 * 正規化処理の入れ替え考慮は先頭 2 件の更新情報を対象とした相互参照に限定しています。
 */
public final class PipelinePhoneNumberRegistrationImpl implements PipelinePhoneNumberRegistration {

    /** 契約お客さま電話番号＜導管＞を参照するデータアクセスです。 */
    private final KyoteldDataAccess kyoteldDataAccess;

    /** 導管電話番号の更新・追加・削除機能を呼び出す下位 SV です。 */
    private final PhoneNumberUpdateAddDeleteFunctionPipeline phoneNumberUpdateAddDeleteFunctionPipeline;

    /** 導管用電話番号追加更新を呼び出す下位 SV です。 */
    private final PhoneNumberAdditionalUpdatePipeline phoneNumberAdditionalUpdatePipeline;

    /** 電話番号削除時に下位 SV へ渡す処理区分です。 */
    private final String deleteProcessingDivision;

    /**
     * 概要: 依存コンポーネントを受け取ってサービス実装を初期化します。
     *
     * 主な仕様:
     * 渡された依存オブジェクトはすべて null を禁止し、生成時点で検証します。
     *
     * @param kyoteldDataAccess 契約お客さま電話番号＜導管＞データアクセス
     * @param phoneNumberUpdateAddDeleteFunctionPipeline 電話番号更新・追加・削除機能-導管
     * @param phoneNumberAdditionalUpdatePipeline 導管用電話番号追加更新
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     */
    public PipelinePhoneNumberRegistrationImpl(
            KyoteldDataAccess kyoteldDataAccess,
            PhoneNumberUpdateAddDeleteFunctionPipeline phoneNumberUpdateAddDeleteFunctionPipeline,
            PhoneNumberAdditionalUpdatePipeline phoneNumberAdditionalUpdatePipeline,
            KaiheisenConstantService kaiheisenConstantService) {
        this.kyoteldDataAccess = Objects.requireNonNull(kyoteldDataAccess, "kyoteldDataAccess");
        this.phoneNumberUpdateAddDeleteFunctionPipeline = Objects.requireNonNull(
                phoneNumberUpdateAddDeleteFunctionPipeline,
                "phoneNumberUpdateAddDeleteFunctionPipeline");
        this.phoneNumberAdditionalUpdatePipeline = Objects.requireNonNull(
                phoneNumberAdditionalUpdatePipeline,
                "phoneNumberAdditionalUpdatePipeline");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.deleteProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.DELETE);
    }

    /**
     * 概要: 導管用の電話番号登録更新を実行します。
     *
     * 主な仕様:
     * 電話番号情報を正規化したうえで各要素を順次評価し、削除または追加更新を必要最小限だけ呼び出します。
     */
    @Override
    public void execute(
            Long uketkjCstRegNo,
            Long rakKyoTrkNo,
            ArrayList<TelnoUpdInf> telnoUpdInfList) {
        List<TelnoUpdInf> normalizedEntries = normalizeEntries(!Objects.equals(uketkjCstRegNo, rakKyoTrkNo), telnoUpdInfList);
        for (TelnoUpdInf entry : normalizedEntries) {
            processEntry(rakKyoTrkNo, entry);
        }
    }

    /**
     * 概要: 電話番号更新情報を下位処理向けに正規化します。
     *
     * 主な仕様:
     * 受付時登録番号と落成時登録番号が不一致の場合は旧電話番号を無効化します。
     * 先頭2件について新旧電話番号の相互参照があれば旧電話番号を無効化します。
     *
     * 既知の制約:
     * 入れ替わり判定は2件までを前提とし、3件以上は扱いません。
     */
    private List<TelnoUpdInf> normalizeEntries(boolean invalidateOldNumbers, List<TelnoUpdInf> originalEntries) {
        ArrayList<TelnoUpdInf> normalized = new ArrayList<>(originalEntries.size());
        if (invalidateOldNumbers) {
            for (TelnoUpdInf entry : originalEntries) {
                normalized.add(new TelnoUpdInf(entry.newTelInf(), null));
            }
        } else {
            normalized.addAll(originalEntries);
        }
        if (!invalidateOldNumbers && normalized.size() >= 2) {
            TelnoUpdInf first = normalized.get(0);
            TelnoUpdInf second = normalized.get(1);
            String firstNewNo = getNewPhoneNo(first);
            String secondNewNo = getNewPhoneNo(second);
            if (first.oldKyoTelNo() != null && Objects.equals(first.oldKyoTelNo(), secondNewNo)) {
                normalized.set(0, new TelnoUpdInf(first.newTelInf(), null));
            }
            if (second.oldKyoTelNo() != null && Objects.equals(second.oldKyoTelNo(), firstNewNo)) {
                normalized.set(1, new TelnoUpdInf(second.newTelInf(), null));
            }
        }
        return normalized;
    }

    /**
     * 概要: 正規化済みの 1 件の電話番号更新情報を処理します。
     *
     * 主な仕様:
     * 旧電話番号と新電話番号の存在有無、番号一致、属性差分の有無で削除または追加更新の呼出しを分岐します。
     *
     * 既知の制約:
     * 新旧電話番号の両方が null の場合は何も行わず終了します。
     */
    private void processEntry(Long rakKyoTrkNo, TelnoUpdInf entry) {
        String oldKyoTelNo = entry.oldKyoTelNo();
        Optional<KyoteldRecord> oldRecord = findRecord(rakKyoTrkNo, oldKyoTelNo);
        boolean oldPhoneExists = oldKyoTelNo != null && oldRecord.isPresent();
        if (oldKyoTelNo != null && !oldPhoneExists) {
            oldKyoTelNo = null;
        }

        NewTelInf newTelInf = entry.newTelInf();
        String newKyoTelNo = getNewPhoneNo(entry);
        boolean newPhoneExists = determineNewPhoneExists(rakKyoTrkNo, oldKyoTelNo, newKyoTelNo, oldPhoneExists);

        if (oldKyoTelNo != null && newKyoTelNo == null) {
            if (oldPhoneExists) {
                callDeletePipelineService(rakKyoTrkNo, oldKyoTelNo);
            }
            return;
        }

        if (oldKyoTelNo == null && newKyoTelNo != null) {
            callAdditionalUpdatePipeline(newPhoneExists, rakKyoTrkNo, newTelInf);
            return;
        }

        if (oldKyoTelNo != null && newKyoTelNo != null) {
            if (Objects.equals(oldKyoTelNo, newKyoTelNo)) {
                if (oldRecord.isPresent() && hasDifferentPhoneAttributes(oldRecord.orElseThrow(), newTelInf)) {
                    callAdditionalUpdatePipeline(newPhoneExists, rakKyoTrkNo, newTelInf);
                }
                return;
            }
            if (oldPhoneExists) {
                callDeletePipelineService(rakKyoTrkNo, oldKyoTelNo);
            }
            callAdditionalUpdatePipeline(newPhoneExists, rakKyoTrkNo, newTelInf);
        }
    }

    /**
     * 概要: 新電話番号が既存登録済みかどうかを判定します。
     *
     * 主な仕様:
     * 新旧電話番号が同一の場合は旧電話番号の存在判定を再利用し、それ以外は DA を参照します。
     *
     * 既知の制約:
     * 電話番号が null の場合は未登録として扱います。
     */
    private boolean determineNewPhoneExists(
            Long rakKyoTrkNo,
            String oldKyoTelNo,
            String newKyoTelNo,
            boolean oldPhoneExists) {
        if (newKyoTelNo == null) {
            return false;
        }
        if (Objects.equals(newKyoTelNo, oldKyoTelNo)) {
            return oldPhoneExists;
        }
        return findRecord(rakKyoTrkNo, newKyoTelNo).isPresent();
    }

    /**
     * 概要: 契約番号と電話番号で既存電話情報を検索します。
     *
     * 主な仕様:
     * 電話番号が null の場合は DA を呼び出さず空結果を返します。
     *
     * 既知の制約:
     * DA 例外はここでは吸収せず、そのまま呼出元へ伝播します。
     */
    private Optional<KyoteldRecord> findRecord(Long rakKyoTrkNo, String kyoTelNo) {
        if (kyoTelNo == null) {
            return Optional.empty();
        }
        return kyoteldDataAccess.findByKyoTrkNoAndKyoTelNo(rakKyoTrkNo, kyoTelNo);
    }

    /**
     * 概要: 既存電話情報と新電話情報の属性差分を判定します。
     *
     * 主な仕様:
     * 電話種別、連絡先種別、電話確認済種別、電話取得手段種別、取得日、提供者名、電話続柄種別、電話申出区分を比較します。
     *
     * 既知の制約:
     * 新電話情報が null の場合は差分なしとして扱います。
     */
    private boolean hasDifferentPhoneAttributes(KyoteldRecord existingRecord, NewTelInf newTelInf) {
        if (newTelInf == null) {
            return false;
        }
        return !Objects.equals(existingRecord.telShbt(), newTelInf.telShbt())
                || !Objects.equals(existingRecord.telRrskShbt(), newTelInf.telRrskShbt())
                || !Objects.equals(existingRecord.telKnzShbt(), newTelInf.telKnzShbt())
                || !Objects.equals(existingRecord.telStsdShbt(), newTelInf.telStsdShbt())
                || !Objects.equals(existingRecord.telShtkYmd(), newTelInf.telShtkYmd())
                || !Objects.equals(existingRecord.tetsCtnmKj(), newTelInf.tetsCtnmKj())
                || !Objects.equals(existingRecord.telZkgrShbt(), newTelInf.telZkgrShbt())
                || !Objects.equals(existingRecord.telMusdKbn(), newTelInf.telMusdKbn());
    }

    /**
     * 概要: 旧電話番号の削除を下位 SV に委譲します。
     *
     * 主な仕様:
     * 処理区分には削除を示す固定値を使用し、電話属性は渡しません。
     *
     * 既知の制約:
     * 下位 SV の仕様上、削除対象以外の属性を補完せず null のまま引き渡します。
     */
    private void callDeletePipelineService(Long rakKyoTrkNo, String oldKyoTelNo) {
        phoneNumberUpdateAddDeleteFunctionPipeline.execute(
                deleteProcessingDivision,
                rakKyoTrkNo,
                oldKyoTelNo,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    /**
     * 概要: 新電話情報の追加または更新を下位 SV に委譲します。
     *
     * 主な仕様:
     * 新電話情報が存在する場合のみ下位 SV を呼び出し、電話属性一式をそのまま引き渡します。
     *
     * 既知の制約:
     * 新電話情報が null の場合は何も行わず終了します。
     */
    private void callAdditionalUpdatePipeline(Boolean telNoUmFlg, Long rakKyoTrkNo, NewTelInf newTelInf) {
        if (newTelInf == null) {
            return;
        }
        phoneNumberAdditionalUpdatePipeline.execute(
                telNoUmFlg,
                rakKyoTrkNo,
                newTelInf.kyoTelNo(),
                newTelInf.telShbt(),
                newTelInf.telRrskShbt(),
                newTelInf.telKnzShbt(),
                newTelInf.telStsdShbt(),
                newTelInf.telShtkYmd(),
                newTelInf.tetsCtnmKj(),
                newTelInf.telZkgrShbt(),
                newTelInf.telMusdKbn());
    }

    /**
     * 概要: 更新情報から新電話番号を取得します。
     *
     * 主な仕様:
     * 更新情報または新電話情報が null の場合は null を返します。
     *
     * 既知の制約:
     * 電話番号の形式補正やトリムは行いません。
     */
    private String getNewPhoneNo(TelnoUpdInf entry) {
        return entry == null || entry.newTelInf() == null ? null : entry.newTelInf().kyoTelNo();
    }
}
