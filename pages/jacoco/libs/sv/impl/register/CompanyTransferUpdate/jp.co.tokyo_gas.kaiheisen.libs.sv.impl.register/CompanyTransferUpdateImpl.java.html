package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.ShyfrkeDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.write.ShyfrkeColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.write.ShyfrkeUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.write.ShyfrkeWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.CompanyTransferUpdate;

/**
 * 概要: SV-030（社用振替更新）のサービス実装。
 * 主な仕様:
 * - 社用区分が社用以外の場合は更新を行わず、そのまま終了する。
 * - 社用区分が社用の場合は SHYFRKE の適用終了年月日（TYEN_YMD）を更新し、監査情報に固定更新ユーザー ID と報告元機能 ID を設定する。
 * - 更新件数が 0 件の場合はデータ不整合例外を送出する。
 * 既知の制約:
 * - RequestContext の値設定と妥当性確認は上位層責務とし、本クラスでは補完や追加検証を行わない。
 * - 社用判定に使う CIRIUS 定数はコンストラクタ初期化時に fail-fast で解決する。
 */
public class CompanyTransferUpdateImpl implements CompanyTransferUpdate {

    // 監査更新で使用する固定の更新ユーザー ID。
    static final String SERVICE_USER_ID = "KHSS030";

    // SHYFRKE 更新を担当するデータアクセス。
    private final ShyfrkeDataAccess shyfrkeDataAccess;

    // 実行時の監査・システム情報を保持するリクエストコンテキスト。
    private final RequestContext requestContext;

    // 報告元機能 ID の復号に使用する開閉栓定数サービス。
    private final KaiheisenConstantService kaiheisenConstantService;

    // 社用区分判定に使用する CIRIUS の社用定数値。
    private final String companyUseDivision;

    /**
     * 概要: 社用振替更新に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - SHYFRKE 更新、RequestContext、開閉栓定数サービスを保持する。
     * - 社用区分判定に必要な CIRIUS 定数を初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は初期化時点で NullPointerException を送出する。
     * - 必須 CIRIUS 定数が不足している場合はコンストラクタ初期化時点で失敗する。
     * @param shyfrkeDataAccess SHYFRKE データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public CompanyTransferUpdateImpl(
            ShyfrkeDataAccess shyfrkeDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.shyfrkeDataAccess = Objects.requireNonNull(shyfrkeDataAccess, "shyfrkeDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.companyUseDivision = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.Corporate.TYPE,
                ConstantLookupKeys.Cirius.Corporate.FOR_COMPANY_USE);
    }

    /**
     * 概要: 社用区分に該当する契約の振替更新を実行する。
     * 主な仕様:
     * - 実行前に RequestContext の必須値を検証する。
     * - 社用区分が社用以外の場合は更新せずそのまま終了する。
     * - 社用区分が社用の場合は指定契約番号の適用終了年月日を更新し、更新件数 0 件はデータ不整合として扱う。
     * 既知の制約:
     * - 引数の null 検証は明示的に行わず、下位処理で実行時例外となる可能性がある。
     * - 更新対象は SHYFRKE の TYEN_YMD 列に限定される。
     * @param syaKbn 社用区分
     * @param sihKiyNo 支払契約番号
     * @param updYmd 更新後の適用終了年月日
     * @param tyenYmd 検索条件に使用する適用終了年月日
     */
    @Override
    public void execute(
            String syaKbn,
            Long sihKiyNo,
            LocalDate updYmd,
            LocalDate tyenYmd) {
        if (!companyUseDivision.equals(syaKbn)) {
            return;
        }

        String decodedOriginFunctionId = decodeOriginFunctionId();
        int updatedCount = shyfrkeDataAccess.updateBySihKiyNoAndTkysYmdAndTyenYmd(
                new ShyfrkeWriteRecord(sihKiyNo, null, updYmd, null, null, null, null, null),
                Set.of(ShyfrkeColumn.TYEN_YMD),
                new ShyfrkeUpdateKey(sihKiyNo, tyenYmd),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    shyfrkeDataAccess.tableName(),
                    ShyfrkeDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_TYEN_YMD
                            .identifier(),
                    new FlatJsonBuilder()
                            .put("sihKiyNo", sihKiyNo)
                            .put("knskKjnYmd", tyenYmd)
                            .put("tyenYmd", updYmd)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    /**
     * 概要: RequestContext の報告元システム種別から監査用の機能 ID を取得する。
     * 主な仕様:
     * - RequestContext から hukkSysShbt を取得し、開閉栓定数サービスで名称へ変換する。
     * - 返却値は監査情報の updKtnKinoId に設定する。
     * 既知の制約:
        * - hukkSysShbt の値設定と妥当性確認は上位層責務とし、本メソッドでは補完しない。
     * - 定数変換結果の妥当性は KaiheisenConstantService の解決結果に依存する。
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                                requestContext.getHukkSysShbt()
                                                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
