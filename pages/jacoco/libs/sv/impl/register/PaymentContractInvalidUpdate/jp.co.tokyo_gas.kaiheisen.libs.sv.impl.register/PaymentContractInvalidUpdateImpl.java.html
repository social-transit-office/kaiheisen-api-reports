package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.FlatJsonBuilder;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.DsqkjhDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.HrikmiDataAccess;

/**
 * 概要: SV-064（支払契約関連無効更新処理）のサービス実装。
 * 主な仕様:
 * - 開栓年月日以降に有効な支払契約を取得し、各レコードを無効化更新する。
 * - 請求書表示制御管理番号が有効な場合のみ請求書表示制御を削除する。
 * - 支払方法種別に応じて関連情報 delete を 1 系統だけ実行する。
 * 主な仕様補足:
 * - 支払契約の無効化更新は `DA-075-U002` を利用する。
 * - `updUsrId` / `updKtnKinoId` は `AuditContext` で DA へ連携する。
 */
public class PaymentContractInvalidUpdateImpl implements PaymentContractInvalidUpdate {

    static final String SERVICE_USER_ID = "KHSS064";

    private final SihakyDataAccess sihakyDataAccess;

    private final SqshyjsDataAccess sqshyjsDataAccess;

    private final HrikmiDataAccess hrikmiDataAccess;

    private final KuzfrkDataAccess kuzfrkDataAccess;

    private final CrdhriDataAccess crdhriDataAccess;

    private final DsqkjhDataAccess dsqkjhDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String invalidPaymentContractStatus;

    private final String paymentSlipPaymentMethodType;

    private final String accountTransferPaymentMethodType;

    private final String cardPaymentMethodType;

    private final String electronicBillingPaymentMethodType;

    /**
     * 概要: SV-064 に必要な依存と必須定数を初期化する。
     *
     * @param sihakyDataAccess 支払契約 DA
     * @param sqshyjsDataAccess 請求書表示制御 DA
     * @param hrikmiDataAccess 払込関連情報 DA
     * @param kuzfrkDataAccess 口座振替関連情報 DA
     * @param crdhriDataAccess カード払い関連情報 DA
     * @param dsqkjhDataAccess 電子請求関連情報 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public PaymentContractInvalidUpdateImpl(
            SihakyDataAccess sihakyDataAccess,
            SqshyjsDataAccess sqshyjsDataAccess,
            HrikmiDataAccess hrikmiDataAccess,
            KuzfrkDataAccess kuzfrkDataAccess,
            CrdhriDataAccess crdhriDataAccess,
            DsqkjhDataAccess dsqkjhDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.sihakyDataAccess = Objects.requireNonNull(sihakyDataAccess, "sihakyDataAccess");
        this.sqshyjsDataAccess = Objects.requireNonNull(sqshyjsDataAccess, "sqshyjsDataAccess");
        this.hrikmiDataAccess = Objects.requireNonNull(hrikmiDataAccess, "hrikmiDataAccess");
        this.kuzfrkDataAccess = Objects.requireNonNull(kuzfrkDataAccess, "kuzfrkDataAccess");
        this.crdhriDataAccess = Objects.requireNonNull(crdhriDataAccess, "crdhriDataAccess");
        this.dsqkjhDataAccess = Objects.requireNonNull(dsqkjhDataAccess, "dsqkjhDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.invalidPaymentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentContractStatus.TYPE,
                ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID);
        this.paymentSlipPaymentMethodType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.PAYMENT);
        this.accountTransferPaymentMethodType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ACCOUNT_TRANSFER);
        this.cardPaymentMethodType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.CARD_PAYMENT);
        this.electronicBillingPaymentMethodType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PaymentMethodType.TYPE,
                ConstantLookupKeys.Cirius.PaymentMethodType.ELECTRONIC_BILLING);
    }

    @Override
    public void execute(Long sihKiyNo, LocalDate kasYmd) {
        requireRequestContextValues();
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);
        List<SihakyRecord> paymentContracts = sihakyDataAccess.findAllBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
                sihKiyNo,
                kasYmd,
                invalidPaymentContractStatus);
        if (paymentContracts.isEmpty()) {
            throw DataInconsistencySupport.create(
                    sihakyDataAccess.tableName(),
                    SihakyDataAccess.CallSpecs.FIND_ALL_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_NOT_SIH_KIY_JTKB.identifier(),
                    new FlatJsonBuilder()
                            .put("sihKiyNo", sihKiyNo)
                            .put("tkysYmd", kasYmd)
                            .put("sihKiyJtkb", invalidPaymentContractStatus));
        }
        for (SihakyRecord paymentContract : paymentContracts) {
            invalidatePaymentContract(paymentContract, auditContext, decodedOriginFunctionId);
            deleteInvoiceDisplayControlIfNeeded(paymentContract, kasYmd);
            deletePaymentMethodSpecificInfo(paymentContract);
        }
    }

    private void invalidatePaymentContract(
            SihakyRecord paymentContract,
            AuditContext auditContext,
            String decodedOriginFunctionId) {
        int updatedCount = sihakyDataAccess.updateBySihKiyNoAndSkno(
                new SihakyWriteRecord(
                        paymentContract.sihKiyNo(),
                        paymentContract.skno(),
                        null,
                        null,
                        invalidPaymentContractStatus,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null,
                        null),
                Set.of(SihakyColumn.SIH_KIY_JTKB),
                new SihakySknoUpdateKey(paymentContract.sihKiyNo(), paymentContract.skno()),
                auditContext);
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(
                    sihakyDataAccess.tableName(),
                    SihakyDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SKNO.identifier(),
                    new FlatJsonBuilder()
                            .put("sihKiyNo", paymentContract.sihKiyNo())
                            .put("skno", paymentContract.skno())
                            .put("sihKiyJtkb", invalidPaymentContractStatus)
                            .put("updUsrId", SERVICE_USER_ID)
                            .put("updKtnKinoId", decodedOriginFunctionId));
        }
    }

    private void deleteInvoiceDisplayControlIfNeeded(SihakyRecord paymentContract, LocalDate kasYmd) {
        if (paymentContract.skshHyjsKrno() == null || paymentContract.skshHyjsKrno() == 0L) {
            return;
        }
        sqshyjsDataAccess.deleteBySkshHyjsKrnoAndTkysYmd(paymentContract.skshHyjsKrno(), kasYmd, null);
    }

    private void deletePaymentMethodSpecificInfo(SihakyRecord paymentContract) {
        if (paymentSlipPaymentMethodType.equals(paymentContract.sihHuhuShbt())) {
            hrikmiDataAccess.deleteBySihKiyNoAndSihKiySkno(paymentContract.sihKiyNo(), paymentContract.skno(), null);
            return;
        }
        if (accountTransferPaymentMethodType.equals(paymentContract.sihHuhuShbt())) {
            kuzfrkDataAccess.deleteBySihKiyNoAndSihKiySkno(paymentContract.sihKiyNo(), paymentContract.skno(), null);
            return;
        }
        if (cardPaymentMethodType.equals(paymentContract.sihHuhuShbt())) {
            crdhriDataAccess.deleteBySihKiyNoAndSihKiySkno(paymentContract.sihKiyNo(), paymentContract.skno(), null);
            return;
        }
        if (electronicBillingPaymentMethodType.equals(paymentContract.sihHuhuShbt())) {
            dsqkjhDataAccess.deleteBySihKiyNoAndSihKiySkno(paymentContract.sihKiyNo(), paymentContract.skno(), null);
        }
    }

    private void requireRequestContextValues() {
        requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }
}
