package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.SvmstkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.write.SvmstkrColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.write.SvmstkrSknoUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.svmstkr.write.SvmstkrWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.OpeningCancelServiceContractMeterLocationUpdate;

/**
 * 概要: SV-059（開栓取消時サービス契約・メーター設置場所関連更新）のサービス実装。
 * 主な仕様:
 * - 主関連フラグとサービス契約開始事由種別に応じて、DA-041-U002 へ渡す更新値を切り替える。
 * - 一次分岐では料金計算関連状態区分を関連有効待ち、料金計算開始年月日を最大日付へ更新する。
 * - 二次分岐では料金計算関連状態区分を無効へ更新する。
 * 既知の制約:
 * - hukkSysShbt は RequestContext から取得し、未設定時は早期失敗する。
 * - 必須定数はコンストラクタ初期化時に fail-fast で解決する。
 */
public class OpeningCancelServiceContractMeterLocationUpdateImpl
        implements OpeningCancelServiceContractMeterLocationUpdate {

    static final String SERVICE_USER_ID = "KHSS059";

    static final LocalDate MAX_DATE = LocalDate.of(9999, 12, 31);

    private static final Set<SvmstkrColumn> PRIMARY_TARGET_COLUMNS = Set.of(
            SvmstkrColumn.RKKI_KNRN_JTKB,
            SvmstkrColumn.RKKI_STT_YMD);

    private static final Set<SvmstkrColumn> SECONDARY_TARGET_COLUMNS = Set.of(
            SvmstkrColumn.RKKI_KNRN_JTKB);

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String continuationOfPreviousContract;

    private final String relatedValidWaiting;

    private final String invalidChargeCalculationRelatedStatus;

    /**
     * 概要: SV-059 に必要な依存オブジェクトと必須定数を初期化する。
     * 主な仕様:
     * - SVMSTKR DA、RequestContext、開閉栓定数サービス、CIRIUS 定数サービスを保持する。
     * - 分岐条件と更新値に必要な J0567/J0611 定数を初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は NullPointerException を送出する。
     * - 必須定数が不足している場合はコンストラクタ初期化時点で失敗する。
     *
     * @param svmstkrDataAccess サービス契約・メーター設置場所関連 DA
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     * @param ciriusConstantService CIRIUS 定数サービス
     */
    public OpeningCancelServiceContractMeterLocationUpdateImpl(
            SvmstkrDataAccess svmstkrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.continuationOfPreviousContract = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStartReasonType.CONTINUATION_OF_PREVIOUS_CONTRACT);
        this.relatedValidWaiting = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_01);
        this.invalidChargeCalculationRelatedStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.INVALID);
    }

    /**
     * 概要: SV-059 のサービス契約・メーター設置場所関連を更新する。
     * 主な仕様:
     * - 一次分岐条件は、主関連フラグが true かつサービス契約開始事由種別が前契約継続以外であること。
     * - 一次分岐・二次分岐ともに SvmstkrDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_SVKY_NO_AND_SKNO_KEY_ONLY を使用し、
     *   SvmstkrSknoUpdateKey で更新する。
     * - 監査情報は AuditContext を使用し、ユーザー ID は SERVICE_USER_ID、
     *   報告元機能 ID は RequestContext.hukkSysShbt をデコードした値を設定する。
     * 既知の制約:
     * - 更新件数が 0 件の場合のエラー化は DA 側の契約に委譲する。
     * - RequestContext.hukkSysShbt が未設定の場合は監査情報生成時に失敗する。
     *
     * @param mtstNo メーター設置場所番号
     * @param svkyNo サービス契約番号
     * @param skno 世代管理番号
     * @param sknrFlg 主関連フラグ
     * @param svkySttJysb サービス契約開始事由種別
     */
    @Override
    public void execute(Long mtstNo, Long svkyNo, Integer skno, Boolean sknrFlg, String svkySttJysb) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        if (isPrimaryBranch(sknrFlg, svkySttJysb)) {
            svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                    buildPrimaryWriteRecord(mtstNo, svkyNo, skno),
                    PRIMARY_TARGET_COLUMNS,
                    new SvmstkrSknoUpdateKey(mtstNo, svkyNo, skno, null),
                    auditContext);
            return;
        }

        svmstkrDataAccess.updateByMtstNoAndSvkyNoAndSkno(
                buildSecondaryWriteRecord(mtstNo, svkyNo, skno),
                SECONDARY_TARGET_COLUMNS,
                new SvmstkrSknoUpdateKey(mtstNo, svkyNo, skno, null),
                auditContext);
    }

    /**
     * 概要: 一次分岐に入る条件を判定する。
     * 主な仕様:
     * - 主関連フラグが true であることを必須とする。
     * - サービス契約開始事由種別が前契約継続以外であることを必須とする。
     *
     * @param sknrFlg 主関連フラグ
     * @param svkySttJysb サービス契約開始事由種別
     * @return 一次分岐に入る場合 true
     */
    private boolean isPrimaryBranch(Boolean sknrFlg, String svkySttJysb) {
        return Boolean.TRUE.equals(sknrFlg)
                && !continuationOfPreviousContract.equals(svkySttJysb);
    }

        /**
         * 概要: 一次分岐用の更新レコードを組み立てる。
         * 主な仕様:
         * - 料金計算関連状態区分を関連有効待ちへ更新する。
         * - 料金計算開始年月日を最大日付へ更新する。
         *
         * @param mtstNo メーター設置場所番号
         * @param svkyNo サービス契約番号
         * @param skno 世代管理番号
         * @return 更新レコード
         */
    private SvmstkrWriteRecord buildPrimaryWriteRecord(Long mtstNo, Long svkyNo, Integer skno) {
        return new SvmstkrWriteRecord(
                mtstNo,
                svkyNo,
                skno,
                null,
                null,
                relatedValidWaiting,
                MAX_DATE,
                null,
                null);
    }

        /**
         * 概要: 二次分岐用の更新レコードを組み立てる。
         * 主な仕様:
         * - 料金計算関連状態区分を無効へ更新する。
         * - 料金計算開始年月日は更新しない。
         *
         * @param mtstNo メーター設置場所番号
         * @param svkyNo サービス契約番号
         * @param skno 世代管理番号
         * @return 更新レコード
         */
    private SvmstkrWriteRecord buildSecondaryWriteRecord(Long mtstNo, Long svkyNo, Integer skno) {
        return new SvmstkrWriteRecord(
                mtstNo,
                svkyNo,
                skno,
                null,
                null,
                invalidChargeCalculationRelatedStatus,
                null,
                null,
                null);
    }

    /**
     * 概要: RequestContext の報告システム種別から監査用の機能 ID を取得する。
     * 主な仕様:
     * - RequestContext.hukkSysShbt を必須値として扱う。
     * - 開閉栓定数サービスで報告元機能 ID 名へ変換する。
     *
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext
                        .getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
}
