package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

/**
 * SV-329（解約共通処理-共通-他小売）のサービス実装。
 */
public final class CancellationCommonProcessCommonOtherRetailImpl
        implements CancellationCommonProcessCommonOtherRetail {

    static final String SERVICE_USER_ID = "KHSS329";

    private static final Set<TaksmtrColumn> TARGET_COLUMNS = EnumSet.of(TaksmtrColumn.TAKS_KY_JT_CD);

    private final TransportServiceContractMeterLocationUpdateForTermination sv043;

    private final TaksmtrDataAccess taksmtrDataAccess;

    private final RequestContext requestContext;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String dedicatedGasMeterStatus;

    private final String canceledConsignmentContractStatus;

    /**
     * コンストラクタ。
     */
    public CancellationCommonProcessCommonOtherRetailImpl(
            TransportServiceContractMeterLocationUpdateForTermination sv043,
            TaksmtrDataAccess taksmtrDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.sv043 = Objects.requireNonNull(sv043, "sv043");
        this.taksmtrDataAccess = Objects.requireNonNull(taksmtrDataAccess, "taksmtrDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.dedicatedGasMeterStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.DEDICATED_GAS_METER);
        this.canceledConsignmentContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.CANCELED);
    }

    @Override
    public void execute(
            Long gasMtstNo,
            java.time.LocalDate tyenYmd,
            String tksuSvkeJysb,
            CancellationCommonProcessCommonOtherRetailGmtsetdInf gmtsetdInf,
            CancellationCommonProcessCommonOtherRetailTskgmstInf tskgmstInf,
            CancellationCommonProcessCommonOtherRetailTkssvkyInf tkssvkyInf,
            List<CancellationCommonProcessCommonOtherRetailSnyInf> snyInfList,
            CancellationCommonProcessCommonOtherRetailTaksmtrInf taksmtrInf) {
        String decodedOriginFunctionId = decodeOriginFunctionId();
        if (shouldCallSv043(gmtsetdInf)) {
            sv043.execute(
                    gasMtstNo,
                    tyenYmd,
                    tksuSvkeJysb,
                    toSv043TskgmstInf(tskgmstInf),
                    toSv043TkssvkyInf(tkssvkyInf),
                    toSv043SnyInfList(snyInfList));
        }

        TaksmtrUpdateKey updateKey = new TaksmtrUpdateKey(
                taksmtrInf.tkykNo(),
                taksmtrInf.hkicNo(),
                gasMtstNo,
                taksmtrInf.recUpdJnjNo());
        taksmtrDataAccess.updateByTakskyNoAndHkkmItiNoAndGmtSetNoAndRecUpdJnjNo(
                buildTaksmtrWriteRecord(),
                TARGET_COLUMNS,
                updateKey,
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }

    private boolean shouldCallSv043(CancellationCommonProcessCommonOtherRetailGmtsetdInf gmtsetdInf) {
        return !dedicatedGasMeterStatus.equals(gmtsetdInf.ckrtGmscJtkb());
    }

    private TaksmtrWriteRecord buildTaksmtrWriteRecord() {
        return new TaksmtrWriteRecord(
                canceledConsignmentContractStatus,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TskgmstInf toSv043TskgmstInf(
            CancellationCommonProcessCommonOtherRetailTskgmstInf input) {
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TskgmstInf(
                input.tksuSvkyNo(),
                input.skno(),
                input.recUpdJnjNo());
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TkssvkyInf toSv043TkssvkyInf(
            CancellationCommonProcessCommonOtherRetailTkssvkyInf input) {
        return new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TkssvkyInf(
                input.tksuSvkyNo(),
                input.skno() == null ? null : Long.valueOf(input.skno()),
                input.recUpdJnjNo());
    }

    private List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf> toSv043SnyInfList(
            List<CancellationCommonProcessCommonOtherRetailSnyInf> input) {
        return input == null ? null : input.stream()
                .map(value -> new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.SnyInf(
                        new jp.co.tokyo_gas.kaiheisen.libs.sv.api.closing.model.TransportServiceContractGmtsetdInf(
                                value.gasMtstNo())))
                .toList();
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requireHukkSysShbt());
    }

    private String requireHukkSysShbt() {
        return requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }
}
