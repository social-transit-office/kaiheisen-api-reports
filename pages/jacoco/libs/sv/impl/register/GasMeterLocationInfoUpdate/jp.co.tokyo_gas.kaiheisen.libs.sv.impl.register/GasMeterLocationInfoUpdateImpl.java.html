/**
 * 概要: SV-119（ガスメーター設置場所情報更新）のサービス実装。
 * 主な仕様:
 * - 排他区分に応じて楽観排他あり/なしの更新経路を切り替える。
 * - GMTSETD、必要時のみ MTSETK、GMTSETK を順に更新し、実行した更新件数が 0 件ならデータ不整合例外を送出する。
 * - 監査情報には固定更新ユーザー ID と RequestContext 由来の報告元機能 ID を設定する。
 * 既知の制約:
 * - RequestContext の値設定と妥当性確認は上位層責務とし、本クラスでは補完や追加検証を行わない。
 * - 排他あり定数はコンストラクタ初期化時に fail-fast で解決する。
 */
public class GasMeterLocationInfoUpdateImpl implements GasMeterLocationInfoUpdate {

    static final String SERVICE_USER_ID = "KHSS119";

    private static final Set<GmtsetdColumn> GMTSETD_TARGET_COLUMNS = Set.of(

            GmtsetdColumn.KAS_YMD,

            GmtsetdColumn.HESN_JYSB,
            GmtsetdColumn.HESN_SDN_SHBT);

    private static final Set<MtsetkColumn> MTSETK_TARGET_COLUMNS = Set.of(

            GmtsetkColumn.KAS_YMD,

    /**
     * 概要: SV-119 実行に必要な依存オブジェクトと定数を初期化する。
     * 主な仕様:
     * - 3 つの DA、RequestContext、開閉栓定数サービスを保持する。
     * - 排他あり判定に使用する開閉栓定数を初期化時に解決する。
     * 既知の制約:
     * - 依存オブジェクトが null の場合は NullPointerException を送出する。
     * - 必須定数が不足している場合はコンストラクタ初期化時点で失敗する。
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param mtsetkDataAccess メーター設置場所＜小売＞データアクセス
     * @param gmtsetkDataAccess ガスメーター設置場所＜小売＞データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    public GasMeterLocationInfoUpdateImpl(
            GmtsetdDataAccess gmtsetdDataAccess,
            MtsetkDataAccess mtsetkDataAccess,
            GmtsetkDataAccess gmtsetkDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService) {
        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.mtsetkDataAccess = Objects.requireNonNull(mtsetkDataAccess, "mtsetkDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.kaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.withExclusionDivision = RequiredConstantResolver.requireKaiheisenConstant(
                this.kaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.Exclusive.TYPE,
                ConstantLookupKeys.Kaiheisen.Exclusive.WITH_EXCLUSION);
    }

    /**
     * 概要: ガスメーター設置場所関連情報を更新する。
     * 主な仕様:
        * - RequestContext の値をそのまま利用して監査情報を組み立てる。
     * - 排他区分に応じて更新キーへ recUpdJnjNo を設定する。
     * - MTSETK の更新キーが不足する場合は MTSETK 更新のみスキップする。
     * - 実行した更新件数が 0 件の場合はデータ不整合例外を送出する。
     * @param haitaKbn 排他区分
     * @param gmtsetdInf ガスメーター設置場所＜導管＞情報
     * @param gmtsetkInf ガスメーター設置場所＜小売＞情報
     * @param mtsetkInf メーター設置場所＜小売＞情報
     * @param gmtsetdUpdInf ガスメーター設置場所＜導管＞更新情報
     * @param mtsetkUpdInf メーター設置場所＜小売＞更新情報
     */
    @Override
    public void execute(
            String haitaKbn,
            GmtsetdInf gmtsetdInf,
            GmtsetkInf gmtsetkInf,
            MtsetkInf mtsetkInf,
            GmtsetdUpdInf gmtsetdUpdInf,
            MtsetkUpdInf mtsetkUpdInf) {
        boolean withExclusion = isWithExclusion(haitaKbn);
        GmtsetdUpdInf normalizedGmtsetdUpdInf = normalizeGmtsetdUpdInf(gmtsetdUpdInf);
        MtsetkUpdInf normalizedMtsetkUpdInf = normalizeMtsetkUpdInf(mtsetkUpdInf);
        String decodedOriginFunctionId = decodeOriginFunctionId();
        AuditContext auditContext = AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId);

        int gmtsetdUpdatedCount = gmtsetdDataAccess.updateByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
                buildGmtsetdWriteRecord(gmtsetdInf, normalizedGmtsetdUpdInf),
                GMTSETD_TARGET_COLUMNS,
                new GmtsetdTkysJnjUpdateKey(
                        gmtsetdInf.gasMtstNo(),
                        gmtsetdInf.tkysYmd(),
                        withExclusion ? gmtsetdInf.recUpdJnjNo() : null),
                auditContext);
        ensureUpdatedCount(
                gmtsetdUpdatedCount,
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                buildGmtsetdParameterJson(gmtsetdInf, normalizedGmtsetdUpdInf, decodedOriginFunctionId, withExclusion));

        if (hasMtsetkUpdateTarget(mtsetkInf)) {
            int mtsetkUpdatedCount = mtsetkDataAccess.updateByMtstNoAndTkysYmdAndRecUpdJnjNo(
                    buildMtsetkWriteRecord(mtsetkInf, normalizedMtsetkUpdInf),
                    MTSETK_TARGET_COLUMNS,
                    new MtsetkUpdateKey(
                            mtsetkInf.mtstNo(),
                            mtsetkInf.tkysYmd(),
                            withExclusion ? mtsetkInf.recUpdJnjNo() : null),
                    auditContext);
            ensureUpdatedCount(
                    mtsetkUpdatedCount,
                    mtsetkDataAccess.tableName(),
                    MtsetkDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                    buildMtsetkParameterJson(mtsetkInf, normalizedMtsetkUpdInf, decodedOriginFunctionId, withExclusion));
        }

        int gmtsetkUpdatedCount = gmtsetkDataAccess.updateByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
                buildGmtsetkWriteRecord(gmtsetkInf, normalizedGmtsetdUpdInf),
                GMTSETK_TARGET_COLUMNS,
                new GmtsetkTkysJnjUpdateKey(
                        gmtsetkInf.gasMtstNo(),
                        gmtsetkInf.tkysYmd(),
                        withExclusion ? gmtsetkInf.recUpdJnjNo() : null),
                auditContext);
        ensureUpdatedCount(
                gmtsetkUpdatedCount,
                gmtsetkDataAccess.tableName(),
                GmtsetkDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                buildGmtsetkParameterJson(gmtsetkInf, normalizedGmtsetdUpdInf, decodedOriginFunctionId, withExclusion));
    }

    /**
     * 概要: 排他区分が排他ありかを判定する。
     * @param haitaKbn 排他区分
     * @return 排他ありの場合 true
     */
    private boolean isWithExclusion(String haitaKbn) {
        return withExclusionDivision.equals(haitaKbn);
    }

    private boolean hasMtsetkUpdateTarget(MtsetkInf mtsetkInf) {
        return mtsetkInf != null
                && mtsetkInf.mtstNo() != null
                && mtsetkInf.tkysYmd() != null;
    }

    private GmtsetdUpdInf normalizeGmtsetdUpdInf(GmtsetdUpdInf gmtsetdUpdInf) {
        if (gmtsetdUpdInf == null) {
            return new GmtsetdUpdInf(null, null, null, null, null, null);
        }
        return gmtsetdUpdInf;
    }

    private MtsetkUpdInf normalizeMtsetkUpdInf(MtsetkUpdInf mtsetkUpdInf) {
        if (mtsetkUpdInf == null) {
            return new MtsetkUpdInf(null, null, null, null, null);
        }
        return mtsetkUpdInf;
    }

    /**
     * 概要: 更新件数を検証し、0 件時はデータ不整合例外を送出する。
     * @param updatedCount 更新件数
     * @param tableName テーブル名
     * @param daIdentifier DA 識別子
     * @param parameterJsonBuilder パラメータ JSON
     */
    private void ensureUpdatedCount(
            int updatedCount,
            String tableName,
            String daIdentifier,
            FlatJsonBuilder parameterJsonBuilder) {
        if (updatedCount == 0) {
            throw DataInconsistencySupport.create(tableName, daIdentifier, parameterJsonBuilder);
        }
    }

    /**
     * 概要: GMTSETD 更新時のデータ不整合例外用 JSON を生成する。
     * @param gmtsetdInf キー情報
     * @param gmtsetdUpdInf 更新情報
     * @param decodedOriginFunctionId 報告元機能 ID
     * @param withExclusion 排他あり判定
     * @return JSON 文字列
     */
    private FlatJsonBuilder buildGmtsetdParameterJson(
            GmtsetdInf gmtsetdInf,
            GmtsetdUpdInf gmtsetdUpdInf,
            String decodedOriginFunctionId,
            boolean withExclusion) {
        return new FlatJsonBuilder()
                .put("gasMtstNo", gmtsetdInf.gasMtstNo())
                .put("tkysYmd", gmtsetdInf.tkysYmd())
                .putIfNonNull("recUpdJnjNo", withExclusion ? gmtsetdInf.recUpdJnjNo() : null)
                .put("gmtKhsJtkb", gmtsetdUpdInf.gmtKhsJtkb())
                .put("kasYmd", gmtsetdUpdInf.kasYmd())
                .put("hesnYmd", gmtsetdUpdInf.hesnYmd())
                .put("kasJysb", gmtsetdUpdInf.kasJysb())
                .put("hesnJysb", gmtsetdUpdInf.hesnJysb())
                .put("hesnSdnShbt", gmtsetdUpdInf.hesnSdnShbt())
                .put("updUsrId", SERVICE_USER_ID)
                .put("updKtnKinoId", decodedOriginFunctionId);
    }

    /**
     * 概要: RequestContext の報告元システム種別から監査用の機能 ID を取得する。
     * @return 監査用の報告元機能 ID
     */
    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }
