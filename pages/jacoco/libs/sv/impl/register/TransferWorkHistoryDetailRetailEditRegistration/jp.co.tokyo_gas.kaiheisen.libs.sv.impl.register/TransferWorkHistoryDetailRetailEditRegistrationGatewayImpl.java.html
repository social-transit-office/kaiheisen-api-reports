package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import jakarta.inject.Inject;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.idousagyourireki_kouri_jouhou.IdousagyourirekiKouriJouhouRequest;

/**
 * SV-159 が編集した異動・作業履歴＜小売＞情報を CIRIUS 連携 request へ変換する gateway 実装。
 */
public final class TransferWorkHistoryDetailRetailEditRegistrationGatewayImpl
        implements TransferWorkHistoryDetailRetailEditRegistrationGateway {

    private static final DateTimeFormatter DATE_FORMATTER = DateTimeFormatter.BASIC_ISO_DATE;

    private final IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter;

    /**
     * コンストラクタ。
     *
     * @param ciriusAdapter 小売向け異動・作業履歴 CIRIUS adapter
     */
    @Inject
    public TransferWorkHistoryDetailRetailEditRegistrationGatewayImpl(
            IdousagyourirekiKouriJouhouCiriusAdapter ciriusAdapter) {
        this.ciriusAdapter = Objects.requireNonNull(ciriusAdapter, "ciriusAdapter");
    }

    @Override
    public void register(TransferWorkHistoryDetailRetailRegistrationRequest request) {
        Objects.requireNonNull(request, "request");
        TransferWorkHistoryRetailCommonInfo commonInfo = Objects.requireNonNull(request.commonInfo(), "commonInfo");
        ciriusAdapter.register(new IdousagyourirekiKouriJouhouRequest.IdousagyourirekiKouriJouhouRenkei(
                new IdousagyourirekiKouriJouhouRequest.IdousagyourirekigyoumuKyoutsuuKouri(
                        toText(commonInfo.artyCstNo()),
                        commonInfo.akrkKbn(),
                        commonInfo.akrkTrskShbt(),
                        commonInfo.isrrGymCd(),
                        commonInfo.isrrGmssCd(),
                        commonInfo.isrrHkkrKbn(),
                        commonInfo.isrrSktCd(),
                        formatDate(commonInfo.isrrTriYmd()),
                        formatTime(commonInfo.isrrTriHms()),
                        formatDate(commonInfo.idsgYmd()),
                        nullToEmpty(commonInfo.isrrHusrId()),
                        nullToEmpty(commonInfo.isrrHkunKj()),
                        commonInfo.isrrHjsgId()),
                toDetailList(request.detailInfoList()),
                commonInfo.cstJskhFlg(),
                nullToEmpty(commonInfo.ckjEvntFleId()),
                formatDate(commonInfo.ctksKjnDd())));
    }

    private List<IdousagyourirekiKouriJouhouRequest.IdousagyourirekiShousaiKouri> toDetailList(
            List<TransferWorkHistoryRetailDetailInfo> detailInfoList) {
        if (detailInfoList == null) {
            return List.of();
        }
        return detailInfoList.stream()
                .map(detailInfo -> new IdousagyourirekiKouriJouhouRequest.IdousagyourirekiShousaiKouri(
                        detailInfo.hjjnNo() == null ? null : BigDecimal.valueOf(detailInfo.hjjnNo()),
                        nullToEmpty(detailInfo.hukkKmnm()),
                        nullToEmpty(detailInfo.hukmJhu()),
                        nullToEmpty(detailInfo.hkkgJhu())))
                .toList();
    }

    private String formatDate(LocalDate value) {
        return value == null ? null : DATE_FORMATTER.format(value);
    }

    private String formatTime(Integer value) {
        return value == null ? null : String.format("%06d", value);
    }

    private String toText(Long value) {
        return value == null ? null : String.valueOf(value);
    }

    private String nullToEmpty(String value) {
        return value == null ? "" : value;
    }
}
