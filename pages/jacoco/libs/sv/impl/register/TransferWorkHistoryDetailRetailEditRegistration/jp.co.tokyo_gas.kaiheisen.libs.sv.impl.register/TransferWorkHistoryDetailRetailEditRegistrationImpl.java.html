    /**
     * 概要: 異動・作業履歴詳細＜小売＞を編集して登録ゲートウェイへ引き渡す。
     * 主な仕様:
     * - RequestContext の必須値を検証したうえで共通情報と 29 行明細を組み立てる。
     * - 東京ガス判定や業務コード判定に応じて下位 SV を呼び分け、変更前後情報を補完する。
     * - 編集結果は戻り値にせず、登録ゲートウェイへそのまま委譲する。
     * 既知の制約:
     * - 引数値の null / blank 妥当性は追加検証しない。
     * - 下位 SV や登録ゲートウェイの例外は変換せずに上位へ伝播する。
     */
    @Override
    public void execute(
            List<IsrrShsiTrkInf> isrrShsiTrkInfList,
            String tksuKijsCd,
            String rshhShjKbn,
            String gymCd,
            String hesnSdnShbt,
            LocalDate hssgYmd,
            String kkysRyuShbt,
            Long mejsGmsbNo,
            String sihHuhuShbt,
            String ctnmKn1,
            String ctnmKj1,
            String ctnm1HujnShbt,
            String ctnm1HshiShbt,
            Boolean ctnmKjhkFlg,
            String kiyShbt,
            String syaKbn,
            Integer suguGmtKnsu,
            String sziKziKbn,
            String sikSvcShbt,
            Long artyCstNo,
            LocalDate isrrTriYmd,
            String isrrGymCd,
            String isrrGmssCd,
            String isrrHkkrKbn,
            String hukkUsrId,
            String hukkUnmKj) {
        RequestContextValues requestContextValues = requireRequestContext();

        boolean tokyoGas = tokyoGasRetailerCode.equals(tksuKijsCd);
        boolean hasClosingMeansType = hesnSdnShbt != null && !hesnSdnShbt.isEmpty();
        boolean supplyOutageBusinessCode = supplyOutageRetailBusinessCode.equals(isrrGymCd);
        boolean supplyOutageBusinessDetailCode =
                supplyOutageReportRetailDetailCode.equals(isrrGmssCd)
                        || closingErrorFollowSupplyOutageDetailCode.equals(isrrGmssCd);
        LocalDate reservedPeriodDate = tokyoGas && supplyOutageBusinessDetailCode && hssgYmd != null
                ? hssgYmd.plusDays(RESERVED_PERIOD_DAYS)
                : null;
        CustomerNameEditOutput editedCustomerName = tokyoGas
                ? customerNameEdit.execute(
                        ctnmKn1,
                        ctnmKj1,
                        ctnm1HujnShbt,
                        ctnm1HshiShbt,
                        ctnmKjhkFlg,
                        nameEditingProcessName1)
                : null;
        String settlementPaymentMethodName = tokyoGas
                        && rshhShjKbn != null
                        && !ConstantLookupKeys.Cirius.FeePaymentMethodInstructionCategory.NO_CONCEPT.equals(
                                rshhShjKbn)
                ? resolveConstantContent(
                        ConstantLookupKeys.Cirius.FeePaymentMethodInstructionCategory.TYPE,
                        rshhShjKbn)
                : null;
        String closingMeansName = hasClosingMeansType
                ? resolveConstantContent(ConstantLookupKeys.Cirius.ClosingMeansType.TYPE, hesnSdnShbt)
                : null;
        String supplyOutageReasonName = tokyoGas && supplyOutageBusinessCode
                ? resolveConstantContent(ConstantLookupKeys.Cirius.SupplySuspensionReasonType.TYPE, kkysRyuShbt)
                : null;
        String paymentMethodName = tokyoGas
                ? resolveConstantContent(ConstantLookupKeys.Cirius.PaymentMethodType.TYPE, sihHuhuShbt)
                : null;
        String contractTypeName = tokyoGas
                ? resolveConstantContent(ConstantLookupKeys.Cirius.ContractType.TYPE, kiyShbt)
                : null;
        String billingServiceName = tokyoGas
                ? resolveConstantContent(ConstantLookupKeys.Cirius.BillingServiceType.TYPE, sikSvcShbt)
                : null;
        String dutyFreeName = tokyoGas && dutyFreeConsumptionTax.equals(sziKziKbn) ? "免税" : null;

        List<TransferWorkHistoryRetailDetailInfo> detailInfoList = buildDetailInfoList(
                isrrShsiTrkInfList,
                tokyoGas,
                hasClosingMeansType,
                supplyOutageBusinessCode,
                settlementPaymentMethodName,
                closingMeansName,
                hssgYmd,
                requestContextValues.systemDatetime().toLocalDate(),
                reservedPeriodDate,
                supplyOutageReasonName,
                mejsGmsbNo,
                paymentMethodName,
                editedCustomerName,
                contractTypeName,
                syaKbn,
                suguGmtKnsu,
                dutyFreeName,
                billingServiceName);

        TransferWorkHistoryRetailCommonInfo commonInfo = new TransferWorkHistoryRetailCommonInfo(
                artyCstNo,
                isrrTriYmd,
                caseHistoryResultTransfer,
                itemHistoryRegistrationDestinationCustomer,
                isrrGymCd,
                isrrGmssCd,
                isrrHkkrKbn,
                transferWorkHistoryInquiryCodeSubject,
                toHms(requestContextValues.systemDatetime()),
                hssgYmd,
                hukkUsrId,
                hukkUnmKj,
                TRANSFER_WORK_HISTORY_DETAIL_SCREEN_ID,
                customerInfoAcquisitionPossible,
                "",
                hssgYmd);

        registrationGateway.register(new TransferWorkHistoryDetailRetailRegistrationRequest(
                commonInfo,
                detailInfoList,
                requestContextValues.correlationId(),
                requestContextValues.hukkSysShbt(),
                requestContextValues.systemDatetime()));
    }

    /**
     * 概要: 明細 29 行分の登録内容を編集する。
     * 主な仕様:
     * - 設計書の行番号に従って固定 29 行の明細レコードを生成する。
     * - 条件を満たす行のみ変更項目名や変更前後情報を設定し、それ以外は空欄のままとする。
     * - 入力一覧の不足分は空行扱いとし、追加の例外は送出しない。
     * 既知の制約:
     * - 29 行目は請求サービス内容を設定し、条件不成立時は空欄のまま出力する。
     * - 条件分岐に必要な下位 SV 実行結果は呼び出し元で事前解決した値を利用する。
     *
     * @return 編集済み明細一覧
     */
    private List<TransferWorkHistoryRetailDetailInfo> buildDetailInfoList(
            List<IsrrShsiTrkInf> isrrShsiTrkInfList,
            boolean tokyoGas,
            boolean hasClosingMeansType,
            boolean supplyOutageBusinessCode,
            String settlementPaymentMethodName,
            String closingMeansName,
            LocalDate hssgYmd,
            LocalDate systemDate,
            LocalDate reservedPeriodDate,
            String supplyOutageReasonName,
            Long mejsGmsbNo,
            String paymentMethodName,
            CustomerNameEditOutput editedCustomerName,
            String contractTypeName,
            String syaKbn,
            Integer suguGmtKnsu,
            String dutyFreeName,
            String billingServiceName) {
        List<TransferWorkHistoryRetailDetailInfo> detailInfoList = new ArrayList<>(TOTAL_DETAIL_ROWS);
        for (int detailNumber = 1; detailNumber <= TOTAL_DETAIL_ROWS; detailNumber++) {
            detailInfoList.add(buildDetailInfo(
                    detailNumber,
                    isrrShsiTrkInfList,
                    tokyoGas,
                    hasClosingMeansType,
                    supplyOutageBusinessCode,
                    settlementPaymentMethodName,
                    closingMeansName,
                    hssgYmd,
                    systemDate,
                    reservedPeriodDate,
                    supplyOutageReasonName,
                    mejsGmsbNo,
                    paymentMethodName,
                    editedCustomerName,
                    contractTypeName,
                    syaKbn,
                    suguGmtKnsu,
                    dutyFreeName,
                    billingServiceName));
        }
        return detailInfoList;
    }

    /**
     * 概要: 明細 1 行分の登録内容を編集する。
     * 主な仕様:
     * - 行番号ごとに設計書の条件分岐を適用する。
     * - 通常移送行は入力一覧の同じ行または指定オフセット行をそのままコピーする。
     * - 特殊編集行は定数名やおなまえ編集結果、予定期間計算結果を反映する。
     * 既知の制約:
     * - 明細番号は 1 始まりで固定とし、入力一覧側は不足分を空行扱いとする。
     */
    private TransferWorkHistoryRetailDetailInfo buildDetailInfo(
            int detailNumber,
            List<IsrrShsiTrkInf> isrrShsiTrkInfList,
            boolean tokyoGas,
            boolean hasClosingMeansType,
            boolean supplyOutageBusinessCode,
            String settlementPaymentMethodName,
            String closingMeansName,
            LocalDate hssgYmd,
            LocalDate systemDate,
            LocalDate reservedPeriodDate,
            String supplyOutageReasonName,
            Long mejsGmsbNo,
            String paymentMethodName,
            CustomerNameEditOutput editedCustomerName,
            String contractTypeName,
            String syaKbn,
            Integer suguGmtKnsu,
            String dutyFreeName,
            String billingServiceName) {
        if (detailNumber <= 10) {
            return copyDetailInfo(detailNumber, detailAt(isrrShsiTrkInfList, detailNumber));
        }
        return switch (detailNumber) {
            case 11 -> buildDetailInfo(detailNumber, "精算料金支払方法", null, settlementPaymentMethodName);
            case 12 -> buildDetailInfo(detailNumber, "供給停止方法", null, hasClosingMeansType ? closingMeansName : null);
            case 13 -> buildDetailInfo(
                    detailNumber,
                    "供給停止計算保留予定期間（開始）",
                    null,
                    shouldSetReservedPeriodDates(systemDate, reservedPeriodDate) ? formatDate(hssgYmd) : null);
            case 14 -> buildDetailInfo(
                    detailNumber,
                    "供給停止計算保留予定期間（終了）",
                    null,
                    shouldSetReservedPeriodDates(systemDate, reservedPeriodDate)
                            ? formatDate(reservedPeriodDate)
                            : null);
            case 15 -> buildDetailInfo(detailNumber, "供給停止理由", null, supplyOutageReasonName);
            case 16 -> buildDetailInfo(
                    detailNumber,
                    "前住所ガスメータ設置場所番号",
                    null,
                    shouldFormatPreviousMeterLocationNumber(tokyoGas, supplyOutageBusinessCode, mejsGmsbNo)
                            ? formatMeterLocationNumber(mejsGmsbNo)
                            : null);
            case 17, 18 -> copyDetailInfo(detailNumber, detailAt(isrrShsiTrkInfList, detailNumber - 6));
            case 19, 20 -> {
                IsrrShsiTrkInf sourceDetail = detailAt(isrrShsiTrkInfList, detailNumber - 6);
                yield buildDetailInfo(
                        detailNumber,
                        sourceDetail.hukkKmnm(),
                        tokyoGas ? sourceDetail.hukmJhu() : null,
                        sourceDetail.hkkgJhu());
            }
            case 21 -> buildDetailInfo(detailNumber, "支払方法", tokyoGas ? paymentMethodName : null, null);
            case 22 -> buildDetailInfo(detailNumber, "支払者名", tokyoGas ? resolveEditedCustomerName(editedCustomerName) : null, null);
            case 23, 24, 25 -> copyDetailInfo(detailNumber, detailAt(isrrShsiTrkInfList, detailNumber - 8));
            case 26 -> buildDetailInfo(detailNumber, "契約種別", tokyoGas ? contractTypeName : null, null);
            case 27 -> buildDetailInfo(
                    detailNumber,
                    "社用／総合",
                    tokyoGas ? resolveCompanyOrComprehensiveName(syaKbn, suguGmtKnsu) : null,
                    null);
            case 28 -> buildDetailInfo(detailNumber, "免税", tokyoGas ? dutyFreeName : null, null);
            case 29 -> buildDetailInfo(detailNumber, "請求サービス内容", tokyoGas ? billingServiceName : null, null);
            default -> buildDetailInfo(detailNumber, null, null, null);
        };
    }

    private boolean shouldSetReservedPeriodDates(LocalDate systemDate, LocalDate reservedPeriodDate) {
        return reservedPeriodDate != null && !reservedPeriodDate.isBefore(systemDate);
    }

    private boolean shouldFormatPreviousMeterLocationNumber(
            boolean tokyoGas,
            boolean supplyOutageBusinessCode,
            Long mejsGmsbNo) {
        return tokyoGas && supplyOutageBusinessCode && mejsGmsbNo != null;
    }

        /**
         * 概要: 会社区分 / 総合区分の変更前情報を決定する。
         * 主な仕様:
         * - 社用区分が社用の場合は固定文言「社用」を返す。
         * - 社用でなく、総合ガスメーター個数が 1 以上の場合は「総合」を返す。
         * - それ以外は null を返す。
         * 既知の制約:
         * - 設計書に下位 SV 呼び出し記載がないため、社用文言は定数マスタから解決しない。
         */
    private String resolveCompanyOrComprehensiveName(
            String syaKbn,
            Integer suguGmtKnsu) {
        if (companyUseDivision.equals(syaKbn)) {
            return COMPANY_USE_LABEL;
        }
        if (suguGmtKnsu != null && suguGmtKnsu >= 1) {
            return COMPREHENSIVE_LABEL;
        }
        return null;
    }

    /**
     * 概要: おなまえ編集結果から変更前情報へ採用する値を決定する。
     * 主な仕様:
     * - 漢字編集結果が存在する場合は漢字を優先する。
     * - 漢字が未設定の場合はカナ編集結果を採用する。
     * - 編集結果自体が未設定の場合は null を返す。
     * 既知の制約:
     * - blank 判定は行わず、CustomerNameEdit の戻り値をそのまま扱う。
     */
    private String resolveEditedCustomerName(CustomerNameEditOutput editedCustomerName) {
        if (editedCustomerName == null) {
            return null;
        }
        if (editedCustomerName.hnsgCtnmKj1() != null) {
            return editedCustomerName.hnsgCtnmKj1();
        }
        return editedCustomerName.hnsgCtnmKn1();
    }

    /**
     * 概要: 定数マスタから内容名を取得する。
     * 主な仕様:
     * - SV-089 を呼び出し、内容名 `cntn` をそのまま返却する。
     * - コードが未設定の場合は SV-089 の仕様に従い null を返す。
     * 既知の制約:
     * - 下位 SV の例外は変換せずに上位へ伝播する。
     */
    private String resolveConstantContent(String constantType, String constantCode) {
        return constantMasterInfoAcquisition.execute(constantType, constantCode).cntn();
    }

    /**
     * 概要: 明細入力一覧の指定行を取得する。
     * 主な仕様:
     * - 入力一覧が null、または指定行が範囲外の場合は空行を返す。
     * - 行番号は 1 始まりとして扱う。
     * - 取得した入力モデルは編集せず、そのまま呼び出し元へ返す。
     * 既知の制約:
     * - 入力一覧内の null 要素までは防御しない。
     */
    private IsrrShsiTrkInf detailAt(List<IsrrShsiTrkInf> isrrShsiTrkInfList, int detailNumber) {
        List<IsrrShsiTrkInf> safeDetailInfoList = isrrShsiTrkInfList == null
                ? Collections.emptyList()
                : isrrShsiTrkInfList;
        if (detailNumber < 1 || detailNumber > safeDetailInfoList.size()) {
            return EMPTY_DETAIL_SOURCE;
        }
        return safeDetailInfoList.get(detailNumber - 1);
    }

    /**
     * 概要: 入力明細をそのまま登録明細へ複写する。
     * 主な仕様:
     * - 変更項目名、変更前情報、変更後情報を入力明細からそのまま転記する。
     * - 明細番号は呼び出し元で指定された値をそのまま設定する。
     * - 入力明細が空行の場合は空欄行として扱う。
     * 既知の制約:
     * - 入力明細自体が null の場合は NullPointerException となる。
     */
    private TransferWorkHistoryRetailDetailInfo copyDetailInfo(int detailNumber, IsrrShsiTrkInf detailInfo) {
        return buildDetailInfo(detailNumber, detailInfo.hukkKmnm(), detailInfo.hukmJhu(), detailInfo.hkkgJhu());
    }

    /**
     * 概要: 登録明細レコードを生成する。
     * 主な仕様:
     * - 明細番号、変更項目名、変更前情報、変更後情報をそのまま保持する。
     * - 呼び出し元で決定済みの値を単純に詰め替えるだけとする。
     * - 29 行固定編集の共通生成口として利用する。
     * 既知の制約:
     * - 値の trim や blank 補正は行わない。
     */
    private TransferWorkHistoryRetailDetailInfo buildDetailInfo(
            int detailNumber,
            String itemName,
            String beforeInfo,
            String afterInfo) {
        return new TransferWorkHistoryRetailDetailInfo(detailNumber, itemName, beforeInfo, afterInfo);
    }

    /**
     * 概要: RequestContext の必須値を検証して取得する。
     * 主な仕様:
     * - correlationId、hukkSysShbt、systemDatetime の順で必須チェックを行う。
     * - いずれかが未設定の場合は IllegalStateException を送出して処理を中断する。
     * - 検証済みの値をレコードへまとめて返却する。
     * 既知の制約:
     * - 値の形式妥当性までは検証しない。
     */
    private RequestContextValues requireRequestContext() {
        String correlationId = requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        return new RequestContextValues(correlationId, hukkSysShbt, systemDatetime);
    }

        /**
         * 概要: 日付を `yyyy/MM/dd` 形式へ整形する。
         * 主な仕様:
         * - 値が存在する場合のみ書式化した文字列を返す。
         * - 値が null の場合は null を返す。
         * - 書式は実装と同じ `yyyy/MM/dd` を使用する。
         * 既知の制約:
         * - タイムゾーン変換は行わない。
         */
    private String formatDate(LocalDate value) {
        return value == null ? null : DATE_FORMATTER.format(value);
    }

    /**
     * 概要: 前住所ガスメーター設置場所番号へハイフンを挿入する。
     * 主な仕様:
     * - 4 桁目と 5 桁目の間、および 7 桁目と 8 桁目の間にハイフンを挿入する。
     * - 7 桁以下の値は挿入可能な位置までのみハイフンを付与する。
     * - 値が null の場合は null を返す。
     * 既知の制約:
     * - 桁不足時の 0 埋めは行わず、入力値の文字列表現をそのまま編集する。
     */
    private String formatMeterLocationNumber(Long mejsGmsbNo) {
        if (mejsGmsbNo == null) {
            return null;
        }
        String rawValue = Long.toString(mejsGmsbNo);
        if (rawValue.length() <= 4) {
            return rawValue;
        }
        if (rawValue.length() <= 7) {
            return rawValue.substring(0, 4) + "-" + rawValue.substring(4);
        }
        return rawValue.substring(0, 4) + "-" + rawValue.substring(4, 7) + "-" + rawValue.substring(7);
    }

    /**
     * 概要: システム日時から時分秒整数を生成する。
     * 主な仕様:
     * - `HHmmss` 形式で整形した文字列を整数化して返却する。
     * - 案件履歴共通情報の登録依頼時分秒に利用する。
     * - タイムゾーン変換は行わない。
     * 既知の制約:
     * - 24 時間制 `HH` での整形結果に依存する。
     */
    private Integer toHms(LocalDateTime systemDatetime) {
        return Integer.valueOf(HMS_FORMATTER.format(systemDatetime));
    }

    }
