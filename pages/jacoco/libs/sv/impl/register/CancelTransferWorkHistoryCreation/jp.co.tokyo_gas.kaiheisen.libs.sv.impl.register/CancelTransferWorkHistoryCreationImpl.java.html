/**
 * 概要: SV-061（取消時異動・作業履歴作成処理）の実装クラス。
 * 主な仕様:
 * - 開栓取消と閉栓取消の 2 ルートを切り替え、導管登録・小売登録・SV-256 を順序どおりに実行する。
 * - CIRIUS request 組立は補助資料を正とし、(2) 付き資料を通常版の代用にしない。
 * - RequestContext の `systemDatetime` と `hukkSysShbt` を必須値として fail-fast する。
 */
public final class CancelTransferWorkHistoryCreationImpl implements CancelTransferWorkHistoryCreation {

    /**
     * SV-061 の実装を構築する。
     *
     * @param sameDayCancel SV-256
     * @param doukanAdapter 導管向け CIRIUS adapter
     * @param kouriAdapter 小売向け CIRIUS adapter
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓基盤定数取得サービス
     * @param ciriusConstantService CIRIUS 定数取得サービス
     */
    public CancelTransferWorkHistoryCreationImpl(
            TransferWorkHistoryIntegrationManagementSameDayCancel sameDayCancel,
            IdousagyourirekiDoukanJouhouCiriusAdapter doukanAdapter,
            IdousagyourirekiKouriJouhouCiriusAdapter kouriAdapter,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.sameDayCancel = Objects.requireNonNull(sameDayCancel, "sameDayCancel");
        this.doukanAdapter = Objects.requireNonNull(doukanAdapter, "doukanAdapter");
        this.kouriAdapter = Objects.requireNonNull(kouriAdapter, "kouriAdapter");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        KaiheisenConstantService nonNullKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        CiriusConstantService nonNullCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.openingProcessingDivision = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.CancelTransferWorkHistoryCreationProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.CancelTransferWorkHistoryCreationProcessingDivision.OPENING_PROCESS);
        this.caseHistoryResultTransfer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.CaseHistory.TYPE,
                ConstantLookupKeys.Cirius.CaseHistory.RESULT_TRANSFER);
        this.itemHistoryRegistrationDestinationPlace = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.TYPE,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.PLACE);
        this.itemHistoryRegistrationDestinationCustomer = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.TYPE,
                ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.CUSTOMER);
        this.transferWorkHistoryInquiryCodeSubject = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.SUBJECT);
        this.transferWorkHistoryInquiryCodeDetailsOnlyNotApplicable = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.NOT_APPLICABLE_TO_DETAILS_ONLY);
        this.customerInfoAcquisitionPossible = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.PossibilityFlag.TYPE,

    @Override
    public void execute(
            String tksjIsrrSksiShrKbn,
            Long gasMtstNo,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrGymCd,
            String isrrGmssCd,
            String isrrHkkrKbn,
            LocalDate trkKnsYmd,
            String isrrHusrId,
            String isrrHkunKj,
            String rirekiMemo,
            String tksuKijsCd,
            Boolean svkyUpdFlg,
            Boolean suguDelFlg,
            Boolean svkyFkktFlg,
            List<SuguGmtInf> suguGmtInfList,
            IsrrkrUpdInf isrrkrUpdInf) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(trkKnsYmd, "trkKnsYmd");
        LocalDateTime systemDatetime = requireSystemDatetime();
        requireHukkSysShbt();
        if (openingProcessingDivision.equals(tksjIsrrSksiShrKbn)) {
            IsrrkrUpdInf requiredIsrrkrUpdInf = Objects.requireNonNull(isrrkrUpdInf, "isrrkrUpdInf");
            executeOpening(
                    gasMtstNo,
                    duknIsrgCd,
                    duknIsrsCd,
                    isrrGymCd,
                    isrrGmssCd,
                    isrrHkkrKbn,
                    trkKnsYmd,
                    isrrHusrId,
                    isrrHkunKj,
                    rirekiMemo,
                    tksuKijsCd,
                    svkyUpdFlg,
                    suguDelFlg,
                    requiredIsrrkrUpdInf,
                    systemDatetime);
            return;
        }
        executeClosing(
                gasMtstNo,
                duknIsrgCd,
                duknIsrsCd,
                isrrGymCd,
                isrrGmssCd,
                isrrHkkrKbn,
                trkKnsYmd,
                isrrHusrId,
                isrrHkunKj,
                rirekiMemo,
                tksuKijsCd,
                svkyFkktFlg,
                suguGmtInfList,
                isrrkrUpdInf,
                systemDatetime);
    }

    private void executeOpening(
            Long gasMtstNo,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrGymCd,
            String isrrGmssCd,
            String isrrHkkrKbn,
            LocalDate trkKnsYmd,
            String isrrHusrId,
            String isrrHkunKj,
            String rirekiMemo,
            String tksuKijsCd,
            Boolean svkyUpdFlg,
            Boolean suguDelFlg,
            IsrrkrUpdInf isrrkrUpdInf,
            LocalDateTime systemDatetime) {
        doukanAdapter.register(buildOpeningDoukanRequest(
                gasMtstNo,
                duknIsrgCd,
                duknIsrsCd,
                isrrHkkrKbn,
                trkKnsYmd,
                isrrHusrId,
                isrrHkunKj,
                rirekiMemo,
                systemDatetime));
        sameDayCancel.execute(
                gasMtstNo,
                systemDatetime.toLocalDate(),
                isrrkrUpdInf.duknIsrgCd(),
                isrrkrUpdInf.duknIsrsCd());
        kouriAdapter.register(buildOpeningRetailRequest(
                gasMtstNo,
                isrrGymCd,
                isrrGmssCd,
                isrrHkkrKbn,
                trkKnsYmd,
                isrrHusrId,
                isrrHkunKj,
                rirekiMemo,
                systemDatetime));
        if (isTokyoGasRetailer(tksuKijsCd)) {
            sameDayCancel.execute(
                    gasMtstNo,
                    systemDatetime.toLocalDate(),
                    isrrkrUpdInf.isrrGymCd(),
                    isrrkrUpdInf.isrrGmssCd());
        }
        if (shouldRegisterOpeningAdditional(svkyUpdFlg, suguDelFlg)) {
            kouriAdapter.register(buildOpeningAdditionalRetailRequest(
                    gasMtstNo,
                    isrrHkkrKbn,
                    trkKnsYmd,
                    isrrHusrId,
                    isrrHkunKj,
                    systemDatetime));
        }
    }

    private void executeClosing(
            Long gasMtstNo,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrGymCd,
            String isrrGmssCd,
            String isrrHkkrKbn,
            LocalDate trkKnsYmd,
            String isrrHusrId,
            String isrrHkunKj,
            String rirekiMemo,
            String tksuKijsCd,
            Boolean svkyFkktFlg,
            List<SuguGmtInf> suguGmtInfList,
            IsrrkrUpdInf isrrkrUpdInf,
            LocalDateTime systemDatetime) {
        doukanAdapter.register(buildClosingDoukanRequest(
                gasMtstNo,
                duknIsrgCd,
                duknIsrsCd,
                isrrHkkrKbn,
                trkKnsYmd,
                isrrHusrId,
                isrrHkunKj,
                rirekiMemo,
                systemDatetime));
        if (isrrkrUpdInf != null) {
            sameDayCancel.execute(
                    gasMtstNo,
                    systemDatetime.toLocalDate(),
                    isrrkrUpdInf.duknIsrgCd(),
                    isrrkrUpdInf.duknIsrsCd());
        }

        String customerInfoFlag = isTokyoGasRetailer(tksuKijsCd)
                ? customerInfoAcquisitionPossible
                : customerInfoAcquisitionNo;
        kouriAdapter.register(buildClosingRetailRequest(
                gasMtstNo,
                isrrGymCd,
                isrrGmssCd,
                isrrHkkrKbn,
                trkKnsYmd,
                isrrHusrId,
                isrrHkunKj,
                rirekiMemo,
                systemDatetime,
                customerInfoFlag));

        if (isTokyoGasRetailer(tksuKijsCd) && isrrkrUpdInf != null) {
            sameDayCancel.execute(
                    gasMtstNo,
                    systemDatetime.toLocalDate(),
                    isrrkrUpdInf.isrrGymCd(),
                    isrrkrUpdInf.isrrGmssCd());
        }
        if (Boolean.TRUE.equals(svkyFkktFlg)) {
            for (SuguGmtInf suguGmtInf : normalizeSuguGmtInfList(suguGmtInfList)) {
                kouriAdapter.register(buildClosingAdditionalRetailRequest(
                        requireSuguGasMtstNo(suguGmtInf),
                        isrrHkkrKbn,
                        trkKnsYmd,
                        isrrHusrId,
                        isrrHkunKj,
                        systemDatetime));
            }
        }
    }

    private boolean shouldRegisterOpeningAdditional(Boolean svkyUpdFlg, Boolean suguDelFlg) {
        return Boolean.TRUE.equals(svkyUpdFlg) || Boolean.TRUE.equals(suguDelFlg);
    }

    private boolean isTokyoGasRetailer(String tksuKijsCd) {
        return tokyoGasRetailerCode.equals(tksuKijsCd);
    }

    private IdousagyourirekiDoukanJouhouRequest buildDoukanRequest(
            Long gasMtstNo,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrHkkrKbn,
            LocalDate trkKnsYmd,

            String isrrHkunKj,

            String customerInfoFlag) {
        return new IdousagyourirekiDoukanJouhouRequest(
                new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekiDoukanJouhouRenkei(
                        new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekigyoumuKyoutsuuDoukan(
                                toCustomerNumber(gasMtstNo),
                                caseHistoryResultTransfer,
                                itemHistoryRegistrationDestinationPlace,
                                duknIsrgCd,
                                duknIsrsCd,
                                isrrHkkrKbn,
                                transferWorkHistoryInquiryCodeSubject,
                                toDateText(systemDatetime.toLocalDate()),
                                toTimeText(systemDatetime),
                                toDateText(trkKnsYmd),
                                normalizeText(isrrHusrId),
                                normalizeText(isrrHkunKj),
                                PIPELINE_SCREEN_ID),
                        List.of(new IdousagyourirekiDoukanJouhouRequest.IdousagyourirekiShousaiDoukan(
                                BigDecimal.ONE,
                                HISTORY_MEMO_LABEL,
                                "",
                                normalizeText(rirekiMemo))),
                        customerInfoFlag,
                        toDateText(systemDatetime.toLocalDate())));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private void requireHukkSysShbt() {
        requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
    }

    private List<SuguGmtInf> normalizeSuguGmtInfList(List<SuguGmtInf> suguGmtInfList) {
        return suguGmtInfList == null ? List.of() : suguGmtInfList;
    }

    private Long requireSuguGasMtstNo(SuguGmtInf suguGmtInf) {
        SuguGmtInf nonNullSuguGmtInf = Objects.requireNonNull(suguGmtInf, "suguGmtInf");
        return Objects.requireNonNull(nonNullSuguGmtInf.gasMtstNo(), "suguGmtInf.gasMtstNo");
    }

    private String normalizeText(String value) {
        return value == null ? "" : value;
    }

    private String toCustomerNumber(Long gasMtstNo) {
        return String.format("%011d", gasMtstNo);
    }

    private String toDateText(LocalDate date) {
        return DATE_FORMATTER.format(date);
    }

    private String toTimeText(LocalDateTime datetime) {
        return TIME_FORMATTER.format(datetime);
    }
