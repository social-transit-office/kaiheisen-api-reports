package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.KyotelkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.PhoneNumberAdditionalUpdateRetail;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.PhoneNumberUpdateAddDeleteFunctionRetail;

/** TG小売用電話番号追加更新サービス実装。 */
public class PhoneNumberAdditionalUpdateRetailImpl implements PhoneNumberAdditionalUpdateRetail {

    private static final long MAX_PHONE_NUMBER_COUNT = 2L;

    private final KyotelkDataAccess kyotelkDataAccess;

    private final PhoneNumberUpdateAddDeleteFunctionRetail phoneNumberUpdateAddDeleteFunctionRetail;

    private final String processingDivisionAdd;

    private final String processingDivisionUpdate;

    private final String processingDivisionDelete;

    private final String rejectedConfirmedType;

    /**
     * コンストラクタ。
     *
     * @param kyotelkDataAccess 契約お客さま電話番号＜小売＞データアクセス
     * @param phoneNumberUpdateAddDeleteFunctionRetail 電話番号更新・追加・削除機能-小売
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public PhoneNumberAdditionalUpdateRetailImpl(
            KyotelkDataAccess kyotelkDataAccess,
            PhoneNumberUpdateAddDeleteFunctionRetail phoneNumberUpdateAddDeleteFunctionRetail,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.kyotelkDataAccess = Objects.requireNonNull(kyotelkDataAccess, "kyotelkDataAccess");
        this.phoneNumberUpdateAddDeleteFunctionRetail = Objects.requireNonNull(
                phoneNumberUpdateAddDeleteFunctionRetail,
                "phoneNumberUpdateAddDeleteFunctionRetail");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.processingDivisionAdd = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.ADDITION);
        this.processingDivisionUpdate = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.UPDATE);
        this.processingDivisionDelete = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.DELETE);
        this.rejectedConfirmedType = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.PhoneConfirmedType.TYPE,
                ConstantLookupKeys.Cirius.PhoneConfirmedType.REJECTED_CONFIRMED);
    }

    @Override
        public void execute(
            Boolean telNoUmFlg,
            Long rakKyoTrkNo,
            String kyoTelNo,
            String telShbt,
            String telRrskShbt,
            String telKnzShbt,
            String telStsdShbt,
            LocalDate telShtkYmd,
            String tetsCtnmKj,
            String telZkgrShbt,
            String telMusdKbn) {
        if (Boolean.FALSE.equals(telNoUmFlg)) {
            long count = kyotelkDataAccess.countByKyoTrkNo(rakKyoTrkNo);
            if (count < MAX_PHONE_NUMBER_COUNT) {
                callRetailService(
                        processingDivisionAdd,
                        rakKyoTrkNo,
                        kyoTelNo,
                        telShbt,
                        telRrskShbt,
                        telKnzShbt,
                        telStsdShbt,
                        telShtkYmd,
                        tetsCtnmKj,
                        telZkgrShbt,
                        telMusdKbn);
            }
            if (rejectedConfirmedType.equals(telKnzShbt)) {
                callDeleteRetailService(rakKyoTrkNo, kyoTelNo);
            }
            return;
        }

        callRetailService(
                processingDivisionUpdate,
                rakKyoTrkNo,
                kyoTelNo,
                telShbt,
                telRrskShbt,
                telKnzShbt,
                telStsdShbt,
                telShtkYmd,
                tetsCtnmKj,
                telZkgrShbt,
                telMusdKbn);

        if (rejectedConfirmedType.equals(telKnzShbt)) {
            callDeleteRetailService(rakKyoTrkNo, kyoTelNo);
        }
    }

    private void callRetailService(
            String telNoSyrKbn,
            Long rakKyoTrkNo,
            String kyoTelNo,
            String telShbt,
            String telRrskShbt,
            String telKnzShbt,
            String telStsdShbt,
            LocalDate telShtkYmd,
            String tetsCtnmKj,
            String telZkgrShbt,
            String telMusdKbn) {
        phoneNumberUpdateAddDeleteFunctionRetail.execute(
                telNoSyrKbn,
                rakKyoTrkNo,
                kyoTelNo,
                telShbt,
                telRrskShbt,
                telKnzShbt,
                telStsdShbt,
                telShtkYmd,
                tetsCtnmKj,
                telZkgrShbt,
                telMusdKbn);
    }

    private void callDeleteRetailService(Long rakKyoTrkNo, String kyoTelNo) {
        phoneNumberUpdateAddDeleteFunctionRetail.execute(
                processingDivisionDelete,
                rakKyoTrkNo,
                kyoTelNo,
                null,
                null,
                null,
                null,
                null,
                null,
                null,
                null);
    }
}
