package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.EnumSet;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencySupport;

/** 電話番号更新・追加・削除機能-導管サービス実装。 */
public class PhoneNumberUpdateAddDeleteFunctionPipelineImpl
        implements PhoneNumberUpdateAddDeleteFunctionPipeline {

    static final String SERVICE_USER_ID = "KHSS336";

    private static final int HISTORY_SEQUENCE_MAX = 99999;

    private final KyoteldDataAccess kyoteldDataAccess;

    private final KyotlrdDataAccess kyotlrdDataAccess;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final CiriusConstantService ciriusConstantService;

    private final RequestContext requestContext;

    private final String processingDivisionAdd;

    private final String processingDivisionUpdate;

    private final String processingDivisionDelete;

    private final String changeCancelDivisionChange;

    private final String changeCancelDivisionDelete;

    /**
     * コンストラクタ。
     *
     * @param kyoteldDataAccess 契約お客さま電話番号＜導管＞データアクセス
     * @param kyotlrdDataAccess 契約お客さま電話番号履歴＜導管＞データアクセス
     * @param requestContext リクエストコンテキスト
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public PhoneNumberUpdateAddDeleteFunctionPipelineImpl(
            KyoteldDataAccess kyoteldDataAccess,
            KyotlrdDataAccess kyotlrdDataAccess,
            RequestContext requestContext,
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.kyoteldDataAccess = Objects.requireNonNull(kyoteldDataAccess, "kyoteldDataAccess");
        this.kyotlrdDataAccess = Objects.requireNonNull(kyotlrdDataAccess, "kyotlrdDataAccess");
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        KaiheisenConstantService resolvedKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        this.ciriusConstantService = resolvedCiriusConstantService;
        this.processingDivisionAdd = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.ADDITION);
        this.processingDivisionUpdate = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.UPDATE);
        this.processingDivisionDelete = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.PhoneNumberProcessingDivision.DELETE);
        this.changeCancelDivisionChange = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TelephoneChangeCancellation.TYPE,
                ConstantLookupKeys.Cirius.TelephoneChangeCancellation.CHANGE);
        this.changeCancelDivisionDelete = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.TelephoneChangeCancellation.TYPE,
                ConstantLookupKeys.Cirius.TelephoneChangeCancellation.DELETE);
    }

    @Override
        public void execute(
            String telNoSyrKbn,
            Long cstRegNo,
            String kyoTelNo,
            String telShbt,
            String telRrskShbt,
            String telKnzShbt,
            String telStsdShbt,
            LocalDate telShtkYmd,
            String tetsCtnmKj,
            String telZkgrShbt,
            String telMusdKbn) {
        Optional<KyoteldRecord> currentRecord = kyoteldDataAccess.findByKyoTrkNoAndKyoTelNo(cstRegNo, kyoTelNo);
        boolean exists = currentRecord.isPresent();

        if (!exists && isUpdateOrDelete(telNoSyrKbn)) {
            throw DataInconsistencySupport.create(
                    kyoteldDataAccess.tableName(),
                    KyoteldDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO_AND_KYO_TEL_NO.identifier(),
                    new FlatJsonBuilder()
                            .put("kyoTrkNo", cstRegNo)
                            .put("kyoTelNo", kyoTelNo));
        }

        if (processingDivisionAdd.equals(telNoSyrKbn)) {
            if (!exists) {
                String decodedOriginFunctionId = decodeOriginFunctionId();
                kyoteldDataAccess.insert(
                        new KyoteldWriteRecord(
                                kyoTelNo,
                                cstRegNo,
                                telShbt,
                                telRrskShbt,
                                telKnzShbt,
                                telStsdShbt,
                                telShtkYmd,
                                tetsCtnmKj,
                                telZkgrShbt,
                                telMusdKbn,
                                SERVICE_USER_ID),
                        AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
            }
            return;
        }

        if (processingDivisionUpdate.equals(telNoSyrKbn)) {
            KyoteldRecord existingRecord = currentRecord.orElseThrow();
            String decodedOriginFunctionId = decodeOriginFunctionId();
            LocalDate historyDate = requireSystemDatetime().toLocalDate();
            KyoteldWriteRecord updateRecord = new KyoteldWriteRecord(
                    existingRecord.kyoTelNo(),
                    existingRecord.kyoTrkNo(),
                    telShbt,
                    telRrskShbt,
                    telKnzShbt,
                    telStsdShbt,
                    telShtkYmd,
                    tetsCtnmKj,
                    telZkgrShbt,
                    telMusdKbn,
                    existingRecord.ttrsUsrId());
            Set<KyoteldColumn> targetColumns = buildUpdateTargetColumns(
                    telShbt,
                    telRrskShbt,
                    telKnzShbt,
                    telStsdShbt,
                    telShtkYmd,
                    tetsCtnmKj,
                    telZkgrShbt,
                    telMusdKbn);
            if (targetColumns.isEmpty()) {
                return;
            }
            KyoteldUpdateKey updateKey = new KyoteldUpdateKey(
                    existingRecord.kyoTrkNo(),
                    existingRecord.kyoTelNo(),
                    existingRecord.recUpdJnjNo());
            int updatedCount = kyoteldDataAccess.updateByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
                    updateRecord,
                    targetColumns,
                    updateKey,
                    AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
            if (updatedCount == 0) {
                throw DataInconsistencySupport.create(
                        kyoteldDataAccess.tableName(),
                        KyoteldDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("kyoTelNo", existingRecord.kyoTelNo())
                                .put("kyoTrkNo", existingRecord.kyoTrkNo())
                                .put("recUpdJnjNo", existingRecord.recUpdJnjNo())
                                .putIfNonNull("telShbt", telShbt)
                                .putIfNonNull("telRrskShbt", telRrskShbt)
                                .putIfNonNull("telKnzShbt", telKnzShbt)
                                .putIfNonNull("telStsdShbt", telStsdShbt)
                                .putIfNonNull("telShtkYmd", telShtkYmd)
                                .putIfNonNull("tetsCtnmKj", tetsCtnmKj)
                                .putIfNonNull("telZkgrShbt", telZkgrShbt)
                                .putIfNonNull("telMusdKbn", telMusdKbn)
                                .put("updUsrId", SERVICE_USER_ID)
                                .put("updKtnKinoId", decodedOriginFunctionId));
            }
            insertHistory(
                    existingRecord,
                    changeCancelDivisionChange,
                    historyDate,
                    decodedOriginFunctionId);
            return;
        }

        if (processingDivisionDelete.equals(telNoSyrKbn)) {
            KyoteldRecord existingRecord = currentRecord.orElseThrow();
            String decodedOriginFunctionId = decodeOriginFunctionId();
            LocalDate historyDate = requireSystemDatetime().toLocalDate();
            int deletedCount = kyoteldDataAccess.deleteByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
                    existingRecord.kyoTrkNo(),
                    existingRecord.kyoTelNo(),
                    existingRecord.recUpdJnjNo());
            if (deletedCount == 0) {
                throw DataInconsistencySupport.create(
                        kyoteldDataAccess.tableName(),
                        KyoteldDataAccess.CallSpecs.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("kyoTelNo", existingRecord.kyoTelNo())
                                .put("kyoTrkNo", existingRecord.kyoTrkNo())
                                .put("recUpdJnjNo", existingRecord.recUpdJnjNo()));
            }
            insertHistory(
                    existingRecord,
                    changeCancelDivisionDelete,
                    historyDate,
                    decodedOriginFunctionId);
            return;
        }
    }

    private boolean isUpdateOrDelete(String telNoSyrKbn) {
        return processingDivisionUpdate.equals(telNoSyrKbn)
                || processingDivisionDelete.equals(telNoSyrKbn);
    }

    private String decodeOriginFunctionId() {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                requestContext.getHukkSysShbt()
                        .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
    }

    private LocalDateTime requireSystemDatetime() {
        return requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
    }

    private Set<KyoteldColumn> buildUpdateTargetColumns(
            String telShbt,
            String telRrskShbt,
            String telKnzShbt,
            String telStsdShbt,
            LocalDate telShtkYmd,
            String tetsCtnmKj,
            String telZkgrShbt,
            String telMusdKbn) {
        EnumSet<KyoteldColumn> targetColumns = EnumSet.noneOf(KyoteldColumn.class);
        if (telShbt != null) {
            targetColumns.add(KyoteldColumn.TEL_SHBT);
        }
        if (telRrskShbt != null) {
            targetColumns.add(KyoteldColumn.TEL_RRSK_SHBT);
        }
        if (telKnzShbt != null) {
            targetColumns.add(KyoteldColumn.TEL_KNZ_SHBT);
        }
        if (telStsdShbt != null) {
            targetColumns.add(KyoteldColumn.TEL_STSD_SHBT);
        }
        if (telShtkYmd != null) {
            targetColumns.add(KyoteldColumn.TEL_SHTK_YMD);
        }
        if (tetsCtnmKj != null) {
            targetColumns.add(KyoteldColumn.TETS_CTNM_KJ);
        }
        if (telZkgrShbt != null) {
            targetColumns.add(KyoteldColumn.TEL_ZKGR_SHBT);
        }
        if (telMusdKbn != null) {
            targetColumns.add(KyoteldColumn.TEL_MUSD_KBN);
        }
        return targetColumns;
    }

    private void insertHistory(
            KyoteldRecord existingRecord,
            String changeCancelDivision,
            LocalDate historyDate,
            String decodedOriginFunctionId) {
        int historySequenceNo = kyotlrdDataAccess
                .findLatestByKyoTelNoAndKyoTrkNo(existingRecord.kyoTelNo(), existingRecord.kyoTrkNo())
                .map(KyotlrdRecord::telNoTsno)
                .map(sequenceNo -> sequenceNo >= HISTORY_SEQUENCE_MAX ? 1 : sequenceNo + 1)
                .orElse(1);
        kyotlrdDataAccess.insert(
                new KyotlrdWriteRecord(
                        existingRecord.kyoTelNo(),
                        existingRecord.kyoTrkNo(),
                        historySequenceNo,
                        existingRecord.telShbt(),
                        existingRecord.telRrskShbt(),
                        existingRecord.telKnzShbt(),
                        existingRecord.telStsdShbt(),
                        existingRecord.telShtkYmd(),
                        existingRecord.tetsCtnmKj(),
                        existingRecord.telZkgrShbt(),
                        existingRecord.telMusdKbn(),
                        changeCancelDivision,
                        historyDate,
                        existingRecord.ttrsUsrId(),
                        SERVICE_USER_ID),
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));
    }
}
