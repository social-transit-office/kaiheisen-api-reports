package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.register;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.anzntnk.AnzntnkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.homsyk.HomsykDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.SupportInfoDeletion;

/**
 * SV-287 支援情報削除サービス実装。
 */
public class SupportInfoDeletionImpl implements SupportInfoDeletion {

    private final KenssieDataAccess kenssieDataAccess;

    private final HomsykDataAccess homsykDataAccess;

    private final AnzntnkDataAccess anzntnkDataAccess;

    private final String idoMasyKhiCd;

    /**
     * コンストラクタ。
     *
     * @param kenssieDataAccess 検針支援データアクセス
     * @param homsykDataAccess 訪問集金支援情報データアクセス
     * @param anzntnkDataAccess 安全点検支援情報データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public SupportInfoDeletionImpl(
            KenssieDataAccess kenssieDataAccess,
            HomsykDataAccess homsykDataAccess,
            AnzntnkDataAccess anzntnkDataAccess,
            CiriusConstantService ciriusConstantService) {
        this.kenssieDataAccess = Objects.requireNonNull(kenssieDataAccess, "kenssieDataAccess");
        this.homsykDataAccess = Objects.requireNonNull(homsykDataAccess, "homsykDataAccess");
        this.anzntnkDataAccess = Objects.requireNonNull(anzntnkDataAccess, "anzntnkDataAccess");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");
        this.idoMasyKhiCd = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
            ConstantLookupKeys.Cirius.TransferDeletionAvailabilityCode.TYPE,
            ConstantLookupKeys.Cirius.TransferDeletionAvailabilityCode.DELETE);
    }

    /**
     * 支援情報削除を実行する。
     *
     * @param digwrHntFlg 代替判定結果フラグ
     * @param gasMtstNo ガスメーター設置場所番号
     */
    @Override
    public void execute(Boolean digwrHntFlg, Long gasMtstNo) {
        if (!Boolean.TRUE.equals(digwrHntFlg)) {
            return;
        }

        kenssieDataAccess.deleteByGmtSetNoAndIdoMasyKhiCd(gasMtstNo, idoMasyKhiCd, null);
        homsykDataAccess.deleteByGmtSetNoAndIdoMasyKhiCd(gasMtstNo, idoMasyKhiCd, null);
        anzntnkDataAccess.deleteByGmtSetNoAndIdoMasyKhiCd(gasMtstNo, idoMasyKhiCd, null);
    }
}
