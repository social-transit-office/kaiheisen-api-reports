package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.ServiceSuspensionCancellationReasonTypeAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.ServiceSuspensionCancellationReasonTypeAcquisitionOutput;

/**
 * サービス停止・解約事由種別取得サービス実装クラスです.
 */

public final class ServiceSuspensionCancellationReasonTypeAcquisitionImpl
    implements ServiceSuspensionCancellationReasonTypeAcquisition {

    /** 移転解約理由コードです. */
    private final String transferCancellationTerminationReasonCode;

    /** 不使用解約理由コードです. */
    private final String nonUseCancellationTerminationReasonCode;

    /** 当社都合解約理由コードです. */
    private final String cancellationAtOurDiscretionTerminationReasonCode;

    /** 供給停止閉栓理由コードです. */
    private final String supplyOutageClosureReasonCode;

    /** 停止・解約該当なし理由コードです. */
    private final String noServiceSuspensionOrCancellationReasonCode;

    /** 一時未使用停止・解約理由コードです. */
    private final String temporarilyUnusedSuspensionCancellationReasonCode;

    /** 供給停止・停止/解約理由コードです. */
    private final String supplyOutageSuspensionCancellationReasonCode;

    /** 移転閉栓・停止/解約理由コードです. */
    private final String relocationClosureSuspensionCancellationReasonCode;

    /** 当社都合・停止/解約理由コードです. */
    private final String cancellationAtOurDiscretionSuspensionCancellationReasonCode;

    /**
     * 判定と返却に必要な CIRIUS 定数を初期化します.
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービスです.
     */
    public ServiceSuspensionCancellationReasonTypeAcquisitionImpl(
        final CiriusConstantService ciriusConstantService
    ) {
        final CiriusConstantService nonNullCiriusConstantService =
            Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService"
            );
        // 定数をローカル変数に分割
        final String contractType =
            ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                .TYPE;

        final String transferCancel =
            ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                .TRANSFER_CANCELLATION;

        final String nonUseCancel =
            ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                .NON_USE_CANCELLATION;

        final String atOurDiscretionCancel =
            ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                .CANCELLATION_AT_OUR_DISCRETION;

        final String closureType =
            ConstantLookupKeys.Cirius.ClosureReasonType
                .TYPE;

        final String supplyOutage =
            ConstantLookupKeys.Cirius.ClosureReasonType
                .SUPPLY_OUTAGE;

        final String susCanType =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .TYPE;

        final String noSusCan =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .NO_SERVICE_SUSPENSION_OR_CANCELLATION;

        final String tempUnused =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .TEMPORARILY_UNUSED_CLOSURE;

        final String susCanSupplyOutage =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .SUPPLY_OUTAGE;

        final String relocationClosure =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .RELOCATION_CLOSURE;

        final String susCanAtOurDiscretion =
            ConstantLookupKeys.Cirius.ServiceSuspensionCancellationReasonType
                .CANCELLATION_AT_OUR_DISCRETION;

        this.transferCancellationTerminationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                contractType,
                transferCancel
            );

        this.nonUseCancellationTerminationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                contractType,
                nonUseCancel
            );

        this.cancellationAtOurDiscretionTerminationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                contractType,
                atOurDiscretionCancel
            );

        this.supplyOutageClosureReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                closureType,
                supplyOutage
            );

        this.noServiceSuspensionOrCancellationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                susCanType,
                noSusCan
            );

        this.temporarilyUnusedSuspensionCancellationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                susCanType,
                tempUnused
            );

        this.supplyOutageSuspensionCancellationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                susCanType,
                susCanSupplyOutage
            );

        this.relocationClosureSuspensionCancellationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                susCanType,
                relocationClosure
            );

        this.cancellationAtOurDiscretionSuspensionCancellationReasonCode =
            RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                susCanType,
                susCanAtOurDiscretion
            );
    }

    /**
     * サービス停止・解約事由種別を決定します.
     *
     * @param svkyEndJysb サービス契約終了事由種別です.
     * @param hesnJysb 閉栓事由種別です.
     * @return サービス停止・解約事由種別です.
     */
    @Override
    public ServiceSuspensionCancellationReasonTypeAcquisitionOutput execute(
        final String svkyEndJysb,
        final String hesnJysb
    ) {
        if (
            transferCancellationTerminationReasonCode
                .equals(
                    svkyEndJysb
                )
        ) {
            return output(
                relocationClosureSuspensionCancellationReasonCode
            );
        }

        if (
            nonUseCancellationTerminationReasonCode
                .equals(
                    svkyEndJysb
                )
        ) {
            return output(
                temporarilyUnusedSuspensionCancellationReasonCode
            );
        }

        if (
            cancellationAtOurDiscretionTerminationReasonCode
                .equals(
                    svkyEndJysb
                )
        ) {
            return output(
                cancellationAtOurDiscretionSuspensionCancellationReasonCode
            );
        }

        if (
            supplyOutageClosureReasonCode
                .equals(
                    hesnJysb
                )
        ) {
            return output(
                supplyOutageSuspensionCancellationReasonCode
            );
        }

        return output(
            noServiceSuspensionOrCancellationReasonCode
        );
    }

    /**
     * 戻り値 DTO を生成します.
     *
     * @param svkjyShbt サービス停止・解約事由種別です.
     * @return 戻り値 DTOです。
     */
    private ServiceSuspensionCancellationReasonTypeAcquisitionOutput output(
        final String svkjyShbt
    ) {
        return new ServiceSuspensionCancellationReasonTypeAcquisitionOutput(
            svkjyShbt
        );
    }
}
