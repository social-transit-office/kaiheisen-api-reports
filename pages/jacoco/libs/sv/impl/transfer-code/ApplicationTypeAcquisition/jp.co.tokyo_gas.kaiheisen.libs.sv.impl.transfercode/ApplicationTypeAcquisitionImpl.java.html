package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.ApplicationTypeAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.ApplicationTypeAcquisitionOutput;

/**
 * 申込種別取得サービス実装クラス。
 *
 * <p>開栓事由種別、閉栓事由種別、判定用報告システム種別から
 * 申込種別を決定する。
 */
public final class ApplicationTypeAcquisitionImpl
        implements ApplicationTypeAcquisition {

    private final String alternativeOpeningReasonCode;

    private final String newlyOpenedTapReasonCode;

    private final String supplyResumedReasonCode;

    private final String continuedUseReasonCode;

    private final String relocationClosureReasonCode;

    private final String unusedClosureReasonCode;

    private final String startApplicationTypeCode;

    private final String cancellationApplicationTypeCode;

    private final String supplyResumedApplicationTypeCode;

    private final String continuedUseAndOpeningApplicationTypeCode;

    private final String unusedClosureSeasonalClosureApplicationTypeCode;

    private final String remoteSupplyResumedApplicationTypeCode;

    private final String smartMeterReportingSystemType;

    /**
     * 判定に必要な CIRIUS 定数と開閉栓定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS 定数取得サービス
     * @param kaiheisenConstantService 開閉栓定数取得サービス
     */
    public ApplicationTypeAcquisitionImpl(
            final CiriusConstantService ciriusConstantService,
            final KaiheisenConstantService kaiheisenConstantService) {
        final CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(
                        ciriusConstantService,
                        "ciriusConstantService");
        final KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(
                        kaiheisenConstantService,
                        "kaiheisenConstantService");

        this.alternativeOpeningReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .ALTERNATIVE_OPENING);
        this.newlyOpenedTapReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .NEWLY_OPENED_TAP);
        this.supplyResumedReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .SUPPLY_RESUMED);
        this.continuedUseReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.OpeningReasonType.TYPE,
                        ConstantLookupKeys.Cirius.OpeningReasonType
                                .CONTINUED_USE);
        this.relocationClosureReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType
                                .RELOCATION_CLOSURE);
        this.unusedClosureReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType
                                .UNUSED_CLOSURE);
        this.startApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType.START);
        this.cancellationApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType.CANCELLATION);
        this.supplyResumedApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType
                                .SUPPLY_RESUMED);
        this.continuedUseAndOpeningApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType
                                .CONTINUED_USE_AND_OPENING);
        this.unusedClosureSeasonalClosureApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType
                                .UNUSED_CLOSURE_SEASONAL_CLOSURE);
        this.remoteSupplyResumedApplicationTypeCode =
                RequiredConstantResolver.requireCiriusConstant(
                        nonNullCiriusConstantService,
                        ConstantLookupKeys.Cirius.ApplicationType.TYPE,
                        ConstantLookupKeys.Cirius.ApplicationType
                                .SUPPLY_RESUMED_REMOTE);
        this.smartMeterReportingSystemType =
                RequiredConstantResolver.requireKaiheisenConstant(
                        nonNullKaiheisenConstantService,
                        ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                        ConstantLookupKeys.Kaiheisen.ReportingSystemType.GASTA);
    }

    /**
     * 申込種別を決定する。
     *
     * @param kasJysb 開栓事由種別
     * @param hesnJysb 閉栓事由種別
     * @param htyoSysShbtCd 判定用報告システム種別
     * @return 申込種別
     */
    @Override
    public ApplicationTypeAcquisitionOutput execute(
            final String kasJysb,
            final String hesnJysb,
            final String htyoSysShbtCd) {
        if (kasJysb == null || kasJysb.isEmpty()) {
            return output(
                    resolveByClosureReason(hesnJysb));
        }

        return output(
                resolveByOpeningReason(kasJysb, htyoSysShbtCd));
    }

    private String resolveByClosureReason(final String hesnJysb) {
        if (relocationClosureReasonCode.equals(hesnJysb)) {
            return cancellationApplicationTypeCode;
        }

        if (unusedClosureReasonCode.equals(hesnJysb)) {
            return unusedClosureSeasonalClosureApplicationTypeCode;
        }

        return null;
    }

    private String resolveByOpeningReason(
            final String kasJysb,
            final String htyoSysShbtCd) {
        if (isStartOpeningReason(kasJysb)) {
            return startApplicationTypeCode;
        }

        if (supplyResumedReasonCode.equals(kasJysb)) {
            return resolveSupplyResumedApplicationType(htyoSysShbtCd);
        }

        if (continuedUseReasonCode.equals(kasJysb)) {
            return continuedUseAndOpeningApplicationTypeCode;
        }

        return null;
    }

    private boolean isStartOpeningReason(final String kasJysb) {
        return alternativeOpeningReasonCode.equals(kasJysb)
                || newlyOpenedTapReasonCode.equals(kasJysb);
    }

    private String resolveSupplyResumedApplicationType(
            final String htyoSysShbtCd) {
        if (smartMeterReportingSystemType.equals(htyoSysShbtCd)) {
            return remoteSupplyResumedApplicationTypeCode;
        }

        return supplyResumedApplicationTypeCode;
    }

    private ApplicationTypeAcquisitionOutput output(final String mkSbt) {
        return new ApplicationTypeAcquisitionOutput(mkSbt);
    }
}
