package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;


import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.InquiryLevelCodeEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.InquiryLevelCodeEditOutput;

/**
 * 概要: SV-079（照会レベルコード編集編集）サービス実装。
 * 主な仕様:
 * - 照会レベルコードのブランクを判定し、11300コード"0"を返却する。
 * - 上記に該当しない場合は引数.照会レベルコードをそのまま返却する。
 * - 判定値/返却値は CiriusConstantService で必須定数を解決した値を使用する。
 * 既知の制約:
 * - 相関ID/報告システム種別/システム日時は現行仕様では判定に使用しない。
 * - nullや想定外入力はエラー化せず、条件外として未設置コードを返却する。
 */
public class InquiryLevelCodeEditImpl implements InquiryLevelCodeEdit {

    private static final String BLANK = "";

    private final String displayedSyokaiLevelStatus;

     /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public InquiryLevelCodeEditImpl(CiriusConstantService ciriusConstantService) {
        // コンストラクタの初期化
        this.displayedSyokaiLevelStatus = RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.InquiryLevelType.TYPE,
                ConstantLookupKeys.Cirius.InquiryLevelType.ALL_CAN_BE_QUERIED_AND_DISPLAYED);
    }

    @Override
    public InquiryLevelCodeEditOutput execute(
            String syokaiLevelCd) {

        // ブランクの場合、照会レベルコード（＝0）を返す
        if (BLANK.equals(syokaiLevelCd) || syokaiLevelCd == null) {
            return new InquiryLevelCodeEditOutput(
                    displayedSyokaiLevelStatus);
        }

        return new InquiryLevelCodeEditOutput(
               syokaiLevelCd);
    }

}
