package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.GhpMeterInstallationCodeEdit;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.GhpMeterInstallationCodeEditOutput;

/**
 * 概要: SV-010（ＧＨＰメータ設置コード編集）サービス実装。 主な仕様: - 直列ガスメーター設置状態区分が元/専用の順で優先判定し、対応する00800コードを返却する。 -
 * 上記に該当しない場合は機器用ガスメーター区分を判定し、最終的に未設置コードを返却する。 - 判定値/返却値は CiriusConstantService で必須定数を解決した値を使用する。
 * 既知の制約: - 相関ID/報告システム種別/システム日時は現行仕様では判定に使用しない。 - nullや想定外入力はエラー化せず、条件外として未設置コードを返却する。
 */
public class GhpMeterInstallationCodeEditImpl implements GhpMeterInstallationCodeEdit {

    private final String seriesOriginalGasMeterStatus;

    private final String seriesDedicatedGasMeterStatus;

    private final String equipmentGasMeterForGhp;

    private final String ghpMeterInstallationCodeSeriesOriginal;

    private final String ghpMeterInstallationCodeSeriesDedicated;

    private final String ghpMeterInstallationCodeParallel;

    private final String ghpMeterInstallationCodeNone;

    /**
     * コンストラクタ。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public GhpMeterInstallationCodeEditImpl(CiriusConstantService ciriusConstantService) {
        this.seriesOriginalGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                                .ORIGINAL_GAS_METER);
        this.seriesDedicatedGasMeterStatus =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus.TYPE,
                        ConstantLookupKeys.Cirius.SeriesGasMeterInstallationStatus
                                .DEDICATED_GAS_METER);
        this.equipmentGasMeterForGhp =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.EquipmentGasMeter.TYPE,
                        ConstantLookupKeys.Cirius.EquipmentGasMeter.FOR_GHP_EQUIPMENT);
        this.ghpMeterInstallationCodeSeriesOriginal =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.TYPE,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.SERIES_ORIGINAL_METER);
        this.ghpMeterInstallationCodeSeriesDedicated =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.TYPE,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.SERIES_DEDICATED_METER);
        this.ghpMeterInstallationCodeParallel =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.TYPE,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.PARALLEL);
        this.ghpMeterInstallationCodeNone =
                RequiredConstantResolver.requireCiriusConstant(
                        ciriusConstantService,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.TYPE,
                        ConstantLookupKeys.Cirius.GhpMeterInstallationCode.NO_GHP_INSTALLATION);
    }

    @Override
    public GhpMeterInstallationCodeEditOutput execute(String ckrtGmscJtkb, String kkyuGmtKbn) {
        if (seriesOriginalGasMeterStatus.equals(ckrtGmscJtkb)) {
            return new GhpMeterInstallationCodeEditOutput(ghpMeterInstallationCodeSeriesOriginal);
        }

        if (seriesDedicatedGasMeterStatus.equals(ckrtGmscJtkb)) {
            return new GhpMeterInstallationCodeEditOutput(ghpMeterInstallationCodeSeriesDedicated);
        }

        if (equipmentGasMeterForGhp.equals(kkyuGmtKbn)) {
            return new GhpMeterInstallationCodeEditOutput(ghpMeterInstallationCodeParallel);
        }

        return new GhpMeterInstallationCodeEditOutput(ghpMeterInstallationCodeNone);
    }
}
