package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransferWorkHistoryDetailCodeRetailAcquisition2;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransferWorkHistoryDetailCodeRetailAcquisition2Output;

/**
 * SV-270（異動・作業履歴業務詳細コード（小売）取得２）サービス実装。
 *
 * <p>主な仕様:
 * <ul>
 *   <li>移転解約は `CCBG6401`、不使用解約は `CCBG6402`、当社判断解約は `CCBG6403` を返却する。</li>
 *   <li>上記以外のサービス契約終了事由種別では `null` を返却する。</li>
 *   <li>判定値と返却値は `CiriusConstantService` で必須定数を解決した値を使用する。</li>
 * </ul>
 */
public final class TransferWorkHistoryDetailCodeRetailAcquisition2Impl
        implements TransferWorkHistoryDetailCodeRetailAcquisition2 {

    /** サービス契約終了事由種別コード（移転解約）。 */
    private final String transferCancellationReasonCode;

    /** サービス契約終了事由種別コード（不使用解約）。 */
    private final String nonUseCancellationReasonCode;

    /** サービス契約終了事由種別コード（当社判断解約）。 */
    private final String cancellationAtOurDiscretionReasonCode;

    /** 異動・作業履歴業務詳細コード（小売）：移転解約。 */
    private final String transferCancellationHistoryDetailCodeRetail;

    /** 異動・作業履歴業務詳細コード（小売）：不使用解約。 */
    private final String nonUseCancellationHistoryDetailCodeRetail;

    /** 異動・作業履歴業務詳細コード（小売）：当社判断解約。 */
    private final String cancellationAtOurDiscretionHistoryDetailCodeRetail;

    /**
     * 判定と返却に必要な CIRIUS 定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public TransferWorkHistoryDetailCodeRetailAcquisition2Impl(
            final CiriusConstantService ciriusConstantService) {
        final CiriusConstantService requiredService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        final String terminationReasonType =
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE;
        final String historyDetailType =
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.TYPE;

        this.transferCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .TRANSFER_CANCELLATION);

        this.nonUseCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .NON_USE_CANCELLATION);

        this.cancellationAtOurDiscretionReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .CANCELLATION_AT_OUR_DISCRETION);

        this.transferCancellationHistoryDetailCodeRetail =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail
                                .TRANSFER_CANCELLATION_CCBG6401);

        this.nonUseCancellationHistoryDetailCodeRetail =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail
                                .NON_USE_CANCELLATION_CCBG6402);

        this.cancellationAtOurDiscretionHistoryDetailCodeRetail =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail
                                .CANCELLATION_AT_OUR_DISCRETION);
    }

    /**
     * サービス契約終了事由種別に応じて異動・作業履歴業務詳細コード（小売）を返却する。
     *
     * @param svkyEndJysb サービス契約終了事由種別
     * @return 異動・作業履歴業務詳細コード（小売）
     */
    @Override
    public TransferWorkHistoryDetailCodeRetailAcquisition2Output execute(
            final String svkyEndJysb) {
        if (transferCancellationReasonCode.equals(svkyEndJysb)) {
            return output(transferCancellationHistoryDetailCodeRetail);
        }

        if (nonUseCancellationReasonCode.equals(svkyEndJysb)) {
            return output(nonUseCancellationHistoryDetailCodeRetail);
        }

        if (cancellationAtOurDiscretionReasonCode.equals(svkyEndJysb)) {
            return output(cancellationAtOurDiscretionHistoryDetailCodeRetail);
        }

        return output(null);
    }

    private TransferWorkHistoryDetailCodeRetailAcquisition2Output output(
            final String isrrGmssCd) {
        return new TransferWorkHistoryDetailCodeRetailAcquisition2Output(isrrGmssCd);
    }
}
