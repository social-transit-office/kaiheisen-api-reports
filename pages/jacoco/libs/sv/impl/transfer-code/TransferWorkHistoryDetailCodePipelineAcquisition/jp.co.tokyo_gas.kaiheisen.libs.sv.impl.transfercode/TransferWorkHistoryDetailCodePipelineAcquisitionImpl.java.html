package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransferWorkHistoryDetailCodePipelineAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransferWorkHistoryDetailCodePipelineAcquisitionOutput;

/**
 * SV-280（異動・作業履歴業務詳細コード（導管）取得）サービス実装。
 *
 * <p>主な仕様:
 * <ul>
 *   <li>移転解約は `CCJG1401`、不使用解約は `CCJG1402`、当社判断解約は `CCJG1403` を返却する。</li>
 *   <li>上記以外のサービス契約終了事由種別では `null` を返却する。</li>
 *   <li>判定値と返却値は `CiriusConstantService` で必須定数を解決した値を使用する。</li>
 * </ul>
 */
public final class TransferWorkHistoryDetailCodePipelineAcquisitionImpl
        implements TransferWorkHistoryDetailCodePipelineAcquisition {

    /** サービス契約終了事由種別コード（移転解約）。 */
    private final String transferCancellationReasonCode;

    /** サービス契約終了事由種別コード（不使用解約）。 */
    private final String nonUseCancellationReasonCode;

    /** サービス契約終了事由種別コード（当社判断解約）。 */
    private final String cancellationAtOurDiscretionReasonCode;

    /** 異動・作業履歴業務詳細コード（導管）：移転解約。 */
    private final String transferCancellationHistoryDetailCodeConduit;

    /** 異動・作業履歴業務詳細コード（導管）：不使用解約。 */
    private final String nonUseCancellationHistoryDetailCodeConduit;

    /** 異動・作業履歴業務詳細コード（導管）：当社判断解約。 */
    private final String retailDecisionCancellationHistoryDetailCodeConduit;

    /**
     * 判定と返却に必要な CIRIUS 定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public TransferWorkHistoryDetailCodePipelineAcquisitionImpl(
            final CiriusConstantService ciriusConstantService) {
        final CiriusConstantService requiredService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        final String terminationReasonType =
                ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE;
        final String historyDetailType =
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.TYPE;

        this.transferCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .TRANSFER_CANCELLATION);

        this.nonUseCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .NON_USE_CANCELLATION);

        this.cancellationAtOurDiscretionReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        terminationReasonType,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .CANCELLATION_AT_OUR_DISCRETION);

        this.transferCancellationHistoryDetailCodeConduit =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit
                                .TRANSFER_CANCELLATION);

        this.nonUseCancellationHistoryDetailCodeConduit =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit
                                .NON_USE_CANCELLATION);

        this.retailDecisionCancellationHistoryDetailCodeConduit =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        historyDetailType,
                        ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit
                                .RETAIL_DECISION_CANCELLATION);
    }

    /**
     * サービス契約終了事由種別に応じて異動・作業履歴業務詳細コード（導管）を返却する。
     *
     * @param svkyEndJysb サービス契約終了事由種別
     * @return 異動・作業履歴業務詳細コード（導管）
     */
    @Override
    public TransferWorkHistoryDetailCodePipelineAcquisitionOutput execute(
            final String svkyEndJysb) {
        if (transferCancellationReasonCode.equals(svkyEndJysb)) {
            return output(transferCancellationHistoryDetailCodeConduit);
        }

        if (nonUseCancellationReasonCode.equals(svkyEndJysb)) {
            return output(nonUseCancellationHistoryDetailCodeConduit);
        }

        if (cancellationAtOurDiscretionReasonCode.equals(svkyEndJysb)) {
            return output(retailDecisionCancellationHistoryDetailCodeConduit);
        }

        return output(null);
    }

    private TransferWorkHistoryDetailCodePipelineAcquisitionOutput output(
            final String duknIsrsCd) {
        return new TransferWorkHistoryDetailCodePipelineAcquisitionOutput(duknIsrsCd);
    }
}
