package jp.co.tokyo_gas.kaiheisen.libs.sv.impl.transfercode;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.RequiredConstantResolver;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransportServiceContractTerminationReasonTypeAcquisition;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.transfercode.TransportServiceContractTerminationReasonTypeAcquisitionOutput;

/**
 * SV-199（託送サービス契約終了事由種別取得）サービス実装。
 *
 * <p>主な仕様:
 * <ul>
 *   <li>移転解約は `J0690 01`、不使用解約は `J0690 02`、当社判断解約は `J0690 03` を返却する。</li>
 *   <li>上記以外のサービス契約終了事由種別では `null` を返却する。</li>
 *   <li>判定値と返却値は `CiriusConstantService` で必須定数を解決した値を使用する。</li>
 * </ul>
 */
public final class TransportServiceContractTerminationReasonTypeAcquisitionImpl
        implements TransportServiceContractTerminationReasonTypeAcquisition {

    /** サービス契約終了事由種別コード（移転解約）。 */
    private final String transferCancellationReasonCode;

    /** サービス契約終了事由種別コード（不使用解約）。 */
    private final String nonUseCancellationReasonCode;

    /** サービス契約終了事由種別コード（当社判断解約）。 */
    private final String cancellationAtOurDiscretionReasonCode;

    /** 託送サービス契約終了事由種別コード（移転解約）。 */
    private final String transportTransferCancellationReasonCode;

    /** 託送サービス契約終了事由種別コード（不使用解約）。 */
    private final String transportNonUseCancellationReasonCode;

    /** 託送サービス契約終了事由種別コード（小売判断解約）。 */
    private final String transportRetailDecisionCancellationReasonCode;

    /**
     * 判定と返却に必要な CIRIUS 定数を初期化する。
     *
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    public TransportServiceContractTerminationReasonTypeAcquisitionImpl(
            final CiriusConstantService ciriusConstantService) {
        final CiriusConstantService requiredService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.transferCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .TRANSFER_CANCELLATION);

        this.nonUseCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .NON_USE_CANCELLATION);

        this.cancellationAtOurDiscretionReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ServiceContractTerminationReasonType
                                .CANCELLATION_AT_OUR_DISCRETION);

        this.transportTransferCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract.TYPE,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract
                                .TRANSFER_CANCELLATION);

        this.transportNonUseCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract.TYPE,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract
                                .NON_USE_CANCELLATION);

        this.transportRetailDecisionCancellationReasonCode =
                RequiredConstantResolver.requireCiriusConstant(
                        requiredService,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract.TYPE,
                        ConstantLookupKeys.Cirius
                                .TypeOfReasonForTerminationOfTransportationServiceContract
                                .RETAIL_DECISION_CANCELLATION);
    }

    /**
     * サービス契約終了事由種別に応じて託送サービス契約終了事由種別を返却する。
     *
     * @param svkyEndJysb サービス契約終了事由種別
     * @return 託送サービス契約終了事由種別
     */
    @Override
    public TransportServiceContractTerminationReasonTypeAcquisitionOutput execute(
            final String svkyEndJysb) {
        if (transferCancellationReasonCode.equals(svkyEndJysb)) {
            return output(transportTransferCancellationReasonCode);
        }

        if (nonUseCancellationReasonCode.equals(svkyEndJysb)) {
            return output(transportNonUseCancellationReasonCode);
        }

        if (cancellationAtOurDiscretionReasonCode.equals(svkyEndJysb)) {
            return output(transportRetailDecisionCancellationReasonCode);
        }

        return output(null);
    }

    private TransportServiceContractTerminationReasonTypeAcquisitionOutput output(
            final String tksuSvkeJysb) {
        return new TransportServiceContractTerminationReasonTypeAcquisitionOutput(tksuSvkeJysb);
    }
}
