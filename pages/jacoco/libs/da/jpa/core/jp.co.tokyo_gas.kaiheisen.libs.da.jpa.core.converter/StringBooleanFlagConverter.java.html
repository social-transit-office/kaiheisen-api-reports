package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

import java.util.Locale;

/** フラグ文字列（"1"/"Y"/"T"/"true" 等）を {@link Boolean} に変換する converter。 */
public final class StringBooleanFlagConverter implements Converter<String, Boolean> {

    /** シングルトンインスタンス。 */
    public static final StringBooleanFlagConverter INSTANCE = new StringBooleanFlagConverter();

    private StringBooleanFlagConverter() { }

    @Override
    public Boolean convert(String source) {
        if (source == null) {
            return null;
        }
        String normalized = source.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        String lower = normalized.toLowerCase(Locale.ROOT);
        if (lower.equals("1") || lower.equals("y") || lower.equals("t") || lower.equals("true")) {
            return Boolean.TRUE;
        }
        if (lower.equals("0") || lower.equals("n") || lower.equals("f") || lower.equals("false")) {
            return Boolean.FALSE;
        }
        return null;
    }
}
