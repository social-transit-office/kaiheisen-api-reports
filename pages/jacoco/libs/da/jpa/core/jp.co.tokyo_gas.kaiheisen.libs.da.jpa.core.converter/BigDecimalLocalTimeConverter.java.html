package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

import java.math.BigDecimal;
import java.time.LocalTime;

/** 数値時刻の {@link BigDecimal} を {@link LocalTime} に変換する。 */
public final class BigDecimalLocalTimeConverter implements Converter<BigDecimal, LocalTime> {

    public static final BigDecimalLocalTimeConverter INSTANCE = new BigDecimalLocalTimeConverter();

    private BigDecimalLocalTimeConverter() {
    }

    @Override
    public LocalTime convert(BigDecimal source) {
        if (source == null) {
            return null;
        }
        BigDecimal normalized = source.stripTrailingZeros();
        if (normalized.scale() > 0) {
            throw new IllegalArgumentException("Fractional LocalTime value is not supported: " + source);
        }
        final long numeric;
        try {
            numeric = normalized.longValueExact();
        } catch (ArithmeticException ex) {
            throw new IllegalArgumentException("HHmm LocalTime value is out of range: " + source, ex);
        }
        if (numeric <= 0L) {
            return null;
        }
        if (numeric > 2400L) {
            throw new IllegalArgumentException("HHmm LocalTime value is out of range: " + source);
        }
        if (numeric == 2400L) {
            return LocalTime.MIDNIGHT;
        }
        long hour = numeric / 100L;
        long minute = numeric % 100L;
        if (hour > 23L || minute > 59L) {
            throw new IllegalArgumentException("HHmm LocalTime value is out of range: " + source);
        }
        return LocalTime.of((int) hour, (int) minute);
    }
}
