package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

/** {@link Boolean} を "1"/"0" のフラグ文字列に変換する converter。 */
public final class BooleanFlagStringConverter implements Converter<Boolean, String> {

    /** シングルトンインスタンス。 */
    public static final BooleanFlagStringConverter INSTANCE = new BooleanFlagStringConverter();

    private BooleanFlagStringConverter() { }

    @Override
    public String convert(Boolean source) {
        if (source == null) {
            return null;
        }
        return source ? "1" : "0";
    }
}
