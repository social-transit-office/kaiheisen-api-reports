package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

/** {@link LocalDate} を yyyyMMdd 形式の {@link BigDecimal} に変換する。 */
public final class LocalDateYmdBigDecimalConverter implements Converter<LocalDate, BigDecimal> {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    public static final LocalDateYmdBigDecimalConverter INSTANCE =
            new LocalDateYmdBigDecimalConverter();

    private LocalDateYmdBigDecimalConverter() { }

    @Override
    public BigDecimal convert(LocalDate source) {
        if (source == null) {
            return null;
        }
        return new BigDecimal(source.format(BASIC_DATE));
    }
}
