package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

/** {@link Short} を {@link Integer} に変換し、0 は {@code null} として扱う。 */
public final class ShortNullIfZeroIntegerConverter implements Converter<Short, Integer> {

    public static final ShortNullIfZeroIntegerConverter INSTANCE =
            new ShortNullIfZeroIntegerConverter();

    private ShortNullIfZeroIntegerConverter() { }

    @Override
    public Integer convert(Short source) {
        if (source == null) {
            return null;
        }
        int converted = source.intValue();
        if (converted == 0) {
            return null;
        }
        return converted;
    }
}
