package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

/** yyyyMMdd 形式の {@link BigDecimal} を {@link LocalDate} に変換する。 */
public final class YmdBigDecimalLocalDateConverter implements Converter<BigDecimal, LocalDate> {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    public static final YmdBigDecimalLocalDateConverter INSTANCE =
            new YmdBigDecimalLocalDateConverter();

    private YmdBigDecimalLocalDateConverter() { }

    @Override
    public LocalDate convert(BigDecimal source) {
        if (source == null) {
            return null;
        }
        long ymd = source.longValue();
        if (ymd <= 0L) {
            return null;
        }
        return LocalDate.parse(String.format("%08d", ymd), BASIC_DATE);
    }
}
