package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter;

import java.math.BigDecimal;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;

/** {@link BigDecimal}（yyyyMM 形式）を {@link YearMonth} に変換する converter。 */
public final class BigDecimalYearMonthConverter implements Converter<BigDecimal, YearMonth> {

    /** シングルトンインスタンス。 */
    public static final BigDecimalYearMonthConverter INSTANCE = new BigDecimalYearMonthConverter();

    private BigDecimalYearMonthConverter() { }

    @Override
    public YearMonth convert(BigDecimal source) {
        if (source == null) {
            return null;
        }
        int ym = source.intValue();
        if (ym <= 0) {
            return null;
        }
        int year = ym / 100;
        int month = ym % 100;
        try {
            return YearMonth.of(year, month);
        } catch (IllegalArgumentException ex) {
            return null;
        }
    }
}
