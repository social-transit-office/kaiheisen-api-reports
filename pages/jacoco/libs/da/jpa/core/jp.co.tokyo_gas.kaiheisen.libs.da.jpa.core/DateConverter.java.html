package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;

/**
 * 日付（YYYYMMDD 形式）を {@link LocalDate} に変換するユーティリティ。
 *
 * @deprecated {@code converter/YmdBigDecimalLocalDateConverter} および {@code
 *     converter/LocalDateYmdBigDecimalConverter} を使用すること。
 */
@Deprecated(forRemoval = true)
public final class DateConverter {

    /** YYYYMMDD 形式のフォーマッタ。 */
    private static final DateTimeFormatter YMD_FORMAT = DateTimeFormatter.BASIC_ISO_DATE;

    /** 外部からのインスタンス生成を禁止する。 */
    private DateConverter() { }

    /**
     * YYYYMMDD 形式の数値を {@link LocalDate} へ変換する。
     *
     * <p>null または 0 の場合は null を返す。
     *
     * @param ymd YYYYMMDD 形式の数値
     * @return 変換後の {@link LocalDate}、変換できない場合は null
     */
    public static LocalDate toLocalDate(BigDecimal ymd) {
        if (ymd == null || ymd.compareTo(BigDecimal.ZERO) == 0) {
            return null;
        }

        return LocalDate.parse(ymd.toBigInteger().toString(), YMD_FORMAT);
    }

    /**
     * {@link LocalDate} を YYYYMMDD 形式の数値へ変換する。
     *
     * <p>null の場合は 0 を返す。
     *
     * @param date 日付
     * @return YYYYMMDD 形式の数値
     */
    public static BigDecimal toYmd(LocalDate date) {

        if (date == null) {
            return BigDecimal.ZERO;
        }

        String ymd = date.format(YMD_FORMAT);

        return new BigDecimal(ymd);
    }
}
