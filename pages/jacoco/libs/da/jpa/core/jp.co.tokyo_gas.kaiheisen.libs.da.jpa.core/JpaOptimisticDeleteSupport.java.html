package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.ExclusiveException;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.SqlCapture;

/** native DELETE 向けの排他制御 helper。 */
public final class JpaOptimisticDeleteSupport {

    private final EntityManager entityManager;

    private final String tableName;

    /**
     * helper を生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param tableName 物理テーブル名
     */
    public JpaOptimisticDeleteSupport(EntityManager entityManager, String tableName) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.tableName = Objects.requireNonNull(tableName, "tableName");
    }

    /**
     * 排他制御なしの delete を実行する。
     *
     * @param where WHERE 句 helper
     * @return 削除件数
     */
    public int delete(PreparedWhere where) {
        return delete(where, null, null, () -> new DataInconsistencyException(tableName, null, "{}"));
    }

    /**
     * 排他制御付き delete を実行する。
     *
     * @param where 排他制御列を除く WHERE 句 helper
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @return 削除件数
     */
    public int delete(PreparedWhere where, Integer expectedRecUpdJnjNo, String daIdentifier) {
        return delete(
                where,
                expectedRecUpdJnjNo,
                daIdentifier,
                () -> new DataInconsistencyException(tableName, daIdentifier, "{}"));
    }

    /**
     * 排他制御付き delete を実行する。
     *
     * @param where 排他制御列を除く WHERE 句 helper
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param notFoundExceptionSupplier 対象データなし時の例外生成
     * @return 削除件数
     */
    public int delete(
            PreparedWhere where,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Supplier<? extends RuntimeException> notFoundExceptionSupplier) {
        Objects.requireNonNull(where, "where");
        if (expectedRecUpdJnjNo != null) {
            Objects.requireNonNull(daIdentifier, "daIdentifier");
        }

        String sql =
                "DELETE FROM "
                        + tableName
                        + " WHERE "
                        + where.clause()
                        + (expectedRecUpdJnjNo == null ? "" : " AND REC_UPD_JNJ_NO = ?");
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        int nextIndex = where.bind(query, 1);
        if (expectedRecUpdJnjNo != null) {
            query.setParameter(nextIndex, JpaDbWriteValues.bigDecimal(expectedRecUpdJnjNo));
        }
        int deleted = query.executeUpdate();
        if (deleted > 0 || expectedRecUpdJnjNo == null) {
            return deleted;
        }
        if (existsByBaseKey(where)) {
            throw new ExclusiveException(
                    daIdentifier,
                    sql,
                    optimisticConflictBindParameters(where, expectedRecUpdJnjNo));
        }
        throw Objects.requireNonNull(notFoundExceptionSupplier, "notFoundExceptionSupplier").get();
    }

    private Object[] optimisticConflictBindParameters(
            PreparedWhere where, Integer expectedRecUpdJnjNo) {
        List<Object> bindParameters = new ArrayList<>(where.bindValues());
        bindParameters.add(
                JpaDbWriteValues.bigDecimal(
                        Objects.requireNonNull(expectedRecUpdJnjNo, "expectedRecUpdJnjNo")));
        return bindParameters.toArray();
    }

    private boolean existsByBaseKey(PreparedWhere where) {
        String sql = "SELECT 1 FROM " + tableName + " WHERE " + where.clause();
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        where.bind(query, 1);
        query.setMaxResults(1);
        return !query.getResultList().isEmpty();
    }
}
