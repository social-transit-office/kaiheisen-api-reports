package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import java.util.function.BiConsumer;
import java.util.function.Function;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.ExclusiveException;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.SqlCapture;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.TableWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WriteColumnResolver;

/**
 * targetColumns 指定型 UPDATE を実行する reusable helper。
 *
 * @param <RecordT> API レコード型
 * @param <EntityT> JPA エンティティ型
 * @param <ColumnT> 列型
 */
public final class JpaTargetColumnUpdateSupport<
        RecordT, EntityT, ColumnT extends Enum<ColumnT> & DataAccessColumn> {

    private final EntityManager entityManager;

    private final String tableName;

    private final TableWriteDefinition<RecordT, ColumnT> definition;

    private final JpaEntityWriteSupport<RecordT, EntityT> entityWriteSupport;

    /**
     * helper を生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param tableName 物理テーブル名
     * @param definition テーブル write 定義
     * @param entityClass JPA エンティティ型
     * @param entityMapper Record/Entity 変換
     */
    public JpaTargetColumnUpdateSupport(
            EntityManager entityManager,
            String tableName,
            TableWriteDefinition<RecordT, ColumnT> definition,
            Class<EntityT> entityClass,
            Function<RecordT, EntityT> entityMapper) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.tableName = Objects.requireNonNull(tableName, "tableName");
        this.definition = Objects.requireNonNull(definition, "definition");
        this.entityWriteSupport = new JpaEntityWriteSupport<>(entityClass, entityMapper);
    }

    /**
     * targetColumns 指定型 UPDATE を実行する。
     *
     * @param record 更新値
     * @param targetColumns SET 対象列
     * @param updateDefaultsApplier update 前の DA 管理列設定
     * @param where WHERE 句 helper
     * @return 更新件数
     */
    public int update(
            RecordT record,
            Set<ColumnT> targetColumns,
            BiConsumer<EntityT, List<? extends DataAccessColumn>> updateDefaultsApplier,
            PreparedWhere where) {
        return update(record, targetColumns, updateDefaultsApplier, where, null, null);
    }

    /**
     * 排他制御付き targetColumns 指定型 UPDATE を実行する。
     *
     * @param record 更新値
     * @param targetColumns SET 対象列
     * @param updateDefaultsApplier update 前の DA 管理列設定
     * @param where 排他制御列を除く WHERE 句 helper
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @return 更新件数
     */
    public int update(
            RecordT record,
            Set<ColumnT> targetColumns,
            BiConsumer<EntityT, List<? extends DataAccessColumn>> updateDefaultsApplier,
            PreparedWhere where,
            Integer expectedRecUpdJnjNo,
            String daIdentifier) {
        return update(
                record,
                targetColumns,
                updateDefaultsApplier,
                where,
                expectedRecUpdJnjNo,
                daIdentifier,
                () -> new DataInconsistencyException(tableName, daIdentifier, "{}"));
    }

    /**
     * 排他制御付き targetColumns 指定型 UPDATE を実行する。
     *
     * @param record 更新値
     * @param targetColumns SET 対象列
     * @param updateDefaultsApplier update 前の DA 管理列設定
     * @param where 排他制御列を除く WHERE 句 helper
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param notFoundExceptionSupplier 対象データなし時の例外生成
     * @return 更新件数
     */
    public int update(
            RecordT record,
            Set<ColumnT> targetColumns,
            BiConsumer<EntityT, List<? extends DataAccessColumn>> updateDefaultsApplier,
            PreparedWhere where,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Supplier<? extends RuntimeException> notFoundExceptionSupplier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateDefaultsApplier, "updateDefaultsApplier");
        Objects.requireNonNull(where, "where");
        if (expectedRecUpdJnjNo != null) {
            Objects.requireNonNull(daIdentifier, "daIdentifier");
        }

        List<ColumnT> orderedPublicColumns =
                WriteColumnResolver.resolveUpdatableColumns(definition, targetColumns, false);
        if (orderedPublicColumns.isEmpty()) {
            return 0;
        }

        List<ColumnT> effectiveColumns =
                WriteColumnResolver.effectiveColumns(
                        orderedPublicColumns, definition.updateManagedColumns());
        EntityT entity = entityWriteSupport.toEntity(record);
        updateDefaultsApplier.accept(entity, effectiveColumns);
        entityWriteSupport.complementNotNullColumns(entity, effectiveColumns);
        PreparedNativeUpdate prepared =
                PreparedNativeUpdate.of(
                        effectiveColumns,
                        entityWriteSupport.toBindValues(entity, effectiveColumns));

        String effectiveWhereClause =
                expectedRecUpdJnjNo == null
                        ? where.clause()
                        : where.clause() + " AND REC_UPD_JNJ_NO = ?";
        String sql =
                prepared.includesRecUpdJnjNo()
                        ? prepared.sqlIncrementingRecUpdJnjNo(tableName, effectiveWhereClause)
                        : prepared.sql(tableName, effectiveWhereClause);
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        int nextIndex =
                prepared.includesRecUpdJnjNo()
                        ? prepared.bindSetParametersSkippingRecUpdJnjNo(query, 1)
                        : prepared.bindSetParameters(query, 1);
        nextIndex = where.bind(query, nextIndex);
        if (expectedRecUpdJnjNo != null) {
            query.setParameter(nextIndex, JpaDbWriteValues.bigDecimal(expectedRecUpdJnjNo));
        }
        int updated = query.executeUpdate();
        if (updated > 0 || expectedRecUpdJnjNo == null) {
            return updated;
        }
        if (existsByBaseKey(where)) {
            throw new ExclusiveException(
                    daIdentifier,
                    sql,
                    optimisticConflictBindParameters(prepared, where, expectedRecUpdJnjNo));
        }
        throw Objects.requireNonNull(notFoundExceptionSupplier, "notFoundExceptionSupplier").get();
    }

    private Object[] optimisticConflictBindParameters(
            PreparedNativeUpdate prepared, PreparedWhere where, Integer expectedRecUpdJnjNo) {
        List<Object> bindParameters = new ArrayList<>(prepared.effectiveSetBindValues());
        bindParameters.addAll(where.bindValues());
        bindParameters.add(
                JpaDbWriteValues.bigDecimal(
                        Objects.requireNonNull(expectedRecUpdJnjNo, "expectedRecUpdJnjNo")));
        return bindParameters.toArray();
    }

    private boolean existsByBaseKey(PreparedWhere where) {
        String sql = "SELECT 1 FROM " + tableName + " WHERE " + where.clause();
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        where.bind(query, 1);
        query.setMaxResults(1);
        return !query.getResultList().isEmpty();
    }
}
