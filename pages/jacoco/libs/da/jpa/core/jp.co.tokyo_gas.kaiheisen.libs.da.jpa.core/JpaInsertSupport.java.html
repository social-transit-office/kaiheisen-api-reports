package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import java.util.EnumMap;
import java.util.List;
import java.util.Objects;
import java.util.function.Consumer;
import java.util.function.Function;
import java.util.stream.Collectors;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.SqlCapture;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.TableWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WriteColumnResolver;

/**
 * write record から解決できた公開列と insert 管理列を対象に INSERT を実行する reusable helper。
 *
 * @param <RecordT> API レコード型
 * @param <EntityT> JPA エンティティ型
 * @param <ColumnT> 列型
 */
public final class JpaInsertSupport<
        RecordT, EntityT, ColumnT extends Enum<ColumnT> & DataAccessColumn> {

    private final EntityManager entityManager;

    private final String tableName;

    private final TableWriteDefinition<RecordT, ColumnT> definition;

    private final JpaEntityWriteSupport<RecordT, EntityT> entityWriteSupport;

    /**
     * helper を生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param tableName 物理テーブル名
     * @param definition テーブル write 定義
     * @param entityClass JPA エンティティ型
     * @param entityMapper Record/Entity 変換
     */
    public JpaInsertSupport(
            EntityManager entityManager,
            String tableName,
            TableWriteDefinition<RecordT, ColumnT> definition,
            Class<EntityT> entityClass,
            Function<RecordT, EntityT> entityMapper) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.tableName = Objects.requireNonNull(tableName, "tableName");
        this.definition = Objects.requireNonNull(definition, "definition");
        this.entityWriteSupport = new JpaEntityWriteSupport<>(entityClass, entityMapper);
    }

    /**
     * write record から解決できた公開列と insert 管理列で INSERT を実行する。
     *
     * @param record 登録値
     * @return 登録件数
     */
    public int insert(RecordT record) {
        return insert(record, entity -> { });
    }

    /**
     * write record から解決できた公開列と insert 管理列で INSERT を実行する。
     *
     * @param record 登録値
     * @param insertDefaultsApplier insert 前の DA 管理列設定
     * @return 登録件数
     * @throws IllegalArgumentException INSERT 対象列が 1 件も解決できない場合
     */
    public int insert(RecordT record, Consumer<EntityT> insertDefaultsApplier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(insertDefaultsApplier, "insertDefaultsApplier");
        List<ColumnT> insertColumns = resolveInsertColumns(record);
        if (insertColumns.isEmpty()) {
            throw new IllegalArgumentException(
                    "No insert columns resolved for table: " + tableName);
        }
        EntityT entity = entityWriteSupport.toEntity(record);
        insertDefaultsApplier.accept(entity);
        entityWriteSupport.complementNotNullColumns(entity, insertColumns);

        String sql = buildInsertSql(insertColumns);
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        entityWriteSupport.bindColumnValues(query, insertColumns, entity, 1);
        return query.executeUpdate();
    }

    private List<ColumnT> resolveInsertColumns(RecordT record) {
        EnumMap<ColumnT, Object> publicValues = WriteColumnResolver.extractValues(definition, record);
        List<ColumnT> publicColumns = definition.columnOrder().stream()
                .filter(publicValues::containsKey)
                .toList();
        List<ColumnT> effectiveColumns =
            WriteColumnResolver.effectiveColumns(
                publicColumns, definition.insertManagedColumns());
        return definition.columnOrder().stream()
            .filter(effectiveColumns::contains)
            .toList();
    }

    private String buildInsertSql(List<? extends DataAccessColumn> insertColumns) {
        String columnList =
                insertColumns.stream()
                        .map(DataAccessColumn::columnName)
                        .collect(Collectors.joining(", "));
        String values = insertColumns.stream().map(ignored -> "?").collect(Collectors.joining(", "));
        return "INSERT INTO " + tableName + " (" + columnList + ") VALUES (" + values + ")";
    }
}
