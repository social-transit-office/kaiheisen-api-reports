package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.Column;
import java.lang.reflect.Field;
import java.math.BigDecimal;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentMap;

/** JPA エンティティの {@link Column} メタデータを解決するユーティリティ。 */
public final class EntityColumnMetadataResolver {

    private static final ConcurrentMap<Class<?>, Map<String, EntityColumnMetadata>> CACHE =
            new ConcurrentHashMap<>();

    private EntityColumnMetadataResolver() { }

    /**
     * エンティティクラスの列メタデータを列名キーで返す。
     *
     * @param entityClass エンティティクラス
     * @return 列メタデータ
     */
    public static Map<String, EntityColumnMetadata> resolve(Class<?> entityClass) {
        Objects.requireNonNull(entityClass, "entityClass");
        return CACHE.computeIfAbsent(entityClass, EntityColumnMetadataResolver::scan);
    }

    private static Map<String, EntityColumnMetadata> scan(Class<?> entityClass) {
        Map<String, EntityColumnMetadata> resolved = new LinkedHashMap<>();
        Class<?> current = entityClass;
        while (current != null && current != Object.class) {
            for (Field field : current.getDeclaredFields()) {
                Column column = field.getAnnotation(Column.class);
                if (column == null) {
                    continue;
                }
                field.setAccessible(true);
                resolved.putIfAbsent(
                        column.name(),
                        new EntityColumnMetadata(column.name(), field, column.nullable()));
            }
            current = current.getSuperclass();
        }
        return Map.copyOf(resolved);
    }

    /**
     * 列メタデータ。
     *
     * @param columnName 列物理名
     * @param field 対応フィールド
     * @param nullable nullable 属性
     */
    public record EntityColumnMetadata(String columnName, Field field, boolean nullable) {

        /**
         * フィールド値を取得する。
         *
         * @param entity 対象エンティティ
         * @return 現在値
         */
        public Object getValue(Object entity) {
            try {
                return field.get(entity);
            } catch (IllegalAccessException ex) {
                throw new IllegalStateException("Failed to access field: " + field.getName(), ex);
            }
        }

        /**
         * フィールド値を設定する。
         *
         * @param entity 対象エンティティ
         * @param value 設定値
         */
        public void setValue(Object entity, Object value) {
            try {
                field.set(entity, value);
            } catch (IllegalAccessException ex) {
                throw new IllegalStateException("Failed to access field: " + field.getName(), ex);
            }
        }

        /**
         * 列の Java 型を返す。
         *
         * @return Java 型
         */
        public Class<?> javaType() {
            return field.getType();
        }

        /**
         * NOT NULL 補完に使う中立値を返す。
         *
         * @return 補完値。対応外型は {@code null}
         */
        public Object neutralValue() {
            if (nullable) {
                return null;
            }
            Class<?> type = javaType();
            if (type == String.class) {
                return "";
            }
            if (type == BigDecimal.class) {
                return BigDecimal.ZERO;
            }
            if (type == Integer.class) {
                return 0;
            }
            if (type == Long.class) {
                return 0L;
            }
            if (type == Short.class) {
                return (short) 0;
            }
            return null;
        }
    }
}
