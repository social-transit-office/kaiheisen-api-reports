package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import com.blazebit.persistence.BaseWhereBuilder;
import java.util.Collection;
import java.util.Objects;
import java.util.function.Function;

/** Blaze-Persistence の WHERE 句組み立てを簡潔にする helper。 */
public final class JpaWhere {

    private JpaWhere() { }

    /**
     * 等価条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T eq(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).eq(value);
    }

    /**
     * 値変換を行ってから等価条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T eq(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return eq(builder, expression, converter.apply(source));
    }

    /**
     * 非等価条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T notEq(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).notEq(value);
    }

    /**
     * 値変換を行ってから非等価条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T notEq(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return notEq(builder, expression, converter.apply(source));
    }

    /**
     * 大なり条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T gt(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).gt(value);
    }

    /**
     * 値変換を行ってから大なり条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T gt(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return gt(builder, expression, converter.apply(source));
    }

    /**
     * 以上条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T ge(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).ge(value);
    }

    /**
     * 値変換を行ってから以上条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T ge(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return ge(builder, expression, converter.apply(source));
    }

    /**
     * 小なり条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T lt(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).lt(value);
    }

    /**
     * 値変換を行ってから小なり条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T lt(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return lt(builder, expression, converter.apply(source));
    }

    /**
     * 以下条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param value 比較値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T le(
            BaseWhereBuilder<T> builder, String expression, Object value) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).le(value);
    }

    /**
     * 値変換を行ってから以下条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param source 変換前の値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T le(
            BaseWhereBuilder<T> builder,
            String expression,
            S source,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return le(builder, expression, converter.apply(source));
    }

    /**
     * BETWEEN 条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param start 開始値
     * @param end 終了値
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T between(
            BaseWhereBuilder<T> builder, String expression, Object start, Object end) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        return builder.where(expression).between(start).and(end);
    }

    /**
     * 値変換を行ってから BETWEEN 条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param startSource 変換前の開始値
     * @param endSource 変換前の終了値
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T between(
            BaseWhereBuilder<T> builder,
            String expression,
            S startSource,
            S endSource,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return between(builder, expression, converter.apply(startSource), converter.apply(endSource));
    }

    /**
     * IN 条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param values 比較値コレクション
     * @param <T> builder 型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>> T in(
            BaseWhereBuilder<T> builder, String expression, Collection<?> values) {
        Objects.requireNonNull(builder, "builder");
        Objects.requireNonNull(expression, "expression");
        Objects.requireNonNull(values, "values");
        return builder.where(expression).in(values);
    }

    /**
     * 値変換を行ってから IN 条件を追加する。
     *
     * @param builder 対象 builder
     * @param expression 条件式
     * @param sources 変換前の値コレクション
     * @param converter 値変換
     * @param <T> builder 型
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return builder
     */
    public static <T extends BaseWhereBuilder<T>, S, V> T in(
            BaseWhereBuilder<T> builder,
            String expression,
            Collection<S> sources,
            Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return in(builder, expression, sources.stream().map(converter).toList());
    }
}
