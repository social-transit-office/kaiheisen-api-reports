package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.Query;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;

/**
 * JPA write helper 間で共有する mapper / metadata / bind 処理を提供する内部支援クラス。
 *
 * @param <RecordT> API レコード型
 * @param <EntityT> JPA エンティティ型
 */
final class JpaEntityWriteSupport<RecordT, EntityT> {

    private final Function<RecordT, EntityT> entityMapper;

    private final Map<String, EntityColumnMetadataResolver.EntityColumnMetadata>
            metadataByColumnName;

    JpaEntityWriteSupport(Class<EntityT> entityClass, Function<RecordT, EntityT> entityMapper) {
        this.entityMapper = Objects.requireNonNull(entityMapper, "entityMapper");
        this.metadataByColumnName =
                EntityColumnMetadataResolver.resolve(
                        Objects.requireNonNull(entityClass, "entityClass"));
    }

    EntityT toEntity(RecordT record) {
        Objects.requireNonNull(record, "record");
        return entityMapper.apply(record);
    }

    void complementNotNullColumns(EntityT entity, List<? extends DataAccessColumn> columns) {
        Objects.requireNonNull(entity, "entity");
        Objects.requireNonNull(columns, "columns");
        for (DataAccessColumn column : columns) {
            EntityColumnMetadataResolver.EntityColumnMetadata metadata = metadata(column);
            if (metadata.getValue(entity) != null) {
                continue;
            }
            Object neutralValue = metadata.neutralValue();
            if (neutralValue != null) {
                metadata.setValue(entity, neutralValue);
            }
        }
    }

    int bindColumnValues(
            Query query, List<? extends DataAccessColumn> columns, EntityT entity, int startIndex) {
        Objects.requireNonNull(query, "query");
        Objects.requireNonNull(columns, "columns");
        Objects.requireNonNull(entity, "entity");
        int nextIndex = startIndex;
        for (DataAccessColumn column : columns) {
            EntityColumnMetadataResolver.EntityColumnMetadata metadata = metadata(column);
            query.setParameter(nextIndex++, metadata.getValue(entity));
        }
        return nextIndex;
    }

    List<Object> toBindValues(EntityT entity, List<? extends DataAccessColumn> columns) {
        Objects.requireNonNull(entity, "entity");
        Objects.requireNonNull(columns, "columns");
        return columns.stream()
                .map(this::metadata)
                .map(metadata -> metadata.getValue(entity))
                .toList();
    }

    private EntityColumnMetadataResolver.EntityColumnMetadata metadata(DataAccessColumn column) {
        Objects.requireNonNull(column, "column");
        EntityColumnMetadataResolver.EntityColumnMetadata metadata =
                metadataByColumnName.get(column.columnName());
        if (metadata == null) {
            throw new IllegalStateException("Column metadata not found: " + column.columnName());
        }
        return metadata;
    }
}
