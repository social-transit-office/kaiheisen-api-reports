package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import java.lang.reflect.Constructor;
import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.lang.reflect.RecordComponent;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalTime;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.PreserveZeroValue;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalYearMonthConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BooleanFlagStringConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.StringBooleanFlagConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.YearMonthBigDecimalConverter;

/**
 * Entity と Record の変換をリフレクションで自動実行するマッパー。
 *
 * <p>Entity フィールドと Record コンポーネントを名前照合し型変換を自動適用する。
 *
 * <p><b>Entity → Record 変換ルール:</b>
 *
 * <ul>
 *   <li>BigDecimal → Long: {@link PreserveZeroValue} ありは 0 を保持、なしは null-if-zero
 *   <li>BigDecimal → Integer: {@link PreserveZeroValue} ありは 0 を保持、なしは null-if-zero
 *   <li>BigDecimal → LocalDate: YYYYMMDD 変換
 *   <li>BigDecimal → LocalTime: 数値時刻変換
 *   <li>BigDecimal → YearMonth: YYYYMM 変換
 *   <li>Short → Integer: plain conversion
 *   <li>String → String: 末尾の半角/全角空白除去
 *   <li>String → Boolean: フラグ変換
 *   <li>同一型: identity
 * </ul>
 *
 * <p><b>Record → Entity 変換ルール:</b>
 *
 * <ul>
 *   <li>Long → BigDecimal
 *   <li>Integer → BigDecimal
 *   <li>LocalDate → BigDecimal (YYYYMMDD)。{@link PreserveZeroValue} ありは null を 0 として書き込む
 *   <li>LocalTime → BigDecimal (HHmm)
 *   <li>YearMonth → BigDecimal (YYYYMM)
 *   <li>Integer → Short
 *   <li>Boolean → String: フラグ変換
 *   <li>String → String: identity
 *   <li>同一型: identity
 * </ul>
 *
 * @param <RecordT> Record 型（Java record）
 * @param <EntityT> JPA Entity 型
 */
public final class ReflectiveRecordEntityMapper<RecordT extends Record, EntityT>
        implements RecordEntityMapper<RecordT, EntityT> {

    private final Class<RecordT> recordClass;

    private final FieldMapping[] fieldMappings;

    private final Constructor<RecordT> recordConstructor;

    private final Constructor<EntityT> entityConstructor;

    private ReflectiveRecordEntityMapper(
            Class<RecordT> recordClass,
            FieldMapping[] fieldMappings,
            Constructor<RecordT> recordConstructor,
            Constructor<EntityT> entityConstructor) {
        this.recordClass = recordClass;
        this.fieldMappings = fieldMappings;
        this.recordConstructor = recordConstructor;
        this.entityConstructor = entityConstructor;
    }

    /**
     * マッパーを構築する。
     *
     * @param <R> Record 型
     * @param <E> Entity 型
     * @param recordClass Record クラス
     * @param entityClass Entity クラス（引数なしコンストラクタ必須）
     * @return マッパーインスタンス
     */
    public static <R extends Record, E> ReflectiveRecordEntityMapper<R, E> create(
            Class<R> recordClass, Class<E> entityClass) {
        Objects.requireNonNull(recordClass, "recordClass");
        Objects.requireNonNull(entityClass, "entityClass");

        Map<String, Field> entityFields = new LinkedHashMap<>();
        for (Class<?> cls = entityClass;
                cls != null && cls != Object.class;
                cls = cls.getSuperclass()) {
            for (Field f : cls.getDeclaredFields()) {
                if (java.lang.reflect.Modifier.isStatic(f.getModifiers())) {
                    continue;
                }
                f.setAccessible(true);
                entityFields.putIfAbsent(f.getName(), f);
            }
        }

        RecordComponent[] components = recordClass.getRecordComponents();
        FieldMapping[] mappings = new FieldMapping[components.length];
        for (int i = 0; i < components.length; i++) {
            RecordComponent comp = components[i];
            Method accessor = comp.getAccessor();
            accessor.setAccessible(true);
            Field entityField = entityFields.get(comp.getName());
            if (entityField != null) {
                Function<Object, Object> toRecord =
                    resolveToRecordConverter(comp, entityField.getType(), comp.getType());
                Function<Object, Object> toEntity =
                        resolveToEntityConverter(comp.getType(), entityField.getType(), comp);
                mappings[i] = new FieldMapping(accessor, entityField, toRecord, toEntity);
            }
        }

        Constructor<R> recordCtor;
        try {
            Class<?>[] paramTypes = new Class<?>[components.length];
            for (int i = 0; i < components.length; i++) {
                paramTypes[i] = components[i].getType();
            }
            recordCtor = recordClass.getDeclaredConstructor(paramTypes);
            recordCtor.setAccessible(true);
        } catch (NoSuchMethodException e) {
            throw new IllegalArgumentException(
                    "Record class must have a canonical constructor: " + recordClass.getName(), e);
        }

        Constructor<E> ctor;
        try {
            ctor = entityClass.getDeclaredConstructor();
            ctor.setAccessible(true);
        } catch (NoSuchMethodException e) {
            throw new IllegalArgumentException(
                    "Entity class must have a no-arg constructor: " + entityClass.getName(), e);
        }

        return new ReflectiveRecordEntityMapper<>(recordClass, mappings, recordCtor, ctor);
    }

    @Override
    public RecordT toRecord(EntityT entity) {
        Objects.requireNonNull(entity, "entity");
        Object[] values = new Object[fieldMappings.length];
        for (int i = 0; i < fieldMappings.length; i++) {
            FieldMapping mapping = fieldMappings[i];
            if (mapping == null) {
                values[i] = null;
                continue;
            }
            try {
                Object entityValue = mapping.entityField.get(entity);
                values[i] = mapping.toRecord.apply(entityValue);
            } catch (IllegalAccessException e) {
                throw new IllegalStateException(
                        "Failed to read entity field: " + mapping.entityField.getName(), e);
            }
        }
        try {
            return recordConstructor.newInstance(values);
        } catch (ReflectiveOperationException e) {
            throw new IllegalStateException(
                    "Failed to construct record: " + recordClass.getName(), e);
        }
    }

    @Override
    public EntityT toEntity(RecordT record) {
        Objects.requireNonNull(record, "record");
        try {
            EntityT entity = entityConstructor.newInstance();
            for (FieldMapping mapping : fieldMappings) {
                if (mapping == null) {
                    continue;
                }
                Object recordValue = mapping.recordAccessor.invoke(record);
                Object entityValue = mapping.toEntity.apply(recordValue);
                mapping.entityField.set(entity, entityValue);
            }
            return entity;
        } catch (ReflectiveOperationException e) {
            throw new IllegalStateException(
                    "Failed to convert record to entity: " + recordClass.getName(), e);
        }
    }

    @SuppressWarnings("java:S3776")
    private static Function<Object, Object> resolveToRecordConverter(
            RecordComponent component, Class<?> entityType, Class<?> recordType) {
        boolean preserveZero = component.isAnnotationPresent(PreserveZeroValue.class);
        if (entityType == recordType) {
            if (entityType == String.class) {
                return v -> FixedLengthFormatter.suppressSpace((String) v);
            }
            return Function.identity();
        }
        if (entityType == BigDecimal.class && recordType == Long.class) {
            return preserveZero
                    ? v -> JpaDbReadValues.longValue((BigDecimal) v)
                    : v -> JpaDbReadValues.longOrNullIfZero((BigDecimal) v);
        }
        if (entityType == BigDecimal.class && recordType == Integer.class) {
            return preserveZero
                    ? v -> JpaDbReadValues.integerValue((BigDecimal) v)
                    : v -> JpaDbReadValues.integerOrNullIfZero((BigDecimal) v);
        }
        if (entityType == BigDecimal.class && recordType == LocalDate.class) {
            return v -> JpaDbReadValues.localDate((BigDecimal) v);
        }
        if (entityType == BigDecimal.class && recordType == LocalTime.class) {
            return v -> JpaDbReadValues.localTime((BigDecimal) v);
        }
        if (entityType == BigDecimal.class && recordType == YearMonth.class) {
            return v -> BigDecimalYearMonthConverter.INSTANCE.convert((BigDecimal) v);
        }
        if ((entityType == Short.class || entityType == short.class)
                && recordType == Integer.class) {
            return v -> v == null ? null : ((Short) v).intValue();
        }
        if (entityType == String.class && recordType == Boolean.class) {
            return v -> StringBooleanFlagConverter.INSTANCE.convert((String) v);
        }
        return Function.identity();
    }

    private static Function<Object, Object> resolveToEntityConverter(
            Class<?> recordType, Class<?> entityType, RecordComponent component) {
        boolean preserveZero = component.isAnnotationPresent(PreserveZeroValue.class);
        if (recordType == entityType) {
            return Function.identity();
        }
        if (recordType == Long.class && entityType == BigDecimal.class) {
            return v -> JpaDbWriteValues.bigDecimal((Long) v);
        }
        if (recordType == Integer.class && entityType == BigDecimal.class) {
            return v -> JpaDbWriteValues.bigDecimal((Integer) v);
        }
        if (recordType == LocalDate.class && entityType == BigDecimal.class) {
            return v -> {
                if (v == null && preserveZero) {
                    return BigDecimal.ZERO;
                }
                return JpaDbWriteValues.ymd((LocalDate) v);
            };
        }
        if (recordType == LocalTime.class && entityType == BigDecimal.class) {
            return v -> JpaDbWriteValues.localTime((LocalTime) v);
        }
        if (recordType == YearMonth.class && entityType == BigDecimal.class) {
            return v -> YearMonthBigDecimalConverter.INSTANCE.convert((YearMonth) v);
        }
        if (recordType == Integer.class
                && (entityType == Short.class || entityType == short.class)) {
            return v -> JpaDbWriteValues.smallInt((Integer) v);
        }
        if (recordType == Boolean.class && entityType == String.class) {
            return v -> BooleanFlagStringConverter.INSTANCE.convert((Boolean) v);
        }
        return Function.identity();
    }

    private record FieldMapping(
            Method recordAccessor,
            Field entityField,
            Function<Object, Object> toRecord,
            Function<Object, Object> toEntity) { }
}
