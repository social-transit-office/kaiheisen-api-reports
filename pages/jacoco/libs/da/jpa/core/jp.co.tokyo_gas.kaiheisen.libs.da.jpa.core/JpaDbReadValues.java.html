package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalTime;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalIntegerConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalLocalTimeConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalLongConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalNullIfZeroIntegerConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BigDecimalNullIfZeroLongConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.ShortNullIfZeroIntegerConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.YmdBigDecimalLocalDateConverter;

/** JPA 向けの DB→API 値変換 sugar。 */
public final class JpaDbReadValues {

    private JpaDbReadValues() { }

    /**
     * DB 数値を 0 なら null として Long へ変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static Long longOrNullIfZero(BigDecimal value) {
        return BigDecimalNullIfZeroLongConverter.INSTANCE.convert(value);
    }

    /**
     * DB 数値を Long へそのまま変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static Long longValue(BigDecimal value) {
        return BigDecimalLongConverter.INSTANCE.convert(value);
    }

    /**
     * DB 数値を 0 なら null として Integer へ変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static Integer integerOrNullIfZero(BigDecimal value) {
        return BigDecimalNullIfZeroIntegerConverter.INSTANCE.convert(value);
    }

    /**
     * DB 数値を Integer へそのまま変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static Integer integerValue(BigDecimal value) {
        return BigDecimalIntegerConverter.INSTANCE.convert(value);
    }

    /**
     * DB の smallint を 0 なら null として Integer へ変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static Integer integerOrNullIfZero(Short value) {
        return ShortNullIfZeroIntegerConverter.INSTANCE.convert(value);
    }

    /**
     * YYYYMMDD 数値を LocalDate へ変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static LocalDate localDate(BigDecimal value) {
        return YmdBigDecimalLocalDateConverter.INSTANCE.convert(value);
    }

    /**
     * 数値時刻を LocalTime へ変換する。
     *
     * @param value DB 値
     * @return API 値
     */
    public static LocalTime localTime(BigDecimal value) {
        return BigDecimalLocalTimeConverter.INSTANCE.convert(value);
    }
}
