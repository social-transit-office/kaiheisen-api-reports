package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.Query;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;
import java.util.Objects;

/** native query 向けの WHERE 句と bind 処理を保持する immutable helper。 */
public final class PreparedWhere {

    private final String clause;

    private final List<Object> bindValues;

    private final Binder binder;

    private PreparedWhere(String clause, List<Object> bindValues, Binder binder) {
        this.clause = Objects.requireNonNull(clause, "clause");
        this.bindValues =
                Collections.unmodifiableList(
                        new ArrayList<>(Objects.requireNonNull(bindValues, "bindValues")));
        this.binder = Objects.requireNonNull(binder, "binder");
    }

    /**
     * WHERE 句 helper を生成する。
     *
     * @param clause WHERE 句
     * @param binder bind 処理
     * @return 生成結果
     */
    public static PreparedWhere of(String clause, Binder binder) {
        return new PreparedWhere(clause, List.of(), binder);
    }

    /**
     * WHERE 句 helper を生成する。
     *
     * @param clause WHERE 句
     * @param binder bind 処理
     * @param bindValues bind 値
     * @return 生成結果
     */
    public static PreparedWhere of(String clause, Binder binder, Object... bindValues) {
        Object[] clonedBindValues = bindValues == null ? new Object[0] : bindValues.clone();
        return new PreparedWhere(clause, Arrays.asList(clonedBindValues), binder);
    }

    /**
     * WHERE 句を返す。
     *
     * @return WHERE 句
     */
    public String clause() {
        return clause;
    }

    /**
     * WHERE 句の bind 値を返す。
     *
     * @return bind 値
     */
    public List<Object> bindValues() {
        return bindValues;
    }

    /**
     * WHERE 句の bind を行い、次の index を返す。
     *
     * @param query 対象 query
     * @param startIndex 開始 index
     * @return 次の index
     */
    public int bind(Query query, int startIndex) {
        Objects.requireNonNull(query, "query");
        return binder.bind(query, startIndex);
    }

    /** WHERE 句の bind を行う関数型インターフェース。 */
    @FunctionalInterface
    public interface Binder {

        /**
         * WHERE 句の bind を行い、次の index を返す。
         *
         * @param query 対象 query
         * @param startIndex 開始 index
         * @return 次の index
         */
        int bind(Query query, int startIndex);
    }
}
