package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalTime;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.IntegerBigDecimalConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.IntegerShortConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LocalDateYmdBigDecimalConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LocalDateYmdIntegerConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LocalDateYmdLongConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LocalDateYmdStringConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LocalTimeBigDecimalConverter;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.LongBigDecimalConverter;

/** JPA 向けの API→DB 値変換 sugar。 */
public final class JpaDbWriteValues {

    private JpaDbWriteValues() { }

    /**
     * Long を DB 数値へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static BigDecimal bigDecimal(Long value) {
        return LongBigDecimalConverter.INSTANCE.convert(value);
    }

    /**
     * Integer を DB 数値へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static BigDecimal bigDecimal(Integer value) {
        return IntegerBigDecimalConverter.INSTANCE.convert(value);
    }

    /**
     * Integer を DB の smallint 相当へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static Short smallInt(Integer value) {
        return IntegerShortConverter.INSTANCE.convert(value);
    }

    /**
     * LocalDate を YYYYMMDD 数値へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static BigDecimal ymd(LocalDate value) {
        return LocalDateYmdBigDecimalConverter.INSTANCE.convert(value);
    }

    /**
     * LocalDate を YYYYMMDD 整数値へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static Integer ymdInt(LocalDate value) {
        return LocalDateYmdIntegerConverter.INSTANCE.convert(value);
    }

    /**
     * LocalDate を YYYYMMDD 長整数値へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static Long ymdLong(LocalDate value) {
        return LocalDateYmdLongConverter.INSTANCE.convert(value);
    }

    /**
     * LocalDate を YYYYMMDD 文字列へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static String ymdString(LocalDate value) {
        return LocalDateYmdStringConverter.INSTANCE.convert(value);
    }

    /**
     * LocalTime を数値時刻へ変換する。
     *
     * @param value API 値
     * @return DB 値
     */
    public static BigDecimal localTime(LocalTime value) {
        return LocalTimeBigDecimalConverter.INSTANCE.convert(value);
    }
}
