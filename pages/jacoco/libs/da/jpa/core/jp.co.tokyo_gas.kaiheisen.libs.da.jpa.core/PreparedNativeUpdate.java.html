package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.Query;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.stream.Collectors;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;

/** native UPDATE の SET 句と SET バインド値を保持する immutable helper。 */
public final class PreparedNativeUpdate {

    private static final String REC_UPD_JNJ_NO = "REC_UPD_JNJ_NO";

    private static final PreparedNativeUpdate EMPTY =
            new PreparedNativeUpdate(List.of(), List.of());

    private final List<DataAccessColumn> columns;

    private final List<Object> bindValues;

    private PreparedNativeUpdate(
            List<? extends DataAccessColumn> columns, List<Object> bindValues) {
        this.columns = List.copyOf(columns);
        this.bindValues = List.copyOf(bindValues);
        if (this.columns.size() != this.bindValues.size()) {
            throw new IllegalArgumentException("columns size must match bindValues size");
        }
    }

    /**
     * SET 句情報を生成する。
     *
     * @param columns SET 対象列
     * @param bindValues SET バインド値
     * @return 生成結果
     */
    public static PreparedNativeUpdate of(
            List<? extends DataAccessColumn> columns, List<Object> bindValues) {
        Objects.requireNonNull(columns, "columns");
        Objects.requireNonNull(bindValues, "bindValues");
        return new PreparedNativeUpdate(columns, bindValues);
    }

    /**
     * 空の UPDATE 情報を返す。
     *
     * @return 空の UPDATE 情報
     */
    public static PreparedNativeUpdate empty() {
        return EMPTY;
    }

    /**
     * SET 対象列を返す。
     *
     * @return SET 対象列
     */
    public List<DataAccessColumn> columns() {
        return columns;
    }

    /**
     * SET バインド値を返す。
     *
     * @return SET バインド値
     */
    public List<Object> bindValues() {
        return bindValues;
    }

    /**
     * 実際に SET 句へ bind する値を返す。
     *
     * @return SET 句の実効 bind 値
     */
    public List<Object> effectiveSetBindValues() {
        if (!includesRecUpdJnjNo()) {
            return bindValues;
        }
        List<Object> effectiveBindValues = new ArrayList<>(bindValues.size() - 1);
        for (int i = 0; i < columns.size(); i++) {
            if (REC_UPD_JNJ_NO.equals(columns.get(i).columnName())) {
                continue;
            }
            effectiveBindValues.add(bindValues.get(i));
        }
        return effectiveBindValues;
    }

    /**
     * SET 対象列が空かどうかを返す。
     *
     * @return 空の場合は true
     */
    public boolean isEmpty() {
        return columns.isEmpty();
    }

    /**
     * UPDATE 文の SET 句を返す。
     *
     * @return SET 句
     */
    public String setClause() {
        return columns.stream()
                .map(column -> column.columnName() + " = ?")
                .collect(Collectors.joining(", "));
    }

    /**
     * REC_UPD_JNJ_NO を自己加算する SET 句を返す。
     *
     * @return SET 句
     */
    public String setClauseIncrementingRecUpdJnjNo() {
        return columns.stream()
                .map(
                        column ->
                                REC_UPD_JNJ_NO.equals(column.columnName())
                                        ? REC_UPD_JNJ_NO + " = " + REC_UPD_JNJ_NO + " + 1"
                                        : column.columnName() + " = ?")
                .collect(Collectors.joining(", "));
    }

    /**
     * WHERE 句付き UPDATE SQL を生成する。
     *
     * @param tableName 物理テーブル名
     * @param whereClause WHERE 句
     * @return native UPDATE SQL
     */
    public String sql(String tableName, String whereClause) {
        Objects.requireNonNull(tableName, "tableName");
        Objects.requireNonNull(whereClause, "whereClause");
        return "UPDATE " + tableName + " SET " + setClause() + " WHERE " + whereClause;
    }

    /**
     * REC_UPD_JNJ_NO を自己加算する WHERE 句付き UPDATE SQL を生成する。
     *
     * @param tableName 物理テーブル名
     * @param whereClause WHERE 句
     * @return native UPDATE SQL
     */
    public String sqlIncrementingRecUpdJnjNo(String tableName, String whereClause) {
        Objects.requireNonNull(tableName, "tableName");
        Objects.requireNonNull(whereClause, "whereClause");
        String setClause = setClauseIncrementingRecUpdJnjNo();
        return "UPDATE " + tableName + " SET " + setClause + " WHERE " + whereClause;
    }

    /**
     * SET 句の値をクエリへ bind する。
     *
     * @param query 対象クエリ
     * @param startIndex 開始位置
     * @return 次の bind 位置
     */
    public int bindSetParameters(Query query, int startIndex) {
        Objects.requireNonNull(query, "query");
        int nextIndex = startIndex;
        for (Object bindValue : bindValues) {
            query.setParameter(nextIndex++, bindValue);
        }
        return nextIndex;
    }

    /**
     * REC_UPD_JNJ_NO を除く SET 句の値をクエリへ bind する。
     *
     * @param query 対象クエリ
     * @param startIndex 開始位置
     * @return 次の bind 位置
     */
    public int bindSetParametersSkippingRecUpdJnjNo(Query query, int startIndex) {
        Objects.requireNonNull(query, "query");
        int nextIndex = startIndex;
        for (int i = 0; i < columns.size(); i++) {
            if (REC_UPD_JNJ_NO.equals(columns.get(i).columnName())) {
                continue;
            }
            query.setParameter(nextIndex++, bindValues.get(i));
        }
        return nextIndex;
    }

    /**
     * REC_UPD_JNJ_NO が含まれるかどうかを返す。
     *
     * @return 含まれる場合 true
     */
    public boolean includesRecUpdJnjNo() {
        return columns.stream().anyMatch(column -> REC_UPD_JNJ_NO.equals(column.columnName()));
    }
}
