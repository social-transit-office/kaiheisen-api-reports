package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.List;

/** 業務テーブルを使わずに共通の問い合わせ処理を初期化する。 */
public final class JpaStartupQueryWarmup {

    /** 業務DAと区別する起動準備ステップ識別子。 */
    public static final String FUNCTION_ID = "STARTUP-ORM";

    private JpaStartupQueryWarmup() {
    }

    /**
     * 実際の問い合わせ基盤で定数1を取得する。
     *
     * <p>Blazeの公開APIで式を組み立て、JPAの定数問い合わせでHibernateとDB接続を準備する。
     * 呼出し元のトランザクションに参加し、業務データの参照や更新は行わない。
     * BlazeのVALUES問い合わせは組立てまでとし、拡張SQLの実行経路へは進まない。</p>
     *
     * @param entityManager アプリケーションが使用するEntityManager
     * @param criteriaBuilderFactory アプリケーションが使用するBlazeのファクトリ
     * @return 接続確認値1
     */
    public static int execute(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        criteriaBuilderFactory.create(entityManager, Integer.class)
                .fromValues(Integer.class, "warmupValue", List.of(1))
                .select("warmupValue")
                .getQueryString();
        return entityManager.createQuery("select 1", Integer.class).getSingleResult();
    }
}
