package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import jakarta.persistence.Query;
import java.util.Objects;
import java.util.function.Function;

/** native query の bind を簡潔にする helper。 */
public final class JpaBind {

    private JpaBind() { }

    /**
     * 値を bind して次の index を返す。
     *
     * @param query 対象 query
     * @param index bind index
     * @param value bind 値
     * @return 次の index
     */
    public static int set(Query query, int index, Object value) {
        Objects.requireNonNull(query, "query");
        query.setParameter(index, value);
        return index + 1;
    }

    /**
     * 値変換を行ってから bind し、次の index を返す。
     *
     * @param query 対象 query
     * @param index bind index
     * @param source 変換前の値
     * @param converter 値変換
     * @param <S> 変換前の型
     * @param <V> 変換後の型
     * @return 次の index
     */
    public static <S, V> int set(
            Query query, int index, S source, Function<? super S, ? extends V> converter) {
        Objects.requireNonNull(converter, "converter");
        return set(query, index, converter.apply(source));
    }
}
