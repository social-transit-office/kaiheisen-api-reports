package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core;

import java.math.BigDecimal;
import java.sql.Timestamp;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditColumnAccess;

/**
 * JPA エンティティを {@link AuditColumnAccess} として扱う binder。
 *
 * @param <EntityT> エンティティ型
 */
public final class JpaAuditColumnAccessBinder<EntityT> {

    private final Map<String, EntityColumnMetadataResolver.EntityColumnMetadata>
            metadataByColumnName;

    /**
     * binder を生成する。
     *
     * @param entityClass エンティティ型
     */
    public JpaAuditColumnAccessBinder(Class<EntityT> entityClass) {
        this.metadataByColumnName =
                EntityColumnMetadataResolver.resolve(
                        Objects.requireNonNull(entityClass, "entityClass"));
    }

    /**
     * エンティティを監査列アクセスへ bind する。
     *
     * @param entity エンティティ
     * @return 監査列アクセス
     */
    public AuditColumnAccess bind(EntityT entity) {
        Objects.requireNonNull(entity, "entity");
        return new AuditColumnAccess() {
            @Override
            public boolean hasColumn(String columnName) {
                return metadataByColumnName.containsKey(columnName);
            }

            @Override
            public void setValue(String columnName, Object value) {
                EntityColumnMetadataResolver.EntityColumnMetadata metadata =
                        metadataByColumnName.get(columnName);
                if (metadata == null) {
                    throw new IllegalArgumentException("Unknown column: " + columnName);
                }
                metadata.setValue(entity, convertValue(value, metadata.javaType()));
            }
        };
    }

    private static Object convertValue(Object value, Class<?> targetType) {
        Objects.requireNonNull(targetType, "targetType");
        if (value == null || targetType.isInstance(value)) {
            return value;
        }
        return switch (value) {
            case LocalDate localDate -> convertLocalDate(localDate, targetType);
            case LocalDateTime localDateTime -> convertLocalDateTime(localDateTime, targetType);
            case Integer integer -> convertInteger(integer, targetType);
            case Long longValue -> convertLong(longValue, targetType);
            case String text -> convertString(text, targetType);
            default -> throw unsupported(value, targetType);
        };
    }

    private static Object convertLocalDate(LocalDate value, Class<?> targetType) {
        if (targetType == BigDecimal.class) {
            return JpaDbWriteValues.ymd(value);
        }
        if (targetType == Integer.class) {
            return JpaDbWriteValues.ymdInt(value);
        }
        if (targetType == Long.class) {
            return JpaDbWriteValues.ymdLong(value);
        }
        if (targetType == String.class) {
            return JpaDbWriteValues.ymdString(value);
        }
        if (targetType == LocalDate.class) {
            return value;
        }
        throw unsupported(value, targetType);
    }

    private static Object convertLocalDateTime(LocalDateTime value, Class<?> targetType) {
        if (targetType == LocalDateTime.class) {
            return value;
        }
        if (targetType == Timestamp.class) {
            return Timestamp.valueOf(value);
        }
        throw unsupported(value, targetType);
    }

    private static Object convertInteger(Integer value, Class<?> targetType) {
        if (targetType == BigDecimal.class) {
            return JpaDbWriteValues.bigDecimal(value);
        }
        if (targetType == Short.class) {
            return JpaDbWriteValues.smallInt(value);
        }
        if (targetType == Integer.class) {
            return value;
        }
        if (targetType == Long.class) {
            return value.longValue();
        }
        throw unsupported(value, targetType);
    }

    private static Object convertLong(Long value, Class<?> targetType) {
        if (targetType == BigDecimal.class) {
            return JpaDbWriteValues.bigDecimal(value);
        }
        if (targetType == Long.class) {
            return value;
        }
        throw unsupported(value, targetType);
    }

    private static Object convertString(String value, Class<?> targetType) {
        if (targetType == String.class) {
            return value;
        }
        throw unsupported(value, targetType);
    }

    private static IllegalStateException unsupported(Object value, Class<?> targetType) {
        return new IllegalStateException(
                "Unsupported audit value conversion: valueType="
                        + value.getClass().getName()
                        + ", targetType="
                        + targetType.getName());
    }
}
