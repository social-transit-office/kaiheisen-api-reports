package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.dialect;

import org.hibernate.dialect.DatabaseVersion;
import org.hibernate.dialect.sql.ast.DB2zSqlAstTranslator;
import org.hibernate.engine.spi.SessionFactoryImplementor;
import org.hibernate.sql.ast.tree.Statement;
import org.hibernate.sql.exec.spi.JdbcOperation;

/** Db2 for z/OS の OFFSET/FETCH 件数をリテラルとして出力する SQL AST translator。 */
public class LegacyDb2zSqlAstTranslator<T extends JdbcOperation>
        extends DB2zSqlAstTranslator<T> {

    /**
     * コンストラクタ。
     *
     * @param sessionFactory SessionFactory
     * @param statement SQL AST statement
     * @param version Db2 version
     */
    public LegacyDb2zSqlAstTranslator(
            SessionFactoryImplementor sessionFactory,
            Statement statement,
            DatabaseVersion version) {
        super(sessionFactory, statement, version);
    }

    @Override
    protected boolean supportsParameterOffsetFetchExpression() {
        return false;
    }
}
