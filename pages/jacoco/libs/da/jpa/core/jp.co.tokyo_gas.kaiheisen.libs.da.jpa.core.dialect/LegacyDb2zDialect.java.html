package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.dialect;

import org.hibernate.dialect.DB2zDialect;
import org.hibernate.dialect.pagination.LimitHandler;
import org.hibernate.dialect.pagination.OffsetFetchLimitHandler;
import org.hibernate.engine.spi.SessionFactoryImplementor;
import org.hibernate.sql.ast.SqlAstTranslator;
import org.hibernate.sql.ast.SqlAstTranslatorFactory;
import org.hibernate.sql.ast.spi.StandardSqlAstTranslatorFactory;
import org.hibernate.sql.ast.tree.Statement;
import org.hibernate.sql.exec.spi.JdbcOperation;

/**
 * Db2 for z/OS の旧 APPLCOMPAT 向け Dialect。
 *
 * <p>OFFSET/FETCH の件数だけをリテラルとして出力し、その他の SQL 生成は {@link DB2zDialect}
 * に委譲する。
 */
public class LegacyDb2zDialect extends DB2zDialect {

    private static final LimitHandler LIMIT_HANDLER = new OffsetFetchLimitHandler(false);

    @Override
    public LimitHandler getLimitHandler() {
        return LIMIT_HANDLER;
    }

    @Override
    public SqlAstTranslatorFactory getSqlAstTranslatorFactory() {
        return new StandardSqlAstTranslatorFactory() {
            @Override
            protected <T extends JdbcOperation> SqlAstTranslator<T> buildTranslator(
                    SessionFactoryImplementor sessionFactory, Statement statement) {
                return new LegacyDb2zSqlAstTranslator<>(
                        sessionFactory, statement, LegacyDb2zDialect.this.getVersion());
            }
        };
    }
}
