package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.sysibm.sysdummy1;

import jakarta.persistence.EntityManager;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1Record;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.SqlCapture;

/** DBウォームアップ用データアクセスのJPA実装。 */
public final class Sysdummy1DataAccessImpl extends DataAccessBase
        implements Sysdummy1DataAccess {

    static final String SQL = "SELECT 1 FROM SYSIBM.SYSDUMMY1";

    private final EntityManager entityManager;

    /**
     * コンストラクタ。
     *
     * @param entityManager EntityManager
     */
    public Sysdummy1DataAccessImpl(final EntityManager entityManager) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
    }

    @Override
    public Sysdummy1Record fetchWarmupValue() {
        SqlCapture.set(SQL);
        List<?> results = entityManager.createNativeQuery(SQL).getResultList();
        if (results.size() != 1) {
            throw new IllegalStateException(
                    "DB warmup query must return exactly one row: actual=" + results.size());
        }
        Object result = results.getFirst();
        if (!(result instanceof Number number)) {
            throw new IllegalStateException("DB warmup query returned a non-numeric value");
        }
        int value = number.intValue();
        if (value != 1) {
            throw new IllegalStateException(
                    "DB warmup query must return the value 1: actual=" + value);
        }
        return new Sysdummy1Record(value);
    }
}
