package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.atlkinf;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.atlkinf.AtlkinfDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.atlkinf.AtlkinfRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** オートロック情報テーブル（ATLKINF）データアクセスの JPA 実装。 */
public class AtlkinfDataAccessImpl extends DataAccessBase implements AtlkinfDataAccess {

    private static final int BLD_NO_LENGTH = 11;

    private static final ReflectiveRecordEntityMapper<AtlkinfRecord, AtlkinfEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(AtlkinfRecord.class, AtlkinfEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public AtlkinfDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<AtlkinfRecord> findByBldNo(String bldNo) {
        String safeBldNo = WhereParamGuard.fixedLengthNotNull(bldNo, "bldNo", BLD_NO_LENGTH);

        var builder = criteriaBuilderFactory.create(entityManager, AtlkinfEntity.class);
        JpaWhere.eq(builder, "bldNo", safeBldNo);
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(MAPPER::toRecord)
                .findFirst();
    }
}
