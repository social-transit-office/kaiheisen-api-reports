public class TaksmtrDataAccessImpl extends DataAccessBase implements TaksmtrDataAccess {

    private static final String TABLE_NAME = "TAKSMTR";

    private static final int TAKSKY_NO_LENGTH = 11;

    private static final String WHERE_BY_PK = "TAKSKY_NO = ? AND HKKM_ITI_NO = ? AND GMT_SET_NO = ?";

    private static final ReflectiveRecordEntityMapper<TaksmtrRecord, TaksmtrEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(TaksmtrRecord.class, TaksmtrEntity.class);

    private static final ReflectiveRecordEntityMapper<TaksmtrWriteRecord, TaksmtrEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(TaksmtrWriteRecord.class, TaksmtrEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<TaksmtrEntity> auditColumnAccessBinder;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaTargetColumnUpdateSupport<TaksmtrWriteRecord, TaksmtrEntity, TaksmtrColumn>
            updateSupport;

    /**
     * JPA ベースの TAKSMTR データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public TaksmtrDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new NcisCisAuditColumnAutoSetter());
    }

    TaksmtrDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(TaksmtrEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        TaksmtrWriteDefinition.INSTANCE,
                        TaksmtrEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNo(Long gmtSetNo) {
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        var builder = criteriaBuilderFactory.create(entityManager, TaksmtrEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", safeGmtSetNo);
        builder.orderByDesc("taksEndYmd");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCd(Long gmtSetNo, String taksKyJtCd) {
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        String safeTaksKyJtCd = safeFixedCode(taksKyJtCd, "taksKyJtCd", 1);
        var builder = criteriaBuilderFactory.create(entityManager, TaksmtrEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", safeGmtSetNo);
        JpaWhere.eq(builder, "taksKyJtCd", safeTaksKyJtCd);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCdList(
            Long gmtSetNo, List<String> taksKyJtCdList) {
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        List<String> safeTaksKyJtCdList = safeFixedCodeList(taksKyJtCdList, "taksKyJtCdList", 1);
        if (safeTaksKyJtCdList.isEmpty()) {
            return List.of();
        }
        var builder = criteriaBuilderFactory.create(entityManager, TaksmtrEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", safeGmtSetNo);
        JpaWhere.in(builder, "taksKyJtCd", safeTaksKyJtCdList);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCdListAndTaksCisHntCd(
            Long gmtSetNo,
            List<String> taksKyJtCdList,
            String taksCisHntCd) {
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        List<String> safeTaksKyJtCdList = safeFixedCodeList(taksKyJtCdList, "taksKyJtCdList", 1);
        String safeTaksCisHntCd = safeFixedCode(taksCisHntCd, "taksCisHntCd", 1);
        if (safeTaksKyJtCdList.isEmpty()) {
            return List.of();
        }
        return entityManager.createQuery(
                        """
                        SELECT taksmtr
                          FROM TaksmtrEntity taksmtr
                         WHERE taksmtr.gmtSetNo = :gmtSetNo
                           AND taksmtr.taksKyJtCd IN :taksKyJtCdList
                           AND taksmtr.taksCisHntCd = :taksCisHntCd
                        """,
                        TaksmtrEntity.class)
                .setParameter("gmtSetNo", safeGmtSetNo)
                .setParameter("taksKyJtCdList", safeTaksKyJtCdList)
                .setParameter("taksCisHntCd", safeTaksCisHntCd)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .toList();
    }

        return entityManager.createQuery(

    /** {@inheritDoc} */
    @Override
    public int updateByTakskyNoAndHkkmItiNoAndGmtSetNoAndRecUpdJnjNo(
            TaksmtrWriteRecord record,
            Set<TaksmtrColumn> targetColumns,
            TaksmtrUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhereByPk(updateKey.takskyNo(), updateKey.hkkmItiNo(), updateKey.gmtSetNo()),
                updateKey.recUpdJnjNo(),
                Identifiers.UPDATE_BY_TAKSKY_NO_AND_HKKM_ITI_NO_AND_GMT_SET_NO_AND_REC_UPD_JNJ_NO);
    }

    /** {@inheritDoc} */
    @Override
    public int updateByTakskyNoAndHkkmItiNoAndGmtSetNo(
            TaksmtrWriteRecord record,
            Set<TaksmtrColumn> targetColumns,
            TaksmtrUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Optional<Integer> currentRecUpdJnjNo =
            findCurrentRecUpdJnjNo(
                updateKey.takskyNo(),
                updateKey.hkkmItiNo(),
                updateKey.gmtSetNo());
        if (currentRecUpdJnjNo.isEmpty()) {
            return 0;
        }

        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo.orElseThrow()),
                preparedWhereByPk(updateKey.takskyNo(), updateKey.hkkmItiNo(), updateKey.gmtSetNo()));
    }

    private PreparedWhere preparedWhereByPk(String takskyNo, Integer hkkmItiNo, Long gmtSetNo) {
        String safeTakskyNo = safeTakskyNo(takskyNo);
        BigDecimal safeHkkmItiNo = safeInteger(hkkmItiNo, "hkkmItiNo");
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        return PreparedWhere.of(
                WHERE_BY_PK,
                (query, startIndex) ->
                        bindWhereByPk(
                                query,
                                startIndex,
                                safeTakskyNo,
                                safeHkkmItiNo,
                                safeGmtSetNo),
                safeTakskyNo,
                safeHkkmItiNo,
                safeGmtSetNo);
    }

    private static int bindWhereByPk(
            Query query,
            int startIndex,
            String takskyNo,
            BigDecimal hkkmItiNo,
            BigDecimal gmtSetNo) {
        int nextIndex = JpaBind.set(query, startIndex, takskyNo);
        nextIndex = JpaBind.set(query, nextIndex, hkkmItiNo);
        return JpaBind.set(query, nextIndex, gmtSetNo);
    }

    private Optional<Integer> findCurrentRecUpdJnjNo(
            String takskyNo,
            Integer hkkmItiNo,
            Long gmtSetNo) {
        String safeTakskyNo = safeTakskyNo(takskyNo);
        BigDecimal safeHkkmItiNo = safeInteger(hkkmItiNo, "hkkmItiNo");
        BigDecimal safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");

        TaksmtrId id = new TaksmtrId(safeTakskyNo, safeHkkmItiNo, safeGmtSetNo);
        TaksmtrEntity entity = entityManager.find(TaksmtrEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(toRecUpdJnjNo(entity.getRecUpdJnjNo()));
    }

    private static Integer toRecUpdJnjNo(Object value) {
        if (value == null) {
            return 0;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.parseInt(value.toString());
    }

    private static String safeTakskyNo(String value) {
        return safeFixedCode(value, "takskyNo", TAKSKY_NO_LENGTH);
    }

    private static String safeFixedCode(String value, String paramName, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }

    private static List<String> safeFixedCodeList(List<String> values, String paramName, int length) {
        Objects.requireNonNull(values, paramName);
        if (values.isEmpty()) {
            return List.of();
        }
        return values.stream()
                .map(value -> safeFixedCode(value, paramName, length))
                .toList();
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
