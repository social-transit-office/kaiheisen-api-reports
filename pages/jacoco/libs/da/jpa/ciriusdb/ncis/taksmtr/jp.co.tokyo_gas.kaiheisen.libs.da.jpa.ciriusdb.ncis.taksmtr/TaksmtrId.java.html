package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.taksmtr;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * TAKSMTR の複合主キーを表現する ID クラス。
 */
public class TaksmtrId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String takskyNo;

    private BigDecimal hkkmItiNo;

    private BigDecimal gmtSetNo;

    /** デフォルトコンストラクタ。 */
    public TaksmtrId() { }

    /** 全フィールド指定コンストラクタ。 */
    public TaksmtrId(String takskyNo, BigDecimal hkkmItiNo, BigDecimal gmtSetNo) {
        this.takskyNo = takskyNo;
        this.hkkmItiNo = hkkmItiNo;
        this.gmtSetNo = gmtSetNo;
    }

    /** takskyNo を取得する。 */
    public String getTakskyNo() {
        return takskyNo;
    }

    /** takskyNo を設定する。 */
    public void setTakskyNo(String takskyNo) {
        this.takskyNo = takskyNo;
    }

    /** hkkmItiNo を取得する。 */
    public BigDecimal getHkkmItiNo() {
        return hkkmItiNo;
    }

    /** hkkmItiNo を設定する。 */
    public void setHkkmItiNo(BigDecimal hkkmItiNo) {
        this.hkkmItiNo = hkkmItiNo;
    }

    /** gmtSetNo を取得する。 */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /** gmtSetNo を設定する。 */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /** 主キー項目の値で同値判定を行う。 */
    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof TaksmtrId that)) {
            return false;
        }
        return compareString(takskyNo, that.takskyNo)
                && compareBigDecimal(hkkmItiNo, that.hkkmItiNo)
                && compareBigDecimal(gmtSetNo, that.gmtSetNo);
    }

    private static boolean compareString(String left, String right) {
        if (left == null) {
            return right == null;
        }
        return left.equals(right);
    }

    private static boolean compareBigDecimal(BigDecimal left, BigDecimal right) {
        if (left == null) {
            return right == null;
        }
        return right != null && left.compareTo(right) == 0;
    }

    /** 主キー項目からハッシュ値を計算する。 */
    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (takskyNo == null ? 0 : takskyNo.hashCode());
        result = 31 * result + (hkkmItiNo == null ? 0 : hkkmItiNo.stripTrailingZeros().hashCode());
        result = 31 * result + (gmtSetNo == null ? 0 : gmtSetNo.stripTrailingZeros().hashCode());
        return result;
    }
}
