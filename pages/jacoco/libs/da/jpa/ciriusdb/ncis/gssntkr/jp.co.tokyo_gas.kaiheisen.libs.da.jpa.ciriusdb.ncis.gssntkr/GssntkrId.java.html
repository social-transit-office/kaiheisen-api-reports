package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.gssntkr;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** GSSNTKR の複合主キー。 */
public class GssntkrId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gmtSetNo;

    private String cstBrSign;

    /** デフォルトコンストラクタ。 */
    public GssntkrId() { }

    /** 全フィールド指定コンストラクタ。 */
    public GssntkrId(BigDecimal gmtSetNo, String cstBrSign) {
        this.gmtSetNo = gmtSetNo;
        this.cstBrSign = cstBrSign;
    }

    /** gmtSetNoを取得する。 */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /** gmtSetNoを設定する。 */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /** cstBrSignを取得する。 */
    public String getCstBrSign() {
        return cstBrSign;
    }

    /** cstBrSignを設定する。 */
    public void setCstBrSign(String cstBrSign) {
        this.cstBrSign = cstBrSign;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof GssntkrId that)) {
            return false;
        }
        return Objects.equals(gmtSetNo, that.gmtSetNo) && Objects.equals(cstBrSign, that.cstBrSign);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gmtSetNo, cstBrSign);
    }
}
