package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.gssntkr;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.gssntkr.GssntkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.gssntkr.GssntkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GSSNTKR データアクセスの JPA 実装。 */
public class GssntkrDataAccessImpl extends DataAccessBase implements GssntkrDataAccess {

    private static final ReflectiveRecordEntityMapper<GssntkrRecord, GssntkrEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(GssntkrRecord.class, GssntkrEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public GssntkrDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<GssntkrRecord> findAllByGmtSetNo(Long gmtSetNo) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        var builder = criteriaBuilderFactory.create(entityManager, GssntkrEntity.class);
        builder.where("gmtSetNo")
                .eq(JpaDbWriteValues.bigDecimal(gmtSetNo))
                .orderByDesc("recRegYmd")
                .orderByDesc("cstBrSign");
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public Optional<GssntkrRecord> findByPK(Long gmtSetNo, String cstBrSign) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(cstBrSign, "cstBrSign");

        GssntkrId id = new GssntkrId(JpaDbWriteValues.bigDecimal(gmtSetNo), cstBrSign);
        GssntkrEntity entity = entityManager.find(GssntkrEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }
}
