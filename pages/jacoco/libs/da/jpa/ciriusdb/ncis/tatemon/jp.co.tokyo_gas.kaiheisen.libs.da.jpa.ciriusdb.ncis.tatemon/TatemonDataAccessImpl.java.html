package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.tatemon;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.TatemonDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.TatemonRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.write.TatemonColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.write.TatemonUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.write.TatemonWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.tatemon.write.TatemonWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** 建物（TATEMON）データアクセスの JPA 実装。 */
public class TatemonDataAccessImpl extends DataAccessBase implements TatemonDataAccess {

    private static final String TABLE_NAME = "TATEMON";

    private static final String UPDATE_BASE_WHERE_CLAUSE = "BLD_NO = ?";

    private static final ReflectiveRecordEntityMapper<TatemonRecord, TatemonEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(TatemonRecord.class, TatemonEntity.class);

    private static final ReflectiveRecordEntityMapper<TatemonWriteRecord, TatemonEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            TatemonWriteRecord.class, TatemonEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<TatemonEntity> auditColumnAccessBinder;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaTargetColumnUpdateSupport<TatemonWriteRecord, TatemonEntity, TatemonColumn>
            updateSupport;

    /**
     * JPA ベースの TATEMON データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public TatemonDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new NcisCisAuditColumnAutoSetter());
    }

    TatemonDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(TatemonEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        TatemonWriteDefinition.INSTANCE,
                        TatemonEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public Optional<TatemonRecord> findByBldNo(String bldNo) {
        Objects.requireNonNull(bldNo, "bldNo");
        var builder = criteriaBuilderFactory.create(entityManager, TatemonEntity.class);
        return builder.where("bldNo").eq(bldNo).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByBldNoAndRecUpdJnjNo(
            TatemonWriteRecord record,
            Set<TatemonColumn> targetColumns,
            TatemonUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                PreparedWhere.of(
                        UPDATE_BASE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            return JpaBind.set(query, startIndex, updateKey.bldNo());
                        },
                        updateKey.bldNo()),
                updateKey.recUpdJnjNo(),
                TatemonDataAccess.Identifiers.UPDATE_BY_BLD_NO_AND_REC_UPD_JNJ_NO);
    }
}
