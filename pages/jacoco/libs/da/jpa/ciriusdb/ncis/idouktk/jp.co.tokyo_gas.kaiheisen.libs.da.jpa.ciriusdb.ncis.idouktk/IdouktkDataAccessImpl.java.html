package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.idouktk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import jakarta.persistence.Query;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalTime;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.idouktk.IdouktkDataAccess;

/**
 * IDOUKTK（異動受付）データアクセスの JPA 実装。
 */
public class IdouktkDataAccessImpl extends DataAccessBase implements IdouktkDataAccess {

    private static final String TABLE_NAME = "IDOUKTK";

    private static final String FIND_MAX_SEQ_SQL =
            "SELECT MAX(IDO_RKK_SEQ_NO) FROM IDOUKTK"
                    + " WHERE OKYAKU_NO = :okyakuNo"
                    + " AND IDOH_HKK_YMD = :idohHkkYmd"
                    + " AND IDO_HKOKU_JKK = :idoHkokuJkk";

    private static final ReflectiveRecordEntityMapper<IdouktkWriteRecord, IdouktkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            IdouktkWriteRecord.class, IdouktkEntity.class);

    private final EntityManager entityManager;

    private final JpaAuditColumnAccessBinder<IdouktkEntity> auditColumnAccessBinder;

    private final SystemDateTimeProvider clock;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<IdouktkWriteRecord, IdouktkEntity, IdouktkColumn> insertSupport;

    /**
     * JPA ベースの IDOUKTK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public IdouktkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new NcisCisAuditColumnAutoSetter());
    }

    IdouktkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new NcisCisAuditColumnAutoSetter());
    }

    IdouktkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(IdouktkEntity.class);
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        IdouktkWriteDefinition.INSTANCE,
                        IdouktkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(IdouktkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<Integer> findByOkyakuNoAndIdohHkkYmdAndIdoHkokuJkk(
            Long okyakuNo, LocalDate idohHkkYmd, LocalTime idoHkokuJkk) {
        Long safeOkyakuNo = WhereParamGuard.complement(okyakuNo, "okyakuNo", 0L);
        BigDecimal safeIdohHkkYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(idohHkkYmd), "idohHkkYmd", BigDecimal.ZERO);
        BigDecimal safeIdoHkokuJkk =
                WhereParamGuard.complement(
                        JpaDbWriteValues.localTime(idoHkokuJkk),
                        "idoHkokuJkk",
                        BigDecimal.ZERO);

        SqlCapture.set(FIND_MAX_SEQ_SQL);
        Query query = entityManager.createNativeQuery(FIND_MAX_SEQ_SQL);
        query.setParameter("okyakuNo", JpaDbWriteValues.bigDecimal(safeOkyakuNo));
        query.setParameter("idohHkkYmd", safeIdohHkkYmd);
        query.setParameter("idoHkokuJkk", safeIdoHkokuJkk);

        Object value = query.getSingleResult();
        if (value == null) {
            return Optional.empty();
        }
        if (value instanceof BigDecimal bigDecimal) {
            return Optional.ofNullable(JpaDbReadValues.integerOrNullIfZero(bigDecimal));
        }
        if (value instanceof Number number) {
            return Optional.of(number.intValue());
        }
        String text = value.toString();
        if (text.isBlank()) {
            return Optional.empty();
        }
        return Optional.of(Integer.parseInt(text));
    }
}
