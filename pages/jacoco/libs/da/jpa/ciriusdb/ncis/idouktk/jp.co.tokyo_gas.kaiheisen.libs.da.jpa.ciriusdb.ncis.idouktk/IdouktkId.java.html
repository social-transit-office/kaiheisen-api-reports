package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.idouktk;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * IDOUKTK の複合主キーを表現する ID クラス。
 */
public class IdouktkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal okyakuNo;

    private BigDecimal idohHkkYmd;

    private BigDecimal idoHkokuJkk;

    private BigDecimal idoRkkSeqNo;

    private String idoHkkGymCd;

    /** デフォルトコンストラクタ。 */
    public IdouktkId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public IdouktkId(
            BigDecimal okyakuNo,
            BigDecimal idohHkkYmd,
            BigDecimal idoHkokuJkk,
            BigDecimal idoRkkSeqNo,
            String idoHkkGymCd) {
        this.okyakuNo = okyakuNo;
        this.idohHkkYmd = idohHkkYmd;
        this.idoHkokuJkk = idoHkokuJkk;
        this.idoRkkSeqNo = idoRkkSeqNo;
        this.idoHkkGymCd = idoHkkGymCd;
    }

    /**
     * お客さま番号を返す。
     *
     * @return お客さま番号
     */
    public BigDecimal getOkyakuNo() {
        return okyakuNo;
    }

    /**
     * お客さま番号を設定する。
     *
     * @param okyakuNo お客さま番号
     */
    public void setOkyakuNo(BigDecimal okyakuNo) {
        this.okyakuNo = okyakuNo;
    }

    /**
     * 異動報告年月日を返す。
     *
     * @return 異動報告年月日
     */
    public BigDecimal getIdohHkkYmd() {
        return idohHkkYmd;
    }

    /**
     * 異動報告年月日を設定する。
     *
     * @param idohHkkYmd 異動報告年月日
     */
    public void setIdohHkkYmd(BigDecimal idohHkkYmd) {
        this.idohHkkYmd = idohHkkYmd;
    }

    /**
     * 異動報告時刻を返す。
     *
     * @return 異動報告時刻
     */
    public BigDecimal getIdoHkokuJkk() {
        return idoHkokuJkk;
    }

    /**
     * 異動報告時刻を設定する。
     *
     * @param idoHkokuJkk 異動報告時刻
     */
    public void setIdoHkokuJkk(BigDecimal idoHkokuJkk) {
        this.idoHkokuJkk = idoHkokuJkk;
    }

    /**
     * 異動履歴通し番号を返す。
     *
     * @return 異動履歴通し番号
     */
    public BigDecimal getIdoRkkSeqNo() {
        return idoRkkSeqNo;
    }

    /**
     * 異動履歴通し番号を設定する。
     *
     * @param idoRkkSeqNo 異動履歴通し番号
     */
    public void setIdoRkkSeqNo(BigDecimal idoRkkSeqNo) {
        this.idoRkkSeqNo = idoRkkSeqNo;
    }

    /**
     * 異動報告業務コードを返す。
     *
     * @return 異動報告業務コード
     */
    public String getIdoHkkGymCd() {
        return idoHkkGymCd;
    }

    /**
     * 異動報告業務コードを設定する。
     *
     * @param idoHkkGymCd 異動報告業務コード
     */
    public void setIdoHkkGymCd(String idoHkkGymCd) {
        this.idoHkkGymCd = idoHkkGymCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof IdouktkId that)) {
            return false;
        }
        return compareBigDecimal(okyakuNo, that.okyakuNo)
                && compareBigDecimal(idohHkkYmd, that.idohHkkYmd)
                && compareBigDecimal(idoHkokuJkk, that.idoHkokuJkk)
                && compareBigDecimal(idoRkkSeqNo, that.idoRkkSeqNo)
                && java.util.Objects.equals(idoHkkGymCd, that.idoHkkGymCd);
    }

    private static boolean compareBigDecimal(BigDecimal left, BigDecimal right) {
        if (left == null) {
            return right == null;
        }
        return right != null && left.compareTo(right) == 0;
    }

    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (okyakuNo == null ? 0 : okyakuNo.stripTrailingZeros().hashCode());
        result =
                31 * result
                        + (idohHkkYmd == null ? 0 : idohHkkYmd.stripTrailingZeros().hashCode());
        result =
                31 * result
                        + (idoHkokuJkk == null ? 0 : idoHkokuJkk.stripTrailingZeros().hashCode());
        result =
                31 * result
                        + (idoRkkSeqNo == null ? 0 : idoRkkSeqNo.stripTrailingZeros().hashCode());
        result = 31 * result + (idoHkkGymCd == null ? 0 : idoHkkGymCd.hashCode());
        return result;
    }
}
