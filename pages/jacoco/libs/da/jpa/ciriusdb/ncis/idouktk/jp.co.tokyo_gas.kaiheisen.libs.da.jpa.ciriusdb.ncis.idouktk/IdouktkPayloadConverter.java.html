package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.idouktk;

import jakarta.persistence.AttributeConverter;
import jakarta.persistence.Converter;
import java.nio.charset.StandardCharsets;

/** IDOUKTKのバイト領域と既存の1バイト対1文字表現を可逆変換する。 */
@Converter
public class IdouktkPayloadConverter implements AttributeConverter<String, byte[]> {

    @Override
    public byte[] convertToDatabaseColumn(String value) {
        if (value == null) {
            return null;
        }
        for (int i = 0; i < value.length(); i++) {
            if (value.charAt(i) > 0xFF) {
                throw new IllegalArgumentException("Payload must use one character per byte");
            }
        }
        return value.getBytes(StandardCharsets.ISO_8859_1);
    }

    @Override
    public String convertToEntityAttribute(byte[] value) {
        return value == null ? null : new String(value, StandardCharsets.ISO_8859_1);
    }
}
