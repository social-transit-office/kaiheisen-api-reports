package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.itensak;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * 概要: ITENSAK の複合主キーを表現する ID クラス。
 *
 * <p>主な仕様: - {@code GMT_SET_NO/SYOKY_NO/ITEN_SAK_SEQ_NO} の 3 項目で同一性を判定する。 - JPA {@code @IdClass}
 * 要件に合わせて no-args コンストラクタを公開する。
 *
 * <p>既知の制約: - 値の妥当性（桁数・範囲）は本クラスでは検証しない。
 */
public class ItensakId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gmtSetNo;

    private BigDecimal syokyNo;

    private BigDecimal itenSakSeqNo;

    /** デフォルトコンストラクタ。 */
    public ItensakId() { }

    /** 全フィールド指定コンストラクタ。 */
    public ItensakId(BigDecimal gmtSetNo, BigDecimal syokyNo, BigDecimal itenSakSeqNo) {
        this.gmtSetNo = gmtSetNo;
        this.syokyNo = syokyNo;
        this.itenSakSeqNo = itenSakSeqNo;
    }

    /** gmtSetNoを取得する。 */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /** gmtSetNoを設定する。 */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /** syokyNoを取得する。 */
    public BigDecimal getSyokyNo() {
        return syokyNo;
    }

    /** syokyNoを設定する。 */
    public void setSyokyNo(BigDecimal syokyNo) {
        this.syokyNo = syokyNo;
    }

    /** itenSakSeqNoを取得する。 */
    public BigDecimal getItenSakSeqNo() {
        return itenSakSeqNo;
    }

    /** itenSakSeqNoを設定する。 */
    public void setItenSakSeqNo(BigDecimal itenSakSeqNo) {
        this.itenSakSeqNo = itenSakSeqNo;
    }

    /**
     * 概要: 主キー 3 項目の値で同値判定を行う。
     *
     * <p>主な仕様: - すべてのキー項目が一致する場合のみ true を返す。 - null 安全に比較する。
     *
     * <p>既知の制約: - 型が異なるオブジェクトとの比較は常に false になる。
     */
    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof ItensakId that)) {
            return false;
        }
        return compareBigDecimal(gmtSetNo, that.gmtSetNo)
                && compareBigDecimal(syokyNo, that.syokyNo)
                && compareBigDecimal(itenSakSeqNo, that.itenSakSeqNo);
    }

    private static boolean compareBigDecimal(BigDecimal a, BigDecimal b) {
        if (a == null) {
            return b == null;
        }
        return b != null && a.compareTo(b) == 0;
    }

    /**
     * 概要: 主キー 3 項目を用いてハッシュ値を計算する。
     *
     * <p>主な仕様: - equals と整合するよう各キー値を同一比率で合成する。
     *
     * <p>既知の制約: - ハッシュ衝突回避は Java 標準の hashCode 契約に依存する。
     */
    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (gmtSetNo == null ? 0 : gmtSetNo.stripTrailingZeros().hashCode());
        result = 31 * result + (syokyNo == null ? 0 : syokyNo.stripTrailingZeros().hashCode());
        result =
                31 * result
                        + (itenSakSeqNo == null ? 0 : itenSakSeqNo.stripTrailingZeros().hashCode());
        return result;
    }
}
