package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.itensak;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.write.ItensakColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.write.ItensakUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.write.ItensakWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.write.ItensakWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** ITENSAK（移転先管理）データアクセスの JPA 実装。 */
public class ItensakDataAccessImpl extends DataAccessBase implements ItensakDataAccess {

    private static final String TABLE_NAME = "ITENSAK";

    private static final String UPDATE_BASE_WHERE_CLAUSE =
            "GMT_SET_NO = ? AND SYOKY_NO = ? AND ITEN_SAK_SEQ_NO = ?";

    private static final ReflectiveRecordEntityMapper<ItensakRecord, ItensakEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(ItensakRecord.class, ItensakEntity.class);

    private static final ReflectiveRecordEntityMapper<ItensakWriteRecord, ItensakEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            ItensakWriteRecord.class, ItensakEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<ItensakEntity> auditColumnAccessBinder;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<ItensakWriteRecord, ItensakEntity, ItensakColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<ItensakWriteRecord, ItensakEntity, ItensakColumn>
            updateSupport;

    /**
     * JPA ベースの ITENSAK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public ItensakDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new NcisCisAuditColumnAutoSetter());
    }

    ItensakDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new NcisCisAuditColumnAutoSetter());
    }

    ItensakDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(ItensakEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        ItensakWriteDefinition.INSTANCE,
                        ItensakEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        ItensakWriteDefinition.INSTANCE,
                        ItensakEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(ItensakWriteRecord record, AuditContext context) {
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
            Long gmtSetNo, Long syokyNo, Integer itenSakSeqNo, Integer recUpdJnjNo) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeSyokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        var safeItenSakSeqNo = WhereParamGuard.complement(itenSakSeqNo, "itenSakSeqNo", 0);

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                UPDATE_BASE_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    startIndex,
                                                    safeGmtSetNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    nextIndex,
                                                    safeSyokyNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    return JpaBind.set(
                                            query,
                                            nextIndex,
                                            safeItenSakSeqNo,
                                            JpaDbWriteValues::bigDecimal);
                                },
                                JpaDbWriteValues.bigDecimal(safeGmtSetNo),
                                JpaDbWriteValues.bigDecimal(safeSyokyNo),
                                JpaDbWriteValues.bigDecimal(safeItenSakSeqNo)),
                        recUpdJnjNo,
                        ItensakDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_SYOKY_NO_AND_ITEN_SAK_SEQ_NO_AND_REC_UPD_JNJ_NO.identifier());
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndSyokyNo(Long gmtSetNo, Long syokyNo) {
        gmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        syokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);

        var builder = criteriaBuilderFactory.create(entityManager, ItensakEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", gmtSetNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "syokyNo", syokyNo, JpaDbWriteValues::bigDecimal);
        builder.orderByDesc("itenSakSeqNo");

        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public java.util.Optional<ItensakRecord> findByGmtSetNoAndSyokyNoAndItenSakSeqNo(
            Long gmtSetNo, Long syokyNo, Integer itenSakSeqNo) {
        gmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        syokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        itenSakSeqNo = WhereParamGuard.complement(itenSakSeqNo, "itenSakSeqNo", 0);

        var builder = criteriaBuilderFactory.create(entityManager, ItensakEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", gmtSetNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "syokyNo", syokyNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "itenSakSeqNo", itenSakSeqNo, JpaDbWriteValues::bigDecimal);

        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndSyokyNoAndItenTksCd(
            Long gmtSetNo, Long syokyNo, String itenTksCd) {
        gmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        syokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        WhereParamGuard.requireNonNull(itenTksCd, "itenTksCd");

        var builder = criteriaBuilderFactory.create(entityManager, ItensakEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", gmtSetNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "syokyNo", syokyNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "itenTksCd", itenTksCd);
        builder.orderByAsc("gmtSetNo");
        builder.orderByAsc("syokyNo");
        builder.orderByDesc("itenSakSeqNo");

        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndItenSShrkyNoAndItenTksCd(
            Long gmtSetNo, Long itenSShrkyNo, String itenTksCd) {
        gmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        itenSShrkyNo = WhereParamGuard.complement(itenSShrkyNo, "itenSShrkyNo", 0L);
        WhereParamGuard.requireNonNull(itenTksCd, "itenTksCd");

        var builder = criteriaBuilderFactory.create(entityManager, ItensakEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", gmtSetNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "itenSShrkyNo", itenSShrkyNo, JpaDbWriteValues::bigDecimal);
        JpaWhere.eq(builder, "itenTksCd", itenTksCd);

        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public int updateByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
            ItensakWriteRecord record,
            Set<ItensakColumn> targetColumns,
            ItensakUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                PreparedWhere.of(
                        UPDATE_BASE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(
                                            query,
                                            startIndex,
                                            updateKey.gmtSetNo(),
                                            JpaDbWriteValues::bigDecimal);
                            nextIndex =
                                    JpaBind.set(
                                            query,
                                            nextIndex,
                                            updateKey.syokyNo(),
                                            JpaDbWriteValues::bigDecimal);
                            nextIndex =
                                    JpaBind.set(
                                            query,
                                            nextIndex,
                                            updateKey.itenSakSeqNo(),
                                            JpaDbWriteValues::bigDecimal);
                            return nextIndex;
                        },
                        JpaDbWriteValues.bigDecimal(updateKey.gmtSetNo()),
                        JpaDbWriteValues.bigDecimal(updateKey.syokyNo()),
                        JpaDbWriteValues.bigDecimal(updateKey.itenSakSeqNo())),
                updateKey.recUpdJnjNo(),
                ItensakDataAccess.CallSpecs.UPDATE_BY_GMT_SET_NO_AND_SYOKY_NO_AND_ITEN_SAK_SEQ_NO_AND_REC_UPD_JNJ_NO.identifier());
    }
}
