package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.kenssie;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaBind;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * KENSSIE データアクセスの JPA 実装。
 *
 * <p>主な仕様: DA-009-D001 の削除、DA-009-R001 の単一取得、DA-009-R002 の一覧取得を提供する。
 *
 * <p>制限: 例外のラップや SQL ログは Proxy 層（DataAccessProxy）側で行う。
 */
public class KenssieDataAccessImpl extends DataAccessBase implements KenssieDataAccess {

    private static final String TABLE_NAME = "KENSSIE";

    private static final ReflectiveRecordEntityMapper<KenssieRecord, KenssieEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(KenssieRecord.class, KenssieEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public KenssieDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GMT_SET_NO = ? AND IDO_MASY_KHI_CD = ?",
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    startIndex,
                                                    gmtSetNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    return JpaBind.set(query, nextIndex, idoMasyKhiCd);
                                },
                                JpaDbWriteValues.bigDecimal(gmtSetNo),
                                idoMasyKhiCd),
                            recUpdJnjNo,
                            KenssieDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier());
    }

    @Override
    public Optional<KenssieRecord> findByGmtSetNoAndKnsnSienBrCd(
            Long gmtSetNo, String knsnSienBrCd) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(knsnSienBrCd, "knsnSienBrCd");
        var builder = criteriaBuilderFactory.create(entityManager, KenssieEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", JpaDbWriteValues.bigDecimal(gmtSetNo));
        JpaWhere.eq(builder, "knsnSienBrCd", knsnSienBrCd);
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    @Override
    public List<KenssieRecord> findAllByGmtSetNoAndKnsnSienBrCd(
            Long gmtSetNo, List<String> knsnSienBrCdList) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(knsnSienBrCdList, "knsnSienBrCdList");
        if (knsnSienBrCdList.isEmpty()) {
            return List.of();
        }

        var builder = criteriaBuilderFactory.create(entityManager, KenssieEntity.class);
        JpaWhere.eq(builder, "gmtSetNo", JpaDbWriteValues.bigDecimal(gmtSetNo));
        JpaWhere.in(builder, "knsnSienBrCd", knsnSienBrCdList);
        builder.orderByAsc("knsnSienBrCd");

        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }
}
