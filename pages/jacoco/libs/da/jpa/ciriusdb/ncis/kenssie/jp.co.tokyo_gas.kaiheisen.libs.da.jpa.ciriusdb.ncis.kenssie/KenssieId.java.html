package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.kenssie;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * KENSSIE の複合主キー。
 *
 * <p>主な仕様: (GMT_SET_NO, KNSN_SIEN_BR_CD) を主キーとして表現する。
 *
 * <p>制限: JPA の要件により public no-args ctor を持つ。
 */
public class KenssieId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gmtSetNo;

    private String knsnSienBrCd;

    /** デフォルトコンストラクタ。 */
    public KenssieId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KenssieId(BigDecimal gmtSetNo, String knsnSienBrCd) {
        this.gmtSetNo = gmtSetNo;
        this.knsnSienBrCd = knsnSienBrCd;
    }

    /** gmtSetNoを取得する。 */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /** gmtSetNoを設定する。 */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /** knsnSienBrCdを取得する。 */
    public String getKnsnSienBrCd() {
        return knsnSienBrCd;
    }

    /** knsnSienBrCdを設定する。 */
    public void setKnsnSienBrCd(String knsnSienBrCd) {
        this.knsnSienBrCd = knsnSienBrCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KenssieId that)) {
            return false;
        }

        if (gmtSetNo == null) {
            if (that.gmtSetNo != null) {
                return false;
            }
        } else if (!gmtSetNo.equals(that.gmtSetNo)) {
            return false;
        }
        if (knsnSienBrCd == null) {
            return that.knsnSienBrCd == null;
        }
        return knsnSienBrCd.equals(that.knsnSienBrCd);
    }

    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (gmtSetNo == null ? 0 : gmtSetNo.hashCode());
        result = 31 * result + (knsnSienBrCd == null ? 0 : knsnSienBrCd.hashCode());
        return result;
    }
}
