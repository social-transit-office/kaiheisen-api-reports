package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.cstsvc;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * CSTSVC の複合主キー。
 */
public class CstsvcId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gmtSetNo;

    private String svcRyuBnCd;

    /** デフォルトコンストラクタ。 */
    public CstsvcId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public CstsvcId(BigDecimal gmtSetNo, String svcRyuBnCd) {
        this.gmtSetNo = gmtSetNo;
        this.svcRyuBnCd = svcRyuBnCd;
    }

    /**
     * ガスメータ設置場所番号を返す。
     *
     * @return ガスメータ設置場所番号
     */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /**
     * ガスメータ設置場所番号を設定する。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /**
     * サービス留意分類コードを返す。
     *
     * @return サービス留意分類コード
     */
    public String getSvcRyuBnCd() {
        return svcRyuBnCd;
    }

    /**
     * サービス留意分類コードを設定する。
     *
     * @param svcRyuBnCd サービス留意分類コード
     */
    public void setSvcRyuBnCd(String svcRyuBnCd) {
        this.svcRyuBnCd = svcRyuBnCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof CstsvcId that)) {
            return false;
        }
        if (gmtSetNo == null) {
            if (that.gmtSetNo != null) {
                return false;
            }
        } else if (!gmtSetNo.equals(that.gmtSetNo)) {
            return false;
        }
        if (svcRyuBnCd == null) {
            return that.svcRyuBnCd == null;
        }
        return svcRyuBnCd.equals(that.svcRyuBnCd);
    }

    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (gmtSetNo == null ? 0 : gmtSetNo.hashCode());
        result = 31 * result + (svcRyuBnCd == null ? 0 : svcRyuBnCd.hashCode());
        return result;
    }
}
