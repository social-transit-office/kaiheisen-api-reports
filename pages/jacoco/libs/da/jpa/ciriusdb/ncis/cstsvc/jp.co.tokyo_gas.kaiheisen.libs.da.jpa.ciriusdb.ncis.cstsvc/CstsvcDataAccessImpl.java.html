package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.cstsvc;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.cstsvc.CstsvcDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaBind;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;

/**
 * CSTSVC データアクセスの JPA 実装。
 */
public class CstsvcDataAccessImpl extends DataAccessBase implements CstsvcDataAccess {

    /** 操作対象テーブル名。 */
    private static final String TABLE_NAME = "CSTSVC";

    /** 異動時抹消可否コードの固定長。 */
    private static final int IDO_MASY_KHI_CD_LENGTH = 1;

    /** JPA 操作用の EntityManager。 */
    private final EntityManager entityManager;

    /** Criteria API を構築するファクトリ。 */
    @SuppressWarnings("unused")
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /**
     * コンストラクタ。
     *
     * @param entityManager JPA 操作用の EntityManager
     * @param criteriaBuilderFactory Criteria API を構築するファクトリ
     */
    public CstsvcDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    /**
     * ガスメータ設置場所番号と異動時抹消可否コードを条件に行を削除する。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     * @param idoMasyKhiCd 異動時抹消可否コード
     * @param recUpdJnjNo 排他制御用のレコード更新順序番号
     * @return 削除件数
     */
    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Long safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        String safeIdoMasyKhiCd =
                WhereParamGuard.fixedLengthNotNull(
                        idoMasyKhiCd, "idoMasyKhiCd", IDO_MASY_KHI_CD_LENGTH);

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GMT_SET_NO = ? AND IDO_MASY_KHI_CD = ?",
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    startIndex,
                                                    safeGmtSetNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    return JpaBind.set(query, nextIndex, safeIdoMasyKhiCd);
                                },
                                JpaDbWriteValues.bigDecimal(safeGmtSetNo),
                                safeIdoMasyKhiCd),
                        recUpdJnjNo,
                        CstsvcDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier());
    }
}
