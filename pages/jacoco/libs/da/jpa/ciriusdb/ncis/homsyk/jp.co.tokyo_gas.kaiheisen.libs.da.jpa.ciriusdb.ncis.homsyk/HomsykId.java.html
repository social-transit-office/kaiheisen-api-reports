package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.ncis.homsyk;

import java.io.Serializable;
import java.math.BigDecimal;

/** HOMSYK の複合主キー。 */
public class HomsykId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gmtSetNo;

    private String hmsSykSienCd;

    /** デフォルトコンストラクタ。 */
    public HomsykId() {
    }

    /**
     * 全フィールド指定コンストラクタ。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     * @param hmsSykSienCd 訪問集金支援分類コード
     */
    public HomsykId(BigDecimal gmtSetNo, String hmsSykSienCd) {
        this.gmtSetNo = gmtSetNo;
        this.hmsSykSienCd = hmsSykSienCd;
    }

    /**
     * ガスメータ設置場所番号を返す。
     *
     * @return ガスメータ設置場所番号
     */
    public BigDecimal getGmtSetNo() {
        return gmtSetNo;
    }

    /**
     * ガスメータ設置場所番号を設定する。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     */
    public void setGmtSetNo(BigDecimal gmtSetNo) {
        this.gmtSetNo = gmtSetNo;
    }

    /**
     * 訪問集金支援分類コードを返す。
     *
     * @return 訪問集金支援分類コード
     */
    public String getHmsSykSienCd() {
        return hmsSykSienCd;
    }

    /**
     * 訪問集金支援分類コードを設定する。
     *
     * @param hmsSykSienCd 訪問集金支援分類コード
     */
    public void setHmsSykSienCd(String hmsSykSienCd) {
        this.hmsSykSienCd = hmsSykSienCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof HomsykId that)) {
            return false;
        }
        if (gmtSetNo == null) {
            if (that.gmtSetNo != null) {
                return false;
            }
        } else if (!gmtSetNo.equals(that.gmtSetNo)) {
            return false;
        }
        if (hmsSykSienCd == null) {
            return that.hmsSykSienCd == null;
        }
        return hmsSykSienCd.equals(that.hmsSykSienCd);
    }

    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + (gmtSetNo == null ? 0 : gmtSetNo.hashCode());
        result = 31 * result + (hmsSykSienCd == null ? 0 : hmsSykSienCd.hashCode());
        return result;
    }
}
