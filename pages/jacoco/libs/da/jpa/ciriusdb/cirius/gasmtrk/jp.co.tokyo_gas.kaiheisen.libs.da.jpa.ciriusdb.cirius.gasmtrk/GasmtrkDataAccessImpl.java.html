package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gasmtrk;

import com.blazebit.persistence.CriteriaBuilder;
import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GASMTRK データアクセスの JPA 実装 */
public class GasmtrkDataAccessImpl extends DataAccessBase implements GasmtrkDataAccess {

    private static final ReflectiveRecordEntityMapper<GasmtrkRecord, GasmtrkEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(GasmtrkRecord.class, GasmtrkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public GasmtrkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<GasmtrkRecord> findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate knskKjnYmd) {
        Objects.requireNonNull(ttkGmsbNo, "ttkGmsbNo");
        var builder = createBuilderForDa026R002(knskKjnYmd);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(ttkGmsbNo));
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public List<GasmtrkRecord> findAllByTtkGmsbNoAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        Objects.requireNonNull(ttkGmsbNo, "ttkGmsbNo");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrkEntity.class);
        BigDecimal gmtThkgYmdValue =
            gmtThkgYmd == null ? BigDecimal.ZERO : JpaDbWriteValues.ymd(gmtThkgYmd);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(ttkGmsbNo));
        JpaWhere.eq(builder, "gmtThkgYmd", gmtThkgYmdValue);
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public List<GasmtrkRecord> findAllByGmbAndGmtkYmdAndGmtThkgYmd(
            String gmb, LocalDate knskKjnYmd) {
        Objects.requireNonNull(gmb, "gmb");
        var builder = createBuilderForDa026R003(knskKjnYmd);
        JpaWhere.eq(builder, "gmb", gmb);
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    private CriteriaBuilder<GasmtrkEntity> createBuilderForDa026R002(LocalDate knskKjnYmd) {
        var builder = createBaseBuilderByBaseDate(knskKjnYmd);
        builder.orderByDesc("gmtkYmd")
                .orderByAsc("gmtThkgYmd")
                .orderByDesc("gmtThkgJysb")
                .orderByAsc("gmb")
                .orderByAsc("gmbTsno");
        return builder;
    }

    private CriteriaBuilder<GasmtrkEntity> createBuilderForDa026R003(LocalDate knskKjnYmd) {
        var builder = createBaseBuilderByBaseDate(knskKjnYmd);
        builder.orderByDesc("gmtkYmd")
                .orderByDesc("gmtThkgYmd")
                .orderByDesc("gmtThkgJysb")
                .orderByAsc("gmb")
                .orderByAsc("gmbTsno");
        return builder;
    }

    private CriteriaBuilder<GasmtrkEntity> createBaseBuilderByBaseDate(LocalDate knskKjnYmd) {
        Objects.requireNonNull(knskKjnYmd, "knskKjnYmd");
        var baseDateValue = JpaDbWriteValues.ymd(knskKjnYmd);
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrkEntity.class);
        JpaWhere.le(builder, "gmtkYmd", baseDateValue);
        builder.whereOr()
                .where("gmtThkgYmd")
                .ge(baseDateValue)
                .where("gmtThkgYmd")
                .eq(BigDecimal.ZERO)
                .endOr();
        return builder;
    }

    @Override
    public Optional<GasmtrkRecord> findByPK(String gmb, Integer gmbTsno) {
        Objects.requireNonNull(gmb, "gmb");
        Objects.requireNonNull(gmbTsno, "gmbTsno");

        GasmtrkId id = new GasmtrkId(gmb, JpaDbWriteValues.bigDecimal(gmbTsno));
        GasmtrkEntity entity = entityManager.find(GasmtrkEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }
}
