/** NTKNSK データアクセスの JPA 実装。 */
public class NtknskDataAccessImpl extends DataAccessBase implements NtknskDataAccess {

    private static final int NWJG_CD_LENGTH = 5;

    private static final int UTLT_KBN_LENGTH = 1;

    private static final int TREK_SSKB_NO_LENGTH = 2;

    private static final int TREK_SDKB_NO_LENGTH = 2;

    private static final int TKKB_NO_LENGTH = 2;

    private static final int SHTI_EKKB_NO_LENGTH = 2;

    private static final int SHTI_HKSD_KBN_LENGTH = 2;

    private static final int SHTI_EKSD_KBN_LENGTH = 2;

    private static final ReflectiveRecordEntityMapper<NtknskRecord, NtknskEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(NtknskRecord.class, NtknskEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public NtknskDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<NtknskRecord> findAllByKymYmdAndNwjgCdAndUtltKbnAndTkkbNo(
            LocalDate kymYmdFrom,
            LocalDate kymYmdTo,
            String nwjgCd,
            String utltKbn,
            String tkkbNo) {
        BigDecimal safeKymYmdFrom = safeYmd(kymYmdFrom, "kymYmdFrom");
        BigDecimal safeKymYmdTo = safeYmd(kymYmdTo, "kymYmdTo");
        String safeNwjgCd = safeNwjgCd(nwjgCd, "nwjgCd");
        String safeUtltKbn = safeUtltKbn(utltKbn, "utltKbn");
        String safeTkkbNo = safeTkkbNo(tkkbNo, "tkkbNo");

        var builder = criteriaBuilderFactory.create(entityManager, NtknskEntity.class);
        JpaWhere.ge(builder, "kymYmd", safeKymYmdFrom);
        JpaWhere.le(builder, "kymYmd", safeKymYmdTo);
        JpaWhere.eq(builder, "nwjgCd", safeNwjgCd);
        JpaWhere.eq(builder, "utltKbn", safeUtltKbn);
        JpaWhere.eq(builder, "tkkbNo", safeTkkbNo);

        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public List<NtknskRecord> findAllByKymYmdAndNwjgCdAndUtltKbnAndTrekSskbNoAndTrekSdkbNo(
            LocalDate kymYmdFrom,
            LocalDate kymYmdTo,
            String nwjgCd,
            String utltKbn,
            String trekSskbNo,
            String trekSdkbNo) {
        BigDecimal safeKymYmdFrom = safeYmd(kymYmdFrom, "kymYmdFrom");
        BigDecimal safeKymYmdTo = safeYmd(kymYmdTo, "kymYmdTo");
        String safeNwjgCd = safeNwjgCd(nwjgCd, "nwjgCd");
        String safeUtltKbn = safeUtltKbn(utltKbn, "utltKbn");
        String safeTrekSskbNo = safeTrekSskbNo(trekSskbNo, "trekSskbNo");
        String safeTrekSdkbNo = safeTrekSdkbNo(trekSdkbNo, "trekSdkbNo");

        var builder = criteriaBuilderFactory.create(entityManager, NtknskEntity.class);
        JpaWhere.ge(builder, "kymYmd", safeKymYmdFrom);
        JpaWhere.le(builder, "kymYmd", safeKymYmdTo);
        JpaWhere.eq(builder, "nwjgCd", safeNwjgCd);
        JpaWhere.eq(builder, "utltKbn", safeUtltKbn);
        builder.whereOr()
                .where("trekSskbNo")
            .eq(safeTrekSskbNo)
                .where("trekSdkbNo")
            .eq(safeTrekSdkbNo)
                .endOr();

        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public Optional<NtknskRecord> findByPK(LocalDate kymYmd, String nwjgCd, String utltKbn) {
        BigDecimal safeKymYmd = safeYmd(kymYmd, "kymYmd");
        String safeNwjgCd = safeNwjgCd(nwjgCd, "nwjgCd");
        String safeUtltKbn = safeUtltKbn(utltKbn, "utltKbn");

        NtknskId id = new NtknskId(safeKymYmd, safeNwjgCd, safeUtltKbn);
        NtknskEntity entity = entityManager.find(NtknskEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }

    private static String safeNwjgCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, NWJG_CD_LENGTH);
    }

    private static String safeUtltKbn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, UTLT_KBN_LENGTH);
    }

    private static String safeTrekSskbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TREK_SSKB_NO_LENGTH);
    }

    private static String safeTrekSdkbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TREK_SDKB_NO_LENGTH);

    private static String safeTkkbNo(String value, String paramName) {

    }

    private static String safeShtiEkkbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SHTI_EKKB_NO_LENGTH);
    }

    private static String safeShtiHksdKbn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SHTI_HKSD_KBN_LENGTH);
    }

    private static String safeShtiEksdKbn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SHTI_EKSD_KBN_LENGTH);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
