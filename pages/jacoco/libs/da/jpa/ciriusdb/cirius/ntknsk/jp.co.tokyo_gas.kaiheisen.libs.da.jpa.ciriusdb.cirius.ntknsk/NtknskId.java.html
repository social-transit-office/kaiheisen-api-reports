package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ntknsk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 日程管理検針日テーブルの複合主キー。 */
public class NtknskId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal kymYmd;

    private String nwjgCd;

    private String utltKbn;

    /** デフォルトコンストラクタ。 */
    public NtknskId() { }

    /** 全フィールド指定コンストラクタ。 */
    public NtknskId(BigDecimal kymYmd, String nwjgCd, String utltKbn) {
        this.kymYmd = kymYmd;
        this.nwjgCd = nwjgCd;
        this.utltKbn = utltKbn;
    }

    /** kymYmdを取得する。 */
    public BigDecimal getKymYmd() {
        return kymYmd;
    }

    /** kymYmdを設定する。 */
    public void setKymYmd(BigDecimal kymYmd) {
        this.kymYmd = kymYmd;
    }

    /** nwjgCdを取得する。 */
    public String getNwjgCd() {
        return nwjgCd;
    }

    /** nwjgCdを設定する。 */
    public void setNwjgCd(String nwjgCd) {
        this.nwjgCd = nwjgCd;
    }

    /** utltKbnを取得する。 */
    public String getUtltKbn() {
        return utltKbn;
    }

    /** utltKbnを設定する。 */
    public void setUtltKbn(String utltKbn) {
        this.utltKbn = utltKbn;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof NtknskId that)) {
            return false;
        }
        return Objects.equals(kymYmd, that.kymYmd)
                && Objects.equals(nwjgCd, that.nwjgCd)
                && Objects.equals(utltKbn, that.utltKbn);
    }

    @Override
    public int hashCode() {
        return Objects.hash(kymYmd, nwjgCd, utltKbn);
    }
}
