package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kyoteld;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.EnumSet;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.KyoteldDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.KyoteldRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.write.KyoteldColumn;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.write.KyoteldWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyoteld.write.KyoteldWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;

/** KYOTELD データアクセスの JPA 実装。 */
public class KyoteldDataAccessImpl extends DataAccessBase implements KyoteldDataAccess {

    private static final String TABLE_NAME = "KYOTELD";

    private static final String PK_WHERE_CLAUSE = "KYO_TRK_NO = ? AND KYO_TEL_NO = ?";

    private static final ReflectiveRecordEntityMapper<KyoteldRecord, KyoteldEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KyoteldRecord.class, KyoteldEntity.class);

    private static final ReflectiveRecordEntityMapper<KyoteldWriteRecord, KyoteldEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KyoteldWriteRecord.class, KyoteldEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KyoteldEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KyoteldWriteRecord, KyoteldEntity, KyoteldColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KyoteldWriteRecord, KyoteldEntity, KyoteldColumn>
            updateSupport;

    /**
     * JPA 実装を生成する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public KyoteldDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KyoteldDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KyoteldDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KyoteldEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KyoteldWriteDefinition.INSTANCE,
                        KyoteldEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KyoteldWriteDefinition.INSTANCE,
                        KyoteldEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KyoteldWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            Long kyoTrkNo, String kyoTelNo, Integer recUpdJnjNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                PK_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeKyoTrkNo);
                                    return JpaBind.set(query, nextIndex, safeKyoTelNo);
                                },
                                safeKyoTrkNo,
                                safeKyoTelNo),
                        recUpdJnjNo,
                        KyoteldDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO);
    }

    @Override
    public long countByKyoTrkNo(Long kyoTrkNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var query =
                entityManager.createNativeQuery(
                        "SELECT COUNT(*) AS count FROM " + TABLE_NAME + " WHERE KYO_TRK_NO = :kyoTrkNo");
        query.setParameter("kyoTrkNo", safeKyoTrkNo);
        return ((Number) query.getSingleResult()).longValue();
    }

    @Override
    public Optional<KyoteldRecord> findByKyoTrkNoAndKyoTelNo(Long kyoTrkNo, String kyoTelNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        var builder = criteriaBuilderFactory.create(entityManager, KyoteldEntity.class);
        builder.where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .where("kyoTelNo")
                .eq(safeKyoTelNo);
        builder.setMaxResults(1);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            KyoteldWriteRecord record,
            Set<KyoteldColumn> targetColumns,
            KyoteldUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        BigDecimal safeKyoTrkNo = safeLong(updateKey.kyoTrkNo(), "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(updateKey.kyoTelNo());
        Set<KyoteldColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        return updateSupport.update(
                record,
                canonicalTargetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        PK_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeKyoTrkNo);
                            return JpaBind.set(query, nextIndex, safeKyoTelNo);
                        },
                        safeKyoTrkNo,
                        safeKyoTelNo),
                updateKey.recUpdJnjNo(),
                KyoteldDataAccess.Identifiers.UPDATE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", 13);
    }

    private static Set<KyoteldColumn> canonicalTargetColumns(Set<KyoteldColumn> targetColumns) {
        EnumSet<KyoteldColumn> canonicalColumns = EnumSet.noneOf(KyoteldColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }
}
