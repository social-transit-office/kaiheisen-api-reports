package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kbagmst;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kbagmst.KbagmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kbagmst.KbagmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** KBAGMST データアクセスの JPA 実装。 */
public class KbagmstDataAccessImpl extends DataAccessBase implements KbagmstDataAccess {

    private static final ReflectiveRecordEntityMapper<KbagmstRecord, KbagmstEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(KbagmstRecord.class, KbagmstEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public KbagmstDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<KbagmstRecord> findByGasMtstNoAndTkysYmAndTyenYm(
            Long gasMtstNo, YearMonth knskKjnYm) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(knskKjnYm, "knskKjnYm");

        var knskKjnYmValue = JpaDbWriteValues.bigDecimal(knskKjnYm.toYmInt());
        var builder = criteriaBuilderFactory.create(entityManager, KbagmstEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", JpaDbWriteValues.bigDecimal(gasMtstNo));
        JpaWhere.le(builder, "tkysYm", knskKjnYmValue);
        JpaWhere.ge(builder, "tyenYm", knskKjnYmValue);
        builder.orderByDesc("tkysYm");
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<KbagmstRecord> findByPK(Long gasMtstNo, YearMonth tkysYm) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(tkysYm, "tkysYm");

        KbagmstId id =
                new KbagmstId(
                        JpaDbWriteValues.bigDecimal(gasMtstNo),
                        JpaDbWriteValues.bigDecimal(tkysYm.toYmInt()));
        KbagmstEntity entity = entityManager.find(KbagmstEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }
}
