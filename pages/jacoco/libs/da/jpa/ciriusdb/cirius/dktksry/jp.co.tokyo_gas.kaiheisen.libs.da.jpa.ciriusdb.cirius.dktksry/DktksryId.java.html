package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.dktksry;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 導管通知確定使用量テーブルの複合主キー。 */
public class DktksryId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal knsYmd;

    private BigDecimal knsKkaTsno;

    /** デフォルトコンストラクタ。 */
    public DktksryId() { }

    /** 全フィールド指定コンストラクタ。 */
    public DktksryId(BigDecimal gasMtstNo, BigDecimal knsYmd, BigDecimal knsKkaTsno) {
        this.gasMtstNo = gasMtstNo;
        this.knsYmd = knsYmd;
        this.knsKkaTsno = knsKkaTsno;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** knsYmdを取得する。 */
    public BigDecimal getKnsYmd() {
        return knsYmd;
    }

    /** knsYmdを設定する。 */
    public void setKnsYmd(BigDecimal knsYmd) {
        this.knsYmd = knsYmd;
    }

    /** knsKkaTsnoを取得する。 */
    public BigDecimal getKnsKkaTsno() {
        return knsKkaTsno;
    }

    /** knsKkaTsnoを設定する。 */
    public void setKnsKkaTsno(BigDecimal knsKkaTsno) {
        this.knsKkaTsno = knsKkaTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof DktksryId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo)
                && Objects.equals(knsYmd, that.knsYmd)
                && Objects.equals(knsKkaTsno, that.knsKkaTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, knsYmd, knsKkaTsno);
    }
}
