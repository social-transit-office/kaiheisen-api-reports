/** DKTKSRY データアクセスの JPA 実装。 */
public class DktksryDataAccessImpl extends DataAccessBase implements DktksryDataAccess {

    private static final String TABLE_NAME = "DKTKSRY";

    private static final String U001_WHERE_CLAUSE =
            "GAS_MTST_NO = ? AND KNS_YMD = ? AND KNS_KKA_TSNO = ?";

    private static final ReflectiveRecordEntityMapper<DktksryRecord, DktksryEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(DktksryRecord.class, DktksryEntity.class);

    private static final ReflectiveRecordEntityMapper<DktksryWriteRecord, DktksryEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            DktksryWriteRecord.class, DktksryEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<DktksryEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<DktksryWriteRecord, DktksryEntity, DktksryColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<DktksryWriteRecord, DktksryEntity, DktksryColumn>
            updateSupport;

    /**
     * JPA ベースの DKTKSRY データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public DktksryDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    DktksryDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    DktksryDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(DktksryEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        DktksryWriteDefinition.INSTANCE,
                        DktksryEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        DktksryWriteDefinition.INSTANCE,
                        DktksryEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(DktksryWriteRecord record, AuditContext context) {
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno, Integer recUpdJnjNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        BigDecimal safeKnsKkaTsno = safeInt(knsKkaTsno, "knsKkaTsno");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GAS_MTST_NO = ? AND KNS_YMD = ? AND KNS_KKA_TSNO = ?",
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(query, startIndex, safeGasMtstNo);
                                    nextIndex =
                                            JpaBind.set(query, nextIndex, safeKnsYmd);
                                    return JpaBind.set(query, nextIndex, safeKnsKkaTsno);
                                },
                                safeGasMtstNo,
                                safeKnsYmd,
                                safeKnsKkaTsno),
                        recUpdJnjNo,
                        DktksryDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO.identifier());
    }

    @Override
    public List<DktksryRecord> findAll(Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        BigDecimal safeKnsKkaTsno = safeInt(knsKkaTsno, "knsKkaTsno");
        SqlCapture.set(
                "SELECT * FROM DKTKSRY WHERE GAS_MTST_NO = :gasMtstNo AND KNS_YMD = :knsYmd AND"
                    + " KNS_KKA_TSNO = :knsKkaTsno");
        var builder = criteriaBuilderFactory.create(entityManager, DktksryEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .eq(safeKnsYmd)
                .where("knsKkaTsno")
                .eq(safeKnsKkaTsno);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmd(Long gasMtstNo, LocalDate knsYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        SqlCapture.set(
                "SELECT * FROM DKTKSRY WHERE GAS_MTST_NO = :gasMtstNo AND KNS_YMD >= :knsYmd ORDER"
                    + " BY KNS_YMD DESC FETCH FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, DktksryEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .ge(safeKnsYmd)
                .orderByDesc("knsYmd");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
            Long gasMtstNo, LocalDate knsYmd, String knkkSksiJysb) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        String safeKnkkSksiJysb =
                WhereParamGuard.fixedLengthNotNull(knkkSksiJysb, "knkkSksiJysb", 3);
        SqlCapture.set(
                "SELECT * FROM DKTKSRY WHERE GAS_MTST_NO = :gasMtstNo AND KNS_YMD = :knsYmd AND"
                    + " KNKK_SKSI_JYSB = :knkkSksiJysb ORDER BY KNS_KKA_TSNO DESC FETCH FIRST 1"
                    + " ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, DktksryEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .eq(safeKnsYmd)
                .where("knkkSksiJysb")
                .eq(safeKnkkSksiJysb)
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdRange(
            Long gasMtstNo, LocalDate knsYmdFrom, LocalDate knsYmdTo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        SqlCapture.set(
                "SELECT * FROM DKTKSRY WHERE GAS_MTST_NO = :gasMtstNo AND KNS_YMD >= :knsYmdFrom"
                    + " AND KNS_YMD <= :knsYmdTo ORDER BY KNS_YMD DESC, KNS_KKA_TSNO DESC"
                    + " FETCH FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, DktksryEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .ge(safeKnsYmdFrom)
                .where("knsYmd")
                .le(safeKnsYmdTo)
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
            Long gasMtstNo, LocalDate knsYmdFrom, LocalDate knsYmdTo, String knkkSksiJysb) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        String safeKnkkSksiJysb =
                WhereParamGuard.fixedLengthNotNull(knkkSksiJysb, "knkkSksiJysb", 3);
        SqlCapture.set(
                "SELECT * FROM DKTKSRY WHERE GAS_MTST_NO = :gasMtstNo AND KNS_YMD >= :knsYmdFrom"
                    + " AND KNS_YMD <= :knsYmdTo AND KNKK_SKSI_JYSB = :knkkSksiJysb ORDER BY"
                    + " KNS_YMD DESC, KNS_KKA_TSNO DESC FETCH FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, DktksryEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .ge(safeKnsYmdFrom)
                .where("knsYmd")
                .le(safeKnsYmdTo)
                .where("knkkSksiJysb")
                .eq(safeKnkkSksiJysb)
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            DktksryWriteRecord record,
            Set<DktksryColumn> targetColumns,
            DktksryUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo =
                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        BigDecimal safeGasMtstNo = safeLong(updateKey.gasMtstNo(), "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(updateKey.knsYmd(), "knsYmd");
        BigDecimal safeKnsKkaTsno = safeInt(updateKey.knsKkaTsno(), "knsKkaTsno");

        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        U001_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(query, startIndex, safeGasMtstNo);
                            nextIndex =
                                    JpaBind.set(query, nextIndex, safeKnsYmd);
                            return JpaBind.set(query, nextIndex, safeKnsKkaTsno);
                        },
                        safeGasMtstNo,
                        safeKnsYmd,
                        safeKnsKkaTsno),
                updateKey.recUpdJnjNo(),
                DktksryDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO.identifier());
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
