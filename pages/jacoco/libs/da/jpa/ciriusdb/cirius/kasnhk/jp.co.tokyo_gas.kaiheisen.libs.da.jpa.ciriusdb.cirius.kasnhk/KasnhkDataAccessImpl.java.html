/** KASNHK データアクセスの JPA 実装。 */
public class KasnhkDataAccessImpl extends DataAccessBase implements KasnhkDataAccess {

    private static final String TABLE_NAME = "KASNHK";

    private static final int GMB_LENGTH = 15;

    private static final int LVLT_ANKN_NO_LENGTH = 10;

    private static final int HUKK_SYS_SHBT_LENGTH = 1;

    private static final int KAS_HUKK_JTKB_LENGTH = 1;

    private static final TableWriteDefinition<KasnhkWriteRecord, KasnhkColumn> WRITE_DEFINITION =
            KasnhkWriteDefinition.INSTANCE;

    private static final String PK_WHERE_CLAUSE =
            "GMB = ? AND KASG_YMD = ? AND KAS_HUKK_YMD = ? AND HUKK_SYS_SHBT = ? AND LVLT_ANKN_NO = ?";

    private static final ReflectiveRecordEntityMapper<KasnhkRecord, KasnhkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KasnhkRecord.class, KasnhkEntity.class);

    private static final ReflectiveRecordEntityMapper<KasnhkWriteRecord, KasnhkEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(KasnhkWriteRecord.class, KasnhkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KasnhkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KasnhkWriteRecord, KasnhkEntity, KasnhkColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KasnhkWriteRecord, KasnhkEntity, KasnhkColumn>
            updateSupport;

    /** JPA ベースの KASNHK データアクセスを生成する。 */
    public KasnhkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KasnhkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KasnhkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KasnhkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                WRITE_DEFINITION,
                        KasnhkEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                WRITE_DEFINITION,
                        KasnhkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    /** 1 行登録する。 */
    @Override
    public int insert(KasnhkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    /** 4条件と任意排他制御順序番号で一致行を削除する。 */
    @Override
    public int deleteByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNo(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            Integer recUpdJnjNo) {
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        preparedWhereBy4Conditions(
                                safeGmb(gmb),
                                safeYmd(kasgYmd, "kasgYmd"),
                                safeYmd(kasHukkYmd, "kasHukkYmd"),
                                safeLvltAnknNo(lvltAnknNo)),
                        recUpdJnjNo,
                        KasnhkDataAccess.CallSpecs
                                .DELETE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_LVLT_ANKN_NO
                                .identifier());
    }

    /** 開栓報告状態区分を単一値で指定して 1 件取得する。 */
    @Override
    public Optional<KasnhkRecord> findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            String kasHukkJtkb) {
        var builder = criteriaBuilderFactory.create(entityManager, KasnhkEntity.class);
        JpaWhere.eq(builder, "gmb", safeGmb(gmb));
        JpaWhere.eq(builder, "kasgYmd", safeYmd(kasgYmd, "kasgYmd"));
        JpaWhere.eq(builder, "kasHukkYmd", safeYmd(kasHukkYmd, "kasHukkYmd"));
        JpaWhere.eq(builder, "lvltAnknNo", safeLvltAnknNo(lvltAnknNo));
        JpaWhere.eq(builder, "kasHukkJtkb", safeKasHukkJtkb(kasHukkJtkb, "kasHukkJtkb"));
        SqlCapture.set(builder.getQueryString());
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    /** 開栓報告状態区分を複数値で指定して 1 件取得する。 */
    @Override
    public Optional<KasnhkRecord> findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            List<String> kasHukkJtkbList) {
        List<String> safeKasHukkJtkbList = safeKasHukkJtkbList(kasHukkJtkbList);
        if (safeKasHukkJtkbList.isEmpty()) {
            return Optional.empty();
        }
        var builder = criteriaBuilderFactory.create(entityManager, KasnhkEntity.class);
        JpaWhere.eq(builder, "gmb", safeGmb(gmb));
        JpaWhere.eq(builder, "kasgYmd", safeYmd(kasgYmd, "kasgYmd"));
        JpaWhere.eq(builder, "kasHukkYmd", safeYmd(kasHukkYmd, "kasHukkYmd"));
        JpaWhere.eq(builder, "lvltAnknNo", safeLvltAnknNo(lvltAnknNo));
        builder.where("kasHukkJtkb").in(safeKasHukkJtkbList);
        SqlCapture.set(builder.getQueryString());
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    /** 主キー検索の実処理を行う。 */
    @Override
    public Optional<KasnhkRecord> findByPrimaryKey(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String hukkSysShbt,
            String lvltAnknNo) {
        KasnhkId id = new KasnhkId(
                                safeGmb(gmb),
                                safeYmd(kasgYmd, "kasgYmd"),
                                safeYmd(kasHukkYmd, "kasHukkYmd"),
                                safeHukkSysShbt(hukkSysShbt),
                                safeLvltAnknNo(lvltAnknNo));
        KasnhkEntity entity = entityManager.find(KasnhkEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    /** 報告システム種別を含む条件の対象行を更新する。 */
    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndHukkSysShbtAndLvltAnknNo(
            KasnhkWriteRecord record,
            Set<KasnhkUpdatableColumn> targetColumns,
            KasnhkU001UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        String safeGmb = safeGmb(updateKey.gmb());
        BigDecimal safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");
        BigDecimal safeKasHukkYmd = safeYmd(updateKey.kasHukkYmd(), "kasHukkYmd");
        String safeHukkSysShbt = safeHukkSysShbt(updateKey.hukkSysShbt());
        String safeLvltAnknNo = safeLvltAnknNo(updateKey.lvltAnknNo());
        return updateSupport.update(
                record,
                canonicalTargetColumns(targetColumns),
                (entity, ignoredColumns) -> {
                    Integer currentRecUpdJnjNo = updateKey.recUpdJnjNo();
                    if (currentRecUpdJnjNo == null) {
                        currentRecUpdJnjNo =
                                entity.getRecUpdJnjNo() == null
                                        ? 0
                                        : entity.getRecUpdJnjNo().intValueExact();
                    }
                    auditColumnAutoSetter.applyForUpdate(
                            auditColumnAccessBinder.bind(entity),
                            context,
                            clock,
                            currentRecUpdJnjNo);
                },
                preparedWhere(
                        safeGmb,
                        safeKasgYmd,
                        safeKasHukkYmd,
                        safeHukkSysShbt,
                        safeLvltAnknNo),
                updateKey.recUpdJnjNo(),
                KasnhkDataAccess.CallSpecs
                        .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_HUKK_SYS_SHBT_AND_LVLT_ANKN_NO_U001
                        .identifier());
    }

    /** 4条件と任意排他で対象行を更新する。 */
    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndRecUpdJnjNo(
            KasnhkWriteRecord record,
            Set<KasnhkUpdatableColumn> targetColumns,
            KasnhkU002UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        String safeGmb = safeGmb(updateKey.gmb());
        BigDecimal safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");
        BigDecimal safeKasHukkYmd = safeYmd(updateKey.kasHukkYmd(), "kasHukkYmd");
        String safeLvltAnknNo = safeLvltAnknNo(updateKey.lvltAnknNo());
        return updateSupport.update(
                record,
                canonicalTargetColumns(targetColumns),
                (entity, ignoredColumns) -> {
                    Integer currentRecUpdJnjNo = updateKey.recUpdJnjNo();
                    if (currentRecUpdJnjNo == null) {
                        currentRecUpdJnjNo =
                                entity.getRecUpdJnjNo() == null
                                        ? 0
                                        : entity.getRecUpdJnjNo().intValueExact();
                    }
                    auditColumnAutoSetter.applyForUpdate(
                            auditColumnAccessBinder.bind(entity),
                            context,
                            clock,
                            currentRecUpdJnjNo);
                },
                preparedWhereBy4Conditions(safeGmb, safeKasgYmd, safeKasHukkYmd, safeLvltAnknNo),
                updateKey.recUpdJnjNo(),
                KasnhkDataAccess.CallSpecs
                        .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_LVLT_ANKN_NO_AND_REC_UPD_JNJ_NO_U002
                        .identifier());
    }

    /** 更新対象列を EnumSet に正規化する。 */
    private static Set<KasnhkColumn> canonicalTargetColumns(
            Set<KasnhkUpdatableColumn> targetColumns) {
        EnumSet<KasnhkColumn> canonicalColumns = EnumSet.noneOf(KasnhkColumn.class);
        for (KasnhkUpdatableColumn targetColumn : targetColumns) {
            canonicalColumns.add(KasnhkColumn.valueOf(targetColumn.name()));
        }
        return canonicalColumns;
    }

    /** 主キー条件の WHERE 定義を生成する。 */
    private static PreparedWhere preparedWhere(
            String safeGmb,
            BigDecimal safeKasgYmd,
            BigDecimal safeKasHukkYmd,
            String safeHukkSysShbt,
            String safeLvltAnknNo) {
        return PreparedWhere.of(
                PK_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeGmb);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasgYmd);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasHukkYmd);
                    nextIndex = JpaBind.set(query, nextIndex, safeHukkSysShbt);
                    return JpaBind.set(query, nextIndex, safeLvltAnknNo);
                },
                safeGmb,
                safeKasgYmd,
                safeKasHukkYmd,
                safeHukkSysShbt,
                safeLvltAnknNo);
    }

    /** HUKK_SYS_SHBT を含まない4条件 WHERE 定義を生成する。 */
    private static PreparedWhere preparedWhereBy4Conditions(
            String safeGmb,
            BigDecimal safeKasgYmd,
            BigDecimal safeKasHukkYmd,
            String safeLvltAnknNo) {
        return PreparedWhere.of(
                "GMB = ? AND KASG_YMD = ? AND KAS_HUKK_YMD = ? AND LVLT_ANKN_NO = ?",
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeGmb);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasgYmd);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasHukkYmd);
                    return JpaBind.set(query, nextIndex, safeLvltAnknNo);
                },
                safeGmb,
                safeKasgYmd,
                safeKasHukkYmd,
                safeLvltAnknNo);
    }

    /** GMB を WHERE 用に正規化する。 */
    private static String safeGmb(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "gmb", GMB_LENGTH);
    }

    /** LIV@LIT 案件番号を WHERE 用に正規化する。 */
    private static String safeLvltAnknNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "lvltAnknNo", LVLT_ANKN_NO_LENGTH);
    }

    /** 報告システム種別を WHERE 用に正規化する。 */
    private static String safeHukkSysShbt(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "hukkSysShbt", HUKK_SYS_SHBT_LENGTH);
    }

    /** 開栓報告状態区分を WHERE 用に正規化する。 */
    private static String safeKasHukkJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, KAS_HUKK_JTKB_LENGTH);
    }

    /** 開栓報告状態区分リストを WHERE 用に正規化する。 */
    private static List<String> safeKasHukkJtkbList(List<String> values) {
        if (values == null) {
            throw new IllegalArgumentException("kasHukkJtkbList must not be null");
        }
        if (values.isEmpty()) {
            return List.of();
        }
        return values.stream().map(KasnhkDataAccessImpl::normalizeInKasHukkJtkb).toList();
    }

    /** IN 句要素の開栓報告状態区分を固定長条件値へ正規化する。 */
    private static String normalizeInKasHukkJtkb(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "kasHukkJtkbList must not contain null or blank-only");
        }
        return safeKasHukkJtkb(value, "kasHukkJtkbList");
    }

    /** 年月日を WHERE 用に数値化する。 */
    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
