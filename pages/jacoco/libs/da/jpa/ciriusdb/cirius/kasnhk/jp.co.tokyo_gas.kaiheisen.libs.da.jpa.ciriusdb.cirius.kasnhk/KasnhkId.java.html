package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kasnhk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 開栓報告（KASNHK）の複合主キー。 */
public class KasnhkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String gmb;

    private BigDecimal kasgYmd;

    private BigDecimal kasHukkYmd;

    private String hukkSysShbt;

    private String lvltAnknNo;

    /** デフォルトコンストラクタ。 */
    public KasnhkId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KasnhkId(
            String gmb,
            BigDecimal kasgYmd,
            BigDecimal kasHukkYmd,
            String hukkSysShbt,
            String lvltAnknNo) {
        this.gmb = gmb;
        this.kasgYmd = kasgYmd;
        this.kasHukkYmd = kasHukkYmd;
        this.hukkSysShbt = hukkSysShbt;
        this.lvltAnknNo = lvltAnknNo;
    }

    /** gmbを取得する。 */
    public String getGmb() {
        return gmb;
    }

    /** gmbを設定する。 */
    public void setGmb(String gmb) {
        this.gmb = gmb;
    }

    /** kasgYmdを取得する。 */
    public BigDecimal getKasgYmd() {
        return kasgYmd;
    }

    /** kasgYmdを設定する。 */
    public void setKasgYmd(BigDecimal kasgYmd) {
        this.kasgYmd = kasgYmd;
    }

    /** kasHukkYmdを取得する。 */
    public BigDecimal getKasHukkYmd() {
        return kasHukkYmd;
    }

    /** kasHukkYmdを設定する。 */
    public void setKasHukkYmd(BigDecimal kasHukkYmd) {
        this.kasHukkYmd = kasHukkYmd;
    }

    /** hukkSysShbtを取得する。 */
    public String getHukkSysShbt() {
        return hukkSysShbt;
    }

    /** hukkSysShbtを設定する。 */
    public void setHukkSysShbt(String hukkSysShbt) {
        this.hukkSysShbt = hukkSysShbt;
    }

    /** lvltAnknNoを取得する。 */
    public String getLvltAnknNo() {
        return lvltAnknNo;
    }

    /** lvltAnknNoを設定する。 */
    public void setLvltAnknNo(String lvltAnknNo) {
        this.lvltAnknNo = lvltAnknNo;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KasnhkId that)) {
            return false;
        }
        return Objects.equals(gmb, that.gmb)
                && Objects.equals(kasgYmd, that.kasgYmd)
                && Objects.equals(kasHukkYmd, that.kasHukkYmd)
                && Objects.equals(hukkSysShbt, that.hukkSysShbt)
                && Objects.equals(lvltAnknNo, that.lvltAnknNo);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gmb, kasgYmd, kasHukkYmd, hukkSysShbt, lvltAnknNo);
    }
}
