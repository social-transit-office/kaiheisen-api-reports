package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.grksbhw;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** ガス料金契約種別振分（GRKSBHW）の複合主キー。 */
public class GrksbhwId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String grkKsbHwsb;

    private String grkKiyShbt;

    private String rknChkCd;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public GrksbhwId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public GrksbhwId(String grkKsbHwsb, String grkKiyShbt, String rknChkCd, BigDecimal tkysYmd) {
        this.grkKsbHwsb = grkKsbHwsb;
        this.grkKiyShbt = grkKiyShbt;
        this.rknChkCd = rknChkCd;
        this.tkysYmd = tkysYmd;
    }

    /** ガス料金契約種別振分種別を返す。 */
    public String getGrkKsbHwsb() {
        return grkKsbHwsb;
    }

    /** ガス料金契約種別振分種別を設定する。 */
    public void setGrkKsbHwsb(String grkKsbHwsb) {
        this.grkKsbHwsb = grkKsbHwsb;
    }

    /** ガス料金契約種別を返す。 */
    public String getGrkKiyShbt() {
        return grkKiyShbt;
    }

    /** ガス料金契約種別を設定する。 */
    public void setGrkKiyShbt(String grkKiyShbt) {
        this.grkKiyShbt = grkKiyShbt;
    }

    /** 料金地区コードを返す。 */
    public String getRknChkCd() {
        return rknChkCd;
    }

    /** 料金地区コードを設定する。 */
    public void setRknChkCd(String rknChkCd) {
        this.rknChkCd = rknChkCd;
    }

    /** 適用開始年月日を返す。 */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /** 適用開始年月日を設定する。 */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof GrksbhwId that)) {
            return false;
        }
        return Objects.equals(grkKsbHwsb, that.grkKsbHwsb)
                && Objects.equals(grkKiyShbt, that.grkKiyShbt)
                && Objects.equals(rknChkCd, that.rknChkCd)
                && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(grkKsbHwsb, grkKiyShbt, rknChkCd, tkysYmd);
    }
}
