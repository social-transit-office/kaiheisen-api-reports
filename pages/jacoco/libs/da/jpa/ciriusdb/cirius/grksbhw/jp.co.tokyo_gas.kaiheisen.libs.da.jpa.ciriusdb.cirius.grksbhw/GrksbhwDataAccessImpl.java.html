package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.grksbhw;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GRKSBHW データアクセスの JPA 実装。 */
public class GrksbhwDataAccessImpl extends DataAccessBase implements GrksbhwDataAccess {

    private static final int GRK_KSB_HWSB_LENGTH = 4;

    private static final int GRK_KIY_SHBT_LENGTH = 5;

    private static final int RKN_CHK_CD_LENGTH = 1;

    private static final ReflectiveRecordEntityMapper<GrksbhwRecord, GrksbhwEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(GrksbhwRecord.class, GrksbhwEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public GrksbhwDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<GrksbhwRecord> findAllByGrkKsbHwsbListAndGrkKiyShbtAndKnskKjnYmd(
            List<String> grkKsbHwsbList,
            String grkKiyShbt,
            LocalDate knskKjnYmd) {
        List<String> safeGrkKsbHwsbList = normalizeGrkKsbHwsbList(grkKsbHwsbList);
        if (safeGrkKsbHwsbList.isEmpty()) {
            return List.of();
        }
        String safeGrkKiyShbt = safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GrksbhwEntity.class);
        JpaWhere.in(builder, "grkKsbHwsb", safeGrkKsbHwsbList);
        JpaWhere.eq(builder, "grkKiyShbt", safeGrkKiyShbt);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByAsc("grkKsbHwsb")
                .orderByAsc("grkKiyShbt")
                .orderByAsc("rknChkCd")
                .orderByDesc("tkysYmd");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public Optional<GrksbhwRecord> findByGrkKiyShbtAndGrkKsbHwsbAndRknChkCdAndKnskKjnYmdOrderByTkysYmdDesc(
            String grkKiyShbt,
            String grkKsbHwsb,
            String rknChkCd,
            LocalDate knskKjnYmd) {
        String safeGrkKiyShbt = safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt");
        String safeGrkKsbHwsb = safeGrkKsbHwsb(grkKsbHwsb, "grkKsbHwsb");
        String safeRknChkCd = safeRknChkCd(rknChkCd, "rknChkCd");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, GrksbhwEntity.class);
        JpaWhere.eq(builder, "grkKiyShbt", safeGrkKiyShbt);
        JpaWhere.eq(builder, "grkKsbHwsb", safeGrkKsbHwsb);
        JpaWhere.eq(builder, "rknChkCd", safeRknChkCd);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<GrksbhwRecord> findByPK(
            String grkKsbHwsb,
            String grkKiyShbt,
            String rknChkCd,
            LocalDate tkysYmd) {
        GrksbhwId id = new GrksbhwId(
                safeGrkKsbHwsb(grkKsbHwsb, "grkKsbHwsb"),
                safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt"),
                safeRknChkCd(rknChkCd, "rknChkCd"),
                safeYmd(tkysYmd, "tkysYmd"));
        GrksbhwEntity entity = entityManager.find(GrksbhwEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    private static String safeGrkKsbHwsb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GRK_KSB_HWSB_LENGTH);
    }

    private static String safeGrkKiyShbt(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GRK_KIY_SHBT_LENGTH);
    }

    private static String safeRknChkCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, RKN_CHK_CD_LENGTH);
    }

    private static List<String> normalizeGrkKsbHwsbList(List<String> grkKsbHwsbList) {
        if (grkKsbHwsbList == null) {
            throw new IllegalArgumentException("grkKsbHwsbList must not be null");
        }
        if (grkKsbHwsbList.isEmpty()) {
            return List.of();
        }
        return grkKsbHwsbList.stream().map(GrksbhwDataAccessImpl::normalizeInGrkKsbHwsb).toList();
    }

    private static String normalizeInGrkKsbHwsb(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "grkKsbHwsbList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "grkKsbHwsbList", GRK_KSB_HWSB_LENGTH);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
