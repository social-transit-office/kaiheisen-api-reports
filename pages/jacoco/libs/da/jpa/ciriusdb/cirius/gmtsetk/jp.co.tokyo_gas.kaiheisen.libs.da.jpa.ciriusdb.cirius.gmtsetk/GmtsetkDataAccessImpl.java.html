package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gmtsetk;

/** GMTSETK データアクセスの JPA 実装。 */
public class GmtsetkDataAccessImpl extends DataAccessBase implements GmtsetkDataAccess {

    private static final String TABLE_NAME = "GMTSETK";

    private static final String U001_BASE_WHERE_CLAUSE = "GAS_MTST_NO = ? AND TKYS_YMD = ?";

    private static final String U002_WHERE_CLAUSE = "GAS_MTST_NO = ? AND TYEN_YMD = ?";

    private static final ReflectiveRecordEntityMapper<GmtsetkRecord, GmtsetkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(GmtsetkRecord.class, GmtsetkEntity.class);

    private static final ReflectiveRecordEntityMapper<GmtsetkWriteRecord, GmtsetkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            GmtsetkWriteRecord.class, GmtsetkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<GmtsetkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<GmtsetkWriteRecord, GmtsetkEntity, GmtsetkColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<GmtsetkWriteRecord, GmtsetkEntity, GmtsetkColumn>
            updateSupport;

    /**
     * JPA ベースの GMTSETK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     */
    public GmtsetkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    GmtsetkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    GmtsetkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(GmtsetkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        GmtsetkWriteDefinition.INSTANCE,
                        GmtsetkEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        GmtsetkWriteDefinition.INSTANCE,
                        GmtsetkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(GmtsetkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<GmtsetkRecord> findByGasMtstNoAndTkysYmd(Long gasMtstNo, LocalDate tkysYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        GmtsetkId id = new GmtsetkId(safeGasMtstNo, safeTkysYmd);
        GmtsetkEntity entity = entityManager.find(GmtsetkEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public Optional<GmtsetkRecord> findByGasMtstNoAndTkysYmdAndTyenYmd(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, GmtsetkEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
            GmtsetkWriteRecord record,
            Set<GmtsetkColumn> targetColumns,
            GmtsetkTkysJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                PreparedWhere.of(
                        U001_BASE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(
                                            query,
                                            startIndex,
                                            updateKey.gasMtstNo(),
                                            JpaDbWriteValues::bigDecimal);
                            nextIndex =
                                    JpaBind.set(
                                            query,
                                            nextIndex,
                                            updateKey.tkysYmd(),
                                            JpaDbWriteValues::ymd);
                            return nextIndex;
                        },
                        JpaDbWriteValues.bigDecimal(updateKey.gasMtstNo()),
                        JpaDbWriteValues.ymd(updateKey.tkysYmd())),
                updateKey.recUpdJnjNo(),
                GmtsetkDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO);
    }

    @Override
    public int updateByGasMtstNoAndTyenYmd(
            GmtsetkWriteRecord record,
            Set<GmtsetkColumn> targetColumns,
            GmtsetkTyenUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo =
                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        U002_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(
                                            query,
                                            startIndex,
                                            updateKey.gasMtstNo(),
                                            JpaDbWriteValues::bigDecimal);
                            return JpaBind.set(
                                    query, nextIndex, updateKey.tyenYmd(), JpaDbWriteValues::ymd);
                        },
                        JpaDbWriteValues.bigDecimal(updateKey.gasMtstNo()),
                        JpaDbWriteValues.ymd(updateKey.tyenYmd())),
                updateKey.recUpdJnjNo(),
                GmtsetkDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_TYEN_YMD);
    }

    @Override
    public int deleteByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
            Long gasMtstNo, LocalDate tkysYmd, Integer recUpdJnjNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                U001_BASE_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    query.setParameter(startIndex, safeGasMtstNo);
                                    query.setParameter(startIndex + 1, safeTkysYmd);
                                    return startIndex + 2;
                                },
                                safeGasMtstNo,
                                safeTkysYmd),
                        recUpdJnjNo,
                        GmtsetkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return JpaDbWriteValues.bigDecimal(WhereParamGuard.complement(value, paramName, 0L));
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

}
