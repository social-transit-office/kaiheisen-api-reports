        String safeRkkiKnrnJtkb = safeRkkiKnrnJtkb(rkkiKnrnJtkb, "rkkiKnrnJtkb");
        return entityManager
                .createNativeQuery(
                        """
                        SELECT MTST_NO, SVKY_NO, SKNO, SUGU_KBN, SKNR_FLG,
                               RKKI_KNRN_JTKB, RKKI_STT_YMD, RKKI_END_YMD, HISH_YTI_YMD,
                               REG_TSMP, REG_USR_ID, UPD_TSMP, UPD_USR_ID, UPD_KTN_KINO_ID,
                               REC_UPD_JNJ_NO
                          FROM SVMSTKR
                         WHERE MTST_NO = :mtstNo
                           AND RKKI_STT_YMD <= :rkkiSttYmd
                           AND RKKI_KNRN_JTKB <> :rkkiKnrnJtkb
                         ORDER BY RKKI_STT_YMD DESC
                        """)
                    .setParameter("mtstNo", safeMtstNo)
                    .setParameter("rkkiSttYmd", safeRkkiSttYmd)
                    .setParameter("rkkiKnrnJtkb", safeRkkiKnrnJtkb)
                .setMaxResults(1)
                .getResultList()
                .stream()
                .map(result -> toRecordFromNativeRow((Object[]) result))
                .findFirst();
    }

    private static SvmstkrRecord toRecordFromNativeRow(Object[] row) {
        return new SvmstkrRecord(
                JpaDbReadValues.longOrNullIfZero((BigDecimal) row[0]),
                JpaDbReadValues.longOrNullIfZero((BigDecimal) row[1]),
                JpaDbReadValues.integerOrNullIfZero((BigDecimal) row[2]),
                stringValue(row[3]),
                StringBooleanFlagConverter.INSTANCE.convert(stringValue(row[4])),
                stringValue(row[5]),
                JpaDbReadValues.localDate((BigDecimal) row[6]),
                JpaDbReadValues.localDate((BigDecimal) row[7]),
                JpaDbReadValues.localDate((BigDecimal) row[8]),
                toLocalDateTime(row[9]),
                stringValue(row[10]),
                toLocalDateTime(row[11]),
                stringValue(row[12]),
                stringValue(row[13]),
                JpaDbReadValues.integerOrNullIfZero((BigDecimal) row[14]));
    }

    private static String stringValue(Object value) {
        if (value == null) {
            return null;
        }
        return FixedLengthFormatter.suppressSpace(String.valueOf(value));
    }

    private static LocalDateTime toLocalDateTime(Object value) {
        if (value == null) {
            return null;
        }
        if (value instanceof LocalDateTime localDateTime) {
            return localDateTime;
        }
        if (value instanceof Timestamp timestamp) {
            return timestamp.toLocalDateTime();
        }
        throw new IllegalArgumentException("Unsupported timestamp type: " + value.getClass().getName());
    }

            SvmstkrWriteRecord record,

    private Optional<ResolvedUpdateTarget> resolvePrimaryKeyTarget(
            Long mtstNo, Long svkyNo, Integer skno) {
        Objects.requireNonNull(mtstNo, "mtstNo");
        Objects.requireNonNull(svkyNo, "svkyNo");
        Objects.requireNonNull(skno, "skno");

        SvmstkrId id =
                new SvmstkrId(
                        JpaDbWriteValues.bigDecimal(mtstNo),
                        JpaDbWriteValues.bigDecimal(svkyNo),
                        JpaDbWriteValues.bigDecimal(skno));
        SvmstkrEntity entity = entityManager.find(SvmstkrEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        Integer currentRecUpdJnjNo = currentRecUpdJnjNo(entity);
        entityManager.detach(entity);
        return Optional.of(
                new ResolvedUpdateTarget(
                        entity,
                        currentRecUpdJnjNo,
                        preparedWhereByPrimaryKey(entity)));
    }

    private ResolvedOptimisticTargets resolveOptimisticTargets(
            List<SvmstkrEntity> entities, Integer expectedRecUpdJnjNo) {
        Objects.requireNonNull(entities, "entities");

        List<ResolvedUpdateTarget> matchedTargets = new java.util.ArrayList<>();
        ResolvedUpdateTarget firstMismatchedTarget = null;
        for (SvmstkrEntity entity : entities) {
            Integer currentRecUpdJnjNo = currentRecUpdJnjNo(entity);
            entityManager.detach(entity);
            ResolvedUpdateTarget resolvedTarget =
                    new ResolvedUpdateTarget(
                            entity,
                            currentRecUpdJnjNo,
                            preparedWhereByPrimaryKey(entity));
            if (expectedRecUpdJnjNo != null
                    && !Objects.equals(currentRecUpdJnjNo, expectedRecUpdJnjNo)) {
                if (firstMismatchedTarget == null) {
                    firstMismatchedTarget = resolvedTarget;
                }
                continue;
            }
            matchedTargets.add(resolvedTarget);
        }
        return new ResolvedOptimisticTargets(matchedTargets, firstMismatchedTarget);
    }

    private Integer currentRecUpdJnjNo(SvmstkrEntity entity) {
        Objects.requireNonNull(entity, "entity");
        return Objects.requireNonNull(entity.getRecUpdJnjNo(), "entity.recUpdJnjNo")
                .intValueExact();
    }

    }
