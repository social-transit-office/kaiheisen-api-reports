package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.svmstkr;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** SVMSTKR の複合主キー。 */
public class SvmstkrId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal mtstNo;

    private BigDecimal svkyNo;

    private BigDecimal skno;

    /** デフォルトコンストラクタ。 */
    public SvmstkrId() { }

    /** 全フィールド指定コンストラクタ。 */
    public SvmstkrId(BigDecimal mtstNo, BigDecimal svkyNo, BigDecimal skno) {
        this.mtstNo = mtstNo;
        this.svkyNo = svkyNo;
        this.skno = skno;
    }

    /** mtstNoを取得する。 */
    public BigDecimal getMtstNo() {
        return mtstNo;
    }

    /** mtstNoを設定する。 */
    public void setMtstNo(BigDecimal mtstNo) {
        this.mtstNo = mtstNo;
    }

    /** svkyNoを取得する。 */
    public BigDecimal getSvkyNo() {
        return svkyNo;
    }

    /** svkyNoを設定する。 */
    public void setSvkyNo(BigDecimal svkyNo) {
        this.svkyNo = svkyNo;
    }

    /** sknoを取得する。 */
    public BigDecimal getSkno() {
        return skno;
    }

    /** sknoを設定する。 */
    public void setSkno(BigDecimal skno) {
        this.skno = skno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof SvmstkrId that)) {
            return false;
        }
        return Objects.equals(mtstNo, that.mtstNo)
                && Objects.equals(svkyNo, that.svkyNo)
                && Objects.equals(skno, that.skno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(mtstNo, svkyNo, skno);
    }
}
