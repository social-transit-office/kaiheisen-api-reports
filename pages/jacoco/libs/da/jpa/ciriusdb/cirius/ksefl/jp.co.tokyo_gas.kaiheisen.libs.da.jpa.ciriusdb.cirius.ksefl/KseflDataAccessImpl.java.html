    KseflDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KseflEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KseflWriteDefinition.INSTANCE,
                        KseflEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KseflWriteDefinition.INSTANCE,
                        KseflEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KseflWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public List<KseflRecord> findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdAndKasgYmdAndFlwStCd(
            String flttCktkCdFrom,
            String flttCktkCdTo,
            String flttKtnCd,
            String kasErryCd,
            LocalDate kasgYmd,
            String flwStCd) {
        String safeFlttCktkCdFrom =
                WhereParamGuard.fixedLengthNotNull(flttCktkCdFrom, "flttCktkCdFrom", 3);
        String safeFlttCktkCdTo =
                WhereParamGuard.fixedLengthNotNull(flttCktkCdTo, "flttCktkCdTo", 3);
        String safeKasErryCd =
                WhereParamGuard.fixedLengthNotNull(kasErryCd, "kasErryCd", 2);
        BigDecimal safeKasgYmd = safeYmd(kasgYmd, "kasgYmd");
        String safeFlwStCd =
                WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);

        var builder = criteriaBuilderFactory.create(entityManager, KseflEntity.class);
        JpaWhere.between(builder, "flttCktkCd", safeFlttCktkCdFrom, safeFlttCktkCdTo);
        if (flttKtnCd != null) {
            String blankFlttKtnCd =
                    WhereParamGuard.fixedLengthNotNull("", "flttKtnCd", 1);
            JpaWhere.in(
                    builder,
                    "flttKtnCd",
                    List.of(
                            WhereParamGuard.fixedLengthNotNull(flttKtnCd, "flttKtnCd", 1),
                            blankFlttKtnCd));
        }
        JpaWhere.eq(builder, "kasErryCd", safeKasErryCd);
        JpaWhere.ge(builder, "kasgYmd", safeKasgYmd);
        JpaWhere.eq(builder, "flwStCd", safeFlwStCd);
        builder.orderByAsc("kasgYmd")
                .orderByAsc("kasHukkYmd")
                .orderByAsc("kasgNwkCd")
                .orderByAsc("gmb");
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public List<KseflRecord> findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdOrKasErryCdAndKasgYmdAndFlwStCd(
            String flttCktkCdFrom,
            String flttCktkCdTo,
            String flttKtnCd,
            String kasErryCd,
            LocalDate kasgYmd,
            String flwStCd) {
        String safeFlttCktkCdFrom =
                WhereParamGuard.fixedLengthNotNull(flttCktkCdFrom, "flttCktkCdFrom", 3);
        String safeFlttCktkCdTo =
                WhereParamGuard.fixedLengthNotNull(flttCktkCdTo, "flttCktkCdTo", 3);
        String safeKasErryCd =
                WhereParamGuard.fixedLengthNotNull(kasErryCd, "kasErryCd", 2);
        BigDecimal kasgYmdValue = safeYmd(kasgYmd, "kasgYmd");
        String safeFlwStCd =
                WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);

        var builder = criteriaBuilderFactory.create(entityManager, KseflEntity.class);
        JpaWhere.between(builder, "flttCktkCd", safeFlttCktkCdFrom, safeFlttCktkCdTo);
        if (flttKtnCd != null) {
            String blankFlttKtnCd =
                    WhereParamGuard.fixedLengthNotNull("", "flttKtnCd", 1);
            JpaWhere.in(
                    builder,
                    "flttKtnCd",
                    List.of(
                            WhereParamGuard.fixedLengthNotNull(flttKtnCd, "flttKtnCd", 1),
                            blankFlttKtnCd));
        }
        var orBuilder = builder.whereOr();
        orBuilder.where("kasErryCd").notEq(safeKasErryCd);
        orBuilder.whereAnd()
                .where("kasErryCd").eq(safeKasErryCd)
                .where("kasgYmd").le(kasgYmdValue)
                .endAnd();
        orBuilder.endOr();
        JpaWhere.eq(builder, "flwStCd", safeFlwStCd);

        builder.orderByAsc("kasgYmd")
                .orderByAsc("kasHukkYmd")
                .orderByAsc("kasgNwkCd")
                .orderByAsc("gmb");
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public List<KseflRecord> findAllByGmbAndFlwStCdAndKasgYmdAndTsigKasgYmd(
            String gmb, String flwStCd, LocalDate kasgYmd, LocalDate tsigKsgYmd) {
        String safeGmb = WhereParamGuard.fixedLengthNotNull(gmb, "gmb", 15);
        String safeFlwStCd = WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);
        BigDecimal kasgYmdValue = safeYmd(kasgYmd, "kasgYmd");

        BigDecimal tsigKasgYmdValue = safeYmd(tsigKsgYmd, "tsigKsgYmd");
        var builder = criteriaBuilderFactory.create(entityManager, KseflEntity.class);
        JpaWhere.eq(builder, "gmb", safeGmb);
        JpaWhere.eq(builder, "flwStCd", safeFlwStCd);

        // ((KASG_YMD = kasgYmd AND TSIG_KASG_YMD = 0) OR (TSIG_KASG_YMD = tsigKsgYmd AND TSIG_KASG_YMD <> 0))
        var orBuilder = builder.whereOr();
        orBuilder.whereAnd()
                .where("kasgYmd").eq(kasgYmdValue)
                .where("tsigKasgYmd").eq(BigDecimal.ZERO)
                .endAnd();
        orBuilder.whereAnd()
                .where("tsigKasgYmd").eq(tsigKasgYmdValue)
                .where("tsigKasgYmd").notEq(BigDecimal.ZERO)
                .endAnd();
        orBuilder.endOr();

        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override

            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String hukkSysShbt,
            String sgyAnknNo) {
        Objects.requireNonNull(gmb, "gmb");
        Objects.requireNonNull(kasgYmd, "kasgYmd");
        Objects.requireNonNull(kasHukkYmd, "kasHukkYmd");
        Objects.requireNonNull(hukkSysShbt, "hukkSysShbt");
        Objects.requireNonNull(sgyAnknNo, "sgyAnknNo");

        KseflId id = new KseflId(
                gmb,
                JpaDbWriteValues.ymd(kasgYmd),
                JpaDbWriteValues.ymd(kasHukkYmd),
                hukkSysShbt,
                sgyAnknNo);
        KseflEntity entity = entityManager.find(KseflEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndHukkSysShbtAndSgyAnknNoAndRecUpdJnjNo(
            KseflWriteRecord record,
            Set<KseflColumn> targetColumns,
            KseflU001UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        String safeGmb = WhereParamGuard.fixedLengthNotNull(updateKey.gmb(), "gmb", 15);
        BigDecimal safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");
        BigDecimal safeKasHukkYmd = safeYmd(updateKey.kasHukkYmd(), "kasHukkYmd");
        String safeHukkSysShbt =
                WhereParamGuard.fixedLengthNotNull(updateKey.hukkSysShbt(), "hukkSysShbt", 1);
        String safeSgyAnknNo =
                WhereParamGuard.fixedLengthNotNull(updateKey.sgyAnknNo(), "sgyAnknNo", 10);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity), context, clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        U001_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeGmb);
                            nextIndex = JpaBind.set(query, nextIndex, safeKasgYmd);
                            nextIndex = JpaBind.set(query, nextIndex, safeKasHukkYmd);
                            nextIndex = JpaBind.set(query, nextIndex, safeHukkSysShbt);
                            return JpaBind.set(query, nextIndex, safeSgyAnknNo);
                        },
                        safeGmb,
                        safeKasgYmd,
                        safeKasHukkYmd,
                        safeHukkSysShbt,
                        safeSgyAnknNo),
                updateKey.recUpdJnjNo(),
                KseflDataAccess.CallSpecs
                        .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_HUKK_SYS_SHBT_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO
                        .identifier());
    }

    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNoAndRecUpdJnjNo(

            Set<KseflColumn> targetColumns,
            KseflU002UpdateKey updateKey,

        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return updateSupport.update(

                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity), context, clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                u002Where(updateKey),
                updateKey.recUpdJnjNo(),
                KseflDataAccess.CallSpecs.UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO.identifier());
    }

    private PreparedWhere u002Where(KseflU002UpdateKey updateKey) {
        String safeGmb = WhereParamGuard.fixedLengthNotNull(updateKey.gmb(), "gmb", 15);
        BigDecimal safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");
        BigDecimal safeKasHukkYmd = safeYmd(updateKey.kasHukkYmd(), "kasHukkYmd");
        String safeSgyAnknNo =
                WhereParamGuard.fixedLengthNotNull(updateKey.sgyAnknNo(), "sgyAnknNo", 10);
        return PreparedWhere.of(
                U002_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeGmb);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasgYmd);
                    nextIndex = JpaBind.set(query, nextIndex, safeKasHukkYmd);
                    return JpaBind.set(query, nextIndex, safeSgyAnknNo);
                },
                safeGmb,
                safeKasgYmd,
                safeKasHukkYmd,
                safeSgyAnknNo);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
