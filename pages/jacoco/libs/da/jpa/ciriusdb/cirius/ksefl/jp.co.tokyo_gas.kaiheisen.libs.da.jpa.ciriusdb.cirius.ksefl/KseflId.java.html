package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ksefl;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 開栓エラーフォロー件名（KSEFL）の複合主キー。 */
public class KseflId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String gmb;

    private BigDecimal kasgYmd;

    private BigDecimal kasHukkYmd;

    private String hukkSysShbt;

    private String sgyAnknNo;

    /** デフォルトコンストラクタ。 */
    public KseflId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KseflId(String gmb, BigDecimal kasgYmd, BigDecimal kasHukkYmd,
            String hukkSysShbt, String sgyAnknNo) {
        this.gmb = gmb;
        this.kasgYmd = kasgYmd;
        this.kasHukkYmd = kasHukkYmd;
        this.hukkSysShbt = hukkSysShbt;
        this.sgyAnknNo = sgyAnknNo;
    }

    /** gmbを取得する。 */
    public String getGmb() {
        return gmb;
    }

    /** gmbを設定する。 */
    public void setGmb(String gmb) {
        this.gmb = gmb;
    }

    /** kasgYmdを取得する。 */
    public BigDecimal getKasgYmd() {
        return kasgYmd;
    }

    /** kasgYmdを設定する。 */
    public void setKasgYmd(BigDecimal kasgYmd) {
        this.kasgYmd = kasgYmd;
    }

    /** kasHukkYmdを取得する。 */
    public BigDecimal getKasHukkYmd() {
        return kasHukkYmd;
    }

    /** kasHukkYmdを設定する。 */
    public void setKasHukkYmd(BigDecimal kasHukkYmd) {
        this.kasHukkYmd = kasHukkYmd;
    }

    /** hukkSysShbtを取得する。 */
    public String getHukkSysShbt() {
        return hukkSysShbt;
    }

    /** hukkSysShbtを設定する。 */
    public void setHukkSysShbt(String hukkSysShbt) {
        this.hukkSysShbt = hukkSysShbt;
    }

    /** sgyAnknNoを取得する。 */
    public String getSgyAnknNo() {
        return sgyAnknNo;
    }

    /** sgyAnknNoを設定する。 */
    public void setSgyAnknNo(String sgyAnknNo) {
        this.sgyAnknNo = sgyAnknNo;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KseflId that)) {
            return false;
        }
        return Objects.equals(gmb, that.gmb)
                && Objects.equals(kasgYmd, that.kasgYmd)
                && Objects.equals(kasHukkYmd, that.kasHukkYmd)
                && Objects.equals(hukkSysShbt, that.hukkSysShbt)
                && Objects.equals(sgyAnknNo, that.sgyAnknNo);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gmb, kasgYmd, kasHukkYmd, hukkSysShbt, sgyAnknNo);
    }
}
