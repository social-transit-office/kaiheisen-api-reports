package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.knsyti;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 検針予定テーブルの複合主キー。 */
public class KnsytiId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal tkysYm;

    /** デフォルトコンストラクタ。 */
    public KnsytiId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KnsytiId(BigDecimal gasMtstNo, BigDecimal tkysYm) {
        this.gasMtstNo = gasMtstNo;
        this.tkysYm = tkysYm;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** tkysYmを取得する。 */
    public BigDecimal getTkysYm() {
        return tkysYm;
    }

    /** tkysYmを設定する。 */
    public void setTkysYm(BigDecimal tkysYm) {
        this.tkysYm = tkysYm;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KnsytiId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo) && Objects.equals(tkysYm, that.tkysYm);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, tkysYm);
    }
}
