package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.knsyti;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/** KNSYTI データアクセスの JPA 実装。 */
public class KnsytiDataAccessImpl extends DataAccessBase implements KnsytiDataAccess {

    private static final String TABLE_NAME = "KNSYTI";

    private static final String U001_BASE_WHERE_CLAUSE = "GAS_MTST_NO = ? AND TKYS_YM = ?";

    private static final ReflectiveRecordEntityMapper<KnsytiRecord, KnsytiEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(KnsytiRecord.class, KnsytiEntity.class);

    private static final ReflectiveRecordEntityMapper<KnsytiWriteRecord, KnsytiEntity>
        WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(
                KnsytiWriteRecord.class, KnsytiEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KnsytiEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaTargetColumnUpdateSupport<KnsytiWriteRecord, KnsytiEntity, KnsytiColumn>
        updateSupport;

    /** コンストラクタ。 */
    public KnsytiDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に時刻プロバイダを差し替えて生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     * @param clock システム日時プロバイダ
     */
    KnsytiDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えて生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     * @param clock システム日時プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    KnsytiDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KnsytiEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KnsytiWriteDefinition.INSTANCE,
                        KnsytiEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public List<KnsytiRecord> findAllByGasMtstNoAndTkysYmAndTyenYm(
            Long gasMtstNo, YearMonth knskKjnYm) {
        BigDecimal safeGasMtstNo = safeGasMtstNo(gasMtstNo, "gasMtstNo");
        BigDecimal knskKjnYmValue = safeYearMonth(knskKjnYm, "knskKjnYm");
        flushIfTransactionActive();

        var builder = criteriaBuilderFactory.create(entityManager, KnsytiEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.le(builder, "tkysYm", knskKjnYmValue);
        JpaWhere.ge(builder, "tyenYm", knskKjnYmValue);
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public Optional<KnsytiRecord> findByPK(Long gasMtstNo, YearMonth tkysYm) {
        KnsytiId id =
                new KnsytiId(
                        safeGasMtstNo(gasMtstNo, "gasMtstNo"),
                        safeYearMonth(tkysYm, "tkysYm"));
        KnsytiEntity entity = entityManager.find(KnsytiEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }

    @Override
    public int updateByGasMtstNoAndTkysYmAndRecUpdJnjNo(
            KnsytiWriteRecord record,
            Set<KnsytiColumn> targetColumns,
            KnsytiUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        BigDecimal safeGasMtstNo = safeGasMtstNo(updateKey.gasMtstNo(), "gasMtstNo");
        BigDecimal safeTkysYm = safeYearMonth(updateKey.tkysYm(), "tkysYm");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                PreparedWhere.of(
                        U001_BASE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeGasMtstNo, value -> value);
                            return JpaBind.set(query, nextIndex, safeTkysYm, value -> value);
                        },
                        safeGasMtstNo,
                        safeTkysYm),
                updateKey.recUpdJnjNo(),
                KnsytiDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YM_AND_REC_UPD_JNJ_NO.identifier());
    }

    /**
     * WHERE 条件用のガスメーター設置場所番号を DB 値へ変換する。
     *
     * @param value API 値
     * @param paramName パラメータ名
     * @return DB バインド値
     */
    private static BigDecimal safeGasMtstNo(Long value, String paramName) {
        return WhereParamGuard.complement(
            JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    /**
     * WHERE 条件用の適用年月を DB 値へ変換する。
     *
     * @param value API 値
     * @param paramName パラメータ名
     * @return DB バインド値
     */
    private static BigDecimal safeYearMonth(YearMonth value, String paramName) {
        Integer yearMonthValue = value == null ? null : value.toYmInt();
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(yearMonthValue),
                paramName,
                BigDecimal.ZERO);
    }

    private void flushIfTransactionActive() {
        try {
            if (entityManager.isJoinedToTransaction()) {
                entityManager.flush();
                return;
            }
        } catch (IllegalStateException ignore) {
        }

        try {
            if (entityManager.getTransaction().isActive()) {
                entityManager.flush();
            }
        } catch (IllegalStateException ignore) {
        }
    }
}
