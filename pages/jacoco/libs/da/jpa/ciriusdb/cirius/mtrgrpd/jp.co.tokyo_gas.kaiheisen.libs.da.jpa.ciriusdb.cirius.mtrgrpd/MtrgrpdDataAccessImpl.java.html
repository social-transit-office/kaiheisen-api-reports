package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mtrgrpd;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import jakarta.persistence.TypedQuery;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtrgrpd.MtrgrpdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtrgrpd.MtrgrpdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** MTRGRPD データアクセスの JPA 実装。 */
public class MtrgrpdDataAccessImpl extends DataAccessBase implements MtrgrpdDataAccess {

    private static final ReflectiveRecordEntityMapper<MtrgrpdRecord, MtrgrpdEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(MtrgrpdRecord.class, MtrgrpdEntity.class);

    private final EntityManager entityManager;

    /**
     * JPA ベースの MTRGRPD データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public MtrgrpdDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<MtrgrpdRecord> findByGasMtstNoAndTkysYmdAndTyenYmd(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        TypedQuery<MtrgrpdEntity> query =
                entityManager.createQuery(
                        "SELECT e FROM MtrgrpdEntity e "
                                + "WHERE e.gasMtstNo = :gasMtstNo "
                                + "AND e.tkysYmd <= :knskKjnYmd "
                                + "AND e.tyenYmd >= :knskKjnYmd "
                                + "ORDER BY e.tkysYmd DESC, e.hrtGrpNo ASC",
                        MtrgrpdEntity.class);
        query.setParameter("gasMtstNo", safeGasMtstNo(gasMtstNo));
        query.setParameter("knskKjnYmd", safeYmd(knskKjnYmd, "knskKjnYmd"));
        query.setMaxResults(1);
        List<MtrgrpdEntity> results = query.getResultList();
        if (results.isEmpty()) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(results.getFirst()));
    }

    @Override
    public Optional<MtrgrpdRecord> findByPK(Integer hrtGrpNo, Long gasMtstNo, LocalDate tkysYmd) {
        MtrgrpdId id = new MtrgrpdId(
                safeHrtGrpNo(hrtGrpNo),
                safeGasMtstNo(gasMtstNo),
                safeYmd(tkysYmd, "tkysYmd"));
        MtrgrpdEntity entity = entityManager.find(MtrgrpdEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    private static BigDecimal safeHrtGrpNo(Integer value) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), "hrtGrpNo", BigDecimal.ZERO);
    }

    private static BigDecimal safeGasMtstNo(Long value) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), "gasMtstNo", BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
