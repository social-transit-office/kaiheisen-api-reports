package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mtrgrpd;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** メーターグループ＜導管＞の複合主キー。 */
public class MtrgrpdId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal hrtGrpNo;

    private BigDecimal gasMtstNo;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public MtrgrpdId() { }

    /** 全フィールド指定コンストラクタ。 */
    public MtrgrpdId(BigDecimal hrtGrpNo, BigDecimal gasMtstNo, BigDecimal tkysYmd) {
        this.hrtGrpNo = hrtGrpNo;
        this.gasMtstNo = gasMtstNo;
        this.tkysYmd = tkysYmd;
    }

    /** hrtGrpNoを取得する。 */
    public BigDecimal getHrtGrpNo() {
        return hrtGrpNo;
    }

    /** hrtGrpNoを設定する。 */
    public void setHrtGrpNo(BigDecimal hrtGrpNo) {
        this.hrtGrpNo = hrtGrpNo;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** tkysYmdを取得する。 */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /** tkysYmdを設定する。 */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof MtrgrpdId that)) {
            return false;
        }
        return Objects.equals(hrtGrpNo, that.hrtGrpNo)
                && Objects.equals(gasMtstNo, that.gasMtstNo)
                && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(hrtGrpNo, gasMtstNo, tkysYmd);
    }
}
