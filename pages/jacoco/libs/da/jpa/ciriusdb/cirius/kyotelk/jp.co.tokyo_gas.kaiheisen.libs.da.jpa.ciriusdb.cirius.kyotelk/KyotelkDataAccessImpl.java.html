package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kyotelk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.EnumSet;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.KyotelkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.KyotelkRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotelk.write.KyotelkWriteDefinition;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** KYOTELK データアクセスの JPA 実装。 */
public class KyotelkDataAccessImpl extends DataAccessBase implements KyotelkDataAccess {

    private static final String TABLE_NAME = "KYOTELK";

    private static final String PK_WHERE_CLAUSE = "KYO_TRK_NO = ? AND KYO_TEL_NO = ?";

    private static final ReflectiveRecordEntityMapper<KyotelkRecord, KyotelkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KyotelkRecord.class, KyotelkEntity.class);

    private static final ReflectiveRecordEntityMapper<KyotelkWriteRecord, KyotelkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KyotelkWriteRecord.class, KyotelkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KyotelkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KyotelkWriteRecord, KyotelkEntity, KyotelkColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KyotelkWriteRecord, KyotelkEntity, KyotelkColumn>
            updateSupport;

    /**
     * JPA 実装を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public KyotelkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に時刻プロバイダを差し替えて生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock 時刻プロバイダ
     */
    KyotelkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えて生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock 時刻プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    KyotelkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KyotelkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KyotelkWriteDefinition.INSTANCE,
                        KyotelkEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KyotelkWriteDefinition.INSTANCE,
                        KyotelkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KyotelkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            Long kyoTrkNo, String kyoTelNo, Integer recUpdJnjNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                PK_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeKyoTrkNo);
                                    return JpaBind.set(query, nextIndex, safeKyoTelNo);
                                },
                                safeKyoTrkNo,
                                safeKyoTelNo),
                        recUpdJnjNo,
                        KyotelkDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO);
    }

    @Override
    public List<KyotelkRecord> findAllByKyoTrkNo(Long kyoTrkNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var builder = criteriaBuilderFactory.create(entityManager, KyotelkEntity.class);
        builder.where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .orderByDesc("telKnzShbt")
                .orderByAsc("telRrskShbt")
                .orderByAsc("telShbt")
                .orderByDesc("telShtkYmd")
                .orderByAsc("kyoTelNo");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public Optional<KyotelkRecord> findByKyoTrkNo(Long kyoTrkNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var builder = criteriaBuilderFactory.create(entityManager, KyotelkEntity.class);
        builder.where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .orderByDesc("telKnzShbt")
                .orderByAsc("telRrskShbt")
                .orderByAsc("telShbt")
                .orderByDesc("telShtkYmd")
                .orderByAsc("kyoTelNo");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public long countByKyoTrkNo(Long kyoTrkNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var query =
                entityManager.createNativeQuery(
                        "SELECT COUNT(*) AS count FROM " + TABLE_NAME + " WHERE KYO_TRK_NO = :kyoTrkNo");
        query.setParameter("kyoTrkNo", safeKyoTrkNo);
        return ((Number) query.getSingleResult()).longValue();
    }

    @Override
    public Optional<KyotelkRecord> findByKyoTrkNoAndKyoTelNo(Long kyoTrkNo, String kyoTelNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        var builder = criteriaBuilderFactory.create(entityManager, KyotelkEntity.class);
        builder.where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .where("kyoTelNo")
                .eq(safeKyoTelNo);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            KyotelkWriteRecord record,
            Set<KyotelkColumn> targetColumns,
            KyotelkUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        BigDecimal safeKyoTrkNo = safeLong(updateKey.kyoTrkNo(), "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(updateKey.kyoTelNo());
        Set<KyotelkColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        return updateSupport.update(
                record,
                canonicalTargetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        PK_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeKyoTrkNo);
                            return JpaBind.set(query, nextIndex, safeKyoTelNo);
                        },
                        safeKyoTrkNo,
                        safeKyoTelNo),
                updateKey.recUpdJnjNo(),
                KyotelkDataAccess.Identifiers.UPDATE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", 13);
    }

    private static Set<KyotelkColumn> canonicalTargetColumns(Set<KyotelkColumn> targetColumns) {
        EnumSet<KyotelkColumn> canonicalColumns = EnumSet.noneOf(KyotelkColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }
}
