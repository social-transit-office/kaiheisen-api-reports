package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kyotelk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** KYOTELK テーブルの複合主キー。 */
public class KyotelkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String kyoTelNo;

    private BigDecimal kyoTrkNo;

    /** JPA 用のデフォルトコンストラクタ。 */
    public KyotelkId() {
    }

    /**
     * 複合主キーを生成する。
     *
     * @param kyoTelNo 契約お客さま電話番号
     * @param kyoTrkNo 契約お客さま登録番号
     */
    public KyotelkId(String kyoTelNo, BigDecimal kyoTrkNo) {
        this.kyoTelNo = kyoTelNo;
        this.kyoTrkNo = kyoTrkNo;
    }

    /**
     * 契約お客さま電話番号を返す。
     *
     * @return 契約お客さま電話番号
     */
    public String getKyoTelNo() {
        return kyoTelNo;
    }

    /**
     * 契約お客さま電話番号を設定する。
     *
     * @param kyoTelNo 契約お客さま電話番号
     */
    public void setKyoTelNo(String kyoTelNo) {
        this.kyoTelNo = kyoTelNo;
    }

    /**
     * 契約お客さま登録番号を返す。
     *
     * @return 契約お客さま登録番号
     */
    public BigDecimal getKyoTrkNo() {
        return kyoTrkNo;
    }

    /**
     * 契約お客さま登録番号を設定する。
     *
     * @param kyoTrkNo 契約お客さま登録番号
     */
    public void setKyoTrkNo(BigDecimal kyoTrkNo) {
        this.kyoTrkNo = kyoTrkNo;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KyotelkId that)) {
            return false;
        }
        return Objects.equals(kyoTelNo, that.kyoTelNo)
                && Objects.equals(kyoTrkNo, that.kyoTrkNo);
    }

    @Override
    public int hashCode() {
        return Objects.hash(kyoTelNo, kyoTrkNo);
    }
}
