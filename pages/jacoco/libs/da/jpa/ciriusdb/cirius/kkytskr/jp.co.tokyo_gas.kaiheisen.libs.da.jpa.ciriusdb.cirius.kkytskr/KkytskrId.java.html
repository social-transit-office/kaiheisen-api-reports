package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kkytskr;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 供給停止管理テーブルの複合主キー。 */
public class KkytskrId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal hesnYmd;

    /** デフォルトコンストラクタ。 */
    public KkytskrId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KkytskrId(BigDecimal gasMtstNo, BigDecimal hesnYmd) {
        this.gasMtstNo = gasMtstNo;
        this.hesnYmd = hesnYmd;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** hesnYmdを取得する。 */
    public BigDecimal getHesnYmd() {
        return hesnYmd;
    }

    /** hesnYmdを設定する。 */
    public void setHesnYmd(BigDecimal hesnYmd) {
        this.hesnYmd = hesnYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KkytskrId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo) && Objects.equals(hesnYmd, that.hesnYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, hesnYmd);
    }
}
