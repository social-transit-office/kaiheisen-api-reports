/** KKYTSKR データアクセスの JPA 実装。 */
public class KkytskrDataAccessImpl extends DataAccessBase implements KkytskrDataAccess {

    private static final String TABLE_NAME = "KKYTSKR";

    private static final String U001_WHERE_CLAUSE = "GAS_MTST_NO = ? AND HESN_YMD = ?";

    private static final ReflectiveRecordEntityMapper<KkytskrRecord, KkytskrEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KkytskrRecord.class, KkytskrEntity.class);

    private static final ReflectiveRecordEntityMapper<KkytskrWriteRecord, KkytskrEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KkytskrWriteRecord.class, KkytskrEntity.class);

    private static final BigDecimal ZERO_YMD = BigDecimal.ZERO;

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KkytskrEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KkytskrWriteRecord, KkytskrEntity, KkytskrColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KkytskrWriteRecord, KkytskrEntity, KkytskrColumn>
            updateSupport;

    /**
     * JPA ベースの KKYTSKR データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     */
    public KkytskrDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KkytskrDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KkytskrDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KkytskrEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KkytskrWriteDefinition.INSTANCE,
                        KkytskrEntity.class,
                        KkytskrDataAccessImpl::toWriteEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KkytskrWriteDefinition.INSTANCE,
                        KkytskrEntity.class,
                        KkytskrDataAccessImpl::toWriteEntity);
    }

    @Override
    public int insert(KkytskrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByGasMtstNoAndHesnYmd(Long gasMtstNo, LocalDate hesnYmd, Integer recUpdJnjNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeHesnYmd = safeYmd(hesnYmd, "hesnYmd");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GAS_MTST_NO = ? AND HESN_YMD = ?",
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(query, startIndex, safeGasMtstNo);
                                    return JpaBind.set(query, nextIndex, safeHesnYmd);
                                },
                                safeGasMtstNo,
                                safeHesnYmd),
                        recUpdJnjNo,
                        KkytskrDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_HESN_YMD);
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNo(Long gasMtstNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        var builder = criteriaBuilderFactory.create(entityManager, KkytskrEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .orderByDesc("hesnYmd");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndHesnYmd(Long gasMtstNo, LocalDate hesnYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeHesnYmd = safeYmd(hesnYmd, "hesnYmd");
        var builder = criteriaBuilderFactory.create(entityManager, KkytskrEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("hesnYmd")
                .eq(safeHesnYmd);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
            Long gasMtstNo, LocalDate kasYmd, String kkysSisnJtkb) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKasYmd = safeYmd(kasYmd, "kasYmd");
        String safeKkysSisnJtkb =
                WhereParamGuard.fixedLengthNotNull(kkysSisnJtkb, "kkysSisnJtkb", 1);
        var builder = criteriaBuilderFactory.create(entityManager, KkytskrEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("kasYmd")
                .eq(safeKasYmd)
                .where("kkysSisnJtkb")
                .eq(safeKkysSisnJtkb)
                .orderByDesc("hesnYmd");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndHesnYmdAndKkysSisnJtkb(
            Long gasMtstNo, LocalDate hesnYmd, String kkysSisnJtkb) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeHesnYmd = safeYmd(hesnYmd, "hesnYmd");
        String safeKkysSisnJtkb =
                WhereParamGuard.fixedLengthNotNull(kkysSisnJtkb, "kkysSisnJtkb", 1);
        var builder = criteriaBuilderFactory.create(entityManager, KkytskrEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("hesnYmd")
                .eq(safeHesnYmd)
                .where("kkysSisnJtkb")
                .eq(safeKkysSisnJtkb);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateByGasMtstNoAndHesnYmd(
            KkytskrWriteRecord record,
            Set<KkytskrColumn> targetColumns,
            KkytskrHesnUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo =
                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        BigDecimal safeGasMtstNo = safeLong(updateKey.gasMtstNo(), "gasMtstNo");
        BigDecimal safeHesnYmd = safeYmd(updateKey.hesnYmd(), "hesnYmd");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        U001_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(query, startIndex, safeGasMtstNo);
                            return JpaBind.set(query, nextIndex, safeHesnYmd);
                        },
                        safeGasMtstNo,
                        safeHesnYmd),
                updateKey.recUpdJnjNo(),
                KkytskrDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD);
    }

    @Override
    public int updateByGasMtstNoAndHesnYmdAndRecUpdJnjNo(
            KkytskrWriteRecord record,
            Set<KkytskrColumn> targetColumns,
            KkytskrHesnJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                u002Where(updateKey),
                updateKey.recUpdJnjNo(),
                KkytskrDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD_AND_REC_UPD_JNJ_NO);
    }

    private PreparedWhere u002Where(KkytskrHesnJnjUpdateKey updateKey) {
        BigDecimal safeGasMtstNo = safeLong(updateKey.gasMtstNo(), "gasMtstNo");
        BigDecimal safeHesnYmd = safeYmd(updateKey.hesnYmd(), "hesnYmd");
        return PreparedWhere.of(
                U001_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeGasMtstNo);
                    return JpaBind.set(query, nextIndex, safeHesnYmd);
                },
                safeGasMtstNo,
                safeHesnYmd);
    }

    private static KkytskrEntity toWriteEntity(KkytskrWriteRecord record) {
        KkytskrEntity entity = WRITE_MAPPER.toEntity(record);
        if (record.kasYmd() == null) {
            entity.setKasYmd(ZERO_YMD);
        }
        return entity;
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
