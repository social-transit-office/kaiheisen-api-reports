package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.crdhri;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.CrdhriDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.CrdhriRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.write.CrdhriColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.write.CrdhriSihKiyJnjUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.write.CrdhriSihKiyUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.write.CrdhriWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.crdhri.write.CrdhriWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CiriusAuditColumnAutoSetter;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaBind;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * CRDHRI データアクセスの JPA 実装.
 */
public class CrdhriDataAccessImpl
        extends DataAccessBase implements CrdhriDataAccess {

    /** 読み取り用 mapper. */
    static final ReflectiveRecordEntityMapper<CrdhriRecord, CrdhriEntity>
            READ_MAPPER = ReflectiveRecordEntityMapper.create(
                    CrdhriRecord.class,
                    CrdhriEntity.class);

    /** テーブル名. */
    private static final String TABLE_NAME = "CRDHRI";

    /** 主キー条件句. */
    private static final String WHERE_BY_PK =
            "SIH_KIY_NO = ? AND SIH_KIY_SKNO = ?";

    /** 書き込み用 mapper. */
    private static final ReflectiveRecordEntityMapper<
            CrdhriWriteRecord,
            CrdhriEntity> WRITE_MAPPER = ReflectiveRecordEntityMapper.create(
                    CrdhriWriteRecord.class,
                    CrdhriEntity.class);

    /** エンティティマネージャ. */
    private final EntityManager entityManager;

    /** CriteriaBuilderFactory. */
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** システム時刻プロバイダ. */
    private final SystemDateTimeProvider clock;

    /** 監査項目アクセサバインダ. */
    private final JpaAuditColumnAccessBinder<CrdhriEntity>
            auditColumnAccessBinder;

    /** 監査項目自動設定. */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** insert helper. */
    private final JpaInsertSupport<
            CrdhriWriteRecord,
            CrdhriEntity,
            CrdhriColumn>
            insertSupport;

    /** update helper. */
    private final JpaTargetColumnUpdateSupport<
            CrdhriWriteRecord,
            CrdhriEntity,
            CrdhriColumn> updateSupport;

    /**
     * 標準依存で初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     */
    public CrdhriDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の時刻依存を含めて初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     */
    CrdhriDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                systemClock,
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の依存を含めて初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査項目自動設定
     */
    CrdhriDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.entityManager =
                Objects.requireNonNull(jpaEntityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(
                        jpaCriteriaBuilderFactory,
                        "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(systemClock, "clock");
        this.auditColumnAccessBinder =
                new JpaAuditColumnAccessBinder<>(CrdhriEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(
                        ciriusAuditColumnAutoSetter,
                        "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        CrdhriWriteDefinition.INSTANCE,
                        CrdhriEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        CrdhriWriteDefinition.INSTANCE,
                        CrdhriEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    /** {@inheritDoc} */
    @Override
    public int insert(
            final CrdhriWriteRecord record,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity -> auditColumnAutoSetter.applyForInsert(
                        auditColumnAccessBinder.bind(entity),
                        context,
                        clock));
    }

    /** {@inheritDoc} */
    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                WHERE_BY_PK,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(
                                            query,
                                            startIndex,
                                            safeSihKiyNo);
                                    return JpaBind.set(
                                            query,
                                            nextIndex,
                                            safeSihKiySkno);
                                },
                                safeSihKiyNo,
                                safeSihKiySkno),
                        recUpdJnjNo,
                        CrdhriDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    /** {@inheritDoc} */
    @Override
    public Optional<CrdhriRecord> findBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        var builder = criteriaBuilderFactory.create(
                entityManager,
                CrdhriEntity.class);
        builder.where("sihKiyNo")
                .eq(safeSihKiyNo)
                .where("sihKiySkno")
                .eq(safeSihKiySkno);
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            final CrdhriWriteRecord record,
            final Set<CrdhriColumn> targetColumns,
            final CrdhriSihKiyUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                CrdhriDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            final CrdhriWriteRecord record,
            final Set<CrdhriColumn> targetColumns,
            final CrdhriSihKiyJnjUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                CrdhriDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO.identifier());
    }

    private int updateByKey(
            final CrdhriWriteRecord record,
            final Set<CrdhriColumn> targetColumns,
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo,
            final AuditContext context,
            final String dataAccessIdentifier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(context, "context");
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(recUpdJnjNo, 0)),
                PreparedWhere.of(
                        WHERE_BY_PK,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(
                                    query,
                                    startIndex,
                                    safeSihKiyNo);
                            return JpaBind.set(
                                    query,
                                    nextIndex,
                                    safeSihKiySkno);
                        },
                        safeSihKiyNo,
                        safeSihKiySkno),
                recUpdJnjNo,
                dataAccessIdentifier);
    }

    private static BigDecimal safeLong(
            final Long value,
            final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(
            final Integer value,
            final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value),
                paramName,
                BigDecimal.ZERO);
    }
}
