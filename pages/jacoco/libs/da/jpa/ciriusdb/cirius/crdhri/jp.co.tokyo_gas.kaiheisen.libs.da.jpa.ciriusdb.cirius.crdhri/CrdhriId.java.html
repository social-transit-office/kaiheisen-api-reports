package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.crdhri;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/**
 * カード払い関連情報テーブルの複合主キー.
 */
public class CrdhriId implements Serializable {

    /** シリアルバージョン UID. */
    private static final long serialVersionUID = 1L;

    /** 支払契約番号. */
    private BigDecimal sihKiyNo;

    /** 支払契約世代管理番号. */
    private BigDecimal sihKiySkno;

    /** デフォルトコンストラクタ. */
    public CrdhriId() {
    }

    /**
     * 主キー項目を指定して初期化する.
     *
     * @param primarySihKiyNo 支払契約番号
     * @param primarySihKiySkno 支払契約世代管理番号
     */
    public CrdhriId(
            final BigDecimal primarySihKiyNo,
            final BigDecimal primarySihKiySkno) {
        this.sihKiyNo = primarySihKiyNo;
        this.sihKiySkno = primarySihKiySkno;
    }

    /**
     * 支払契約番号を返す.
     *
     * @return 支払契約番号
     */
    public BigDecimal getSihKiyNo() {
        return sihKiyNo;
    }

    /**
     * 支払契約番号を設定する.
     *
     * @param newSihKiyNo 支払契約番号
     */
    public void setSihKiyNo(final BigDecimal newSihKiyNo) {
        this.sihKiyNo = newSihKiyNo;
    }

    /**
     * 支払契約世代管理番号を返す.
     *
     * @return 支払契約世代管理番号
     */
    public BigDecimal getSihKiySkno() {
        return sihKiySkno;
    }

    /**
     * 支払契約世代管理番号を設定する.
     *
     * @param newSihKiySkno 支払契約世代管理番号
     */
    public void setSihKiySkno(final BigDecimal newSihKiySkno) {
        this.sihKiySkno = newSihKiySkno;
    }

    /** {@inheritDoc} */
    @Override
    public boolean equals(final Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof CrdhriId that)) {
            return false;
        }
        return Objects.equals(sihKiyNo, that.sihKiyNo)
                && Objects.equals(sihKiySkno, that.sihKiySkno);
    }

    /** {@inheritDoc} */
    @Override
    public int hashCode() {
        return Objects.hash(sihKiyNo, sihKiySkno);
    }
}
