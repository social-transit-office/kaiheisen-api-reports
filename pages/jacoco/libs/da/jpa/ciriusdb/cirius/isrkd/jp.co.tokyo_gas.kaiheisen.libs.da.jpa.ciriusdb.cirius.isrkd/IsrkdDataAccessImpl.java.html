package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrkd;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** ISRKD データアクセスの JPA 実装。 */
public class IsrkdDataAccessImpl extends DataAccessBase implements IsrkdDataAccess {

    private static final int DUKN_ISRS_CD_LENGTH = 8;

    private static final ReflectiveRecordEntityMapper<IsrkdRecord, IsrkdEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(IsrkdRecord.class, IsrkdEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public IsrkdDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<IsrkdRecord> findByArtyCstNoAndIsrrTriYmdAndDuknIsrsCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> duknIsrsCdList) {
        List<String> safeDuknIsrsCdList = normalizeDuknIsrsCdList(duknIsrsCdList);
        if (safeDuknIsrsCdList.isEmpty()) {
            return Optional.empty();
        }
        BigDecimal safeArtyCstNo =
                JpaDbWriteValues.bigDecimal(WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L));
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        var builder = criteriaBuilderFactory.create(entityManager, IsrkdEntity.class);
        JpaWhere.eq(builder, "artyCstNo", safeArtyCstNo);
        JpaWhere.eq(builder, "isrrTriYmd", safeIsrrTriYmd);
        JpaWhere.in(builder, "duknIsrsCd", safeDuknIsrsCdList);
        builder.orderByDesc("isrrKytuTsno");
        return builder.setMaxResults(1).getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<IsrkdRecord> findByPK(Long artyCstNo, LocalDate isrrTriYmd, Integer isrrKytuTsno) {
        BigDecimal safeArtyCstNo =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L));
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        BigDecimal safeIsrrKytuTsno =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(isrrKytuTsno, "isrrKytuTsno", 0));

        var builder = criteriaBuilderFactory.create(entityManager, IsrkdEntity.class);
        JpaWhere.eq(builder, "artyCstNo", safeArtyCstNo);
        JpaWhere.eq(builder, "isrrTriYmd", safeIsrrTriYmd);
        JpaWhere.eq(builder, "isrrKytuTsno", safeIsrrKytuTsno);
        builder.setMaxResults(1);
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    /**
     * IN 句検索用の業務詳細コード一覧を固定長文字列へ正規化する。
     *
     * <p>主な仕様: null の一覧は拒否し、空一覧はそのまま返却する。
     * 各要素は null/空白-only を入力不正として検証した後、固定長 8 文字へ整形する。
     *
     * <p>既知の制約: 要素に null または空白-only を含む一覧は入力不正として扱う。
     */
    private static List<String> normalizeDuknIsrsCdList(List<String> duknIsrsCdList) {
        if (duknIsrsCdList == null) {
            throw new IllegalArgumentException("duknIsrsCdList must not be null");
        }
        if (duknIsrsCdList.isEmpty()) {
            return List.of();
        }
        return duknIsrsCdList.stream().map(IsrkdDataAccessImpl::normalizeInDuknIsrsCd).toList();
    }

    /**
     * IN 句要素の業務詳細コードを固定長条件値へ正規化する。
     *
     * <p>主な仕様: null と空白-only を入力不正として拒否した上で、fixedLengthNotNull により固定長 8 文字へ整形する。
     *
     * <p>既知の制約: 8 文字超の値は切り詰めず、そのまま返す。
     */
    private static String normalizeInDuknIsrsCd(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "duknIsrsCdList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "duknIsrsCdList", DUKN_ISRS_CD_LENGTH);
    }
}
