/**
 * 請求書表示制御（SQSHYJS）データアクセスの JPA 実装。
 */
public class SqshyjsDataAccessImpl extends DataAccessBase implements SqshyjsDataAccess {

    private static final String TABLE_NAME = "SQSHYJS";

    private static final String WHERE_CLAUSE_BY_PK = "SKSH_HYJS_KRNO = ? AND TKYS_YMD = ?";

    private static final String WHERE_CLAUSE_BY_RANGE =
            "SKSH_HYJS_KRNO = ? AND TKYS_YMD <= ? AND TYEN_YMD >= ?";

    static final ReflectiveRecordEntityMapper<SqshyjsRecord, SqshyjsEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(SqshyjsRecord.class, SqshyjsEntity.class);

    private static final ReflectiveRecordEntityMapper<SqshyjsWriteRecord, SqshyjsEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(SqshyjsWriteRecord.class, SqshyjsEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<SqshyjsEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<SqshyjsWriteRecord, SqshyjsEntity, SqshyjsColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<SqshyjsWriteRecord, SqshyjsEntity, SqshyjsColumn>
            updateSupport;

    /**
     * 標準依存で初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public SqshyjsDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(entityManager, criteriaBuilderFactory, new DefaultSystemDateTimeProvider());
    }

    SqshyjsDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    SqshyjsDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(SqshyjsEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        SqshyjsWriteDefinition.INSTANCE,
                        SqshyjsEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        SqshyjsWriteDefinition.INSTANCE,
                        SqshyjsEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(SqshyjsWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteBySkshHyjsKrnoAndTkysYmd(
            Long skshHyjsKrno, LocalDate tkysYmd, Integer recUpdJnjNo) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "SKSH_HYJS_KRNO = ? AND TKYS_YMD >= ?",
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeSkshHyjsKrno);
                                    return JpaBind.set(query, nextIndex, safeTkysYmd);
                                },
                                safeSkshHyjsKrno,
                                safeTkysYmd),
                        recUpdJnjNo,
                        SqshyjsDataAccess.Identifiers.DELETE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD);
    }

    @Override
    public Optional<SqshyjsRecord> findBySkshHyjsKrnoAndTyenYmd(Long skshHyjsKrno, LocalDate tyenYmd) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeTyenYmd = safeYmd(tyenYmd, "tyenYmd");
        var builder = criteriaBuilderFactory.create(entityManager, SqshyjsEntity.class);
        JpaWhere.eq(builder, "skshHyjsKrno", safeSkshHyjsKrno);
        JpaWhere.eq(builder, "tyenYmd", safeTyenYmd);
        builder.orderByDesc("tkysYmd");
        SqlCapture.set(builder.getQueryString());
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<SqshyjsRecord> findBySkshHyjsKrnoAndKnskKjnYmd(
            Long skshHyjsKrno,
            LocalDate knskKjnYmd) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        var builder = criteriaBuilderFactory.create(entityManager, SqshyjsEntity.class);
        JpaWhere.eq(builder, "skshHyjsKrno", safeSkshHyjsKrno);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        SqlCapture.set(builder.getQueryString());
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<SqshyjsRecord> findByPK(Long skshHyjsKrno, LocalDate tkysYmd) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        SqshyjsEntity entity =
                entityManager.find(
                        SqshyjsEntity.class,
                        new SqshyjsId(safeSkshHyjsKrno, safeTkysYmd));
        return Optional.ofNullable(entity).map(READ_MAPPER::toRecord);
    }

    @Override
    public int updateBySkshHyjsKrnoAndKnskKjnYmd(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsKnskKjnUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhereByRange(updateKey.skshHyjsKrno(), updateKey.knskKjnYmd()),
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_KNSK_KJN_YMD);
    }

    @Override
    public int updateBySkshHyjsKrnoAndTkysYmdAndRecUpdJnjNo(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsTkysYmdJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhereByPk(updateKey.skshHyjsKrno(), updateKey.tkysYmd()),
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO);
    }

    @Override
    public int updateBySkshHyjsKrnoAndTkysYmd(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsTkysYmdUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhereByPk(updateKey.skshHyjsKrno(), updateKey.tkysYmd()),
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD);
    }

    private PreparedWhere preparedWhereByPk(Long skshHyjsKrno, LocalDate tkysYmd) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return PreparedWhere.of(
                WHERE_CLAUSE_BY_PK,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeSkshHyjsKrno);
                    return JpaBind.set(query, nextIndex, safeTkysYmd);
                },
                safeSkshHyjsKrno,
                safeTkysYmd);
    }

    private PreparedWhere preparedWhereByRange(Long skshHyjsKrno, LocalDate knskKjnYmd) {
        BigDecimal safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return PreparedWhere.of(
                WHERE_CLAUSE_BY_RANGE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeSkshHyjsKrno);
                    nextIndex = JpaBind.set(query, nextIndex, safeKnskKjnYmd);
                    return JpaBind.set(query, nextIndex, safeKnskKjnYmd);
                },
                safeSkshHyjsKrno,
                safeKnskKjnYmd,
                safeKnskKjnYmd);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
