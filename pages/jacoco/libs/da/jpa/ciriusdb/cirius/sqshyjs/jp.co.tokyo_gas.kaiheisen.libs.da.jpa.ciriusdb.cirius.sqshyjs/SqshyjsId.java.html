package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.sqshyjs;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/**
 * 請求書表示制御テーブルの複合主キー。
 */
public class SqshyjsId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal skshHyjsKrno;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public SqshyjsId() {
    }

    /**
     * 主キー項目を指定して初期化する。
     *
     * @param skshHyjsKrno 請求書表示制御管理番号
     * @param tkysYmd 適用開始年月日
     */
    public SqshyjsId(BigDecimal skshHyjsKrno, BigDecimal tkysYmd) {
        this.skshHyjsKrno = skshHyjsKrno;
        this.tkysYmd = tkysYmd;
    }

    /**
     * 請求書表示制御管理番号を返す。
     *
     * @return 請求書表示制御管理番号
     */
    public BigDecimal getSkshHyjsKrno() {
        return skshHyjsKrno;
    }

    /**
     * 請求書表示制御管理番号を設定する。
     *
     * @param skshHyjsKrno 請求書表示制御管理番号
     */
    public void setSkshHyjsKrno(BigDecimal skshHyjsKrno) {
        this.skshHyjsKrno = skshHyjsKrno;
    }

    /**
     * 適用開始年月日を返す。
     *
     * @return 適用開始年月日
     */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /**
     * 適用開始年月日を設定する。
     *
     * @param tkysYmd 適用開始年月日
     */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof SqshyjsId that)) {
            return false;
        }
        return Objects.equals(skshHyjsKrno, that.skshHyjsKrno)
                && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(skshHyjsKrno, tkysYmd);
    }
}
