package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gmtsetd;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** ガスメーター設置場所テーブルの複合主キー。 */
public class GmtsetdId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public GmtsetdId() { }

    /** 全フィールド指定コンストラクタ。 */
    public GmtsetdId(BigDecimal gasMtstNo, BigDecimal tkysYmd) {
        this.gasMtstNo = gasMtstNo;
        this.tkysYmd = tkysYmd;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** tkysYmdを取得する。 */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /** tkysYmdを設定する。 */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof GmtsetdId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo) && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, tkysYmd);
    }
}
