package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kiycstd;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kiycstd.KiycstdDataAccess;

/** KIYCSTD データアクセスの JPA 実装。 */
public class KiycstdDataAccessImpl extends DataAccessBase implements KiycstdDataAccess {

    private static final String TABLE_NAME = "KIYCSTD";

    private static final String U001_WHERE_CLAUSE = "KYO_TRK_NO = ?";

    private static final ReflectiveRecordEntityMapper<KiycstdRecord, KiycstdEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KiycstdRecord.class, KiycstdEntity.class);

    private static final ReflectiveRecordEntityMapper<KiycstdWriteRecord, KiycstdEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KiycstdWriteRecord.class, KiycstdEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KiycstdEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KiycstdWriteRecord, KiycstdEntity, KiycstdColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KiycstdWriteRecord, KiycstdEntity, KiycstdColumn>
            updateSupport;

    /**
     * JPA ベースの KIYCSTD データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     */
    public KiycstdDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KiycstdDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KiycstdDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KiycstdEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KiycstdWriteDefinition.INSTANCE,
                        KiycstdEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KiycstdWriteDefinition.INSTANCE,
                        KiycstdEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KiycstdWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<KiycstdRecord> findByKyoTrkNo(Long kyoTrkNo) {
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var builder = criteriaBuilderFactory.create(entityManager, KiycstdEntity.class);
        builder.where("kyoTrkNo").eq(safeKyoTrkNo);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateByKyoTrkNo(
            KiycstdWriteRecord record,
            Set<KiycstdColumn> targetColumns,
            KiycstdUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo =
                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        BigDecimal safeKyoTrkNo = safeLong(updateKey.kyoTrkNo(), "kyoTrkNo");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        U001_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            return JpaBind.set(query, startIndex, safeKyoTrkNo);
                        },
                        safeKyoTrkNo),
                updateKey.recUpdJnjNo(),
                KiycstdDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO.identifier());
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
