package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.tkssvky;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 託送サービス契約テーブルの複合主キー。 */
public class TkssvkyId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal tksuSvkyNo;

    private BigDecimal skno;

    /** デフォルトコンストラクタ。 */
    public TkssvkyId() { }

    /** 全フィールド指定コンストラクタ。 */
    public TkssvkyId(BigDecimal tksuSvkyNo, BigDecimal skno) {
        this.tksuSvkyNo = tksuSvkyNo;
        this.skno = skno;
    }

    /** tksuSvkyNoを取得する。 */
    public BigDecimal getTksuSvkyNo() {
        return tksuSvkyNo;
    }

    /** tksuSvkyNoを設定する。 */
    public void setTksuSvkyNo(BigDecimal tksuSvkyNo) {
        this.tksuSvkyNo = tksuSvkyNo;
    }

    /** sknoを取得する。 */
    public BigDecimal getSkno() {
        return skno;
    }

    /** sknoを設定する。 */
    public void setSkno(BigDecimal skno) {
        this.skno = skno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof TkssvkyId that)) {
            return false;
        }
        return Objects.equals(tksuSvkyNo, that.tksuSvkyNo) && Objects.equals(skno, that.skno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(tksuSvkyNo, skno);
    }
}
