package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gymykiy;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gymykiy.GymykiyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gymykiy.GymykiyRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GYMYKIY データアクセスの JPA 実装。 */
public class GymykiyDataAccessImpl extends DataAccessBase implements GymykiyDataAccess {

    private static final ReflectiveRecordEntityMapper<GymykiyRecord, GymykiyEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(GymykiyRecord.class, GymykiyEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public GymykiyDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<GymykiyRecord> findBySvkyNoAndKyttSknoAndKyttTsno(
            Long svkyNo,
            Integer kyttSkno,
            Integer kyttTsno) {
        var builder = criteriaBuilderFactory.create(entityManager, GymykiyEntity.class);
        builder.where("svkyNo").eq(safeLong(svkyNo, "svkyNo"));
        builder.where("kyttSkno").eq(safeInteger(kyttSkno, "kyttSkno"));
        builder.where("kyttTsno").eq(safeInteger(kyttTsno, "kyttTsno"));
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
