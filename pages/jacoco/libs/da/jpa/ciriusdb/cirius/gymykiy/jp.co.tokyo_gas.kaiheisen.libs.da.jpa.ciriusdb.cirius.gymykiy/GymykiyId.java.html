package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gymykiy;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** GYMYKIY の複合主キー。 */
public class GymykiyId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal svkyNo;

    private BigDecimal kyttSkno;

    private BigDecimal kyttTsno;

    /** デフォルトコンストラクタ。 */
    public GymykiyId() { }

    /** 全フィールド指定コンストラクタ。 */
    public GymykiyId(BigDecimal svkyNo, BigDecimal kyttSkno, BigDecimal kyttTsno) {
        this.svkyNo = svkyNo;
        this.kyttSkno = kyttSkno;
        this.kyttTsno = kyttTsno;
    }

    /** svkyNoを取得する。 */
    public BigDecimal getSvkyNo() {
        return svkyNo;
    }

    /** svkyNoを設定する。 */
    public void setSvkyNo(BigDecimal svkyNo) {
        this.svkyNo = svkyNo;
    }

    /** kyttSknoを取得する。 */
    public BigDecimal getKyttSkno() {
        return kyttSkno;
    }

    /** kyttSknoを設定する。 */
    public void setKyttSkno(BigDecimal kyttSkno) {
        this.kyttSkno = kyttSkno;
    }

    /** kyttTsnoを取得する。 */
    public BigDecimal getKyttTsno() {
        return kyttTsno;
    }

    /** kyttTsnoを設定する。 */
    public void setKyttTsno(BigDecimal kyttTsno) {
        this.kyttTsno = kyttTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof GymykiyId that)) {
            return false;
        }
        return Objects.equals(svkyNo, that.svkyNo)
                && Objects.equals(kyttSkno, that.kyttSkno)
                && Objects.equals(kyttTsno, that.kyttTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(svkyNo, kyttSkno, kyttTsno);
    }
}
