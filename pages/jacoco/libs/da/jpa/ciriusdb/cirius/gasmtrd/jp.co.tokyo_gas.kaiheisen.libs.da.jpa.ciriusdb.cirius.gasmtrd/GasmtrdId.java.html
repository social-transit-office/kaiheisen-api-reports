package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gasmtrd;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** ガスメーター＜導管＞テーブルの複合主キー。 */
public class GasmtrdId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String gmb;

    private BigDecimal gmbTsno;

    /** デフォルトコンストラクタ。 */
    public GasmtrdId() { }

    /** 全フィールド指定コンストラクタ。 */
    public GasmtrdId(String gmb, BigDecimal gmbTsno) {
        this.gmb = gmb;
        this.gmbTsno = gmbTsno;
    }

    /** gmbを取得する。 */
    public String getGmb() {
        return gmb;
    }

    /** gmbを設定する。 */
    public void setGmb(String gmb) {
        this.gmb = gmb;
    }

    /** gmbTsnoを取得する。 */
    public BigDecimal getGmbTsno() {
        return gmbTsno;
    }

    /** gmbTsnoを設定する。 */
    public void setGmbTsno(BigDecimal gmbTsno) {
        this.gmbTsno = gmbTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof GasmtrdId that)) {
            return false;
        }
        return Objects.equals(gmbTsno, that.gmbTsno) && Objects.equals(gmb, that.gmb);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gmb, gmbTsno);
    }
}
