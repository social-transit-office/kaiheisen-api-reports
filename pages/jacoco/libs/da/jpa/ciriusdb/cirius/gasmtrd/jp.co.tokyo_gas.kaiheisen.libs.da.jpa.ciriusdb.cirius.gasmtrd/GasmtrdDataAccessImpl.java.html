package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gasmtrd;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrd.GasmtrdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrd.GasmtrdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GASMTRD（ガスメーター＜導管＞）データアクセスの JPA 実装。 */
public class GasmtrdDataAccessImpl extends DataAccessBase implements GasmtrdDataAccess {

    private static final int GMB_LENGTH = 15;

    private static final ReflectiveRecordEntityMapper<GasmtrdRecord, GasmtrdEntity>
            READ_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            GasmtrdRecord.class, GasmtrdEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public GasmtrdDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNo(Long ttkGmsbNo) {
        Long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(safeTtkGmsbNo));
        builder.orderByAsc("gmbTsno");
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public List<GasmtrdRecord> findAllByTtkGmsbNoAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        Long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        BigDecimal safeGmtThkgYmd = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(safeTtkGmsbNo));
        JpaWhere.eq(builder, "gmtThkgYmd", safeGmtThkgYmd);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNoAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        Long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        BigDecimal safeGmtThkgYmd = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(safeTtkGmsbNo));
        JpaWhere.eq(builder, "gmtThkgYmd", safeGmtThkgYmd);
        builder.orderByDesc("gmbTsno");
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public List<GasmtrdRecord> findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate gmtkYmd, LocalDate gmtThkgYmd) {
        Long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        BigDecimal safeGmtkYmd = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        BigDecimal safeGmtThkgYmd = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(safeTtkGmsbNo));
        JpaWhere.le(builder, "gmtkYmd", safeGmtkYmd);
        JpaWhere.eq(builder, "gmtThkgYmd", safeGmtThkgYmd);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public Optional<GasmtrdRecord> findByGmbAndGmtkYmd(String gmb, LocalDate gmtkYmd) {
        String safeGmb = safeGmb(gmb, "gmb");
        BigDecimal safeGmtkYmd = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "gmb", safeGmb);
        JpaWhere.le(builder, "gmtkYmd", safeGmtkYmd);
        builder.orderByDesc("gmbTsno");
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<GasmtrdRecord> findByGmbAndGmtThkgYmd(String gmb, LocalDate gmtThkgYmd) {
        String safeGmb = safeGmb(gmb, "gmb");
        BigDecimal safeGmtThkgYmd = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "gmb", safeGmb);
        JpaWhere.eq(builder, "gmtThkgYmd", safeGmtThkgYmd);
        builder.orderByDesc("gmbTsno");
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNoAndGmtkYmd(Long ttkGmsbNo, LocalDate gmtkYmd) {
        Long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        BigDecimal safeGmtkYmd = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GasmtrdEntity.class);
        JpaWhere.eq(builder, "ttkGmsbNo", JpaDbWriteValues.bigDecimal(safeTtkGmsbNo));
        JpaWhere.le(builder, "gmtkYmd", safeGmtkYmd);
        builder.orderByDesc("gmtkYmd");
        builder.orderByAsc("gmtThkgYmd");
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    private static Long safeTtkGmsbNo(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static BigDecimal safeGmtThkgYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeGmtkYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static String safeGmb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GMB_LENGTH);
    }
}
