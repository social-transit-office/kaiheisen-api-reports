package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kuzfrk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.KuzfrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.KuzfrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.write.KuzfrkColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.write.KuzfrkSihKiyJnjUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.write.KuzfrkSihKiyUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.write.KuzfrkWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kuzfrk.write.KuzfrkWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * KUZFRK データアクセスの JPA 実装。
 */
public class KuzfrkDataAccessImpl extends DataAccessBase implements KuzfrkDataAccess {

    static final ReflectiveRecordEntityMapper<KuzfrkRecord, KuzfrkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KuzfrkRecord.class, KuzfrkEntity.class);

    private static final String TABLE_NAME = "KUZFRK";

    private static final String WHERE_BY_PK = "SIH_KIY_NO = ? AND SIH_KIY_SKNO = ?";

    private static final ReflectiveRecordEntityMapper<KuzfrkWriteRecord, KuzfrkEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(KuzfrkWriteRecord.class, KuzfrkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KuzfrkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KuzfrkWriteRecord, KuzfrkEntity, KuzfrkColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<KuzfrkWriteRecord, KuzfrkEntity, KuzfrkColumn> updateSupport;

    /**
     * 標準依存で初期化する。
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     */
    public KuzfrkDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の時刻依存を含めて初期化する。
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     */
    KuzfrkDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                systemClock,
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の依存を含めて初期化する。
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査項目自動設定
     */
    KuzfrkDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(jpaEntityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(jpaCriteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(systemClock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KuzfrkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(ciriusAuditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KuzfrkWriteDefinition.INSTANCE,
                        KuzfrkEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KuzfrkWriteDefinition.INSTANCE,
                        KuzfrkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(final KuzfrkWriteRecord record, final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity -> auditColumnAutoSetter.applyForInsert(
                        auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                WHERE_BY_PK,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeSihKiyNo);
                                    return JpaBind.set(query, nextIndex, safeSihKiySkno);
                                },
                                safeSihKiyNo,
                                safeSihKiySkno),
                        recUpdJnjNo,
                        KuzfrkDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    @Override
    public Optional<KuzfrkRecord> findBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        var builder = criteriaBuilderFactory.create(entityManager, KuzfrkEntity.class);
        builder.where("sihKiyNo").eq(safeSihKiyNo).where("sihKiySkno").eq(safeSihKiySkno);
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final KuzfrkSihKiyUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                KuzfrkDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final KuzfrkSihKiyJnjUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                KuzfrkDataAccess.CallSpecs
                        .UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO
                        .identifier());
    }

    private int updateByKey(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo,
            final AuditContext context,
            final String dataAccessIdentifier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(context, "context");
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(recUpdJnjNo, 0)),
                PreparedWhere.of(
                        WHERE_BY_PK,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeSihKiyNo);
                            return JpaBind.set(query, nextIndex, safeSihKiySkno);
                        },
                        safeSihKiyNo,
                        safeSihKiySkno),
                recUpdJnjNo,
                dataAccessIdentifier);
    }

    private static BigDecimal safeLong(final Long value, final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(final Integer value, final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
