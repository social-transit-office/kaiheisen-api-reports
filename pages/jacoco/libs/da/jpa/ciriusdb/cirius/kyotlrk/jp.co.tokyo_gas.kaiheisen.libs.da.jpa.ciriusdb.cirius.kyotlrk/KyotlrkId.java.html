package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kyotlrk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** KYOTLRK テーブルの複合主キー。 */
public class KyotlrkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String kyoTelNo;

    private BigDecimal kyoTrkNo;

    private BigDecimal telNoTsno;

    /** JPA 用のデフォルトコンストラクタ。 */
    public KyotlrkId() {
    }

    /**
     * 複合主キーを生成する。
     *
     * @param kyoTelNo 契約お客さま電話番号
     * @param kyoTrkNo 契約お客さま登録番号
     * @param telNoTsno 電話番号通番
     */
    public KyotlrkId(String kyoTelNo, BigDecimal kyoTrkNo, BigDecimal telNoTsno) {
        this.kyoTelNo = kyoTelNo;
        this.kyoTrkNo = kyoTrkNo;
        this.telNoTsno = telNoTsno;
    }

    /**
     * 契約お客さま電話番号を返す。
     *
     * @return 契約お客さま電話番号
     */
    public String getKyoTelNo() {
        return kyoTelNo;
    }

    /**
     * 契約お客さま電話番号を設定する。
     *
     * @param kyoTelNo 契約お客さま電話番号
     */
    public void setKyoTelNo(String kyoTelNo) {
        this.kyoTelNo = kyoTelNo;
    }

    /**
     * 契約お客さま登録番号を返す。
     *
     * @return 契約お客さま登録番号
     */
    public BigDecimal getKyoTrkNo() {
        return kyoTrkNo;
    }

    /**
     * 契約お客さま登録番号を設定する。
     *
     * @param kyoTrkNo 契約お客さま登録番号
     */
    public void setKyoTrkNo(BigDecimal kyoTrkNo) {
        this.kyoTrkNo = kyoTrkNo;
    }

    /**
     * 電話番号通番を返す。
     *
     * @return 電話番号通番
     */
    public BigDecimal getTelNoTsno() {
        return telNoTsno;
    }

    /**
     * 電話番号通番を設定する。
     *
     * @param telNoTsno 電話番号通番
     */
    public void setTelNoTsno(BigDecimal telNoTsno) {
        this.telNoTsno = telNoTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KyotlrkId that)) {
            return false;
        }
        return Objects.equals(kyoTelNo, that.kyoTelNo)
            && Objects.equals(kyoTrkNo, that.kyoTrkNo)
            && Objects.equals(telNoTsno, that.telNoTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(kyoTelNo, kyoTrkNo, telNoTsno);
    }
}
