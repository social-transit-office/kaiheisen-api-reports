package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.kyotlrk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrk.KyotlrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrk.KyotlrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrk.write.KyotlrkColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrk.write.KyotlrkWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrk.write.KyotlrkWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CiriusAuditColumnAutoSetter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaAuditColumnAccessBinder;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** KYOTLRK データアクセスの JPA 実装。 */
public class KyotlrkDataAccessImpl extends DataAccessBase implements KyotlrkDataAccess {

    private static final String TABLE_NAME = "KYOTLRK";

    private static final ReflectiveRecordEntityMapper<KyotlrkRecord, KyotlrkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KyotlrkRecord.class, KyotlrkEntity.class);

    private static final ReflectiveRecordEntityMapper<KyotlrkWriteRecord, KyotlrkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KyotlrkWriteRecord.class, KyotlrkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KyotlrkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KyotlrkWriteRecord, KyotlrkEntity, KyotlrkColumn> insertSupport;

    /**
     * JPA 実装を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public KyotlrkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に時刻プロバイダを差し替えて生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock 時刻プロバイダ
     */
    KyotlrkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えて生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock 時刻プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    KyotlrkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KyotlrkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KyotlrkWriteDefinition.INSTANCE,
                        KyotlrkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KyotlrkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<KyotlrkRecord> findLatestByKyoTelNoAndKyoTrkNo(String kyoTelNo, Long kyoTrkNo) {
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        var builder = criteriaBuilderFactory.create(entityManager, KyotlrkEntity.class);
        builder.where("kyoTelNo")
                .eq(safeKyoTelNo);
        builder.where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .orderByDesc("telNoTsno");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<KyotlrkRecord> findByPK(String kyoTelNo, Long kyoTrkNo, Integer telNoTsno) {
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        BigDecimal safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        BigDecimal safeTelNoTsno = safeInteger(telNoTsno, "telNoTsno");
        var builder = criteriaBuilderFactory.create(entityManager, KyotlrkEntity.class);
        builder.where("kyoTelNo")
                .eq(safeKyoTelNo)
                .where("kyoTrkNo")
                .eq(safeKyoTrkNo)
                .where("telNoTsno")
                .eq(safeTelNoTsno);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", 13);
    }
}
