package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.irslhs;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslhs.IrslhsDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslhs.IrslhsRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslhs.write.IrslhsColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslhs.write.IrslhsWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslhs.write.IrslhsWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CiriusAuditColumnAutoSetter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaAuditColumnAccessBinder;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * 開閉栓ＩＲＩＳ請求回収基盤送信ログ（閉栓）（IRSLHS）の JPA 実装。
 */
public class IrslhsDataAccessImpl extends DataAccessBase implements IrslhsDataAccess {

    private static final String TABLE_NAME = "IRSLHS";

    private static final ReflectiveRecordEntityMapper<IrslhsRecord, IrslhsEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(IrslhsRecord.class, IrslhsEntity.class);

    private static final ReflectiveRecordEntityMapper<IrslhsWriteRecord, IrslhsEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(IrslhsWriteRecord.class, IrslhsEntity.class);

    private final EntityManager entityManager;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<IrslhsEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<IrslhsWriteRecord, IrslhsEntity, IrslhsColumn> insertSupport;

    /**
     * 標準依存で初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public IrslhsDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this(entityManager, criteriaBuilderFactory, new DefaultSystemDateTimeProvider());
    }

    IrslhsDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    IrslhsDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(IrslhsEntity.class);
        this.auditColumnAutoSetter = Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport = new JpaInsertSupport<>(
                this.entityManager,
                TABLE_NAME,
                IrslhsWriteDefinition.INSTANCE,
                IrslhsEntity.class,
                WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(IrslhsWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity -> auditColumnAutoSetter.applyForInsert(
                        auditColumnAccessBinder.bind(entity),
                        context,
                        clock));
    }

    @Override
    public Optional<IrslhsRecord> findByPK(Integer isrkLogTsno) {
        Objects.requireNonNull(isrkLogTsno, "isrkLogTsno");
        IrslhsEntity entity = entityManager.find(IrslhsEntity.class, BigDecimal.valueOf(isrkLogTsno.longValue()));
        return Optional.ofNullable(entity).map(READ_MAPPER::toRecord);
    }
}
