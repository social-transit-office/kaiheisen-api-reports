package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mnzkiy;

/** MNZKIY データアクセスの JPA 実装。 */
public class MnzkiyDataAccessImpl extends DataAccessBase implements MnzkiyDataAccess {

    /** 操作対象のテーブル名。 */
    private static final String TABLE_NAME = "MNZKIY";

    /** 主キー一致更新・削除で共通利用する WHERE 句。 */
    private static final String PK_WHERE_CLAUSE = "SVKY_NO = ? AND MNZI_TKYS_YMD = ?";

    /** Entity と read record の相互変換を担うマッパー。 */
    private static final ReflectiveRecordEntityMapper<MnzkiyRecord, MnzkiyEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(MnzkiyRecord.class, MnzkiyEntity.class);

    /** Entity と write record の相互変換を担うマッパー。 */
    private static final ReflectiveRecordEntityMapper<MnzkiyWriteRecord, MnzkiyEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            MnzkiyWriteRecord.class, MnzkiyEntity.class);

    /** JPA 操作用の EntityManager。 */
    private final EntityManager entityManager;

    /** 検索条件構築に利用する CriteriaBuilderFactory。 */
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** 監査列補完に利用する現在時刻プロバイダ。 */
    private final SystemDateTimeProvider clock;

    /** エンティティの監査列アクセスを束ねるバインダ。 */
    private final JpaAuditColumnAccessBinder<MnzkiyEntity> auditColumnAccessBinder;

    /** CIRIUS 共通の監査列自動設定ロジック。 */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** Insert 用の JPA 書き込み支援。 */
    private final JpaInsertSupport<MnzkiyWriteRecord, MnzkiyEntity, MnzkiyColumn> insertSupport;

    /** Update 用の JPA 書き込み支援。 */
    private final JpaTargetColumnUpdateSupport<MnzkiyWriteRecord, MnzkiyEntity, MnzkiyColumn>
            updateSupport;

    /** 標準依存で初期化する。 */
    public MnzkiyDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えて初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock 時刻プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    MnzkiyDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(MnzkiyEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        MnzkiyWriteDefinition.INSTANCE,
                        MnzkiyEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        MnzkiyWriteDefinition.INSTANCE,
                        MnzkiyEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    /** 1 行登録する。 */
    @Override
    public int insert(MnzkiyWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    /** 主キーと排他制御順序番号を条件に 1 行削除する。 */
    @Override
    public int deleteBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
            Long svkyNo, LocalDate mnziTkysYmd, Integer recUpdJnjNo) {
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        preparedWhere(svkyNo, mnziTkysYmd),
                        recUpdJnjNo,
                        MnzkiyDataAccess.CallSpecs.DELETE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier());
    }

    /** 主キーで 1 行取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmd(Long svkyNo, LocalDate mnziTkysYmd) {
        SqlCapture.set(
                "SELECT * FROM MNZKIY WHERE SVKY_NO = :svkyNo AND MNZI_TKYS_YMD = :mnziTkysYmd");
        var builder = criteriaBuilderFactory.create(entityManager, MnzkiyEntity.class);
        JpaWhere.eq(builder, "svkyNo", safeLong(svkyNo, "svkyNo"));
        JpaWhere.eq(builder, "mnziTkysYmd", safeYmd(mnziTkysYmd, "mnziTkysYmd"));
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    /** サービス契約番号と免税適用終了年月日で 1 行取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTyenYmd(Long svkyNo, LocalDate mnziTyenYmd) {
        SqlCapture.set(
                "SELECT * FROM MNZKIY WHERE SVKY_NO = :svkyNo AND MNZI_TYEN_YMD = :mnziTyenYmd"
                        + " ORDER BY MNZI_TKYS_YMD DESC FETCH FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, MnzkiyEntity.class);
        JpaWhere.eq(builder, "svkyNo", safeLong(svkyNo, "svkyNo"));
        JpaWhere.eq(builder, "mnziTyenYmd", safeYmd(mnziTyenYmd, "mnziTyenYmd"));
        builder.orderByDesc("mnziTkysYmd");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    /** 指定開始年月日より後の最初の 1 行を取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmdGreaterThan(
            Long svkyNo, LocalDate mnziTkysYmd) {
        SqlCapture.set(
                "SELECT * FROM MNZKIY WHERE SVKY_NO = :svkyNo AND MNZI_TKYS_YMD > :mnziTkysYmd"
                        + " ORDER BY MNZI_TKYS_YMD FETCH FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, MnzkiyEntity.class);
        JpaWhere.eq(builder, "svkyNo", safeLong(svkyNo, "svkyNo"));
        JpaWhere.gt(builder, "mnziTkysYmd", safeYmd(mnziTkysYmd, "mnziTkysYmd"));
        builder.orderByAsc("mnziTkysYmd");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    /** 基準日を含む有効期間のうち最新開始日の 1 行を取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmdAndMnziTyenYmd(
            Long svkyNo, LocalDate knskKjnYmd) {
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        SqlCapture.set(
                "SELECT * FROM MNZKIY WHERE SVKY_NO = :svkyNo AND MNZI_TKYS_YMD <= :knskKjnYmd"
                        + " AND MNZI_TYEN_YMD >= :knskKjnYmd ORDER BY MNZI_TKYS_YMD DESC FETCH"
                        + " FIRST 1 ROWS ONLY");
        var builder = criteriaBuilderFactory.create(entityManager, MnzkiyEntity.class);
        JpaWhere.eq(builder, "svkyNo", safeLong(svkyNo, "svkyNo"));
        JpaWhere.le(builder, "mnziTkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "mnziTyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("mnziTkysYmd");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    /** U001 条件で 1 行更新する。 */
    @Override
    public int updateBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
            MnzkiyWriteRecord record,
            Set<MnzkiyColumn> targetColumns,
            MnzkiySvkyTkysUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return update(
                record,
                targetColumns,
                updateKey.svkyNo(),
                updateKey.mnziTkysYmd(),
                updateKey.recUpdJnjNo(),
                context,
                MnzkiyDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO_U001.identifier());
    }

    /**
     * 主キー一致行を更新する。
     *
     * @param record 更新内容
     * @param targetColumns 更新対象列
     * @param svkyNo サービス契約番号
     * @param mnziTkysYmd 免税適用開始年月日
     * @param recUpdJnjNo 排他制御順序番号
     * @param context 監査コンテキスト
     * @param daId DA 識別子
     * @return 更新件数
     */
    private int update(
            MnzkiyWriteRecord record,
            Set<MnzkiyColumn> targetColumns,
            Long svkyNo,
            LocalDate mnziTkysYmd,
            Integer recUpdJnjNo,
            AuditContext context,
            String daId) {
        Set<MnzkiyColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        return updateSupport.update(
                record,
                canonicalTargetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(recUpdJnjNo, 0)),
                preparedWhere(svkyNo, mnziTkysYmd),
                recUpdJnjNo,
                daId);
    }

    /**
     * 主キー条件に対応する PreparedWhere を生成する。
     *
     * @param svkyNo サービス契約番号
     * @param mnziTkysYmd 免税適用開始年月日
     * @return PreparedWhere
     */
    private PreparedWhere preparedWhere(Long svkyNo, LocalDate mnziTkysYmd) {
        BigDecimal safeSvkyNo = safeLong(svkyNo, "svkyNo");
        BigDecimal safeMnziTkysYmd = safeYmd(mnziTkysYmd, "mnziTkysYmd");
        return PreparedWhere.of(
                PK_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeSvkyNo);
                    return JpaBind.set(query, nextIndex, safeMnziTkysYmd);
                },
                safeSvkyNo,
                safeMnziTkysYmd);
    }

    /**
     * 更新対象列を EnumSet 化して canonical 形式に整える。
     *
     * @param targetColumns 更新対象列
     * @return canonical 化した更新対象列
     */
    private static Set<MnzkiyColumn> canonicalTargetColumns(Set<MnzkiyColumn> targetColumns) {
        EnumSet<MnzkiyColumn> canonicalColumns = EnumSet.noneOf(MnzkiyColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }

    /**
     * Long 条件値を BigDecimal に変換し null-safe に補完する。
     *
     * @param value 入力値
     * @param paramName パラメータ名
     * @return 補完後の値
     */
    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    /**
     * 年月日条件値を BigDecimal に変換し null-safe に補完する。
     *
     * @param value 入力値
     * @param paramName パラメータ名
     * @return 補完後の値
     */
    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
