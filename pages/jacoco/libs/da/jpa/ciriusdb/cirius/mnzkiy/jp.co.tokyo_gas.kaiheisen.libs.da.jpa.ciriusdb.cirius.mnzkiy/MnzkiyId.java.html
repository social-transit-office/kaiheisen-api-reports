package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mnzkiy;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** MNZKIY テーブルの複合主キー。 */
public class MnzkiyId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal svkyNo;

    private BigDecimal mnziTkysYmd;

    /** JPA 用のデフォルトコンストラクタ。 */
    public MnzkiyId() {
    }

    /** 主キー項目を指定して生成する。 */
    public MnzkiyId(BigDecimal svkyNo, BigDecimal mnziTkysYmd) {
        this.svkyNo = svkyNo;
        this.mnziTkysYmd = mnziTkysYmd;
    }

    /**
     * サービス契約番号を返します。
     *
     * @return サービス契約番号
     */
    public BigDecimal getSvkyNo() {
        return svkyNo;
    }

    /**
     * サービス契約番号を設定します。
     *
     * @param svkyNo サービス契約番号
     */
    public void setSvkyNo(BigDecimal svkyNo) {
        this.svkyNo = svkyNo;
    }

    /**
     * 免税適用開始年月日を返します。
     *
     * @return 免税適用開始年月日
     */
    public BigDecimal getMnziTkysYmd() {
        return mnziTkysYmd;
    }

    /**
     * 免税適用開始年月日を設定します。
     *
     * @param mnziTkysYmd 免税適用開始年月日
     */
    public void setMnziTkysYmd(BigDecimal mnziTkysYmd) {
        this.mnziTkysYmd = mnziTkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof MnzkiyId that)) {
            return false;
        }
        return Objects.equals(svkyNo, that.svkyNo)
                && Objects.equals(mnziTkysYmd, that.mnziTkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(svkyNo, mnziTkysYmd);
    }
}
