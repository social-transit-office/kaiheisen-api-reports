package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.shyfrke;

import java.io.Serial;
import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 社用振替（SHYFRKE）の複合主キー。 */
public class ShyfrkeId implements Serializable {

    /** シリアルバージョンUID。 */
    @Serial
    private static final long serialVersionUID = 1L;

    /** 支払契約番号。 */
    private BigDecimal sihKiyNo;

    /** 適用開始年月日。 */
    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public ShyfrkeId() {
    }

    /**
     * コンストラクタ。
     *
     * @param sihKiyNo 支払契約番号
     * @param tkysYmd 適用開始年月日
     */
    public ShyfrkeId(BigDecimal sihKiyNo, BigDecimal tkysYmd) {
        this.sihKiyNo = sihKiyNo;
        this.tkysYmd = tkysYmd;
    }

    /**
     * 支払契約番号を返す。
     *
     * @return 支払契約番号
     */
    public BigDecimal getSihKiyNo() {
        return sihKiyNo;
    }

    /**
     * 支払契約番号を設定する。
     *
     * @param sihKiyNo 支払契約番号
     */
    public void setSihKiyNo(BigDecimal sihKiyNo) {
        this.sihKiyNo = sihKiyNo;
    }

    /**
     * 適用開始年月日を返す。
     *
     * @return 適用開始年月日
     */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /**
     * 適用開始年月日を設定する。
     *
     * @param tkysYmd 適用開始年月日
     */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (!(obj instanceof ShyfrkeId other)) {
            return false;
        }
        return Objects.equals(sihKiyNo, other.sihKiyNo)
                && Objects.equals(tkysYmd, other.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(sihKiyNo, tkysYmd);
    }
}
