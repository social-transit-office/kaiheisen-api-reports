package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.shyfrke;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.ShyfrkeDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.shyfrke.ShyfrkeRecord;

/** SHYFRKE データアクセスの JPA 実装。 */
public class ShyfrkeDataAccessImpl extends DataAccessBase implements ShyfrkeDataAccess {

    private static final String TABLE_NAME = "SHYFRKE";

    private static final String UPDATE_WHERE_CLAUSE =
            "SIH_KIY_NO = ? AND TKYS_YMD <= ? AND TYEN_YMD >= ?";

    private static final ReflectiveRecordEntityMapper<ShyfrkeRecord, ShyfrkeEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(ShyfrkeRecord.class, ShyfrkeEntity.class);

    private static final ReflectiveRecordEntityMapper<ShyfrkeWriteRecord, ShyfrkeEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            ShyfrkeWriteRecord.class, ShyfrkeEntity.class);

    private final EntityManager entityManager;

    @SuppressWarnings("unused")
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<ShyfrkeEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaTargetColumnUpdateSupport<ShyfrkeWriteRecord, ShyfrkeEntity, ShyfrkeColumn>
            updateSupport;

    /**
     * コンストラクタ。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public ShyfrkeDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    ShyfrkeDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(ShyfrkeEntity.class);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        ShyfrkeWriteDefinition.INSTANCE,
                        ShyfrkeEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public Optional<ShyfrkeRecord> findByPK(Long sihKiyNo, LocalDate tkysYmd) {
        Objects.requireNonNull(sihKiyNo, "sihKiyNo");
        Objects.requireNonNull(tkysYmd, "tkysYmd");
        ShyfrkeId id =
                new ShyfrkeId(JpaDbWriteValues.bigDecimal(sihKiyNo), JpaDbWriteValues.ymd(tkysYmd));
        ShyfrkeEntity entity = entityManager.find(ShyfrkeEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public int updateBySihKiyNoAndTkysYmdAndTyenYmd(
            ShyfrkeWriteRecord record,
            Set<ShyfrkeColumn> targetColumns,
            ShyfrkeUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        BigDecimal safeSihKiyNo =
                WhereParamGuard.complement(
                        JpaDbWriteValues.bigDecimal(updateKey.sihKiyNo()),
                        "sihKiyNo",
                        BigDecimal.ZERO);
        BigDecimal safeKnskKjnYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(updateKey.knskKjnYmd()),
                        "knskKjnYmd",
                        BigDecimal.ZERO);
        int currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);

        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                PreparedWhere.of(
                        UPDATE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(query, startIndex, safeSihKiyNo);
                            nextIndex = JpaBind.set(query, nextIndex, safeKnskKjnYmd);
                            return JpaBind.set(query, nextIndex, safeKnskKjnYmd);
                        },
                        safeSihKiyNo,
                        safeKnskKjnYmd,
                        safeKnskKjnYmd),
                updateKey.recUpdJnjNo(),
                ShyfrkeDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_TYEN_YMD);
    }
}
