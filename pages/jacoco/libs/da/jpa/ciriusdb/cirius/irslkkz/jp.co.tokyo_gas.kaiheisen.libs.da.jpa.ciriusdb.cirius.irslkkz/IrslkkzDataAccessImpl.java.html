package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.irslkkz;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.IrslkkzDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.IrslkkzRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.write.IrslkkzColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.write.IrslkkzWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.write.IrslkkzWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CiriusAuditColumnAutoSetter;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaAuditColumnAccessBinder;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * IRSLKKZ データアクセスの JPA 実装.
 */
public class IrslkkzDataAccessImpl extends DataAccessBase implements IrslkkzDataAccess {

    private static final String TABLE_NAME = "IRSLKKZ";

    static final ReflectiveRecordEntityMapper<IrslkkzRecord, IrslkkzEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(IrslkkzRecord.class, IrslkkzEntity.class);

    private static final ReflectiveRecordEntityMapper<IrslkkzWriteRecord, IrslkkzEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            IrslkkzWriteRecord.class, IrslkkzEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<IrslkkzEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<IrslkkzWriteRecord, IrslkkzEntity, IrslkkzColumn> insertSupport;

    /**
     * 標準依存で初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public IrslkkzDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(entityManager, criteriaBuilderFactory, new DefaultSystemDateTimeProvider());
    }

    IrslkkzDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    IrslkkzDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(IrslkkzEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        IrslkkzWriteDefinition.INSTANCE,
                        IrslkkzEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(IrslkkzWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<IrslkkzRecord> findByPK(Integer isrkLogTsno) {
        BigDecimal safeIsrkLogTsno =
                WhereParamGuard.complement(
                        JpaDbWriteValues.bigDecimal(isrkLogTsno), "isrkLogTsno", BigDecimal.ZERO);
        var builder = criteriaBuilderFactory.create(entityManager, IrslkkzEntity.class);
        builder.where("isrkLogTsno").eq(safeIsrkLogTsno);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }
}
