    @Override
    public Optional<GmkskkdRecord> findByGasMtstNoAndKnsYmd(Long gasMtstNo, LocalDate knsYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GmkskkdEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.eq(builder, "knsYmd", safeKnsYmd);
        builder.orderByDesc("knsKkaTsno");
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    @Override
    public List<GmkskkdRecord> findAllByKnsYmdAndMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
            LocalDate knsYmdFrom,
            LocalDate knsYmdTo,
            Long mtgMtstNo,
            LocalDate mgmtKnsYmd,
            Integer mgmtKnkkTsno) {
        BigDecimal safeMtgMtstNo = safeLong(mtgMtstNo, "mtgMtstNo");
        BigDecimal safeMgmtKnkkTsno = safeInteger(mgmtKnkkTsno, "mgmtKnkkTsno");

        if (BigDecimal.ZERO.compareTo(safeMtgMtstNo) == 0) {
            return List.of();
        }

        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        BigDecimal safeMgmtKnsYmd = safeYmd(mgmtKnsYmd, "mgmtKnsYmd");
        var builder = criteriaBuilderFactory.create(entityManager, GmkskkdEntity.class);
        JpaWhere.ge(builder, "knsYmd", safeKnsYmdFrom);
        JpaWhere.le(builder, "knsYmd", safeKnsYmdTo);
        JpaWhere.eq(builder, "mtgMtstNo", safeMtgMtstNo);
        JpaWhere.notEq(builder, "mtgMtstNo", JpaDbWriteValues.bigDecimal(0L));
        JpaWhere.eq(builder, "mgmtKnsYmd", safeMgmtKnsYmd);
        JpaWhere.eq(builder, "mgmtKnkkTsno", safeMgmtKnkkTsno);
        builder.orderByAsc("gasMtstNo").orderByDesc("knsYmd").orderByDesc("knsKkaTsno");

        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return JpaDbWriteValues.bigDecimal(WhereParamGuard.complement(value, paramName, 0L));
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return JpaDbWriteValues.bigDecimal(WhereParamGuard.complement(value, paramName, 0));
    }

    @Override
    public List<GmkskkdKkmtKnsResultRecord> findAllByGasMtstNoAndKnsYmdWithKkmtFlgAndKnsKkaTsno(
            Long gasMtstNo, LocalDate knsYmdFrom, LocalDate knsYmdTo, Integer knsKkaTsno) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        BigDecimal safeKnsKkaTsno = safeInteger(knsKkaTsno, "knsKkaTsno");

        var gmkskkdBuilder = criteriaBuilderFactory.create(entityManager, GmkskkdEntity.class);
        JpaWhere.eq(gmkskkdBuilder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.ge(gmkskkdBuilder, "knsYmd", safeKnsYmdFrom);
        JpaWhere.le(gmkskkdBuilder, "knsYmd", safeKnsYmdTo);
        gmkskkdBuilder.orderByDesc("knsYmd").orderByDesc("knsKkaTsno");

        var ktmgmkkBuilder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        JpaWhere.eq(ktmgmkkBuilder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.ge(ktmgmkkBuilder, "knsYmd", safeKnsYmdFrom);
        ktmgmkkBuilder.orderByDesc("knsYmd").orderByDesc("knsKkaTsno");
        List<KtmgmkkEntity> ktmgmkkEntities = ktmgmkkBuilder.getResultList().stream()
                .filter(entity -> isBeforeOrSameResult(entity, safeKnsYmdTo, safeKnsKkaTsno))
                .toList();

        return concatKkmtResults(ktmgmkkEntities, gmkskkdBuilder.getResultList());
    }

    @Override
    public List<GmkskkdKnsResultRecord> findAllByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            Long gasMtstNo, LocalDate knsYmdFrom, LocalDate knsYmdTo, Integer knsKkaTsno) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal knsYmdToValue = safeYmd(knsYmdTo, "knsYmdTo");
        BigDecimal safeKnsKkaTsno = safeInteger(knsKkaTsno, "knsKkaTsno");
        var builder = criteriaBuilderFactory.create(entityManager, GmkskkdEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.ge(builder, "knsYmd", safeKnsYmdFrom);
        JpaWhere.le(builder, "knsYmd", knsYmdToValue);
        builder.whereOr()
                .where("knsYmd")
                .lt(knsYmdToValue)
                .where("knsKkaTsno")
                .le(safeKnsKkaTsno)
                .endOr()
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");

        return builder.getResultList().stream()
                .map(
                        entity ->
                                new GmkskkdKnsResultRecord(
                                        DateConverter.toLocalDate(entity.getKnsYmd()),
                                        JpaDbReadValues.integerOrNullIfZero(entity.getKnsKkaTsno()),
                                        FixedLengthFormatter.suppressSpace(
                                                entity.getKnkkSksiJysb()),
                                    JpaDbReadValues.longValue(entity.getGmtShjSu()),
                                        entity.getGasSyry(),
                                        entity.getSgyjKytiSyry(),
                                    JpaDbReadValues.integerValue(entity.getShyDdsu()),
                                        FixedLengthFormatter.suppressSpace(entity.getSjskKbn()),
                                        FixedLengthFormatter.suppressSpace(entity.getKnsJsjkCd())))
                .toList();
    }

    @Override
    public Optional<GmkskkdRecord> findByPK(Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(knsYmd, "knsYmd");
        Objects.requireNonNull(knsKkaTsno, "knsKkaTsno");

        GmkskkdId id =
                new GmkskkdId(
                        JpaDbWriteValues.bigDecimal(gasMtstNo),
                        JpaDbWriteValues.ymd(knsYmd),
                        JpaDbWriteValues.bigDecimal(knsKkaTsno));
        GmkskkdEntity entity = entityManager.find(GmkskkdEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }

    private static List<GmkskkdKkmtKnsResultRecord> concatKkmtResults(
            List<KtmgmkkEntity> ktmgmkkEntities,
            List<GmkskkdEntity> gmkskkdEntities) {
        return Stream.concat(
                        ktmgmkkEntities.stream().map(GmkskkdDataAccessImpl::toKkmtKnsResultRecord),
                        gmkskkdEntities.stream().map(GmkskkdDataAccessImpl::toKkmtKnsResultRecord))
                .toList();
    }

    private static GmkskkdKkmtKnsResultRecord toKkmtKnsResultRecord(KtmgmkkEntity entity) {
        return new GmkskkdKkmtKnsResultRecord(
                "1",
                DateConverter.toLocalDate(entity.getKnsYmd()),
                JpaDbReadValues.integerOrNullIfZero(entity.getKnsKkaTsno()),
                FixedLengthFormatter.suppressSpace(entity.getKnkkSksiJysb()),
                JpaDbReadValues.longValue(entity.getGmtShjSu()),
                entity.getGasSyry(),
                entity.getSgyjKytiSyry(),
                JpaDbReadValues.integerValue(entity.getShyDdsu()),
                FixedLengthFormatter.suppressSpace(entity.getSjskKbn()),
                FixedLengthFormatter.suppressSpace(entity.getKnsJsjkCd()));
    }

    private static GmkskkdKkmtKnsResultRecord toKkmtKnsResultRecord(GmkskkdEntity entity) {
        return new GmkskkdKkmtKnsResultRecord(
                "0",
                DateConverter.toLocalDate(entity.getKnsYmd()),
                JpaDbReadValues.integerOrNullIfZero(entity.getKnsKkaTsno()),
                FixedLengthFormatter.suppressSpace(entity.getKnkkSksiJysb()),
                JpaDbReadValues.longValue(entity.getGmtShjSu()),
                entity.getGasSyry(),
                entity.getSgyjKytiSyry(),
                JpaDbReadValues.integerValue(entity.getShyDdsu()),
                FixedLengthFormatter.suppressSpace(entity.getSjskKbn()),
                FixedLengthFormatter.suppressSpace(entity.getKnsJsjkCd()));
    }

    private static boolean isBeforeOrSameResult(
            KtmgmkkEntity entity,
            BigDecimal knsYmd,
            BigDecimal knsKkaTsno) {
        int ymdComparison = entity.getKnsYmd().compareTo(knsYmd);
        return ymdComparison < 0
                || (ymdComparison == 0 && entity.getKnsKkaTsno().compareTo(knsKkaTsno) <= 0);
    }
