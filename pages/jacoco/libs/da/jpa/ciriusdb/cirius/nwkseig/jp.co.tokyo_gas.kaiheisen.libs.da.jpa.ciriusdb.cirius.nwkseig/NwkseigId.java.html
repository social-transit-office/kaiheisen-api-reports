package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.nwkseig;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 営業情報ＮＷ箇所マスター（NWKSEIG）の複合主キー。 */
public class NwkseigId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String nwkCd;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public NwkseigId() { }

    /** 全フィールド指定コンストラクタ。 */
    public NwkseigId(String nwkCd, BigDecimal tkysYmd) {
        this.nwkCd = nwkCd;
        this.tkysYmd = tkysYmd;
    }

    /** nwkCd を返す。 */
    public String getNwkCd() {
        return nwkCd;
    }

    /** nwkCd を設定する。 */
    public void setNwkCd(String nwkCd) {
        this.nwkCd = nwkCd;
    }

    /** tkysYmd を返す。 */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /** tkysYmd を設定する。 */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof NwkseigId that)) {
            return false;
        }
        return Objects.equals(nwkCd, that.nwkCd) && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(nwkCd, tkysYmd);
    }
}
