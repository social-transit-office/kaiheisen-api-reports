package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.nwkseig;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.nwkseig.NwkseigDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.nwkseig.NwkseigRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** 営業情報ＮＷ箇所マスター（NWKSEIG）データアクセスの JPA 実装。 */
public class NwkseigDataAccessImpl extends DataAccessBase implements NwkseigDataAccess {

    private static final int NWK_CD_LENGTH = 7;

    private static final ReflectiveRecordEntityMapper<NwkseigRecord, NwkseigEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(NwkseigRecord.class, NwkseigEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /**
     * コンストラクタ。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public NwkseigDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<NwkseigRecord> findByNwkCdAndTkysYmdAndTyenYmd(
            String nwkCd, LocalDate knskKjnYmd) {
        String safeNwkCd = safeNwkCd(nwkCd, "nwkCd");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, NwkseigEntity.class);
        JpaWhere.eq(builder, "nwkCd", safeNwkCd);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<NwkseigRecord> findByPK(String nwkCd, LocalDate tkysYmd) {
        NwkseigId id = new NwkseigId(safeNwkCd(nwkCd, "nwkCd"), safeYmd(tkysYmd, "tkysYmd"));
        NwkseigEntity entity = entityManager.find(NwkseigEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }

    private static String safeNwkCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, NWK_CD_LENGTH);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
