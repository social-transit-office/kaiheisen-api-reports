package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ssvsbhw;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ssvsbhw.SsvsbhwDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ssvsbhw.SsvsbhwRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** SSVSBHW データアクセスの JPA 実装。 */
public class SsvsbhwDataAccessImpl extends DataAccessBase implements SsvsbhwDataAccess {

    private static final int SQSVS_HWSB_LENGTH = 4;

    private static final int SIK_SVC_SHBT_LENGTH = 2;

    private static final ReflectiveRecordEntityMapper<SsvsbhwRecord, SsvsbhwEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(SsvsbhwRecord.class, SsvsbhwEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public SsvsbhwDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<SsvsbhwRecord> findBySqsvsHwsbAndSikSvcShbtAndKnskKjnYmdOrderByTkysYmdDesc(
            String sqsvsHwsb,
            String sikSvcShbt,
            LocalDate knskKjnYmd) {
        String safeSqsvsHwsb =
                WhereParamGuard.fixedLengthNotNull(sqsvsHwsb, "sqsvsHwsb", SQSVS_HWSB_LENGTH);
        String safeSikSvcShbt =
                WhereParamGuard.fixedLengthNotNull(sikSvcShbt, "sikSvcShbt", SIK_SVC_SHBT_LENGTH);
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, SsvsbhwEntity.class);
        JpaWhere.eq(builder, "sqsvsHwsb", safeSqsvsHwsb);
        JpaWhere.eq(builder, "sikSvcShbt", safeSikSvcShbt);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<SsvsbhwRecord> findByPK(String sqsvsHwsb, String sikSvcShbt, LocalDate tkysYmd) {
        String safeSqsvsHwsb =
                WhereParamGuard.fixedLengthNotNull(sqsvsHwsb, "sqsvsHwsb", SQSVS_HWSB_LENGTH);
        String safeSikSvcShbt =
                WhereParamGuard.fixedLengthNotNull(sikSvcShbt, "sikSvcShbt", SIK_SVC_SHBT_LENGTH);
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        SsvsbhwEntity entity = entityManager.find(
                SsvsbhwEntity.class,
                new SsvsbhwId(safeSqsvsHwsb, safeSikSvcShbt, safeTkysYmd));
        return Optional.ofNullable(entity).map(READ_MAPPER::toRecord);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
