package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ssvsbhw;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 請求サービス種別振分（SSVSBHW）の複合主キー。 */
public class SsvsbhwId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String sqsvsHwsb;

    private String sikSvcShbt;

    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public SsvsbhwId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public SsvsbhwId(String sqsvsHwsb, String sikSvcShbt, BigDecimal tkysYmd) {
        this.sqsvsHwsb = sqsvsHwsb;
        this.sikSvcShbt = sikSvcShbt;
        this.tkysYmd = tkysYmd;
    }

    /**
     * 請求サービス種別振分種別を返す。
     *
     * @return 請求サービス種別振分種別
     */
    public String getSqsvsHwsb() {
        return sqsvsHwsb;
    }

    /**
     * 請求サービス種別振分種別を設定する。
     *
     * @param sqsvsHwsb 請求サービス種別振分種別

    public void setSqsvsHwsb(String sqsvsHwsb) {
        this.sqsvsHwsb = sqsvsHwsb;
    }

    /**
     * 請求サービス種別を返す。
     *
     * @return 請求サービス種別
     */
    public String getSikSvcShbt() {
        return sikSvcShbt;
    }

    /**
     * 請求サービス種別を設定する。
     *
     * @param sikSvcShbt 請求サービス種別

    public void setSikSvcShbt(String sikSvcShbt) {
        this.sikSvcShbt = sikSvcShbt;
    }

    /**
     * 適用開始年月日を返す。
     *
     * @return 適用開始年月日
     */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /**
     * 適用開始年月日を設定する。
     *
     * @param tkysYmd 適用開始年月日
     */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof SsvsbhwId that)) {
            return false;
        }
        return Objects.equals(sqsvsHwsb, that.sqsvsHwsb)
                && Objects.equals(sikSvcShbt, that.sikSvcShbt)
                && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(sqsvsHwsb, sikSvcShbt, tkysYmd);
    }
}
