    TskgmstDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(TskgmstEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(

                        TskgmstWriteDefinition.INSTANCE,
                        TskgmstEntity.class,

        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(

    }

    @Override
    public Optional<TskgmstRecord> findByGasMtstNoAndTksuSvkkJtkbList(
            Long gasMtstNo, List<String> tksuSvkkJtkbList) {
        R003Parameters parameters = DataAccessPhaseRecorder.measure(
                DataAccessPhaseRecorder.PARAMETER_CONVERSION,
                () -> {
                    Objects.requireNonNull(tksuSvkkJtkbList, "tksuSvkkJtkbList");
                    if (tksuSvkkJtkbList.isEmpty()) {
                        return new R003Parameters(null, List.of());
                    }
                    BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
                    List<String> safeStatusList = tksuSvkkJtkbList.stream()
                            .map(value -> safeTksuSvkkJtkb(value, "tksuSvkkJtkbList"))
                            .toList();
                    return new R003Parameters(safeGasMtstNo, safeStatusList);
                });
        if (parameters.tksuSvkkJtkbList().isEmpty()) {
            return Optional.empty();
        }
        var builder = DataAccessPhaseRecorder.measure(
                DataAccessPhaseRecorder.QUERY_CONSTRUCTION,
                () -> {
                    var queryBuilder =
                            criteriaBuilderFactory.create(entityManager, TskgmstEntity.class);
                    JpaWhere.eq(queryBuilder, "gasMtstNo", parameters.gasMtstNo());
                    JpaWhere.in(
                            queryBuilder,
                            "tksuSvkkJtkb",
                            parameters.tksuSvkkJtkbList());
                    queryBuilder.orderByDesc("tkysYmd");
                    return queryBuilder.setMaxResults(1);
                });
        List<TskgmstEntity> entities = DataAccessPhaseRecorder.measure(
                DataAccessPhaseRecorder.SQL_EXECUTION_RESULT_FETCH,
                builder::getResultList);
        return DataAccessPhaseRecorder.measure(
                DataAccessPhaseRecorder.RECORD_CONVERSION,
                () -> entities.stream().map(READ_MAPPER::toRecord).findFirst());
    }

                TskgmstDataAccess.CallSpecs
