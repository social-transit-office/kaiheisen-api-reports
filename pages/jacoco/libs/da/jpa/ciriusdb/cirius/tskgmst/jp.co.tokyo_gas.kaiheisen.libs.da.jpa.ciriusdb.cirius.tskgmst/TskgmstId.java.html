package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.tskgmst;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 託送ガスメーター設置場所テーブルの複合主キー。 */
public class TskgmstId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal tksuSvkyNo;

    private BigDecimal skno;

    /** デフォルトコンストラクタ。 */
    public TskgmstId() { }

    /** 全フィールド指定コンストラクタ。 */
    public TskgmstId(BigDecimal gasMtstNo, BigDecimal tksuSvkyNo, BigDecimal skno) {
        this.gasMtstNo = gasMtstNo;
        this.tksuSvkyNo = tksuSvkyNo;
        this.skno = skno;
    }

    /** gasMtstNoを取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNoを設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** tksuSvkyNoを取得する。 */
    public BigDecimal getTksuSvkyNo() {
        return tksuSvkyNo;
    }

    /** tksuSvkyNoを設定する。 */
    public void setTksuSvkyNo(BigDecimal tksuSvkyNo) {
        this.tksuSvkyNo = tksuSvkyNo;
    }

    /** sknoを取得する。 */
    public BigDecimal getSkno() {
        return skno;
    }

    /** sknoを設定する。 */
    public void setSkno(BigDecimal skno) {
        this.skno = skno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof TskgmstId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo)
                && Objects.equals(tksuSvkyNo, that.tksuSvkyNo)
                && Objects.equals(skno, that.skno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, tksuSvkyNo, skno);
    }
}
