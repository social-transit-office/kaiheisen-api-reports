package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mtsetk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.MtsetkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.MtsetkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.write.MtsetkColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.write.MtsetkUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.write.MtsetkWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtsetk.write.MtsetkWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;

/** メーター設置場所＜小売＞（MTSETK）のデータアクセス実装。 */
public class MtsetkDataAccessImpl extends DataAccessBase implements MtsetkDataAccess {

    private static final String TABLE_NAME = "MTSETK";

    private static final String U001_BASE_WHERE_CLAUSE = "MTST_NO = ? AND TKYS_YMD = ?";

    private static final ReflectiveRecordEntityMapper<MtsetkRecord, MtsetkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(MtsetkRecord.class, MtsetkEntity.class);

    private static final ReflectiveRecordEntityMapper<MtsetkWriteRecord, MtsetkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            MtsetkWriteRecord.class, MtsetkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<MtsetkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<MtsetkWriteRecord, MtsetkEntity, MtsetkColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<MtsetkWriteRecord, MtsetkEntity, MtsetkColumn>
            updateSupport;

    /**
     * JPA ベースの MTSETK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze Persistence CriteriaBuilderFactory
     */
    public MtsetkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    MtsetkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    MtsetkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(MtsetkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        MtsetkWriteDefinition.INSTANCE,
                        MtsetkEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        MtsetkWriteDefinition.INSTANCE,
                        MtsetkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(MtsetkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByMtstNoAndTkysYmd(Long mtstNo, LocalDate tkysYmd, Integer recUpdJnjNo) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                U001_BASE_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    startIndex,
                                                    safeMtstNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    return JpaBind.set(query, nextIndex, safeTkysYmd);
                                },
                                JpaDbWriteValues.bigDecimal(safeMtstNo),
                                safeTkysYmd),
                        recUpdJnjNo,
                        MtsetkDataAccess.CallSpecs.DELETE_BY_MTST_NO_AND_TKYS_YMD.identifier());
    }

    @Override
    public Optional<MtsetkRecord> findByMtstNoAndTkysYmd(Long mtstNo, LocalDate tkysYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        BigDecimal safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        MtsetkId id = new MtsetkId(JpaDbWriteValues.bigDecimal(safeMtstNo), safeTkysYmd);
        MtsetkEntity entity = entityManager.find(MtsetkEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public List<MtsetkRecord> findAllByMtstNoAndTkysYmdAndTyenYmd(
            Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, MtsetkEntity.class);
        JpaWhere.eq(builder, "mtstNo", JpaDbWriteValues.bigDecimal(safeMtstNo));
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);

        List<MtsetkEntity> entities = builder.getResultList();

        return entities.stream().map(READ_MAPPER::toRecord).toList();
    }

    @Override
    public long countByMtstNoAndTkysYmdAndTyenYmd(Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, MtsetkEntity.class);
        JpaWhere.eq(builder, "mtstNo", JpaDbWriteValues.bigDecimal(safeMtstNo));
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        return builder.getResultList().size();
    }

    @Override
    public Optional<MtsetkRecord> findByMtstNoAndTkysYmdAndTyenYmd(
            Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        var builder = criteriaBuilderFactory.create(entityManager, MtsetkEntity.class);
        JpaWhere.eq(builder, "mtstNo", JpaDbWriteValues.bigDecimal(safeMtstNo));
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        builder.orderByDesc("tkysYmd");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByMtstNoAndTkysYmdAndRecUpdJnjNo(
            MtsetkWriteRecord record,
            Set<MtsetkColumn> targetColumns,
            MtsetkUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0)),
                PreparedWhere.of(
                        U001_BASE_WHERE_CLAUSE,
                        (query, startIndex) -> {
                            int nextIndex =
                                    JpaBind.set(
                                            query,
                                            startIndex,
                                            updateKey.mtstNo(),
                                            JpaDbWriteValues::bigDecimal);
                            nextIndex =
                                    JpaBind.set(
                                            query,
                                            nextIndex,
                                            updateKey.tkysYmd(),
                                            JpaDbWriteValues::ymd);
                            return nextIndex;
                        },
                        JpaDbWriteValues.bigDecimal(updateKey.mtstNo()),
                        JpaDbWriteValues.ymd(updateKey.tkysYmd())),
                updateKey.recUpdJnjNo(),
                MtsetkDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier());
    }

    private static Long safeMtstNo(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
