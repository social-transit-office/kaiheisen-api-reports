package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.mtsetk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** メーター設置場所＜小売＞（MTSETK）の複合主キー。 */
public class MtsetkId implements Serializable {

    private static final long serialVersionUID = 1L;

    /** メーター設置場所番号 */
    private BigDecimal mtstNo;

    /** 適用開始年月日 */
    private BigDecimal tkysYmd;

    /** デフォルトコンストラクタ。 */
    public MtsetkId() { }

    /** 全フィールド指定コンストラクタ。 */
    public MtsetkId(BigDecimal mtstNo, BigDecimal tkysYmd) {
        this.mtstNo = mtstNo;
        this.tkysYmd = tkysYmd;
    }

    /** mtstNoを取得する。 */
    public BigDecimal getMtstNo() {
        return mtstNo;
    }

    /** mtstNoを設定する。 */
    public void setMtstNo(BigDecimal mtstNo) {
        this.mtstNo = mtstNo;
    }

    /** tkysYmdを取得する。 */
    public BigDecimal getTkysYmd() {
        return tkysYmd;
    }

    /** tkysYmdを設定する。 */
    public void setTkysYmd(BigDecimal tkysYmd) {
        this.tkysYmd = tkysYmd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof MtsetkId that)) {
            return false;
        }
        return Objects.equals(mtstNo, that.mtstNo) && Objects.equals(tkysYmd, that.tkysYmd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(mtstNo, tkysYmd);
    }
}
