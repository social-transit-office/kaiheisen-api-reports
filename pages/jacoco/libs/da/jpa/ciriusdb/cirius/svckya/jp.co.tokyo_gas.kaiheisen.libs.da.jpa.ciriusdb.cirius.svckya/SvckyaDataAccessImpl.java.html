    @Override
    public Optional<Integer> findMaxKyttSknoBySvkyNo(final Long svkyNo) {
        BigDecimal safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");

        SqlCapture.set("SELECT MAX(KYTT_SKNO) FROM SVCKYA WHERE SVKY_NO = :svkyNo");
        Query query = entityManager.createNativeQuery(
                "SELECT MAX(KYTT_SKNO) FROM SVCKYA WHERE SVKY_NO = :svkyNo");
        query.setParameter("svkyNo", safeSvkyNo);
        Object value = query.getSingleResult();
        if (value == null) {
            return Optional.empty();
        }
        if (value instanceof java.math.BigDecimal bigDecimal) {
            return Optional.ofNullable(JpaDbReadValues.integerOrNullIfZero(bigDecimal));
        }
        if (value instanceof Number number) {
            return Optional.of(number.intValue());
        }
        return Optional.of(Integer.parseInt(value.toString()));
    }

    private static BigDecimal safeSvkyNo(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeSkno(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(java.time.LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static String safeSvkyJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SVKY_JTKB_LENGTH);
    }

    private static List<String> safeSvkyJtkbList(List<String> values, String paramName) {
        return values.stream().map(value -> safeSvkyJtkb(value, paramName)).toList();
    }
