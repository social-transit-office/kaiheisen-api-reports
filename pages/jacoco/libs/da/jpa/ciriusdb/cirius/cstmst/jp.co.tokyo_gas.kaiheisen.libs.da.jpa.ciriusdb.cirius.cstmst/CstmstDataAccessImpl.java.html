package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.cstmst;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** CSTMST データアクセスの JPA 実装。 */
public class CstmstDataAccessImpl extends DataAccessBase implements CstmstDataAccess {

    private static final int CNST_SHBT_CD_LENGTH = 5;

    private static final int CNST_CD_LENGTH = 10;

    private static final int TIGN_SGN_LENGTH = 1;

    private static final ReflectiveRecordEntityMapper<CstmstRecord, CstmstEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(CstmstRecord.class, CstmstEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public CstmstDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<CstmstRecord> findAllByCnstShbtCd(String cnstShbtCd) {
        String safeCnstShbtCd = safeCnstShbtCd(cnstShbtCd, "cnstShbtCd");
        var builder = criteriaBuilderFactory.create(entityManager, CstmstEntity.class);
        JpaWhere.eq(builder, "cnstShbtCd", safeCnstShbtCd);
        builder.orderByAsc("cnstCd");
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public Optional<CstmstRecord> findByCnstShbtCdAndCnstCdAndTkysYmdAndTyenYmdAndTignSgn(
            String cnstShbtCd, String cnstCd, LocalDate knskKjnYmd, String tignSgn) {
        if (tignSgn == null || tignSgn.isBlank()) {
            return Optional.empty();
        }
        String safeCnstShbtCd = safeCnstShbtCd(cnstShbtCd, "cnstShbtCd");
        String safeCnstCd = safeCnstCd(cnstCd, "cnstCd");
        BigDecimal safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        String safeTignSgn = safeTignSgn(tignSgn, "tignSgn");
        var builder = criteriaBuilderFactory.create(entityManager, CstmstEntity.class);
        JpaWhere.eq(builder, "cnstShbtCd", safeCnstShbtCd);
        JpaWhere.eq(builder, "cnstCd", safeCnstCd);
        JpaWhere.le(builder, "tkysYmd", safeKnskKjnYmd);
        JpaWhere.ge(builder, "tyenYmd", safeKnskKjnYmd);
        JpaWhere.notEq(builder, "tignSgn", safeTignSgn);
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(MAPPER::toRecord)
                .findFirst();
    }

    private static String safeCnstShbtCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, CNST_SHBT_CD_LENGTH);
    }

    private static String safeCnstCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, CNST_CD_LENGTH);
    }

    private static String safeTignSgn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TIGN_SGN_LENGTH);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }
}
