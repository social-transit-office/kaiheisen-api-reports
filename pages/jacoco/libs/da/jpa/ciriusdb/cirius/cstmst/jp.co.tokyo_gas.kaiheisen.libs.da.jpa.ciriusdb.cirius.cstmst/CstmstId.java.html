package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.cstmst;

import java.io.Serializable;
import java.util.Objects;

/** コンスタントマスター（CSTMST）の複合主キー。 */
public class CstmstId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String cnstShbtCd;

    private String cnstCd;

    /** デフォルトコンストラクタ。 */
    public CstmstId() { }

    /** 全フィールド指定コンストラクタ。 */
    public CstmstId(String cnstShbtCd, String cnstCd) {
        this.cnstShbtCd = cnstShbtCd;
        this.cnstCd = cnstCd;
    }

    /** cnstShbtCdを取得する。 */
    public String getCnstShbtCd() {
        return cnstShbtCd;
    }

    /** cnstShbtCdを設定する。 */
    public void setCnstShbtCd(String cnstShbtCd) {
        this.cnstShbtCd = cnstShbtCd;
    }

    /** cnstCdを取得する。 */
    public String getCnstCd() {
        return cnstCd;
    }

    /** cnstCdを設定する。 */
    public void setCnstCd(String cnstCd) {
        this.cnstCd = cnstCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof CstmstId that)) {
            return false;
        }
        return Objects.equals(cnstShbtCd, that.cnstShbtCd) && Objects.equals(cnstCd, that.cnstCd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(cnstShbtCd, cnstCd);
    }
}
