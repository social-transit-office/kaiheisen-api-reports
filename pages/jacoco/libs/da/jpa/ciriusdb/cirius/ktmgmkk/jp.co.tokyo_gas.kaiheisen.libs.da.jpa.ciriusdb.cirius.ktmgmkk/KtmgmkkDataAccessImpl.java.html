/** KTMGMKK（確定前ガスメーター検針結果）データアクセスの JPA 実装。 */
public class KtmgmkkDataAccessImpl extends DataAccessBase implements KtmgmkkDataAccess {

    private static final String TABLE_NAME = "KTMGMKK";

    private static final int KNKK_SKSI_JYSB_LENGTH = 3;

    private static final ReflectiveRecordEntityMapper<KtmgmkkRecord, KtmgmkkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KtmgmkkRecord.class, KtmgmkkEntity.class);

    private static final ReflectiveRecordEntityMapper<KtmgmkkWriteRecord, KtmgmkkEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            KtmgmkkWriteRecord.class, KtmgmkkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KtmgmkkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KtmgmkkWriteRecord, KtmgmkkEntity, KtmgmkkColumn> insertSupport;

    /**
     * JPA ベースの KTMGMKK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public KtmgmkkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KtmgmkkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KtmgmkkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KtmgmkkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KtmgmkkWriteDefinition.INSTANCE,
                        KtmgmkkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KtmgmkkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno, Integer recUpdJnjNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        BigDecimal safeKnsKkaTsno = safeInteger(knsKkaTsno, "knsKkaTsno");

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GAS_MTST_NO = ? AND KNS_YMD = ? AND KNS_KKA_TSNO = ?",
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeGasMtstNo);
                                    nextIndex = JpaBind.set(query, nextIndex, safeKnsYmd);
                                    return JpaBind.set(query, nextIndex, safeKnsKkaTsno);
                                },
                                safeGasMtstNo,
                                safeKnsYmd,
                                safeKnsKkaTsno),
                        recUpdJnjNo,
                        KtmgmkkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO);
    }

    @Override
    public Optional<KtmgmkkRecord> findByPK(Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        BigDecimal safeKnsKkaTsno = safeInteger(knsKkaTsno, "knsKkaTsno");

        KtmgmkkId id =
                new KtmgmkkId(safeGasMtstNo, safeKnsYmd, safeKnsKkaTsno);
        KtmgmkkEntity entity = entityManager.find(KtmgmkkEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNo(Long gasMtstNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");

        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        builder.orderByDesc("knsYmd").orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmd(Long gasMtstNo, LocalDate knsYmd) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");

        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.eq(builder, "knsYmd", safeKnsYmd);
        builder.orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmdAndTksuSvkyNo(
            Long gasMtstNo, LocalDate knsYmd, Long tksuSvkyNo) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        BigDecimal safeTksuSvkyNo = safeLong(tksuSvkyNo, "tksuSvkyNo");

        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.eq(builder, "knsYmd", safeKnsYmd);
        JpaWhere.eq(builder, "tksuSvkyNo", safeTksuSvkyNo);
        builder.orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
            Long gasMtstNo, LocalDate knsYmd, List<String> knkkSksiJysbList) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return Optional.empty();
        }
        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .eq(safeKnsYmd)
                .where("knkkSksiJysb")
                .in(safeKnkkSksiJysbList)
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
            Long gasMtstNo,
            LocalDate knsYmdFrom,
            LocalDate knsYmdTo,
            List<String> knkkSksiJysbList) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        BigDecimal safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        BigDecimal safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return Optional.empty();
        }
        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knsYmd")
                .ge(safeKnsYmdFrom)
                .where("knsYmd")
                .le(safeKnsYmdTo)
                .where("knkkSksiJysb")
                .in(safeKnkkSksiJysbList)
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnkkSksiJysb(
            Long gasMtstNo, String knkkSksiJysb) {
        if (knkkSksiJysb == null || knkkSksiJysb.isBlank()) {
            return Optional.empty();
        }
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        String safeKnkkSksiJysb = safeKnkkSksiJysb(knkkSksiJysb, "knkkSksiJysb");

        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        builder.where("gasMtstNo")
                .eq(safeGasMtstNo)
                .where("knkkSksiJysb")
                .notEq(safeKnkkSksiJysb)
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");
        return builder.setMaxResults(1)
                .getResultList()
                .stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public long countByGasMtstNoAndKnkkSksiJysb(Long gasMtstNo, List<String> knkkSksiJysbList) {
        BigDecimal safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return 0L;
        }
        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        JpaWhere.eq(builder, "gasMtstNo", safeGasMtstNo);
        JpaWhere.in(builder, "knkkSksiJysb", safeKnkkSksiJysbList);
        return builder.getResultList().size();
    }

    @Override
    public List<KtmgmkkRecord> findAllByMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
            Long mtgMtstNo, LocalDate mgmtKnsYmd, Integer mgmtKnkkTsno) {
        BigDecimal safeMtgMtstNo = safeLong(mtgMtstNo, "mtgMtstNo");
        BigDecimal safeMgmtKnsYmd = safeYmd(mgmtKnsYmd, "mgmtKnsYmd");
        BigDecimal safeMgmtKnkkTsno = safeInteger(mgmtKnkkTsno, "mgmtKnkkTsno");

        var builder = criteriaBuilderFactory.create(entityManager, KtmgmkkEntity.class);
        builder.where("mtgMtstNo")
                .eq(safeMtgMtstNo)
                .where("mtgMtstNo")
                .notEq(BigDecimal.ZERO)
                .where("mgmtKnsYmd")
                .eq(safeMgmtKnsYmd)
                .where("mgmtKnkkTsno")
                .eq(safeMgmtKnkkTsno)
                .orderByAsc("gasMtstNo")
                .orderByDesc("knsYmd")
                .orderByDesc("knsKkaTsno");
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

    private static String safeKnkkSksiJysb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, KNKK_SKSI_JYSB_LENGTH);
    }

    private static List<String> normalizeKnkkSksiJysbList(List<String> knkkSksiJysbList) {
        List<String> nonNullKnkkSksiJysbList =
                Objects.requireNonNull(knkkSksiJysbList, "knkkSksiJysbList");
        if (nonNullKnkkSksiJysbList.isEmpty()) {
            return List.of();
        }
        return nonNullKnkkSksiJysbList.stream()
                .map(KtmgmkkDataAccessImpl::normalizeInKnkkSksiJysb)
                .toList();
    }

    private static String normalizeInKnkkSksiJysb(String value) {
        return safeKnkkSksiJysb(value, "knkkSksiJysbList");
    }
}
