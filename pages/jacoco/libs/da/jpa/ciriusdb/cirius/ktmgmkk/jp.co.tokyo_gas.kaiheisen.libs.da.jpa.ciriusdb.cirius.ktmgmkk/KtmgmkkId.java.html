package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ktmgmkk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 確定前ガスメーター検針結果（KTMGMKK）の複合主キー。 */
public class KtmgmkkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal gasMtstNo;

    private BigDecimal knsYmd;

    private BigDecimal knsKkaTsno;

    /** デフォルトコンストラクタ。 */
    public KtmgmkkId() { }

    /** 全フィールド指定コンストラクタ。 */
    public KtmgmkkId(BigDecimal gasMtstNo, BigDecimal knsYmd, BigDecimal knsKkaTsno) {
        this.gasMtstNo = gasMtstNo;
        this.knsYmd = knsYmd;
        this.knsKkaTsno = knsKkaTsno;
    }

    /** gasMtstNo を取得する。 */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /** gasMtstNo を設定する。 */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /** knsYmd を取得する。 */
    public BigDecimal getKnsYmd() {
        return knsYmd;
    }

    /** knsYmd を設定する。 */
    public void setKnsYmd(BigDecimal knsYmd) {
        this.knsYmd = knsYmd;
    }

    /** knsKkaTsno を取得する。 */
    public BigDecimal getKnsKkaTsno() {
        return knsKkaTsno;
    }

    /** knsKkaTsno を設定する。 */
    public void setKnsKkaTsno(BigDecimal knsKkaTsno) {
        this.knsKkaTsno = knsKkaTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof KtmgmkkId that)) {
            return false;
        }
        return Objects.equals(gasMtstNo, that.gasMtstNo)
                && Objects.equals(knsYmd, that.knsYmd)
                && Objects.equals(knsKkaTsno, that.knsKkaTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, knsYmd, knsKkaTsno);
    }
}
