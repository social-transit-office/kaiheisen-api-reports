package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gssihhh;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.GssihhhDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.GssihhhRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.write.GssihhhColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.write.GssihhhWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gssihhh.write.GssihhhWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CiriusAuditColumnAutoSetter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaAuditColumnAccessBinder;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaBind;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** GSSIHHH データアクセスの JPA 実装。 */
public class GssihhhDataAccessImpl extends DataAccessBase implements GssihhhDataAccess {

    private static final String TABLE_NAME = "GSSIHHH";

    private static final ReflectiveRecordEntityMapper<GssihhhRecord, GssihhhEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(GssihhhRecord.class, GssihhhEntity.class);

    private static final ReflectiveRecordEntityMapper<GssihhhWriteRecord, GssihhhEntity>
            WRITE_MAPPER =
                    ReflectiveRecordEntityMapper.create(
                            GssihhhWriteRecord.class, GssihhhEntity.class);

    private final EntityManager entityManager;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<GssihhhEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<GssihhhWriteRecord, GssihhhEntity, GssihhhColumn> insertSupport;

    /**
     * JPA ベースの GSSIHHH データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public GssihhhDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    GssihhhDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    GssihhhDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(GssihhhEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        GssihhhWriteDefinition.INSTANCE,
                        GssihhhEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(GssihhhWriteRecord record, AuditContext context) {
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteBySvkyNo(Long svkyNo, Integer recUpdJnjNo) {
        Objects.requireNonNull(svkyNo, "svkyNo");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "SVKY_NO = ?",
                                (query, startIndex) ->
                                        JpaBind.set(
                                                query,
                                                startIndex,
                                                svkyNo,
                                                JpaDbWriteValues::bigDecimal),
                                JpaDbWriteValues.bigDecimal(svkyNo)),
                        recUpdJnjNo,
                        GssihhhDataAccess.Identifiers.DELETE_BY_SVKY_NO);
    }

    @Override
    public Optional<GssihhhRecord> findByPK(Long svkyNo) {
        Objects.requireNonNull(svkyNo, "svkyNo");

        GssihhhEntity entity =
                entityManager.find(GssihhhEntity.class, JpaDbWriteValues.bigDecimal(svkyNo));
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }
}
