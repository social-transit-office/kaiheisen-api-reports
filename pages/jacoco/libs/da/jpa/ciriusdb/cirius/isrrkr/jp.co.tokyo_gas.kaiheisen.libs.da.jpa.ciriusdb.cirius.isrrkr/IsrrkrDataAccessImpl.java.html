/** ISRRKR データアクセスの JPA 実装。 */
public class IsrrkrDataAccessImpl extends DataAccessBase implements IsrrkrDataAccess {

    private static final String TABLE_NAME = "ISRRKR";

    private static final int ISRR_GYM_CD_LENGTH = 6;

    private static final int ISRR_GMSS_CD_LENGTH = 8;

    private static final int TJT_TRKS_FLG_LENGTH = 1;

    private static final String PK_WHERE_CLAUSE =
            "CST_NO = ? AND ISRR_TRI_YMD = ? AND ISRR_RKKR_TSNO = ?";

    private static final ReflectiveRecordEntityMapper<IsrrkrRecord, IsrrkrEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(IsrrkrRecord.class, IsrrkrEntity.class);

    private static final ReflectiveRecordEntityMapper<IsrrkrWriteRecord, IsrrkrEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(IsrrkrWriteRecord.class, IsrrkrEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<IsrrkrEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<IsrrkrWriteRecord, IsrrkrEntity, IsrrkrColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<IsrrkrWriteRecord, IsrrkrEntity, IsrrkrColumn> updateSupport;

    /** JPA ベースの ISRRKR データアクセスを生成する。 */
    public IsrrkrDataAccessImpl(EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    IsrrkrDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    IsrrkrDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory = Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(IsrrkrEntity.class);
        this.auditColumnAutoSetter = Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        IsrrkrWriteDefinition.INSTANCE,
                        IsrrkrEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        IsrrkrWriteDefinition.INSTANCE,
                        IsrrkrEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(IsrrkrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<IsrrkrRecord> findByCstNoAndIsrrTriYmdAndIsrrGymCdAndIsrrGmssCdAndNotTjtTrksFlg(
            Long cstNo,
            LocalDate isrrTriYmd,
            String isrrGymCd,
            String isrrGmssCd,
            Boolean tjtTrksFlg) {
        var builder = criteriaBuilderFactory.create(entityManager, IsrrkrEntity.class);
        JpaWhere.eq(builder, "cstNo", safeLong(cstNo, "cstNo"));
        JpaWhere.eq(builder, "isrrTriYmd", safeYmd(isrrTriYmd, "isrrTriYmd"));
        JpaWhere.eq(builder, "isrrGymCd", safeIsrrGymCd(isrrGymCd));
        JpaWhere.eq(builder, "isrrGmssCd", safeIsrrGmssCd(isrrGmssCd));
        JpaWhere.notEq(builder, "tjtTrksFlg", safeFlag(tjtTrksFlg, "tjtTrksFlg"));
        builder.orderByDesc("isrrRkkrTsno");
        SqlCapture.set(builder.getQueryString());
        return builder.setMaxResults(1).getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<Integer> findMaxIsrrRkkrTsnoByCstNoAndIsrrTriYmd(Long cstNo, LocalDate isrrTriYmd) {
        BigDecimal safeCstNo = safeLong(cstNo, "cstNo");
        BigDecimal safeIsrrTriYmd = safeYmd(isrrTriYmd, "isrrTriYmd");
        String sql = "SELECT MAX(ISRR_RKKR_TSNO) FROM ISRRKR WHERE CST_NO = ? AND ISRR_TRI_YMD = ?";
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        int nextIndex = JpaBind.set(query, 1, safeCstNo);
        JpaBind.set(query, nextIndex, safeIsrrTriYmd);
        return readMaxIsrrRkkrTsno(query.getSingleResult());
    }

    @Override
    public int updateByCstNoAndIsrrTriYmdAndIsrrRkkrTsnoAndRecUpdJnjNo(
            IsrrkrWriteRecord record,
            Set<IsrrkrUpdatableColumn> targetColumns,
            IsrrkrU002UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        String daIdentifier = IsrrkrDataAccess.CallSpecs.U002.identifier();

        return updateSupport.update(
                record,
                canonicalTargetColumns(targetColumns),
                (entity, ignoredColumns) -> {
                Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
                    auditColumnAutoSetter.applyForUpdate(
                            auditColumnAccessBinder.bind(entity),
                            context,
                            clock,
                            currentRecUpdJnjNo);
                },
                preparedWhere(updateKey.cstNo(), updateKey.isrrTriYmd(), updateKey.isrrRkkrTsno()),
                updateKey.recUpdJnjNo(),
                daIdentifier);
    }

    @Override
    public Optional<IsrrkrRecord> findByPrimaryKey(Long cstNo, LocalDate isrrTriYmd, Integer isrrRkkrTsno) {
        IsrrkrId id = new IsrrkrId(
                safeLong(cstNo, "cstNo"),
                safeYmd(isrrTriYmd, "isrrTriYmd"),
                safeInteger(isrrRkkrTsno, "isrrRkkrTsno"));
        IsrrkrEntity entity = entityManager.find(IsrrkrEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    private static Optional<Integer> readMaxIsrrRkkrTsno(Object result) {
        if (result == null) {
            return Optional.empty();
        }
        if (result instanceof BigDecimal value) {
            return Optional.ofNullable(JpaDbReadValues.integerOrNullIfZero(value));
        }
        if (result instanceof Number value) {
            return Optional.ofNullable(JpaDbReadValues.integerOrNullIfZero(new BigDecimal(value.toString())));
        }
        if (result instanceof String value) {
            String trimmedValue = value.trim();
            if (trimmedValue.isEmpty()) {
                return Optional.empty();
            }
            try {
                return Optional.ofNullable(
                        JpaDbReadValues.integerOrNullIfZero(new BigDecimal(trimmedValue)));
            } catch (NumberFormatException ignored) {
                return Optional.empty();
            }
        }
        return Optional.empty();
    }

    private static Set<IsrrkrColumn> canonicalTargetColumns(Set<IsrrkrUpdatableColumn> targetColumns) {
        EnumSet<IsrrkrColumn> canonicalColumns = EnumSet.noneOf(IsrrkrColumn.class);
        for (IsrrkrUpdatableColumn targetColumn : targetColumns) {
            canonicalColumns.add(IsrrkrColumn.valueOf(targetColumn.name()));
        }
        return canonicalColumns;
    }

    private static PreparedWhere preparedWhere(Long cstNo, LocalDate isrrTriYmd, Integer isrrRkkrTsno) {
        BigDecimal safeCstNo = safeLong(cstNo, "cstNo");
        BigDecimal safeIsrrTriYmd = safeYmd(isrrTriYmd, "isrrTriYmd");
        BigDecimal safeIsrrRkkrTsno = safeInteger(isrrRkkrTsno, "isrrRkkrTsno");
        return PreparedWhere.of(
                PK_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeCstNo);
                    nextIndex = JpaBind.set(query, nextIndex, safeIsrrTriYmd);
                    return JpaBind.set(query, nextIndex, safeIsrrRkkrTsno);
                },
                safeCstNo,
                safeIsrrTriYmd,
                safeIsrrRkkrTsno);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static String safeIsrrGymCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGymCd", ISRR_GYM_CD_LENGTH);
    }

    private static String safeIsrrGmssCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGmssCd", ISRR_GMSS_CD_LENGTH);
    }

    private static String safeFlag(Boolean value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(
                BooleanFlagStringConverter.INSTANCE.convert(value),
                paramName,
                TJT_TRKS_FLG_LENGTH);
    }
}
