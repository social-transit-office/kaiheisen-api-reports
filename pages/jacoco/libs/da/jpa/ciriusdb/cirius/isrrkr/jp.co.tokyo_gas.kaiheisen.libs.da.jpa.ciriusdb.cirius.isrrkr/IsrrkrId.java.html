package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrrkr;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 異動・作業履歴連携管理（ISRRKR）の複合主キー。 */
public class IsrrkrId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal cstNo;

    private BigDecimal isrrTriYmd;

    private BigDecimal isrrRkkrTsno;

    /** デフォルトコンストラクタ。 */
    public IsrrkrId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public IsrrkrId(BigDecimal cstNo, BigDecimal isrrTriYmd, BigDecimal isrrRkkrTsno) {
        this.cstNo = cstNo;
        this.isrrTriYmd = isrrTriYmd;
        this.isrrRkkrTsno = isrrRkkrTsno;
    }

    /** お客さま番号を返す。 */
    public BigDecimal getCstNo() {
        return cstNo;
    }

    /** お客さま番号を設定する。 */
    public void setCstNo(BigDecimal cstNo) {
        this.cstNo = cstNo;
    }

    /** 異動・作業履歴登録依頼年月日を返す。 */
    public BigDecimal getIsrrTriYmd() {
        return isrrTriYmd;
    }

    /** 異動・作業履歴登録依頼年月日を設定する。 */
    public void setIsrrTriYmd(BigDecimal isrrTriYmd) {
        this.isrrTriYmd = isrrTriYmd;
    }

    /** 異動・作業履歴連携管理通し番号を返す。 */
    public BigDecimal getIsrrRkkrTsno() {
        return isrrRkkrTsno;
    }

    /** 異動・作業履歴連携管理通し番号を設定する。 */
    public void setIsrrRkkrTsno(BigDecimal isrrRkkrTsno) {
        this.isrrRkkrTsno = isrrRkkrTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof IsrrkrId that)) {
            return false;
        }
        return Objects.equals(cstNo, that.cstNo)
                && Objects.equals(isrrTriYmd, that.isrrTriYmd)
                && Objects.equals(isrrRkkrTsno, that.isrrRkkrTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(cstNo, isrrTriYmd, isrrRkkrTsno);
    }
}
