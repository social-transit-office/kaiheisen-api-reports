package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrkk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** ISRKK データアクセスの JPA 実装。 */
public class IsrkkDataAccessImpl extends DataAccessBase implements IsrkkDataAccess {

    private static final int ISRR_GMSS_CD_LENGTH = 8;

    private static final ReflectiveRecordEntityMapper<IsrkkRecord, IsrkkEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(IsrkkRecord.class, IsrkkEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public IsrkkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public List<IsrkkRecord> findAllByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> isrrGmssCdList) {
        List<String> safeIsrrGmssCdList = normalizeIsrrGmssCdList(isrrGmssCdList);
        if (safeIsrrGmssCdList.isEmpty()) {
            return List.of();
        }
        BigDecimal safeArtyCstNo =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L));
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        var builder = criteriaBuilderFactory.create(entityManager, IsrkkEntity.class);
        JpaWhere.eq(builder, "artyCstNo", safeArtyCstNo);
        JpaWhere.eq(builder, "isrrTriYmd", safeIsrrTriYmd);
        JpaWhere.in(builder, "isrrGmssCd", safeIsrrGmssCdList);
        builder.orderByAsc("isrrKytuTsno");
        return builder.getResultList().stream().map(MAPPER::toRecord).toList();
    }

    @Override
    public Optional<IsrkkRecord> findByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> isrrGmssCdList) {
        List<String> safeIsrrGmssCdList = normalizeIsrrGmssCdList(isrrGmssCdList);
        if (safeIsrrGmssCdList.isEmpty()) {
            return Optional.empty();
        }
        BigDecimal safeArtyCstNo =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L));
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        var builder = criteriaBuilderFactory.create(entityManager, IsrkkEntity.class);
        JpaWhere.eq(builder, "artyCstNo", safeArtyCstNo);
        JpaWhere.eq(builder, "isrrTriYmd", safeIsrrTriYmd);
        JpaWhere.in(builder, "isrrGmssCd", safeIsrrGmssCdList);
        builder.orderByDesc("isrrKytuTsno");
        return builder.setMaxResults(1).getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    @Override
    public Optional<IsrkkRecord> findByPK(Long artyCstNo, LocalDate isrrTriYmd, Integer isrrKytuTsno) {
        BigDecimal safeArtyCstNo =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L));
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                        JpaDbWriteValues.ymd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        BigDecimal safeIsrrKytuTsno =
                JpaDbWriteValues.bigDecimal(
                        WhereParamGuard.complement(isrrKytuTsno, "isrrKytuTsno", 0));

        var builder = criteriaBuilderFactory.create(entityManager, IsrkkEntity.class);
        JpaWhere.eq(builder, "artyCstNo", safeArtyCstNo);
        JpaWhere.eq(builder, "isrrTriYmd", safeIsrrTriYmd);
        JpaWhere.eq(builder, "isrrKytuTsno", safeIsrrKytuTsno);
        builder.setMaxResults(1);
        return builder.getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

    /**
     * IN 句検索用の業務詳細コード一覧を固定長文字列へ正規化する。
     *
     * @param isrrGmssCdList 異動・作業履歴業務詳細コード一覧
     * @return 正規化後の一覧
     */
    private static List<String> normalizeIsrrGmssCdList(List<String> isrrGmssCdList) {
        if (isrrGmssCdList == null) {
            throw new IllegalArgumentException("isrrGmssCdList must not be null");
        }
        if (isrrGmssCdList.isEmpty()) {
            return List.of();
        }
        return isrrGmssCdList.stream().map(IsrkkDataAccessImpl::normalizeInIsrrGmssCd).toList();
    }

    /**
     * IN 句要素の業務詳細コードを固定長条件値へ正規化する。
     *
     * @param value 業務詳細コード
     * @return 正規化後の業務詳細コード
     */
    private static String normalizeInIsrrGmssCd(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "isrrGmssCdList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGmssCdList", ISRR_GMSS_CD_LENGTH);
    }
}
