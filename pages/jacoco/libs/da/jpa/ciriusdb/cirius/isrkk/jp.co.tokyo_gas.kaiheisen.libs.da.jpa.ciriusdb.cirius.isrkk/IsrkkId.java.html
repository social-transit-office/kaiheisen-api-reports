package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrkk;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 異動・作業履歴（共通）＜小売＞（ISRKK）の複合主キー。 */
public class IsrkkId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal artyCstNo;

    private BigDecimal isrrTriYmd;

    private BigDecimal isrrKytuTsno;

    /** デフォルトコンストラクタ。 */
    public IsrkkId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public IsrkkId(BigDecimal artyCstNo, BigDecimal isrrTriYmd, BigDecimal isrrKytuTsno) {
        this.artyCstNo = artyCstNo;
        this.isrrTriYmd = isrrTriYmd;
        this.isrrKytuTsno = isrrKytuTsno;
    }

    /** 案件履歴特定用お客さま番号を取得する。 */
    public BigDecimal getArtyCstNo() {
        return artyCstNo;
    }

    /** 案件履歴特定用お客さま番号を設定する。 */
    public void setArtyCstNo(BigDecimal artyCstNo) {
        this.artyCstNo = artyCstNo;
    }

    /** 異動・作業履歴登録依頼年月日を取得する。 */
    public BigDecimal getIsrrTriYmd() {
        return isrrTriYmd;
    }

    /** 異動・作業履歴登録依頼年月日を設定する。 */
    public void setIsrrTriYmd(BigDecimal isrrTriYmd) {
        this.isrrTriYmd = isrrTriYmd;
    }

    /** 異動・作業履歴（共通）通し番号を取得する。 */
    public BigDecimal getIsrrKytuTsno() {
        return isrrKytuTsno;
    }

    /** 異動・作業履歴（共通）通し番号を設定する。 */
    public void setIsrrKytuTsno(BigDecimal isrrKytuTsno) {
        this.isrrKytuTsno = isrrKytuTsno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof IsrkkId that)) {
            return false;
        }
        return Objects.equals(artyCstNo, that.artyCstNo)
                && Objects.equals(isrrTriYmd, that.isrrTriYmd)
                && Objects.equals(isrrKytuTsno, that.isrrKytuTsno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(artyCstNo, isrrTriYmd, isrrKytuTsno);
    }
}
