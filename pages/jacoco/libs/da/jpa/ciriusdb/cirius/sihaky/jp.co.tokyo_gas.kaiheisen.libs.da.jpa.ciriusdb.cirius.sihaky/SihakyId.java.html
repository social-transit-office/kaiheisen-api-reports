package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.sihaky;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/**
 * 支払契約テーブルの複合主キー。
 */
public class SihakyId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal sihKiyNo;

    private BigDecimal skno;

    /** デフォルトコンストラクタ。 */
    public SihakyId() {
    }

    /**
     * 主キー項目を指定して初期化する。
     *
     * @param sihKiyNo 支払契約番号
     * @param skno 世代管理番号
     */
    public SihakyId(BigDecimal sihKiyNo, BigDecimal skno) {
        this.sihKiyNo = sihKiyNo;
        this.skno = skno;
    }

    /**
     * 支払契約番号を返す。
     *
     * @return 支払契約番号
     */
    public BigDecimal getSihKiyNo() {
        return sihKiyNo;
    }

    /**
     * 支払契約番号を設定する。
     *
     * @param sihKiyNo 支払契約番号
     */
    public void setSihKiyNo(BigDecimal sihKiyNo) {
        this.sihKiyNo = sihKiyNo;
    }

    /**
     * 世代管理番号を返す。
     *
     * @return 世代管理番号
     */
    public BigDecimal getSkno() {
        return skno;
    }

    /**
     * 世代管理番号を設定する。
     *
     * @param skno 世代管理番号
     */
    public void setSkno(BigDecimal skno) {
        this.skno = skno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof SihakyId that)) {
            return false;
        }
        return Objects.equals(sihKiyNo, that.sihKiyNo)
                && Objects.equals(skno, that.skno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(sihKiyNo, skno);
    }
}
