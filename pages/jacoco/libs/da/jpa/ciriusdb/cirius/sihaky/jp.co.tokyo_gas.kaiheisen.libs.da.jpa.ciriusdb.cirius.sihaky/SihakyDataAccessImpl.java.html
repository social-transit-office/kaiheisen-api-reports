    SihakyDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(SihakyEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        SihakyWriteDefinition.INSTANCE,
                        SihakyEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        SihakyWriteDefinition.INSTANCE,
                        SihakyEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(SihakyWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndSihKiyJtkb(Long sihKiyNo, String sihKiyJtkb) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        String safeSihKiyJtkb = safeFixedLength(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        var builder = criteriaBuilderFactory.create(entityManager, SihakyEntity.class);
        JpaWhere.eq(builder, "sihKiyNo", safeSihKiyNo);
        JpaWhere.eq(builder, "sihKiyJtkb", safeSihKiyJtkb);
        builder.orderByDesc("tkysYmd").orderByDesc("skno");
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSknoAndRecUpdJnjNo(
            SihakyWriteRecord record,
            Set<SihakyColumn> targetColumns,
            SihakySknoJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Optional<Integer> currentRecUpdJnjNo =
            updateKey.recUpdJnjNo() == null
                ? findCurrentRecUpdJnjNo(updateKey.sihKiyNo(), updateKey.skno())
                : Optional.of(updateKey.recUpdJnjNo());
        if (currentRecUpdJnjNo.isEmpty()) {
            return 0;
        }
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                    currentRecUpdJnjNo.orElseThrow()),
                preparedWhereBySkno(updateKey.sihKiyNo(), updateKey.skno()),
                updateKey.recUpdJnjNo(),
                SihakyDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO);
    }

    @Override
    public int updateBySihKiyNoAndSkno(
            SihakyWriteRecord record,
            Set<SihakyColumn> targetColumns,
            SihakySknoUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Optional<Integer> currentRecUpdJnjNo = findCurrentRecUpdJnjNo(updateKey.sihKiyNo(), updateKey.skno());
        if (currentRecUpdJnjNo.isEmpty()) {
            return 0;
        }
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo.orElseThrow()),
                preparedWhereBySkno(updateKey.sihKiyNo(), updateKey.skno()),
                null,
                SihakyDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SKNO);
    }

    private PreparedWhere preparedWhereBySkno(Long sihKiyNo, Integer skno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSkno = safeInt(skno, "skno");
        return PreparedWhere.of(
                WHERE_CLAUSE_BY_SKNO,
                (query, startIndex) -> bindWhereBySkno(query, startIndex, safeSihKiyNo, safeSkno),
                safeSihKiyNo,
                safeSkno);
    }

    private static int bindWhereBySkno(
            Query query,
            int startIndex,
            BigDecimal sihKiyNo,
            BigDecimal skno) {
        int nextIndex = JpaBind.set(query, startIndex, sihKiyNo);
        return JpaBind.set(query, nextIndex, skno);
    }

    private Optional<Integer> findCurrentRecUpdJnjNo(Long sihKiyNo, Integer skno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSkno = safeInt(skno, "skno");

        String sql =
                "SELECT REC_UPD_JNJ_NO FROM "
                        + TABLE_NAME
                        + " WHERE SIH_KIY_NO = :sihKiyNo AND SKNO = :skno";
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        query.setParameter("sihKiyNo", safeSihKiyNo);
        query.setParameter("skno", safeSkno);

        List<?> results = query.getResultList();
        if (results.isEmpty()) {
            return Optional.empty();
        }

        Object value = results.getFirst();
        if (value == null) {
            return Optional.of(0);
        }
        if (value instanceof BigDecimal bigDecimal) {
            return Optional.of(bigDecimal.intValueExact());
        }
        if (value instanceof Number number) {
            return Optional.of(number.intValue());
        }
        return Optional.of(Integer.parseInt(value.toString()));
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(JpaDbWriteValues.ymd(value), paramName, BigDecimal.ZERO);
    }

    private static String safeFixedLength(String value, String paramName, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }
