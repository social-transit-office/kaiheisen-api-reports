package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.dsqkjh;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.DsqkjhDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.DsqkjhRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.write.DsqkjhColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.write.DsqkjhSihKiyJnjUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.write.DsqkjhSihKiyUpdateKey;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.write.DsqkjhWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.dsqkjh.write.DsqkjhWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;

/**
 * DSQKJH データアクセスの JPA 実装.
 */
public class DsqkjhDataAccessImpl
        extends DataAccessBase implements DsqkjhDataAccess {

    /** 読み取り用 mapper. */
    static final ReflectiveRecordEntityMapper<DsqkjhRecord, DsqkjhEntity>
            READ_MAPPER = ReflectiveRecordEntityMapper.create(
                    DsqkjhRecord.class,
                    DsqkjhEntity.class);

    /** テーブル名. */
    private static final String TABLE_NAME = "DSQKJH";

    /** 主キー条件句. */
    private static final String WHERE_BY_PK =
            "SIH_KIY_NO = ? AND SIH_KIY_SKNO = ?";

    /** 書き込み用 mapper. */
    private static final ReflectiveRecordEntityMapper<
            DsqkjhWriteRecord,
            DsqkjhEntity> WRITE_MAPPER = ReflectiveRecordEntityMapper.create(
                    DsqkjhWriteRecord.class,
                    DsqkjhEntity.class);

    /** エンティティマネージャ. */
    private final EntityManager entityManager;

    /** CriteriaBuilderFactory. */
    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** システム時刻プロバイダ. */
    private final SystemDateTimeProvider clock;

    /** 監査項目アクセサバインダ. */
    private final JpaAuditColumnAccessBinder<DsqkjhEntity>
            auditColumnAccessBinder;

    /** 監査項目自動設定. */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** insert helper. */
    private final JpaInsertSupport<
            DsqkjhWriteRecord,
            DsqkjhEntity,
            DsqkjhColumn>
            insertSupport;

    /** update helper. */
    private final JpaTargetColumnUpdateSupport<
            DsqkjhWriteRecord,
            DsqkjhEntity,
            DsqkjhColumn> updateSupport;

    /**
     * 標準依存で初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     */
    public DsqkjhDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の時刻依存を含めて初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     */
    DsqkjhDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock) {
        this(
                jpaEntityManager,
                jpaCriteriaBuilderFactory,
                systemClock,
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の依存を含めて初期化する.
     *
     * @param jpaEntityManager エンティティマネージャ
     * @param jpaCriteriaBuilderFactory CriteriaBuilderFactory
     * @param systemClock システム時刻プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査項目自動設定
     */
    DsqkjhDataAccessImpl(
            final EntityManager jpaEntityManager,
            final CriteriaBuilderFactory jpaCriteriaBuilderFactory,
            final SystemDateTimeProvider systemClock,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.entityManager =
                Objects.requireNonNull(jpaEntityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(
                        jpaCriteriaBuilderFactory,
                        "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(systemClock, "clock");
        this.auditColumnAccessBinder =
                new JpaAuditColumnAccessBinder<>(DsqkjhEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(
                        ciriusAuditColumnAutoSetter,
                        "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        DsqkjhWriteDefinition.INSTANCE,
                        DsqkjhEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        DsqkjhWriteDefinition.INSTANCE,
                        DsqkjhEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    /** {@inheritDoc} */
    @Override
    public int insert(
            final DsqkjhWriteRecord record,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity -> auditColumnAutoSetter.applyForInsert(
                        auditColumnAccessBinder.bind(entity),
                        context,
                        clock));
    }

    /** {@inheritDoc} */
    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                WHERE_BY_PK,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(
                                            query,
                                            startIndex,
                                            safeSihKiyNo);
                                    return JpaBind.set(
                                            query,
                                            nextIndex,
                                            safeSihKiySkno);
                                },
                                safeSihKiyNo,
                                safeSihKiySkno),
                        recUpdJnjNo,
                        DsqkjhDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    /** {@inheritDoc} */
    @Override
    public Optional<DsqkjhRecord> findBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        var builder = criteriaBuilderFactory.create(
                entityManager,
                DsqkjhEntity.class);
        JpaWhere.eq(builder, "sihKiyNo", safeSihKiyNo);
        JpaWhere.eq(builder, "sihKiySkno", safeSihKiySkno);
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream()
                .map(READ_MAPPER::toRecord)
                .findFirst();
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final DsqkjhSihKiyUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                DsqkjhDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final DsqkjhSihKiyJnjUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                DsqkjhDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO.identifier());
    }

    private int updateByKey(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo,
            final AuditContext context,
            final String dataAccessIdentifier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(context, "context");
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                Objects.requireNonNullElse(recUpdJnjNo, 0)),
                PreparedWhere.of(
                        WHERE_BY_PK,
                        (query, startIndex) -> {
                            int nextIndex = JpaBind.set(
                                    query,
                                    startIndex,
                                    safeSihKiyNo);
                            return JpaBind.set(
                                    query,
                                    nextIndex,
                                    safeSihKiySkno);
                        },
                        safeSihKiyNo,
                        safeSihKiySkno),
                recUpdJnjNo,
                dataAccessIdentifier);
    }

    private static BigDecimal safeLong(
            final Long value,
            final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInteger(
            final Integer value,
            final String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value),
                paramName,
                BigDecimal.ZERO);
    }
}
