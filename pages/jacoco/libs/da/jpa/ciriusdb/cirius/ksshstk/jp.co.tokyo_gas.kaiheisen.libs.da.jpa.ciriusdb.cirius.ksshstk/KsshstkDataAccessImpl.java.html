package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.ksshstk;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.KsshstkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.KsshstkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.write.KsshstkColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.write.KsshstkWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ksshstk.write.KsshstkWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * KSSHSTK データアクセスの JPA 実装。
 */
public class KsshstkDataAccessImpl extends DataAccessBase implements KsshstkDataAccess {

    private static final String TABLE_NAME = "KSSHSTK";

    private static final ReflectiveRecordEntityMapper<KsshstkRecord, KsshstkEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KsshstkRecord.class, KsshstkEntity.class);

    private static final ReflectiveRecordEntityMapper<KsshstkWriteRecord, KsshstkEntity>
            WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(KsshstkWriteRecord.class, KsshstkEntity.class);

    private final EntityManager entityManager;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<KsshstkEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<KsshstkWriteRecord, KsshstkEntity, KsshstkColumn> insertSupport;

    /**
     * JPA ベースの KSSHSTK データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public KsshstkDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    KsshstkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    KsshstkDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(KsshstkEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        KsshstkWriteDefinition.INSTANCE,
                        KsshstkEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(KsshstkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<KsshstkRecord> findBySvkyNo(Long svkyNo) {
        BigDecimal safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        KsshstkEntity entity = entityManager.find(KsshstkEntity.class, safeSvkyNo);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(READ_MAPPER.toRecord(entity));
    }

    @Override
    public int deleteBySvkyNo(Long svkyNo, Integer recUpdJnjNo) {
        BigDecimal safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "SVKY_NO = ?",
                                (query, startIndex) ->
                                        JpaBind.set(query, startIndex, safeSvkyNo, value -> value),
                                safeSvkyNo),
                        recUpdJnjNo,
                        KsshstkDataAccess.Identifiers.DELETE_BY_SVKY_NO);
    }

    private static BigDecimal safeSvkyNo(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
