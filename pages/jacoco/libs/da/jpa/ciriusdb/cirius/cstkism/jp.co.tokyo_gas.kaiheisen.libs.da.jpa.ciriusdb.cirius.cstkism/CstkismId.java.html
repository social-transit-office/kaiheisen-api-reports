package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.cstkism;

import java.io.Serializable;
import java.util.Objects;

/** コンスタント階層構造マスター（CSTKISM）の複合主キー。 */
public class CstkismId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String jiCtshCd;

    private String jiCnstCd;

    private String kaiCtshCd;

    private String kaiCnstCd;

    /** デフォルトコンストラクタ。 */
    public CstkismId() {
    }

    /** 全フィールド指定コンストラクタ。 */
    public CstkismId(String jiCtshCd, String jiCnstCd, String kaiCtshCd, String kaiCnstCd) {
        this.jiCtshCd = jiCtshCd;
        this.jiCnstCd = jiCnstCd;
        this.kaiCtshCd = kaiCtshCd;
        this.kaiCnstCd = kaiCnstCd;
    }

    /**
     * 上位コンスタント種別コードを返す。
     *
     * @return 上位コンスタント種別コード
     */
    public String getJiCtshCd() {
        return jiCtshCd;
    }

    /**
     * 上位コンスタント種別コードを設定する。
     *
     * @param jiCtshCd 上位コンスタント種別コード
     */
    public void setJiCtshCd(String jiCtshCd) {
        this.jiCtshCd = jiCtshCd;
    }

    /**
     * 上位コンスタントコードを返す。
     *
     * @return 上位コンスタントコード
     */
    public String getJiCnstCd() {
        return jiCnstCd;
    }

    /**
     * 上位コンスタントコードを設定する。
     *
     * @param jiCnstCd 上位コンスタントコード
     */
    public void setJiCnstCd(String jiCnstCd) {
        this.jiCnstCd = jiCnstCd;
    }

    /**
     * 下位コンスタント種別コードを返す。
     *
     * @return 下位コンスタント種別コード
     */
    public String getKaiCtshCd() {
        return kaiCtshCd;
    }

    /**
     * 下位コンスタント種別コードを設定する。
     *
     * @param kaiCtshCd 下位コンスタント種別コード
     */
    public void setKaiCtshCd(String kaiCtshCd) {
        this.kaiCtshCd = kaiCtshCd;
    }

    /**
     * 下位コンスタントコードを返す。
     *
     * @return 下位コンスタントコード
     */
    public String getKaiCnstCd() {
        return kaiCnstCd;
    }

    /**
     * 下位コンスタントコードを設定する。
     *
     * @param kaiCnstCd 下位コンスタントコード
     */
    public void setKaiCnstCd(String kaiCnstCd) {
        this.kaiCnstCd = kaiCnstCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof CstkismId that)) {
            return false;
        }
        return Objects.equals(jiCtshCd, that.jiCtshCd)
                && Objects.equals(jiCnstCd, that.jiCnstCd)
                && Objects.equals(kaiCtshCd, that.kaiCtshCd)
                && Objects.equals(kaiCnstCd, that.kaiCnstCd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(jiCtshCd, jiCnstCd, kaiCtshCd, kaiCnstCd);
    }
}
