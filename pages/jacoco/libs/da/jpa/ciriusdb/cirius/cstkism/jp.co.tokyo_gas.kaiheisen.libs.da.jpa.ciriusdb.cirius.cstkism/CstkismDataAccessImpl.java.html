package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.cstkism;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstkism.CstkismDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstkism.CstkismRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** CSTKISM データアクセスの JPA 実装。 */
public class CstkismDataAccessImpl extends DataAccessBase implements CstkismDataAccess {

    private static final int JI_CTSH_CD_LENGTH = 5;

    private static final int JI_CNST_CD_LENGTH = 10;

    private static final int KAI_CTSH_CD_LENGTH = 5;

    private static final ReflectiveRecordEntityMapper<CstkismRecord, CstkismEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(CstkismRecord.class, CstkismEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public CstkismDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<CstkismRecord> findByJiCtshCdAndJiCnstCdAndKaiCtshCd(
            String jiCtshCd,
            String jiCnstCd,
            String kaiCtshCd) {
        String safeJiCtshCd =
                WhereParamGuard.fixedLengthNotNull(jiCtshCd, "jiCtshCd", JI_CTSH_CD_LENGTH);
        String safeJiCnstCd =
                WhereParamGuard.fixedLengthNotNull(jiCnstCd, "jiCnstCd", JI_CNST_CD_LENGTH);
        String safeKaiCtshCd =
                WhereParamGuard.fixedLengthNotNull(kaiCtshCd, "kaiCtshCd", KAI_CTSH_CD_LENGTH);

        var builder = criteriaBuilderFactory.create(entityManager, CstkismEntity.class);
        JpaWhere.eq(builder, "jiCtshCd", safeJiCtshCd);
        JpaWhere.eq(builder, "jiCnstCd", safeJiCnstCd);
        JpaWhere.eq(builder, "kaiCtshCd", safeKaiCtshCd);
        return builder.setMaxResults(1).getResultList().stream().map(MAPPER::toRecord).findFirst();
    }
}
