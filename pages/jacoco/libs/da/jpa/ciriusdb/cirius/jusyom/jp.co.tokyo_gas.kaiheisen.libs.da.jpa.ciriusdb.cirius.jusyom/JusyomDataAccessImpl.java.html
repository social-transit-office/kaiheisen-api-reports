package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.jusyom;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.jusyom.JusyomDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.jusyom.JusyomRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** JUSYOM データアクセスの JPA 実装。 */
public class JusyomDataAccessImpl extends DataAccessBase implements JusyomDataAccess {

    private static final int GKK_CD_LENGTH = 5;

    private static final int MCH_CD_LENGTH = 3;

    private static final int CMAZ_CD_LENGTH = 3;

    private static final ReflectiveRecordEntityMapper<JusyomRecord, JusyomEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(JusyomRecord.class, JusyomEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /** コンストラクタ。 */
    public JusyomDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<JusyomRecord> findByGkkCdAndMchCdAndCmazCd(
            String gkkCd,
            String mchCd,
            String cmazCd) {
        String safeGkkCd = WhereParamGuard.fixedLengthNotNull(gkkCd, "gkkCd", GKK_CD_LENGTH);
        String safeMchCd = WhereParamGuard.fixedLengthNotNull(mchCd, "mchCd", MCH_CD_LENGTH);
        String safeCmazCd = WhereParamGuard.fixedLengthNotNull(cmazCd, "cmazCd", CMAZ_CD_LENGTH);

        var builder = criteriaBuilderFactory.create(entityManager, JusyomEntity.class);
        JpaWhere.eq(builder, "gkkCd", safeGkkCd);
        JpaWhere.eq(builder, "mchCd", safeMchCd);
        JpaWhere.eq(builder, "cmazCd", safeCmazCd);
        return builder.setMaxResults(1).getResultList().stream().map(MAPPER::toRecord).findFirst();
    }

}
