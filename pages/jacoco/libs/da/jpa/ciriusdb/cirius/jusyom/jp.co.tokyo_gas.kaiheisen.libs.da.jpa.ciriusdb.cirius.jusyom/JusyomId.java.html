package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.jusyom;

import java.io.Serializable;
import java.util.Objects;

/** JUSYOM の複合主キー。 */
public class JusyomId implements Serializable {

    private static final long serialVersionUID = 1L;

    private String gkkCd;

    private String mchCd;

    private String cmazCd;

    /** デフォルトコンストラクタ。 */
    public JusyomId() { }

    /** 全フィールド指定コンストラクタ。 */
    public JusyomId(String gkkCd, String mchCd, String cmazCd) {
        this.gkkCd = gkkCd;
        this.mchCd = mchCd;
        this.cmazCd = cmazCd;
    }

    /**
     * 行政区画コードを返す。
     *
     * @return 行政区画コード
     */
    public String getGkkCd() {
        return gkkCd;
    }

    /**
     * 行政区画コードを設定する。
     *
     * @param gkkCd 行政区画コード
     */
    public void setGkkCd(String gkkCd) {
        this.gkkCd = gkkCd;
    }

    /**
     * 町コードを返す。
     *
     * @return 町コード
     */
    public String getMchCd() {
        return mchCd;
    }

    /**
     * 町コードを設定する。
     *
     * @param mchCd 町コード
     */
    public void setMchCd(String mchCd) {
        this.mchCd = mchCd;
    }

    /**
     * 丁目・字コードを返す。
     *
     * @return 丁目・字コード
     */
    public String getCmazCd() {
        return cmazCd;
    }

    /**
     * 丁目・字コードを設定する。
     *
     * @param cmazCd 丁目・字コード
     */
    public void setCmazCd(String cmazCd) {
        this.cmazCd = cmazCd;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof JusyomId that)) {
            return false;
        }
        return Objects.equals(gkkCd, that.gkkCd)
                && Objects.equals(mchCd, that.mchCd)
                && Objects.equals(cmazCd, that.cmazCd);
    }

    @Override
    public int hashCode() {
        return Objects.hash(gkkCd, mchCd, cmazCd);
    }
}
