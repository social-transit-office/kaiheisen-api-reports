package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.hrikmi;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.HrikmiDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.HrikmiRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.write.HrikmiColumn;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.write.HrikmiWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.hrikmi.write.HrikmiWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaInsertSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaTargetColumnUpdateSupport;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/** HRIKMI データアクセスの JPA 実装。 */
public class HrikmiDataAccessImpl extends DataAccessBase implements HrikmiDataAccess {

    private static final String TABLE_NAME = "HRIKMI";

    private static final String UPDATE_WHERE_CLAUSE = "SIH_KIY_NO = ? AND SIH_KIY_SKNO = ?";

    private static final ReflectiveRecordEntityMapper<HrikmiRecord, HrikmiEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(HrikmiRecord.class, HrikmiEntity.class);

    private static final ReflectiveRecordEntityMapper<HrikmiWriteRecord, HrikmiEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(HrikmiWriteRecord.class, HrikmiEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<HrikmiEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<HrikmiWriteRecord, HrikmiEntity, HrikmiColumn> insertSupport;

    private final JpaTargetColumnUpdateSupport<HrikmiWriteRecord, HrikmiEntity, HrikmiColumn>
            updateSupport;

    /**
     * 標準依存で初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public HrikmiDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト差し替え用の依存を含めて初期化する。
     *
     * @param entityManager エンティティマネージャ
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @param clock システム時刻プロバイダ
     * @param auditColumnAutoSetter 監査項目自動設定
     */
    HrikmiDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(HrikmiEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        HrikmiWriteDefinition.INSTANCE,
                        HrikmiEntity.class,
                        WRITE_MAPPER::toEntity);
        this.updateSupport =
                new JpaTargetColumnUpdateSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        HrikmiWriteDefinition.INSTANCE,
                        HrikmiEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(HrikmiWriteRecord record, AuditContext context) {
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public int deleteBySihKiyNoAndSihKiySkno(Long sihKiyNo, Integer sihKiySkno, Integer recUpdJnjNo) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                UPDATE_WHERE_CLAUSE,
                                (query, startIndex) -> {
                                    int nextIndex = JpaBind.set(query, startIndex, safeSihKiyNo);
                                    return JpaBind.set(query, nextIndex, safeSihKiySkno);
                                },
                                safeSihKiyNo,
                                safeSihKiySkno),
                        recUpdJnjNo,
                        HrikmiDataAccess.Identifiers.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO);
    }

    @Override
    public Optional<HrikmiRecord> findBySihKiyNoAndSihKiySkno(Long sihKiyNo, Integer sihKiySkno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        SqlCapture.set(
                "SELECT * FROM HRIKMI WHERE SIH_KIY_NO = :sihKiyNo AND SIH_KIY_SKNO = :sihKiySkno");
        var builder = criteriaBuilderFactory.create(entityManager, HrikmiEntity.class);
        builder.where("sihKiyNo").eq(safeSihKiyNo).where("sihKiySkno").eq(safeSihKiySkno);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            HrikmiWriteRecord record,
            Set<HrikmiColumn> targetColumns,
            HrikmiSihKiyUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhere(updateKey.sihKiyNo(), updateKey.sihKiySkno()),
                updateKey.recUpdJnjNo(),
                HrikmiDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO);
    }

    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            HrikmiWriteRecord record,
            Set<HrikmiColumn> targetColumns,
            HrikmiSihKiyJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Integer currentRecUpdJnjNo = Objects.requireNonNullElse(updateKey.recUpdJnjNo(), 0);
        return updateSupport.update(
                record,
                targetColumns,
                (entity, ignoredColumns) ->
                        auditColumnAutoSetter.applyForUpdate(
                                auditColumnAccessBinder.bind(entity),
                                context,
                                clock,
                                currentRecUpdJnjNo),
                preparedWhere(updateKey.sihKiyNo(), updateKey.sihKiySkno()),
                updateKey.recUpdJnjNo(),
                HrikmiDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO);
    }

    private PreparedWhere preparedWhere(Long sihKiyNo, Integer sihKiySkno) {
        BigDecimal safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        BigDecimal safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        return PreparedWhere.of(
                UPDATE_WHERE_CLAUSE,
                (query, startIndex) -> {
                    int nextIndex = JpaBind.set(query, startIndex, safeSihKiyNo);
                    return JpaBind.set(query, nextIndex, safeSihKiySkno);
                },
                safeSihKiyNo,
                safeSihKiySkno);
    }

    private static BigDecimal safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }

    private static BigDecimal safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
