package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.hrikmi;

import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 払込関連情報テーブルの複合主キー。 */
public class HrikmiId implements Serializable {

    private static final long serialVersionUID = 1L;

    private BigDecimal sihKiyNo;

    private BigDecimal sihKiySkno;

    /** デフォルトコンストラクタ。 */
    public HrikmiId() {
    }

    /**
     * 主キー項目を指定して初期化する。
     *
     * @param sihKiyNo 支払契約番号
     * @param sihKiySkno 支払契約世代管理番号
     */
    public HrikmiId(BigDecimal sihKiyNo, BigDecimal sihKiySkno) {
        this.sihKiyNo = sihKiyNo;
        this.sihKiySkno = sihKiySkno;
    }

    /**
     * 支払契約番号を返す。
     *
     * @return 支払契約番号
     */
    public BigDecimal getSihKiyNo() {
        return sihKiyNo;
    }

    /**
     * 支払契約番号を設定する。
     *
     * @param sihKiyNo 支払契約番号
     */
    public void setSihKiyNo(BigDecimal sihKiyNo) {
        this.sihKiyNo = sihKiyNo;
    }

    /**
     * 支払契約世代管理番号を返す。
     *
     * @return 支払契約世代管理番号
     */
    public BigDecimal getSihKiySkno() {
        return sihKiySkno;
    }

    /**
     * 支払契約世代管理番号を設定する。
     *
     * @param sihKiySkno 支払契約世代管理番号
     */
    public void setSihKiySkno(BigDecimal sihKiySkno) {
        this.sihKiySkno = sihKiySkno;
    }

    @Override
    public boolean equals(Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof HrikmiId that)) {
            return false;
        }
        return Objects.equals(sihKiyNo, that.sihKiyNo)
                && Objects.equals(sihKiySkno, that.sihKiySkno);
    }

    @Override
    public int hashCode() {
        return Objects.hash(sihKiyNo, sihKiySkno);
    }
}
