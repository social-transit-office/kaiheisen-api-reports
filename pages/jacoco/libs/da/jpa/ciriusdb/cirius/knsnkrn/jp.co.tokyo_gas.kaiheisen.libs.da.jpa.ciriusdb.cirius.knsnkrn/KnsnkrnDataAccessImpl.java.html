package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.knsnkrn;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * KNSNKRN データアクセスの JPA 実装。
 */
public class KnsnkrnDataAccessImpl extends DataAccessBase implements KnsnkrnDataAccess {

    private static final ReflectiveRecordEntityMapper<KnsnkrnRecord, KnsnkrnEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(KnsnkrnRecord.class, KnsnkrnEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    /**
     * JPA ベースの KNSNKRN データアクセスを生成する。
     *
     * @param entityManager JPA エンティティマネージャ
     * @param criteriaBuilderFactory Blaze-Persistence の criteria builder factory
     */
    public KnsnkrnDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public Optional<KnsnkrnRecord> findByGasMtstNo(Long gasMtstNo) {
        BigDecimal safeGasMtstNo = safeGasMtstNo(gasMtstNo);
        var builder = criteriaBuilderFactory.create(entityManager, KnsnkrnEntity.class);
        builder.where("gasMtstNo").eq(safeGasMtstNo);
        builder.setMaxResults(1);
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    /**
     * WHERE 条件に使用するガスメーター設置場所番号を補完する。
     *
     * @param value 入力値
     * @return 補完後の DB 値
     */
    private static BigDecimal safeGasMtstNo(Long value) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), "gasMtstNo", BigDecimal.ZERO);
    }
}
