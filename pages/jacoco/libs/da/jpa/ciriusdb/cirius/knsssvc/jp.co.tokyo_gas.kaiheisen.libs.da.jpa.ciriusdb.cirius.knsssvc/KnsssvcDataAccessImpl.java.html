package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.knsssvc;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsssvc.KnsssvcDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsssvc.KnsssvcRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaBind;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaDbWriteValues;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaOptimisticDeleteSupport;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.PreparedWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.converter.BooleanFlagStringConverter;

/** KNSSSVC データアクセスの JPA 実装。 */
public class KnsssvcDataAccessImpl extends DataAccessBase implements KnsssvcDataAccess {

    private static final String TABLE_NAME = "KNSSSVC";

    private static final ReflectiveRecordEntityMapper<KnsssvcRecord, KnsssvcEntity> MAPPER =
            ReflectiveRecordEntityMapper.create(KnsssvcRecord.class, KnsssvcEntity.class);

    private final EntityManager entityManager;

    /** コンストラクタ。 */
    public KnsssvcDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
    }

    @Override
    public int deleteByGasMtstNoAndIdujMskhFlg(Long gasMtstNo, Boolean idujMskhFlg, Integer recUpdJnjNo) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(idujMskhFlg, "idujMskhFlg");

        return new JpaOptimisticDeleteSupport(entityManager, TABLE_NAME)
                .delete(
                        PreparedWhere.of(
                                "GAS_MTST_NO = ? AND IDUJ_MSKH_FLG = ?",
                                (query, startIndex) -> {
                                    int nextIndex =
                                            JpaBind.set(
                                                    query,
                                                    startIndex,
                                                    gasMtstNo,
                                                    JpaDbWriteValues::bigDecimal);
                                    return JpaBind.set(
                                            query,
                                            nextIndex,
                                            BooleanFlagStringConverter.INSTANCE.convert(
                                                    idujMskhFlg));
                                },
                                JpaDbWriteValues.bigDecimal(gasMtstNo),
                                BooleanFlagStringConverter.INSTANCE.convert(idujMskhFlg)),
                        recUpdJnjNo,
                        KnsssvcDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_IDUJ_MSKH_FLG);
    }

    @Override
    public Optional<KnsssvcRecord> findByPK(Long gasMtstNo, String ksseBnriShbt) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(ksseBnriShbt, "ksseBnriShbt");

        KnsssvcId id = new KnsssvcId(JpaDbWriteValues.bigDecimal(gasMtstNo), ksseBnriShbt);
        KnsssvcEntity entity = entityManager.find(KnsssvcEntity.class, id);
        if (entity == null) {
            return Optional.empty();
        }
        return Optional.of(MAPPER.toRecord(entity));
    }
}
