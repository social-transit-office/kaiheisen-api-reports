package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.knsssvc;

import java.io.Serial;
import java.io.Serializable;
import java.math.BigDecimal;
import java.util.Objects;

/** 検針支援サービス情報（KNSSSVC）の複合主キー。 */
public class KnsssvcId implements Serializable {

    /** シリアルバージョンUID。 */
    @Serial private static final long serialVersionUID = 1L;

    /** ガスメーター設置場所番号。 */
    private BigDecimal gasMtstNo;

    /** 検針支援分類種別。 */
    private String ksseBnriShbt;

    /** デフォルトコンストラクタ。 */
    public KnsssvcId() { }

    /**
     * コンストラクタ。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param ksseBnriShbt 検針支援分類種別
     */
    public KnsssvcId(BigDecimal gasMtstNo, String ksseBnriShbt) {
        this.gasMtstNo = gasMtstNo;
        this.ksseBnriShbt = ksseBnriShbt;
    }

    /**
     * ガスメーター設置場所番号を返す。
     *
     * @return ガスメーター設置場所番号
     */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /**
     * ガスメーター設置場所番号を設定する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     */
    public void setGasMtstNo(BigDecimal gasMtstNo) {
        this.gasMtstNo = gasMtstNo;
    }

    /**
     * 検針支援分類種別を返す。
     *
     * @return 検針支援分類種別
     */
    public String getKsseBnriShbt() {
        return ksseBnriShbt;
    }

    /**
     * 検針支援分類種別を設定する。
     *
     * @param ksseBnriShbt 検針支援分類種別
     */
    public void setKsseBnriShbt(String ksseBnriShbt) {
        this.ksseBnriShbt = ksseBnriShbt;
    }

    /**
     * 同値判定。
     *
     * @param obj 比較対象
     * @return 同値なら true
     */
    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (!(obj instanceof KnsssvcId other)) {
            return false;
        }
        return Objects.equals(gasMtstNo, other.gasMtstNo)
                && Objects.equals(ksseBnriShbt, other.ksseBnriShbt);
    }

    /**
     * ハッシュコードを返す。
     *
     * @return ハッシュコード
     */
    @Override
    public int hashCode() {
        return Objects.hash(gasMtstNo, ksseBnriShbt);
    }
}
