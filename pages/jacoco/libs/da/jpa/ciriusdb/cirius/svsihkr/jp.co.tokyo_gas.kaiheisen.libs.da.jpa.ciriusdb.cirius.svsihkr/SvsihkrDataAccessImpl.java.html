    private int executeUpdateBySihKiyNoAndSvkyNoAndSkno(final UpdateCommand command) {
        Objects.requireNonNull(command, "command");
        Objects.requireNonNull(command.record(), "record");
        Objects.requireNonNull(command.targetColumns(), "targetColumns");
        Objects.requireNonNull(command.updateKey(), "updateKey");
        Objects.requireNonNull(command.updateKey().sihKiyNo(), "sihKiyNo");
        Objects.requireNonNull(command.updateKey().svkyNo(), "svkyNo");
        Objects.requireNonNull(command.updateKey().skno(), "skno");
        Objects.requireNonNull(command.auditContext(), "auditContext");

        Optional<Integer> currentRecUpdJnjNo =

        if (currentRecUpdJnjNo.isEmpty()) {

    }

    private Optional<Integer> resolveCurrentRecUpdJnjNoForUpdate(final UpdateKey updateKey) {
        Objects.requireNonNull(updateKey, "updateKey");

        Objects.requireNonNull(updateKey.svkyNo(), "svkyNo");
        Objects.requireNonNull(updateKey.skno(), "skno");

            return Optional.of(updateKey.recUpdJnjNo());
        }
        return findCurrentRecUpdJnjNo(
                updateKey.sihKiyNo(), updateKey.svkyNo(), updateKey.skno());
    }

    private int executeResolvedUpdateBySihKiyNoAndSvkyNoAndSkno(
            final UpdateCommand command, final Integer currentRecUpdJnjNo) {
        Objects.requireNonNull(command, "command");
        Objects.requireNonNull(command.record(), "record");
        Objects.requireNonNull(command.targetColumns(), "targetColumns");
        Objects.requireNonNull(command.updateKey(), "updateKey");
        Objects.requireNonNull(currentRecUpdJnjNo, "currentRecUpdJnjNo");
        Objects.requireNonNull(command.updateKey().daIdentifier(), "daIdentifier");
        Objects.requireNonNull(command.auditContext(), "auditContext");

        return updateSupport.update(
        command.record(),
        command.targetColumns(),
                (entity, ignoredColumns) ->
            applyAuditForResolvedUpdate(
                entity, command.auditContext(), currentRecUpdJnjNo),
        preparedWhereBySkno(
            command.updateKey().sihKiyNo(),
            command.updateKey().svkyNo(),
            command.updateKey().skno()),
        command.updateKey().recUpdJnjNo(),
        command.updateKey().daIdentifier());
    }

    private static int bindWhereBySkno(
            final Query query,
            final int startIndex,
            final Long sihKiyNo,
            final Long svkyNo,
            final Integer skno) {
        int nextIndex =
                JpaBind.set(query, startIndex, sihKiyNo, JpaDbWriteValues::bigDecimal);
        nextIndex = JpaBind.set(query, nextIndex, svkyNo, JpaDbWriteValues::bigDecimal);
        return JpaBind.set(query, nextIndex, skno, JpaDbWriteValues::bigDecimal);
    }

    private Optional<Integer> findCurrentRecUpdJnjNo(
            final Long sihKiyNo, final Long svkyNo, final Integer skno) {
        Objects.requireNonNull(sihKiyNo, "sihKiyNo");
        Objects.requireNonNull(svkyNo, "svkyNo");
        Objects.requireNonNull(skno, "skno");

        String sql =
                "SELECT REC_UPD_JNJ_NO FROM "
                        + TABLE_NAME
                        + " WHERE SIH_KIY_NO = :sihKiyNo AND SVKY_NO = :svkyNo AND SKNO = :skno";
        SqlCapture.set(sql);
        Query query = entityManager.createNativeQuery(sql);
        query.setParameter("sihKiyNo", JpaDbWriteValues.bigDecimal(sihKiyNo));
        query.setParameter("svkyNo", JpaDbWriteValues.bigDecimal(svkyNo));
        query.setParameter("skno", JpaDbWriteValues.bigDecimal(skno));

        List<?> results = query.getResultList();
        if (results.isEmpty()) {
            return Optional.empty();
        }

        Object value = results.getFirst();
        if (value == null) {
            return Optional.of(0);
        }
        if (value instanceof java.math.BigDecimal bigDecimal) {
            Integer converted = JpaDbReadValues.integerOrNullIfZero(bigDecimal);
            return Optional.of(converted == null ? 0 : converted);
        }
        if (value instanceof Number number) {
