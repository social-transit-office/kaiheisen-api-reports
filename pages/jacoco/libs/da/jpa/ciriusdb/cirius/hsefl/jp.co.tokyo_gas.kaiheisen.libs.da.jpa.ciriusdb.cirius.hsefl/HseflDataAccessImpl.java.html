                this.entityManager,

    @Override
    public int insert(HseflWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                normalizeInsertRecord(record),
                entity -> {
                    if (record.tsigHesnSgyYmd() == null) {
                        entity.setTsigHesnSgyYmd(DEFAULT_CORRECTED_CLOSING_WORK_DATE);
                    }
                    auditColumnAutoSetter.applyForInsert(auditColumnAccessBinder.bind(entity), context, clock);
                });
    }

    @Override
    public List<HseflRecord> findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
            String flttCktkCd,
            String flttKtnCd,
            String flwStCd,
            List<String> hesnJysbList) {
        String safeFlttCktkCd = WhereParamGuard.fixedLengthNotNull(flttCktkCd, "flttCktkCd", 3);
        String safeFlwStCd = WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);
        String blankFlttKtnCd = WhereParamGuard.fixedLengthNotNull("", "flttKtnCd", 1);
        List<String> safeHesnJysbList = safeHesnJysbList(hesnJysbList);
        var builder = criteriaBuilderFactory.create(entityManager, HseflEntity.class);
        JpaWhere.eq(builder, "flttCktkCd", safeFlttCktkCd);
        if (flttKtnCd != null) {
            JpaWhere.in(
                    builder,
                    "flttKtnCd",
                    List.of(WhereParamGuard.fixedLengthNotNull(flttKtnCd, "flttKtnCd", 1), blankFlttKtnCd));
        }
        JpaWhere.eq(builder, "flwStCd", safeFlwStCd);
        if (!safeHesnJysbList.isEmpty()) {
            JpaWhere.in(builder, "hesnJysb", safeHesnJysbList);
        }
        builder.orderByAsc("hssgYmd")
                .orderByAsc("hesnHukkYmd")
                .orderByAsc("regTsno")
                .orderByAsc("hssgNwkCd")
                .orderByAsc("gasMtstNo");
        SqlCapture.set(builder.getQueryString());
        return builder.getResultList().stream().map(READ_MAPPER::toRecord).toList();
    }

            where,
            updateKey.recUpdJnjNo(),

    private static Optional<Integer> toRecUpdJnjNo(List<?> results) {
        if (results.isEmpty()) {
            return Optional.empty();
        }
        Object value = results.getFirst();
        if (value == null) {
            return Optional.of(0);
        }
        if (value instanceof BigDecimal bigDecimal) {
            return Optional.of(bigDecimal.intValue());
        }
        if (value instanceof Number number) {
            return Optional.of(number.intValue());
        }
        return Optional.of(Integer.parseInt(value.toString()));
    }

    private static List<String> safeHesnJysbList(List<String> hesnJysbList) {
        if (hesnJysbList == null || hesnJysbList.isEmpty()) {
            return List.of();
        }
        return hesnJysbList.stream()
                .filter(Objects::nonNull)
                .filter(value -> !value.isBlank())
                .map(HseflDataAccessImpl::validateAndFormatHesnJysb)
                .toList();
    }

        return WhereParamGuard.fixedLengthNotNull(value, "hesnJysbList", HESN_JYSB_LENGTH);
    }
