package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.hsefl;

import java.io.Serializable;
import java.math.BigDecimal;

/**
 * 閉栓エラーフォロー件名（HSEFL）の複合主キー.
 */
public final class HseflId implements Serializable {

    /** シリアルバージョン UID. */
    private static final long serialVersionUID = 1L;

    /** ガスメーター設置場所番号. */
    private BigDecimal gasMtstNo;

    /** 閉栓作業年月日. */
    private BigDecimal hssgYmd;

    /** 閉栓報告年月日. */
    private BigDecimal hesnHukkYmd;

    /** 登録通し番号. */
    private BigDecimal regTsno;

    /** デフォルトコンストラクタ. */
    public HseflId() { }

    /**
     * 全フィールド指定コンストラクタ.
     *
      * @param primaryGasMtstNo ガスメーター設置場所番号
      * @param primaryHssgYmd 閉栓作業年月日
      * @param primaryHesnHukkYmd 閉栓報告年月日
      * @param primaryRegTsno 登録通し番号
     */
    public HseflId(
            final BigDecimal primaryGasMtstNo,
            final BigDecimal primaryHssgYmd,
            final BigDecimal primaryHesnHukkYmd,
            final BigDecimal primaryRegTsno) {
        this.gasMtstNo = primaryGasMtstNo;
        this.hssgYmd = primaryHssgYmd;
        this.hesnHukkYmd = primaryHesnHukkYmd;
        this.regTsno = primaryRegTsno;
    }

    /**
     * ガスメーター設置場所番号を返す.
     *
     * @return ガスメーター設置場所番号
     */
    public BigDecimal getGasMtstNo() {
        return gasMtstNo;
    }

    /**
     * ガスメーター設置場所番号を設定する.
     *
     * @param newGasMtstNo ガスメーター設置場所番号
     */
    public void setGasMtstNo(final BigDecimal newGasMtstNo) {
        this.gasMtstNo = newGasMtstNo;
    }

    /**
     * 閉栓作業年月日を返す.
     *
     * @return 閉栓作業年月日
     */
    public BigDecimal getHssgYmd() {
        return hssgYmd;
    }

    /**
     * 閉栓作業年月日を設定する.
     *
     * @param newHssgYmd 閉栓作業年月日
     */
    public void setHssgYmd(final BigDecimal newHssgYmd) {
        this.hssgYmd = newHssgYmd;
    }

    /**
     * 閉栓報告年月日を返す.
     *
     * @return 閉栓報告年月日
     */
    public BigDecimal getHesnHukkYmd() {
        return hesnHukkYmd;
    }

    /**
     * 閉栓報告年月日を設定する.
     *
     * @param newHesnHukkYmd 閉栓報告年月日
     */
    public void setHesnHukkYmd(final BigDecimal newHesnHukkYmd) {
        this.hesnHukkYmd = newHesnHukkYmd;
    }

    /**
     * 登録通し番号を返す.
     *
     * @return 登録通し番号
     */
    public BigDecimal getRegTsno() {
        return regTsno;
    }

    /**
     * 登録通し番号を設定する.
     *
     * @param newRegTsno 登録通し番号
     */
    public void setRegTsno(final BigDecimal newRegTsno) {
        this.regTsno = newRegTsno;
    }

    /** {@inheritDoc} */
    @Override
    public boolean equals(final Object other) {
        if (this == other) {
            return true;
        }
        if (!(other instanceof HseflId)) {
            return false;
        }
        final HseflId that = (HseflId) other;
        return compareBigDecimal(this.gasMtstNo, that.gasMtstNo)
                && compareBigDecimal(this.hssgYmd, that.hssgYmd)
                && compareBigDecimal(this.hesnHukkYmd, that.hesnHukkYmd)
                && compareBigDecimal(this.regTsno, that.regTsno);
    }

    private static boolean compareBigDecimal(final BigDecimal left, final BigDecimal right) {
        if (left == null) {
            return right == null;
        }
        return right != null && left.compareTo(right) == 0;
    }

    private static int hashBigDecimal(final BigDecimal value) {
        return value == null ? 0 : value.stripTrailingZeros().hashCode();
    }

    /** {@inheritDoc} */
    @Override
    public int hashCode() {
        int result = 17;
        result = 31 * result + hashBigDecimal(gasMtstNo);
        result = 31 * result + hashBigDecimal(hssgYmd);
        result = 31 * result + hashBigDecimal(hesnHukkYmd);
        result = 31 * result + hashBigDecimal(regTsno);
        return result;
    }
}
