package jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.flknmei;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.persistence.EntityManager;
import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.flknmei.FlknmeiDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.flknmei.FlknmeiRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.flknmei.write.FlknmeiColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.flknmei.write.FlknmeiWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.flknmei.write.FlknmeiWriteRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.AuditContext;

import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaWhere;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.ReflectiveRecordEntityMapper;

/**
 * FLKNMEI データアクセスの JPA 実装。
 */
public class FlknmeiDataAccessImpl extends DataAccessBase implements FlknmeiDataAccess {

    private static final String TABLE_NAME = "FLKNMEI";

    private static final ReflectiveRecordEntityMapper<FlknmeiRecord, FlknmeiEntity> READ_MAPPER =
            ReflectiveRecordEntityMapper.create(FlknmeiRecord.class, FlknmeiEntity.class);

    private static final ReflectiveRecordEntityMapper<FlknmeiWriteRecord, FlknmeiEntity> WRITE_MAPPER =
            ReflectiveRecordEntityMapper.create(FlknmeiWriteRecord.class, FlknmeiEntity.class);

    private final EntityManager entityManager;

    private final CriteriaBuilderFactory criteriaBuilderFactory;

    private final SystemDateTimeProvider clock;

    private final JpaAuditColumnAccessBinder<FlknmeiEntity> auditColumnAccessBinder;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final JpaInsertSupport<FlknmeiWriteRecord, FlknmeiEntity, FlknmeiColumn> insertSupport;

    /**
     * JPA 実装を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     */
    public FlknmeiDataAccessImpl(
            EntityManager entityManager, CriteriaBuilderFactory criteriaBuilderFactory) {
        this(
                entityManager,
                criteriaBuilderFactory,
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    FlknmeiDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock) {
        this(entityManager, criteriaBuilderFactory, clock, new CiriusAuditColumnAutoSetter());
    }

    FlknmeiDataAccessImpl(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory,
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.entityManager = Objects.requireNonNull(entityManager, "entityManager");
        this.criteriaBuilderFactory =
                Objects.requireNonNull(criteriaBuilderFactory, "criteriaBuilderFactory");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAccessBinder = new JpaAuditColumnAccessBinder<>(FlknmeiEntity.class);
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new JpaInsertSupport<>(
                        this.entityManager,
                        TABLE_NAME,
                        FlknmeiWriteDefinition.INSTANCE,
                        FlknmeiEntity.class,
                        WRITE_MAPPER::toEntity);
    }

    @Override
    public int insert(FlknmeiWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        return insertSupport.insert(
                record,
                entity ->
                        auditColumnAutoSetter.applyForInsert(
                                auditColumnAccessBinder.bind(entity), context, clock));
    }

    @Override
    public Optional<FlknmeiRecord> findByPK(Integer flknNo) {
        BigDecimal safeFlknNo = safeInteger(flknNo, "flknNo");
        var builder = criteriaBuilderFactory.create(entityManager, FlknmeiEntity.class);
        JpaWhere.eq(builder, "flknNo", safeFlknNo);
        return builder.setMaxResults(1).getResultList().stream().map(READ_MAPPER::toRecord).findFirst();
    }

    private static BigDecimal safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(
                JpaDbWriteValues.bigDecimal(value), paramName, BigDecimal.ZERO);
    }
}
