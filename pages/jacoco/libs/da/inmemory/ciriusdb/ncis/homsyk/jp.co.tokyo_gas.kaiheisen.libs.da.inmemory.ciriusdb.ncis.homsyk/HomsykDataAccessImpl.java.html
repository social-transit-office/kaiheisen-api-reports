package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.homsyk;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.homsyk.HomsykDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticDeleteSupport;

/** DA-020-D001 の SQL を模したインメモリ実装。 */
public class HomsykDataAccessImpl extends DataAccessBase implements HomsykDataAccess {

    private static final int IDO_MASY_KHI_CD_LENGTH = 1;

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public HomsykDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Long safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        String safeIdoMasyKhiCd =
                WhereParamGuard.fixedLengthNotNull(
                        idoMasyKhiCd, "idoMasyKhiCd", IDO_MASY_KHI_CD_LENGTH);
        String daIdentifier =
                HomsykDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier();

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> safeGmtSetNo.equals(row.gmtSetNo()) && safeIdoMasyKhiCd.equals(row.idoMasyKhiCd()),
                InternalRow::recUpdJnjNo,
                recUpdJnjNo,
                daIdentifier,
                () -> new DataInconsistencyException("HOMSYK", daIdentifier, "{}"),
                safeGmtSetNo,
                safeIdoMasyKhiCd);
    }

    /** HOMSYK のテストデータ行。 */
    public static record InternalRow(
            Long gmtSetNo,
            String hmsSykSienCd,
            String torokuNaiyoMx,
            String idoMasyKhiCd,
            String renrakskSybCd,
            String renrakskTelNo,
            Integer hmsYksStrDd,
            LocalTime hmsYksStrJkk,
            LocalTime hmsYksEndJkk,
            String hmnYksJknCd,
            LocalDate recRegYmd,
            LocalDate recUpdYmd,
            Integer recUpdJnjNo,
            String kosinUserId,
            String kosinTranNo,
            Integer kosinTaskNo) {

        /**
         * 主要キーと排他番号だけを指定してテストデータ行を生成する。
         *
         * @param gmtSetNo ガスメータ設置場所番号
         * @param hmsSykSienCd 訪問集金支援分類コード
         * @param idoMasyKhiCd 異動時抹消可否コード
         * @param recUpdJnjNo レコード更新排他制御順序番号
         */
        public InternalRow(Long gmtSetNo, String hmsSykSienCd, String idoMasyKhiCd, Integer recUpdJnjNo) {
            this(
                    gmtSetNo,
                    hmsSykSienCd,
                    "memo",
                    idoMasyKhiCd,
                    "1",
                    "090123456789",
                    15,
                    LocalTime.of(9, 30),
                    LocalTime.of(10, 0),
                    "1",
                    LocalDate.of(2024, 4, 1),
                    LocalDate.of(2024, 4, 2),
                    recUpdJnjNo,
                    "USER0001",
                    "TRN1",
                    1);
        }

        /** レコード生成時に必須項目が欠落していないことを検証する。 */
        public InternalRow {
            Objects.requireNonNull(gmtSetNo, "gmtSetNo");
            Objects.requireNonNull(hmsSykSienCd, "hmsSykSienCd");
            Objects.requireNonNull(torokuNaiyoMx, "torokuNaiyoMx");
            Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");
            Objects.requireNonNull(renrakskSybCd, "renrakskSybCd");
            Objects.requireNonNull(renrakskTelNo, "renrakskTelNo");
            Objects.requireNonNull(hmsYksStrDd, "hmsYksStrDd");
            Objects.requireNonNull(hmsYksStrJkk, "hmsYksStrJkk");
            Objects.requireNonNull(hmsYksEndJkk, "hmsYksEndJkk");
            Objects.requireNonNull(hmnYksJknCd, "hmnYksJknCd");
            Objects.requireNonNull(recRegYmd, "recRegYmd");
            Objects.requireNonNull(recUpdYmd, "recUpdYmd");
            Objects.requireNonNull(recUpdJnjNo, "recUpdJnjNo");
            Objects.requireNonNull(kosinUserId, "kosinUserId");
            Objects.requireNonNull(kosinTranNo, "kosinTranNo");
            Objects.requireNonNull(kosinTaskNo, "kosinTaskNo");
        }
    }
}
