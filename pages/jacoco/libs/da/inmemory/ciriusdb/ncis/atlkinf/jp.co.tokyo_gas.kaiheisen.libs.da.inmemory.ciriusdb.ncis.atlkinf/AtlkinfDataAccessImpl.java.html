package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.atlkinf;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.atlkinf.AtlkinfDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.atlkinf.AtlkinfRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-015-R001 の SQL を模したインメモリ実装。 */
public class AtlkinfDataAccessImpl extends DataAccessBase implements AtlkinfDataAccess {

    private static final int BLD_NO_LENGTH = 11;

    private final List<AtlkinfRecord> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public AtlkinfDataAccessImpl addRow(AtlkinfRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<AtlkinfRecord> findByBldNo(String bldNo) {
        String safeBldNo = WhereParamGuard.fixedLengthNotNull(bldNo, "bldNo", BLD_NO_LENGTH);
        return rows.stream()
                .filter(row -> Objects.equals(normalizeFixed(row.bldNo()), safeBldNo))
                .findFirst();
    }

    private static String normalizeFixed(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", BLD_NO_LENGTH);
    }
}
