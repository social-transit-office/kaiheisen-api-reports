/** DA-007（R001/U001）の SQL 振る舞いを模した TATEMON インメモリ実装。 */
public class TatemonDataAccessImpl extends DataAccessBase implements TatemonDataAccess {

    private static final ReflectiveInMemoryRowMapper<TatemonColumn, TatemonRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(TatemonColumn.class, TatemonRecord.class);

    private static final AutoNeutralValueResolver<TatemonColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(TatemonColumn.class, TatemonRecord.class);

    private final List<EnumMap<TatemonColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryTargetColumnUpdateSupport<TatemonWriteRecord, TatemonColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public TatemonDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new NcisCisAuditColumnAutoSetter());
    }

    TatemonDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new NcisCisAuditColumnAutoSetter());
    }

    TatemonDataAccessImpl(
            SystemDateTimeProvider clock, NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        TatemonWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public TatemonDataAccessImpl addRow(TatemonRecord row) {
        EnumMap<TatemonColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, TatemonWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public Optional<TatemonRecord> findByBldNo(String bldNo) {
        Objects.requireNonNull(bldNo, "bldNo");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TatemonColumn.BLD_NO), bldNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByBldNoAndRecUpdJnjNo(
            TatemonWriteRecord record,
            Set<TatemonColumn> targetColumns,
            TatemonUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(row.get(TatemonColumn.BLD_NO), updateKey.bldNo()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        TatemonColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(TatemonColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                TatemonColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                TatemonDataAccess.Identifiers.UPDATE_BY_BLD_NO_AND_REC_UPD_JNJ_NO,
                updateKey.bldNo());
    }
}
