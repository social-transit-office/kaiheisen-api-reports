package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.itensak;

import java.util.ArrayList;

import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakDataAccess;

import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticUpdateSupport;

/** DA-004（C001/D001/R001-R004/U001）の SQL 振る舞いを模した ITENSAK インメモリ実装。 */
public class ItensakDataAccessImpl extends DataAccessBase implements ItensakDataAccess {

    private static final ReflectiveInMemoryRowMapper<ItensakColumn, ItensakRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(ItensakColumn.class, ItensakRecord.class);

    private static final AutoNeutralValueResolver<ItensakColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(ItensakColumn.class, ItensakRecord.class);

    private final List<EnumMap<ItensakColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<ItensakWriteRecord, ItensakColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<ItensakWriteRecord, ItensakColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public ItensakDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new NcisCisAuditColumnAutoSetter());
    }

    ItensakDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new NcisCisAuditColumnAutoSetter());
    }

    ItensakDataAccessImpl(
            SystemDateTimeProvider clock, NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        ItensakWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        ItensakWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public ItensakDataAccessImpl addRow(ItensakRecord row) {
        EnumMap<ItensakColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, ItensakWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(ItensakWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                ItensakColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
            Long gmtSetNo, Long syokyNo, Integer itenSakSeqNo, Integer recUpdJnjNo) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeSyokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        var safeItenSakSeqNo = WhereParamGuard.complement(itenSakSeqNo, "itenSakSeqNo", 0);

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        Objects.equals(row.get(ItensakColumn.GMT_SET_NO), safeGmtSetNo)
                                && Objects.equals(row.get(ItensakColumn.SYOKY_NO), safeSyokyNo)
                                && Objects.equals(
                                        row.get(ItensakColumn.ITEN_SAK_SEQ_NO),
                                        safeItenSakSeqNo),
                ItensakColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                ItensakDataAccess.CallSpecs
                        .DELETE_BY_GMT_SET_NO_AND_SYOKY_NO_AND_ITEN_SAK_SEQ_NO_AND_REC_UPD_JNJ_NO
                        .identifier(),
                () ->
                        new DataInconsistencyException(
                                "ITENSAK",
                                ItensakDataAccess.CallSpecs
                                        .DELETE_BY_GMT_SET_NO_AND_SYOKY_NO_AND_ITEN_SAK_SEQ_NO_AND_REC_UPD_JNJ_NO
                                        .identifier(),
                                "{}"),
                safeGmtSetNo,
                safeSyokyNo,
                safeItenSakSeqNo);
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndSyokyNo(Long gmtSetNo, Long syokyNo) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeSyokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);

        return rows.stream()
                .filter(
                        row ->
                                Objects.equals(row.get(ItensakColumn.GMT_SET_NO), safeGmtSetNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.SYOKY_NO), safeSyokyNo))
                .sorted(
                        Comparator.comparing(
                                row -> (Integer) row.get(ItensakColumn.ITEN_SAK_SEQ_NO),
                                Comparator.nullsLast(Comparator.reverseOrder())))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<ItensakRecord> findByGmtSetNoAndSyokyNoAndItenSakSeqNo(
            Long gmtSetNo, Long syokyNo, Integer itenSakSeqNo) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeSyokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        var safeItenSakSeqNo = WhereParamGuard.complement(itenSakSeqNo, "itenSakSeqNo", 0);

        return rows.stream()
                .filter(
                        row ->
                                Objects.equals(row.get(ItensakColumn.GMT_SET_NO), safeGmtSetNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.SYOKY_NO), safeSyokyNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.ITEN_SAK_SEQ_NO),
                                                safeItenSakSeqNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndSyokyNoAndItenTksCd(
            Long gmtSetNo, Long syokyNo, String itenTksCd) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeSyokyNo = WhereParamGuard.complement(syokyNo, "syokyNo", 0L);
        WhereParamGuard.requireNonNull(itenTksCd, "itenTksCd");

        Comparator<EnumMap<ItensakColumn, Object>> comparator =
                Comparator.<EnumMap<ItensakColumn, Object>, Long>comparing(
                                row -> (Long) row.get(ItensakColumn.GMT_SET_NO),
                                Comparator.nullsLast(Comparator.naturalOrder()))
                        .thenComparing(
                                row -> (Long) row.get(ItensakColumn.SYOKY_NO),
                                Comparator.nullsLast(Comparator.naturalOrder()))
                        .thenComparing(
                                row -> (Integer) row.get(ItensakColumn.ITEN_SAK_SEQ_NO),
                                Comparator.nullsLast(Comparator.reverseOrder()));

        return rows.stream()
                .filter(
                        row ->
                                Objects.equals(row.get(ItensakColumn.GMT_SET_NO), safeGmtSetNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.SYOKY_NO), safeSyokyNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.ITEN_TKS_CD), itenTksCd))
                .sorted(comparator)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public List<ItensakRecord> findAllByGmtSetNoAndItenSShrkyNoAndItenTksCd(
            Long gmtSetNo, Long itenSShrkyNo, String itenTksCd) {
        var safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        var safeItenSShrkyNo = WhereParamGuard.complement(itenSShrkyNo, "itenSShrkyNo", 0L);
        WhereParamGuard.requireNonNull(itenTksCd, "itenTksCd");

        return rows.stream()
                .filter(
                        row ->
                                Objects.equals(row.get(ItensakColumn.GMT_SET_NO), safeGmtSetNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.ITEN_S_SHRKY_NO),
                                                safeItenSShrkyNo)
                                        && Objects.equals(
                                                row.get(ItensakColumn.ITEN_TKS_CD), itenTksCd))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public int updateByGmtSetNoAndSyokyNoAndItenSakSeqNoAndRecUpdJnjNo(
            ItensakWriteRecord record,
            Set<ItensakColumn> targetColumns,
            ItensakUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        Objects.equals(row.get(ItensakColumn.GMT_SET_NO), updateKey.gmtSetNo())
                                && Objects.equals(row.get(ItensakColumn.SYOKY_NO), updateKey.syokyNo())
                                && Objects.equals(
                                        row.get(ItensakColumn.ITEN_SAK_SEQ_NO), updateKey.itenSakSeqNo()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        ItensakColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(ItensakColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                ItensakColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                ItensakDataAccess.CallSpecs.UPDATE_BY_GMT_SET_NO_AND_SYOKY_NO_AND_ITEN_SAK_SEQ_NO_AND_REC_UPD_JNJ_NO.identifier(),
                updateKey.gmtSetNo(),
                updateKey.syokyNo(),
                updateKey.itenSakSeqNo());
    }

    private boolean matchesWhere(
            Map<ItensakColumn, Object> row,
            Long whereGmtSetNo,
            Long whereSyokyNo,
            Integer whereItenSakSeqNo,
            Integer whereRecUpdJnjNo) {
        return Objects.equals(row.get(ItensakColumn.GMT_SET_NO), whereGmtSetNo)
                && Objects.equals(row.get(ItensakColumn.SYOKY_NO), whereSyokyNo)
                && Objects.equals(row.get(ItensakColumn.ITEN_SAK_SEQ_NO), whereItenSakSeqNo)
                && Objects.equals(row.get(ItensakColumn.REC_UPD_JNJ_NO), whereRecUpdJnjNo);
    }
}
