package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.idouktk;

import java.util.Optional;

/**
 * DA-137（C001/R001）の SQL 振る舞いを模した IDOUKTK インメモリ実装。
 */
public class IdouktkDataAccessImpl extends DataAccessBase implements IdouktkDataAccess {

    private static final ReflectiveInMemoryRowMapper<IdouktkColumn, IdouktkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(IdouktkColumn.class, IdouktkRecord.class);

    private static final AutoNeutralValueResolver<IdouktkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(IdouktkColumn.class, IdouktkRecord.class);

    private final List<EnumMap<IdouktkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<IdouktkWriteRecord, IdouktkColumn> insertSupport;

    /** デフォルトのシステム時刻プロバイダで初期化する。 */
    public IdouktkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new NcisCisAuditColumnAutoSetter());
    }

    IdouktkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new NcisCisAuditColumnAutoSetter());
    }

    IdouktkDataAccessImpl(
            SystemDateTimeProvider clock, NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        IdouktkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public IdouktkDataAccessImpl addRow(IdouktkRecord row) {
        EnumMap<IdouktkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, IdouktkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(IdouktkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                IdouktkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<Integer> findByOkyakuNoAndIdohHkkYmdAndIdoHkokuJkk(
            Long okyakuNo, LocalDate idohHkkYmd, LocalTime idoHkokuJkk) {
        Long safeOkyakuNo = WhereParamGuard.complement(okyakuNo, "okyakuNo", 0L);
        LocalDate safeIdohHkkYmd =
                WhereParamGuard.complement(idohHkkYmd, "idohHkkYmd", LocalDate.MIN);
        Integer safeIdoHkokuJkk =
                WhereParamGuard.complement(toComparableDbTime(idoHkokuJkk), "idoHkokuJkk", 0);

        return rows.stream()
                .filter(row -> Objects.equals(row.get(IdouktkColumn.OKYAKU_NO), safeOkyakuNo))
                .filter(
                        row ->
                                Objects.equals(
                                        row.get(IdouktkColumn.IDOH_HKK_YMD), safeIdohHkkYmd))
                .filter(
                        row ->
                                Objects.equals(
                                        toComparableDbTime(
                                                (LocalTime) row.get(IdouktkColumn.IDO_HKOKU_JKK)),
                                        safeIdoHkokuJkk))
                .map(row -> (Integer) row.get(IdouktkColumn.IDO_RKK_SEQ_NO))
                .filter(seqNo -> seqNo != 0)
                .max(Integer::compareTo);
    }

    private static Integer toComparableDbTime(LocalTime value) {
        if (value == null) {
            return 0;
        }
        if (value.getSecond() != 0 || value.getNano() != 0) {
            throw new IllegalArgumentException(
                    "LocalTime with seconds or nanos is not supported: " + value);
        }
        if (LocalTime.MIDNIGHT.equals(value)) {
            return 2400;
        }
        return value.getHour() * 100 + value.getMinute();
    }

    /**
     * テスト用に保持している行をレコード一覧として返す。
     *
     * @return 保持中のレコード一覧
     */
    public List<IdouktkRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }
}
