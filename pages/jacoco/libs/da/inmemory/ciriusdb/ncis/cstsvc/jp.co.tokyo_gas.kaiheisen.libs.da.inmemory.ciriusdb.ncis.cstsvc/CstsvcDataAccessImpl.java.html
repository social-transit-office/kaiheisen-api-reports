package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.cstsvc;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.cstsvc.CstsvcDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;
import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticDeleteSupport;

/**
 * DA-005-D001 の SQL を模したインメモリ実装。
 */
public class CstsvcDataAccessImpl extends DataAccessBase implements CstsvcDataAccess {

    /** 異動時抹消可否コードの固定長。 */
    private static final int IDO_MASY_KHI_CD_LENGTH = 1;

    /** テスト用に保持する CSTSVC 行データ。 */
    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public CstsvcDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    /**
     * ガスメータ設置場所番号と異動時抹消可否コードを条件に行を削除する。
     *
     * @param gmtSetNo ガスメータ設置場所番号
     * @param idoMasyKhiCd 異動時抹消可否コード
     * @param recUpdJnjNo 排他制御用のレコード更新順序番号
     * @return 削除件数
     */
    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Long safeGmtSetNo = WhereParamGuard.complement(gmtSetNo, "gmtSetNo", 0L);
        String safeIdoMasyKhiCd =
                WhereParamGuard.fixedLengthNotNull(
                        idoMasyKhiCd, "idoMasyKhiCd", IDO_MASY_KHI_CD_LENGTH);

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        safeGmtSetNo.equals(row.gmtSetNo())
                                && safeIdoMasyKhiCd.equals(row.idoMasyKhiCd()),
                InternalRow::recUpdJnjNo,
                recUpdJnjNo,
                CstsvcDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                () ->
                    new DataInconsistencyException(
                        "CSTSVC",
                        CstsvcDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                        "{}"),
                safeGmtSetNo,
                safeIdoMasyKhiCd);
    }

    /**
     * CSTSVC のテストデータ行。
     */
    public static record InternalRow(
            Long gmtSetNo,
            String svcRyuBnCd,
            String torokuNaiyoMx,
            String idoMasyKhiCd,
            LocalDate recRegYmd,
            LocalDate recUpdYmd,
            Integer recUpdJnjNo,
            String kosinUserId,
            String kosinTranNo,
            Integer kosinTaskNo) {

        /**
         * 主要キーと排他番号だけを指定してテストデータ行を生成する。
         *
         * @param gmtSetNo ガスメータ設置場所番号
         * @param svcRyuBnCd サービス留意分類コード
         * @param idoMasyKhiCd 異動時抹消可否コード
         * @param recUpdJnjNo レコード更新排他制御順序番号
         */
        public InternalRow(Long gmtSetNo, String svcRyuBnCd, String idoMasyKhiCd, Integer recUpdJnjNo) {
            this(
                    gmtSetNo,
                    svcRyuBnCd,
                    "memo",
                    idoMasyKhiCd,
                    LocalDate.of(2024, 4, 1),
                    LocalDate.of(2024, 4, 2),
                    recUpdJnjNo,
                    "USER0001",
                    "TRN1",
                    1);
        }

        /**
         * レコード生成時に必須項目が欠落していないことを検証する。
         */
        public InternalRow {
            Objects.requireNonNull(gmtSetNo, "gmtSetNo");
            Objects.requireNonNull(svcRyuBnCd, "svcRyuBnCd");
            Objects.requireNonNull(torokuNaiyoMx, "torokuNaiyoMx");
            Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");
            Objects.requireNonNull(recRegYmd, "recRegYmd");
            Objects.requireNonNull(recUpdYmd, "recUpdYmd");
            Objects.requireNonNull(kosinUserId, "kosinUserId");
            Objects.requireNonNull(kosinTranNo, "kosinTranNo");
            Objects.requireNonNull(kosinTaskNo, "kosinTaskNo");
        }
    }
}
