package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.gssntkr;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.gssntkr.GssntkrDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.gssntkr.GssntkrRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;

/**
 * DA-152-R001 の SQL を模したインメモリ実装。
 */
public class GssntkrDataAccessImpl extends DataAccessBase implements GssntkrDataAccess {

    private static final Comparator<InternalRow> ORDER_BY_SPEC = Comparator
            .comparingInt(InternalRow::recRegYmd).reversed()
            .thenComparing(InternalRow::cstBrSign, Comparator.reverseOrder());

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GssntkrDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public List<GssntkrRecord> findAllByGmtSetNo(Long gmtSetNo) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        return rows.stream()
                .filter(row -> row.gmtSetNo() == gmtSetNo)
                .sorted(ORDER_BY_SPEC)
                .map(GssntkrDataAccessImpl::toRecord)
                .toList();
    }

    private static GssntkrRecord toRecord(InternalRow row) {
        return new GssntkrRecord(
                row.gmtSetNo(),
                row.tnknTsyKbCd(),
                row.stkJkCd(),
                row.tsyGssnSign(),
                row.tsyGssnSu(),
                row.konroSetbJoho(),
                row.yoteGymCd(),
                row.cstBrSign(),
                toLocalDate(row.recRegYmd()),
                toLocalDate(row.recUpdYmd()),
                row.recUpdJnjNo());
    }

    /**
     * GSSNTKR のテストデータ行。
     */
    public static record InternalRow(
            long gmtSetNo,
            String tnknTsyKbCd,
            String stkJkCd,
            String tsyGssnSign,
            String tsyGssnSu,
            String konroSetbJoho,
            String yoteGymCd,
            String cstBrSign,
            int recRegYmd,
            int recUpdYmd,
            int recUpdJnjNo
    ) {
        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(tnknTsyKbCd, "tnknTsyKbCd");
            Objects.requireNonNull(stkJkCd, "stkJkCd");
            Objects.requireNonNull(tsyGssnSign, "tsyGssnSign");
            Objects.requireNonNull(tsyGssnSu, "tsyGssnSu");
            Objects.requireNonNull(konroSetbJoho, "konroSetbJoho");
            Objects.requireNonNull(yoteGymCd, "yoteGymCd");
            Objects.requireNonNull(cstBrSign, "cstBrSign");
        }
    }

    @Override
    public Optional<GssntkrRecord> findByPK(Long gmtSetNo, String cstBrSign) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(cstBrSign, "cstBrSign");

        for (InternalRow row : rows) {
            if (gmtSetNo.equals(row.gmtSetNo())
                    && cstBrSign.equals(row.cstBrSign())) {
                return Optional.of(toRecord(row));
            }
        }
        return Optional.empty();
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }
}
