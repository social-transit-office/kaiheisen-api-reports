package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.kenssie;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.kenssie.KenssieRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticDeleteSupport;

/**
 * KENSSIE の SQL を模したインメモリ実装。
 *
 * <p>主な仕様:</p>
 * <ul>
 *   <li>DA-009-D001: gmtSetNo と idoMasyKhiCd 一致行を削除し、削除件数を返す。</li>
 *   <li>DA-009-R001: 主キー（gmtSetNo, knsnSienBrCd）一致で 1 件取得する。</li>
 *   <li>DA-009-R002: gmtSetNo 一致 + knsnSienBrCd IN で一覧取得し、knsnSienBrCd 昇順で返す。</li>
 * </ul>
 *
 * <p>制限: テスト用途の簡易実装であり、DB の制約や例外ラップは再現しない。</p>
 */
public class KenssieDataAccessImpl extends DataAccessBase implements KenssieDataAccess {

    private static final Comparator<InternalRow> ORDER_BY_SPEC = Comparator.comparing(InternalRow::knsnSienBrCd);

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public KenssieDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> row.gmtSetNo() == gmtSetNo.longValue() && Objects.equals(row.idoMasyKhiCd(), idoMasyKhiCd),
                InternalRow::recUpdJnjNo,
                recUpdJnjNo,
                KenssieDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                () ->
                        new DataInconsistencyException(
                                "KENSSIE",
                                KenssieDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                                "{}"),
                gmtSetNo,
                idoMasyKhiCd);
    }

    @Override
    public Optional<KenssieRecord> findByGmtSetNoAndKnsnSienBrCd(Long gmtSetNo, String knsnSienBrCd) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(knsnSienBrCd, "knsnSienBrCd");

        return rows.stream()
                .filter(row -> row.gmtSetNo() == gmtSetNo.longValue())
                .filter(row -> Objects.equals(row.knsnSienBrCd(), knsnSienBrCd))
                .map(KenssieDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public List<KenssieRecord> findAllByGmtSetNoAndKnsnSienBrCd(Long gmtSetNo, List<String> knsnSienBrCdList) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(knsnSienBrCdList, "knsnSienBrCdList");
        if (knsnSienBrCdList.isEmpty()) {
            return List.of();
        }

        return rows.stream()
                .filter(row -> row.gmtSetNo() == gmtSetNo.longValue())
                .filter(row -> knsnSienBrCdList.contains(row.knsnSienBrCd()))
                .sorted(ORDER_BY_SPEC)
                .map(KenssieDataAccessImpl::toRecord)
                .toList();
    }

    private static KenssieRecord toRecord(InternalRow row) {
        return new KenssieRecord(
                row.gmtSetNo(),
                row.knsnSienBrCd(),
                row.renrakskSybCd(),
                row.idoMasyKhiCd(),
                row.renrakskTelNo(),
                row.torokuNaiyoMx(),
                row.recRegYmd(),
                row.recUpdYmd(),
                row.recUpdJnjNo(),
                row.kosinUserId(),
                row.kosinTranNo(),
                row.kosinTaskNo()
        );
    }

    /**
     * KENSSIE のテストデータ行。
     */
    public static record InternalRow(
            long gmtSetNo,
            String knsnSienBrCd,
            String renrakskSybCd,
            String idoMasyKhiCd,
            String renrakskTelNo,
            String torokuNaiyoMx,
            LocalDate recRegYmd,
            LocalDate recUpdYmd,
            int recUpdJnjNo,
            String kosinUserId,
            String kosinTranNo,
            int kosinTaskNo
    ) {
        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(knsnSienBrCd, "knsnSienBrCd");
            Objects.requireNonNull(renrakskSybCd, "renrakskSybCd");
            Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");
            Objects.requireNonNull(renrakskTelNo, "renrakskTelNo");
            Objects.requireNonNull(torokuNaiyoMx, "torokuNaiyoMx");
            Objects.requireNonNull(recRegYmd, "recRegYmd");
            Objects.requireNonNull(recUpdYmd, "recUpdYmd");
            Objects.requireNonNull(kosinUserId, "kosinUserId");
            Objects.requireNonNull(kosinTranNo, "kosinTranNo");
        }
    }
}
