/**
 * 託送メータ情報（TAKSMTR）の SQL を模したインメモリ実装。
 */
public class TaksmtrDataAccessImpl extends DataAccessBase implements TaksmtrDataAccess {

    private static final int TAKSKY_NO_LENGTH = 11;

    private static final ReflectiveInMemoryRowMapper<TaksmtrColumn, TaksmtrRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(TaksmtrColumn.class, TaksmtrRecord.class);

    private static final AutoNeutralValueResolver<TaksmtrColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(TaksmtrColumn.class, TaksmtrRecord.class);

    private static final Comparator<Map<TaksmtrColumn, Object>> ORDER_BY_TAKS_END_YMD_DESC =
            Comparator.comparing(
                            TaksmtrDataAccessImpl::getTaksEndYmd,
                            Comparator.nullsLast(Comparator.reverseOrder()))
                .thenComparing(
                    TaksmtrDataAccessImpl::getTakskyNo,
                    Comparator.nullsLast(String::compareTo))
                .thenComparing(
                    TaksmtrDataAccessImpl::getHkkmItiNo,
                    Comparator.nullsLast(Integer::compareTo))
                .thenComparing(
                    TaksmtrDataAccessImpl::getGmtSetNo,
                    Comparator.nullsLast(Long::compareTo));

    private static final Comparator<Map<TaksmtrColumn, Object>> ORDER_BY_TAKS_START_YMD_DESC =
            Comparator.comparing(
                            TaksmtrDataAccessImpl::getTaksStartYmd,
                            Comparator.nullsLast(Comparator.reverseOrder()))
                .thenComparing(
                    TaksmtrDataAccessImpl::getTakskyNo,
                    Comparator.nullsLast(String::compareTo))
                .thenComparing(
                    TaksmtrDataAccessImpl::getHkkmItiNo,
                    Comparator.nullsLast(Integer::compareTo))
                .thenComparing(
                    TaksmtrDataAccessImpl::getGmtSetNo,
                    Comparator.nullsLast(Long::compareTo));

    private final List<EnumMap<TaksmtrColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final NcisCisAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryTargetColumnUpdateSupport<TaksmtrWriteRecord, TaksmtrColumn> updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public TaksmtrDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new NcisCisAuditColumnAutoSetter());
    }

    TaksmtrDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new NcisCisAuditColumnAutoSetter());
    }

    TaksmtrDataAccessImpl(
            SystemDateTimeProvider clock,
            NcisCisAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        TaksmtrWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public TaksmtrDataAccessImpl addRow(TaksmtrRecord row) {
        EnumMap<TaksmtrColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, TaksmtrWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNo(Long gmtSetNo) {
        Long safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), safeGmtSetNo))
                .sorted(ORDER_BY_TAKS_END_YMD_DESC)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCd(
            Long gmtSetNo,
            String taksKyJtCd) {
        Long safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        String safeTaksKyJtCd =
                safeFixedCode(taksKyJtCd, "taksKyJtCd", 1);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), safeGmtSetNo))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_KY_JT_CD),
                                                1),
                                        safeTaksKyJtCd))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCdList(
            Long gmtSetNo,
            List<String> taksKyJtCdList) {
        Long safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        List<String> safeTaksKyJtCdList =
                safeFixedCodeList(taksKyJtCdList, "taksKyJtCdList", 1);
        if (safeTaksKyJtCdList.isEmpty()) {
            return List.of();
        }
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), safeGmtSetNo))
                .filter(
                        row ->
                                safeTaksKyJtCdList.contains(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_KY_JT_CD),
                                                1)))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCdListAndTaksCisHntCd(
            Long gmtSetNo,
            List<String> taksKyJtCdList,
            String taksCisHntCd) {
        Long safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        List<String> safeTaksKyJtCdList =
                safeFixedCodeList(taksKyJtCdList, "taksKyJtCdList", 1);
        String safeTaksCisHntCd =
                safeFixedCode(taksCisHntCd, "taksCisHntCd", 1);
        if (safeTaksKyJtCdList.isEmpty()) {
            return List.of();
        }
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), safeGmtSetNo))
                .filter(
                        row ->
                                safeTaksKyJtCdList.contains(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_KY_JT_CD),
                                                1)))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_CIS_HNT_CD),
                                                1),
                                        safeTaksCisHntCd))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /** {@inheritDoc} */
    @Override
    public List<TaksmtrRecord> findAllByGmtSetNoAndTaksKyJtCdListAndTaksCisHntCdOrderByTaksStartYmdDesc(
            Long gmtSetNo,
            List<String> taksKyJtCdList,
            String taksCisHntCd) {
        Long safeGmtSetNo = safeLong(gmtSetNo, "gmtSetNo");
        List<String> safeTaksKyJtCdList =
                safeFixedCodeList(taksKyJtCdList, "taksKyJtCdList", 1);
        String safeTaksCisHntCd =
                safeFixedCode(taksCisHntCd, "taksCisHntCd", 1);
        if (safeTaksKyJtCdList.isEmpty()) {
            return List.of();
        }
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), safeGmtSetNo))
                .filter(
                        row ->
                                safeTaksKyJtCdList.contains(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_KY_JT_CD),
                                                1)))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeFixed(
                                                row.get(TaksmtrColumn.TAKS_CIS_HNT_CD),
                                                1),
                                        safeTaksCisHntCd))
                .sorted(ORDER_BY_TAKS_START_YMD_DESC)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /** {@inheritDoc} */
    @Override
    public int updateByTakskyNoAndHkkmItiNoAndGmtSetNoAndRecUpdJnjNo(
            TaksmtrWriteRecord record,
            Set<TaksmtrColumn> targetColumns,
            TaksmtrUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        String safeTakskyNo = safeTakskyNo(updateKey.takskyNo());
        Integer safeHkkmItiNo = safeInteger(updateKey.hkkmItiNo(), "hkkmItiNo");
        Long safeGmtSetNo = safeLong(updateKey.gmtSetNo(), "gmtSetNo");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, safeTakskyNo, safeHkkmItiNo, safeGmtSetNo),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(TaksmtrColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? currentRecUpdJnjNo(row)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                TaksmtrColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                Identifiers.UPDATE_BY_TAKSKY_NO_AND_HKKM_ITI_NO_AND_GMT_SET_NO_AND_REC_UPD_JNJ_NO,
                safeTakskyNo,
                safeHkkmItiNo,
                safeGmtSetNo);
    }

    /** {@inheritDoc} */
    @Override
    public int updateByTakskyNoAndHkkmItiNoAndGmtSetNo(
            TaksmtrWriteRecord record,
            Set<TaksmtrColumn> targetColumns,
            TaksmtrUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        String safeTakskyNo = safeTakskyNo(updateKey.takskyNo());
        Integer safeHkkmItiNo = safeInteger(updateKey.hkkmItiNo(), "hkkmItiNo");
        Long safeGmtSetNo = safeLong(updateKey.gmtSetNo(), "gmtSetNo");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, safeTakskyNo, safeHkkmItiNo, safeGmtSetNo),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(TaksmtrColumn.class, values),
                                                context,
                                                clock,
                                                currentRecUpdJnjNo(row))),
                updateSupport::apply,
                TaksmtrColumn.REC_UPD_JNJ_NO,
                null,
                Identifiers.UPDATE_BY_TAKSKY_NO_AND_HKKM_ITI_NO_AND_GMT_SET_NO,
                safeTakskyNo,
                safeHkkmItiNo,
                safeGmtSetNo);
    }

    private static boolean matchesPk(
            Map<TaksmtrColumn, Object> row,
            String takskyNo,
            Integer hkkmItiNo,
            Long gmtSetNo) {
        return Objects.equals(
                normalizeFixed(
                    row.get(TaksmtrColumn.TAKSKY_NO),
                    TAKSKY_NO_LENGTH),
                takskyNo)
                && Objects.equals(row.get(TaksmtrColumn.HKKM_ITI_NO), hkkmItiNo)
                && Objects.equals(row.get(TaksmtrColumn.GMT_SET_NO), gmtSetNo);
    }

    private static String normalizeFixed(Object value, int length) {
        return WhereParamGuard.fixedLengthNotNull((String) value, "value", length);
    }

    private static String safeTakskyNo(String value) {
        return safeFixedCode(value, "takskyNo", TAKSKY_NO_LENGTH);
    }

    private static String safeFixedCode(String value, String paramName, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }

    private static List<String> safeFixedCodeList(List<String> values, String paramName, int length) {
        Objects.requireNonNull(values, paramName);
        if (values.isEmpty()) {
            return List.of();
        }
        return values.stream().map(value -> safeFixedCode(value, paramName, length)).toList();
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate getTaksStartYmd(Map<TaksmtrColumn, Object> row) {
        return (LocalDate) row.get(TaksmtrColumn.TAKS_START_YMD);
    }

    private static LocalDate getTaksEndYmd(Map<TaksmtrColumn, Object> row) {
        return (LocalDate) row.get(TaksmtrColumn.TAKS_END_YMD);
    }

    private static String getTakskyNo(Map<TaksmtrColumn, Object> row) {
        return (String) row.get(TaksmtrColumn.TAKSKY_NO);
    }

    private static Integer getHkkmItiNo(Map<TaksmtrColumn, Object> row) {
        return (Integer) row.get(TaksmtrColumn.HKKM_ITI_NO);
    }

    private static Long getGmtSetNo(Map<TaksmtrColumn, Object> row) {
        return (Long) row.get(TaksmtrColumn.GMT_SET_NO);
    }

    private static Integer currentRecUpdJnjNo(Map<TaksmtrColumn, Object> row) {
        Integer currentRecUpdJnjNo = (Integer) row.get(TaksmtrColumn.REC_UPD_JNJ_NO);
        return currentRecUpdJnjNo == null ? 0 : currentRecUpdJnjNo;
    }
