package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.ncis.anzntnk;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.anzntnk.AnzntnkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticDeleteSupport;

/**
 * DA-008-D001 の SQL を模したインメモリ実装。
 */
public class AnzntnkDataAccessImpl extends DataAccessBase implements AnzntnkDataAccess {

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public AnzntnkDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public int deleteByGmtSetNoAndIdoMasyKhiCd(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {
        Objects.requireNonNull(gmtSetNo, "gmtSetNo");
        Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> gmtSetNo.equals(row.gmtSetNo()) && idoMasyKhiCd.equals(row.idoMasyKhiCd()),
                InternalRow::recUpdJnjNo,
                recUpdJnjNo,
                AnzntnkDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                () ->
                    new DataInconsistencyException(
                        "ANZNTNK",
                        AnzntnkDataAccess.CallSpecs.DELETE_BY_GMT_SET_NO_AND_IDO_MASY_KHI_CD.identifier(),
                        "{}"),
                gmtSetNo,
                idoMasyKhiCd);
    }

    /**
     * ANZNTNK のテストデータ行。
     */
    public static record InternalRow(Long gmtSetNo, String idoMasyKhiCd, Integer recUpdJnjNo) {

        /**
         * 排他制御順序番号なしのテストデータ行を生成する。
         *
         * @param gmtSetNo GMT設定番号
         * @param idoMasyKhiCd 移動増差機器コード
         */
        public InternalRow(Long gmtSetNo, String idoMasyKhiCd) {
            this(gmtSetNo, idoMasyKhiCd, null);
        }

        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(gmtSetNo, "gmtSetNo");
            Objects.requireNonNull(idoMasyKhiCd, "idoMasyKhiCd");
        }
    }
}
