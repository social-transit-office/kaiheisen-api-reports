/** DA-158（ISRRKR）のインメモリ実装。 */
public class IsrrkrDataAccessImpl extends DataAccessBase implements IsrrkrDataAccess {

    private static final int ISRR_GYM_CD_LENGTH = 6;

    private static final int ISRR_GMSS_CD_LENGTH = 8;

    private static final int TJT_TRKS_FLG_LENGTH = 1;

    private static final ReflectiveInMemoryRowMapper<IsrrkrColumn, IsrrkrRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(IsrrkrColumn.class, IsrrkrRecord.class);

    private static final AutoNeutralValueResolver<IsrrkrColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(IsrrkrColumn.class, IsrrkrRecord.class);

    private static final Comparator<EnumMap<IsrrkrColumn, Object>> DESC_TSNO =
            Comparator.<EnumMap<IsrrkrColumn, Object>, Integer>comparing(
                            row -> normalizeInteger(row.get(IsrrkrColumn.ISRR_RKKR_TSNO)),
                            Comparator.nullsLast(Comparator.reverseOrder()));

    private final List<EnumMap<IsrrkrColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<IsrrkrWriteRecord, IsrrkrColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<IsrrkrWriteRecord, IsrrkrColumn> updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public IsrrkrDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    IsrrkrDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    IsrrkrDataAccessImpl(SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter = Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport = new InMemoryInsertSupport<>(IsrrkrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(IsrrkrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /** テストデータを 1 行追加する。 */
    public IsrrkrDataAccessImpl addRow(IsrrkrRecord record) {
        EnumMap<IsrrkrColumn, Object> values = ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, IsrrkrWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(IsrrkrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(IsrrkrColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<IsrrkrRecord> findByCstNoAndIsrrTriYmdAndIsrrGymCdAndIsrrGmssCdAndNotTjtTrksFlg(
            Long cstNo,
            LocalDate isrrTriYmd,
            String isrrGymCd,
            String isrrGmssCd,
            Boolean tjtTrksFlg) {
        long safeCstNo = normalizeLong(cstNo);
        int safeIsrrTriYmd = toYmdIntOrZero(isrrTriYmd);
        String safeIsrrGymCd = safeIsrrGymCd(isrrGymCd);
        String safeIsrrGmssCd = safeIsrrGmssCd(isrrGmssCd);
        String safeFlag = safeFlag(tjtTrksFlg, "tjtTrksFlg");
        return rows.stream()
                .filter(row -> normalizeLong(row.get(IsrrkrColumn.CST_NO)) == safeCstNo)
                .filter(row -> toYmdIntOrZero((LocalDate) row.get(IsrrkrColumn.ISRR_TRI_YMD)) == safeIsrrTriYmd)
            .filter(
                row ->
                    normalizeFixed(row.get(IsrrkrColumn.ISRR_GYM_CD), ISRR_GYM_CD_LENGTH)
                        .equals(safeIsrrGymCd))
            .filter(
                row ->
                    normalizeFixed(row.get(IsrrkrColumn.ISRR_GMSS_CD), ISRR_GMSS_CD_LENGTH)
                        .equals(safeIsrrGmssCd))
                .filter(row -> !normalizeFlag(row.get(IsrrkrColumn.TJT_TRKS_FLG)).equals(safeFlag))
                .sorted(DESC_TSNO)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<Integer> findMaxIsrrRkkrTsnoByCstNoAndIsrrTriYmd(Long cstNo, LocalDate isrrTriYmd) {
        long safeCstNo = normalizeLong(cstNo);
        int safeIsrrTriYmd = toYmdIntOrZero(isrrTriYmd);
        return rows.stream()
                .filter(row -> normalizeLong(row.get(IsrrkrColumn.CST_NO)) == safeCstNo)
                .filter(row -> toYmdIntOrZero((LocalDate) row.get(IsrrkrColumn.ISRR_TRI_YMD)) == safeIsrrTriYmd)
                .map(row -> normalizeInteger(row.get(IsrrkrColumn.ISRR_RKKR_TSNO)))
                .max(Comparator.naturalOrder())
                .filter(value -> value != 0);
    }

    @Override
    public int updateByCstNoAndIsrrTriYmdAndIsrrRkkrTsnoAndRecUpdJnjNo(
            IsrrkrWriteRecord record,
            Set<IsrrkrUpdatableColumn> targetColumns,
            IsrrkrU002UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Set<IsrrkrColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, updateKey.cstNo(), updateKey.isrrTriYmd(), updateKey.isrrRkkrTsno()),
                row ->
                        updateSupport.prepare(
                                record,
                                canonicalTargetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(IsrrkrColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? normalizeInteger(row.get(IsrrkrColumn.REC_UPD_JNJ_NO))
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                IsrrkrColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                IsrrkrDataAccess.CallSpecs.U002.identifier(),
                updateKey.cstNo(),
                updateKey.isrrTriYmd(),
                updateKey.isrrRkkrTsno());
    }

    @Override
    public Optional<IsrrkrRecord> findByPrimaryKey(Long cstNo, LocalDate isrrTriYmd, Integer isrrRkkrTsno) {
        return rows.stream()
                .filter(row -> matchesPk(row, cstNo, isrrTriYmd, isrrRkkrTsno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private static Set<IsrrkrColumn> canonicalTargetColumns(Set<IsrrkrUpdatableColumn> targetColumns) {
        EnumSet<IsrrkrColumn> canonicalColumns = EnumSet.noneOf(IsrrkrColumn.class);
        for (IsrrkrUpdatableColumn targetColumn : targetColumns) {
            canonicalColumns.add(IsrrkrColumn.valueOf(targetColumn.name()));
        }
        return canonicalColumns;
    }

    private static boolean matchesPk(Map<IsrrkrColumn, Object> row, Long cstNo, LocalDate isrrTriYmd, Integer isrrRkkrTsno) {
        return normalizeLong(row.get(IsrrkrColumn.CST_NO)) == normalizeLong(cstNo)
                && toYmdIntOrZero((LocalDate) row.get(IsrrkrColumn.ISRR_TRI_YMD)) == toYmdIntOrZero(isrrTriYmd)
                && normalizeInteger(row.get(IsrrkrColumn.ISRR_RKKR_TSNO)) == normalizeInteger(isrrRkkrTsno);
    }

    private static long normalizeLong(Object value) {
        if (value == null) {
            return 0L;
        }
        return ((Number) value).longValue();
    }

    private static int normalizeInteger(Object value) {
        if (value == null) {
            return 0;
        }
        return ((Number) value).intValue();
    }

    private static int toYmdIntOrZero(LocalDate value) {
        if (value == null) {
            return 0;
        }
        return value.getYear() * 10000 + value.getMonthValue() * 100 + value.getDayOfMonth();
    }

    private static String normalizeFixed(Object value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value == null ? null : value.toString(), "value", length);
    }

    private static String normalizeFlag(Object value) {
        if (value instanceof Boolean booleanValue) {
            return booleanValue ? "1" : "0";
        }
        return WhereParamGuard.fixedLengthNotNull(value == null ? null : value.toString(), "value", TJT_TRKS_FLG_LENGTH);
    }

    private static String safeIsrrGymCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGymCd", ISRR_GYM_CD_LENGTH);
    }

    private static String safeIsrrGmssCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGmssCd", ISRR_GMSS_CD_LENGTH);
    }

    private static String safeFlag(Boolean value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value == null ? null : (value ? "1" : "0"), paramName, TJT_TRKS_FLG_LENGTH);
    }
}
