/**
 * 請求書表示制御（SQSHYJS）のインメモリ実装。
 */
public class SqshyjsDataAccessImpl extends DataAccessBase implements SqshyjsDataAccess {

    private static final DateTimeFormatter BASIC_ISO_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.of(1, 1, 1);

    static final ReflectiveInMemoryRowMapper<SqshyjsColumn, SqshyjsRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SqshyjsColumn.class, SqshyjsRecord.class);

    private static final AutoNeutralValueResolver<SqshyjsColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SqshyjsColumn.class, SqshyjsRecord.class);

    private final List<EnumMap<SqshyjsColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SqshyjsWriteRecord, SqshyjsColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<SqshyjsWriteRecord, SqshyjsColumn>
            updateSupport;

    /** デフォルト依存で初期化する。 */
    public SqshyjsDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    SqshyjsDataAccessImpl(
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(SqshyjsWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        SqshyjsWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param row 追加行
     * @return このインスタンス
     */
    public SqshyjsDataAccessImpl addRow(SqshyjsRecord row) {
        EnumMap<SqshyjsColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        normalizeComplementedDates(values);
        NEUTRAL_VALUE_RESOLVER.complement(values, SqshyjsWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * 保持中の行を返す。
     *
     * @return 保持行
     */
    public List<SqshyjsRecord> getRows() {
        return rows.stream().map(SqshyjsDataAccessImpl::toPublicRecord).toList();
    }

    @Override
    public int insert(SqshyjsWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        EnumMap<SqshyjsColumn, Object> values =
                insertSupport.prepare(
                        record,
                        rawValues ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(SqshyjsColumn.class, rawValues),
                                        context,
                                        clock));
        normalizeComplementedDates(values);
        rows.add(values);
        return 1;
    }

    @Override
    public int deleteBySkshHyjsKrnoAndTkysYmd(
            Long skshHyjsKrno,
            LocalDate tkysYmd,
            Integer recUpdJnjNo) {
        Long safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), safeSkshHyjsKrno)
                        && isNotBefore(row, SqshyjsColumn.TKYS_YMD, safeTkysYmd),
                SqshyjsColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                SqshyjsDataAccess.Identifiers.DELETE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD,
                safeSkshHyjsKrno,
                safeTkysYmd);
    }

    @Override
    public Optional<SqshyjsRecord> findBySkshHyjsKrnoAndTyenYmd(Long skshHyjsKrno, LocalDate tyenYmd) {
        Long safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        LocalDate safeTyenYmd = safeYmd(tyenYmd, "tyenYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), safeSkshHyjsKrno))
                .filter(row -> sameComplementedYmd(row.get(SqshyjsColumn.TYEN_YMD), safeTyenYmd))
                .sorted(orderByTkysYmdDesc())
                .map(SqshyjsDataAccessImpl::toPublicRecord)
                .findFirst();
    }

    @Override
    public Optional<SqshyjsRecord> findBySkshHyjsKrnoAndKnskKjnYmd(
            Long skshHyjsKrno,
            LocalDate knskKjnYmd) {
        Long safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), safeSkshHyjsKrno))
                .filter(row -> isNotAfter(row, SqshyjsColumn.TKYS_YMD, safeKnskKjnYmd))
                .filter(row -> isNotBefore(row, SqshyjsColumn.TYEN_YMD, safeKnskKjnYmd))
                .sorted(orderByTkysYmdDesc())
                .map(SqshyjsDataAccessImpl::toPublicRecord)
                .findFirst();
    }

    @Override
    public Optional<SqshyjsRecord> findByPK(Long skshHyjsKrno, LocalDate tkysYmd) {
        Long safeSkshHyjsKrno = safeLong(skshHyjsKrno, "skshHyjsKrno");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), safeSkshHyjsKrno))
                .filter(row -> sameComplementedYmd(row.get(SqshyjsColumn.TKYS_YMD), safeTkysYmd))
                .map(SqshyjsDataAccessImpl::toPublicRecord)
                .findFirst();
    }

    @Override
    public int updateBySkshHyjsKrnoAndKnskKjnYmd(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsKnskKjnUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Long safeSkshHyjsKrno = safeLong(updateKey.skshHyjsKrno(), "updateKey.skshHyjsKrno");
        LocalDate safeKnskKjnYmd = safeYmd(updateKey.knskKjnYmd(), "updateKey.knskKjnYmd");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesRange(row, safeSkshHyjsKrno, safeKnskKjnYmd),
                row -> {
                    var prepared =
                            updateSupport.prepare(
                                    record,
                                    targetColumns,
                                    values ->
                                            auditColumnAutoSetter.applyForUpdate(
                                                    new MapBackedAuditColumnAccess<>(SqshyjsColumn.class, values),
                                                    context,
                                                    clock,
                                                    currentRecUpdJnjNo(row)));
                    normalizeComplementedDates(prepared.values());
                    return prepared;
                },
                updateSupport::apply,
                SqshyjsColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_KNSK_KJN_YMD,
                safeSkshHyjsKrno,
                safeKnskKjnYmd);
    }

    @Override
    public int updateBySkshHyjsKrnoAndTkysYmdAndRecUpdJnjNo(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsTkysYmdJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Long safeSkshHyjsKrno = safeLong(updateKey.skshHyjsKrno(), "updateKey.skshHyjsKrno");
        LocalDate safeTkysYmd = safeYmd(updateKey.tkysYmd(), "updateKey.tkysYmd");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, safeSkshHyjsKrno, safeTkysYmd),
                row -> {
                    var prepared =
                            updateSupport.prepare(
                                    record,
                                    targetColumns,
                                    values ->
                                            auditColumnAutoSetter.applyForUpdate(
                                                    new MapBackedAuditColumnAccess<>(SqshyjsColumn.class, values),
                                                    context,
                                                    clock,
                                                    currentRecUpdJnjNo(row)));
                    normalizeComplementedDates(prepared.values());
                    return prepared;
                },
                updateSupport::apply,
                SqshyjsColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO,
                safeSkshHyjsKrno,
                safeTkysYmd);
    }

    @Override
    public int updateBySkshHyjsKrnoAndTkysYmd(
            SqshyjsWriteRecord record,
            Set<SqshyjsColumn> targetColumns,
            SqshyjsTkysYmdUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        Long safeSkshHyjsKrno = safeLong(updateKey.skshHyjsKrno(), "updateKey.skshHyjsKrno");
        LocalDate safeTkysYmd = safeYmd(updateKey.tkysYmd(), "updateKey.tkysYmd");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, safeSkshHyjsKrno, safeTkysYmd),
                row -> {
                    var prepared =
                            updateSupport.prepare(
                                    record,
                                    targetColumns,
                                    values ->
                                            auditColumnAutoSetter.applyForUpdate(
                                                    new MapBackedAuditColumnAccess<>(SqshyjsColumn.class, values),
                                                    context,
                                                    clock,
                                                    currentRecUpdJnjNo(row)));
                    normalizeComplementedDates(prepared.values());
                    return prepared;
                },
                updateSupport::apply,
                SqshyjsColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SqshyjsDataAccess.Identifiers.UPDATE_BY_SKSH_HYJS_KRNO_AND_TKYS_YMD,
                safeSkshHyjsKrno,
                safeTkysYmd);
    }

    private static boolean matchesPk(
            Map<SqshyjsColumn, Object> row,
            Long skshHyjsKrno,
            LocalDate tkysYmd) {
        return Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), skshHyjsKrno)
                && sameComplementedYmd(row.get(SqshyjsColumn.TKYS_YMD), tkysYmd);
    }

    private static boolean matchesRange(
            Map<SqshyjsColumn, Object> row,
            Long skshHyjsKrno,
            LocalDate knskKjnYmd) {
        return Objects.equals(row.get(SqshyjsColumn.SKSH_HYJS_KRNO), skshHyjsKrno)
                && isNotAfter(row, SqshyjsColumn.TKYS_YMD, knskKjnYmd)
                && isNotBefore(row, SqshyjsColumn.TYEN_YMD, knskKjnYmd);
    }

    private static Comparator<Map<SqshyjsColumn, Object>> orderByTkysYmdDesc() {
        return Comparator.comparingInt(
                        (Map<SqshyjsColumn, Object> row) ->
                                complementedYmdValue(getLocalDateValue(row, SqshyjsColumn.TKYS_YMD)))
                .reversed();
    }

    private static boolean isNotAfter(
            Map<SqshyjsColumn, Object> row,
            SqshyjsColumn column,
            LocalDate value) {
        return complementedYmdValue(getLocalDateValue(row, column))
                <= complementedYmdValue(value);
    }

    private static boolean isNotBefore(
            Map<SqshyjsColumn, Object> row,
            SqshyjsColumn column,
            LocalDate value) {
        return complementedYmdValue(getLocalDateValue(row, column))
                >= complementedYmdValue(value);
    }

    private static boolean sameComplementedYmd(Object left, LocalDate right) {
        return complementedYmdValue((LocalDate) left) == complementedYmdValue(right);
    }

    private static int complementedYmdValue(LocalDate value) {
        LocalDate normalized = normalizeComplementedYmd(value);
        if (normalized == null) {
            return 0;
        }
        return Integer.parseInt(normalized.format(BASIC_ISO_DATE));
    }

    private static LocalDate normalizeComplementedYmd(LocalDate value) {
        if (value == null || COMPLEMENT_YMD.equals(value)) {
            return null;
        }
        return value;
    }

    private static void normalizeComplementedDates(EnumMap<SqshyjsColumn, Object> values) {
        normalizeComplementedDate(values, SqshyjsColumn.TKYS_YMD);
        normalizeComplementedDate(values, SqshyjsColumn.TYEN_YMD);
    }

    private static void normalizeComplementedDate(
            EnumMap<SqshyjsColumn, Object> values,
            SqshyjsColumn column) {
        Object value = values.get(column);
        if (value instanceof LocalDate localDate && COMPLEMENT_YMD.equals(localDate)) {
            values.put(column, null);
        }
    }

    private static SqshyjsRecord toPublicRecord(Map<SqshyjsColumn, Object> row) {
        SqshyjsRecord record = ROW_MAPPER.toRecord(row);
        return new SqshyjsRecord(
                record.skshHyjsKrno(),
                normalizeComplementedYmd(record.tkysYmd()),
                normalizeComplementedYmd(record.tyenYmd()),
                record.syryHjumFlg(),
                record.sqddHjumFlg(),
                record.skshYuskKbn(),
                record.skmHjjnShbt(),
                record.skdlRyumFlg(),
                record.skshAtnKj1(),
                record.skshAtnKj2(),
                record.sqkuYuskKbn(),
                record.kuinStiKbn(),
                record.regTsmp(),
                record.regUsrId(),
                record.updTsmp(),
                record.updUsrId(),
                record.updKtnKinoId(),
                record.recUpdJnjNo());
    }

    private static LocalDate getLocalDateValue(
            Map<SqshyjsColumn, Object> row,
            SqshyjsColumn column) {
        return (LocalDate) row.get(column);
    }

    private static Integer currentRecUpdJnjNo(Map<SqshyjsColumn, Object> row) {
        Object value = row.get(SqshyjsColumn.REC_UPD_JNJ_NO);
        if (value == null) {
            return 0;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.parseInt(value.toString());
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return normalizeComplementedYmd(
                WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD));
    }
