package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.irslkkz;

import java.util.EnumMap;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslkkz.IrslkkzDataAccess;

/**
 * DA-030（IRSLKKZ）のインメモリ実装.
 */
public class IrslkkzDataAccessImpl extends DataAccessBase implements IrslkkzDataAccess {

    private static final ReflectiveInMemoryRowMapper<IrslkkzColumn, IrslkkzRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(IrslkkzColumn.class, IrslkkzRecord.class);

    private static final AutoNeutralValueResolver<IrslkkzColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(IrslkkzColumn.class, IrslkkzRecord.class);

    private final List<EnumMap<IrslkkzColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<IrslkkzWriteRecord, IrslkkzColumn> insertSupport;

    private int nextIsrkLogTsno = 1;

    /** デフォルト依存で初期化する。 */
    public IrslkkzDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    IrslkkzDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(IrslkkzWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param row 追加行
     * @return このインスタンス
     */
    public IrslkkzDataAccessImpl addRow(IrslkkzRecord row) {
        EnumMap<IrslkkzColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, IrslkkzWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        Integer isrkLogTsno = (Integer) values.get(IrslkkzColumn.ISRK_LOG_TSNO);
        if (isrkLogTsno != null) {
            nextIsrkLogTsno = Math.max(nextIsrkLogTsno, isrkLogTsno + 1);
        }
        return this;
    }

    @Override
    public int insert(IrslkkzWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        EnumMap<IrslkkzColumn, Object> values =
                insertSupport.prepare(
                        record,
                        rawValues ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(IrslkkzColumn.class, rawValues),
                                        context,
                                        clock));
        values.put(IrslkkzColumn.ISRK_LOG_TSNO, nextIsrkLogTsno++);
        rows.add(values);
        return 1;
    }

    @Override
    public Optional<IrslkkzRecord> findByPK(Integer isrkLogTsno) {
        Integer safeIsrkLogTsno = WhereParamGuard.complement(isrkLogTsno, "isrkLogTsno", 0);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(IrslkkzColumn.ISRK_LOG_TSNO), safeIsrkLogTsno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }
}
