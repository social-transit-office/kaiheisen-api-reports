    /** R004 の並び順. */
    private static final Comparator<EnumMap<HseflColumn, Object>> R004_ORDER =
            Comparator
                    .<EnumMap<HseflColumn, Object>, LocalDate>comparing(
                            row -> (LocalDate) row.get(HseflColumn.HSSG_YMD),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row ->
                                    (LocalDate) row.get(
                                            HseflColumn.HESN_HUKK_YMD),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row -> (Integer) row.get(HseflColumn.REG_TSNO),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row -> (String) row.get(HseflColumn.HSSG_NWK_CD),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row -> (Long) row.get(HseflColumn.GAS_MTST_NO),
                            Comparator.nullsLast(Comparator.naturalOrder()));

    /**
     * デフォルト依存で初期化する.
     */
    public HseflDataAccessImpl() {
        this(
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * 依存注入用に初期化する.
     *
     * @param systemDateTimeProvider システム日時プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査列自動設定サポート
     */
    HseflDataAccessImpl(
            final SystemDateTimeProvider systemDateTimeProvider,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(
                systemDateTimeProvider,
                "systemDateTimeProvider");
        this.auditColumnAutoSetter = Objects.requireNonNull(
                ciriusAuditColumnAutoSetter,
                "ciriusAuditColumnAutoSetter");
        this.insertSupport = new InMemoryInsertSupport<>(
                HseflWriteDefinition.INSTANCE,
                NEUTRAL_VALUE_RESOLVER);
        this.updateSupport = new InMemoryTargetColumnUpdateSupport<>(
                HseflWriteDefinition.INSTANCE,
                NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に 1 行追加する.
     *
     * @param row 追加レコード
     * @return このインスタンス
     */
    public HseflDataAccessImpl addRow(final HseflRecord row) {
        final EnumMap<HseflColumn, Object> values = ROW_MAPPER.toRawValueMap(
                Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(
                values,
                HseflWriteDefinition.INSTANCE.columnOrder());
        this.rows.add(values);
        return this;
    }

    @Override
    public int insert(
            final HseflWriteRecord record,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        this.rows.add(this.insertSupport.prepare(
                normalizeInsertRecord(record),
                values -> this.auditColumnAutoSetter.applyForInsert(
                        new MapBackedAuditColumnAccess<>(
                                HseflColumn.class,
                                values),
                        context,
                        this.clock)));
        return 1;
    }

    private static HseflWriteRecord normalizeInsertRecord(
            final HseflWriteRecord record) {
        return new HseflWriteRecord(
                record.gasMtstNo(),
                record.hssgYmd(),
                record.hesnHukkYmd(),
                record.regTsno(),
                record.hukkSysShbt(),
                record.sgyAnknNo(),
                record.flwStCd(),
                Objects.requireNonNullElse(record.fltkYmd(), DEFAULT_FOLLOW_TARGET_DATE),
                record.syoKyoTrno(),
                record.gmb(),
                record.hesnJysb(),
                record.hesnErryCd(),
                record.flttCktkCd(),
                record.flttKtnCd(),
                record.hssgNwkCd(),
                record.sgytSinNo(),
                record.sgytSinnKj(),
                record.hsnjGmsSu(),
                record.hsnjGasSyry(),
                record.sjskKbn(),
                Objects.requireNonNullElse(record.rshhSjsb(), DEFAULT_CONTACT_RESIDENCE_TYPE),
                Objects.requireNonNullElse(record.hkshHshhShbt(), DEFAULT_OPEN_CLOSE_METHOD_TYPE),
                Objects.requireNonNullElse(record.hesnSdnShbt(), DEFAULT_CLOSING_MEANS_TYPE),
                record.utidxNo(),
                record.ctnmKn1(),
                record.ctnmKj1(),
                record.ctnmKjhkFlg(),
                record.ctnm1HujnShbt(),
                record.ctnm1HshiShbt(),
                Objects.requireNonNullElse(record.tsigHesnSgyYmd(), DEFAULT_CORRECTED_CLOSING_WORK_DATE),
                Objects.requireNonNullElse(record.tsigGmb(), DEFAULT_CORRECTED_GAS_METER_SERIAL_NUMBER),
                Objects.requireNonNullElse(record.sihKiyNo(), DEFAULT_PAYMENT_CONTRACT_NUMBER),
                Objects.requireNonNullElse(record.svkyNo(), DEFAULT_SERVICE_CONTRACT_NUMBER),
                Objects.requireNonNullElse(record.tkjMem(), DEFAULT_SPECIAL_NOTES));
    }

    @Override
    public int deleteByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsnoAndRecUpdJnjNo(
            final Long gasMtstNo,
            final LocalDate hssgYmd,
            final LocalDate hesnHukkYmd,
            final Integer regTsno,
            final Integer recUpdJnjNo) {
        final Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        final LocalDate safeHssgYmd = safeYmd(hssgYmd, "hssgYmd");
        final LocalDate safeHesnHukkYmd = safeYmd(hesnHukkYmd, "hesnHukkYmd");
        final Integer safeRegTsno = safeInt(regTsno, "regTsno");
        return InMemoryOptimisticDeleteSupport.delete(
                this.rows,
                row -> Objects.equals(
                        normalizeRowLong(
                                row.get(HseflColumn.GAS_MTST_NO),
                                "gasMtstNo"),
                        safeGasMtstNo)
                        && Objects.equals(
                                normalizeRowYmd(
                                        row.get(HseflColumn.HSSG_YMD),
                                        "hssgYmd"),
                                safeHssgYmd)
                        && Objects.equals(
                                normalizeRowYmd(
                                        row.get(HseflColumn.HESN_HUKK_YMD),
                                        "hesnHukkYmd"),
                                safeHesnHukkYmd)
                        && Objects.equals(
                                normalizeRowInt(
                                        row.get(HseflColumn.REG_TSNO),
                                        "regTsno"),
                                safeRegTsno),
                HseflColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                D001_IDENTIFIER,
                () -> new DataInconsistencyException("HSEFL", D001_IDENTIFIER, "{}"),
                safeGasMtstNo,
                safeHssgYmd,
                safeHesnHukkYmd,
                safeRegTsno);
    }

    @Override
    public Optional<HseflRecord>
            findByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsno(
            final Long gasMtstNo,
            final LocalDate hssgYmd,
            final LocalDate hesnHukkYmd,
            final Integer regTsno) {
        final Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");

    @Override
    public List<HseflRecord> findAllByGasMtstNoAndFlwStCd(

        final String safeFlwStCd = WhereParamGuard.fixedLengthNotNull(

                        safeFlwStCd))

                .toList();
    }

    @Override
    public List<HseflRecord>
            findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                    final String flttCktkCd,
                    final String flttKtnCd,
                    final String flwStCd,
                    final List<String> hesnJysbList) {
        final String safeFlttCktkCd = WhereParamGuard.fixedLengthNotNull(
                flttCktkCd,
                "flttCktkCd",
                FLTT_CKTK_CD_LENGTH);
        final String safeFlttKtnCd = flttKtnCd == null
                ? null
                : WhereParamGuard.fixedLengthNotNull(
                        flttKtnCd,
                        "flttKtnCd",
                        FLTT_KTN_CD_LENGTH);
        final String safeFlwStCd = WhereParamGuard.fixedLengthNotNull(
                flwStCd,
                "flwStCd",
                FLW_ST_CD_LENGTH);
        final String blankFlttKtnCd = WhereParamGuard.fixedLengthNotNull(
                "",
                "flttKtnCd",
                FLTT_KTN_CD_LENGTH);
        final List<String> safeHesnJysbList = safeHesnJysbList(hesnJysbList);
        return this.rows.stream()
                .filter(row -> Objects.equals(
                        normalizeRowFixedLength(
                                (String) row.get(HseflColumn.FLTT_CKTK_CD),
                                "flttCktkCd",
                                FLTT_CKTK_CD_LENGTH),
                        safeFlttCktkCd))
                .filter(row -> {
                    final String normalizedFlttKtnCd = normalizeRowFixedLength(
                            (String) row.get(HseflColumn.FLTT_KTN_CD),
                            "flttKtnCd",
                            FLTT_KTN_CD_LENGTH);
                    return safeFlttKtnCd == null
                            || Objects.equals(
                                    normalizedFlttKtnCd,
                                    safeFlttKtnCd)
                            || Objects.equals(
                                    normalizedFlttKtnCd,
                                    blankFlttKtnCd);
                })
                .filter(row -> Objects.equals(
                        normalizeRowFixedLength(
                                (String) row.get(HseflColumn.FLW_ST_CD),
                                "flwStCd",
                                FLW_ST_CD_LENGTH),
                        safeFlwStCd))
                .filter(row -> safeHesnJysbList.isEmpty()
                        || safeHesnJysbList.contains(
                                normalizeRowFixedLength(
                                        (String) row.get(HseflColumn.HESN_JYSB),
                                        "hesnJysb",
                                        HESN_JYSB_LENGTH)))
                .sorted(R004_ORDER)
                .map(HseflDataAccessImpl::toReadRecord)
                .toList();
    }

    private static HseflRecord toReadRecord(
            final EnumMap<HseflColumn, Object> row) {
        final HseflRecord actual = ROW_MAPPER.toRecord(row);
        final LocalDate fltkYmd = COMPLEMENT_YMD.equals(actual.fltkYmd())
                ? null
                : actual.fltkYmd();
        final LocalDate tsigHesnSgyYmd = COMPLEMENT_YMD.equals(actual.tsigHesnSgyYmd())
                ? null
                : actual.tsigHesnSgyYmd();
        if (Objects.equals(fltkYmd, actual.fltkYmd())
                && Objects.equals(tsigHesnSgyYmd, actual.tsigHesnSgyYmd())) {
            return actual;
        }
        return new HseflRecord(
                actual.gasMtstNo(),
                actual.hssgYmd(),
                actual.hesnHukkYmd(),
                actual.regTsno(),
                actual.hukkSysShbt(),
                actual.sgyAnknNo(),
                actual.flwStCd(),
                fltkYmd,
                actual.syoKyoTrno(),
                actual.gmb(),
                actual.hesnJysb(),
                actual.hesnErryCd(),
                actual.flttCktkCd(),
                actual.flttKtnCd(),
                actual.hssgNwkCd(),
                actual.sgytSinNo(),
                actual.sgytSinnKj(),
                actual.hsnjGmsSu(),
                actual.hsnjGasSyry(),
                actual.sjskKbn(),
                actual.rshhSjsb(),
                actual.hkshHshhShbt(),
                actual.hesnSdnShbt(),
                actual.utidxNo(),
                actual.ctnmKn1(),
                actual.ctnmKj1(),
                actual.ctnmKjhkFlg(),
                actual.ctnm1HujnShbt(),
                actual.ctnm1HshiShbt(),
                tsigHesnSgyYmd,
                actual.tsigGmb(),
                actual.sihKiyNo(),
                actual.svkyNo(),
                actual.tkjMem(),
                actual.regTsmp(),
                actual.regUsrId(),
                actual.updTsmp(),
                actual.updUsrId(),
                actual.updKtnKinoId(),
                actual.recUpdJnjNo());
    }

    /**
     * 固定長文字列を正規化する.
     *
     * @param value 値
     * @param paramName パラメータ名
     * @param length 固定長
     * @return 正規化値
     */
    private static String normalizeRowFixedLength(
            final String value,
            final String paramName,
            final int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }

    /**
     * Long 値を正規化する.
     *
     * @param value 値
     * @param paramName パラメータ名
     * @return 正規化値
     */
    private static Long normalizeRowLong(
            final Object value,
            final String paramName) {
        return WhereParamGuard.complement((Long) value, paramName, 0L);
    }

    /**
     * Integer 値を正規化する.
     *
     * @param value 値
     * @param paramName パラメータ名
     * @return 正規化値
     */
    private static Integer normalizeRowInt(
            final Object value,
            final String paramName) {
        return WhereParamGuard.complement((Integer) value, paramName, 0);
    }

    /**
     * 日付値を正規化する.

            final Object value,

    private static LocalDate safeYmd(

    /**
     * 閉栓住種別一覧を安全補完する.
     *
     * @param hesnJysbList 閉栓住種別一覧
     * @return 補完後一覧
     */
    private static List<String> safeHesnJysbList(
            final List<String> hesnJysbList) {
        if (hesnJysbList == null || hesnJysbList.isEmpty()) {
            return List.of();
        }
        return hesnJysbList.stream()
                .filter(Objects::nonNull)
                .filter(value -> !value.isBlank())
                .map(HseflDataAccessImpl::validateAndFormatHesnJysb)
                .toList();
    }

    /**
     * 閉栓住種別の入力値を検証し固定長整形する.
     *
     * @param value 閉栓住種別
     * @return 固定長整形後の値
     */
    private static String validateAndFormatHesnJysb(
            final String value) {
        return WhereParamGuard.fixedLengthNotNull(
                value,
                "hesnJysbList",
                HESN_JYSB_LENGTH);
    }
