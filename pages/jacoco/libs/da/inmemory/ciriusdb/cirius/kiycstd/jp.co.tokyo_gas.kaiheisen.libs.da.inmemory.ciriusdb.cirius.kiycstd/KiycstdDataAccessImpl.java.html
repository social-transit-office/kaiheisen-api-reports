/** DA-104（C001/R001/U001）の SQL 振る舞いを模した KIYCSTD インメモリ実装。 */
public class KiycstdDataAccessImpl extends DataAccessBase implements KiycstdDataAccess {

    private static final ReflectiveInMemoryRowMapper<KiycstdColumn, KiycstdRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KiycstdColumn.class, KiycstdRecord.class);

    private static final AutoNeutralValueResolver<KiycstdColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KiycstdColumn.class, KiycstdRecord.class);

    private final List<EnumMap<KiycstdColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KiycstdWriteRecord, KiycstdColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KiycstdWriteRecord, KiycstdColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KiycstdDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KiycstdDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KiycstdDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KiycstdWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KiycstdWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public KiycstdDataAccessImpl addRow(KiycstdRecord record) {
        EnumMap<KiycstdColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KiycstdWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KiycstdWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KiycstdColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<KiycstdRecord> findByKyoTrkNo(Long kyoTrkNo) {
        Long safeKyoTrkNo = WhereParamGuard.complement(kyoTrkNo, "kyoTrkNo", 0L);
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KiycstdColumn.KYO_TRK_NO), safeKyoTrkNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByKyoTrkNo(
            KiycstdWriteRecord record,
            Set<KiycstdColumn> targetColumns,
            KiycstdUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(row.get(KiycstdColumn.KYO_TRK_NO), updateKey.kyoTrkNo()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KiycstdColumn.class, values),
                                                context,
                                                clock,
                                                (Integer) row.get(KiycstdColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                KiycstdColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KiycstdDataAccess.CallSpecs.UPDATE_BY_KYO_TRK_NO.identifier(),
                updateKey.kyoTrkNo());
    }
}
