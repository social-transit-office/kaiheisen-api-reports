/** DA-038（C001/R001-R013/U001-U003）の SQL 振る舞いを模した SVCKYA インメモリ実装。 */
public class SvckyaDataAccessImpl extends DataAccessBase implements SvckyaDataAccess {

    private static final int SVKY_JTKB_LENGTH = 2;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private static final ReflectiveInMemoryRowMapper<SvckyaColumn, SvckyaRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SvckyaColumn.class, SvckyaRecord.class);

    private static final AutoNeutralValueResolver<SvckyaColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SvckyaColumn.class, SvckyaRecord.class);

    private final List<EnumMap<SvckyaColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SvckyaWriteRecord, SvckyaColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<SvckyaWriteRecord, SvckyaColumn> updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public SvckyaDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    SvckyaDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    SvckyaDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(SvckyaWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        SvckyaWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public SvckyaDataAccessImpl addRow(SvckyaRecord record) {
        EnumMap<SvckyaColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, SvckyaWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * テスト用に保持している登録済みレコードを返す。
     *
     * @return 登録済みレコード一覧
     */
    public List<SvckyaRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(SvckyaWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                SvckyaColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<SvckyaRecord> findFirstBySvkyNoAndSvkyJtkb(Long svkyNo, String svkyJtkb) {
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        String safeSvkyJtkb = safeSvkyJtkb(svkyJtkb, "svkyJtkb");
        return rows.stream()
                .filter(r -> Objects.equals(normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)), safeSvkyNo))
                .filter(r -> Objects.equals(normalizeSvkyJtkb((String) r.get(SvckyaColumn.SVKY_JTKB)), safeSvkyJtkb))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SvckyaRecord> findLatestBySvkyNoAndSvkyJtkbNot(Long svkyNo) {
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        return rows.stream()
                .filter(r -> Objects.equals(normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)), safeSvkyNo))
                .max(
                        Comparator.<EnumMap<SvckyaColumn, Object>, Integer>comparing(
                                r -> normalizeSkno((Integer) r.get(SvckyaColumn.SKNO))))
                .map(ROW_MAPPER::toRecord);
    }

    @Override
    public Optional<SvckyaRecord> findLatestBySvkyNoAndSvkyJtkbList(
            Long svkyNo, List<String> svkyJtkbList) {
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        Objects.requireNonNull(svkyJtkbList, "svkyJtkbList");
        if (svkyJtkbList.isEmpty()) {
            return Optional.empty();
        }
        return rows.stream()
                .filter(r -> Objects.equals(normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)), safeSvkyNo))
                .filter(r -> svkyJtkbList.contains(r.get(SvckyaColumn.SVKY_JTKB)))
                .max(
                        Comparator.<EnumMap<SvckyaColumn, Object>, LocalDate>comparing(
                                r -> normalizeYmd((LocalDate) r.get(SvckyaColumn.RKKI_STT_YMD))))
                .map(ROW_MAPPER::toRecord);
    }

    @Override
    public List<SvckyaRecord> findAllBySvkyNoAndRkkiSttYmdAfterAndSvkyJtkbNotOrderByRkkiSttYmd(
            Long svkyNo, LocalDate rkkiSttYmd, String svkyJtkb) {
        if (svkyJtkb == null || svkyJtkb.isBlank()) {
            return List.of();
        }
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        LocalDate safeRkkiSttYmd = safeYmd(rkkiSttYmd, "rkkiSttYmd");
        String safeSvkyJtkb = safeSvkyJtkb(svkyJtkb, "svkyJtkb");
        return rows.stream()
                .filter(r -> Objects.equals(normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)), safeSvkyNo))
                .filter(r -> normalizeYmd((LocalDate) r.get(SvckyaColumn.RKKI_STT_YMD)).isAfter(safeRkkiSttYmd))
                .filter(r -> !Objects.equals(normalizeSvkyJtkb((String) r.get(SvckyaColumn.SVKY_JTKB)), safeSvkyJtkb))
                .sorted(
                        Comparator.<EnumMap<SvckyaColumn, Object>, LocalDate>comparing(
                                r -> normalizeYmd((LocalDate) r.get(SvckyaColumn.RKKI_STT_YMD))))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<SvckyaRecord> findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNotOrderByRkkiSttYmd(
            Long svkyNo, LocalDate knskKjnYmd, String svkyJtkb) {
        if (svkyJtkb == null || svkyJtkb.isBlank()) {
            return Optional.empty();
        }
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        String safeSvkyJtkb = safeSvkyJtkb(svkyJtkb, "svkyJtkb");
        return rows.stream()
                .filter(
                        r ->
                                Objects.equals(
                                        normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)),
                                        safeSvkyNo))
                .filter(
                        r ->
                                !Objects.equals(
                                        normalizeSvkyJtkb(
                                                (String) r.get(SvckyaColumn.SVKY_JTKB)),
                                        safeSvkyJtkb))
                .filter(
                        r ->
                                !normalizeYmd((LocalDate) r.get(SvckyaColumn.RKKI_STT_YMD))
                                        .isAfter(safeKnskKjnYmd))
                .max(
                        Comparator.<EnumMap<SvckyaColumn, Object>, LocalDate>comparing(
                                r -> normalizeYmd((LocalDate) r.get(SvckyaColumn.RKKI_STT_YMD))))
                .map(ROW_MAPPER::toRecord);
    }

    @Override
    public Optional<Integer> findMaxKyttSknoBySvkyNo(Long svkyNo) {
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        return rows.stream()
                .filter(
                        r ->
                                Objects.equals(
                                        normalizeSvkyNo((Long) r.get(SvckyaColumn.SVKY_NO)),
                                        safeSvkyNo))
                .map(r -> (Integer) r.get(SvckyaColumn.KYTT_SKNO))
                .max(Integer::compareTo);
    }

    private static Long safeSvkyNo(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeSkno(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String safeSvkyJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SVKY_JTKB_LENGTH);
    }

    private static Long normalizeSvkyNo(Long value) {
        return WhereParamGuard.complement(value, "svkyNo", 0L);
    }

    private static Integer normalizeSkno(Integer value) {
        return WhereParamGuard.complement(value, "skno", 0);
    }

    private static LocalDate normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(value, "ymd", COMPLEMENT_YMD);
    }

    private static String normalizeSvkyJtkb(String value) {
        return safeSvkyJtkb(value, "svkyJtkb");
    }

    @Override
    public int updateBySvkyNoAndSkno(
            SvckyaWriteRecord record,
            Set<SvckyaColumn> targetColumns,
            SvckyaSknoUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        Objects.equals(row.get(SvckyaColumn.SVKY_NO), updateKey.svkyNo())
                                && Objects.equals(row.get(SvckyaColumn.SKNO), updateKey.skno()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        SvckyaColumn.class, values),
                                                context,
                                                clock,
                                                (Integer) row.get(SvckyaColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                SvckyaColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SvckyaDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_SKNO.identifier(),
                updateKey.svkyNo(),
                updateKey.skno());
    }
