/** DA-044 の SQL 挙動を再現するインメモリ実装。 */
public class SvsihkrDataAccessImpl extends DataAccessBase implements SvsihkrDataAccess {

    private static final int SIH_KIY_JTKB_LENGTH = 2;

    private static final int SHZI_KBN_LENGTH = 2;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private static final ReflectiveInMemoryRowMapper<SvsihkrColumn, SvsihkrRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SvsihkrColumn.class, SvsihkrRecord.class);

    private static final AutoNeutralValueResolver<SvsihkrColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SvsihkrColumn.class, SvsihkrRecord.class);

    private final List<EnumMap<SvsihkrColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SvsihkrWriteRecord, SvsihkrColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<SvsihkrWriteRecord, SvsihkrColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public SvsihkrDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用の依存差し替え付きで初期化する。
     *
     * @param clock システム日時プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    SvsihkrDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(SvsihkrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        SvsihkrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用にレコードを追加する。
     *
     * @param record レコード
     * @return このインスタンス
     */
    public SvsihkrDataAccessImpl addRow(SvsihkrRecord record) {
        EnumMap<SvsihkrColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, SvsihkrWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * テスト用に保持している登録済みレコードを返す。
     *
     * @return 登録済みレコード一覧
     */
    public List<SvsihkrRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(SvsihkrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(SvsihkrColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public List<SvsihkrRecord> findAllBySihKiyNoAndSvkyNoAndSihKiyJtkbAndShziKbn(
            Long sihKiyNo, Long svkyNo, String sihKiyJtkb, String shziKbn) {
        if (svkyNo == null) {
            return List.of();
        }
        return rows.stream()
                .filter(row -> hasValue(row, SvsihkrColumn.SIH_KIY_NO, sihKiyNo))
                .filter(row -> !hasValue(row, SvsihkrColumn.SVKY_NO, svkyNo))
                .filter(row -> hasValue(row, SvsihkrColumn.SIH_KIY_JTKB, sihKiyJtkb))
                .filter(row -> hasValue(row, SvsihkrColumn.SHZI_KBN, shziKbn))
                .sorted(Comparator.comparing(row -> normalizedLongValue(row, SvsihkrColumn.SVKY_NO)))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public List<SvsihkrRecord> findAllBySvkyNoAndSihKiyJtkb(Long svkyNo, String sihKiyJtkb) {
        return rows.stream()
                .filter(row -> hasValue(row, SvsihkrColumn.SVKY_NO, svkyNo))
                .filter(row -> hasValue(row, SvsihkrColumn.SIH_KIY_JTKB, sihKiyJtkb))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<SvsihkrRecord> findBySihKiyNoAndSvkyNoAndSihKiyJtkb(
            Long sihKiyNo, Long svkyNo, String sihKiyJtkb) {
        if (sihKiyJtkb == null || sihKiyJtkb.isBlank()) {
            return Optional.empty();
        }
        return rows.stream()
                .filter(row -> hasValue(row, SvsihkrColumn.SIH_KIY_NO, sihKiyNo))
                .filter(row -> hasValue(row, SvsihkrColumn.SVKY_NO, svkyNo))
                .filter(row -> !hasValue(row, SvsihkrColumn.SIH_KIY_JTKB, sihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                        (Map<SvsihkrColumn, Object> row) ->
                                                normalizedIntegerValue(row, SvsihkrColumn.SKNO))
                                .reversed())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SvsihkrRecord> findBySvkyNoAndTkysYmdAndSihKiyJtkb(
            Long svkyNo, LocalDate tkysYmd, String sihKiyJtkb) {
        return rows.stream()
                .filter(row -> hasValue(row, SvsihkrColumn.SVKY_NO, svkyNo))
                .filter(row -> isNotAfter(row, SvsihkrColumn.TKYS_YMD, tkysYmd))
                .filter(row -> hasValue(row, SvsihkrColumn.SIH_KIY_JTKB, sihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                        (Map<SvsihkrColumn, Object> row) ->
                                                normalizedLocalDateValue(row, SvsihkrColumn.TKYS_YMD))
                                .reversed())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSvkyNoAndSkno(
            SvsihkrWriteRecord record,
            Set<SvsihkrColumn> targetColumns,
            SvsihkrSknoOptionalUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesSkno(row, updateKey.sihKiyNo(), updateKey.svkyNo(), updateKey.skno()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        SvsihkrColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? currentRecUpdJnjNo(row)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                SvsihkrColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SvsihkrDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SVKY_NO_AND_SKNO.identifier(),
                updateKey.sihKiyNo(),
                updateKey.svkyNo(),
                updateKey.skno());
    }

    @Override
    public Optional<SvsihkrRecord> findByPK(Long sihKiyNo, Long svkyNo, Integer skno) {
        return rows.stream()
                .filter(row -> matchesSkno(row, sihKiyNo, svkyNo, skno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private static boolean matchesSkno(
            Map<SvsihkrColumn, Object> row, Long sihKiyNo, Long svkyNo, Integer skno) {
        return hasValue(row, SvsihkrColumn.SIH_KIY_NO, sihKiyNo)
                && hasValue(row, SvsihkrColumn.SVKY_NO, svkyNo)
                && hasValue(row, SvsihkrColumn.SKNO, skno);
    }

    private static boolean hasValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column, Object value) {
        return Objects.equals(normalizeValue(column, row.get(column)), normalizeValue(column, value));
    }

    private static boolean isNotAfter(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column, LocalDate baseDate) {
        LocalDate value = normalizeYmd(getLocalDateValue(row, column));
        return !value.isAfter(normalizeYmd(baseDate));
    }

    private static boolean isNotBefore(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column, LocalDate baseDate) {
        LocalDate value = normalizeYmd(getLocalDateValue(row, column));
        return !value.isBefore(normalizeYmd(baseDate));
    }

    private static Long normalizedLongValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (Long) normalizeValue(column, row.get(column));
    }

    private static Integer normalizedIntegerValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (Integer) normalizeValue(column, row.get(column));
    }

    private static LocalDate normalizedLocalDateValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (LocalDate) normalizeValue(column, row.get(column));
    }

    private static Long getLongValue(Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (Long) row.get(column);
    }

    private static Integer getIntegerValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (Integer) row.get(column);
    }

    private static String getStringValue(Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (String) row.get(column);
    }

    private static LocalDate getLocalDateValue(
            Map<SvsihkrColumn, Object> row, SvsihkrColumn column) {
        return (LocalDate) row.get(column);
    }

    private static Object normalizeValue(SvsihkrColumn column, Object value) {
        return switch (column) {
            case SIH_KIY_NO, SVKY_NO, SNSK_SVKY_NO -> safeLong((Long) value, column.name());
            case SKNO -> safeInteger((Integer) value, column.name());
            case TKYS_YMD, TYEN_YMD -> safeYmd((LocalDate) value, column.name());
            case SIH_KIY_JTKB -> safeSihKiyJtkb((String) value, column.name());
            case SHZI_KBN -> safeShziKbn((String) value, column.name());
            default -> value;
        };
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String safeSihKiyJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SIH_KIY_JTKB_LENGTH);
    }

    private static String safeShziKbn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SHZI_KBN_LENGTH);
    }

    private static LocalDate normalizeYmd(LocalDate value) {
        return safeYmd(value, "ymd");
    }

    private static Integer currentRecUpdJnjNo(Map<SvsihkrColumn, Object> row) {
        Integer currentRecUpdJnjNo = (Integer) row.get(SvsihkrColumn.REC_UPD_JNJ_NO);
        return currentRecUpdJnjNo == null ? 0 : currentRecUpdJnjNo;
    }
