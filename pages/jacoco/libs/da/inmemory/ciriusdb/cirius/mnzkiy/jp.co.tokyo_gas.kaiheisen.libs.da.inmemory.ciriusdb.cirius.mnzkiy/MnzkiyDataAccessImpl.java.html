/** DA-123（MNZKIY）のインメモリ実装。 */
public class MnzkiyDataAccessImpl extends DataAccessBase implements MnzkiyDataAccess {

    private static final DateTimeFormatter BASIC_ISO_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.of(1, 1, 1);

    /** 行データとレコードの相互変換を担うマッパー。 */
    private static final ReflectiveInMemoryRowMapper<MnzkiyColumn, MnzkiyRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(MnzkiyColumn.class, MnzkiyRecord.class);

    /** write レコードの未設定項目を補完する中立値リゾルバ。 */
    private static final AutoNeutralValueResolver<MnzkiyColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(MnzkiyColumn.class, MnzkiyRecord.class);

    /** R002 の開始年月日降順ソート順。 */
    private static final Comparator<EnumMap<MnzkiyColumn, Object>> R002_ORDER =
            Comparator.<EnumMap<MnzkiyColumn, Object>, LocalDate>comparing(
                            row -> (LocalDate) row.get(MnzkiyColumn.MNZI_TKYS_YMD),
                            Comparator.nullsLast(Comparator.reverseOrder()));

    /** R003 の開始年月日昇順ソート順。 */
    private static final Comparator<EnumMap<MnzkiyColumn, Object>> R003_ORDER =
            Comparator.<EnumMap<MnzkiyColumn, Object>, LocalDate>comparing(
                            row -> (LocalDate) row.get(MnzkiyColumn.MNZI_TKYS_YMD),
                            Comparator.nullsLast(Comparator.naturalOrder()));

    /** R004 の開始年月日降順ソート順。 */
    private static final Comparator<EnumMap<MnzkiyColumn, Object>> R004_ORDER = R002_ORDER;

    /** インメモリ上で保持する MNZKIY の全行。 */
    private final List<EnumMap<MnzkiyColumn, Object>> rows = new ArrayList<>();

    /** 監査列補完に利用する現在時刻プロバイダ。 */
    private final SystemDateTimeProvider clock;

    /** CIRIUS 共通の監査列自動設定ロジック。 */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** Insert 用の行生成支援。 */
    private final InMemoryInsertSupport<MnzkiyWriteRecord, MnzkiyColumn> insertSupport;

    /** Update 用の更新行生成支援。 */
    private final InMemoryTargetColumnUpdateSupport<MnzkiyWriteRecord, MnzkiyColumn>
            updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public MnzkiyDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えてインスタンスを生成する。
     *
     * @param clock 時刻プロバイダ
     * @param auditColumnAutoSetter 監査列自動設定
     */
    MnzkiyDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(MnzkiyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        MnzkiyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /** テストデータを 1 行追加する。 */
    public MnzkiyDataAccessImpl addRow(MnzkiyRecord row) {
        EnumMap<MnzkiyColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        normalizeComplementedDates(values);
        NEUTRAL_VALUE_RESOLVER.complement(values, MnzkiyWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** 保持中の行をレコードとして返す。 */
    public List<MnzkiyRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    /** 1 行登録する。 */
    @Override
    public int insert(MnzkiyWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(MnzkiyColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    /** 主キーと排他制御順序番号を条件に 1 行削除する。 */
    @Override
    public int deleteBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
            Long svkyNo, LocalDate mnziTkysYmd, Integer recUpdJnjNo) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeMnziTkysYmd = safeYmd(mnziTkysYmd, "mnziTkysYmd");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> matchesPrimaryKey(row, safeSvkyNo, safeMnziTkysYmd),
                MnzkiyColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                MnzkiyDataAccess.CallSpecs.DELETE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                () ->
                        new DataInconsistencyException(
                                "MNZKIY",
                                MnzkiyDataAccess.CallSpecs.DELETE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                                "{}"),
                safeSvkyNo,
                safeMnziTkysYmd);
    }

    /** 主キーで 1 行取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmd(Long svkyNo, LocalDate mnziTkysYmd) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeMnziTkysYmd = safeYmd(mnziTkysYmd, "mnziTkysYmd");
        return rows.stream()
                .filter(row -> matchesPrimaryKey(row, safeSvkyNo, safeMnziTkysYmd))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** サービス契約番号と免税適用終了年月日で 1 行取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTyenYmd(Long svkyNo, LocalDate mnziTyenYmd) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeMnziTyenYmd = safeYmd(mnziTyenYmd, "mnziTyenYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(MnzkiyColumn.SVKY_NO), safeSvkyNo))
                .filter(row -> sameComplementedYmd(row.get(MnzkiyColumn.MNZI_TYEN_YMD), safeMnziTyenYmd))
                .sorted(R002_ORDER)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** 指定開始年月日より後の最初の 1 行を取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmdGreaterThan(
            Long svkyNo, LocalDate mnziTkysYmd) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeMnziTkysYmd = safeYmd(mnziTkysYmd, "mnziTkysYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(MnzkiyColumn.SVKY_NO), safeSvkyNo))
                .filter(
                        row -> {
                            LocalDate rowYmd = (LocalDate) row.get(MnzkiyColumn.MNZI_TKYS_YMD);
                            return complementedYmdValue(rowYmd) > complementedYmdValue(safeMnziTkysYmd);
                        })
                .sorted(R003_ORDER)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** 基準日を含む有効期間のうち最新開始日の 1 行を取得する。 */
    @Override
    public Optional<MnzkiyRecord> findBySvkyNoAndMnziTkysYmdAndMnziTyenYmd(
            Long svkyNo, LocalDate knskKjnYmd) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(MnzkiyColumn.SVKY_NO), safeSvkyNo))
                .filter(row -> isStartOnOrBefore(row, safeKnskKjnYmd))
                .filter(row -> isEndOnOrAfter(row, safeKnskKjnYmd))
                .sorted(R004_ORDER)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** U001 条件で 1 行更新する。 */
    @Override
    public int updateBySvkyNoAndMnziTkysYmdAndRecUpdJnjNo(
            MnzkiyWriteRecord record,
            Set<MnzkiyColumn> targetColumns,
            MnzkiySvkyTkysUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return update(
                record,
                targetColumns,
                updateKey.svkyNo(),
                updateKey.mnziTkysYmd(),
                updateKey.recUpdJnjNo(),
                context,
                MnzkiyDataAccess.CallSpecs.UPDATE_BY_SVKY_NO_AND_MNZI_TKYS_YMD_AND_REC_UPD_JNJ_NO_U001.identifier());
    }

    /**
     * 主キー一致行を更新する。
     *
     * @param record 更新内容
     * @param targetColumns 更新対象列
     * @param svkyNo サービス契約番号
     * @param mnziTkysYmd 免税適用開始年月日
     * @param recUpdJnjNo 排他制御順序番号
     * @param context 監査コンテキスト
     * @param daId DA 識別子
     * @return 更新件数
     */
    private int update(
            MnzkiyWriteRecord record,
            Set<MnzkiyColumn> targetColumns,
            Long svkyNo,
            LocalDate mnziTkysYmd,
            Integer recUpdJnjNo,
            AuditContext context,
            String daId) {
        Long safeSvkyNo = safeLong(svkyNo, "svkyNo");
        LocalDate safeMnziTkysYmd = safeYmd(mnziTkysYmd, "mnziTkysYmd");
        Set<MnzkiyColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPrimaryKey(row, safeSvkyNo, safeMnziTkysYmd),
                row ->
                        updateSupport.prepare(
                                record,
                                canonicalTargetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(MnzkiyColumn.class, values),
                                                context,
                                                clock,
                                                recUpdJnjNo == null
                                                        ? (Integer) row.get(MnzkiyColumn.REC_UPD_JNJ_NO)
                                                        : recUpdJnjNo)),
                updateSupport::apply,
                MnzkiyColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                daId,
                safeSvkyNo,
                safeMnziTkysYmd);
    }

    /**
     * 更新対象列を EnumSet 化して canonical 形式に整える。
     *
     * @param targetColumns 更新対象列
     * @return canonical 化した更新対象列
     */
    private static Set<MnzkiyColumn> canonicalTargetColumns(Set<MnzkiyColumn> targetColumns) {
        EnumSet<MnzkiyColumn> canonicalColumns = EnumSet.noneOf(MnzkiyColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }

    /**
     * 行が主キー条件に一致するか判定する。
     *
     * @param row 判定対象行
     * @param safeSvkyNo サービス契約番号
     * @param safeMnziTkysYmd 免税適用開始年月日
     * @return 主キー一致時 true
     */
    private static boolean matchesPrimaryKey(
            Map<MnzkiyColumn, Object> row, Long safeSvkyNo, LocalDate safeMnziTkysYmd) {
        return Objects.equals(row.get(MnzkiyColumn.SVKY_NO), safeSvkyNo)
                && sameComplementedYmd(row.get(MnzkiyColumn.MNZI_TKYS_YMD), safeMnziTkysYmd);
    }

    /**
     * 行の開始年月日が基準日以前か判定する。
     *
     * @param row 判定対象行
     * @param baseDate 基準日
     * @return 基準日以前の場合 true
     */
    private static boolean isStartOnOrBefore(Map<MnzkiyColumn, Object> row, LocalDate baseDate) {
        LocalDate rowDate = (LocalDate) row.get(MnzkiyColumn.MNZI_TKYS_YMD);
        return complementedYmdValue(rowDate) <= complementedYmdValue(baseDate);
    }

    /**
     * 行の終了年月日が基準日以後か判定する。
     *
     * @param row 判定対象行
     * @param baseDate 基準日
     * @return 基準日以後の場合 true
     */
    private static boolean isEndOnOrAfter(Map<MnzkiyColumn, Object> row, LocalDate baseDate) {
        LocalDate rowDate = (LocalDate) row.get(MnzkiyColumn.MNZI_TYEN_YMD);
        return complementedYmdValue(rowDate) >= complementedYmdValue(baseDate);
    }

    /**
     * 補完用のダミー日付を通常の null 表現へ正規化する。
     *
     * @param values 正規化対象の値マップ
     */
    private static void normalizeComplementedDates(EnumMap<MnzkiyColumn, Object> values) {
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_TKYS_YMD);
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_TYEN_YMD);
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_TYEH_YMD);
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_KYSJ_YMD);
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_TYSJ_YMD);
        normalizeComplementedDate(values, MnzkiyColumn.MNZI_HKSJ_YMD);
    }

    /**
     * 指定列の補完用ダミー日付を null へ置き換える。
     *
     * @param values 変換対象の値マップ
     * @param column 対象列
     */
    private static void normalizeComplementedDate(
            EnumMap<MnzkiyColumn, Object> values, MnzkiyColumn column) {
        Object value = values.get(column);
        if (value instanceof LocalDate localDate && COMPLEMENT_YMD.equals(localDate)) {
            values.put(column, null);
        }
    }

    /**
     * 補完値を考慮して年月日同士の一致を判定する。
     *
     * @param left 行側の年月日
     * @param right 条件側の年月日
     * @return 一致する場合 true
     */
    private static boolean sameComplementedYmd(Object left, LocalDate right) {
        return complementedYmdValue((LocalDate) left) == complementedYmdValue(right);
    }

    /**
     * 年月日を比較用の数値に変換する。
     *
     * @param value 変換対象の年月日
     * @return 比較用数値
     */
    private static int complementedYmdValue(LocalDate value) {
        LocalDate normalized = normalizeComplementedYmd(value);
        if (normalized == null) {
            return 0;
        }
        return Integer.parseInt(normalized.format(BASIC_ISO_DATE));
    }

    /**
     * 補完用のダミー日付を null として扱う。
     *
     * @param value 判定対象の年月日
     * @return 正規化後の年月日
     */
    private static LocalDate normalizeComplementedYmd(LocalDate value) {
        if (value == null || COMPLEMENT_YMD.equals(value)) {
            return null;
        }
        return value;
    }

    /**
     * Long 条件値を null-safe に補完する。
     *
     * @param value 入力値
     * @param paramName パラメータ名
     * @return 補完後の値
     */
    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    /**
     * 年月日条件値を null-safe に補完する。
     *
     * @param value 入力値
     * @param paramName パラメータ名
     * @return 補完後の値
     */
    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return normalizeComplementedYmd(
                WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD));
    }
