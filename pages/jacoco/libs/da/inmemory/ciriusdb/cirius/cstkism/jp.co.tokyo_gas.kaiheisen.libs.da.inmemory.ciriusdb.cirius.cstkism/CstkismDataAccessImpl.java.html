package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.cstkism;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstkism.CstkismDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstkism.CstkismRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-162-R001 の SQL をインメモリで再現する実装。 */
public class CstkismDataAccessImpl extends DataAccessBase implements CstkismDataAccess {

    private static final int JI_CTSH_CD_LENGTH = 5;

    private static final int JI_CNST_CD_LENGTH = 10;

    private static final int KAI_CTSH_CD_LENGTH = 5;

    private final List<CstkismRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public CstkismDataAccessImpl addRow(CstkismRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<CstkismRecord> findByJiCtshCdAndJiCnstCdAndKaiCtshCd(
            String jiCtshCd,
            String jiCnstCd,
            String kaiCtshCd) {
        String safeJiCtshCd =
                WhereParamGuard.fixedLengthNotNull(jiCtshCd, "jiCtshCd", JI_CTSH_CD_LENGTH);
        String safeJiCnstCd =
                WhereParamGuard.fixedLengthNotNull(jiCnstCd, "jiCnstCd", JI_CNST_CD_LENGTH);
        String safeKaiCtshCd =
                WhereParamGuard.fixedLengthNotNull(kaiCtshCd, "kaiCtshCd", KAI_CTSH_CD_LENGTH);

        return rows.stream()
                .filter(row -> normalizeFixed(row.jiCtshCd(), JI_CTSH_CD_LENGTH).equals(safeJiCtshCd))
                .filter(row -> normalizeFixed(row.jiCnstCd(), JI_CNST_CD_LENGTH).equals(safeJiCnstCd))
                .filter(row -> normalizeFixed(row.kaiCtshCd(), KAI_CTSH_CD_LENGTH).equals(safeKaiCtshCd))
                .map(CstkismDataAccessImpl::normalizeRecord)
                .findFirst();
    }

    private static String normalizeFixed(String value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", length);
    }

    private static CstkismRecord normalizeRecord(CstkismRecord row) {
        return new CstkismRecord(
                normalizeResultText(row.jiCtshCd()),
                normalizeResultText(row.jiCnstCd()),
                normalizeResultText(row.kaiCtshCd()),
                normalizeResultText(row.kaiCnstCd()),
                row.regTsmp(),
                normalizeResultText(row.regUsrId()),
                row.updTsmp(),
                normalizeResultText(row.updUsrId()),
                normalizeResultText(row.updKtnKinoId()),
                normalizeResultInteger(row.recUpdJnjNo()));
    }

    private static String normalizeResultText(String value) {
        return FixedLengthFormatter.suppressSpace(value);
    }

    private static Integer normalizeResultInteger(Integer value) {
        if (value == null || value == 0) {
            return null;
        }
        return value;
    }
}
