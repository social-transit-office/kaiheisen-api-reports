import java.util.Objects;
import java.util.Optional;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/** DA-093 の SQL 振る舞いを模した HRIKMI インメモリ実装。 */
public class HrikmiDataAccessImpl extends DataAccessBase implements HrikmiDataAccess {

    private static final ReflectiveInMemoryRowMapper<HrikmiColumn, HrikmiRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(HrikmiColumn.class, HrikmiRecord.class);

    private static final AutoNeutralValueResolver<HrikmiColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(HrikmiColumn.class, HrikmiRecord.class);

    private final List<EnumMap<HrikmiColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<HrikmiWriteRecord, HrikmiColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<HrikmiWriteRecord, HrikmiColumn>
            updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public HrikmiDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    /** テスト用に依存を差し替えてインスタンスを生成する。 */
    HrikmiDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(HrikmiWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        HrikmiWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /** テストデータを 1 行追加する。 */
    public HrikmiDataAccessImpl addRow(HrikmiRecord row) {
        EnumMap<HrikmiColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, HrikmiWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** 保持中の行をレコードとして返す。 */
    public List<HrikmiRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(HrikmiWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(HrikmiColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            Long sihKiyNo, Integer sihKiySkno, Integer recUpdJnjNo) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(row.get(HrikmiColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(row.get(HrikmiColumn.SIH_KIY_SKNO), safeSihKiySkno),
                HrikmiColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                HrikmiDataAccess.Identifiers.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO,
                () -> new DataInconsistencyException("HRIKMI", HrikmiDataAccess.Identifiers.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO, "{}"),
                safeSihKiyNo,
                safeSihKiySkno);
    }

    @Override
    public Optional<HrikmiRecord> findBySihKiyNoAndSihKiySkno(Long sihKiyNo, Integer sihKiySkno) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        return rows.stream()
                .filter(r -> Objects.equals(r.get(HrikmiColumn.SIH_KIY_NO), safeSihKiyNo))
                .filter(r -> Objects.equals(r.get(HrikmiColumn.SIH_KIY_SKNO), safeSihKiySkno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            HrikmiWriteRecord record,
            Set<HrikmiColumn> targetColumns,
            HrikmiSihKiyUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return update(
                rows,
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                HrikmiDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO);
    }

    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            HrikmiWriteRecord record,
            Set<HrikmiColumn> targetColumns,
            HrikmiSihKiyJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return update(
                rows,
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                HrikmiDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO);
    }

    private int update(
            List<EnumMap<HrikmiColumn, Object>> targetRows,
            HrikmiWriteRecord record,
            Set<HrikmiColumn> targetColumns,
            Long sihKiyNo,
            Integer sihKiySkno,
            Integer recUpdJnjNo,
            AuditContext context,
            String daId) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInt(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticUpdateSupport.update(
                targetRows,
                row -> Objects.equals(row.get(HrikmiColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(row.get(HrikmiColumn.SIH_KIY_SKNO), safeSihKiySkno),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(HrikmiColumn.class, values),
                                                context,
                                                clock,
                                                (Integer) row.get(HrikmiColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                HrikmiColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                daId,
                safeSihKiyNo,
                safeSihKiySkno);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }
}
