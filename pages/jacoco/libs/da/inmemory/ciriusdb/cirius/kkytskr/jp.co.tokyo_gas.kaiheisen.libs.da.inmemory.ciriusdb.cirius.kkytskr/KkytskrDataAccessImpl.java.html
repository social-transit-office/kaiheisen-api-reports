package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.kkytskr;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kkytskr.write.KkytskrHesnUpdateKey;

/** DA-055（C001/D001/R001-R004/U001-U002）の SQL 振る舞いを模した KKYTSKR インメモリ実装。 */
public class KkytskrDataAccessImpl extends DataAccessBase implements KkytskrDataAccess {

    private static final DateTimeFormatter YMD_FORMAT = DateTimeFormatter.BASIC_ISO_DATE;

    private static final ReflectiveInMemoryRowMapper<KkytskrColumn, KkytskrRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KkytskrColumn.class, KkytskrRecord.class);

    private static final AutoNeutralValueResolver<KkytskrColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KkytskrColumn.class, KkytskrRecord.class);

    private final List<EnumMap<KkytskrColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KkytskrWriteRecord, KkytskrColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KkytskrWriteRecord, KkytskrColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KkytskrDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KkytskrDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KkytskrDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KkytskrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KkytskrWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public KkytskrDataAccessImpl addRow(KkytskrRecord record) {
        EnumMap<KkytskrColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KkytskrWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KkytskrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KkytskrColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByGasMtstNoAndHesnYmd(Long gasMtstNo, LocalDate hesnYmd, Integer recUpdJnjNo) {
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        Objects.equals(row.get(KkytskrColumn.GAS_MTST_NO), gasMtstNo)
                                && Objects.equals(row.get(KkytskrColumn.HESN_YMD), hesnYmd),
                KkytskrColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KkytskrDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_HESN_YMD,
                () -> new DataInconsistencyException("KKYTSKR", KkytskrDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_HESN_YMD, "{}"),
                gasMtstNo,
                hesnYmd);
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNo(Long gasMtstNo) {
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KkytskrColumn.GAS_MTST_NO), gasMtstNo))
                .sorted(
                        Comparator.<EnumMap<KkytskrColumn, Object>, LocalDate>comparing(
                                r -> (LocalDate) r.get(KkytskrColumn.HESN_YMD),
                                Comparator.nullsLast(Comparator.reverseOrder())))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndHesnYmd(Long gasMtstNo, LocalDate hesnYmd) {
        return rows.stream()
                .filter(
                        r ->
                                Objects.equals(r.get(KkytskrColumn.GAS_MTST_NO), gasMtstNo)
                                        && Objects.equals(r.get(KkytskrColumn.HESN_YMD), hesnYmd))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndKasYmdAndKkysSisnJtkb(
            Long gasMtstNo, LocalDate kasYmd, String kkysSisnJtkb) {
        int safeKasYmd = toYmdIntOrZero(kasYmd);
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KkytskrColumn.GAS_MTST_NO), gasMtstNo))
                .filter(r -> toYmdIntOrZero((LocalDate) r.get(KkytskrColumn.KAS_YMD)) == safeKasYmd)
                .filter(r -> Objects.equals(r.get(KkytskrColumn.KKYS_SISN_JTKB), kkysSisnJtkb))
                .sorted(
                        Comparator.<EnumMap<KkytskrColumn, Object>, LocalDate>comparing(
                                r -> (LocalDate) r.get(KkytskrColumn.HESN_YMD),
                                Comparator.nullsLast(Comparator.reverseOrder())))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KkytskrRecord> findByGasMtstNoAndHesnYmdAndKkysSisnJtkb(
            Long gasMtstNo, LocalDate hesnYmd, String kkysSisnJtkb) {
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KkytskrColumn.GAS_MTST_NO), gasMtstNo))
                .filter(r -> Objects.equals(r.get(KkytskrColumn.HESN_YMD), hesnYmd))
                .filter(r -> Objects.equals(r.get(KkytskrColumn.KKYS_SISN_JTKB), kkysSisnJtkb))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByGasMtstNoAndHesnYmd(
            KkytskrWriteRecord record,
            Set<KkytskrColumn> targetColumns,
            KkytskrHesnUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        Objects.equals(row.get(KkytskrColumn.GAS_MTST_NO), updateKey.gasMtstNo())
                                && Objects.equals(row.get(KkytskrColumn.HESN_YMD), updateKey.hesnYmd()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KkytskrColumn.class, values),
                                                context,
                                                clock,
                                                (Integer) row.get(KkytskrColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                KkytskrColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KkytskrDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD,
                updateKey.gasMtstNo(),
                updateKey.hesnYmd());
    }

    @Override
    public int updateByGasMtstNoAndHesnYmdAndRecUpdJnjNo(
            KkytskrWriteRecord record,
            Set<KkytskrColumn> targetColumns,
            KkytskrHesnJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        Objects.equals(row.get(KkytskrColumn.GAS_MTST_NO), updateKey.gasMtstNo())
                                && Objects.equals(row.get(KkytskrColumn.HESN_YMD), updateKey.hesnYmd()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KkytskrColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(KkytskrColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KkytskrColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KkytskrDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_HESN_YMD_AND_REC_UPD_JNJ_NO,
                updateKey.gasMtstNo(),
                updateKey.hesnYmd());
    }

    private static int toYmdIntOrZero(LocalDate value) {
        if (value == null) {
            return 0;
        }
        return Integer.parseInt(value.format(YMD_FORMAT));
    }
}
