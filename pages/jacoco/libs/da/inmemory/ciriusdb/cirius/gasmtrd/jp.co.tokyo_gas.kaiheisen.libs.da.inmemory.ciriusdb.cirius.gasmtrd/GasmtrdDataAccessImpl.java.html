package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.gasmtrd;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrd.GasmtrdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrd.GasmtrdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/**
 * GASMTRD（ガスメーター＜導管＞）のインメモリ実装。
 */
public class GasmtrdDataAccessImpl extends DataAccessBase implements GasmtrdDataAccess {

    private static final int GMB_LENGTH = 15;

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GasmtrdDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    /** R001: TTK_GMSB_NO = :ttkGmsbNo ORDER BY GMB_TSNO ASC FETCH FIRST 1 */
    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNo(Long ttkGmsbNo) {
        long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == safeTtkGmsbNo)
                .sorted(Comparator.comparingInt(InternalRow::gmbTsno))
                .map(GasmtrdDataAccessImpl::toRecord)
                .findFirst();
    }

    /** R002: TTK_GMSB_NO = :ttkGmsbNo AND GMT_THKG_YMD = :gmtThkgYmd */
    @Override
    public List<GasmtrdRecord> findAllByTtkGmsbNoAndGmtThkgYmd(Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        int safeGmtThkgYmd = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == safeTtkGmsbNo)
            .filter(row -> row.gmtThkgYmd() == safeGmtThkgYmd)
                .map(GasmtrdDataAccessImpl::toRecord)
                .toList();
    }

    /** R003: TTK_GMSB_NO = :ttkGmsbNo AND GMT_THKG_YMD = :gmtThkgYmd ORDER BY GMB_TSNO DESC FETCH FIRST 1 */
    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNoAndGmtThkgYmd(Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        int gmtThkgYmdValue = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == safeTtkGmsbNo)
                .filter(row -> row.gmtThkgYmd() == gmtThkgYmdValue)
                .sorted(Comparator.comparingInt(InternalRow::gmbTsno).reversed())
                .map(GasmtrdDataAccessImpl::toRecord)
                .findFirst();
    }

    /** R004: TTK_GMSB_NO = :ttkGmsbNo AND GMTK_YMD {@literal <=} :gmtkYmd AND GMT_THKG_YMD = :gmtThkgYmd */
    @Override
    public List<GasmtrdRecord> findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(
            Long ttkGmsbNo, LocalDate gmtkYmd, LocalDate gmtThkgYmd) {
        long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        int gmtkYmdValue = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        int gmtThkgYmdValue = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == safeTtkGmsbNo)
                .filter(row -> row.gmtkYmd() <= gmtkYmdValue)
                .filter(row -> row.gmtThkgYmd() == gmtThkgYmdValue)
                .map(GasmtrdDataAccessImpl::toRecord)
                .toList();
    }

    /** R005: GMB = :gmb AND GMTK_YMD {@literal <=} :gmtkYmd ORDER BY GMB_TSNO DESC FETCH FIRST 1 */
    @Override
    public Optional<GasmtrdRecord> findByGmbAndGmtkYmd(String gmb, LocalDate gmtkYmd) {
        String safeGmb = safeGmb(gmb, "gmb");
        int gmtkYmdValue = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        return rows.stream()
                .filter(row -> matchesGmb(row, safeGmb))
                .filter(row -> row.gmtkYmd() <= gmtkYmdValue)
                .sorted(Comparator.comparingInt(InternalRow::gmbTsno).reversed())
                .map(GasmtrdDataAccessImpl::toRecord)
                .findFirst();
    }

    /** R006: GMB = :gmb AND GMT_THKG_YMD = :gmtThkgYmd ORDER BY GMB_TSNO DESC FETCH FIRST 1 */
    @Override
    public Optional<GasmtrdRecord> findByGmbAndGmtThkgYmd(String gmb, LocalDate gmtThkgYmd) {
        String safeGmb = safeGmb(gmb, "gmb");
        int gmtThkgYmdValue = safeGmtThkgYmd(gmtThkgYmd, "gmtThkgYmd");
        return rows.stream()
                .filter(row -> matchesGmb(row, safeGmb))
                .filter(row -> row.gmtThkgYmd() == gmtThkgYmdValue)
                .sorted(Comparator.comparingInt(InternalRow::gmbTsno).reversed())
                .map(GasmtrdDataAccessImpl::toRecord)
                .findFirst();
    }

    /** R007: TTK_GMSB_NO = :ttkGmsbNo AND GMTK_YMD {@literal <=} :gmtkYmd ORDER BY GMTK_YMD DESC, GMT_THKG_YMD ASC FETCH FIRST 1 */
    @Override
    public Optional<GasmtrdRecord> findByTtkGmsbNoAndGmtkYmd(Long ttkGmsbNo, LocalDate gmtkYmd) {
        long safeTtkGmsbNo = safeTtkGmsbNo(ttkGmsbNo, "ttkGmsbNo");
        int gmtkYmdValue = safeGmtkYmd(gmtkYmd, "gmtkYmd");
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == safeTtkGmsbNo)
                .filter(row -> row.gmtkYmd() <= gmtkYmdValue)
                .sorted(Comparator.comparingInt(InternalRow::gmtkYmd).reversed()
                        .thenComparingInt(InternalRow::gmtThkgYmd))
                .map(GasmtrdDataAccessImpl::toRecord)
                .findFirst();
    }

    private static GasmtrdRecord toRecord(InternalRow row) {
        return new GasmtrdRecord(
                row.gmb(),
                row.gmbTsno(),
                row.ttkGmsbNo(),
                toLocalDate(row.gmtkYmd()),
                toLocalDate(row.gmtThkgYmd()),
                "0",
                "00",
                "0",
                "",
                "",
                "",
                toYearMonth(row.ktyYm()),
                row.gmkShbt(),
                row.gmtGusu(),
                BigDecimal.ZERO,
                0,
                row.sjsuMincSu(),
                toLong(row.sjskHticSu()),
                "",
                "",
                "",
                "",
                "",
                "",
                toLocalDate(0),
                "",
                "",
                "",
                "",
                "",
                "",
                "",
                LocalDateTime.of(1970, 1, 1, 0, 0),
                "",
                LocalDateTime.of(1970, 1, 1, 0, 0),
                "",
                "",
                0
        );
    }

    /**
     * GASMTRD のテストデータ行。
     */
    public static record InternalRow(
            String gmb,
            int gmbTsno,
            long ttkGmsbNo,
            String gmkShbt,
            BigDecimal gmtGusu,
            int ktyYm,
            BigDecimal sjsuMincSu,
            BigDecimal sjskHticSu,
            int gmtkYmd,
            int gmtThkgYmd
    ) {
        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(gmb, "gmb");
            Objects.requireNonNull(gmkShbt, "gmkShbt");
            Objects.requireNonNull(gmtGusu, "gmtGusu");
            Objects.requireNonNull(sjsuMincSu, "sjsuMincSu");
            Objects.requireNonNull(sjskHticSu, "sjskHticSu");
        }
    }

    private static int toYmdInt(LocalDate date) {
        return date.getYear() * 10000 + date.getMonthValue() * 100 + date.getDayOfMonth();
    }

    private static long safeTtkGmsbNo(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static int safeGmtThkgYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value == null ? null : toYmdInt(value), paramName, 0);
    }

    private static int safeGmtkYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value == null ? null : toYmdInt(value), paramName, 0);
    }

    private static String safeGmb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GMB_LENGTH);
    }

    private static boolean matchesGmb(InternalRow row, String safeGmb) {
        return safeGmb.equals(safeGmb(row.gmb(), "row.gmb"));
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }

    private static YearMonth toYearMonth(int ym) {
        if (ym <= 0) {
            return null;
        }
        return YearMonth.of(ym / 100, ym % 100);
    }

    private static Long toLong(BigDecimal value) {
        if (value == null) {
            return null;
        }
        return value.longValue();
    }
}
