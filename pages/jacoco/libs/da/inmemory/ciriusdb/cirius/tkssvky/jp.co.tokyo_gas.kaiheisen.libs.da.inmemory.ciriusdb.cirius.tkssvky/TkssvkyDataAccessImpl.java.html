/** TKSSVKY の SQL 挙動を模したインメモリ実装。 */
public class TkssvkyDataAccessImpl extends DataAccessBase implements TkssvkyDataAccess {

    private static final String INVALID_STATUS = "00";

    private static final LocalDate COMPLEMENT_YMD = LocalDate.of(1, 1, 1);

    private static final ReflectiveInMemoryRowMapper<TkssvkyColumn, TkssvkyRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(TkssvkyColumn.class, TkssvkyRecord.class);

    private static final AutoNeutralValueResolver<TkssvkyColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(TkssvkyColumn.class, TkssvkyRecord.class);

    private final List<EnumMap<TkssvkyColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<TkssvkyWriteRecord, TkssvkyColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<TkssvkyWriteRecord, TkssvkyColumn>
            updateSupport;

    /** デフォルト設定で初期化する。 */
    public TkssvkyDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    TkssvkyDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    TkssvkyDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(TkssvkyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        TkssvkyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public TkssvkyDataAccessImpl addRow(InternalRow row) {
        Objects.requireNonNull(row, "row");
        return addRecord(
                new TkssvkyRecord(
                        row.tksuSvkyNo(),
                        row.skno(),
                        row.status(),
                        row.startReason(),
                        row.endReason(),
                        toLocalDate(row.tkysYmd()),
                        toLocalDate(row.tyenYmd()),
                        toLocalDate(row.hesnYtiYmd()),
                        row.contractType(),
                        row.supplierCode(),
                        toBooleanFlag(row.estimationFlag()),
                        row.customerId(),
                        row.regTsmp(),
                        row.regUsrId(),
                        row.updTsmp(),
                        row.updUsrId(),
                        row.updKtnKinoId(),
                        row.recUpdJnjNo()));
    }

    private TkssvkyDataAccessImpl addRecord(TkssvkyRecord row) {
        TkssvkyRecord safeRow = Objects.requireNonNull(row, "row");
        EnumMap<TkssvkyColumn, Object> values = ROW_MAPPER.toRawValueMap(safeRow);
        NEUTRAL_VALUE_RESOLVER.complement(values, TkssvkyWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * 保持している行を read record で返す。
     *
     * @return 行一覧
     */
    public List<TkssvkyRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(TkssvkyWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                TkssvkyColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<TkssvkyRecord> findAtBaseDateExcludeInvalid(
            Long tksuSvkyNo, LocalDate baseDate) {
        Long safeTksuSvkyNo = safeLong(tksuSvkyNo, "tksuSvkyNo");
        LocalDate safeBaseDate = safeYmd(baseDate, "baseDate");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(TkssvkyColumn.TKSU_SVKY_NO), safeTksuSvkyNo))
                .filter(row -> notAfter(row.get(TkssvkyColumn.TYEN_YMD), safeBaseDate))
                .filter(row -> !INVALID_STATUS.equals(row.get(TkssvkyColumn.TKSU_SVKY_JTKB)))
                .sorted(descComparator(TkssvkyColumn.TYEN_YMD))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private static Comparator<EnumMap<TkssvkyColumn, Object>> descComparator(TkssvkyColumn column) {
        return (left, right) -> compareDesc(left.get(column), right.get(column));
    }

    @SuppressWarnings({"rawtypes", "unchecked"})
    private static int compareDesc(Object left, Object right) {
        if (left == right) {
            return 0;
        }
        if (left == null) {
            return 1;
        }
        if (right == null) {
            return -1;
        }
        return ((Comparable) right).compareTo(left);
    }

    private static boolean notAfter(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && !localDate.isAfter(baseDate);
    }

    private static boolean notBefore(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && !localDate.isBefore(baseDate);
    }

    private static boolean isBefore(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && localDate.isBefore(baseDate);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
