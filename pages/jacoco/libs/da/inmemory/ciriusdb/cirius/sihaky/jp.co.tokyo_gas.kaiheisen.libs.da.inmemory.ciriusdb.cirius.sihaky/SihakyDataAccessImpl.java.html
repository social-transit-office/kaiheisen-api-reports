/**
 * 支払契約（SIHAKY）のインメモリ実装。
 */
public class SihakyDataAccessImpl extends DataAccessBase implements SihakyDataAccess {

    private static final int SIH_KIY_JTKB_LENGTH = 2;

    private static final ReflectiveInMemoryRowMapper<SihakyColumn, SihakyRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SihakyColumn.class, SihakyRecord.class);

    private static final AutoNeutralValueResolver<SihakyColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SihakyColumn.class, SihakyRecord.class);

    private final List<EnumMap<SihakyColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SihakyWriteRecord, SihakyColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<SihakyWriteRecord, SihakyColumn>
            updateSupport;

    /** デフォルト依存で初期化する。 */
    public SihakyDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    SihakyDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(SihakyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        SihakyWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param row 追加行
     * @return このインスタンス
     */
    public SihakyDataAccessImpl addRow(SihakyRecord row) {
        EnumMap<SihakyColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, SihakyWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * 保持中の行を返す。
     *
     * @return 保持行
     */
    public List<SihakyRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(SihakyWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(SihakyColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndSihKiyJtkb(Long sihKiyNo, String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(orderByTkysYmdDescAndSknoDesc())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndNotSihKiyJtkb(Long sihKiyNo, String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> !hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                (Map<SihakyColumn, Object> row) -> getIntegerValue(row, SihakyColumn.SKNO),
                                Comparator.reverseOrder()))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public List<SihakyRecord> findAllBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
            Long sihKiyNo,
            LocalDate tkysYmd,
            String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        LocalDate safeTkysYmd = ymdOrMin(tkysYmd, "tkysYmd");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> isNotBefore(row, SihakyColumn.TKYS_YMD, safeTkysYmd))
                .filter(row -> !hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                (Map<SihakyColumn, Object> row) -> getLocalDateValue(row, SihakyColumn.TKYS_YMD),
                                Comparator.reverseOrder()))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndTkysYmdAndNotSihKiyJtkb(
            Long sihKiyNo,
            LocalDate tkysYmd,
            String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        LocalDate safeTkysYmd = ymdOrMin(tkysYmd, "tkysYmd");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> isNotAfter(row, SihakyColumn.TKYS_YMD, safeTkysYmd))
                .filter(row -> !hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                (Map<SihakyColumn, Object> row) -> getLocalDateValue(row, SihakyColumn.TKYS_YMD),
                                Comparator.reverseOrder()))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDesc(
            Long sihKiyNo,
            LocalDate knskKjnYmd,
            String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        LocalDate safeKnskKjnYmd = ymdOrMin(knskKjnYmd, "knskKjnYmd");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> isNotAfter(row, SihakyColumn.TKYS_YMD, safeKnskKjnYmd))
                .filter(row -> isNotBefore(row, SihakyColumn.TYEN_YMD, safeKnskKjnYmd))
                .filter(row -> !hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(
                        Comparator.comparing(
                                (Map<SihakyColumn, Object> row) -> getLocalDateValue(row, SihakyColumn.TKYS_YMD),
                                Comparator.reverseOrder()))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDescAndSknoDesc(
            Long sihKiyNo,
            LocalDate knskKjnYmd,
            String sihKiyJtkb) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        LocalDate safeKnskKjnYmd = ymdOrMin(knskKjnYmd, "knskKjnYmd");
        String safeSihKiyJtkb = fixedLengthOrSpaces(sihKiyJtkb, "sihKiyJtkb", SIH_KIY_JTKB_LENGTH);
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .filter(row -> isNotAfter(row, SihakyColumn.TKYS_YMD, safeKnskKjnYmd))
                .filter(row -> isNotBefore(row, SihakyColumn.TYEN_YMD, safeKnskKjnYmd))
                .filter(row -> !hasValue(row, SihakyColumn.SIH_KIY_JTKB, safeSihKiyJtkb))
                .sorted(orderByTkysYmdDescAndSknoDesc())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SihakyRecord> findBySihKiyNoOrderByTkysYmdDescAndSknoDesc(Long sihKiyNo) {
        Long safeSihKiyNo = longOrZero(sihKiyNo, "sihKiyNo");
        return rows.stream()
                .filter(row -> hasValue(row, SihakyColumn.SIH_KIY_NO, safeSihKiyNo))
                .sorted(orderByTkysYmdDescAndSknoDesc())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<SihakyRecord> findByPK(Long sihKiyNo, Integer skno) {
        Long safeSihKiyNo = WhereParamGuard.complement(sihKiyNo, "sihKiyNo", 0L);
        Integer safeSkno = WhereParamGuard.complement(skno, "skno", 0);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(SihakyColumn.SIH_KIY_NO), safeSihKiyNo))
                .filter(row -> Objects.equals(row.get(SihakyColumn.SKNO), safeSkno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSknoAndRecUpdJnjNo(
            SihakyWriteRecord record,
            Set<SihakyColumn> targetColumns,
            SihakySknoJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, updateKey.sihKiyNo(), updateKey.skno()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(SihakyColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() != null
                                                        ? updateKey.recUpdJnjNo()
                                                        : currentRecUpdJnjNo(row))),
                updateSupport::apply,
                SihakyColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                SihakyDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO,
                updateKey.sihKiyNo(),
                updateKey.skno());
    }

    @Override
    public int updateBySihKiyNoAndSkno(
            SihakyWriteRecord record,
            Set<SihakyColumn> targetColumns,
            SihakySknoUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPk(row, updateKey.sihKiyNo(), updateKey.skno()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(SihakyColumn.class, values),
                                                context,
                                                clock,
                                                currentRecUpdJnjNo(row))),
                updateSupport::apply,
                SihakyColumn.REC_UPD_JNJ_NO,
                null,
                SihakyDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_SKNO,
                updateKey.sihKiyNo(),
                updateKey.skno());
    }

    private static boolean matchesPk(Map<SihakyColumn, Object> row, Long sihKiyNo, Integer skno) {
        return Objects.equals(row.get(SihakyColumn.SIH_KIY_NO), sihKiyNo)
                && Objects.equals(row.get(SihakyColumn.SKNO), skno);
    }

    private static Comparator<Map<SihakyColumn, Object>> orderByTkysYmdDescAndSknoDesc() {
        return Comparator.comparing(
                        (Map<SihakyColumn, Object> row) -> getLocalDateValue(row, SihakyColumn.TKYS_YMD),
                        Comparator.reverseOrder())
                .thenComparing(
                        row -> getIntegerValue(row, SihakyColumn.SKNO),
                        Comparator.reverseOrder());
    }

    private static boolean hasValue(Map<SihakyColumn, Object> row, SihakyColumn column, Object value) {
        return Objects.equals(row.get(column), value);
    }

    private static boolean isNotAfter(Map<SihakyColumn, Object> row, SihakyColumn column, LocalDate value) {
        LocalDate actual = getLocalDateValue(row, column);
        return actual != null && !actual.isAfter(value);
    }

    private static boolean isNotBefore(Map<SihakyColumn, Object> row, SihakyColumn column, LocalDate value) {
        LocalDate actual = getLocalDateValue(row, column);
        return actual != null && !actual.isBefore(value);
    }

    private static LocalDate getLocalDateValue(Map<SihakyColumn, Object> row, SihakyColumn column) {
        return (LocalDate) row.get(column);
    }

    private static Integer getIntegerValue(Map<SihakyColumn, Object> row, SihakyColumn column) {
        Object value = row.get(column);
        if (value == null) {
            return null;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.valueOf(value.toString());
    }

    private static Integer currentRecUpdJnjNo(Map<SihakyColumn, Object> row) {
        Integer currentRecUpdJnjNo = getIntegerValue(row, SihakyColumn.REC_UPD_JNJ_NO);
        return Objects.requireNonNullElse(currentRecUpdJnjNo, 0);
    }

    private static Long longOrZero(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static LocalDate ymdOrMin(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, LocalDate.MIN);
    }

    private static String fixedLengthOrSpaces(String value, String paramName, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }
