package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.kyotelk;

/** DA-062（KYOTELK）のインメモリ実装。 */
public class KyotelkDataAccessImpl extends DataAccessBase implements KyotelkDataAccess {

    private static final int KYO_TEL_NO_LENGTH = 13;

    private static final ReflectiveInMemoryRowMapper<KyotelkColumn, KyotelkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KyotelkColumn.class, KyotelkRecord.class);

    private static final AutoNeutralValueResolver<KyotelkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KyotelkColumn.class, KyotelkRecord.class);

    private static final Comparator<Map<KyotelkColumn, Object>> R001_ORDER =
            Comparator.<Map<KyotelkColumn, Object>, String>comparing(
                            row -> (String) row.get(KyotelkColumn.TEL_KNZ_SHBT),
                            Comparator.nullsLast(Comparator.reverseOrder()))
                    .thenComparing(
                            row -> (String) row.get(KyotelkColumn.TEL_RRSK_SHBT),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row -> (String) row.get(KyotelkColumn.TEL_SHBT),
                            Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            row -> (java.time.LocalDate) row.get(KyotelkColumn.TEL_SHTK_YMD),
                            Comparator.nullsLast(Comparator.reverseOrder()))
                    .thenComparing(
                            row -> (String) row.get(KyotelkColumn.KYO_TEL_NO),
                            Comparator.nullsLast(Comparator.naturalOrder()));

    private final List<EnumMap<KyotelkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KyotelkWriteRecord, KyotelkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KyotelkWriteRecord, KyotelkColumn>
            updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public KyotelkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    /**
         * テスト用に時刻プロバイダを差し替えてインスタンスを生成する。
         *
         * @param clock 時刻プロバイダ
         */
    KyotelkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    /**
         * テスト用に依存を差し替えてインスタンスを生成する。
         *
         * @param clock 時刻プロバイダ
         * @param auditColumnAutoSetter 監査列自動設定
         */
    KyotelkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KyotelkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KyotelkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * 初期データとして 1 行追加する。
     *
     * @param record 追加対象レコード
     * @return 自身
     */
    public KyotelkDataAccessImpl addRow(KyotelkRecord record) {
        EnumMap<KyotelkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KyotelkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KyotelkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KyotelkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            Long kyoTrkNo, String kyoTelNo, Integer recUpdJnjNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> matchesPrimaryKey(row, safeKyoTrkNo, safeKyoTelNo),
                KyotelkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KyotelkDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                () ->
                        new DataInconsistencyException(
                                "KYOTELK",
                                KyotelkDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                                "{}"),
                safeKyoTrkNo,
                safeKyoTelNo);
    }

    @Override
    public List<KyotelkRecord> findAllByKyoTrkNo(Long kyoTrkNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        return rows.stream()
                .filter(r -> matchesKyoTrkNo(r, safeKyoTrkNo))
                .sorted(R001_ORDER)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<KyotelkRecord> findByKyoTrkNo(Long kyoTrkNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        return rows.stream()
                .filter(r -> matchesKyoTrkNo(r, safeKyoTrkNo))
                .sorted(R001_ORDER)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public long countByKyoTrkNo(Long kyoTrkNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        return rows.stream()
                .filter(r -> matchesKyoTrkNo(r, safeKyoTrkNo))
                .count();
    }

    @Override
    public Optional<KyotelkRecord> findByKyoTrkNoAndKyoTelNo(Long kyoTrkNo, String kyoTelNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return rows.stream()
                .filter(r -> matchesPrimaryKey(r, safeKyoTrkNo, safeKyoTelNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            KyotelkWriteRecord record,
            Set<KyotelkColumn> targetColumns,
            KyotelkUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Long safeKyoTrkNo = safeLong(updateKey.kyoTrkNo(), "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(updateKey.kyoTelNo());
        Set<KyotelkColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPrimaryKey(row, safeKyoTrkNo, safeKyoTelNo),
                row ->
                        updateSupport.prepare(
                                record,
                                canonicalTargetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KyotelkColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(KyotelkColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KyotelkColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KyotelkDataAccess.Identifiers.UPDATE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                safeKyoTrkNo,
                safeKyoTelNo);
    }

    private static Set<KyotelkColumn> canonicalTargetColumns(Set<KyotelkColumn> targetColumns) {
        EnumSet<KyotelkColumn> canonicalColumns = EnumSet.noneOf(KyotelkColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }

    private static boolean matchesKyoTrkNo(Map<KyotelkColumn, Object> row, Long safeKyoTrkNo) {
        return Objects.equals(row.get(KyotelkColumn.KYO_TRK_NO), safeKyoTrkNo);
    }

    private static boolean matchesPrimaryKey(
            Map<KyotelkColumn, Object> row, Long safeKyoTrkNo, String safeKyoTelNo) {
        return matchesKyoTrkNo(row, safeKyoTrkNo) && rowKyoTelNo(row).equals(safeKyoTelNo);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", KYO_TEL_NO_LENGTH);
    }

    private static String rowKyoTelNo(Map<KyotelkColumn, Object> row) {
        return WhereParamGuard.fixedLengthNotNull(
                FixedLengthFormatter.padSpace(
                        (String) row.get(KyotelkColumn.KYO_TEL_NO), KYO_TEL_NO_LENGTH),
                "kyoTelNo",
                KYO_TEL_NO_LENGTH);
    }
}
