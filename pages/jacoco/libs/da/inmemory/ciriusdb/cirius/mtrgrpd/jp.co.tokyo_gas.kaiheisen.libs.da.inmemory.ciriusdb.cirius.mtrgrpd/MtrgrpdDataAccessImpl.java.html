package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.mtrgrpd;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtrgrpd.MtrgrpdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.mtrgrpd.MtrgrpdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-113-R001 の SQL 振る舞いを模した MTRGRPD インメモリ実装。 */
public class MtrgrpdDataAccessImpl extends DataAccessBase implements MtrgrpdDataAccess {

    private final List<MtrgrpdRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public MtrgrpdDataAccessImpl addRow(MtrgrpdRecord row) {
        rows.add(java.util.Objects.requireNonNull(row, "row"));
        return this;
    }

    /**
     * 保持行を返す。
     *
     * @return 保持行
     */
    public List<MtrgrpdRecord> rows() {
        return List.copyOf(rows);
    }

    @Override
    public Optional<MtrgrpdRecord> findByGasMtstNoAndTkysYmdAndTyenYmd(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        long safeGasMtstNo = longOrZero(gasMtstNo, "gasMtstNo");
        int safeKnskKjnYmd = ymdIntOrZero(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(row -> longOrZero(row.gasMtstNo(), "row.gasMtstNo") == safeGasMtstNo)
                .filter(row -> ymdIntOrZero(row.tkysYmd(), "row.tkysYmd") <= safeKnskKjnYmd)
                .filter(row -> ymdIntOrZero(row.tyenYmd(), "row.tyenYmd") >= safeKnskKjnYmd)
                .sorted(Comparator.comparingInt((MtrgrpdRecord row) -> ymdIntOrZero(row.tkysYmd(), "row.tkysYmd"))
                        .reversed()
                        .thenComparingInt(row -> intOrZero(row.hrtGrpNo(), "row.hrtGrpNo")))
                .findFirst();
    }

    @Override
    public Optional<MtrgrpdRecord> findByPK(Integer hrtGrpNo, Long gasMtstNo, LocalDate tkysYmd) {
        int safeHrtGrpNo = intOrZero(hrtGrpNo, "hrtGrpNo");
        long safeGasMtstNo = longOrZero(gasMtstNo, "gasMtstNo");
        int safeTkysYmd = ymdIntOrZero(tkysYmd, "tkysYmd");
        return rows.stream()
                .filter(row -> intOrZero(row.hrtGrpNo(), "row.hrtGrpNo") == safeHrtGrpNo)
                .filter(row -> longOrZero(row.gasMtstNo(), "row.gasMtstNo") == safeGasMtstNo)
                .filter(row -> ymdIntOrZero(row.tkysYmd(), "row.tkysYmd") == safeTkysYmd)
                .findFirst();
    }

    private static int intOrZero(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static long longOrZero(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static int ymdIntOrZero(LocalDate value, String paramName) {
        LocalDate safeValue = WhereParamGuard.complement(value, paramName, LocalDate.MIN);
        if (LocalDate.MIN.equals(safeValue)) {
            return 0;
        }
        return safeValue.getYear() * 10000 + safeValue.getMonthValue() * 100 + safeValue.getDayOfMonth();
    }
}
