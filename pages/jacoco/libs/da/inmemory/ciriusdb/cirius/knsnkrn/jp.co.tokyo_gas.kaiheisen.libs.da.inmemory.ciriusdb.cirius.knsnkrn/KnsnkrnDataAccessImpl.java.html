package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.knsnkrn;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsnkrn.KnsnkrnRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/**
 * DA-063-R001 の SQL をインメモリで再現する実装。
 */
public class KnsnkrnDataAccessImpl extends DataAccessBase implements KnsnkrnDataAccess {

    private final List<KnsnkrnRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public KnsnkrnDataAccessImpl addRow(KnsnkrnRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    /**
     * 保持行を参照する。
     *
     * @return 保持行
     */
    public List<KnsnkrnRecord> rows() {
        return List.copyOf(rows);
    }

    @Override
    public Optional<KnsnkrnRecord> findByGasMtstNo(Long gasMtstNo) {
        Long safeGasMtstNo = safeGasMtstNo(gasMtstNo);
        return rows.stream()
                .filter(row -> Objects.equals(safeGasMtstNo(row.gasMtstNo()), safeGasMtstNo))
                .findFirst();
    }

    /**
     * WHERE 条件に使用するガスメーター設置場所番号を補完する。
     *
     * @param value 入力値
     * @return 補完後の値
     */
    private static Long safeGasMtstNo(Long value) {
        return WhereParamGuard.complement(value, "gasMtstNo", 0L);
    }
}
