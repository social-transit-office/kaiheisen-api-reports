package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.flknmei;

/**
 * DA-024（FLKNMEI）のインメモリ実装。
 */
public class FlknmeiDataAccessImpl extends DataAccessBase implements FlknmeiDataAccess {

    private static final int INITIAL_FLKN_NO = 200000001;

    private static final ReflectiveInMemoryRowMapper<FlknmeiColumn, FlknmeiRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(FlknmeiColumn.class, FlknmeiRecord.class);

    private static final AutoNeutralValueResolver<FlknmeiColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(FlknmeiColumn.class, FlknmeiRecord.class);

    private final List<EnumMap<FlknmeiColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<FlknmeiWriteRecord, FlknmeiColumn> insertSupport;

    private int nextFlknNo = INITIAL_FLKN_NO;

    /**
     * デフォルト依存でインスタンスを生成する。
     */
    public FlknmeiDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    FlknmeiDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    FlknmeiDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(FlknmeiWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public FlknmeiDataAccessImpl addRow(FlknmeiRecord record) {
        EnumMap<FlknmeiColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, FlknmeiWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        advanceSequence(values);
        return this;
    }

    @Override
    public int insert(FlknmeiWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        EnumMap<FlknmeiColumn, Object> values =
                insertSupport.prepare(
                        record,
                        row -> {
                            row.put(FlknmeiColumn.FLKN_NO, nextFlknNo++);
                            auditColumnAutoSetter.applyForInsert(
                                    new MapBackedAuditColumnAccess<>(FlknmeiColumn.class, row),
                                    context,
                                    clock);
                        });
        rows.add(values);
        return 1;
    }

    @Override
    public Optional<FlknmeiRecord> findByPK(Integer flknNo) {
        Integer safeFlknNo = WhereParamGuard.complement(flknNo, "flknNo", 0);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(FlknmeiColumn.FLKN_NO), safeFlknNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private void advanceSequence(EnumMap<FlknmeiColumn, Object> values) {
        Integer flknNo = (Integer) values.get(FlknmeiColumn.FLKN_NO);
        if (flknNo != null && flknNo >= nextFlknNo) {
            nextFlknNo = flknNo + 1;
        }
    }
}
