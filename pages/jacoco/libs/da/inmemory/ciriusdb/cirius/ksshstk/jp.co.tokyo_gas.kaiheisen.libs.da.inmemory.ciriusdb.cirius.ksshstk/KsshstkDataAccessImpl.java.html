/**
 * DA-115 の SQL 振る舞いを模した KSSHSTK インメモリ実装。
 */
public class KsshstkDataAccessImpl extends DataAccessBase implements KsshstkDataAccess {

    private static final ReflectiveInMemoryRowMapper<KsshstkColumn, KsshstkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KsshstkColumn.class, KsshstkRecord.class);

    private static final AutoNeutralValueResolver<KsshstkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KsshstkColumn.class, KsshstkRecord.class);

    private final List<EnumMap<KsshstkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KsshstkWriteRecord, KsshstkColumn> insertSupport;

    /**
     * システム日付プロバイダのデフォルト実装で初期化する。
     */
    public KsshstkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KsshstkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KsshstkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KsshstkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public KsshstkDataAccessImpl addRow(KsshstkRecord record) {
        EnumMap<KsshstkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KsshstkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KsshstkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KsshstkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<KsshstkRecord> findBySvkyNo(Long svkyNo) {
        Long safeSvkyNo = WhereParamGuard.complement(svkyNo, "svkyNo", 0L);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(KsshstkColumn.SVKY_NO), safeSvkyNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int deleteBySvkyNo(Long svkyNo, Integer recUpdJnjNo) {
        Long safeSvkyNo = WhereParamGuard.complement(svkyNo, "svkyNo", 0L);

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(row.get(KsshstkColumn.SVKY_NO), safeSvkyNo),
                KsshstkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KsshstkDataAccess.Identifiers.DELETE_BY_SVKY_NO,
                () -> new DataInconsistencyException(
                        "KSSHSTK",
                        KsshstkDataAccess.Identifiers.DELETE_BY_SVKY_NO,
                        "{}"),
                safeSvkyNo);
    }
}
