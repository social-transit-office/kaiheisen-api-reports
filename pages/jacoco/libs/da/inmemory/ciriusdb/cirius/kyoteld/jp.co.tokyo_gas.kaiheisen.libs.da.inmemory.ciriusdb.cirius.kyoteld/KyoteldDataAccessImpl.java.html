package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.kyoteld;

import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;

/** DA-106（KYOTELD）のインメモリ実装。 */
public class KyoteldDataAccessImpl extends DataAccessBase implements KyoteldDataAccess {

    private static final int KYO_TEL_NO_LENGTH = 13;

    private static final ReflectiveInMemoryRowMapper<KyoteldColumn, KyoteldRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KyoteldColumn.class, KyoteldRecord.class);

    private static final AutoNeutralValueResolver<KyoteldColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KyoteldColumn.class, KyoteldRecord.class);

    private final List<EnumMap<KyoteldColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KyoteldWriteRecord, KyoteldColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KyoteldWriteRecord, KyoteldColumn>
            updateSupport;

    /**
     * デフォルトの日時プロバイダと監査列自動設定でインメモリ実装を生成する。
     */
    public KyoteldDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KyoteldDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KyoteldDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KyoteldWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KyoteldWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に初期行を追加する。
     *
     * @param record 追加するレコード
     * @return 自身
     */
    public KyoteldDataAccessImpl addRow(KyoteldRecord record) {
        EnumMap<KyoteldColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KyoteldWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KyoteldWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KyoteldColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            Long kyoTrkNo, String kyoTelNo, Integer recUpdJnjNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> matchesPrimaryKey(row, safeKyoTrkNo, safeKyoTelNo),
                KyoteldColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KyoteldDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                () ->
                        new DataInconsistencyException(
                                "KYOTELD",
                                KyoteldDataAccess.Identifiers.DELETE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                                "{}"),
                safeKyoTrkNo,
                safeKyoTelNo);
    }

    @Override
    public long countByKyoTrkNo(Long kyoTrkNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        return rows.stream().filter(r -> matchesKyoTrkNo(r, safeKyoTrkNo)).count();
    }

    @Override
    public Optional<KyoteldRecord> findByKyoTrkNoAndKyoTelNo(Long kyoTrkNo, String kyoTelNo) {
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        return rows.stream()
                .filter(r -> matchesPrimaryKey(r, safeKyoTrkNo, safeKyoTelNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByKyoTrkNoAndKyoTelNoAndRecUpdJnjNo(
            KyoteldWriteRecord record,
            Set<KyoteldColumn> targetColumns,
            KyoteldUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Long safeKyoTrkNo = safeLong(updateKey.kyoTrkNo(), "kyoTrkNo");
        String safeKyoTelNo = safeKyoTelNo(updateKey.kyoTelNo());
        Set<KyoteldColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matchesPrimaryKey(row, safeKyoTrkNo, safeKyoTelNo),
                row ->
                        updateSupport.prepare(
                                record,
                                canonicalTargetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KyoteldColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(KyoteldColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KyoteldColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KyoteldDataAccess.Identifiers.UPDATE_BY_KYO_TRK_NO_AND_KYO_TEL_NO_AND_REC_UPD_JNJ_NO,
                safeKyoTrkNo,
                safeKyoTelNo);
    }

    private static Set<KyoteldColumn> canonicalTargetColumns(Set<KyoteldColumn> targetColumns) {
        EnumSet<KyoteldColumn> canonicalColumns = EnumSet.noneOf(KyoteldColumn.class);
        canonicalColumns.addAll(targetColumns);
        return canonicalColumns;
    }

    private static boolean matchesKyoTrkNo(Map<KyoteldColumn, Object> row, Long safeKyoTrkNo) {
        return Objects.equals(row.get(KyoteldColumn.KYO_TRK_NO), safeKyoTrkNo);
    }

    private static boolean matchesPrimaryKey(
            Map<KyoteldColumn, Object> row, Long safeKyoTrkNo, String safeKyoTelNo) {
        return matchesKyoTrkNo(row, safeKyoTrkNo) && rowKyoTelNo(row).equals(safeKyoTelNo);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", KYO_TEL_NO_LENGTH);
    }

    private static String rowKyoTelNo(Map<KyoteldColumn, Object> row) {
        return WhereParamGuard.fixedLengthNotNull(
                (String) row.get(KyoteldColumn.KYO_TEL_NO),
                "kyoTelNo",
                KYO_TEL_NO_LENGTH);
    }
}
