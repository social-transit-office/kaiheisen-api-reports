package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.cstmst;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.cstmst.CstmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/**
 * CSTMST データアクセスのインメモリ実装。
 */
public class CstmstDataAccessImpl extends DataAccessBase implements CstmstDataAccess {

    private static final int CNST_SHBT_CD_LENGTH = 5;

    private static final int CNST_CD_LENGTH = 10;

    private static final int TIGN_SGN_LENGTH = 1;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private final List<CstmstRecord> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加するレコード
     * @return このインスタンス
     */
    public CstmstDataAccessImpl addRow(CstmstRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public List<CstmstRecord> findAllByCnstShbtCd(String cnstShbtCd) {
        String safeCnstShbtCd = safeCnstShbtCd(cnstShbtCd, "cnstShbtCd");
        return rows.stream()
                .filter(r -> safeCnstShbtCd.equals(normalizeCnstShbtCd(r.cnstShbtCd())))
                .sorted(Comparator.comparing(CstmstRecord::cnstCd))
                .toList();
    }

    @Override
    public Optional<CstmstRecord> findByCnstShbtCdAndCnstCdAndTkysYmdAndTyenYmdAndTignSgn(
            String cnstShbtCd, String cnstCd, LocalDate knskKjnYmd, String tignSgn) {
        if (tignSgn == null || tignSgn.isBlank()) {
            return Optional.empty();
        }
        String safeCnstShbtCd = safeCnstShbtCd(cnstShbtCd, "cnstShbtCd");
        String safeCnstCd = safeCnstCd(cnstCd, "cnstCd");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        String safeTignSgn = safeTignSgn(tignSgn, "tignSgn");
        return rows.stream()
                .filter(r -> safeCnstShbtCd.equals(normalizeCnstShbtCd(r.cnstShbtCd())))
                .filter(r -> safeCnstCd.equals(normalizeCnstCd(r.cnstCd())))
                .filter(r -> !normalizeYmd(r.tkysYmd()).isAfter(safeKnskKjnYmd))
                .filter(r -> !normalizeYmd(r.tyenYmd()).isBefore(safeKnskKjnYmd))
                .filter(r -> !safeTignSgn.equals(normalizeTignSgn(r.tignSgn())))
                .findFirst();
    }

    private static String safeCnstShbtCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, CNST_SHBT_CD_LENGTH);
    }

    private static String safeCnstCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, CNST_CD_LENGTH);
    }

    private static String safeTignSgn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TIGN_SGN_LENGTH);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String normalizeCnstShbtCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "cnstShbtCd", CNST_SHBT_CD_LENGTH);
    }

    private static String normalizeCnstCd(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "cnstCd", CNST_CD_LENGTH);
    }

    private static String normalizeTignSgn(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "tignSgn", TIGN_SGN_LENGTH);
    }

    private static LocalDate normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(value, "ymd", COMPLEMENT_YMD);
    }
}
