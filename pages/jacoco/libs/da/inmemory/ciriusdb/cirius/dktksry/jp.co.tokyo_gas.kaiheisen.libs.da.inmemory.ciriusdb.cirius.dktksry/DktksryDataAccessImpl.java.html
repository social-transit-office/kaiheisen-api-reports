/**
 * DA-089（C001/D001/R001-R005/U001）の SQL 振る舞いを模した
 * DKTKSRY インメモリ実装.
 */
public class DktksryDataAccessImpl extends DataAccessBase
        implements DktksryDataAccess {

    /** KNKK_SKSI_JYSB の固定長. */
    private static final int KNKK_SKSI_JYSB_LENGTH = 3;

    /** レコードから raw 値マップへ変換する mapper. */
    private static final ReflectiveInMemoryRowMapper<
                    DktksryColumn,
                    DktksryRecord>
            ROW_MAPPER = ReflectiveInMemoryRowMapper.create(
                    DktksryColumn.class,
                    DktksryRecord.class);

    /** neutral 値を補完する resolver. */
    private static final AutoNeutralValueResolver<DktksryColumn>
            NEUTRAL_VALUE_RESOLVER = AutoNeutralValueResolver.create(
                    DktksryColumn.class,
                    DktksryRecord.class);

    /** テスト用に保持する row 一覧. */
    private final List<EnumMap<DktksryColumn, Object>> rows = new ArrayList<>();

    /** システム日時の取得元. */
    private final SystemDateTimeProvider clock;

    /** 監査項目の自動設定ロジック. */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** insert 用の値補完サポート. */
    private final InMemoryInsertSupport<
                    DktksryWriteRecord,
                    DktksryColumn>
            insertSupport;

    /** update 用の値補完サポート. */
    private final InMemoryTargetColumnUpdateSupport<
                    DktksryWriteRecord,
                    DktksryColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する. */
    public DktksryDataAccessImpl() {
        this(
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    DktksryDataAccessImpl(final SystemDateTimeProvider systemDateTimeProvider) {
        this(systemDateTimeProvider, new CiriusAuditColumnAutoSetter());
    }

    DktksryDataAccessImpl(
            final SystemDateTimeProvider systemDateTimeProvider,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(
                systemDateTimeProvider,
                "systemDateTimeProvider");
        this.auditColumnAutoSetter = Objects.requireNonNull(
                ciriusAuditColumnAutoSetter,
                "ciriusAuditColumnAutoSetter");
        this.insertSupport = new InMemoryInsertSupport<>(
                DktksryWriteDefinition.INSTANCE,
                NEUTRAL_VALUE_RESOLVER);
        this.updateSupport = new InMemoryTargetColumnUpdateSupport<>(
                DktksryWriteDefinition.INSTANCE,
                NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する.
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public DktksryDataAccessImpl addRow(final DktksryRecord row) {
        EnumMap<DktksryColumn, Object> values = ROW_MAPPER.toRawValueMap(
                Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(
                values,
                DktksryWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * テスト用に保持している登録済みレコードを返す.
     *
     * @return 登録済みレコード一覧
     */
    public List<DktksryRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(
            final DktksryWriteRecord record,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                DktksryColumn.class,
                                                values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            final Long gasMtstNo,
            final LocalDate knsYmd,
            final Integer knsKkaTsno,
            final Integer recUpdJnjNo) {
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                                == safeLong(gasMtstNo)
                        && sameComplementedYmd(
                                row.get(DktksryColumn.KNS_YMD),
                                knsYmd)
                        && safeInt(row.get(DktksryColumn.KNS_KKA_TSNO))
                                == safeInt(knsKkaTsno),
                DktksryColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                DktksryDataAccess.CallSpecs
                        .DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO
                        .identifier(),
                () -> new DataInconsistencyException(
                        "DKTKSRY",
                        deleteCallIdentifier(),
                        "{}"),
                gasMtstNo,
                knsYmd,
                knsKkaTsno);
    }

    @Override
    public List<DktksryRecord> findAll(
            final Long gasMtstNo,
            final LocalDate knsYmd,
            final Integer knsKkaTsno) {
        return rows.stream()
                .filter(row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                        == safeLong(gasMtstNo))
                .filter(row -> sameComplementedYmd(
                        row.get(DktksryColumn.KNS_YMD),
                        knsYmd))
                .filter(row -> safeInt(row.get(DktksryColumn.KNS_KKA_TSNO))
                        == safeInt(knsKkaTsno))
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmd(
            final Long gasMtstNo,
            final LocalDate knsYmd) {
        return rows.stream()
                .filter(row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                        == safeLong(gasMtstNo))
                .filter(row -> isNotBefore(
                        row,
                        DktksryColumn.KNS_YMD,
                        knsYmd))
                .sorted(knsYmdDescComparator())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
            final Long gasMtstNo,
            final LocalDate knsYmd,
            final String knkkSksiJysb) {
        return rows.stream()
                .filter(row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                        == safeLong(gasMtstNo))
                .filter(row -> sameComplementedYmd(
                        row.get(DktksryColumn.KNS_YMD),
                        knsYmd))
                .filter(row -> Objects.equals(
                        complementKnkkSksiJysb(
                                row.get(DktksryColumn.KNKK_SKSI_JYSB)),
                        complementKnkkSksiJysb(knkkSksiJysb)))
                .sorted(knsKkaTsnoDescComparator())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdRange(
            final Long gasMtstNo,
            final LocalDate knsYmdFrom,
            final LocalDate knsYmdTo) {
        return rows.stream()
                .filter(row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                        == safeLong(gasMtstNo))
                .filter(row -> isNotBefore(
                        row,
                        DktksryColumn.KNS_YMD,
                        knsYmdFrom))
                .filter(row -> isNotAfter(
                        row,
                        DktksryColumn.KNS_YMD,
                        knsYmdTo))
                .sorted(knsYmdAndTsnoDescComparator())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<DktksryRecord> findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
            final Long gasMtstNo,
            final LocalDate knsYmdFrom,
            final LocalDate knsYmdTo,
            final String knkkSksiJysb) {
        return rows.stream()
                .filter(row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                        == safeLong(gasMtstNo))
                .filter(row -> isNotBefore(
                        row,
                        DktksryColumn.KNS_YMD,
                        knsYmdFrom))
                .filter(row -> isNotAfter(
                        row,
                        DktksryColumn.KNS_YMD,
                        knsYmdTo))
                .filter(row -> Objects.equals(
                        complementKnkkSksiJysb(
                                row.get(DktksryColumn.KNKK_SKSI_JYSB)),
                        complementKnkkSksiJysb(knkkSksiJysb)))
                .sorted(knsYmdAndTsnoDescComparator())
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            final DktksryWriteRecord record,
            final Set<DktksryColumn> targetColumns,
            final DktksryUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> safeLong(row.get(DktksryColumn.GAS_MTST_NO))
                                == safeLong(updateKey.gasMtstNo())
                        && sameComplementedYmd(
                                row.get(DktksryColumn.KNS_YMD),
                                updateKey.knsYmd())
                        && safeInt(row.get(DktksryColumn.KNS_KKA_TSNO))
                                == safeInt(updateKey.knsKkaTsno()),
                row -> updateSupport.prepare(
                        record,
                        targetColumns,
                        values -> auditColumnAutoSetter.applyForUpdate(
                                new MapBackedAuditColumnAccess<>(
                                        DktksryColumn.class,
                                        values),
                                context,
                                clock,
                                (Integer)
                                        row.get(DktksryColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                DktksryColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                DktksryDataAccess.CallSpecs
                        .UPDATE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO
                        .identifier(),
                updateKey.gasMtstNo(),
                updateKey.knsYmd(),
                updateKey.knsKkaTsno());
    }

    private static boolean isNotAfter(
            final EnumMap<DktksryColumn, Object> row,
            final DktksryColumn column,
            final LocalDate value) {
        return complementedYmdValue((LocalDate) row.get(column))
                <= complementedYmdValue(value);
    }

    private static boolean isNotBefore(
            final EnumMap<DktksryColumn, Object> row,
            final DktksryColumn column,
            final LocalDate value) {
        return complementedYmdValue((LocalDate) row.get(column))
                >= complementedYmdValue(value);
    }

    private static boolean sameComplementedYmd(
            final Object left,
            final LocalDate right) {
        return complementedYmdValue((LocalDate) left)
                == complementedYmdValue(right);
    }

    private static String deleteCallIdentifier() {
        return DktksryDataAccess.CallSpecs
                .DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO
                .identifier();
    }

    private static Integer knsKkaTsnoValue(
            final EnumMap<DktksryColumn, Object> row) {
        return (Integer) row.get(DktksryColumn.KNS_KKA_TSNO);
    }

    private static Comparator<EnumMap<DktksryColumn, Object>>
            knsYmdDescComparator() {
        return Comparator.comparing(
                DktksryDataAccessImpl::knsYmdValue,
                Comparator.nullsLast(Comparator.reverseOrder()));
    }

    private static Comparator<EnumMap<DktksryColumn, Object>>
            knsKkaTsnoDescComparator() {
        return Comparator.comparing(
                DktksryDataAccessImpl::knsKkaTsnoValue,
                Comparator.nullsLast(Comparator.reverseOrder()));
    }

    private static Comparator<EnumMap<DktksryColumn, Object>>
            knsYmdAndTsnoDescComparator() {
        return knsYmdDescComparator()
                .thenComparing(knsKkaTsnoDescComparator());
    }

    private static LocalDate knsYmdValue(
            final EnumMap<DktksryColumn, Object> row) {
        return (LocalDate) row.get(DktksryColumn.KNS_YMD);
    }

    private static int complementedYmdValue(final LocalDate value) {
        if (value == null) {
            return 0;
        }
        return Integer.parseInt(value.format(BASIC_ISO_DATE));
    }

    private static long safeLong(final Object value) {
        if (value instanceof Number number) {
            return number.longValue();
        }
        if (value == null) {
            return 0L;
        }
        return Long.parseLong(value.toString());
    }

    private static int safeInt(final Object value) {
        if (value instanceof Number number) {
            return number.intValue();
        }
        if (value == null) {
            return 0;
        }
        return Integer.parseInt(value.toString());
    }

    private static String complementKnkkSksiJysb(final Object value) {
        return WhereParamGuard.fixedLengthNotNull(
                (String) value,
                "knkkSksiJysb",
                KNKK_SKSI_JYSB_LENGTH);
    }
}
