/**
 * DA-070 の SQL 振る舞いを模した KUZFRK インメモリ実装。
 */
public class KuzfrkDataAccessImpl extends DataAccessBase implements KuzfrkDataAccess {

    private static final ReflectiveInMemoryRowMapper<KuzfrkColumn, KuzfrkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KuzfrkColumn.class, KuzfrkRecord.class);

    private static final AutoNeutralValueResolver<KuzfrkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KuzfrkColumn.class, KuzfrkRecord.class);

    private final List<EnumMap<KuzfrkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KuzfrkWriteRecord, KuzfrkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KuzfrkWriteRecord, KuzfrkColumn> updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public KuzfrkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に時刻依存を差し替えてインスタンスを生成する。
     *
     * @param systemClock システム時刻プロバイダ
     */
    KuzfrkDataAccessImpl(final SystemDateTimeProvider systemClock) {
        this(systemClock, new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えてインスタンスを生成する。
     *
     * @param systemClock システム時刻プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査項目自動設定
     */
    KuzfrkDataAccessImpl(
            final SystemDateTimeProvider systemClock,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(systemClock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(ciriusAuditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(KuzfrkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KuzfrkWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param record 追加対象レコード
     * @return このインスタンス
     */
    public KuzfrkDataAccessImpl addRow(final KuzfrkRecord record) {
        EnumMap<KuzfrkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KuzfrkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(final KuzfrkWriteRecord record, final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values -> auditColumnAutoSetter.applyForInsert(
                                new MapBackedAuditColumnAccess<>(KuzfrkColumn.class, values),
                                context,
                                clock)));
        return 1;
    }

    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(row.get(KuzfrkColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(row.get(KuzfrkColumn.SIH_KIY_SKNO), safeSihKiySkno),
                KuzfrkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KuzfrkDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                () -> new DataInconsistencyException(
                        "KUZFRK",
                        KuzfrkDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                        "{}"),
                safeSihKiyNo,
                safeSihKiySkno);
    }

    @Override
    public Optional<KuzfrkRecord> findBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(KuzfrkColumn.SIH_KIY_NO), safeSihKiyNo))
                .filter(row -> Objects.equals(row.get(KuzfrkColumn.SIH_KIY_SKNO), safeSihKiySkno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final KuzfrkSihKiyUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                KuzfrkDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final KuzfrkSihKiyJnjUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                KuzfrkDataAccess.CallSpecs
                        .UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO
                        .identifier());
    }

    private int updateByKey(
            final KuzfrkWriteRecord record,
            final Set<KuzfrkColumn> targetColumns,
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo,
            final AuditContext context,
            final String dataAccessIdentifier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(context, "context");
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(row.get(KuzfrkColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(row.get(KuzfrkColumn.SIH_KIY_SKNO), safeSihKiySkno),
                row -> updateSupport.prepare(
                        record,
                        targetColumns,
                        values -> auditColumnAutoSetter.applyForUpdate(
                                new MapBackedAuditColumnAccess<>(KuzfrkColumn.class, values),
                                context,
                                clock,
                                (Integer) row.get(KuzfrkColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                KuzfrkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                dataAccessIdentifier,
                safeSihKiyNo,
                safeSihKiySkno);
    }

    private static Long safeLong(final Long value, final String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(final Integer value, final String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }
}
