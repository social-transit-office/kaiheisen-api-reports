package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.ssvsbhw;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ssvsbhw.SsvsbhwDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.ssvsbhw.SsvsbhwRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-157-R001 の SQL をインメモリで再現する実装。 */
public class SsvsbhwDataAccessImpl extends DataAccessBase implements SsvsbhwDataAccess {

    private static final int SQSVS_HWSB_LENGTH = 4;

    private static final int SIK_SVC_SHBT_LENGTH = 2;

    private final List<SsvsbhwRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public SsvsbhwDataAccessImpl addRow(SsvsbhwRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<SsvsbhwRecord> findBySqsvsHwsbAndSikSvcShbtAndKnskKjnYmdOrderByTkysYmdDesc(
            String sqsvsHwsb,
            String sikSvcShbt,
            LocalDate knskKjnYmd) {
        String safeSqsvsHwsb =
                WhereParamGuard.fixedLengthNotNull(sqsvsHwsb, "sqsvsHwsb", SQSVS_HWSB_LENGTH);
        String safeSikSvcShbt =
                WhereParamGuard.fixedLengthNotNull(sikSvcShbt, "sikSvcShbt", SIK_SVC_SHBT_LENGTH);
        LocalDate safeKnskKjnYmd = WhereParamGuard.complement(knskKjnYmd, "knskKjnYmd", LocalDate.MIN);

        return rows.stream()
                .filter(row -> normalizeFixed(row.sqsvsHwsb(), SQSVS_HWSB_LENGTH).equals(safeSqsvsHwsb))
                .filter(row -> normalizeFixed(row.sikSvcShbt(), SIK_SVC_SHBT_LENGTH).equals(safeSikSvcShbt))
                .filter(row -> normalizeDate(row.tkysYmd()).compareTo(safeKnskKjnYmd) <= 0)
                .filter(row -> normalizeDate(row.tyenYmd()).compareTo(safeKnskKjnYmd) >= 0)
                .sorted(Comparator.comparing(SsvsbhwDataAccessImpl::sortDate, Comparator.reverseOrder()))
                .map(SsvsbhwDataAccessImpl::normalizeRecord)
                .findFirst();
    }

    private static String normalizeFixed(String value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", length);
    }

    private static LocalDate normalizeDate(LocalDate value) {
        return WhereParamGuard.complement(value, "value", LocalDate.MIN);
    }

    private static LocalDate sortDate(SsvsbhwRecord row) {
        return normalizeDate(row.tkysYmd());
    }

    private static SsvsbhwRecord normalizeRecord(SsvsbhwRecord row) {
        return new SsvsbhwRecord(
                normalizeResultText(row.sqsvsHwsb()),
                normalizeResultText(row.sikSvcShbt()),
                normalizeResultDate(row.tkysYmd()),
                normalizeResultDate(row.tyenYmd()),
                normalizeResultText(row.sqsvsHnsb()),
                row.regTsmp(),
                normalizeResultText(row.regUsrId()),
                row.updTsmp(),
                normalizeResultText(row.updUsrId()),
                normalizeResultText(row.updKtnKinoId()),
                normalizeResultInteger(row.recUpdJnjNo()));
    }

    private static String normalizeResultText(String value) {
        return FixedLengthFormatter.suppressSpace(value);
    }

    private static LocalDate normalizeResultDate(LocalDate value) {
        if (value == null || LocalDate.MIN.equals(value)) {
            return null;
        }
        return value;
    }

    private static Integer normalizeResultInteger(Integer value) {
        if (value == null || value == 0) {
            return null;
        }
        return value;
    }
}
