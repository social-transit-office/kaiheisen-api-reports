    @Override
    public List<NtknskRecord> findAllByKymYmdAndNwjgCdAndUtltKbnAndTkkbNo(
            LocalDate kymYmdFrom,
            LocalDate kymYmdTo,
            String nwjgCd,
            String utltKbn,
            String tkkbNo) {
        int safeKymYmdFrom = safeYmd(kymYmdFrom, "kymYmdFrom");
        int safeKymYmdTo = safeYmd(kymYmdTo, "kymYmdTo");
        String safeNwjgCd = safeNwjgCd(nwjgCd, "nwjgCd");
        String safeUtltKbn = safeUtltKbn(utltKbn, "utltKbn");
        String safeTkkbNo = safeTkkbNo(tkkbNo, "tkkbNo");

        return rows.stream()
            .filter(row -> row.kymYmd() >= safeKymYmdFrom)
            .filter(row -> row.kymYmd() <= safeKymYmdTo)
            .filter(row -> Objects.equals(normalizeNwjgCd(row.nwjgCd()), safeNwjgCd))
            .filter(row -> Objects.equals(normalizeUtltKbn(row.utltKbn()), safeUtltKbn))
            .filter(row -> Objects.equals(normalizeTkkbNo(row.tkkbNo()), safeTkkbNo))
            .map(row -> toRecord(row))
                .toList();
    }

    @Override
    public Optional<NtknskRecord> findByPK(LocalDate kymYmd, String nwjgCd, String utltKbn) {
        int safeKymYmd = safeYmd(kymYmd, "kymYmd");
        String safeNwjgCd = safeNwjgCd(nwjgCd, "nwjgCd");
        String safeUtltKbn = safeUtltKbn(utltKbn, "utltKbn");

        for (InternalRow row : rows) {
            if (safeKymYmd == row.kymYmd()
                    && Objects.equals(normalizeNwjgCd(row.nwjgCd()), safeNwjgCd)
                    && Objects.equals(normalizeUtltKbn(row.utltKbn()), safeUtltKbn)) {
                return Optional.of(toRecord(row));
            }
        }
        return Optional.empty();
    }

    private static int safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(toYmdIntOrNull(value), paramName, 0);
    }

    private static String safeNwjgCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, NWJG_CD_LENGTH);
    }

    private static String safeUtltKbn(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, UTLT_KBN_LENGTH);
    }

    private static String safeTrekSskbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TREK_SSKB_NO_LENGTH);
    }

    private static String safeTrekSdkbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TREK_SDKB_NO_LENGTH);
    }

    private static String safeTkkbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, TKKB_NO_LENGTH);
    }

    private static String safeShtiEkkbNo(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, SHTI_EKKB_NO_LENGTH);
    }

    private static String safeShtiHksdKbn(String value, String paramName) {

    }

    private static String safeShtiEksdKbn(String value, String paramName) {

    }

    private static int toYmdInt(LocalDate date) {
        return date.getYear() * 10000 + date.getMonthValue() * 100 + date.getDayOfMonth();
    }

    private static Integer toYmdIntOrNull(LocalDate date) {
        if (date == null || LocalDate.MIN.equals(date)) {
            return null;
        }
        return toYmdInt(date);
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }
