    @Override
    public List<GmkskkdRecord> findAllByKnsYmdAndMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
            LocalDate knsYmdFrom,
            LocalDate knsYmdTo,
            Long mtgMtstNo,
            LocalDate mgmtKnsYmd,
            Integer mgmtKnkkTsno) {
        long safeMtgMtstNo = safeLong(mtgMtstNo, "mtgMtstNo");
        int safeMgmtKnkkTsno = safeInteger(mgmtKnkkTsno, "mgmtKnkkTsno");

        if (safeMtgMtstNo == 0L) {
            return List.of();
        }

        int knsYmdFromValue = safeYmd(knsYmdFrom, "knsYmdFrom");
        int knsYmdToValue = safeYmd(knsYmdTo, "knsYmdTo");
        int mgmtKnsYmdValue = safeYmd(mgmtKnsYmd, "mgmtKnsYmd");
        return rows.stream()
                .filter(row -> row.knsYmd() >= knsYmdFromValue)
                .filter(row -> row.knsYmd() <= knsYmdToValue)
                .filter(row -> row.mtgMtstNo() == safeMtgMtstNo)
                .filter(row -> row.mtgMtstNo() != 0)
                .filter(row -> row.mgmtKnsYmd() == mgmtKnsYmdValue)
                .filter(row -> row.mgmtKnkkTsno() == safeMgmtKnkkTsno)
                .sorted(
                        Comparator.comparingLong(InternalRow::gasMtstNo)
                                .thenComparing(
                                        Comparator.comparingInt(InternalRow::knsYmd).reversed())
                                .thenComparing(
                                        Comparator.comparingInt(InternalRow::knsKkaTsno)
                                                .reversed()))
                .map(GmkskkdDataAccessImpl::toRecord)
                .toList();
    }

        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(knsYmd, "knsYmd");
        Objects.requireNonNull(knsKkaTsno, "knsKkaTsno");

        int knsYmdInt = toYmdInt(knsYmd);
        for (InternalRow row : rows) {
            if (gasMtstNo == row.gasMtstNo()
                    && knsYmdInt == row.knsYmd()
                    && knsKkaTsno == row.knsKkaTsno()) {
                return Optional.of(toRecord(row));
            }
        }
        return Optional.empty();
    }

    private static int safeYmd(LocalDate value, String paramName) {
        LocalDate safeValue = WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
        return safeValue.equals(COMPLEMENT_YMD) ? 0 : toYmdInt(safeValue);
    }

    private static long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static int safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static int toYmdInt(LocalDate date) {
        return date.getYear() * 10000 + date.getMonthValue() * 100 + date.getDayOfMonth();
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }

    private static Boolean toBooleanFlag(String value) {
        if ("1".equals(value)) {
            return Boolean.TRUE;
        }
        if ("0".equals(value)) {
            return Boolean.FALSE;
        }
        return null;
    }
