package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.shyfrke;

import java.util.ArrayList;

import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/** DA-003-U001 の SQL 振る舞いを模した SHYFRKE インメモリ実装。 */
public class ShyfrkeDataAccessImpl extends DataAccessBase implements ShyfrkeDataAccess {

    private static final ReflectiveInMemoryRowMapper<ShyfrkeColumn, ShyfrkeRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(ShyfrkeColumn.class, ShyfrkeRecord.class);

    private static final AutoNeutralValueResolver<ShyfrkeColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(ShyfrkeColumn.class, ShyfrkeRecord.class);

    private final List<EnumMap<ShyfrkeColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryTargetColumnUpdateSupport<ShyfrkeWriteRecord, ShyfrkeColumn>
            updateSupport;

    /** デフォルトコンストラクタ。 */
    public ShyfrkeDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    ShyfrkeDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        ShyfrkeWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public ShyfrkeDataAccessImpl addRow(ShyfrkeRecord row) {
        EnumMap<ShyfrkeColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, ShyfrkeWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public Optional<ShyfrkeRecord> findByPK(Long sihKiyNo, LocalDate tkysYmd) {
        Objects.requireNonNull(sihKiyNo, "sihKiyNo");
        Objects.requireNonNull(tkysYmd, "tkysYmd");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(ShyfrkeColumn.SIH_KIY_NO), sihKiyNo))
                .filter(row -> Objects.equals(row.get(ShyfrkeColumn.TKYS_YMD), tkysYmd))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateBySihKiyNoAndTkysYmdAndTyenYmd(
            ShyfrkeWriteRecord record,
            Set<ShyfrkeColumn> targetColumns,
            ShyfrkeUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Long safeSihKiyNo = Objects.requireNonNullElse(updateKey.sihKiyNo(), 0L);
        LocalDate safeKnskKjnYmd =
                Objects.requireNonNullElse(updateKey.knskKjnYmd(), LocalDate.of(1, 1, 1));

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> matches(row, safeSihKiyNo, safeKnskKjnYmd),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        ShyfrkeColumn.class,
                                                        values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(ShyfrkeColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                ShyfrkeColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                ShyfrkeDataAccess.Identifiers.UPDATE_BY_SIH_KIY_NO_AND_TKYS_YMD_AND_TYEN_YMD,
                safeSihKiyNo,
                safeKnskKjnYmd);
    }

    private boolean matches(
                        Map<ShyfrkeColumn, Object> row, Long safeSihKiyNo, LocalDate safeKnskKjnYmd) {
        if (!Objects.equals(row.get(ShyfrkeColumn.SIH_KIY_NO), safeSihKiyNo)) {
            return false;
        }
        LocalDate tkysYmd = (LocalDate) row.get(ShyfrkeColumn.TKYS_YMD);
        LocalDate tyenYmd = (LocalDate) row.get(ShyfrkeColumn.TYEN_YMD);
        return tkysYmd != null
                && tyenYmd != null
                && !tkysYmd.isAfter(safeKnskKjnYmd)
                && !tyenYmd.isBefore(safeKnskKjnYmd);
    }
}
