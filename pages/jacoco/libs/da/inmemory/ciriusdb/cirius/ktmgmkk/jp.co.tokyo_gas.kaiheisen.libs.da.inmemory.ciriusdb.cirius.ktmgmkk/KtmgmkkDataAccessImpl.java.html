/** DA-052（C001/D001/R001-R008）の SQL 振る舞いを模した KTMGMKK インメモリ実装。 */
public class KtmgmkkDataAccessImpl extends DataAccessBase implements KtmgmkkDataAccess {

    private static final int KNKK_SKSI_JYSB_LENGTH = 3;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private static final ReflectiveInMemoryRowMapper<KtmgmkkColumn, KtmgmkkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KtmgmkkColumn.class, KtmgmkkRecord.class);

    private static final AutoNeutralValueResolver<KtmgmkkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KtmgmkkColumn.class, KtmgmkkRecord.class);

    private final List<EnumMap<KtmgmkkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KtmgmkkWriteRecord, KtmgmkkColumn> insertSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KtmgmkkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KtmgmkkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KtmgmkkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KtmgmkkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public KtmgmkkDataAccessImpl addRow(KtmgmkkRecord row) {
        EnumMap<KtmgmkkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KtmgmkkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KtmgmkkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values -> {
                            auditColumnAutoSetter.applyForInsert(
                                    new MapBackedAuditColumnAccess<>(
                                            KtmgmkkColumn.class, values),
                                    context,
                                    clock);
                            complementTkytYmdZeroSentinel(values);
                        }));
        return 1;
    }

    @Override
    public int deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
            Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno, Integer recUpdJnjNo) {
        Long safeGasMtstNo = WhereParamGuard.complement(gasMtstNo, "gasMtstNo", 0L);
        LocalDate safeKnsYmd = WhereParamGuard.complement(knsYmd, "knsYmd", LocalDate.MIN);
        Integer safeKnsKkaTsno = WhereParamGuard.complement(knsKkaTsno, "knsKkaTsno", 0);

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        Objects.equals(row.get(KtmgmkkColumn.GAS_MTST_NO), safeGasMtstNo)
                                && Objects.equals(row.get(KtmgmkkColumn.KNS_YMD), safeKnsYmd)
                                && Objects.equals(row.get(KtmgmkkColumn.KNS_KKA_TSNO), safeKnsKkaTsno),
                KtmgmkkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KtmgmkkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO,
                () ->
                        new DataInconsistencyException(
                                "KTMGMKK",
                                KtmgmkkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO,
                                "{}"),
                safeGasMtstNo,
                safeKnsYmd,
                safeKnsKkaTsno);
    }

    @Override
    public Optional<KtmgmkkRecord> findByPK(Long gasMtstNo, LocalDate knsYmd, Integer knsKkaTsno) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        Integer safeKnsKkaTsno = safeInteger(knsKkaTsno, "knsKkaTsno");

        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(row -> Objects.equals(normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)), safeKnsYmd))
                .filter(row -> Objects.equals(normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)), safeKnsKkaTsno))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNo(Long gasMtstNo) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");

        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, LocalDate>comparing(
                                        row -> normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)),
                                        Comparator.reverseOrder())
                                .thenComparing(
                                        row -> normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)),
                                        Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmd(Long gasMtstNo, LocalDate knsYmd) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeKnsYmd = safeYmd(knsYmd, "knsYmd");

        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(row -> Objects.equals(normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)), safeKnsYmd))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, Integer>comparing(
                                row -> normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)),
                                Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmdAndKnkkSksiJysb(
            Long gasMtstNo, LocalDate knsYmd, List<String> knkkSksiJysbList) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeKnsYmd = safeYmd(knsYmd, "knsYmd");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return Optional.empty();
        }
        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(row -> Objects.equals(normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)), safeKnsYmd))
                .filter(row -> safeKnkkSksiJysbList.contains(normalizeKnkkSksiJysb((String) row.get(KtmgmkkColumn.KNKK_SKSI_JYSB))))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, Integer>comparing(
                                row -> normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)),
                                Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnsYmdRangeAndKnkkSksiJysb(
            Long gasMtstNo,
            LocalDate knsYmdFrom,
            LocalDate knsYmdTo,
            List<String> knkkSksiJysbList) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeKnsYmdFrom = safeYmd(knsYmdFrom, "knsYmdFrom");
        LocalDate safeKnsYmdTo = safeYmd(knsYmdTo, "knsYmdTo");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return Optional.empty();
        }
        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(
                        row -> {
                            LocalDate ymd = normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD));
                            return !ymd.isBefore(safeKnsYmdFrom);
                        })
                .filter(
                        row -> {
                            LocalDate ymd = normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD));
                            return !ymd.isAfter(safeKnsYmdTo);
                        })
                .filter(row -> safeKnkkSksiJysbList.contains(normalizeKnkkSksiJysb((String) row.get(KtmgmkkColumn.KNKK_SKSI_JYSB))))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, LocalDate>comparing(
                                        row -> normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)),
                                        Comparator.reverseOrder())
                                .thenComparing(
                                        row -> normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)),
                                        Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KtmgmkkRecord> findByGasMtstNoAndKnkkSksiJysb(
            Long gasMtstNo, String knkkSksiJysb) {
        if (knkkSksiJysb == null || knkkSksiJysb.isBlank()) {
            return Optional.empty();
        }
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        String safeKnkkSksiJysb = safeKnkkSksiJysb(knkkSksiJysb, "knkkSksiJysb");

        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(row -> !safeKnkkSksiJysb.equals(normalizeKnkkSksiJysb((String) row.get(KtmgmkkColumn.KNKK_SKSI_JYSB))))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, LocalDate>comparing(
                                        row -> normalizeYmd((LocalDate) row.get(KtmgmkkColumn.KNS_YMD)),
                                        Comparator.reverseOrder())
                                .thenComparing(
                                        row -> normalizeInteger((Integer) row.get(KtmgmkkColumn.KNS_KKA_TSNO)),
                                        Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .findFirst();
    }

    @Override
    public long countByGasMtstNoAndKnkkSksiJysb(Long gasMtstNo, List<String> knkkSksiJysbList) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        List<String> safeKnkkSksiJysbList = normalizeKnkkSksiJysbList(knkkSksiJysbList);

        if (safeKnkkSksiJysbList.isEmpty()) {
            return 0L;
        }
        return rows.stream()
                .filter(row -> Objects.equals(normalizeLong((Long) row.get(KtmgmkkColumn.GAS_MTST_NO)), safeGasMtstNo))
                .filter(row -> safeKnkkSksiJysbList.contains(normalizeKnkkSksiJysb((String) row.get(KtmgmkkColumn.KNKK_SKSI_JYSB))))
                .count();
    }

    @Override
    public List<KtmgmkkRecord> findAllByMtgMtstNoAndMgmtKnsYmdAndMgmtKnkkTsno(
            Long mtgMtstNo, LocalDate mgmtKnsYmd, Integer mgmtKnkkTsno) {
        Long safeMtgMtstNo = safeLong(mtgMtstNo, "mtgMtstNo");
        LocalDate safeMgmtKnsYmd = safeYmd(mgmtKnsYmd, "mgmtKnsYmd");
        Integer safeMgmtKnkkTsno = safeInteger(mgmtKnkkTsno, "mgmtKnkkTsno");

        return rows.stream()
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeLong(
                                                (Long) row.get(KtmgmkkColumn.MTG_MTST_NO)),
                                        safeMtgMtstNo))
                .filter(
                        row ->
                                !Objects.equals(
                                        normalizeLong(
                                                (Long) row.get(KtmgmkkColumn.MTG_MTST_NO)),
                                        0L))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeYmd(
                                                (LocalDate)
                                                        row.get(KtmgmkkColumn.MGMT_KNS_YMD)),
                                        safeMgmtKnsYmd))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeInteger(
                                                (Integer)
                                                        row.get(KtmgmkkColumn.MGMT_KNKK_TSNO)),
                                        safeMgmtKnkkTsno))
                .sorted(
                        Comparator.<EnumMap<KtmgmkkColumn, Object>, Long>comparing(
                                        row ->
                                                normalizeLong(
                                                        (Long)
                                                                row.get(
                                                                        KtmgmkkColumn
                                                                                .GAS_MTST_NO)),
                                        Comparator.naturalOrder())
                                .thenComparing(
                                        row ->
                                                normalizeYmd(
                                                        (LocalDate)
                                                                row.get(
                                                                        KtmgmkkColumn
                                                                                .KNS_YMD)),
                                        Comparator.reverseOrder())
                                .thenComparing(
                                        row ->
                                                normalizeInteger(
                                                        (Integer)
                                                                row.get(
                                                                        KtmgmkkColumn
                                                                                .KNS_KKA_TSNO)),
                                        Comparator.reverseOrder()))
                .map(KtmgmkkDataAccessImpl::toRecord)
                .toList();
    }

    private static void complementTkytYmdZeroSentinel(EnumMap<KtmgmkkColumn, Object> values) {
        if (values.get(KtmgmkkColumn.TKYT_YMD) == null) {
            values.put(KtmgmkkColumn.TKYT_YMD, COMPLEMENT_YMD);
        }
    }

    private static KtmgmkkRecord toRecord(EnumMap<KtmgmkkColumn, Object> row) {
        if (!Objects.equals(row.get(KtmgmkkColumn.TKYT_YMD), COMPLEMENT_YMD)) {
            return ROW_MAPPER.toRecord(row);
        }
        EnumMap<KtmgmkkColumn, Object> normalized = new EnumMap<>(row);
        normalized.put(KtmgmkkColumn.TKYT_YMD, null);
        return ROW_MAPPER.toRecord(normalized);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String safeKnkkSksiJysb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, KNKK_SKSI_JYSB_LENGTH);
    }

    private static List<String> normalizeKnkkSksiJysbList(List<String> knkkSksiJysbList) {
        List<String> nonNullKnkkSksiJysbList =
                Objects.requireNonNull(knkkSksiJysbList, "knkkSksiJysbList");
        if (nonNullKnkkSksiJysbList.isEmpty()) {
            return List.of();
        }
        return nonNullKnkkSksiJysbList.stream()
                .map(KtmgmkkDataAccessImpl::normalizeInKnkkSksiJysb)
                .toList();
    }

    private static String normalizeInKnkkSksiJysb(String value) {
        return safeKnkkSksiJysb(value, "knkkSksiJysbList");
    }

    private static Long normalizeLong(Long value) {
        return WhereParamGuard.complement(value, "value", 0L);
    }

    private static Integer normalizeInteger(Integer value) {
        return WhereParamGuard.complement(value, "value", 0);
    }

    private static LocalDate normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(value, "value", COMPLEMENT_YMD);
    }

    private static String normalizeKnkkSksiJysb(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", KNKK_SKSI_JYSB_LENGTH);
    }

    /**
     * テスト用に保持行を返す。
     *
     * @return 保持行のレコードリスト
     */
    List<KtmgmkkRecord> rows() {
        return rows.stream().map(KtmgmkkDataAccessImpl::toRecord).toList();
    }
