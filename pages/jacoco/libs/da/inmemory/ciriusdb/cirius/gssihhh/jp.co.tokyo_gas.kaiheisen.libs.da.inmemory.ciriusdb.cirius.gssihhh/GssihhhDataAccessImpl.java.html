package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.gssihhh;

/** DA-006（C001/D001）の SQL 振る舞いを模した GSSIHHH インメモリ実装。 */
public class GssihhhDataAccessImpl extends DataAccessBase implements GssihhhDataAccess {

    private static final ReflectiveInMemoryRowMapper<GssihhhColumn, GssihhhRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(GssihhhColumn.class, GssihhhRecord.class);

    private static final AutoNeutralValueResolver<GssihhhColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(GssihhhColumn.class, GssihhhRecord.class);

    private final List<EnumMap<GssihhhColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<GssihhhWriteRecord, GssihhhColumn> insertSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public GssihhhDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    GssihhhDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    GssihhhDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        GssihhhWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public GssihhhDataAccessImpl addRow(GssihhhRecord record) {
        EnumMap<GssihhhColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, GssihhhWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(GssihhhWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                GssihhhColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteBySvkyNo(Long svkyNo, Integer recUpdJnjNo) {
        Objects.requireNonNull(svkyNo, "svkyNo");

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(row.get(GssihhhColumn.SVKY_NO), svkyNo),
                GssihhhColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                GssihhhDataAccess.Identifiers.DELETE_BY_SVKY_NO,
                () -> new DataInconsistencyException("GSSIHHH", GssihhhDataAccess.Identifiers.DELETE_BY_SVKY_NO, "{}"),
                svkyNo);
    }

    @Override
    public Optional<GssihhhRecord> findByPK(Long svkyNo) {
        Objects.requireNonNull(svkyNo, "svkyNo");

        return rows.stream()
                .filter(row -> Objects.equals(row.get(GssihhhColumn.SVKY_NO), svkyNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }
}
