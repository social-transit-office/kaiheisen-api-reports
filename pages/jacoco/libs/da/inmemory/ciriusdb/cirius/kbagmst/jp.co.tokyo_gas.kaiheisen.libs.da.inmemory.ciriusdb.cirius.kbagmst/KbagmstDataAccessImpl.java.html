package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.kbagmst;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kbagmst.KbagmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kbagmst.KbagmstRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;

/**
 * DA-066-R001 の SQL をインメモリで再現する実装。
 */
public class KbagmstDataAccessImpl extends DataAccessBase implements KbagmstDataAccess {

    private final List<KbagmstRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public KbagmstDataAccessImpl addRow(KbagmstRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<KbagmstRecord> findByGasMtstNoAndTkysYmAndTyenYm(Long gasMtstNo, YearMonth knskKjnYm) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(knskKjnYm, "knskKjnYm");

        KbagmstRecord latest = null;
        for (KbagmstRecord row : rows) {
            if (!gasMtstNo.equals(row.gasMtstNo())) {
                continue;
            }
            if (row.tkysYm() == null || row.tyenYm() == null) {
                continue;
            }
            if (row.tkysYm().compareTo(knskKjnYm) > 0) {
                continue;
            }
            if (row.tyenYm().compareTo(knskKjnYm) < 0) {
                continue;
            }
            if (latest == null || row.tkysYm().compareTo(latest.tkysYm()) > 0) {
                latest = row;
            }
        }
        return Optional.ofNullable(latest);
    }

    @Override
    public Optional<KbagmstRecord> findByPK(Long gasMtstNo, YearMonth tkysYm) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(tkysYm, "tkysYm");

        for (KbagmstRecord row : rows) {
            if (gasMtstNo.equals(row.gasMtstNo())
                    && tkysYm.equals(row.tkysYm())) {
                return Optional.of(row);
            }
        }
        return Optional.empty();
    }
}
