/** DA-060（C001/R001/U001）の SQL 振る舞いを模した KIYCSTK インメモリ実装。 */
public class KiycstkDataAccessImpl extends DataAccessBase implements KiycstkDataAccess {

    private static final ReflectiveInMemoryRowMapper<KiycstkColumn, KiycstkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KiycstkColumn.class, KiycstkRecord.class);

    private static final AutoNeutralValueResolver<KiycstkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KiycstkColumn.class, KiycstkRecord.class);

    private final List<EnumMap<KiycstkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KiycstkWriteRecord, KiycstkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KiycstkWriteRecord, KiycstkColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KiycstkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KiycstkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KiycstkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        KiycstkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KiycstkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public KiycstkDataAccessImpl addRow(KiycstkRecord record) {
        EnumMap<KiycstkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KiycstkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KiycstkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                KiycstkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<KiycstkRecord> findByKyoTrkNo(Long kyoTrkNo) {
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KiycstkColumn.KYO_TRK_NO), kyoTrkNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByKyoTrkNo(
            KiycstkWriteRecord record,
            Set<KiycstkColumn> targetColumns,
            KiycstkUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(row.get(KiycstkColumn.KYO_TRK_NO), updateKey.kyoTrkNo()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KiycstkColumn.class, values),
                                                context,
                                                clock,
                                                (Integer) row.get(KiycstkColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                KiycstkColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KiycstkDataAccess.Identifiers.UPDATE_BY_KYO_TRK_NO,
                updateKey.kyoTrkNo());
    }
}
