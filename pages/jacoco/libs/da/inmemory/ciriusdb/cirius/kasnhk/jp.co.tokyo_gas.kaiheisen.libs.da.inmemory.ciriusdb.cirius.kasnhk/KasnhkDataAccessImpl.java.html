/** DA-118（KASNHK）のインメモリ実装。 */
public class KasnhkDataAccessImpl extends DataAccessBase implements KasnhkDataAccess {

    private static final int GMB_LENGTH = 15;

    private static final int LVLT_ANKN_NO_LENGTH = 10;

    private static final int HUKK_SYS_SHBT_LENGTH = 1;

    private static final int KAS_HUKK_JTKB_LENGTH = 1;

    private static final ReflectiveInMemoryRowMapper<KasnhkColumn, KasnhkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KasnhkColumn.class, KasnhkRecord.class);

    private static final AutoNeutralValueResolver<KasnhkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KasnhkColumn.class, KasnhkRecord.class);

    private final List<EnumMap<KasnhkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KasnhkWriteRecord, KasnhkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KasnhkWriteRecord, KasnhkColumn>
            updateSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public KasnhkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KasnhkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KasnhkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(KasnhkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KasnhkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /** テストデータを 1 行追加する。 */
    public KasnhkDataAccessImpl addRow(KasnhkRecord record) {
        EnumMap<KasnhkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KasnhkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** 保持中の行を返す。 */
    public List<KasnhkRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    /** 1 行登録する。 */
    @Override
    public int insert(KasnhkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(KasnhkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

        /** 4条件と任意排他制御順序番号で一致行を削除する。 */
    @Override
    public int deleteByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNo(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            Integer recUpdJnjNo) {
        String safeGmb = safeGmb(gmb);
        LocalDate safeKasgYmd = safeYmd(kasgYmd, "kasgYmd");
        LocalDate safeKasHukkYmd = safeYmd(kasHukkYmd, "kasHukkYmd");
        String safeLvltAnknNo = safeLvltAnknNo(lvltAnknNo);
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> matchesBy4Conditions(row, safeGmb, safeKasgYmd, safeKasHukkYmd, safeLvltAnknNo),
                KasnhkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                KasnhkDataAccess.CallSpecs
                        .DELETE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_LVLT_ANKN_NO
                        .identifier(),
                () ->
                        new DataInconsistencyException(
                                "KASNHK",
                                KasnhkDataAccess.CallSpecs
                                        .DELETE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_LVLT_ANKN_NO
                                        .identifier(),
                                "{}"),
                safeGmb,
                safeKasgYmd,
                safeKasHukkYmd,
                safeLvltAnknNo);
    }

        /** 開栓報告状態区分を単一値で指定して 1 件取得する。 */
    @Override
    public Optional<KasnhkRecord> findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            String kasHukkJtkb) {
        String safeGmb = safeGmb(gmb);
        LocalDate safeKasgYmd = safeYmd(kasgYmd, "kasgYmd");
        LocalDate safeKasHukkYmd = safeYmd(kasHukkYmd, "kasHukkYmd");
        String safeLvltAnknNo = safeLvltAnknNo(lvltAnknNo);
        String safeKasHukkJtkb = safeKasHukkJtkb(kasHukkJtkb, "kasHukkJtkb");
        return rows.stream()
                .filter(
                        row ->
                                matchesBy4Conditions(
                                        row,
                                        safeGmb,
                                        safeKasgYmd,
                                        safeKasHukkYmd,
                                        safeLvltAnknNo))
                .filter(
                        row ->
                                Objects.equals(
                                        normalizeFixed(
                                                row.get(KasnhkColumn.KAS_HUKK_JTKB),
                                                KAS_HUKK_JTKB_LENGTH),
                                        safeKasHukkJtkb))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** 開栓報告状態区分を複数値で指定して 1 件取得する。 */
    @Override
    public Optional<KasnhkRecord> findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String lvltAnknNo,
            List<String> kasHukkJtkbList) {
        String safeGmb = safeGmb(gmb);
        LocalDate safeKasgYmd = safeYmd(kasgYmd, "kasgYmd");
        LocalDate safeKasHukkYmd = safeYmd(kasHukkYmd, "kasHukkYmd");
        String safeLvltAnknNo = safeLvltAnknNo(lvltAnknNo);
        List<String> safeKasHukkJtkbList = safeKasHukkJtkbList(kasHukkJtkbList);
        if (safeKasHukkJtkbList.isEmpty()) {
            return Optional.empty();
        }
        return rows.stream()
                .filter(
                        row ->
                                matchesBy4Conditions(
                                        row,
                                        safeGmb,
                                        safeKasgYmd,
                                        safeKasHukkYmd,
                                        safeLvltAnknNo))
                .filter(
                        row ->
                                safeKasHukkJtkbList.contains(
                                        normalizeFixed(
                                                row.get(KasnhkColumn.KAS_HUKK_JTKB),
                                                KAS_HUKK_JTKB_LENGTH)))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** 主キー検索の実処理を行う。 */
    @Override
    public Optional<KasnhkRecord> findByPrimaryKey(
            String gmb,
            LocalDate kasgYmd,
            LocalDate kasHukkYmd,
            String hukkSysShbt,
            String lvltAnknNo) {
        String safeGmb = safeGmb(gmb);
        LocalDate safeKasgYmd = safeYmd(kasgYmd, "kasgYmd");
        LocalDate safeKasHukkYmd = safeYmd(kasHukkYmd, "kasHukkYmd");
        String safeHukkSysShbt = safeHukkSysShbt(hukkSysShbt);
        String safeLvltAnknNo = safeLvltAnknNo(lvltAnknNo);
        return rows.stream()
                .filter(
                        row ->
                                matchesPk(
                                        row,
                                        safeGmb,
                                        safeKasgYmd,
                                        safeKasHukkYmd,
                                        safeHukkSysShbt,
                                        safeLvltAnknNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** 報告システム種別を含む条件の対象行を更新する。 */
    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndHukkSysShbtAndLvltAnknNo(
            KasnhkWriteRecord record,
            Set<KasnhkUpdatableColumn> targetColumns,
            KasnhkU001UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Set<KasnhkColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);
        String safeGmb = safeGmb(updateKey.gmb());
        LocalDate safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");
        LocalDate safeKasHukkYmd = safeYmd(updateKey.kasHukkYmd(), "kasHukkYmd");
        String safeHukkSysShbt = safeHukkSysShbt(updateKey.hukkSysShbt());
        String safeLvltAnknNo = safeLvltAnknNo(updateKey.lvltAnknNo());
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        matchesPk(
                                row,
                                safeGmb,
                                safeKasgYmd,
                                safeKasHukkYmd,
                                safeHukkSysShbt,
                                safeLvltAnknNo),
                row ->
                        updateSupport.prepare(
                                record,
                                canonicalTargetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(KasnhkColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? toRecUpdJnjNo(row.get(KasnhkColumn.REC_UPD_JNJ_NO))
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KasnhkColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KasnhkDataAccess.CallSpecs
                        .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_HUKK_SYS_SHBT_AND_LVLT_ANKN_NO_U001
                        .identifier(),
                safeGmb,
                safeKasgYmd,
                safeKasHukkYmd,
                safeHukkSysShbt,
                safeLvltAnknNo);
    }

        /** 4条件と任意排他で対象行を更新する。 */
    @Override
        public int updateByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndRecUpdJnjNo(
            KasnhkWriteRecord record,
            Set<KasnhkUpdatableColumn> targetColumns,
            KasnhkU002UpdateKey updateKey,

        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");

        Set<KasnhkColumn> canonicalTargetColumns = canonicalTargetColumns(targetColumns);

        LocalDate safeKasgYmd = safeYmd(updateKey.kasgYmd(), "kasgYmd");

    /** 主キー一致判定を行う。 */
    private static boolean matchesPk(
            Map<KasnhkColumn, Object> row,
            String safeGmb,
            LocalDate safeKasgYmd,
            LocalDate safeKasHukkYmd,
            String safeHukkSysShbt,
            String safeLvltAnknNo) {
        return Objects.equals(
                        normalizeFixed(row.get(KasnhkColumn.GMB), GMB_LENGTH),
                        safeGmb)
                && Objects.equals(normalizeYmd(row.get(KasnhkColumn.KASG_YMD)), safeKasgYmd)
                && Objects.equals(normalizeYmd(row.get(KasnhkColumn.KAS_HUKK_YMD)), safeKasHukkYmd)
                && Objects.equals(
                        normalizeFixed(row.get(KasnhkColumn.HUKK_SYS_SHBT), HUKK_SYS_SHBT_LENGTH),
                        safeHukkSysShbt)
                && Objects.equals(
                        normalizeFixed(
                                row.get(KasnhkColumn.LVLT_ANKN_NO),
                                LVLT_ANKN_NO_LENGTH),
                        safeLvltAnknNo);
    }

    /** HUKK_SYS_SHBT を含まない4条件一致判定を行う。 */
    private static boolean matchesBy4Conditions(
            Map<KasnhkColumn, Object> row,
            String safeGmb,
            LocalDate safeKasgYmd,
            LocalDate safeKasHukkYmd,
            String safeLvltAnknNo) {
        return Objects.equals(
                        normalizeFixed(row.get(KasnhkColumn.GMB), GMB_LENGTH),
                        safeGmb)
                && Objects.equals(normalizeYmd(row.get(KasnhkColumn.KASG_YMD)), safeKasgYmd)
                && Objects.equals(normalizeYmd(row.get(KasnhkColumn.KAS_HUKK_YMD)), safeKasHukkYmd)
                && Objects.equals(
                        normalizeFixed(
                                row.get(KasnhkColumn.LVLT_ANKN_NO),
                                LVLT_ANKN_NO_LENGTH),
                        safeLvltAnknNo);
    }

    /** 更新対象列を EnumSet に正規化する。 */
    private static Set<KasnhkColumn> canonicalTargetColumns(
            Set<KasnhkUpdatableColumn> targetColumns) {
        EnumSet<KasnhkColumn> canonicalColumns = EnumSet.noneOf(KasnhkColumn.class);
        for (KasnhkUpdatableColumn targetColumn : targetColumns) {
            canonicalColumns.add(KasnhkColumn.valueOf(targetColumn.name()));
        }
        return canonicalColumns;
    }

    /** REC_UPD_JNJ_NO を Integer に変換する。 */
    private static Integer toRecUpdJnjNo(Object value) {
        if (value == null) {
            return 0;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.valueOf(value.toString());
    }

    /** GMB を WHERE 用に正規化する。 */
    private static String safeGmb(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "gmb", GMB_LENGTH);
    }

    /** LIV@LIT 案件番号を WHERE 用に正規化する。 */
    private static String safeLvltAnknNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "lvltAnknNo", LVLT_ANKN_NO_LENGTH);
    }

    /** 報告システム種別を WHERE 用に正規化する。 */
    private static String safeHukkSysShbt(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "hukkSysShbt", HUKK_SYS_SHBT_LENGTH);
    }

    /** 開栓報告状態区分を WHERE 用に正規化する。 */
    private static String safeKasHukkJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, KAS_HUKK_JTKB_LENGTH);
    }

    /** 開栓報告状態区分リストを WHERE 用に正規化する。 */
    private static List<String> safeKasHukkJtkbList(List<String> values) {
        if (values == null) {
            throw new IllegalArgumentException("kasHukkJtkbList must not be null");
        }
        if (values.isEmpty()) {
            return List.of();
        }
        return values.stream().map(KasnhkDataAccessImpl::normalizeInKasHukkJtkb).toList();
    }

    /** IN 句要素の開栓報告状態区分を固定長条件値へ正規化する。 */
    private static String normalizeInKasHukkJtkb(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "kasHukkJtkbList must not contain null or blank-only");
        }
        return safeKasHukkJtkb(value, "kasHukkJtkbList");
    }

    /** 年月日を WHERE 用に正規化する。 */
    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, LocalDate.MIN);
    }


    /** 固定長文字列列を比較用に正規化する。 */
    private static String normalizeFixed(Object value, int length) {
        return WhereParamGuard.fixedLengthNotNull((String) value, "value", length);
    }

    /** 年月日列を比較用に正規化する。 */
    private static LocalDate normalizeYmd(Object value) {
        return WhereParamGuard.complement((LocalDate) value, "value", LocalDate.MIN);
    }

    /** 数値列を比較用に正規化する。 */
    private static Integer normalizeInteger(Object value) {
        if (value == null) {
            return 0;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.valueOf(value.toString());
    }
