package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.isrkd;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-101-R001 の SQL をインメモリで再現する実装。 */
public class IsrkdDataAccessImpl extends DataAccessBase implements IsrkdDataAccess {

    private static final int DUKN_ISRS_CD_LENGTH = 8;

    private final List<IsrkdRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public IsrkdDataAccessImpl addRow(IsrkdRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<IsrkdRecord> findByArtyCstNoAndIsrrTriYmdAndDuknIsrsCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> duknIsrsCdList) {
        List<String> safeDuknIsrsCdList = normalizeDuknIsrsCdList(duknIsrsCdList);
        if (safeDuknIsrsCdList.isEmpty()) {
            return Optional.empty();
        }
        Long safeArtyCstNo = WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L);
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                toYmd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);

        return rows.stream()
                .filter(row -> normalizeLong(row.artyCstNo()).equals(safeArtyCstNo))
                .filter(row -> normalizeYmd(row.isrrTriYmd()).equals(safeIsrrTriYmd))
                .filter(row -> safeDuknIsrsCdList.contains(normalizeCode(row.duknIsrsCd())))
                .sorted(
                        Comparator.comparing(
                                        IsrkdDataAccessImpl::normalizeTsno,
                                        Comparator.nullsLast(Integer::compareTo))
                                .reversed())
                .findFirst()
                .map(IsrkdDataAccessImpl::normalizeRecord);
    }

    @Override
    public Optional<IsrkdRecord> findByPK(Long artyCstNo, LocalDate isrrTriYmd, Integer isrrKytuTsno) {
        Long safeArtyCstNo = WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L);
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(
                    toYmd(isrrTriYmd),
                        "isrrTriYmd",
                        BigDecimal.ZERO);
        Integer safeIsrrKytuTsno = WhereParamGuard.complement(isrrKytuTsno, "isrrKytuTsno", 0);

        return rows.stream()
                .filter(row -> normalizeLong(row.artyCstNo()).equals(safeArtyCstNo))
                .filter(row -> normalizeYmd(row.isrrTriYmd()).equals(safeIsrrTriYmd))
                .filter(row -> normalizeTsno(row).equals(safeIsrrKytuTsno))
                .findFirst()
                .map(IsrkdDataAccessImpl::normalizeRecord);
    }

    /**
     * IN 句検索用の業務詳細コード一覧を正規化する。
     *
     * <p>主な仕様: null の一覧は拒否し、空一覧はそのまま返す。
     * 各要素は null/空白-only を入力不正として検証した後、固定長 8 文字へ整形する。
     *
     * <p>既知の制約: 要素に null または空白-only を含む一覧は入力不正として扱う。
     */
    private static List<String> normalizeDuknIsrsCdList(List<String> duknIsrsCdList) {
        if (duknIsrsCdList == null) {
            throw new IllegalArgumentException("duknIsrsCdList must not be null");
        }
        if (duknIsrsCdList.isEmpty()) {
            return List.of();
        }
        return duknIsrsCdList.stream().map(IsrkdDataAccessImpl::normalizeInDuknIsrsCd).toList();
    }

    /**
     * IN 句要素の業務詳細コードを固定長条件値へ正規化する。
     *
     * <p>主な仕様: null と空白-only を入力不正として拒否した上で、fixedLengthNotNull により固定長 8 文字へ整形する。
     *
     * <p>既知の制約: 8 文字超の値は切り詰めず、そのまま返す。
     */
    private static String normalizeInDuknIsrsCd(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "duknIsrsCdList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "duknIsrsCdList", DUKN_ISRS_CD_LENGTH);
    }

    /**
     * 顧客番号条件を補完付きで正規化する。
     *
     * <p>主な仕様: null は 0 に補完する。
     *
     * <p>既知の制約: 条件項目名は固定で value として扱う。
     */
    private static Long normalizeLong(Long value) {
        return WhereParamGuard.complement(value, "value", 0L);
    }

    /**
     * 通し番号を補完付きで正規化する。
     *
     * <p>主な仕様: null は 0 に補完する。
     *
     * <p>既知の制約: レコード本体から通し番号を取得して評価する。
     */
    private static Integer normalizeTsno(IsrkdRecord row) {
        return WhereParamGuard.complement(row.isrrKytuTsno(), "isrrKytuTsno", 0);
    }

    /**
     * 年月日条件を DB 数値形式へ正規化する。
     *
     * <p>主な仕様: LocalDate を yyyymmdd 数値へ変換し、null は 0 に補完する。
     *
     * <p>既知の制約: DB 側比較に合わせて BigDecimal で返す。
     */
    private static BigDecimal normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(toYmd(value), "value", BigDecimal.ZERO);
    }

    /**
     * 年月日を DB 数値形式へ変換する。
     *
     * <p>主な仕様: LocalDate を yyyymmdd 形式の BigDecimal へ変換する。
     *
     * <p>既知の制約: null は null のまま返す。
     */
    private static BigDecimal toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        return BigDecimal.valueOf(value.getYear() * 10000L + value.getMonthValue() * 100L + value.getDayOfMonth());
    }

    /**
     * 業務詳細コードを固定長条件値へ正規化する。
     *
     * <p>主な仕様: 固定長 8 文字の NOT NULL 条件値へ補完する。
     *
     * <p>既知の制約: 入力値が null または空白-only の場合は半角スペース 1 文字へ補完される。
     */
    private static String normalizeCode(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "duknIsrsCd", DUKN_ISRS_CD_LENGTH);
    }

    /**
     * 取得結果レコードを API 公開値へ正規化する。
     *
     * <p>主な仕様: 固定長文字列の後続空白を除去し、数値 0 を null 相当へ変換して取得結果を API 公開表現へそろえる。
     *
     * <p>既知の制約: 日付・日時項目は入力値をそのまま引き継ぐ。
     */
    private static IsrkdRecord normalizeRecord(IsrkdRecord row) {
        return new IsrkdRecord(
                normalizeResultLong(row.artyCstNo()),
                row.isrrTriYmd(),
                normalizeResultInteger(row.isrrKytuTsno()),
                normalizeResultLong(row.kyoTrkNo()),
                normalizeResultLong(row.gasMtstNo()),
                normalizeResultLong(row.tksuSvkyNo()),
                suppress(row.akrkKbn()),
                suppress(row.akrkTrskShbt()),
                suppress(row.duknIsrgCd()),
                suppress(row.duknIsrsCd()),
                suppress(row.isrrHkkrKbn()),
                suppress(row.isrrSktCd()),
                suppress(row.kyonKj1()),
                suppress(row.kyonKn1()),
                suppress(row.kyon1HujnShbt()),
                suppress(row.kyon1HshiShbt()),
                suppress(row.kyonKj2()),
                suppress(row.kyonKn2()),
                suppress(row.kyon2HujnShbt()),
                suppress(row.kyon2HshiShbt()),
                row.kyonKjhkFlg(),
                normalizeResultInteger(row.isrrTriHms()),
                row.idsgYmd(),
                suppress(row.isrrHusrId()),
                suppress(row.isrrHkunKj()),
                suppress(row.isrrHjsgId()),
                row.regTsmp(),
                suppress(row.regUsrId()),
                row.updTsmp(),
                suppress(row.updUsrId()),
                suppress(row.updKtnKinoId()),
                normalizeResultInteger(row.recUpdJnjNo()));
    }

    /**
     * 取得結果の Long 値を API 公開値へ正規化する。
     *
     * <p>主な仕様: 0 は DB 補完値として扱い、null へ変換する。
     *
     * <p>既知の制約: 0 以外の値はそのまま返す。
     */
    private static Long normalizeResultLong(Long value) {
        return value != null && value == 0L ? null : value;
    }

    /**
     * 取得結果の Integer 値を API 公開値へ正規化する。
     *
     * <p>主な仕様: 0 は DB 補完値として扱い、null へ変換する。
     *
     * <p>既知の制約: 0 以外の値はそのまま返す。
     */
    private static Integer normalizeResultInteger(Integer value) {
        return value != null && value == 0 ? null : value;
    }

    /**
     * 固定長文字列の後続空白を抑止する。
     *
     * <p>主な仕様: API 返却値として不要な右側空白を除去する。
     *
     * <p>既知の制約: null は null のまま返す。
     */
    private static String suppress(String value) {
        return FixedLengthFormatter.suppressSpace(value);
    }
}
