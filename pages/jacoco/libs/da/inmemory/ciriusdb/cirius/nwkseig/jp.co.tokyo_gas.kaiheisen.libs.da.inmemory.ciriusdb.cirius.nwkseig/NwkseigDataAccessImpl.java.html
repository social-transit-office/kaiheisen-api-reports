package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.nwkseig;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.nwkseig.NwkseigDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.nwkseig.NwkseigRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** 営業情報ＮＷ箇所マスター（NWKSEIG）のインメモリ実装。 */
public class NwkseigDataAccessImpl extends DataAccessBase implements NwkseigDataAccess {

    private static final int NWK_CD_LENGTH = 7;

    private final List<NwkseigRecord> rows = new ArrayList<>();

    /**
     * テストデータを 1 行追加する。
     *
     * @param row 追加行
     * @return このインスタンス
     */
    public NwkseigDataAccessImpl addRow(NwkseigRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<NwkseigRecord> findByNwkCdAndTkysYmdAndTyenYmd(
            String nwkCd, LocalDate knskKjnYmd) {
        String safeNwkCd = safeNwkCd(nwkCd, "nwkCd");
        int safeKnskKjnYmd = ymdOrZero(knskKjnYmd);
        return rows.stream()
                .filter(row -> normalizeFixedLength(row.nwkCd(), NWK_CD_LENGTH).equals(safeNwkCd))
                .filter(row -> ymdOrZero(row.tkysYmd()) <= safeKnskKjnYmd)
                .filter(row -> ymdOrZero(row.tyenYmd()) >= safeKnskKjnYmd)
                .sorted(Comparator.comparingInt((NwkseigRecord row) -> ymdOrZero(row.tkysYmd())).reversed())
                .findFirst();
    }

    @Override
    public Optional<NwkseigRecord> findByPK(String nwkCd, LocalDate tkysYmd) {
        String safeNwkCd = safeNwkCd(nwkCd, "nwkCd");
        int safeTkysYmd = ymdOrZero(tkysYmd);
        return rows.stream()
                .filter(row -> normalizeFixedLength(row.nwkCd(), NWK_CD_LENGTH).equals(safeNwkCd))
                .filter(row -> ymdOrZero(row.tkysYmd()) == safeTkysYmd)
                .findFirst();
    }

    private static String safeNwkCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, NWK_CD_LENGTH);
    }

    private static String normalizeFixedLength(String value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", length);
    }

    private static int ymdOrZero(LocalDate value) {
        if (value == null) {
            return 0;
        }
        return Integer.parseInt(value.format(java.time.format.DateTimeFormatter.BASIC_ISO_DATE));
    }
}
