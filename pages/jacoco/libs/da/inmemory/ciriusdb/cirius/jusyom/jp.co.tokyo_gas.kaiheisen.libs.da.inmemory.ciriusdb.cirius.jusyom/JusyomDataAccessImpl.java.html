package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.jusyom;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.jusyom.JusyomDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.jusyom.JusyomRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-002-R001 の SQL をインメモリで再現する実装。 */
public class JusyomDataAccessImpl extends DataAccessBase implements JusyomDataAccess {

    private static final int GKK_CD_LENGTH = 5;

    private static final int MCH_CD_LENGTH = 3;

    private static final int CMAZ_CD_LENGTH = 3;

    private final List<JusyomRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public JusyomDataAccessImpl addRow(JusyomRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    /**
     * 保持行を参照する。
     *
     * @return 保持行
     */
    public List<JusyomRecord> rows() {
        return List.copyOf(rows);
    }

    @Override
    public Optional<JusyomRecord> findByGkkCdAndMchCdAndCmazCd(
            String gkkCd,
            String mchCd,
            String cmazCd) {
        String safeGkkCd = WhereParamGuard.fixedLengthNotNull(gkkCd, "gkkCd", GKK_CD_LENGTH);
        String safeMchCd = WhereParamGuard.fixedLengthNotNull(mchCd, "mchCd", MCH_CD_LENGTH);
        String safeCmazCd = WhereParamGuard.fixedLengthNotNull(cmazCd, "cmazCd", CMAZ_CD_LENGTH);

        return rows.stream()
                .filter(row -> normalizeFixed(row.gkkCd(), GKK_CD_LENGTH).equals(safeGkkCd))
                .filter(row -> normalizeFixed(row.mchCd(), MCH_CD_LENGTH).equals(safeMchCd))
                .filter(row -> normalizeFixed(row.cmazCd(), CMAZ_CD_LENGTH).equals(safeCmazCd))
                .findFirst();
    }

    private static String normalizeFixed(String value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", length);
    }
}
