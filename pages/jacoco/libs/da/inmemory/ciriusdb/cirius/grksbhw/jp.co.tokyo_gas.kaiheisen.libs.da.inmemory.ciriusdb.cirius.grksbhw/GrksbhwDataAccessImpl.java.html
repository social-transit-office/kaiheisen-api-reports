package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.grksbhw;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.grksbhw.GrksbhwRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-161 の SQL をインメモリで再現する実装。 */
public class GrksbhwDataAccessImpl extends DataAccessBase implements GrksbhwDataAccess {

    private static final int GRK_KSB_HWSB_LENGTH = 4;

    private static final int GRK_KIY_SHBT_LENGTH = 5;

    private static final int RKN_CHK_CD_LENGTH = 1;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private static final Comparator<GrksbhwRecord> ORDER_BY_R001 =
            Comparator.comparing(
                            (GrksbhwRecord row) -> normalizeFixed(row.grkKsbHwsb(), GRK_KSB_HWSB_LENGTH))
                    .thenComparing(row -> normalizeFixed(row.grkKiyShbt(), GRK_KIY_SHBT_LENGTH))
                    .thenComparing(row -> normalizeFixed(row.rknChkCd(), RKN_CHK_CD_LENGTH))
                    .thenComparing(
                            (GrksbhwRecord row) -> normalizeDate(row.tkysYmd()),
                            Comparator.reverseOrder());

    private static final Comparator<GrksbhwRecord> ORDER_BY_TKYS_YMD_DESC =
            Comparator.comparing(
                    (GrksbhwRecord row) -> normalizeDate(row.tkysYmd()),
                    Comparator.reverseOrder());

    private final List<GrksbhwRecord> rows = new ArrayList<>();

    /**
     * テスト用の行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GrksbhwDataAccessImpl addRow(GrksbhwRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public List<GrksbhwRecord> findAllByGrkKsbHwsbListAndGrkKiyShbtAndKnskKjnYmd(
            List<String> grkKsbHwsbList,
            String grkKiyShbt,
            LocalDate knskKjnYmd) {
        List<String> safeGrkKsbHwsbList = normalizeGrkKsbHwsbList(grkKsbHwsbList);
        if (safeGrkKsbHwsbList.isEmpty()) {
            return List.of();
        }
        String safeGrkKiyShbt = safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        return rows.stream()
                .filter(
                        row ->
                                safeGrkKsbHwsbList.contains(
                                        normalizeFixed(
                                                row.grkKsbHwsb(), GRK_KSB_HWSB_LENGTH)))
                .filter(
                        row ->
                                normalizeFixed(row.grkKiyShbt(), GRK_KIY_SHBT_LENGTH)
                                        .equals(safeGrkKiyShbt))
                .filter(row -> !normalizeDate(row.tkysYmd()).isAfter(safeKnskKjnYmd))
                .filter(row -> !normalizeDate(row.tyenYmd()).isBefore(safeKnskKjnYmd))
                .sorted(ORDER_BY_R001)
                .map(GrksbhwDataAccessImpl::normalizeRecord)
                .toList();
    }

    @Override
    public Optional<GrksbhwRecord> findByGrkKiyShbtAndGrkKsbHwsbAndRknChkCdAndKnskKjnYmdOrderByTkysYmdDesc(
            String grkKiyShbt,
            String grkKsbHwsb,
            String rknChkCd,
            LocalDate knskKjnYmd) {
        String safeGrkKiyShbt = safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt");
        String safeGrkKsbHwsb = safeGrkKsbHwsb(grkKsbHwsb, "grkKsbHwsb");
        String safeRknChkCd = safeRknChkCd(rknChkCd, "rknChkCd");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        return rows.stream()
            .filter(row ->
                normalizeFixed(row.grkKiyShbt(), GRK_KIY_SHBT_LENGTH)
                    .equals(safeGrkKiyShbt)
            )
            .filter(row ->
                normalizeFixed(row.grkKsbHwsb(), GRK_KSB_HWSB_LENGTH)
                    .equals(safeGrkKsbHwsb)
            )
            .filter(row ->
                normalizeFixed(row.rknChkCd(), RKN_CHK_CD_LENGTH)
                    .equals(safeRknChkCd)
            )
            .filter(row ->
                !normalizeDate(row.tkysYmd())
                    .isAfter(safeKnskKjnYmd)
            )
            .filter(row ->
                !normalizeDate(row.tyenYmd())
                    .isBefore(safeKnskKjnYmd)
            )
            .sorted(ORDER_BY_TKYS_YMD_DESC)
            .map(GrksbhwDataAccessImpl::normalizeRecord)
            .findFirst();
    }

    @Override
    public Optional<GrksbhwRecord> findByPK(
            String grkKsbHwsb,
            String grkKiyShbt,
            String rknChkCd,
            LocalDate tkysYmd) {
        String safeGrkKsbHwsb = safeGrkKsbHwsb(grkKsbHwsb, "grkKsbHwsb");
        String safeGrkKiyShbt = safeGrkKiyShbt(grkKiyShbt, "grkKiyShbt");
        String safeRknChkCd = safeRknChkCd(rknChkCd, "rknChkCd");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        return rows.stream()
                .filter(row -> normalizeFixed(row.grkKsbHwsb(), GRK_KSB_HWSB_LENGTH).equals(safeGrkKsbHwsb))
                .filter(row -> normalizeFixed(row.grkKiyShbt(), GRK_KIY_SHBT_LENGTH).equals(safeGrkKiyShbt))
                .filter(row -> normalizeFixed(row.rknChkCd(), RKN_CHK_CD_LENGTH).equals(safeRknChkCd))
                .filter(row -> normalizeDate(row.tkysYmd()).equals(safeTkysYmd))
                .map(GrksbhwDataAccessImpl::normalizeRecord)
                .findFirst();
    }

    private static String safeGrkKsbHwsb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GRK_KSB_HWSB_LENGTH);
    }

    private static String safeGrkKiyShbt(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, GRK_KIY_SHBT_LENGTH);
    }

    private static String safeRknChkCd(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, RKN_CHK_CD_LENGTH);
    }

    private static List<String> normalizeGrkKsbHwsbList(List<String> grkKsbHwsbList) {
        if (grkKsbHwsbList == null) {
            throw new IllegalArgumentException("grkKsbHwsbList must not be null");
        }
        if (grkKsbHwsbList.isEmpty()) {
            return List.of();
        }
        return grkKsbHwsbList.stream().map(GrksbhwDataAccessImpl::normalizeInGrkKsbHwsb).toList();
    }

    private static String normalizeInGrkKsbHwsb(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "grkKsbHwsbList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "grkKsbHwsbList", GRK_KSB_HWSB_LENGTH);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String normalizeFixed(String value, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, "value", length);
    }

    private static LocalDate normalizeDate(LocalDate value) {
        return WhereParamGuard.complement(value, "value", COMPLEMENT_YMD);
    }

    private static GrksbhwRecord normalizeRecord(GrksbhwRecord row) {
        return new GrksbhwRecord(
                normalizeResultText(row.grkKsbHwsb()),
                normalizeResultText(row.grkKiyShbt()),
                normalizeResultText(row.rknChkCd()),
                normalizeResultDate(row.tkysYmd()),
                normalizeResultDate(row.tyenYmd()),
                normalizeResultText(row.grkKsbHnsb()),
                row.regTsmp(),
                normalizeResultText(row.regUsrId()),
                row.updTsmp(),
                normalizeResultText(row.updUsrId()),
                normalizeResultText(row.updKtnKinoId()),
                normalizeResultInteger(row.recUpdJnjNo()));
    }

    private static String normalizeResultText(String value) {
        if (value == null) {
            return null;
        }
        return FixedLengthFormatter.suppressSpace(value);
    }

    private static LocalDate normalizeResultDate(LocalDate value) {
        if (value == null || COMPLEMENT_YMD.equals(value)) {
            return null;
        }
        return value;
    }

    private static Integer normalizeResultInteger(Integer value) {
        if (value == null || value == 0) {
            return null;
        }
        return value;
    }
}
