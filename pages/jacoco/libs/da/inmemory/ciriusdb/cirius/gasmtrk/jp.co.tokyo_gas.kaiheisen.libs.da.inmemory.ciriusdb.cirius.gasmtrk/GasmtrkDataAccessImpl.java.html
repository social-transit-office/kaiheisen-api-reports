package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.gasmtrk;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.domain.shared.time.YearMonth;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;

/**
 * DA-026 の SQL を模したインメモリ実装。
 */
public class GasmtrkDataAccessImpl extends DataAccessBase implements GasmtrkDataAccess {

    private static final Comparator<String> GMB_COMPARATOR = (left, right) -> {
        boolean leftAscii = isAsciiOnly(left);
        boolean rightAscii = isAsciiOnly(right);
        if (leftAscii != rightAscii) {
            return leftAscii ? 1 : -1;
        }
        return left.compareTo(right);
    };

    private static final Comparator<InternalRow> ORDER_BY_DA026_R002 = Comparator
        .comparingInt(InternalRow::gmtkYmd).reversed()
        .thenComparingInt(InternalRow::gmtThkgYmd)
        .thenComparing(Comparator.comparingInt(InternalRow::gmtThkgJysb).reversed())
        .thenComparing(InternalRow::gmb, GMB_COMPARATOR)
        .thenComparingInt(InternalRow::gmbTsno);

    private static final Comparator<InternalRow> ORDER_BY_DA026_R003 = Comparator
        .comparingInt(InternalRow::gmtkYmd).reversed()
        .thenComparing(Comparator.comparingInt(InternalRow::gmtThkgYmd).reversed())
        .thenComparing(Comparator.comparingInt(InternalRow::gmtThkgJysb).reversed())
        .thenComparing(InternalRow::gmb, GMB_COMPARATOR)
        .thenComparingInt(InternalRow::gmbTsno);

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GasmtrkDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    private static boolean isAsciiOnly(String value) {
        for (int i = 0; i < value.length(); i++) {
            if (value.charAt(i) > 0x7F) {
                return false;
            }
        }
        return true;
    }

    @Override
    public List<GasmtrkRecord> findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(Long ttkGmsbNo, LocalDate knskKjnYmd) {
        Objects.requireNonNull(ttkGmsbNo, "ttkGmsbNo");
        Objects.requireNonNull(knskKjnYmd, "knskKjnYmd");
        int baseDateValue = toYmdInt(knskKjnYmd);
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == ttkGmsbNo)
                .filter(row -> row.gmtkYmd() <= baseDateValue)
                .filter(row -> row.gmtThkgYmd() == 0 || row.gmtThkgYmd() >= baseDateValue)
                .sorted(ORDER_BY_DA026_R002)
                .map(GasmtrkDataAccessImpl::toRecord)
                .toList();
    }

    /**
     * 取付ガスメーター設置場所番号とガスメーター取外年月日で一覧取得する。
     *
     * @param ttkGmsbNo  取付ガスメーター設置場所番号
     * @param gmtThkgYmd ガスメーター取外年月日
     * @return 条件に合致するレコード一覧
     */
    @Override
    public List<GasmtrkRecord> findAllByTtkGmsbNoAndGmtThkgYmd(Long ttkGmsbNo, LocalDate gmtThkgYmd) {
        Objects.requireNonNull(ttkGmsbNo, "ttkGmsbNo");
        int gmtThkgYmdValue = gmtThkgYmd == null ? 0 : toYmdInt(gmtThkgYmd);
        return rows.stream()
                .filter(row -> row.ttkGmsbNo() == ttkGmsbNo)
                .filter(row -> row.gmtThkgYmd() == gmtThkgYmdValue)
                .map(GasmtrkDataAccessImpl::toRecord)
                .toList();
    }

    @Override
    public List<GasmtrkRecord> findAllByGmbAndGmtkYmdAndGmtThkgYmd(String gmb, LocalDate knskKjnYmd) {
        Objects.requireNonNull(gmb, "gmb");
        Objects.requireNonNull(knskKjnYmd, "knskKjnYmd");
        int baseDateValue = toYmdInt(knskKjnYmd);
        return rows.stream()
                .filter(row -> gmb.equals(row.gmb()))
                .filter(row -> row.gmtkYmd() <= baseDateValue)
                .filter(row -> row.gmtThkgYmd() == 0 || row.gmtThkgYmd() >= baseDateValue)
                .sorted(ORDER_BY_DA026_R003)
                .map(GasmtrkDataAccessImpl::toRecord)
                .toList();
    }

    private static GasmtrkRecord toRecord(InternalRow row) {
        return new GasmtrkRecord(
                row.gmb(),
                row.gmbTsno(),
                row.ttkGmsbNo(),
                toLocalDate(row.gmtkYmd()),
                toLocalDate(row.gmtThkgYmd()),
                "0",
                "00",
                String.valueOf(row.gmtThkgJysb()),
                "",
                "",
                "",
                toYearMonth(row.ktyYm()),
                row.gmkShbt(),
                row.gmtGusu(),
                BigDecimal.ZERO,
                0,
                row.sjsuMincSu(),
                toLong(row.sjskHticSu()),
                "",
                "",
                "",
                "",
                "",
                "",
                toLocalDate(0),
                "",
                "",
                "",
                "",
                "",
                "",
                "",
                LocalDateTime.of(1970, 1, 1, 0, 0),
                "",
                LocalDateTime.of(1970, 1, 1, 0, 0),
                "",
                "",
                0
        );
    }

    /**
     * GASMTRK のテストデータ行。
     */
    public static record InternalRow(
            String gmb,
            long ttkGmsbNo,
            String gmkShbt,
            BigDecimal gmtGusu,
            int ktyYm,
            BigDecimal sjsuMincSu,
            BigDecimal sjskHticSu,
            int gmtkYmd,
            int gmtThkgYmd,
            int gmtThkgJysb,
            int gmbTsno
    ) {
        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(gmb, "gmb");
            Objects.requireNonNull(gmkShbt, "gmkShbt");
            Objects.requireNonNull(gmtGusu, "gmtGusu");
            Objects.requireNonNull(sjsuMincSu, "sjsuMincSu");
            Objects.requireNonNull(sjskHticSu, "sjskHticSu");
        }
    }

    @Override
    public Optional<GasmtrkRecord> findByPK(String gmb, Integer gmbTsno) {
        Objects.requireNonNull(gmb, "gmb");
        Objects.requireNonNull(gmbTsno, "gmbTsno");

        for (InternalRow row : rows) {
            if (gmb.equals(row.gmb())
                    && gmbTsno == row.gmbTsno()) {
                return Optional.of(toRecord(row));
            }
        }
        return Optional.empty();
    }

    private static int toYmdInt(LocalDate date) {
        return date.getYear() * 10000 + date.getMonthValue() * 100 + date.getDayOfMonth();
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }

    private static YearMonth toYearMonth(int ym) {
        if (ym <= 0) {
            return null;
        }
        return YearMonth.of(ym / 100, ym % 100);
    }

    private static Long toLong(BigDecimal value) {
        if (value == null) {
            return null;
        }
        return value.longValue();
    }
}
