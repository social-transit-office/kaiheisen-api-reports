package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.kyotlrd;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrd.KyotlrdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrd.KyotlrdRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrd.write.KyotlrdColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrd.write.KyotlrdWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.kyotlrd.write.KyotlrdWriteRecord;

import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.ReflectiveInMemoryRowMapper;

/** DA-166（KYOTLRD）のインメモリ実装。 */
public class KyotlrdDataAccessImpl extends DataAccessBase implements KyotlrdDataAccess {

    private static final int KYO_TEL_NO_LENGTH = 13;

    private static final ReflectiveInMemoryRowMapper<KyotlrdColumn, KyotlrdRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KyotlrdColumn.class, KyotlrdRecord.class);

    private static final AutoNeutralValueResolver<KyotlrdColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KyotlrdColumn.class, KyotlrdRecord.class);

    private static final Comparator<Map<KyotlrdColumn, Object>> LATEST_ORDER =
            Comparator.<Map<KyotlrdColumn, Object>, Integer>comparing(
                            row -> (Integer) row.get(KyotlrdColumn.TEL_NO_TSNO),
                            Comparator.nullsLast(Comparator.reverseOrder()));

    private final List<EnumMap<KyotlrdColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KyotlrdWriteRecord, KyotlrdColumn> insertSupport;

    /** デフォルト依存でインスタンスを生成する。 */
    public KyotlrdDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KyotlrdDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KyotlrdDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(KyotlrdWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * 初期データとして 1 行追加する。
     *
     * @param record 追加対象レコード
     * @return 自身
     */
    public KyotlrdDataAccessImpl addRow(KyotlrdRecord record) {
        EnumMap<KyotlrdColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KyotlrdWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KyotlrdWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(KyotlrdColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<KyotlrdRecord> findLatestByKyoTelNoAndKyoTrkNo(String kyoTelNo, Long kyoTrkNo) {
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        return rows.stream()
                .filter(row -> matchesKyoTelNo(row, safeKyoTelNo))
                .filter(row -> Objects.equals(row.get(KyotlrdColumn.KYO_TRK_NO), safeKyoTrkNo))
                .sorted(LATEST_ORDER)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<KyotlrdRecord> findByPK(String kyoTelNo, Long kyoTrkNo, Integer telNoTsno) {
        String safeKyoTelNo = safeKyoTelNo(kyoTelNo);
        Long safeKyoTrkNo = safeLong(kyoTrkNo, "kyoTrkNo");
        Integer safeTelNoTsno = safeInteger(telNoTsno, "telNoTsno");
        return rows.stream()
                .filter(row -> matchesKyoTelNo(row, safeKyoTelNo))
                .filter(row -> Objects.equals(row.get(KyotlrdColumn.KYO_TRK_NO), safeKyoTrkNo))
                .filter(row -> Objects.equals(row.get(KyotlrdColumn.TEL_NO_TSNO), safeTelNoTsno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private static boolean matchesKyoTelNo(Map<KyotlrdColumn, Object> row, String safeKyoTelNo) {
        return WhereParamGuard.fixedLengthNotNull(
                (String) row.get(KyotlrdColumn.KYO_TEL_NO),
                    "kyoTelNo",
                    KYO_TEL_NO_LENGTH)
                .equals(safeKyoTelNo);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static String safeKyoTelNo(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "kyoTelNo", KYO_TEL_NO_LENGTH);
    }
}
