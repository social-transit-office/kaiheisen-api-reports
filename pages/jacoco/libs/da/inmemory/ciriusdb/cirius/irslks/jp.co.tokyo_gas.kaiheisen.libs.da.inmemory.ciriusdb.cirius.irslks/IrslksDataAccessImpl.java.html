package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.irslks;

import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.DefaultSystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.irslks.IrslksDataAccess;

import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.ReflectiveInMemoryRowMapper;

/** DA-025（C001）の振る舞いを模した IRSLKS インメモリ実装。 */
public class IrslksDataAccessImpl extends DataAccessBase implements IrslksDataAccess {

    private static final ReflectiveInMemoryRowMapper<IrslksColumn, IrslksRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(IrslksColumn.class, IrslksRecord.class);

    private static final AutoNeutralValueResolver<IrslksColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(IrslksColumn.class, IrslksRecord.class);

    private final List<EnumMap<IrslksColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<IrslksWriteRecord, IrslksColumn> insertSupport;

    private int nextIdentityValue = 1;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public IrslksDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    IrslksDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    IrslksDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(IrslksWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public IrslksDataAccessImpl addRow(IrslksRecord record) {
        EnumMap<IrslksColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, List.of(IrslksColumn.values()));
        rows.add(values);
        if (record.isrkLogTsno() != null) {
            nextIdentityValue = Math.max(nextIdentityValue, record.isrkLogTsno() + 1);
        }
        return this;
    }

    @Override
    public int insert(IrslksWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values -> {
                            values.put(IrslksColumn.ISRK_LOG_TSNO, nextIdentityValue++);
                            auditColumnAutoSetter.applyForInsert(
                                    new MapBackedAuditColumnAccess<>(IrslksColumn.class, values),
                                    context,
                                    clock);
                        }));
        return 1;
    }

    @Override
    public Optional<IrslksRecord> findByPK(Integer isrkLogTsno) {
        Objects.requireNonNull(isrkLogTsno, "isrkLogTsno");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(IrslksColumn.ISRK_LOG_TSNO), isrkLogTsno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }
}
