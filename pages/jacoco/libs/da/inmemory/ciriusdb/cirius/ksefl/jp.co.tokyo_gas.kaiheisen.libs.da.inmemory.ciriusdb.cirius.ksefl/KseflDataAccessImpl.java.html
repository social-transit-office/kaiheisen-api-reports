/** DA-139（C001/R001-R005/U001/U002）の SQL 振る舞いを模した KSEFL インメモリ実装。 */
public class KseflDataAccessImpl extends DataAccessBase implements KseflDataAccess {

    private static final DateTimeFormatter YMD_FORMAT = DateTimeFormatter.BASIC_ISO_DATE;

    private static final ReflectiveInMemoryRowMapper<KseflColumn, KseflRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(KseflColumn.class, KseflRecord.class);

    private static final AutoNeutralValueResolver<KseflColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(KseflColumn.class, KseflRecord.class);

    private final List<EnumMap<KseflColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<KseflWriteRecord, KseflColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<KseflWriteRecord, KseflColumn> updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KseflDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KseflDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KseflDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(KseflWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KseflWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public KseflDataAccessImpl addRow(KseflRecord record) {
        EnumMap<KseflColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, KseflWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(KseflWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(KseflColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public List<KseflRecord> findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdAndKasgYmdAndFlwStCd(
            String flttCktkCdFrom,
            String flttCktkCdTo,
            String flttKtnCd,
            String kasErryCd,
            LocalDate kasgYmd,
            String flwStCd) {
        String safeFlttKtnCd =
                flttKtnCd == null
                        ? null
                        : WhereParamGuard.fixedLengthNotNull(flttKtnCd, "flttKtnCd", 1);
        String blankFlttKtnCd =
                WhereParamGuard.fixedLengthNotNull("", "flttKtnCd", 1);
        String safeKasErryCd =
                WhereParamGuard.fixedLengthNotNull(kasErryCd, "kasErryCd", 2);
        String safeFlwStCd =
                WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);
        return rows.stream()
                .filter(r -> isInCktkRange(r, flttCktkCdFrom, flttCktkCdTo))
                .filter(r -> isMatchKtnCd(r, safeFlttKtnCd, blankFlttKtnCd))
                .filter(r -> isMatchFixedLength(r, KseflColumn.KAS_ERRY_CD, safeKasErryCd, "kasErryCd", 2))
                .filter(r -> {
                    LocalDate rowYmd = (LocalDate) r.get(KseflColumn.KASG_YMD);
                    return kasgYmd != null && rowYmd != null && !rowYmd.isBefore(kasgYmd);
                })
                .filter(r -> isMatchFixedLength(r, KseflColumn.FLW_ST_CD, safeFlwStCd, "flwStCd", 2))
                .sorted(R001_ORDER)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public List<KseflRecord> findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdOrKasErryCdAndKasgYmdAndFlwStCd(
            String flttCktkCdFrom,
            String flttCktkCdTo,
            String flttKtnCd,
            String kasErryCd,
            LocalDate kasgYmd,
            String flwStCd) {
        String safeFlttKtnCd =
                flttKtnCd == null
                        ? null
                        : WhereParamGuard.fixedLengthNotNull(flttKtnCd, "flttKtnCd", 1);
        String blankFlttKtnCd =
                WhereParamGuard.fixedLengthNotNull("", "flttKtnCd", 1);
        String safeKasErryCd =
                WhereParamGuard.fixedLengthNotNull(kasErryCd, "kasErryCd", 2);
        String safeFlwStCd =
                WhereParamGuard.fixedLengthNotNull(flwStCd, "flwStCd", 2);
        return rows.stream()
                .filter(r -> isInCktkRange(r, flttCktkCdFrom, flttCktkCdTo))
                .filter(r -> isMatchKtnCd(r, safeFlttKtnCd, blankFlttKtnCd))
                .filter(r -> isMatchFixedLength(r, KseflColumn.FLW_ST_CD, safeFlwStCd, "flwStCd", 2))
                .filter(r -> {
                    // 共通条件の上で (KAS_ERRY_CD <> kasErryCd) OR (KAS_ERRY_CD = kasErryCd AND KASG_YMD <= kasgYmd)
                    LocalDate rowKasgYmd = (LocalDate) r.get(KseflColumn.KASG_YMD);
                    String normalizedKasErryCd =
                            normalizeRowFixedLength((String) r.get(KseflColumn.KAS_ERRY_CD), "kasErryCd", 2);
                    boolean condA = !Objects.equals(normalizedKasErryCd, safeKasErryCd);
                    boolean condB = Objects.equals(normalizedKasErryCd, safeKasErryCd)
                            && kasgYmd != null
                            && rowKasgYmd != null
                            && !rowKasgYmd.isAfter(kasgYmd);
                    return condA || condB;
                })
                .sorted(R001_ORDER)
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public List<KseflRecord> findAllByGmbAndFlwStCdAndKasgYmdAndTsigKasgYmd(
            String gmb, String flwStCd, LocalDate kasgYmd, LocalDate tsigKsgYmd) {
        return rows.stream()
                .filter(r -> Objects.equals(r.get(KseflColumn.GMB), gmb))
                .filter(r -> Objects.equals(r.get(KseflColumn.FLW_ST_CD), flwStCd))
                .filter(r -> {
                    LocalDate rowKasgYmd = (LocalDate) r.get(KseflColumn.KASG_YMD);
                    LocalDate rowTsigKasgYmd = (LocalDate) r.get(KseflColumn.TSIG_KASG_YMD);
                    // (KASG_YMD = kasgYmd AND TSIG_KASG_YMD = 0) OR (TSIG_KASG_YMD = tsigKsgYmd AND TSIG_KASG_YMD <> 0)
                    boolean cond1 = Objects.equals(rowKasgYmd, kasgYmd) && rowTsigKasgYmd == null;
                    boolean cond2 = rowTsigKasgYmd != null && Objects.equals(rowTsigKasgYmd, tsigKsgYmd);
                    return cond1 || cond2;
                })
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public int updateByGmbAndKasgYmdAndKasHukkYmdAndHukkSysShbtAndSgyAnknNoAndRecUpdJnjNo(
            KseflWriteRecord record,
            Set<KseflColumn> targetColumns,
            KseflU001UpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                r ->
                        Objects.equals(r.get(KseflColumn.GMB), updateKey.gmb())
                                && Objects.equals(r.get(KseflColumn.KASG_YMD), updateKey.kasgYmd())
                                && Objects.equals(r.get(KseflColumn.KAS_HUKK_YMD), updateKey.kasHukkYmd())
                                && Objects.equals(r.get(KseflColumn.HUKK_SYS_SHBT), updateKey.hukkSysShbt())
                                && Objects.equals(r.get(KseflColumn.SGY_ANKN_NO), updateKey.sgyAnknNo()),
                r ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(KseflColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) r.get(KseflColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KseflColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KseflDataAccess.CallSpecs
                        .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_HUKK_SYS_SHBT_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO
                        .identifier(),
                updateKey.gmb(),
                updateKey.kasgYmd(),
                updateKey.kasHukkYmd(),
                updateKey.hukkSysShbt(),
                updateKey.sgyAnknNo());
    }

    private static boolean isInCktkRange(
            EnumMap<KseflColumn, Object> r, String from, String to) {
        if (from == null || to == null) {
            return false;
        }
        String cktk = (String) r.get(KseflColumn.FLTT_CKTK_CD);
        return cktk != null && cktk.compareTo(from) >= 0 && cktk.compareTo(to) <= 0;
    }

    private static boolean isMatchKtnCd(
            EnumMap<KseflColumn, Object> r, String safeFlttKtnCd, String blankFlttKtnCd) {
        if (safeFlttKtnCd == null) {
            return true;
        }
        String normalizedFlttKtnCd =
                normalizeRowFixedLength((String) r.get(KseflColumn.FLTT_KTN_CD), "flttKtnCd", 1);
        return Objects.equals(normalizedFlttKtnCd, safeFlttKtnCd)
                || Objects.equals(normalizedFlttKtnCd, blankFlttKtnCd);
    }

    private static boolean isMatchFixedLength(
            EnumMap<KseflColumn, Object> r,
            KseflColumn column,
            String safeValue,
            String paramName,
            int length) {
        return Objects.equals(
                normalizeRowFixedLength((String) r.get(column), paramName, length),
                safeValue);
    }

    private static String normalizeRowFixedLength(String value, String paramName, int length) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, length);
    }

    /** R001/R002 の ORDER BY: KASG_YMD, KAS_HUKK_YMD, KASG_NWK_CD, GMB */
    private static final Comparator<EnumMap<KseflColumn, Object>> R001_ORDER =
            Comparator
                    .<EnumMap<KseflColumn, Object>, LocalDate>comparing(
                            r -> (LocalDate) r.get(KseflColumn.KASG_YMD), Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            r -> (LocalDate) r.get(KseflColumn.KAS_HUKK_YMD), Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            r -> (String) r.get(KseflColumn.KASG_NWK_CD), Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            r -> (String) r.get(KseflColumn.GMB), Comparator.nullsLast(Comparator.naturalOrder()));
}
