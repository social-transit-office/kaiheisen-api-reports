/**
 * DA-067-R001/U001 の SQL 振る舞いを模した KNSYTI インメモリ実装。
 */
public class KnsytiDataAccessImpl extends DataAccessBase implements KnsytiDataAccess {

    private static final ReflectiveInMemoryRowMapper<KnsytiColumn, KnsytiRecord> ROW_MAPPER =
        ReflectiveInMemoryRowMapper.create(KnsytiColumn.class, KnsytiRecord.class);

    private static final AutoNeutralValueResolver<KnsytiColumn> NEUTRAL_VALUE_RESOLVER =
        AutoNeutralValueResolver.create(KnsytiColumn.class, KnsytiRecord.class);

    private final List<EnumMap<KnsytiColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryTargetColumnUpdateSupport<KnsytiWriteRecord, KnsytiColumn>
        updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public KnsytiDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    KnsytiDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    KnsytiDataAccessImpl(
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        KnsytiWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public KnsytiDataAccessImpl addRow(InternalRow row) {
        KnsytiRecord record = toRecord(Objects.requireNonNull(row, "row"));
        EnumMap<KnsytiColumn, Object> values = ROW_MAPPER.toRawValueMap(record);
        NEUTRAL_VALUE_RESOLVER.complement(values, KnsytiWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public List<KnsytiRecord> findAllByGasMtstNoAndTkysYmAndTyenYm(Long gasMtstNo, YearMonth knskKjnYm) {
        Long safeGasMtstNo = WhereParamGuard.complement(gasMtstNo, "gasMtstNo", 0L);
        Integer safeKnskKjnYm =
            WhereParamGuard.complement(
                knskKjnYm == null ? null : knskKjnYm.toYmInt(),
                "knskKjnYm",
                0);
        return rows.stream()
            .filter(row -> Objects.equals(row.get(KnsytiColumn.GAS_MTST_NO), safeGasMtstNo))
            .filter(
                row -> {
                    YearMonth tkysYm = (YearMonth) row.get(KnsytiColumn.TKYS_YM);
                    YearMonth tyenYm = (YearMonth) row.get(KnsytiColumn.TYEN_YM);
                    return tkysYm != null
                        && tyenYm != null
                        && tkysYm.toYmInt() <= safeKnskKjnYm
                        && tyenYm.toYmInt() >= safeKnskKjnYm;
                })
            .map(ROW_MAPPER::toRecord)
                .toList();
    }

    /**
     * KNSYTI のテストデータ行。
     */
    public static record InternalRow(
            long gasMtstNo,
            int tkysYm,
            int tyenYm,
            String tkkbNo,
            String knsHuhuKbn,
            String stkbKbn,
            String shtiKskbNo,
            String tdkKskbNo,
            int jkiTkytYmd,
            String nwjgCd,
            String mgksTyumFlg,
            LocalDateTime regTsmp,
            String regUsrId,
            LocalDateTime updTsmp,
            String updUsrId,
            String updKtnKinoId,
            int recUpdJnjNo
    ) {
        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(tkkbNo, "tkkbNo");
            Objects.requireNonNull(knsHuhuKbn, "knsHuhuKbn");
            Objects.requireNonNull(stkbKbn, "stkbKbn");
            Objects.requireNonNull(shtiKskbNo, "shtiKskbNo");
            Objects.requireNonNull(tdkKskbNo, "tdkKskbNo");
            Objects.requireNonNull(nwjgCd, "nwjgCd");
            Objects.requireNonNull(mgksTyumFlg, "mgksTyumFlg");
            Objects.requireNonNull(regTsmp, "regTsmp");
            Objects.requireNonNull(regUsrId, "regUsrId");
            Objects.requireNonNull(updTsmp, "updTsmp");
            Objects.requireNonNull(updUsrId, "updUsrId");
            Objects.requireNonNull(updKtnKinoId, "updKtnKinoId");
        }
    }

    @Override
    public Optional<KnsytiRecord> findByPK(Long gasMtstNo, YearMonth tkysYm) {
        Long safeGasMtstNo = WhereParamGuard.complement(gasMtstNo, "gasMtstNo", 0L);
        Integer safeTkysYm =
            WhereParamGuard.complement(
                tkysYm == null ? null : tkysYm.toYmInt(),
                "tkysYm",
                0);
        return rows.stream()
                .filter(row -> Objects.equals(row.get(KnsytiColumn.GAS_MTST_NO), safeGasMtstNo))
            .filter(
                row -> {
                    YearMonth rowTkysYm = (YearMonth) row.get(KnsytiColumn.TKYS_YM);
                    return rowTkysYm != null && rowTkysYm.toYmInt() == safeTkysYm;
                })
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByGasMtstNoAndTkysYmAndRecUpdJnjNo(
            KnsytiWriteRecord record,
            Set<KnsytiColumn> targetColumns,
            KnsytiUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        Long safeGasMtstNo = WhereParamGuard.complement(updateKey.gasMtstNo(), "gasMtstNo", 0L);
        Integer safeTkysYm =
            WhereParamGuard.complement(
                updateKey.tkysYm() == null ? null : updateKey.tkysYm().toYmInt(),
                "tkysYm",
                0);
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(row.get(KnsytiColumn.GAS_MTST_NO), safeGasMtstNo)
                && matchesTkysYm(row, safeTkysYm),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        KnsytiColumn.class,
                                                        values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(KnsytiColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                KnsytiColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                KnsytiDataAccess.CallSpecs.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YM_AND_REC_UPD_JNJ_NO.identifier(),
                safeGasMtstNo,
                safeTkysYm);
    }

    /**
     * 行の適用開始年月が検索条件と一致するかを判定する。
     *
     * @param row 保存行
     * @param safeTkysYm 補完済みの検索年月
     * @return 一致する場合 true
     */
    private static boolean matchesTkysYm(Map<KnsytiColumn, Object> row, Integer safeTkysYm) {
        YearMonth rowTkysYm = (YearMonth) row.get(KnsytiColumn.TKYS_YM);
        return rowTkysYm != null && rowTkysYm.toYmInt() == safeTkysYm;
    }

    /**
     * テスト用内部行を read record へ変換する。
     *
     * @param row テスト用内部行
     * @return read record
     */
    private static KnsytiRecord toRecord(InternalRow row) {
        return new KnsytiRecord(
                row.gasMtstNo(),
                toYearMonth(row.tkysYm()),
                toYearMonth(row.tyenYm()),
                row.tkkbNo(),
                row.knsHuhuKbn(),
                row.stkbKbn(),
                row.shtiKskbNo(),
                row.tdkKskbNo(),
                toLocalDate(row.jkiTkytYmd()),
                row.nwjgCd(),
                toBooleanFlag(row.mgksTyumFlg()),
                row.regTsmp(),
                row.regUsrId(),
                row.updTsmp(),
                row.updUsrId(),
                row.updKtnKinoId(),
                row.recUpdJnjNo());
    }

    /**
     * 数値型年月を YearMonth へ変換する。
     *
     * @param ym 数値年月
     * @return YearMonth
     */
    private static YearMonth toYearMonth(int ym) {
        if (ym <= 0) {
            return null;
        }
        return YearMonth.of(ym / 100, ym % 100);
    }

    /**
     * 数値型年月日を LocalDate へ変換する。
     *
     * @param ymd 数値年月日
     * @return LocalDate
     */
    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }

    /**
     * 文字列フラグを Boolean へ変換する。
     *
     * @param value フラグ文字列
     * @return Boolean
     */
    private static Boolean toBooleanFlag(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        if ("1".equals(normalized)
                || "Y".equalsIgnoreCase(normalized)
                || "T".equalsIgnoreCase(normalized)
                || "TRUE".equalsIgnoreCase(normalized)) {
            return Boolean.TRUE;
        }
        if ("0".equals(normalized)
                || "N".equalsIgnoreCase(normalized)
                || "F".equalsIgnoreCase(normalized)
                || "FALSE".equalsIgnoreCase(normalized)) {
            return Boolean.FALSE;
        }
        return null;
    }
}
