package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.isrkk;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-102 の SQL をインメモリで再現する実装。 */
public class IsrkkDataAccessImpl extends DataAccessBase implements IsrkkDataAccess {

    private static final int ISRR_GMSS_CD_LENGTH = 8;

    private final List<IsrkkRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public IsrkkDataAccessImpl addRow(IsrkkRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public List<IsrkkRecord> findAllByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> isrrGmssCdList) {
        List<String> safeIsrrGmssCdList = normalizeIsrrGmssCdList(isrrGmssCdList);
        if (safeIsrrGmssCdList.isEmpty()) {
            return List.of();
        }
        Long safeArtyCstNo = WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L);
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(toYmd(isrrTriYmd), "isrrTriYmd", BigDecimal.ZERO);

        return rows.stream()
                .filter(row -> normalizeLong(row.artyCstNo()).equals(safeArtyCstNo))
                .filter(row -> normalizeYmd(row.isrrTriYmd()).equals(safeIsrrTriYmd))
                .filter(row -> safeIsrrGmssCdList.contains(normalizeCode(row.isrrGmssCd())))
                .sorted(
                        Comparator.comparing(
                                IsrkkDataAccessImpl::normalizeTsno,
                                Comparator.nullsLast(Integer::compareTo)))
                .map(IsrkkDataAccessImpl::normalizeRecord)
                .toList();
    }

    @Override
    public Optional<IsrkkRecord> findByArtyCstNoAndIsrrTriYmdAndIsrrGmssCd(
            Long artyCstNo,
            LocalDate isrrTriYmd,
            List<String> isrrGmssCdList) {
        List<String> safeIsrrGmssCdList = normalizeIsrrGmssCdList(isrrGmssCdList);
        if (safeIsrrGmssCdList.isEmpty()) {
            return Optional.empty();
        }
        Long safeArtyCstNo = WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L);
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(toYmd(isrrTriYmd), "isrrTriYmd", BigDecimal.ZERO);

        return rows.stream()
                .filter(row -> normalizeLong(row.artyCstNo()).equals(safeArtyCstNo))
                .filter(row -> normalizeYmd(row.isrrTriYmd()).equals(safeIsrrTriYmd))
                .filter(row -> safeIsrrGmssCdList.contains(normalizeCode(row.isrrGmssCd())))
                .sorted(
                        Comparator.comparing(
                                        IsrkkDataAccessImpl::normalizeTsno,
                                        Comparator.nullsLast(Integer::compareTo))
                                .reversed())
                .findFirst()
                .map(IsrkkDataAccessImpl::normalizeRecord);
    }

    @Override
    public Optional<IsrkkRecord> findByPK(Long artyCstNo, LocalDate isrrTriYmd, Integer isrrKytuTsno) {
        Long safeArtyCstNo = WhereParamGuard.complement(artyCstNo, "artyCstNo", 0L);
        BigDecimal safeIsrrTriYmd =
                WhereParamGuard.complement(toYmd(isrrTriYmd), "isrrTriYmd", BigDecimal.ZERO);
        Integer safeIsrrKytuTsno = WhereParamGuard.complement(isrrKytuTsno, "isrrKytuTsno", 0);

        return rows.stream()
                .filter(row -> normalizeLong(row.artyCstNo()).equals(safeArtyCstNo))
                .filter(row -> normalizeYmd(row.isrrTriYmd()).equals(safeIsrrTriYmd))
                .filter(row -> normalizeTsno(row).equals(safeIsrrKytuTsno))
                .findFirst()
                .map(IsrkkDataAccessImpl::normalizeRecord);
    }

    /**
     * IN 句検索用の業務詳細コード一覧を正規化する。
     *
     * @param isrrGmssCdList 異動・作業履歴業務詳細コード一覧
     * @return 正規化後の一覧
     */
    private static List<String> normalizeIsrrGmssCdList(List<String> isrrGmssCdList) {
        if (isrrGmssCdList == null) {
            throw new IllegalArgumentException("isrrGmssCdList must not be null");
        }
        if (isrrGmssCdList.isEmpty()) {
            return List.of();
        }
        return isrrGmssCdList.stream().map(IsrkkDataAccessImpl::normalizeInIsrrGmssCd).toList();
    }

    /**
     * IN 句要素の業務詳細コードを固定長条件値へ正規化する。
     *
     * @param value 業務詳細コード
     * @return 正規化後の業務詳細コード
     */
    private static String normalizeInIsrrGmssCd(String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(
                    "isrrGmssCdList must not contain null or blank-only");
        }
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGmssCdList", ISRR_GMSS_CD_LENGTH);
    }

    /**
     * 顧客番号条件を補完付きで正規化する。
     *
     * @param value 顧客番号
     * @return 正規化後の顧客番号
     */
    private static Long normalizeLong(Long value) {
        return WhereParamGuard.complement(value, "value", 0L);
    }

    /**
     * 通し番号を補完付きで正規化する。
     *
     * @param row 行データ
     * @return 正規化後の通し番号
     */
    private static Integer normalizeTsno(IsrkkRecord row) {
        return WhereParamGuard.complement(row.isrrKytuTsno(), "isrrKytuTsno", 0);
    }

    /**
     * 年月日条件を DB 数値形式へ正規化する。
     *
     * @param value 年月日
     * @return 正規化後の DB 値
     */
    private static BigDecimal normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(toYmd(value), "value", BigDecimal.ZERO);
    }

    /**
     * 年月日を DB 数値形式へ変換する。
     *
     * @param value 年月日
     * @return 変換後の DB 値
     */
    private static BigDecimal toYmd(LocalDate value) {
        if (value == null) {
            return null;
        }
        return BigDecimal.valueOf(value.getYear() * 10000L + value.getMonthValue() * 100L + value.getDayOfMonth());
    }

    /**
     * 業務詳細コードを固定長条件値へ正規化する。
     *
     * @param value 業務詳細コード
     * @return 正規化後の業務詳細コード
     */
    private static String normalizeCode(String value) {
        return WhereParamGuard.fixedLengthNotNull(value, "isrrGmssCd", ISRR_GMSS_CD_LENGTH);
    }

    /**
     * 取得結果レコードを API 公開値へ正規化する。
     *
     * @param row 行データ
     * @return API 公開値
     */
    private static IsrkkRecord normalizeRecord(IsrkkRecord row) {
        return new IsrkkRecord(
                normalizeResultLong(row.artyCstNo()),
                row.isrrTriYmd(),
                normalizeResultInteger(row.isrrKytuTsno()),
                normalizeResultLong(row.kyoTrkNo()),
                normalizeResultLong(row.gasMtstNo()),
                normalizeResultLong(row.sikKrno()),
                normalizeResultLong(row.svkyNo()),
                normalizeResultLong(row.sihKiyNo()),
                suppress(row.akrkKbn()),
                suppress(row.akrkTrskShbt()),
                suppress(row.isrrGymCd()),
                suppress(row.isrrGmssCd()),
                suppress(row.isrrHkkrKbn()),
                suppress(row.isrrSktCd()),
                suppress(row.shziKbn()),
                suppress(row.kyonKj1()),
                suppress(row.kyonKn1()),
                suppress(row.kyon1HujnShbt()),
                suppress(row.kyon1HshiShbt()),
                suppress(row.kyonKj2()),
                suppress(row.kyonKn2()),
                suppress(row.kyon2HujnShbt()),
                suppress(row.kyon2HshiShbt()),
                row.kyonKjhkFlg(),
                normalizeResultInteger(row.isrrTriHms()),
                row.idsgYmd(),
                suppress(row.isrrHusrId()),
                suppress(row.isrrHkunKj()),
                suppress(row.isrrHjsgId()),
                row.regTsmp(),
                suppress(row.regUsrId()),
                row.updTsmp(),
                suppress(row.updUsrId()),
                suppress(row.updKtnKinoId()),
                normalizeResultInteger(row.recUpdJnjNo()));
    }

    /**
     * 取得結果の Long 値を API 公開値へ正規化する。
     *
     * @param value 数値
     * @return 正規化後の値
     */
    private static Long normalizeResultLong(Long value) {
        return value != null && value == 0L ? null : value;
    }

    /**
     * 取得結果の Integer 値を API 公開値へ正規化する。
     *
     * @param value 数値
     * @return 正規化後の値
     */
    private static Integer normalizeResultInteger(Integer value) {
        return value != null && value == 0 ? null : value;
    }

    /**
     * 固定長文字列の後続空白を抑止する。
     *
     * @param value 文字列
     * @return 正規化後の文字列
     */
    private static String suppress(String value) {
        return FixedLengthFormatter.suppressSpace(value);
    }
}
