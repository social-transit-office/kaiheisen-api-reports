package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.sihtksk;

import java.util.Optional;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/**
 * DA-032 の SQL 振る舞いを模した SIHTKSK インメモリ実装。
 */
public class SihtkskDataAccessImpl extends DataAccessBase implements SihtkskDataAccess {

    private static final ReflectiveInMemoryRowMapper<SihtkskColumn, SihtkskRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SihtkskColumn.class, SihtkskRecord.class);

    private static final AutoNeutralValueResolver<SihtkskColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SihtkskColumn.class, SihtkskRecord.class);

    private final List<EnumMap<SihtkskColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SihtkskWriteRecord, SihtkskColumn> insertSupport;

    /**
     * デフォルト依存でインスタンスを生成する。
     */
    public SihtkskDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    SihtkskDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    SihtkskDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        SihtkskWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param record 追加レコード
     * @return このインスタンス
     */
    public SihtkskDataAccessImpl addRow(SihtkskRecord record) {
        EnumMap<SihtkskColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, SihtkskWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** {@inheritDoc} */
    @Override
    public int insert(SihtkskWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                SihtkskColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    /** {@inheritDoc} */
    @Override
    public Optional<SihtkskRecord> findByPK(Long sihKiyNo) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        return rows.stream()
                .filter(row -> Objects.equals(row.get(SihtkskColumn.SIH_KIY_NO), safeSihKiyNo))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /**
     * long 引数を補完する。
     *
     * @param value 元値
     * @param paramName パラメータ名
     * @return 補完後値
     */
    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }
}
