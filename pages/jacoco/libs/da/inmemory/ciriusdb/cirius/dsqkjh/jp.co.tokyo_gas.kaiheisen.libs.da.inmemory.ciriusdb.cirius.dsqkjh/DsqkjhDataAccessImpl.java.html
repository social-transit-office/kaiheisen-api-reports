package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.dsqkjh;

import java.util.Optional;
import java.util.Set;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/**
 * DA-132 の SQL 振る舞いを模した DSQKJH インメモリ実装.
 */
public class DsqkjhDataAccessImpl
        extends DataAccessBase implements DsqkjhDataAccess {

    /** 行 mapper. */
    private static final ReflectiveInMemoryRowMapper<DsqkjhColumn, DsqkjhRecord>
            ROW_MAPPER = ReflectiveInMemoryRowMapper.create(
                    DsqkjhColumn.class,
                    DsqkjhRecord.class);

    /** 中立値解決器. */
    private static final AutoNeutralValueResolver<DsqkjhColumn>
            NEUTRAL_VALUE_RESOLVER = AutoNeutralValueResolver.create(
                    DsqkjhColumn.class,
                    DsqkjhRecord.class);

    /** 保持行一覧. */
    private final List<EnumMap<DsqkjhColumn, Object>> rows = new ArrayList<>();

    /** システム時刻プロバイダ. */
    private final SystemDateTimeProvider clock;

    /** 監査項目自動設定. */
    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    /** insert helper. */
    private final InMemoryInsertSupport<DsqkjhWriteRecord, DsqkjhColumn>
            insertSupport;

    /** update helper. */
    private final InMemoryTargetColumnUpdateSupport<
            DsqkjhWriteRecord,
            DsqkjhColumn> updateSupport;

    /** デフォルト依存でインスタンスを生成する. */
    public DsqkjhDataAccessImpl() {
        this(
                new DefaultSystemDateTimeProvider(),
                new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に時刻依存を差し替えてインスタンスを生成する.
     *
     * @param systemClock システム時刻プロバイダ
     */
    DsqkjhDataAccessImpl(final SystemDateTimeProvider systemClock) {
        this(systemClock, new CiriusAuditColumnAutoSetter());
    }

    /**
     * テスト用に依存を差し替えてインスタンスを生成する.
     *
     * @param systemClock システム時刻プロバイダ
     * @param ciriusAuditColumnAutoSetter 監査項目自動設定
     */
    DsqkjhDataAccessImpl(
            final SystemDateTimeProvider systemClock,
            final CiriusAuditColumnAutoSetter ciriusAuditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(systemClock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(
                        ciriusAuditColumnAutoSetter,
                        "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        DsqkjhWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        DsqkjhWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する.
     *
     * @param record 追加対象レコード
     * @return このインスタンス
     */
    public DsqkjhDataAccessImpl addRow(final DsqkjhRecord record) {
        EnumMap<DsqkjhColumn, Object> values = ROW_MAPPER.toRawValueMap(
                Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(
                values,
                DsqkjhWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /** {@inheritDoc} */
    @Override
    public int insert(
            final DsqkjhWriteRecord record,
            final AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values -> auditColumnAutoSetter.applyForInsert(
                                new MapBackedAuditColumnAccess<>(
                                        DsqkjhColumn.class,
                                        values),
                                context,
                                clock)));
        return 1;
    }

    /** {@inheritDoc} */
    @Override
    public int deleteBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> Objects.equals(
                        row.get(DsqkjhColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(
                                row.get(DsqkjhColumn.SIH_KIY_SKNO),
                                safeSihKiySkno),
                DsqkjhColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                DsqkjhDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                () -> new DataInconsistencyException(
                        "DSQKJH",
                        DsqkjhDataAccess.CallSpecs.DELETE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier(),
                        "{}"),
                safeSihKiyNo,
                safeSihKiySkno);
    }

    /** {@inheritDoc} */
    @Override
    public Optional<DsqkjhRecord> findBySihKiyNoAndSihKiySkno(
            final Long sihKiyNo,
            final Integer sihKiySkno) {
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return rows.stream()
                .filter(row -> Objects.equals(
                        row.get(DsqkjhColumn.SIH_KIY_NO),
                        safeSihKiyNo))
                .filter(row -> Objects.equals(
                        row.get(DsqkjhColumn.SIH_KIY_SKNO),
                        safeSihKiySkno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySkno(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final DsqkjhSihKiyUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                DsqkjhDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO.identifier());
    }

    /** {@inheritDoc} */
    @Override
    public int updateBySihKiyNoAndSihKiySknoAndRecUpdJnjNo(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final DsqkjhSihKiyJnjUpdateKey updateKey,
            final AuditContext context) {
        Objects.requireNonNull(updateKey, "updateKey");
        return updateByKey(
                record,
                targetColumns,
                updateKey.sihKiyNo(),
                updateKey.sihKiySkno(),
                updateKey.recUpdJnjNo(),
                context,
                DsqkjhDataAccess.CallSpecs.UPDATE_BY_SIH_KIY_NO_AND_SIH_KIY_SKNO_AND_REC_UPD_JNJ_NO.identifier());
    }

    private int updateByKey(
            final DsqkjhWriteRecord record,
            final Set<DsqkjhColumn> targetColumns,
            final Long sihKiyNo,
            final Integer sihKiySkno,
            final Integer recUpdJnjNo,
            final AuditContext context,
            final String dataAccessIdentifier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(targetColumns, "targetColumns");
        Objects.requireNonNull(context, "context");
        Long safeSihKiyNo = safeLong(sihKiyNo, "sihKiyNo");
        Integer safeSihKiySkno = safeInteger(sihKiySkno, "sihKiySkno");
        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row -> Objects.equals(
                        row.get(DsqkjhColumn.SIH_KIY_NO), safeSihKiyNo)
                        && Objects.equals(
                                row.get(DsqkjhColumn.SIH_KIY_SKNO),
                                safeSihKiySkno),
                row -> updateSupport.prepare(
                        record,
                        targetColumns,
                        values -> auditColumnAutoSetter.applyForUpdate(
                                new MapBackedAuditColumnAccess<>(
                                        DsqkjhColumn.class,
                                        values),
                                context,
                                clock,
                                (Integer) row.get(
                                        DsqkjhColumn.REC_UPD_JNJ_NO))),
                updateSupport::apply,
                DsqkjhColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                dataAccessIdentifier,
                safeSihKiyNo,
                safeSihKiySkno);
    }

    private static Long safeLong(final Long value, final String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInteger(
            final Integer value,
            final String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }
}
