package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.gymykiy;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gymykiy.GymykiyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gymykiy.GymykiyRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/** DA-058-R001 の SQL を模したインメモリ実装。 */
public class GymykiyDataAccessImpl extends DataAccessBase implements GymykiyDataAccess {

    private final List<GymykiyRecord> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GymykiyDataAccessImpl addRow(GymykiyRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public Optional<GymykiyRecord> findBySvkyNoAndKyttSknoAndKyttTsno(
            Long svkyNo,
            Integer kyttSkno,
            Integer kyttTsno) {
        Long safeSvkyNo = WhereParamGuard.complement(svkyNo, "svkyNo", 0L);
        Integer safeKyttSkno = WhereParamGuard.complement(kyttSkno, "kyttSkno", 0);
        Integer safeKyttTsno = WhereParamGuard.complement(kyttTsno, "kyttTsno", 0);
        return rows.stream()
                .filter(row -> Objects.equals(row.svkyNo(), safeSvkyNo))
                .filter(row -> Objects.equals(row.kyttSkno(), safeKyttSkno))
                .filter(row -> Objects.equals(row.kyttTsno(), safeKyttTsno))
                .findFirst();
    }
}
