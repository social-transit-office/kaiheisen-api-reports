package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.mtsetk;

/** DA-048（C001/D001/R001-R004/U001）の SQL 振る舞いを模した MTSETK インメモリ実装。 */
public class MtsetkDataAccessImpl extends DataAccessBase implements MtsetkDataAccess {

    private static final ReflectiveInMemoryRowMapper<MtsetkColumn, MtsetkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(MtsetkColumn.class, MtsetkRecord.class);

    private static final AutoNeutralValueResolver<MtsetkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(MtsetkColumn.class, MtsetkRecord.class);

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private final List<EnumMap<MtsetkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<MtsetkWriteRecord, MtsetkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<MtsetkWriteRecord, MtsetkColumn> updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public MtsetkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    MtsetkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    MtsetkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(MtsetkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        MtsetkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param record 追加する行
     * @return このインスタンス
     */
    public MtsetkDataAccessImpl addRow(MtsetkRecord record) {
        EnumMap<MtsetkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(record, "record"));
        NEUTRAL_VALUE_RESOLVER.complement(values, MtsetkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    @Override
    public int insert(MtsetkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                MtsetkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int deleteByMtstNoAndTkysYmd(Long mtstNo, LocalDate tkysYmd, Integer recUpdJnjNo) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        Objects.equals(row.get(MtsetkColumn.MTST_NO), safeMtstNo)
                                && Objects.equals(
                                        normalizeYmd(
                                                (LocalDate) row.get(MtsetkColumn.TKYS_YMD)),
                                        safeTkysYmd),
                MtsetkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                MtsetkDataAccess.CallSpecs.DELETE_BY_MTST_NO_AND_TKYS_YMD.identifier(),
                () ->
                        new DataInconsistencyException(
                                "MTSETK",
                                MtsetkDataAccess.CallSpecs.DELETE_BY_MTST_NO_AND_TKYS_YMD.identifier(),
                                "{}"),
                safeMtstNo,
                safeTkysYmd);
    }

    @Override
    public Optional<MtsetkRecord> findByMtstNoAndTkysYmd(Long mtstNo, LocalDate tkysYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return rows.stream()
                .filter(
                        r ->
                                Objects.equals(r.get(MtsetkColumn.MTST_NO), safeMtstNo)
                                        && Objects.equals(
                                                normalizeYmd(
                                                        (LocalDate)
                                                                r.get(MtsetkColumn.TKYS_YMD)),
                                                safeTkysYmd))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public List<MtsetkRecord> findAllByMtstNoAndTkysYmdAndTyenYmd(
            Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(r -> Objects.equals(r.get(MtsetkColumn.MTST_NO), safeMtstNo))
                .filter(
                        r -> {
                            LocalDate tkysYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TKYS_YMD));
                            LocalDate tyenYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TYEN_YMD));
                            return !tkysYmd.isAfter(safeKnskKjnYmd)
                                    && !tyenYmd.isBefore(safeKnskKjnYmd);
                        })
                .map(ROW_MAPPER::toRecord)
                .toList();
    }

    @Override
    public long countByMtstNoAndTkysYmdAndTyenYmd(Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(r -> Objects.equals(r.get(MtsetkColumn.MTST_NO), safeMtstNo))
                .filter(
                        r -> {
                            LocalDate tkysYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TKYS_YMD));
                            LocalDate tyenYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TYEN_YMD));
                            return !tkysYmd.isAfter(safeKnskKjnYmd)
                                    && !tyenYmd.isBefore(safeKnskKjnYmd);
                        })
                .count();
    }

    @Override
    public Optional<MtsetkRecord> findByMtstNoAndTkysYmdAndTyenYmd(
            Long mtstNo, LocalDate knskKjnYmd) {
        Long safeMtstNo = safeMtstNo(mtstNo, "mtstNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");
        return rows.stream()
                .filter(r -> Objects.equals(r.get(MtsetkColumn.MTST_NO), safeMtstNo))
                .filter(
                        r -> {
                            LocalDate tkysYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TKYS_YMD));
                            LocalDate tyenYmd =
                                    normalizeYmd((LocalDate) r.get(MtsetkColumn.TYEN_YMD));
                            return !tkysYmd.isAfter(safeKnskKjnYmd)
                                    && !tyenYmd.isBefore(safeKnskKjnYmd);
                        })
                .sorted(
                        Comparator.<EnumMap<MtsetkColumn, Object>, LocalDate>comparing(
                                r -> normalizeYmd((LocalDate) r.get(MtsetkColumn.TKYS_YMD)),
                                Comparator.reverseOrder()))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public int updateByMtstNoAndTkysYmdAndRecUpdJnjNo(
            MtsetkWriteRecord record,
            Set<MtsetkColumn> targetColumns,
            MtsetkUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
                row ->
                        Objects.equals(row.get(MtsetkColumn.MTST_NO), updateKey.mtstNo())
                                && Objects.equals(row.get(MtsetkColumn.TKYS_YMD), updateKey.tkysYmd()),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        MtsetkColumn.class, values),
                                                context,
                                                clock,
                                                updateKey.recUpdJnjNo() == null
                                                        ? (Integer) row.get(MtsetkColumn.REC_UPD_JNJ_NO)
                                                        : updateKey.recUpdJnjNo())),
                updateSupport::apply,
                MtsetkColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                MtsetkDataAccess.CallSpecs.UPDATE_BY_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO.identifier(),
                updateKey.mtstNo(),
                updateKey.tkysYmd());
    }

    private static Long safeMtstNo(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static LocalDate normalizeYmd(LocalDate value) {
        return WhereParamGuard.complement(value, "ymd", COMPLEMENT_YMD);
    }
}
