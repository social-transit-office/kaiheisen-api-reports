/** DA-029（C001/D001/R001-R002/U001-U002）の SQL 振る舞いを模した GMTSETK インメモリ実装。 */
public class GmtsetkDataAccessImpl extends DataAccessBase implements GmtsetkDataAccess {

    private static final ReflectiveInMemoryRowMapper<GmtsetkColumn, GmtsetkRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(GmtsetkColumn.class, GmtsetkRecord.class);

    private static final AutoNeutralValueResolver<GmtsetkColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(GmtsetkColumn.class, GmtsetkRecord.class);

    private final List<EnumMap<GmtsetkColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<GmtsetkWriteRecord, GmtsetkColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<GmtsetkWriteRecord, GmtsetkColumn>
            updateSupport;

    /** システム日付プロバイダのデフォルト実装で初期化する。 */
    public GmtsetkDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    GmtsetkDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    GmtsetkDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        GmtsetkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        GmtsetkWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用の初期データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public GmtsetkDataAccessImpl addRow(GmtsetkRecord row) {
        EnumMap<GmtsetkColumn, Object> values =
                ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, GmtsetkWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    List<GmtsetkRecord> rows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(GmtsetkWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");

        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                GmtsetkColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public Optional<GmtsetkRecord> findByGasMtstNoAndTkysYmd(Long gasMtstNo, LocalDate tkysYmd) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");
        return rows.stream()
                .filter(
                        r ->
                                Objects.equals(r.get(GmtsetkColumn.GAS_MTST_NO), safeGasMtstNo)
                                        && Objects.equals(r.get(GmtsetkColumn.TKYS_YMD), safeTkysYmd))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    @Override
    public Optional<GmtsetkRecord> findByGasMtstNoAndTkysYmdAndTyenYmd(
            Long gasMtstNo, LocalDate knskKjnYmd) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeKnskKjnYmd = safeYmd(knskKjnYmd, "knskKjnYmd");

        EnumMap<GmtsetkColumn, Object> latest = null;
        LocalDate latestTkysYmd = null;
        for (EnumMap<GmtsetkColumn, Object> row : rows) {
            if (!Objects.equals(row.get(GmtsetkColumn.GAS_MTST_NO), safeGasMtstNo)) {
                continue;
            }
            LocalDate tkysYmd = (LocalDate) row.get(GmtsetkColumn.TKYS_YMD);
            LocalDate tyenYmd = (LocalDate) row.get(GmtsetkColumn.TYEN_YMD);
            if (tkysYmd == null || tyenYmd == null) {
                continue;
            }
            if (tkysYmd.isAfter(safeKnskKjnYmd) || tyenYmd.isBefore(safeKnskKjnYmd)) {
                continue;
            }
            if (latest == null || tkysYmd.isAfter(latestTkysYmd)) {
                latest = row;
                latestTkysYmd = tkysYmd;
            }
        }
        return Optional.ofNullable(latest).map(ROW_MAPPER::toRecord);
    }

    @Override
    public int updateByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
            GmtsetkWriteRecord record,
            Set<GmtsetkColumn> targetColumns,
            GmtsetkTkysJnjUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
            rows,
            row ->
                Objects.equals(row.get(GmtsetkColumn.GAS_MTST_NO), updateKey.gasMtstNo())
                    && Objects.equals(
                        row.get(GmtsetkColumn.TKYS_YMD), updateKey.tkysYmd()),
            row ->
                updateSupport.prepare(
                    record,
                    targetColumns,
                    values ->
                        auditColumnAutoSetter.applyForUpdate(
                            new MapBackedAuditColumnAccess<>(
                                GmtsetkColumn.class, values),
                            context,
                            clock,
                            updateKey.recUpdJnjNo() == null
                                ? (Integer)
                                    row.get(
                                        GmtsetkColumn.REC_UPD_JNJ_NO)
                                : updateKey.recUpdJnjNo())),
            updateSupport::apply,
            GmtsetkColumn.REC_UPD_JNJ_NO,
            updateKey.recUpdJnjNo(),
            GmtsetkDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO,
            updateKey.gasMtstNo(),
            updateKey.tkysYmd());
    }

    @Override
    public int updateByGasMtstNoAndTyenYmd(
            GmtsetkWriteRecord record,
            Set<GmtsetkColumn> targetColumns,
            GmtsetkTyenUpdateKey updateKey,
            AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(updateKey, "updateKey");
        Objects.requireNonNull(context, "context");

        return InMemoryOptimisticUpdateSupport.update(
            rows,
            row ->
                Objects.equals(row.get(GmtsetkColumn.GAS_MTST_NO), updateKey.gasMtstNo())
                    && Objects.equals(
                        row.get(GmtsetkColumn.TYEN_YMD), updateKey.tyenYmd()),
            row ->
                updateSupport.prepare(
                    record,
                    targetColumns,
                    values ->
                        auditColumnAutoSetter.applyForUpdate(
                            new MapBackedAuditColumnAccess<>(
                                GmtsetkColumn.class, values),
                            context,
                            clock,
                            (Integer) row.get(GmtsetkColumn.REC_UPD_JNJ_NO))),
            updateSupport::apply,
            GmtsetkColumn.REC_UPD_JNJ_NO,
            updateKey.recUpdJnjNo(),
            GmtsetkDataAccess.Identifiers.UPDATE_BY_GAS_MTST_NO_AND_TYEN_YMD,
            updateKey.gasMtstNo(),
            updateKey.tyenYmd());
    }

    @Override
    public int deleteByGasMtstNoAndTkysYmdAndRecUpdJnjNo(
            Long gasMtstNo, LocalDate tkysYmd, Integer recUpdJnjNo) {
        Long safeGasMtstNo = safeLong(gasMtstNo, "gasMtstNo");
        LocalDate safeTkysYmd = safeYmd(tkysYmd, "tkysYmd");

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row ->
                        Objects.equals(row.get(GmtsetkColumn.GAS_MTST_NO), safeGasMtstNo)
                                && Objects.equals(row.get(GmtsetkColumn.TKYS_YMD), safeTkysYmd),
                GmtsetkColumn.REC_UPD_JNJ_NO,
                recUpdJnjNo,
                GmtsetkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO,
                () ->
                        new DataInconsistencyException(
                                "GMTSETK",
                                GmtsetkDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_REC_UPD_JNJ_NO,
                                "{}"),
                safeGasMtstNo,
                safeTkysYmd);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, LocalDate.MIN);
    }
}
