package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.knsssvc;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsssvc.KnsssvcDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.knsssvc.KnsssvcRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core.InMemoryOptimisticDeleteSupport;

/**
 * DA-013-D001 の SQL を模したインメモリ実装。
 */
public class KnsssvcDataAccessImpl extends DataAccessBase implements KnsssvcDataAccess {

    private final List<InternalRow> rows = new ArrayList<>();

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public KnsssvcDataAccessImpl addRow(InternalRow row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }

    @Override
    public int deleteByGasMtstNoAndIdujMskhFlg(Long gasMtstNo, Boolean idujMskhFlg, Integer recUpdJnjNo) {
        Objects.requireNonNull(gasMtstNo, "gasMtstNo");
        Objects.requireNonNull(idujMskhFlg, "idujMskhFlg");

        return InMemoryOptimisticDeleteSupport.delete(
                rows,
                row -> gasMtstNo.equals(row.gasMtstNo()) && idujMskhFlg.equals(row.idujMskhFlg()),
                InternalRow::recUpdJnjNo,
                recUpdJnjNo,
                KnsssvcDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_IDUJ_MSKH_FLG,
                () ->
                        new DataInconsistencyException(
                                "KNSSSVC",
                                KnsssvcDataAccess.Identifiers.DELETE_BY_GAS_MTST_NO_AND_IDUJ_MSKH_FLG,
                                "{}"),
                gasMtstNo,
                toDbFlagString(idujMskhFlg));
    }

    private static String toDbFlagString(Boolean flag) {
        return flag ? "1" : "0";
    }

    /**
     * KNSSSVC のテストデータ行。
     */
    public static record InternalRow(Long gasMtstNo, Boolean idujMskhFlg, Integer recUpdJnjNo) {

        /**
         * 排他制御順序番号なしのテストデータ行を生成する。
         *
         * @param gasMtstNo ガスメーター設置場所番号
         * @param idujMskhFlg 移動需要増減フラグ
         */
        public InternalRow(Long gasMtstNo, Boolean idujMskhFlg) {
            this(gasMtstNo, idujMskhFlg, null);
        }

        /** コンパクトコンストラクタ。 */
        public InternalRow {
            Objects.requireNonNull(gasMtstNo, "gasMtstNo");
            Objects.requireNonNull(idujMskhFlg, "idujMskhFlg");
        }
    }

    @Override
    public Optional<KnsssvcRecord> findByPK(Long gasMtstNo, String ksseBnriShbt) {
        throw new UnsupportedOperationException("findByPK is not supported by this InMemory implementation");
    }
}
