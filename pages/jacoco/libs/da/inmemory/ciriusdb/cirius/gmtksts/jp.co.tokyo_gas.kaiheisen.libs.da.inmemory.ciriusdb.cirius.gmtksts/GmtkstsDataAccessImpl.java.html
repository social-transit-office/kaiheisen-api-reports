package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.gmtksts;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtksts.GmtkstsDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtksts.GmtkstsRecord;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessBase;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WhereParamGuard;

/**
 * DA-036-R001 の SQL をインメモリで再現する実装。
 */
public class GmtkstsDataAccessImpl extends DataAccessBase implements GmtkstsDataAccess {

    private final List<GmtkstsRecord> rows = new ArrayList<>();

    /**
     * テストデータを追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public GmtkstsDataAccessImpl addRow(GmtkstsRecord row) {
        rows.add(Objects.requireNonNull(row, "row"));
        return this;
    }


    /**
     * 保持行を参照する。
     *
     * @return 保持行
     */
    public List<GmtkstsRecord> rows() {
        return List.copyOf(rows);
    }

    @Override
    public Optional<GmtkstsRecord> findByGasMtstNo(Long gasMtstNo) {
        Long safeGasMtstNo = safeGasMtstNo(gasMtstNo);
        return rows.stream()
                .filter(row -> Objects.equals(safeGasMtstNo(row.gasMtstNo()), safeGasMtstNo))
                .findFirst();
    }

    private static Long safeGasMtstNo(Long value) {
        return WhereParamGuard.complement(value, "gasMtstNo", 0L);
    }
}
