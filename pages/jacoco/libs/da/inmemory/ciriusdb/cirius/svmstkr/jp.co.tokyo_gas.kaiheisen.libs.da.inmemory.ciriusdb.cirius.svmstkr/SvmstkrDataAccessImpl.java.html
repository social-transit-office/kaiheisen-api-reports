/** SVMSTKR データアクセスのインメモリ実装。 */

    private static final int RKKI_KNRN_JTKB_LENGTH = 2;

    private static final LocalDate COMPLEMENT_YMD = LocalDate.MIN;

    private static final ReflectiveInMemoryRowMapper<SvmstkrColumn, SvmstkrRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(SvmstkrColumn.class, SvmstkrRecord.class);

    private static final AutoNeutralValueResolver<SvmstkrColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(SvmstkrColumn.class, SvmstkrRecord.class);

    private static final Comparator<EnumMap<SvmstkrColumn, Object>>
            ORDER_BY_RKKI_END_YMD_DESC_MTST_NO =
            Comparator.comparing(
                    (EnumMap<SvmstkrColumn, Object> row) ->
                        normalizeYmd(localDateValue(row, SvmstkrColumn.RKKI_END_YMD)),
                    Comparator.nullsLast(Comparator.reverseOrder()))
                    .thenComparing(
                        row -> normalizeMtstNo(longValue(row, SvmstkrColumn.MTST_NO)),
                            Comparator.nullsLast(Comparator.naturalOrder()));

    private static final Comparator<EnumMap<SvmstkrColumn, Object>>
            ORDER_BY_MTST_NO_RKKI_STT_YMD_DESC =
            Comparator.comparing(
                    (EnumMap<SvmstkrColumn, Object> row) ->
                        normalizeMtstNo(longValue(row, SvmstkrColumn.MTST_NO)),
                    Comparator.nullsLast(Comparator.naturalOrder()))
                    .thenComparing(
                            (EnumMap<SvmstkrColumn, Object> row) ->
                            normalizeYmd(localDateValue(row, SvmstkrColumn.RKKI_STT_YMD)),
                            Comparator.nullsLast(Comparator.reverseOrder()));

    private static final Comparator<EnumMap<SvmstkrColumn, Object>> ORDER_BY_MTST_NO =
            Comparator.comparing(
                    (EnumMap<SvmstkrColumn, Object> row) ->
                    normalizeMtstNo(longValue(row, SvmstkrColumn.MTST_NO)),
                    Comparator.nullsLast(Comparator.naturalOrder()));

    private static final Comparator<EnumMap<SvmstkrColumn, Object>> ORDER_BY_SKNO_DESC =
            Comparator.comparing(
                (EnumMap<SvmstkrColumn, Object> row) -> normalizeSkno(integerValue(row, SvmstkrColumn.SKNO)),
                    Comparator.nullsLast(Comparator.reverseOrder()));

    private static final Comparator<EnumMap<SvmstkrColumn, Object>>
            ORDER_BY_RKKI_STT_YMD_DESC =
                Comparator.comparing(
                        (EnumMap<SvmstkrColumn, Object> row) ->
                        normalizeYmd(localDateValue(row, SvmstkrColumn.RKKI_STT_YMD)),
                        Comparator.nullsLast(Comparator.reverseOrder()));

    private final List<EnumMap<SvmstkrColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<SvmstkrWriteRecord, SvmstkrColumn> insertSupport;

    private final InMemoryTargetColumnUpdateSupport<SvmstkrWriteRecord, SvmstkrColumn> updateSupport;

    /**
     * デフォルトの時計と監査設定で初期化する。
     */
    public SvmstkrDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    SvmstkrDataAccessImpl(
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(
                        SvmstkrWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        SvmstkrWriteDefinition.INSTANCE,
                        NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用に行データを追加する。
     *
     * @param row 追加する行
     * @return このインスタンス
     */
    public SvmstkrDataAccessImpl addRow(InternalRow row) {
        rows.add(toRowMap(Objects.requireNonNull(row, "row")));
        return this;
    }

    @Override
    public int insert(SvmstkrWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values -> auditColumnAutoSetter.applyForInsert(
                                new MapBackedAuditColumnAccess<>(SvmstkrColumn.class, values),
                                context,
                                clock)));
        return 1;
    }

    @Override
    public Optional<SvmstkrRecord> findBySvkyNoAndRkkiSttYmdAndRkkiKnrnJtkb(
            Long svkyNo, LocalDate rkkiSttYmd, String rkkiKnrnJtkb) {
        Long safeSvkyNo = safeSvkyNo(svkyNo, "svkyNo");
        LocalDate safeRkkiSttYmd = safeYmd(rkkiSttYmd, "rkkiSttYmd");
        String safeRkkiKnrnJtkb = safeRkkiKnrnJtkb(rkkiKnrnJtkb, "rkkiKnrnJtkb");
        return rows.stream()
            .filter(row -> Objects.equals(normalizeSvkyNo(longValue(row, SvmstkrColumn.SVKY_NO)), safeSvkyNo))
            .filter(row -> isOnOrBefore(row, SvmstkrColumn.RKKI_STT_YMD, safeRkkiSttYmd))
            .filter(row -> !Objects.equals(normalizeRkkiKnrnJtkb(stringValue(row, SvmstkrColumn.RKKI_KNRN_JTKB)), safeRkkiKnrnJtkb))
                .sorted(ORDER_BY_RKKI_END_YMD_DESC_MTST_NO)
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }

    private static boolean isOnOrBefore(
            Map<SvmstkrColumn, Object> row,
            SvmstkrColumn column,
            LocalDate baseDate) {
        LocalDate value = normalizeYmd(dateValue(row, column));
        return !value.isAfter(normalizeYmd(baseDate));
    }

    private static boolean isOnOrAfter(
            Map<SvmstkrColumn, Object> row,
            SvmstkrColumn column,
            LocalDate baseDate) {
        LocalDate value = normalizeYmd(dateValue(row, column));
        return !value.isBefore(normalizeYmd(baseDate));
    }

        return values;
    }

    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }

    private static Boolean toBooleanFlag(String value) {
        if ("1".equals(value)) {
            return Boolean.TRUE;
        }
        if ("0".equals(value)) {
            return Boolean.FALSE;
        }
        return null;
    }
