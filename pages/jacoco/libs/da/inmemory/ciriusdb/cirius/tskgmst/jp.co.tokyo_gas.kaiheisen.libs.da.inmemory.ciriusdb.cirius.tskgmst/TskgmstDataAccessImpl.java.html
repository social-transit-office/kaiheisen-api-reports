    private static final String INVALID_STATUS_CODE = "00";

    private static final LocalDate COMPLEMENT_YMD = LocalDate.of(1, 1, 1);

    private static final ReflectiveInMemoryRowMapper<TskgmstColumn, TskgmstRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(TskgmstColumn.class, TskgmstRecord.class);

    private static final AutoNeutralValueResolver<TskgmstColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(TskgmstColumn.class, TskgmstRecord.class);

    private final List<EnumMap<TskgmstColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryTargetColumnUpdateSupport<TskgmstWriteRecord, TskgmstColumn>
            updateSupport;

    /** デフォルト設定で初期化する。 */
    public TskgmstDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    TskgmstDataAccessImpl(SystemDateTimeProvider clock) {
        this(clock, new CiriusAuditColumnAutoSetter());
    }

    TskgmstDataAccessImpl(
            SystemDateTimeProvider clock, CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter =
                Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport =
                new InMemoryInsertSupport<>(TskgmstWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
        this.updateSupport =
                new InMemoryTargetColumnUpdateSupport<>(
                        TskgmstWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テスト用行を追加する。
     *
     * @param row 行データ
     * @return このインスタンス
     */
    public TskgmstDataAccessImpl addRow(InternalRow row) {
        Objects.requireNonNull(row, "row");
        return addRecord(
                new TskgmstRecord(
                        row.gasMtstNo(),
                        row.tksuSvkyNo(),
                        row.skno(),
                        row.tksuSvkkJtkb(),
                        toLocalDate(row.tkysYmd()),
                        toLocalDate(row.tyenYmd()),
                        row.regTsmp(),
                        row.regUsrId(),
                        row.updTsmp(),
                        row.updUsrId(),
                        row.updKtnKinoId(),
                        row.recUpdJnjNo()));
    }

    private TskgmstDataAccessImpl addRecord(TskgmstRecord row) {
        TskgmstRecord safeRow = Objects.requireNonNull(row, "row");
        EnumMap<TskgmstColumn, Object> values = ROW_MAPPER.toRawValueMap(safeRow);
        NEUTRAL_VALUE_RESOLVER.complement(values, TskgmstWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        return this;
    }

    /**
     * 保持している行を read record で返す。
     *
     * @return 行一覧
     */
    public List<TskgmstRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(TskgmstWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(
                insertSupport.prepare(
                        record,
                        values ->
                                auditColumnAutoSetter.applyForInsert(
                                        new MapBackedAuditColumnAccess<>(
                                                TskgmstColumn.class, values),
                                        context,
                                        clock)));
        return 1;
    }

    @Override
    public int updateByGasMtstNoAndTksuSvkyNoAndSknoAndRecUpdJnjNo(

        Objects.requireNonNull(targetColumns, "targetColumns");

        Long safeTksuSvkyNo = safeLong(updateKey.tksuSvkyNo(), "tksuSvkyNo");

        return InMemoryOptimisticUpdateSupport.update(
                rows,
            row ->
                Objects.equals(row.get(TskgmstColumn.GAS_MTST_NO), safeGasMtstNo)
                    && Objects.equals(
                        row.get(TskgmstColumn.TKSU_SVKY_NO), safeTksuSvkyNo)
                    && Objects.equals(row.get(TskgmstColumn.SKNO), safeSkno),
                row ->
                        updateSupport.prepare(
                                record,
                                targetColumns,
                                values ->
                                        auditColumnAutoSetter.applyForUpdate(
                                                new MapBackedAuditColumnAccess<>(
                                                        TskgmstColumn.class, values),
                                                context,
                                                clock,
                                                toInteger(row.get(TskgmstColumn.REC_UPD_JNJ_NO)))),
                updateSupport::apply,
                TskgmstColumn.REC_UPD_JNJ_NO,
                updateKey.recUpdJnjNo(),
                TskgmstDataAccess.CallSpecs
                        .UPDATE_BY_GAS_MTST_NO_AND_TKSU_SVKY_NO_AND_SKNO_AND_REC_UPD_JNJ_NO
                        .identifier(),
                () ->
                        new DataInconsistencyException(
                                "TSKGMST",

    private static Comparator<EnumMap<TskgmstColumn, Object>> descComparator(TskgmstColumn column) {
        return (left, right) -> compareDesc(left.get(column), right.get(column));
    }

    @SuppressWarnings({"rawtypes", "unchecked"})
    private static int compareDesc(Object left, Object right) {
        if (left == right) {
            return 0;
        }
        if (left == null) {
            return 1;
        }
        if (right == null) {
            return -1;
        }
        return ((Comparable) right).compareTo(left);
    }

    private static boolean notAfter(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && !localDate.isAfter(baseDate);
    }

    private static boolean notBefore(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && !localDate.isBefore(baseDate);
    }

    private static boolean isBefore(Object value, LocalDate baseDate) {
        LocalDate localDate = (LocalDate) value;
        return localDate != null && localDate.isBefore(baseDate);
    }

    private static Long safeLong(Long value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0L);
    }

    private static Integer safeInt(Integer value, String paramName) {
        return WhereParamGuard.complement(value, paramName, 0);
    }

    private static LocalDate safeYmd(LocalDate value, String paramName) {
        return WhereParamGuard.complement(value, paramName, COMPLEMENT_YMD);
    }

    private static String safeTksuSvkkJtkb(String value, String paramName) {
        return WhereParamGuard.fixedLengthNotNull(value, paramName, 2);
    }

    private static Integer toInteger(Object value) {
        if (value instanceof Number number) {
            return number.intValue();
        }
        return value == null ? null : Integer.valueOf(value.toString());
    }

    private static LocalDate toLocalDate(int ymd) {
        if (ymd <= 0) {
            return null;
        }
        return LocalDate.of(ymd / 10000, (ymd / 100) % 100, ymd % 100);
    }
