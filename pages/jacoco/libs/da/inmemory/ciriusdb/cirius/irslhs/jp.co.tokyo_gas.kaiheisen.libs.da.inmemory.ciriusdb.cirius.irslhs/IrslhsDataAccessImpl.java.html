package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.cirius.irslhs;

import java.util.Optional;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/**
 * 開閉栓ＩＲＩＳ請求回収基盤送信ログ（閉栓）（IRSLHS）の InMemory 実装。
 */
public class IrslhsDataAccessImpl extends DataAccessBase implements IrslhsDataAccess {

    private static final ReflectiveInMemoryRowMapper<IrslhsColumn, IrslhsRecord> ROW_MAPPER =
            ReflectiveInMemoryRowMapper.create(IrslhsColumn.class, IrslhsRecord.class);

    private static final AutoNeutralValueResolver<IrslhsColumn> NEUTRAL_VALUE_RESOLVER =
            AutoNeutralValueResolver.create(IrslhsColumn.class, IrslhsRecord.class);

    private final List<EnumMap<IrslhsColumn, Object>> rows = new ArrayList<>();

    private final SystemDateTimeProvider clock;

    private final CiriusAuditColumnAutoSetter auditColumnAutoSetter;

    private final InMemoryInsertSupport<IrslhsWriteRecord, IrslhsColumn> insertSupport;

    private int nextIsrkLogTsno = 1;

    /** デフォルト依存で初期化する。 */
    public IrslhsDataAccessImpl() {
        this(new DefaultSystemDateTimeProvider(), new CiriusAuditColumnAutoSetter());
    }

    IrslhsDataAccessImpl(
            SystemDateTimeProvider clock,
            CiriusAuditColumnAutoSetter auditColumnAutoSetter) {
        this.clock = Objects.requireNonNull(clock, "clock");
        this.auditColumnAutoSetter = Objects.requireNonNull(auditColumnAutoSetter, "auditColumnAutoSetter");
        this.insertSupport = new InMemoryInsertSupport<>(IrslhsWriteDefinition.INSTANCE, NEUTRAL_VALUE_RESOLVER);
    }

    /**
     * テストデータを 1 行追加する。
     *
     * @param row 追加行
     * @return このインスタンス
     */
    public IrslhsDataAccessImpl addRow(IrslhsRecord row) {
        EnumMap<IrslhsColumn, Object> values = ROW_MAPPER.toRawValueMap(Objects.requireNonNull(row, "row"));
        NEUTRAL_VALUE_RESOLVER.complement(values, IrslhsWriteDefinition.INSTANCE.columnOrder());
        rows.add(values);
        if (row.isrkLogTsno() != null) {
            nextIsrkLogTsno = Math.max(nextIsrkLogTsno, row.isrkLogTsno() + 1);
        }
        return this;
    }

    /**
     * 保持中の行を返す。
     *
     * @return 保持行
     */
    public List<IrslhsRecord> getRows() {
        return rows.stream().map(ROW_MAPPER::toRecord).toList();
    }

    @Override
    public int insert(IrslhsWriteRecord record, AuditContext context) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(context, "context");
        rows.add(insertSupport.prepare(
                record,
                values -> {
                    values.put(IrslhsColumn.ISRK_LOG_TSNO, nextIsrkLogTsno++);
                    auditColumnAutoSetter.applyForInsert(
                            new MapBackedAuditColumnAccess<>(IrslhsColumn.class, values),
                            context,
                            clock);
                }));
        return 1;
    }

    @Override
    public Optional<IrslhsRecord> findByPK(Integer isrkLogTsno) {
        Objects.requireNonNull(isrkLogTsno, "isrkLogTsno");
        return rows.stream()
            .filter(row -> Objects.equals(row.get(IrslhsColumn.ISRK_LOG_TSNO), isrkLogTsno))
                .map(ROW_MAPPER::toRecord)
                .findFirst();
    }
}
