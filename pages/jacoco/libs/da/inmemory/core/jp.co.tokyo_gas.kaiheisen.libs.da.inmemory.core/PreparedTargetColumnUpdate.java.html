package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.util.EnumMap;
import java.util.List;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;

/**
 * targetColumns 指定型 update の更新 payload を保持する immutable helper。
 *
 * @param <TColumn> 列挙型
 */
public record PreparedTargetColumnUpdate<TColumn extends Enum<TColumn> & DataAccessColumn>(
        List<TColumn> targetColumns,
        EnumMap<TColumn, Object> values
) {

    /**
     * 生成時に defensive copy する。
     */
    public PreparedTargetColumnUpdate {
        targetColumns = List.copyOf(Objects.requireNonNull(targetColumns, "targetColumns"));
        values = new EnumMap<>(Objects.requireNonNull(values, "values"));
    }

    /**
     * 更新 payload を生成する。
     *
     * @param targetColumns 更新対象列
     * @param values 列値
     * @param <TColumn> 列挙型
     * @return 更新 payload
     */
    public static <TColumn extends Enum<TColumn> & DataAccessColumn> PreparedTargetColumnUpdate<TColumn> of(
            List<TColumn> targetColumns,
            EnumMap<TColumn, Object> values
    ) {
        return new PreparedTargetColumnUpdate<>(targetColumns, values);
    }

    /**
     * 空の更新 payload を返す。
     *
     * @param columnClass 列挙型
     * @param <TColumn> 列挙型
     * @return 空の更新 payload
     */
    public static <TColumn extends Enum<TColumn> & DataAccessColumn> PreparedTargetColumnUpdate<TColumn> empty(
            Class<TColumn> columnClass
    ) {
        Objects.requireNonNull(columnClass, "columnClass");
        return new PreparedTargetColumnUpdate<>(List.of(), new EnumMap<>(columnClass));
    }

    /**
     * 更新対象列が空かを返す。
     *
     * @return 空の場合は true
     */
    public boolean isEmpty() {
        return targetColumns.isEmpty();
    }
}
