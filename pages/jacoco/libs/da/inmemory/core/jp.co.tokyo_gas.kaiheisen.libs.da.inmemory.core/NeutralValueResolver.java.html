package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.util.Collection;
import java.util.EnumMap;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;

/**
 * 列ごとの neutral value 解決と一括補完を担う契約。
 *
 * @param <TColumn> 列挙型
 */
@FunctionalInterface
public interface NeutralValueResolver<TColumn extends Enum<TColumn> & DataAccessColumn> {

    /**
     * 指定列の neutral value を返す。
     *
     * @param column 列
     * @return neutral value。補完しない場合は empty
     */
    Optional<Object> neutralValue(TColumn column);

    /**
     * 指定列群の null 値を neutral value で補完する。
     *
     * @param values 対象値マップ
     * @param columns 補完対象列
     */
    default void complement(EnumMap<TColumn, Object> values, Collection<? extends TColumn> columns) {
        Objects.requireNonNull(values, "values");
        Objects.requireNonNull(columns, "columns");
        for (TColumn column : columns) {
            TColumn nonNullColumn = Objects.requireNonNull(column, "columns contains null");
            if (values.get(nonNullColumn) != null) {
                continue;
            }
            Optional<Object> resolved = Objects.requireNonNull(
                    neutralValue(nonNullColumn),
                    "neutralValue returned null"
            );
            resolved.ifPresent(value -> values.put(nonNullColumn, value));
        }
    }
}
