package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.lang.reflect.Constructor;
import java.lang.reflect.Method;
import java.lang.reflect.RecordComponent;
import java.util.EnumMap;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.FixedLengthFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.ReflectiveValueExtractors;

/**
 * Column enum と Record コンポーネントの名前照合により InMemory 行変換を自動実行するマッパー。
 *
 * <p>{@code toRawValueMap}: Record の各コンポーネント値をそのまま列マップへ格納する。
 *
 * <p>{@code toRecord}: 列マップから Record を構築する。 String 型は末尾の半角/全角空白除去、Long 型はゼロ→null 変換を適用する。
 *
 * @param <TColumn> Column enum 型
 * @param <TReadRecord> Read Record 型（Java record）
 */
public final class ReflectiveInMemoryRowMapper<
                TColumn extends Enum<TColumn> & DataAccessColumn, TReadRecord extends Record>
        implements InMemoryRowMapper<TColumn, TReadRecord> {

    private final Class<TColumn> columnClass;

    private final Class<TReadRecord> recordClass;

    private final ColumnMapping<TColumn>[] columnMappings;

    private final Constructor<TReadRecord> recordConstructor;

    private final RecordComponent[] recordComponents;

    private ReflectiveInMemoryRowMapper(
            Class<TColumn> columnClass,
            Class<TReadRecord> recordClass,
            ColumnMapping<TColumn>[] columnMappings,
            Constructor<TReadRecord> recordConstructor,
            RecordComponent[] recordComponents) {
        this.columnClass = columnClass;
        this.recordClass = recordClass;
        this.columnMappings = columnMappings;
        this.recordConstructor = recordConstructor;
        this.recordComponents = recordComponents;
    }

    /**
     * マッパーを構築する。
     *
     * @param <C> Column enum 型
     * @param <R> Read Record 型
     * @param columnClass Column enum クラス
     * @param recordClass Read Record クラス
     * @return マッパーインスタンス
     */
    @SuppressWarnings("unchecked")
    public static <C extends Enum<C> & DataAccessColumn, R extends Record>
            ReflectiveInMemoryRowMapper<C, R> create(Class<C> columnClass, Class<R> recordClass) {
        Objects.requireNonNull(columnClass, "columnClass");
        Objects.requireNonNull(recordClass, "recordClass");

        RecordComponent[] components = recordClass.getRecordComponents();
        Map<String, Integer> componentIndexByName = new HashMap<>();
        for (int i = 0; i < components.length; i++) {
            componentIndexByName.put(components[i].getName(), i);
        }

        C[] columns = columnClass.getEnumConstants();
        ColumnMapping<C>[] mappings = new ColumnMapping[components.length];
        for (C column : columns) {
            String camelName = ReflectiveValueExtractors.snakeToCamel(column.name());
            Integer idx = componentIndexByName.get(camelName);
            if (idx != null) {
                components[idx].getAccessor().setAccessible(true);
                mappings[idx] = new ColumnMapping<>(column, components[idx].getAccessor());
            }
        }

        Class<?>[] paramTypes = new Class<?>[components.length];
        for (int i = 0; i < components.length; i++) {
            paramTypes[i] = components[i].getType();
        }
        Constructor<R> ctor;
        try {
            ctor = recordClass.getDeclaredConstructor(paramTypes);
            ctor.setAccessible(true);
        } catch (NoSuchMethodException e) {
            throw new IllegalArgumentException(
                    "Record canonical constructor not found: " + recordClass.getName(), e);
        }

        return new ReflectiveInMemoryRowMapper<>(
                columnClass, recordClass, mappings, ctor, components);
    }

    @Override
    public EnumMap<TColumn, Object> toRawValueMap(TReadRecord record) {
        Objects.requireNonNull(record, "record");
        EnumMap<TColumn, Object> values = new EnumMap<>(columnClass);
        for (ColumnMapping<TColumn> mapping : columnMappings) {
            if (mapping == null) {
                continue;
            }
            try {
                values.put(mapping.column, mapping.accessor.invoke(record));
            } catch (ReflectiveOperationException e) {
                throw new IllegalStateException(
                        "Failed to invoke accessor for column: " + mapping.column.name(), e);
            }
        }
        return values;
    }

    @Override
    public TReadRecord toRecord(Map<TColumn, Object> values) {
        Objects.requireNonNull(values, "values");
        Object[] args = new Object[recordComponents.length];
        for (int i = 0; i < columnMappings.length; i++) {
            ColumnMapping<TColumn> mapping = columnMappings[i];
            if (mapping == null) {
                args[i] = null;
                continue;
            }
            Object raw = values.get(mapping.column);
            args[i] = convertForRead(raw, recordComponents[i].getType());
        }
        try {
            return recordConstructor.newInstance(args);
        } catch (ReflectiveOperationException e) {
            throw new IllegalStateException(
                    "Failed to construct record: " + recordClass.getName(), e);
        }
    }

    /** InMemory マップ値を Read Record 型に変換する。 String は末尾の半角/全角空白除去、Long/Integer はゼロ→null を適用する。 */
    private static Object convertForRead(Object value, Class<?> targetType) {
        if (value == null) {
            return null;
        }
        if (targetType == Boolean.class && value instanceof String s) {
            String normalized = s.trim();
            if (normalized.isEmpty()) {
                return null;
            }
            String lower = normalized.toLowerCase(java.util.Locale.ROOT);
            if (lower.equals("1") || lower.equals("y") || lower.equals("t") || lower.equals("true")) {
                return Boolean.TRUE;
            }
            if (lower.equals("0") || lower.equals("n") || lower.equals("f") || lower.equals("false")) {
                return Boolean.FALSE;
            }
            return null;
        }
        if (targetType == String.class && value instanceof String s) {
            return FixedLengthFormatter.suppressSpace(s);
        }
        if (targetType == Long.class && value instanceof Long l) {
            return l == 0L ? null : l;
        }
        if (targetType == Integer.class && value instanceof Integer i) {
            return i == 0 ? null : i;
        }
        return value;
    }

    private record ColumnMapping<C>(C column, Method accessor) { }
}
