package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.util.EnumMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.function.Consumer;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.TableWriteDefinition;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.WriteColumnResolver;

/**
 * targetColumns 指定型 update payload を準備する reusable helper。
 *
 * @param <TRecord> write record 型
 * @param <TColumn> 列型
 */
public final class InMemoryTargetColumnUpdateSupport<
        TRecord, TColumn extends Enum<TColumn> & DataAccessColumn> {

    private final TableWriteDefinition<TRecord, TColumn> definition;

    private final NeutralValueResolver<TColumn> neutralValueResolver;

    /**
     * helper を生成する。
     *
     * @param definition テーブル write 定義
     * @param neutralValueResolver neutral value 解決
     */
    public InMemoryTargetColumnUpdateSupport(
            TableWriteDefinition<TRecord, TColumn> definition,
            NeutralValueResolver<TColumn> neutralValueResolver) {
        this.definition = Objects.requireNonNull(definition, "definition");
        this.neutralValueResolver =
                Objects.requireNonNull(neutralValueResolver, "neutralValueResolver");
    }

    /**
     * update payload を準備する。
     *
     * @param record write record
     * @param targetColumns 公開 targetColumns
     * @param defaultsApplier default 値適用
     * @return update payload
     */
    public PreparedTargetColumnUpdate<TColumn> prepare(
            TRecord record,
            Set<TColumn> targetColumns,
            Consumer<EnumMap<TColumn, Object>> defaultsApplier) {
        Objects.requireNonNull(record, "record");
        Objects.requireNonNull(defaultsApplier, "defaultsApplier");

        List<TColumn> orderedPublicColumns =
                WriteColumnResolver.resolveUpdatableColumns(definition, targetColumns, false);
        if (orderedPublicColumns.isEmpty()) {
            return PreparedTargetColumnUpdate.empty(definition.columnClass());
        }

        List<TColumn> effectiveColumns =
                WriteColumnResolver.effectiveColumns(
                        orderedPublicColumns, definition.updateManagedColumns());
        EnumMap<TColumn, Object> values = WriteColumnResolver.extractValues(definition, record);
        defaultsApplier.accept(values);
        neutralValueResolver.complement(values, effectiveColumns);
        return PreparedTargetColumnUpdate.of(effectiveColumns, values);
    }

    /**
     * 準備済み payload を保存行へ反映する。
     *
     * @param row 保存行
     * @param prepared 更新 payload
     */
    public void apply(Map<TColumn, Object> row, PreparedTargetColumnUpdate<TColumn> prepared) {
        Objects.requireNonNull(row, "row");
        Objects.requireNonNull(prepared, "prepared");
        for (TColumn column : prepared.targetColumns()) {
            row.put(column, prepared.values().get(column));
        }
    }
}
