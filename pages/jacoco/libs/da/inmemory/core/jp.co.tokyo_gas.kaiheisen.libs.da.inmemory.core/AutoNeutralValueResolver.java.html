package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.lang.reflect.RecordComponent;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.ReflectiveValueExtractors;

/**
 * Record コンポーネントの型情報から neutral value を自動判定する {@link NeutralValueResolver} 実装。
 *
 * <p>判定ルール:
 *
 * <ul>
 *   <li>{@link String} → {@code ""}
 *   <li>{@link Long} / {@code long} → {@code 0L}
 *   <li>{@link Integer} / {@code int} → {@code 0}
 *   <li>{@link Short} / {@code short} → {@code (short) 0}
 *   <li>{@link BigDecimal} → {@link BigDecimal#ZERO}
 *   <li>{@link LocalDate} → 補完しない（{@link Optional#empty()}）
 *   <li>不明型 → 補完しない
 * </ul>
 *
 * @param <TColumn> Column enum 型
 */
public final class AutoNeutralValueResolver<TColumn extends Enum<TColumn> & DataAccessColumn>
        implements NeutralValueResolver<TColumn> {

    private final Map<TColumn, Optional<Object>> cache;

    private AutoNeutralValueResolver(Map<TColumn, Optional<Object>> cache) {
        this.cache = cache;
    }

    /**
     * Column enum と Record クラスから neutral value resolver を構築する。
     *
     * @param <C> Column enum 型
     * @param <R> Record 型（read model）
     * @param columnClass Column enum クラス
     * @param recordClass Record クラス（全列の型を判定するため read model を使用）
     * @return neutral value resolver
     */
    public static <C extends Enum<C> & DataAccessColumn, R extends Record>
            AutoNeutralValueResolver<C> create(Class<C> columnClass, Class<R> recordClass) {
        Objects.requireNonNull(columnClass, "columnClass");
        Objects.requireNonNull(recordClass, "recordClass");

        Map<String, Class<?>> componentTypes = new HashMap<>();
        for (RecordComponent comp : recordClass.getRecordComponents()) {
            componentTypes.put(comp.getName(), comp.getType());
        }

        Map<C, Optional<Object>> result = new HashMap<>();
        for (C column : columnClass.getEnumConstants()) {
            String camelName = ReflectiveValueExtractors.snakeToCamel(column.name());
            Class<?> type = componentTypes.get(camelName);
            result.put(column, resolveNeutralValue(type));
        }

        return new AutoNeutralValueResolver<>(result);
    }

    @Override
    public Optional<Object> neutralValue(TColumn column) {
        Objects.requireNonNull(column, "column");
        Optional<Object> value = cache.get(column);
        if (value == null) {
            return Optional.empty();
        }
        return value;
    }

    private static Optional<Object> resolveNeutralValue(Class<?> type) {
        if (type == null) {
            return Optional.empty();
        }
        if (type == String.class) {
            return Optional.of("");
        }
        if (type == Long.class || type == long.class) {
            return Optional.of(0L);
        }
        if (type == Integer.class || type == int.class) {
            return Optional.of(0);
        }
        if (type == Short.class || type == short.class) {
            return Optional.of((short) 0);
        }
        if (type == BigDecimal.class) {
            return Optional.of(BigDecimal.ZERO);
        }
        if (type == LocalDate.class) {
            return Optional.empty();
        }
        return Optional.empty();
    }
}
