package jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.core;

import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;
import java.util.function.Predicate;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.ExclusiveException;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessColumn;

/** in-memory delete に対して共通の排他チェックを適用する helper。 */
public final class InMemoryOptimisticDeleteSupport {

    private InMemoryOptimisticDeleteSupport() {
    }

    /**
     * 排他制御付き delete を適用する。
     *
     * @param rows 保存行
     * @param baseMatcher 排他列を除く基本一致条件
     * @param recUpdJnjNoColumn 排他列
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param <TColumn> 列型
     * @return 削除件数
     */
    public static <TColumn extends Enum<TColumn> & DataAccessColumn> int delete(
            List<? extends Map<TColumn, Object>> rows,
            Predicate<Map<TColumn, Object>> baseMatcher,
            TColumn recUpdJnjNoColumn,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Object... optimisticBindValues) {
        return delete(
                rows,
                baseMatcher,
                recUpdJnjNoColumn,
                expectedRecUpdJnjNo,
                daIdentifier,
                () -> new DataInconsistencyException("UNKNOWN", daIdentifier, "{}"),
                optimisticBindValues);
    }

    /**
     * 排他制御付き delete を適用する。
     *
     * @param rows 保存行
     * @param baseMatcher 排他列を除く基本一致条件
     * @param recUpdJnjNoExtractor 行から排他制御順序番号を取得する extractor
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param <TRow> 行型
     * @return 削除件数
     */
    public static <TRow> int delete(
            List<TRow> rows,
            Predicate<TRow> baseMatcher,
            Function<TRow, Integer> recUpdJnjNoExtractor,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Object... optimisticBindValues) {
        return delete(
                rows,
                baseMatcher,
                recUpdJnjNoExtractor,
                expectedRecUpdJnjNo,
                daIdentifier,
                () -> new DataInconsistencyException("UNKNOWN", daIdentifier, "{}"),
                optimisticBindValues);
    }

    /**
     * 排他制御付き delete を適用する。
     *
     * @param rows 保存行
     * @param baseMatcher 排他列を除く基本一致条件
     * @param recUpdJnjNoColumn 排他列
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param notFoundExceptionSupplier 対象データなし時の例外生成
     * @param optimisticBindValues 排他不一致時に復旧用として保持する bind 値
     * @param <TColumn> 列型
     * @return 削除件数
     */
    public static <TColumn extends Enum<TColumn> & DataAccessColumn> int delete(
            List<? extends Map<TColumn, Object>> rows,
            Predicate<Map<TColumn, Object>> baseMatcher,
            TColumn recUpdJnjNoColumn,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Supplier<? extends RuntimeException> notFoundExceptionSupplier,
            Object... optimisticBindValues) {
        return deleteInternal(
                rows,
                baseMatcher,
                row -> toInteger(row.get(recUpdJnjNoColumn)),
                expectedRecUpdJnjNo,
                daIdentifier,
                notFoundExceptionSupplier,
                optimisticBindValues,
                recUpdJnjNoColumn,
                "recUpdJnjNoColumn");
    }

    /**
     * 排他制御付き delete を適用する。
     *
     * @param rows 保存行
     * @param baseMatcher 排他列を除く基本一致条件
     * @param recUpdJnjNoExtractor 行から排他制御順序番号を取得する extractor
     * @param expectedRecUpdJnjNo 排他チェック対象の順序番号。null の場合は排他チェックしない
     * @param daIdentifier 例外に設定する DA 識別子
     * @param notFoundExceptionSupplier 対象データなし時の例外生成
     * @param optimisticBindValues 排他不一致時に復旧用として保持する bind 値
     * @param <TRow> 行型
     * @return 削除件数
     */
    public static <TRow> int delete(
            List<TRow> rows,
            Predicate<TRow> baseMatcher,
            Function<TRow, Integer> recUpdJnjNoExtractor,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Supplier<? extends RuntimeException> notFoundExceptionSupplier,
            Object... optimisticBindValues) {
        return deleteInternal(
                rows,
                baseMatcher,
                recUpdJnjNoExtractor,
                expectedRecUpdJnjNo,
                daIdentifier,
                notFoundExceptionSupplier,
                optimisticBindValues,
                recUpdJnjNoExtractor,
                "recUpdJnjNoExtractor");
    }

    private static <TRow> int deleteInternal(
            List<? extends TRow> rows,
            Predicate<TRow> baseMatcher,
            Function<TRow, Integer> recUpdJnjNoExtractor,
            Integer expectedRecUpdJnjNo,
            String daIdentifier,
            Supplier<? extends RuntimeException> notFoundExceptionSupplier,
            Object[] optimisticBindValues,
            Object extractorDependency,
            String extractorDependencyName) {
        Objects.requireNonNull(rows, "rows");
        Objects.requireNonNull(baseMatcher, "baseMatcher");
        Objects.requireNonNull(extractorDependency, extractorDependencyName);

        Object[] effectiveOptimisticBindValues =
                optimisticBindValues == null ? new Object[0] : optimisticBindValues.clone();
        if (expectedRecUpdJnjNo != null) {
            Objects.requireNonNull(daIdentifier, "daIdentifier");
        }

        int deleted = 0;
        boolean baseMatched = false;
        for (int i = rows.size() - 1; i >= 0; i--) {
            TRow row = rows.get(i);
            if (!baseMatcher.test(row)) {
                continue;
            }
            baseMatched = true;
            if (expectedRecUpdJnjNo != null
                    && !Objects.equals(recUpdJnjNoExtractor.apply(row), expectedRecUpdJnjNo)) {
                continue;
            }
            rows.remove(i);
            deleted++;
        }

        if (deleted > 0 || expectedRecUpdJnjNo == null) {
            return deleted;
        }
        if (baseMatched) {
            throw new ExclusiveException(
                    daIdentifier,
                    null,
                    buildExclusiveBindParameters(effectiveOptimisticBindValues, expectedRecUpdJnjNo));
        }
        throw Objects.requireNonNull(notFoundExceptionSupplier, "notFoundExceptionSupplier").get();
    }

    private static Object[] buildExclusiveBindParameters(
            Object[] optimisticBindValues, Integer expectedRecUpdJnjNo) {
        Object[] bindParameters = Arrays.copyOf(optimisticBindValues, optimisticBindValues.length + 1);
        bindParameters[optimisticBindValues.length] = expectedRecUpdJnjNo;
        return bindParameters;
    }

    private static Integer toInteger(Object value) {
        if (value == null) {
            return null;
        }
        if (value instanceof Number number) {
            return number.intValue();
        }
        return Integer.valueOf(value.toString());
    }
}
