package jp.co.tokyo_gas.kaiheisen.libs.da.core;

/**
 * 固定長フィールドの空白・ゼロ埋めおよび除去を行うユーティリティ。
 */
public final class FixedLengthFormatter {

    private FixedLengthFormatter() {
    }

    private static final char HALF_WIDTH_SPACE = ' ';

    private static final char FULL_WIDTH_SPACE = '　';

    /**
     * 文字列末尾の半角スペース・全角スペースを除去する。
     *
     * @param value 対象文字列
     * @return 末尾の半角/全角空白除去後の文字列。全て空白または {@code null} の場合は {@code null}
     */
    public static String suppressSpace(String value) {
        if (value == null) {
            return null;
        }
        int end = value.length();
        while (end > 0 && isSuppressibleSpace(value.charAt(end - 1))) {
            end--;
        }
        if (end == 0) {
            return null;
        }
        return value.substring(0, end);
    }

    private static boolean isSuppressibleSpace(char character) {
        return character == HALF_WIDTH_SPACE || character == FULL_WIDTH_SPACE;
    }

    /**
     * 文字列先頭のゼロを除去する。
     *
     * @param value 対象文字列
     * @return 先頭ゼロ除去後の文字列。全てゼロまたは {@code null} の場合は {@code null}
     */
    public static String suppressZero(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        int start = 0;
        while (start < normalized.length() && normalized.charAt(start) == '0') {
            start++;
        }
        if (start == normalized.length()) {
            return null;
        }
        return normalized.substring(start);
    }

    /**
     * 文字列を指定長まで末尾半角スペースで埋める。
     *
     * @param value  対象文字列
     * @param length 埋め先の長さ
     * @return スペース埋め後の文字列。{@code null} の場合は {@code null}
     */
    public static String padSpace(String value, int length) {
        String normalized = suppressSpace(value);
        if (normalized == null) {
            return null;
        }
        if (normalized.length() >= length) {
            return normalized;
        }
        StringBuilder builder = new StringBuilder(length);
        builder.append(normalized);
        for (int i = normalized.length(); i < length; i++) {
            builder.append(' ');
        }
        return builder.toString();
    }

    /**
     * 文字列を指定長まで先頭ゼロで埋める。
     *
     * @param value  対象文字列
     * @param length 埋め先の長さ
     * @return ゼロ埋め後の文字列。全てゼロまたは {@code null} の場合は {@code null}
     */
    public static String padZero(String value, int length) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        boolean allZero = true;
        for (int i = 0; i < normalized.length(); i++) {
            if (normalized.charAt(i) != '0') {
                allZero = false;
                break;
            }
        }
        if (allZero) {
            return null;
        }
        if (normalized.length() >= length) {
            return normalized;
        }
        StringBuilder builder = new StringBuilder(length);
        for (int i = normalized.length(); i < length; i++) {
            builder.append('0');
        }
        builder.append(normalized);
        return builder.toString();
    }
}
