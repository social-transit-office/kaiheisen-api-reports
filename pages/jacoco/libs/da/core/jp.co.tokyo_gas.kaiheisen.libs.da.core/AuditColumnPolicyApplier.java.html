package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Collection;
import java.util.Objects;
import java.util.Set;
import java.util.function.Predicate;
import java.util.stream.Collectors;

/**
 * 監査列 policy を適用するユーティリティ。
 */
public final class AuditColumnPolicyApplier {

    private AuditColumnPolicyApplier() {
    }

    /**
     * insert 時の監査列 policy を適用する。
     *
     * @param policy 監査列 policy
     * @param access 列アクセス
     */
    public static void applyForInsert(
            AuditColumnPolicy policy,
            AuditColumnAccess access
    ) {
        apply(policy, DataAccessWriteOperation.INSERT, access, ignored -> true);
    }

    /**
     * update 時の監査列 policy を適用する。
     *
     * @param policy 監査列 policy
     * @param access 列アクセス
     */
    public static void applyForUpdate(
            AuditColumnPolicy policy,
            AuditColumnAccess access
    ) {
        apply(policy, DataAccessWriteOperation.UPDATE, access, ignored -> true);
    }

    /**
     * update 時の監査列 policy を対象列付きで適用する。
     *
     * @param policy 監査列 policy
     * @param access 列アクセス
     * @param targetColumns 更新対象列
     */
    public static void applyForUpdate(
            AuditColumnPolicy policy,
            AuditColumnAccess access,
            Collection<? extends DataAccessColumn> targetColumns
    ) {
        Objects.requireNonNull(targetColumns, "targetColumns");
        Set<String> targetColumnNames = targetColumns.stream()
                .map(column -> Objects.requireNonNull(column, "targetColumns contains null"))
                .map(DataAccessColumn::columnName)
                .collect(Collectors.toUnmodifiableSet());
        apply(policy, DataAccessWriteOperation.UPDATE, access, targetColumnNames::contains);
    }

    /**
     * 操作種別に応じて監査列 policy を適用する。
     *
     * @param policy 監査列 policy
     * @param operation 操作種別
     * @param access 列アクセス
     */
    public static void apply(
            AuditColumnPolicy policy,
            DataAccessWriteOperation operation,
            AuditColumnAccess access
    ) {
        apply(policy, operation, access, ignored -> true);
    }

    private static void apply(
            AuditColumnPolicy policy,
            DataAccessWriteOperation operation,
            AuditColumnAccess access,
            Predicate<String> shouldApply
    ) {
        Objects.requireNonNull(policy, "policy");
        Objects.requireNonNull(operation, "operation");
        Objects.requireNonNull(access, "access");
        Objects.requireNonNull(shouldApply, "shouldApply");

        AuditColumnValueResolver valueResolver = policy.valueResolver();
        for (AuditColumnRule rule : policy.pattern().rules()) {
            String columnName = rule.columnName();
            if (!rule.appliesTo(operation) || !shouldApply.test(columnName) || !access.hasColumn(columnName)) {
                continue;
            }
            valueResolver.resolve(rule.valueSlot()).ifPresent(value -> access.setValue(columnName, value));
        }
    }
}
