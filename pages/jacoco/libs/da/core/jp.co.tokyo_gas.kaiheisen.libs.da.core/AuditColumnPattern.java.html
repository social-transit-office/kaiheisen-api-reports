package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.List;
import java.util.Set;

/**
 * 監査列 policy の列パターン定義。
 */
public enum AuditColumnPattern {

    PATTERN1(pattern1Rules()),
    PATTERN2(pattern2Rules()),
    PATTERN3(pattern3Rules());

    private final List<AuditColumnRule> rules;

    AuditColumnPattern(List<AuditColumnRule> rules) {
        this.rules = List.copyOf(rules);
    }

    /**
     * パターンに含まれる監査列ルール一覧を返す。
     *
     * @return 監査列ルール一覧
     */
    public List<AuditColumnRule> rules() {
        return rules;
    }

    private static List<AuditColumnRule> pattern1Rules() {
        return List.of(
                AuditColumnRule.insertOnly("REG_TSMP", AuditColumnValueSlot.CURRENT_DATETIME),
                AuditColumnRule.insertOnly("REG_USR_ID", AuditColumnValueSlot.REGISTER_USER_ID),
                AuditColumnRule.insertAndUpdate("UPD_TSMP", AuditColumnValueSlot.CURRENT_DATETIME),
                AuditColumnRule.insertAndUpdate("UPD_USR_ID", AuditColumnValueSlot.UPDATE_USER_ID),
                AuditColumnRule.insertAndUpdate("UPD_KTN_KINO_ID", AuditColumnValueSlot.UPDATE_FUNCTION_ID),
                AuditColumnRule.insertAndUpdate("REC_UPD_JNJ_NO", AuditColumnValueSlot.UPDATE_SEQUENCE_NUMBER)
        );
    }

    private static List<AuditColumnRule> pattern2Rules() {
        return List.of(
                AuditColumnRule.insertOnly("REC_REG_YMD", AuditColumnValueSlot.BUSINESS_DATE),
                AuditColumnRule.insertAndUpdate("REC_UPD_YMD", AuditColumnValueSlot.BUSINESS_DATE),
                AuditColumnRule.insertAndUpdate("REC_UPD_JNJ_NO", AuditColumnValueSlot.UPDATE_SEQUENCE_NUMBER),
                AuditColumnRule.insertAndUpdate("KOSIN_USER_ID", AuditColumnValueSlot.UPDATE_USER_ID),
                AuditColumnRule.insertAndUpdate("KOSIN_TRAN_NO", AuditColumnValueSlot.TRANSACTION_NUMBER),
                AuditColumnRule.insertAndUpdate("KOSIN_TASK_NO", AuditColumnValueSlot.TASK_NUMBER)
        );
    }

    private static List<AuditColumnRule> pattern3Rules() {
        Set<String> pattern3Columns = Set.of("REC_REG_YMD", "REC_UPD_YMD", "REC_UPD_JNJ_NO");
        return pattern2Rules().stream()
                .filter(rule -> pattern3Columns.contains(rule.columnName()))
                .toList();
    }
}
