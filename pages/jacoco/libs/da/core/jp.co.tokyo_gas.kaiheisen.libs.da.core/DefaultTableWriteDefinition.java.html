package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Arrays;
import java.util.Collections;
import java.util.EnumMap;
import java.util.EnumSet;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import java.util.function.Function;
import java.util.stream.Collectors;

/**
 * {@link TableWriteDefinition} の汎用デフォルト実装。
 *
 * <p>各テーブル固有の {@code TableWriteDefinition} を手書きする代わりに、
 * PK 列セットと監査列パターンから全メタデータを自動導出する。
 * 60 テーブルへの横展開においてボイラープレートを排除するための中核クラス。</p>
 *
 * @param <RecordT> write record 型（Java record）
 * @param <ColumnT> canonical 列 enum 型
 */
public final class DefaultTableWriteDefinition<
        RecordT extends Record,
        ColumnT extends Enum<ColumnT> & DataAccessColumn
        > implements TableWriteDefinition<RecordT, ColumnT> {

    private final Class<ColumnT> columnClass;

    private final List<ColumnT> columnOrder;

    private final List<ColumnT> insertManagedColumns;

    private final List<ColumnT> updateManagedColumns;

    private final Set<ColumnT> updatableColumns;

    private final EnumMap<ColumnT, Function<RecordT, Object>> valueExtractors;

    private DefaultTableWriteDefinition(
            Class<ColumnT> columnClass,
            Class<RecordT> recordClass,
            Set<ColumnT> pkColumns,
            AuditColumnEnum<?> auditColumnEnum
    ) {
        this.columnClass = Objects.requireNonNull(columnClass, "columnClass");
        this.columnOrder = List.copyOf(Arrays.asList(columnClass.getEnumConstants()));
        this.insertManagedColumns = resolveManagedColumns(columnClass, auditColumnEnum, DataAccessWriteOperation.INSERT);
        this.updateManagedColumns = resolveManagedColumns(columnClass, auditColumnEnum, DataAccessWriteOperation.UPDATE);
        this.updatableColumns = buildUpdatableColumns(columnClass, pkColumns, auditColumnEnum);
        this.valueExtractors = ReflectiveValueExtractors.build(columnClass, recordClass);
    }

    /**
     * 汎用 write 定義を生成する。
     *
     * @param columnClass canonical 列 enum クラス
     * @param recordClass write record クラス（Java record）
     * @param pkColumns 主キー列の集合
     * @param auditColumnEnum 監査列 enum（{@link NcisCisAuditColumn} 等）
     * @param <RecordT> write record 型
     * @param <ColumnT> 列 enum 型
     * @return write 定義
     */
    public static <RecordT extends Record, ColumnT extends Enum<ColumnT> & DataAccessColumn>
        DefaultTableWriteDefinition<RecordT, ColumnT> of(
            Class<ColumnT> columnClass,
            Class<RecordT> recordClass,
            Set<ColumnT> pkColumns,
            AuditColumnEnum<?> auditColumnEnum
    ) {
        Objects.requireNonNull(pkColumns, "pkColumns");
        Objects.requireNonNull(auditColumnEnum, "auditColumnEnum");
        return new DefaultTableWriteDefinition<>(columnClass, recordClass, pkColumns, auditColumnEnum);
    }

    @Override
    public Class<ColumnT> columnClass() {
        return columnClass;
    }

    @Override
    public List<ColumnT> columnOrder() {
        return columnOrder;
    }

    @Override
    public List<ColumnT> insertManagedColumns() {
        return insertManagedColumns;
    }

    @Override
    public List<ColumnT> updateManagedColumns() {
        return updateManagedColumns;
    }

    @Override
    public Set<ColumnT> updatableColumns() {
        return updatableColumns;
    }

    @Override
    public EnumMap<ColumnT, Object> extractValues(RecordT record) {
        Objects.requireNonNull(record, "record");
        EnumMap<ColumnT, Object> values = new EnumMap<>(columnClass);
        valueExtractors.forEach((column, extractor) -> values.put(column, extractor.apply(record)));
        return values;
    }

    /**
     * 監査列 enum の操作種別ごとの列一覧を解決する共通契約。
     *
     * @param <A> 監査列 enum 型
     */
    public interface AuditColumnEnum<A extends Enum<A> & DataAccessColumn> {

        /**
         * 操作種別ごとの監査列名一覧を返す。
         *
         * @param operation 操作種別
         * @return 監査列名一覧
         */
        List<String> columnNamesFor(DataAccessWriteOperation operation);

        /**
         * 監査列名の全集合を返す。
         *
         * @return 監査列名集合
         */
        Set<String> allColumnNames();
    }

    /**
     * {@link NcisCisAuditColumn} を {@link AuditColumnEnum} として提供する。
     */
    public static final AuditColumnEnum<NcisCisAuditColumn> NCIS_CIS = new AuditColumnEnum<>() {
        @Override
        public List<String> columnNamesFor(DataAccessWriteOperation operation) {
            return NcisCisAuditColumn.columnsFor(operation).stream()
                    .map(NcisCisAuditColumn::name)
                    .toList();
        }

        @Override
        public Set<String> allColumnNames() {
            return EnumSet.allOf(NcisCisAuditColumn.class).stream()
                    .map(NcisCisAuditColumn::name)
                    .collect(Collectors.toSet());
        }
    };

    /**
     * {@link NcisSubsetAuditColumn} を {@link AuditColumnEnum} として提供する。
     *
     * <p>KOSIN 列を持たない NCIS テーブル（TATEMON 等）に使用する。</p>
     */
    public static final AuditColumnEnum<NcisSubsetAuditColumn> NCIS_SUBSET = new AuditColumnEnum<>() {
        @Override
        public List<String> columnNamesFor(DataAccessWriteOperation operation) {
            return NcisSubsetAuditColumn.columnsFor(operation).stream()
                    .map(NcisSubsetAuditColumn::name)
                    .toList();
        }

        @Override
        public Set<String> allColumnNames() {
            return EnumSet.allOf(NcisSubsetAuditColumn.class).stream()
                    .map(NcisSubsetAuditColumn::name)
                    .collect(Collectors.toSet());
        }
    };

    /**
     * {@link CiriusAuditColumn} を {@link AuditColumnEnum} として提供する。
     */
    public static final AuditColumnEnum<CiriusAuditColumn> CIRIUS = new AuditColumnEnum<>() {
        @Override
        public List<String> columnNamesFor(DataAccessWriteOperation operation) {
            return CiriusAuditColumn.columnsFor(operation).stream()
                    .map(CiriusAuditColumn::name)
                    .toList();
        }

        @Override
        public Set<String> allColumnNames() {
            return EnumSet.allOf(CiriusAuditColumn.class).stream()
                    .map(CiriusAuditColumn::name)
                    .collect(Collectors.toSet());
        }
    };

    private static <ColumnT extends Enum<ColumnT> & DataAccessColumn> List<ColumnT> resolveManagedColumns(
            Class<ColumnT> columnClass,
            AuditColumnEnum<?> auditColumnEnum,
            DataAccessWriteOperation operation
    ) {
        return auditColumnEnum.columnNamesFor(operation).stream()
                .map(name -> Enum.valueOf(columnClass, name))
                .toList();
    }

    private static <ColumnT extends Enum<ColumnT> & DataAccessColumn> Set<ColumnT> buildUpdatableColumns(
            Class<ColumnT> columnClass,
            Set<ColumnT> pkColumns,
            AuditColumnEnum<?> auditColumnEnum
    ) {
        Set<String> auditNames = auditColumnEnum.allColumnNames();
        EnumSet<ColumnT> updatable = EnumSet.allOf(columnClass);
        updatable.removeAll(pkColumns);
        updatable.removeIf(col -> auditNames.contains(col.name()));
        return Collections.unmodifiableSet(updatable);
    }
}
