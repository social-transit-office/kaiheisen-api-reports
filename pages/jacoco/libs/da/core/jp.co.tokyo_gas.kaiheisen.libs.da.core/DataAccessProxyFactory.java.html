package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import jakarta.enterprise.context.ContextNotActiveException;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;
import java.lang.reflect.Proxy;
import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext.DataAccessInvocation;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DatabaseException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;

/**
 * DataAccess インターフェース向けの共通ラッパ生成ファクトリです.
 */
public final class DataAccessProxyFactory {

    /** 開始時のイベント名. */
    private static final String DATA_ACCESS_STARTED_EVENT =
            "dataaccess.started";

    /** 開始時の要約. */
    private static final String DATA_ACCESS_STARTED_SUMMARY =
            "DataAccess execution started";

    /** 失敗時のイベント名. */
    private static final String DATA_ACCESS_FAILED_EVENT = "dataaccess.failed";

    /** 失敗時の要約. */
    private static final String DATA_ACCESS_FAILED_SUMMARY =
            "DataAccess execution failed";

    /** 正常終了時のイベント名. */
    private static final String DATA_ACCESS_COMPLETED_EVENT =
            "dataaccess.completed";

    /** 正常終了時の要約. */
    private static final String DATA_ACCESS_COMPLETED_SUMMARY =
            "DataAccess execution completed";

    /** 機能 ID 属性名. */
    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    /** 名称属性名. */
    private static final String NAME_ATTRIBUTE = "name";

    /** パラメータ属性名. */
    private static final String PARAMS_ATTRIBUTE = "params";

    /** SQL 属性名. */
    private static final String SQL_ATTRIBUTE = "sql";

    private DataAccessProxyFactory() {
    }

    /**
     * DataAccess インターフェースの例外変換プロキシを生成します.
     *
     * <p>プロキシは {@link DataAccess} アノテーション付きメソッドで発生した例外を
     * {@link DatabaseException} へ変換し、開始時および失敗時のログを出力する。</p>
     *
     * @param <T>      DataAccess インターフェースの型
     * @param api      DataAccess インターフェースクラス
     * @param delegate 実際の処理を行うデリゲート
     * @param logger   開始時および失敗時ログの出力先
     * @return 例外変換を行う動的プロキシ
     */
    public static <T> T create(
            final Class<T> api,
            final T delegate,
            final CommonDataAccessLogger logger) {
        return createInternal(api, delegate, logger, null);
    }

    /**
     * DataAccess インターフェースの例外変換プロキシを生成します.
     *
     * <p>プロキシは {@link DataAccess} アノテーション付きメソッドで発生した例外を
     * {@link DatabaseException} へ変換し、開始時および失敗時のログを出力する。</p>
     *
     * @param <T> DataAccess インターフェースの型
     * @param api DataAccess インターフェースクラス
     * @param delegate 実際の処理を行うデリゲート
     * @param logger 開始時および失敗時ログの出力先
     * @param requestContext リクエストコンテキスト
     * @return 例外変換を行う動的プロキシ
     */
    public static <T> T create(
            final Class<T> api,
            final T delegate,
            final CommonDataAccessLogger logger,
            final RequestContext requestContext) {
        Objects.requireNonNull(requestContext, "requestContext");
        return createInternal(api, delegate, logger, requestContext);
    }

    private static <T> T createInternal(
            final Class<T> api,
            final T delegate,
            final CommonDataAccessLogger logger,
            final RequestContext requestContext) {
        Objects.requireNonNull(api, "api");
        Objects.requireNonNull(delegate, "delegate");
        Objects.requireNonNull(logger, "logger");
        if (!api.isInterface()) {
            throw new IllegalArgumentException(
                    "api must be interface: " + api.getName());
        }
        return api.cast(Proxy.newProxyInstance(
                api.getClassLoader(),
                new Class<?>[] {api},
                (proxy, method, args) -> {
                    DataAccess meta = method.getAnnotation(DataAccess.class);
                    if (meta == null) {
                        return invokeWithoutWrap(method, delegate, args);
                    }
                    long startNanos = System.nanoTime();
                    boolean phaseMeasurementEnabled =
                            DataAccessPhaseRecorder.isEnabledFor(meta.identifier());
                    Measurement measurement = beginMeasurement(
                            requestContext,
                            meta.identifier(),
                            phaseMeasurementEnabled);
                    DataAccessPhaseRecorder.Scope phaseScope = DataAccessPhaseRecorder.open(
                            phaseMeasurementEnabled,
                            meta.identifier(),
                            safeGetCorrelationId(requestContext).orElse(null),
                            measurement.sequence(),
                            measurement.unavailable(),
                            startNanos,
                            logger);
                    try {
                        logInfoSafely(logger, buildStartedEvent(
                                meta.identifier(),
                                meta.name(),
                                args,
                                requestContext,
                                measurement));
                        Object result = method.invoke(delegate, args);
                        String sql = SqlCapture.getAndClear();
                        if (measurement.enabled()) {
                            logInfoSafely(logger, buildTerminalEvent(
                                    DATA_ACCESS_COMPLETED_EVENT,
                                    DATA_ACCESS_COMPLETED_SUMMARY,
                                    meta.identifier(),
                                    meta.name(),
                                    args,
                                    sql,
                                    requestContext,
                                    measurement,
                                    elapsedMillis(startNanos)));
                        }
                        return result;
                    } catch (InvocationTargetException ex) {
                        Throwable cause = ex.getCause();
                        String sql = SqlCapture.getAndClear();
                        if (measurement.enabled()) {
                            logErrorSafely(logger, buildTerminalEvent(
                                    DATA_ACCESS_FAILED_EVENT,
                                    DATA_ACCESS_FAILED_SUMMARY,
                                    meta.identifier(),
                                    meta.name(),
                                    args,
                                    sql,
                                    requestContext,
                                    measurement,
                                    elapsedMillis(startNanos)), cause);
                        } else if (!(cause instanceof Error)) {
                            logger.error(buildTerminalEvent(
                                    DATA_ACCESS_FAILED_EVENT,
                                    DATA_ACCESS_FAILED_SUMMARY,
                                    meta.identifier(),
                                    meta.name(),
                                    args,
                                    sql,
                                    requestContext,
                                    measurement,
                                    null), cause);
                        }
                        switch (cause) {
                            case Error error -> throw error;
                            case null, default -> {
                                String message = buildFailureMessage(
                                        meta.identifier(),
                                        meta.name(),
                                        args,
                                        sql,
                                        requestContext);
                                throw new DatabaseException(
                                        meta.identifier(),
                                        sql,
                                        args,
                                        new RuntimeException(message, cause));
                            }
                        }
                    } catch (ReflectiveOperationException | RuntimeException exception) {
                        String sql = SqlCapture.getAndClear();
                        logErrorSafely(logger, buildTerminalEvent(
                                DATA_ACCESS_FAILED_EVENT,
                                DATA_ACCESS_FAILED_SUMMARY,
                                meta.identifier(),
                                meta.name(),
                                args,
                                sql,
                                requestContext,
                                measurement,
                                elapsedMillis(startNanos)), exception);
                        throw exception;
                    } finally {
                        SqlCapture.getAndClear();
                        phaseScope.close();
                        endMeasurement(requestContext, measurement);
                    }
                }
        ));
    }

    private static Object invokeWithoutWrap(
            final Method method,
            final Object delegate,
            final Object[] args)
            throws Throwable {
        try {
            return method.invoke(delegate, args);
        } catch (InvocationTargetException ex) {
            Throwable cause = ex.getCause();
            switch (cause) {
                case Error error -> throw error;
                case RuntimeException runtimeException ->
                    throw runtimeException;
                case null, default -> throw new RuntimeException(cause);
            }
        }
    }

    private static LogEvent buildStartedEvent(
            final String identifier,
            final String name,
            final Object[] params,
            final RequestContext requestContext,
            final Measurement measurement) {
        return new LogEvent(
                DATA_ACCESS_STARTED_EVENT,
                DATA_ACCESS_STARTED_SUMMARY,
                LogLayer.DA,
                null,
                null,
                buildAttributes(
                    identifier,
                    name,
                    params,
                    null,
                    requestContext,
                    measurement,
                    null));
    }

    private static LogEvent buildTerminalEvent(
            final String event,
            final String summary,
            final String identifier,
            final String name,
            final Object[] params,
            final String sql,
            final RequestContext requestContext,
            final Measurement measurement,
            final Long latencyMs) {
        return new LogEvent(
                event,
                summary,
                LogLayer.DA,
                null,
                null,
                buildAttributes(
                        identifier,
                        name,
                        params,
                        sql,
                        requestContext,
                        measurement,
                        latencyMs));
    }

    private static Map<String, Object> buildAttributes(
            final String identifier,
            final String name,
            final Object[] params,
            final String sql,
            final RequestContext requestContext,
            final Measurement measurement,
            final Long latencyMs) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        safeGetCorrelationId(requestContext)
            .ifPresent(correlationId ->
                attributes.put(
                    LogField.CORRELATION_ID.key(),
                    correlationId));
        attributes.put(FUNCTION_ID_ATTRIBUTE, identifier);
        attributes.put(NAME_ATTRIBUTE, name);
        String callPath = safeBuildCallPath(requestContext, identifier);
        if (callPath != null) {
            attributes.put(LogField.CALL_PATH.key(), callPath);
        }
        attributes.put(
            PARAMS_ATTRIBUTE,
            params == null ? null : Arrays.asList(params));
        if (sql != null) {
            attributes.put(SQL_ATTRIBUTE, sql);
        }
        if (measurement.sequence() != null) {
            attributes.put(LogField.DA_INVOCATION_SEQUENCE.key(), measurement.sequence());
        }
        if (latencyMs != null) {
            attributes.put(LogField.LATENCY_MS.key(), latencyMs);
        }
        if (measurement.unavailable()) {
            attributes.put(LogField.MEASUREMENT_UNAVAILABLE.key(), true);
        }
        return attributes;
    }

    private static Measurement beginMeasurement(
            final RequestContext requestContext,
            final String identifier,
            final boolean phaseMeasurementEnabled) {
        if (!PerformanceObservabilitySettings.isEnabled() && !phaseMeasurementEnabled) {
            return Measurement.disabled();
        }
        if (requestContext == null) {
            return Measurement.unavailableMeasurement();
        }
        try {
            DataAccessInvocation invocation = requestContext.beginDataAccess(identifier);
            return new Measurement(true, invocation, invocation.sequence(), false);
        } catch (RuntimeException exception) {
            return Measurement.unavailableMeasurement();
        }
    }

    private static void endMeasurement(
            final RequestContext requestContext,
            final Measurement measurement) {
        if (requestContext == null || measurement.invocation() == null) {
            return;
        }
        try {
            requestContext.endDataAccess(measurement.invocation());
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }

    private static long elapsedMillis(final long startNanos) {
        return Math.max(0L, System.nanoTime() - startNanos) / 1_000_000L;
    }

    private static void logInfoSafely(
            final CommonDataAccessLogger logger,
            final LogEvent event) {
        try {
            logger.info(event);
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }

    private static void logErrorSafely(
            final CommonDataAccessLogger logger,
            final LogEvent event,
            final Throwable cause) {
        try {
            logger.error(event, cause);
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }

    private static String buildFailureMessage(
            final String identifier,
            final String name,
            final Object[] params,
            final String sql,
            final RequestContext requestContext) {
        StringBuilder builder = new StringBuilder();
        appendCorrelationPrefix(builder, requestContext);
        builder.append("DataAccess failure: functionId=").append(identifier)
                .append(", name=").append(name);
        String callPath = safeBuildCallPath(requestContext, identifier);
        if (callPath != null) {
            builder.append(", callPath=").append(callPath);
        }
        builder.append(", params=").append(Arrays.toString(params));
        if (sql != null) {
            builder.append(", sql=").append(sql);
        }
        return builder.toString();
    }

    private static void appendCorrelationPrefix(
            final StringBuilder builder,
            final RequestContext requestContext) {
        safeGetCorrelationId(requestContext)
                .ifPresent(correlationId ->
                        builder.append("correlationId=")
                                .append(correlationId)
                                .append(' '));
    }

    private static Optional<String> safeGetCorrelationId(
            final RequestContext requestContext) {
        if (requestContext == null) {
            return Optional.empty();
        }
        try {
            return requestContext.getCorrelationId();
        } catch (ContextNotActiveException ex) {
            return Optional.empty();
        }
    }

    private static String safeBuildCallPath(
            final RequestContext requestContext,
            final String identifier) {
        if (requestContext == null) {
            return null;
        }
        try {
            return requestContext.buildCallPath(identifier);
        } catch (ContextNotActiveException ex) {
            return null;
        }
    }

    private record Measurement(
            boolean enabled,
            DataAccessInvocation invocation,
            Long sequence,
            boolean unavailable) {

        private static Measurement disabled() {
            return new Measurement(false, null, null, false);
        }

        private static Measurement unavailableMeasurement() {
            return new Measurement(true, null, null, true);
        }
    }
}
