package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Objects;

/**
 * WHERE 句パラメータの null 防御と neutral value 補完を提供するユーティリティ。
 *
 * <p>NOT NULL 列を WHERE 条件に使う場合、API 層の nullable な Java 型（{@code Long}, {@code Integer} 等）で
 * null が渡されることがある。本クラスの {@link #complement} で WHERE 専用の neutral value に置換し、
 * DB 上の NOT NULL 値と安全に比較できるようにする。</p>
 *
 * <p>固定長 NOT NULL 文字列の WHERE 条件項目は write 用 neutral value と切り離し、
 * {@link #fixedLengthNotNull(String, String, int)} で半角スペース 1 文字へ補完する。</p>
 *
 * <p>{@link #requireNonNull} は {@link Objects#requireNonNull} と同等だが、WHERE パラメータの防御を
 * 本クラスに集約する目的で提供する。</p>
 */
public final class WhereParamGuard {

    private WhereParamGuard() {
    }

    /**
     * WHERE 句パラメータが null の場合、neutral value で補完して返す。
     *
     * <p>DB の NOT NULL 列に対応するパラメータで、null を neutral value（{@code 0L}, {@code 0}, {@code ""} 等）
     * に置換して検索条件として安全に使えるようにする。</p>
     *
     * @param value パラメータ値（null 許容）
     * @param paramName パラメータ名（デバッグ/ログ用）
     * @param neutralValue null 時に代わりに返す値（null 不可）
     * @param <T> パラメータ型
     * @return value が非 null ならそのまま、null なら neutralValue
     * @throws NullPointerException neutralValue が null の場合
     */
    public static <T> T complement(T value, String paramName, T neutralValue) {
        Objects.requireNonNull(neutralValue, "neutralValue");
        return (value != null) ? value : neutralValue;
    }

    /**
     * 固定長 NOT NULL 文字列の WHERE 条件項目を補完する。
     *
     * <p>非 null の場合は {@link FixedLengthFormatter#padSpace(String, int)} と同じ固定長整形を行い、
     * null または空白-only の場合は半角スペース 1 文字を返す。</p>
     *
     * <p>固定長 NOT NULL 文字列の WHERE 補完値は write 用 neutral value とは異なり、
     * 空文字列や列長分スペースではなく半角スペース 1 文字を正とする。</p>
     *
     * @param value パラメータ値（null 許容）
     * @param paramName パラメータ名（デバッグ/ログ用）
     * @param length 固定長カラムの長さ
     * @return 非 null なら固定長整形後の値、null または空白-only なら半角スペース 1 文字
     * @throws IllegalArgumentException length が 1 未満の場合
     */
    public static String fixedLengthNotNull(String value, String paramName, int length) {
        if (length < 1) {
            throw new IllegalArgumentException("length must be greater than 0");
        }
        return complement(FixedLengthFormatter.padSpace(value, length), paramName, " ");
    }

    /**
     * WHERE 句パラメータが null でないことを検証する。
     *
     * <p>null が業務的にありえないパラメータ（{@code String} 型の条件値など）に使用する。
     * {@link Objects#requireNonNull} と同等だが、WHERE パラメータの防御を本クラスに集約する。</p>
     *
     * @param value パラメータ値
     * @param paramName パラメータ名（例外メッセージ用）
     * @param <T> パラメータ型
     * @return value そのまま
     * @throws NullPointerException value が null の場合
     */
    public static <T> T requireNonNull(T value, String paramName) {
        return Objects.requireNonNull(value, paramName);
    }
}
