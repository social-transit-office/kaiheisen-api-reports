package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import org.jboss.logging.Logger;

/**
 * JBoss Logging を使ったデータアクセス共通ログ実装。
 */
public final class JbossCommonDataAccessLogger implements CommonDataAccessLogger {

    private final Logger logger;

    private final JbossStructuredLogger structuredLogger;

    /**
     * 指定クラスを元にロガーを生成する。
     *
     * @param owner ログ出力元のクラス
     */
    public JbossCommonDataAccessLogger(Class<?> owner) {
        this(Logger.getLogger(owner));
    }

    /**
     * 指定した JBoss Logger でインスタンスを生成する。
     *
     * @param logger JBoss Logger
     */
    public JbossCommonDataAccessLogger(Logger logger) {
        this.logger = Objects.requireNonNull(logger, "logger");
        this.structuredLogger = new JbossStructuredLogger(logger);
    }

    @Override
    public void info(String message) {
        logger.info(message);
    }

    @Override
    public void info(LogEvent event) {
        structuredLogger.info(event);
    }

    @Override
    public void error(String message, Throwable cause) {
        logger.error(message, cause);
    }

    @Override
    public void error(LogEvent event, Throwable cause) {
        structuredLogger.error(event, cause);
    }
}
