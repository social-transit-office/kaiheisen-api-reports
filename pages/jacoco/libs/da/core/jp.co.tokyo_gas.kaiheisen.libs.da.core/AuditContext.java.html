package jp.co.tokyo_gas.kaiheisen.libs.da.core;

/**
 * DA write 実行時に呼び出し元から受け取る監査コンテキスト。
 *
 * <p>NCIS/CIS テーブルでは {@code userId} のみ使用する。
 * CIRIUS テーブルでは {@code userId} に加えて {@code kinoId}（最終更新起点機能ID）を使用する。</p>
 *
 * @param userId write 実行者ユーザーID
 * @param kinoId 最終更新起点機能ID（CIRIUS テーブルで必須、NCIS/CIS では null 可）
 */
public record AuditContext(String userId, String kinoId) {

    /**
     * write コンテキストを生成する。
     */
    public AuditContext {
        if (userId == null || userId.isBlank()) {
            throw new IllegalArgumentException("userId must not be blank");
        }
        userId = userId.trim();
        if (kinoId != null) {
            kinoId = kinoId.trim();
        }
    }

    /**
     * 更新ユーザーIDから監査コンテキストを生成する（NCIS/CIS 向け）。
     *
     * @param userId write 実行者ユーザーID
     * @return 監査コンテキスト
     */
    public static AuditContext of(String userId) {
        return new AuditContext(userId, null);
    }

    /**
     * 更新ユーザーIDと最終更新起点機能IDから監査コンテキストを生成する（CIRIUS 向け）。
     *
     * @param userId write 実行者ユーザーID
     * @param kinoId 最終更新起点機能ID
     * @return 監査コンテキスト
     */
    public static AuditContext of(String userId, String kinoId) {
        return new AuditContext(userId, kinoId);
    }
}
