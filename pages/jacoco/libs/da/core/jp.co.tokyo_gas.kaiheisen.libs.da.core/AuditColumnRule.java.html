package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.EnumSet;
import java.util.Objects;
import java.util.Set;

/**
 * 監査列物理名、値スロット、適用操作を表す。
 *
 * @param columnName 列物理名
 * @param valueSlot 値スロット
 * @param operations 適用する操作種別
 */
public record AuditColumnRule(String columnName, AuditColumnValueSlot valueSlot, Set<DataAccessWriteOperation> operations) {

    /**
     * 監査列ルールを生成する。
     */
    public AuditColumnRule {
        if (columnName == null || columnName.isBlank()) {
            throw new IllegalArgumentException("columnName must not be blank");
        }
        columnName = columnName.trim();
        Objects.requireNonNull(valueSlot, "valueSlot");
        Objects.requireNonNull(operations, "operations");
        if (operations.isEmpty()) {
            throw new IllegalArgumentException("operations must not be empty");
        }
        operations = Set.copyOf(EnumSet.copyOf(operations));
    }

    /**
     * insert 時のみ適用するルールを生成する。
     *
     * @param columnName 列物理名
     * @param valueSlot 値スロット
     * @return 監査列ルール
     */
    public static AuditColumnRule insertOnly(String columnName, AuditColumnValueSlot valueSlot) {
        return new AuditColumnRule(columnName, valueSlot, EnumSet.of(DataAccessWriteOperation.INSERT));
    }

    /**
     * insert と update の両方で適用するルールを生成する。
     *
     * @param columnName 列物理名
     * @param valueSlot 値スロット
     * @return 監査列ルール
     */
    public static AuditColumnRule insertAndUpdate(String columnName, AuditColumnValueSlot valueSlot) {
        return new AuditColumnRule(
                columnName,
                valueSlot,
                EnumSet.of(DataAccessWriteOperation.INSERT, DataAccessWriteOperation.UPDATE)
        );
    }

    /**
     * 指定した操作種別に適用するかを返す。
     *
     * @param operation 操作種別
     * @return 適用する場合は true
     */
    public boolean appliesTo(DataAccessWriteOperation operation) {
        Objects.requireNonNull(operation, "operation");
        return operations.contains(operation);
    }
}
