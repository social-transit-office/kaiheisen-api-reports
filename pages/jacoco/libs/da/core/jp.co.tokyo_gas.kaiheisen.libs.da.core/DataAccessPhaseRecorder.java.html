package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.DataAccessPhaseObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;

/** DataAccess 内部の処理フェーズを構造化ログへ記録する。 */
public final class DataAccessPhaseRecorder {

    /** パラメータ変換フェーズ。 */
    public static final String PARAMETER_CONVERSION = "parameter-conversion";

    /** クエリ組み立てフェーズ。 */
    public static final String QUERY_CONSTRUCTION = "query-construction";

    /** SQL 実行・結果取得フェーズ。 */
    public static final String SQL_EXECUTION_RESULT_FETCH = "sql-execution-result-fetch";

    /** Record 変換フェーズ。 */
    public static final String RECORD_CONVERSION = "record-conversion";

    private static final String COMPLETED_EVENT = "dataaccess.phase.completed";

    private static final String COMPLETED_SUMMARY = "DataAccess phase completed";

    private static final String FAILED_EVENT = "dataaccess.phase.failed";

    private static final String FAILED_SUMMARY = "DataAccess phase failed";

    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    private static final String PHASE_ATTRIBUTE = "phase";

    private static final String ELAPSED_SINCE_DATA_ACCESS_START_MS_ATTRIBUTE =
            "elapsedSinceDataAccessStartMs";

    private static final Set<String> TARGET_FUNCTION_IDS =
            Set.of("DA-033-R006", "DA-108-R003");

    private static final ThreadLocal<Deque<State>> CURRENT = new ThreadLocal<>();

    private DataAccessPhaseRecorder() {
    }

    /**
     * 現在の DataAccess 呼び出しに紐付けて処理フェーズを計測する。
     *
     * <p>計測が無効、または計測コンテキストがない場合は処理のみを実行する。ログ生成・出力の
     * 失敗は業務処理へ伝播させない。処理自身が投げた {@link RuntimeException} または
     * {@link Error} は、同じインスタンスを呼び出し元へ返す。</p>
     *
     * @param phase フェーズ名
     * @param action 計測対象処理
     * @param <T> 戻り値型
     * @return 処理結果
     */
    public static <T> T measure(final String phase, final Supplier<T> action) {
        State state = currentState();
        if (state == null) {
            return action.get();
        }

        long phaseStartedNanos = System.nanoTime();
        try {
            T result = action.get();
            logSafely(
                    state,
                    COMPLETED_EVENT,
                    COMPLETED_SUMMARY,
                    phase,
                    phaseStartedNanos,
                    null);
            return result;
        } catch (RuntimeException | Error cause) {
            logSafely(
                    state,
                    FAILED_EVENT,
                    FAILED_SUMMARY,
                    phase,
                    phaseStartedNanos,
                    cause);
            throw cause;
        }
    }

    static boolean isEnabledFor(final String functionId) {
        try {
            return TARGET_FUNCTION_IDS.contains(functionId)
                    && DataAccessPhaseObservabilitySettings.isEnabled();
        } catch (RuntimeException exception) {
            return false;
        }
    }

    static Scope open(
            final boolean enabled,
            final String functionId,
            final String correlationId,
            final Long invocationSequence,
            final boolean measurementUnavailable,
            final long dataAccessStartedNanos,
            final CommonDataAccessLogger logger) {
        if (!enabled) {
            return Scope.NOOP;
        }
        try {
            State state = new State(
                    functionId,
                    correlationId,
                    invocationSequence,
                    measurementUnavailable,
                    dataAccessStartedNanos,
                    logger);
            Deque<State> states = CURRENT.get();
            if (states == null) {
                states = new ArrayDeque<>();
                CURRENT.set(states);
            }
            states.push(state);
            return () -> close(state);
        } catch (RuntimeException exception) {
            return Scope.NOOP;
        }
    }

    private static State currentState() {
        try {
            Deque<State> states = CURRENT.get();
            return states == null ? null : states.peek();
        } catch (RuntimeException exception) {
            return null;
        }
    }

    private static void close(final State state) {
        try {
            Deque<State> states = CURRENT.get();
            if (states == null) {
                return;
            }
            if (states.peek() == state) {
                states.pop();
            } else {
                states.removeFirstOccurrence(state);
            }
            if (states.isEmpty()) {
                CURRENT.remove();
            }
        } catch (RuntimeException exception) {
            CURRENT.remove();
        }
    }

    private static void logSafely(
            final State state,
            final String event,
            final String summary,
            final String phase,
            final long phaseStartedNanos,
            final Throwable cause) {
        try {
            long nowNanos = System.nanoTime();
            Map<String, Object> attributes = new LinkedHashMap<>();
            attributes.put(LogField.CORRELATION_ID.key(), state.correlationId());
            attributes.put(FUNCTION_ID_ATTRIBUTE, state.functionId());
            attributes.put(LogField.DA_INVOCATION_SEQUENCE.key(), state.invocationSequence());
            attributes.put(PHASE_ATTRIBUTE, phase);
            attributes.put(LogField.LATENCY_MS.key(), elapsedMillis(phaseStartedNanos, nowNanos));
            attributes.put(
                    ELAPSED_SINCE_DATA_ACCESS_START_MS_ATTRIBUTE,
                    elapsedMillis(state.dataAccessStartedNanos(), nowNanos));
            attributes.put(
                    LogField.MEASUREMENT_UNAVAILABLE.key(),
                    state.measurementUnavailable());
            LogEvent logEvent = new LogEvent(
                    event,
                    summary,
                    LogLayer.DA,
                    null,
                    null,
                    attributes);
            if (cause == null) {
                state.logger().info(logEvent);
            } else {
                state.logger().error(logEvent, cause);
            }
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない。
        }
    }

    private static long elapsedMillis(final long startedNanos, final long nowNanos) {
        return Math.max(0L, nowNanos - startedNanos) / 1_000_000L;
    }

    interface Scope extends AutoCloseable {

        Scope NOOP = () -> {
        };

        @Override
        void close();
    }

    private record State(
            String functionId,
            String correlationId,
            Long invocationSequence,
            boolean measurementUnavailable,
            long dataAccessStartedNanos,
            CommonDataAccessLogger logger) {
    }
}
