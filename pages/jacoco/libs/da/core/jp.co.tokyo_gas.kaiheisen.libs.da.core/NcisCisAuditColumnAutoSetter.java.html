package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import java.util.function.Function;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/**
 * NCIS/CIS 監査列を DA 管理対象として自動設定する adapter。
 */
public final class NcisCisAuditColumnAutoSetter {

    private static final String DEFAULT_TRANSACTION_NUMBER = "KHS";

    private static final DateTimeFormatter TASK_NO_FORMATTER = DateTimeFormatter.ofPattern("HHmmss");

    private final String transactionNumber;

    private final Function<LocalDateTime, LocalDate> businessDateResolver;

    private final Function<LocalDateTime, Integer> taskNumberResolver;

    /**
     * 標準設定の adapter を生成する。
     */
    public NcisCisAuditColumnAutoSetter() {
        this(
                DEFAULT_TRANSACTION_NUMBER,
                LocalDateTime::toLocalDate,
                NcisCisAuditColumnAutoSetter::defaultTaskNumber
        );
    }

    /**
     * トランザクション番号のみ差し替えた adapter を生成する。
     *
     * @param transactionNumber 更新トランザクション番号
     */
    public NcisCisAuditColumnAutoSetter(String transactionNumber) {
        this(
                transactionNumber,
                LocalDateTime::toLocalDate,
                NcisCisAuditColumnAutoSetter::defaultTaskNumber
        );
    }

    /**
     * 監査列設定ルールを指定して adapter を生成する。
     *
     * @param transactionNumber 更新トランザクション番号
     * @param businessDateResolver 業務日付解決
     * @param taskNumberResolver 更新タスク番号解決
     */
    public NcisCisAuditColumnAutoSetter(
            String transactionNumber,
            Function<LocalDateTime, LocalDate> businessDateResolver,
            Function<LocalDateTime, Integer> taskNumberResolver
    ) {
        if (transactionNumber == null || transactionNumber.isBlank()) {
            throw new IllegalArgumentException("transactionNumber must not be blank");
        }
        this.transactionNumber = transactionNumber.trim();
        this.businessDateResolver = Objects.requireNonNull(businessDateResolver, "businessDateResolver");
        this.taskNumberResolver = Objects.requireNonNull(taskNumberResolver, "taskNumberResolver");
    }

    /**
     * insert 時の監査列を設定する。
     *
     * @param access 列アクセス
     * @param context 呼び出し元コンテキスト
     * @param clock システム日時プロバイダ
     */
    public void applyForInsert(
            AuditColumnAccess access,
            AuditContext context,
            SystemDateTimeProvider clock
    ) {
        apply(access, DataAccessWriteOperation.INSERT, context, clock, null);
    }

    /**
     * update 時の監査列を設定する。
     *
     * @param access 列アクセス
     * @param context 呼び出し元コンテキスト
     * @param clock システム日時プロバイダ
     * @param currentRecUpdJnjNo 現在の排他順序番号
     */
    public void applyForUpdate(
            AuditColumnAccess access,
            AuditContext context,
            SystemDateTimeProvider clock,
            Integer currentRecUpdJnjNo
    ) {
        apply(access, DataAccessWriteOperation.UPDATE, context, clock, currentRecUpdJnjNo);
    }

    /**
     * 操作種別に応じて監査列を設定する。
     *
     * @param access 列アクセス
     * @param operation 操作種別
     * @param context 呼び出し元コンテキスト
     * @param clock システム日時プロバイダ
     * @param currentRecUpdJnjNo 現在の排他順序番号。insert 時は不要
     */
    public void apply(
            AuditColumnAccess access,
            DataAccessWriteOperation operation,
            AuditContext context,
            SystemDateTimeProvider clock,
            Integer currentRecUpdJnjNo
    ) {
        Objects.requireNonNull(access, "access");
        Objects.requireNonNull(operation, "operation");
        Objects.requireNonNull(context, "context");
        Objects.requireNonNull(clock, "clock");

        LocalDateTime currentDateTime = Objects.requireNonNull(clock.currentDateTime(), "clock.currentDateTime()");
        Integer nextRecUpdJnjNo = switch (operation) {
            case INSERT -> 1;
            case UPDATE -> Math.addExact(Objects.requireNonNull(currentRecUpdJnjNo, "currentRecUpdJnjNo"), 1);
        };
        AuditColumnPolicyApplier.apply(
                createPolicy(context, currentDateTime, nextRecUpdJnjNo),
                operation,
                access
        );
    }

    private AuditColumnPolicy createPolicy(
            AuditContext context,
            LocalDateTime currentDateTime,
            Integer nextRecUpdJnjNo
    ) {
        AuditColumnValueContext valueContext = new AuditColumnValueContext(
                Objects.requireNonNull(
                        businessDateResolver.apply(currentDateTime),
                        "businessDateResolver.apply(currentDateTime)"
                ),
                currentDateTime,
                context.userId(),
                transactionNumber,
                Objects.requireNonNull(
                        taskNumberResolver.apply(currentDateTime),
                        "taskNumberResolver.apply(currentDateTime)"
                ),
                nextRecUpdJnjNo,
                null
        );
        return AuditColumnPolicy.of(
                AuditColumnPattern.PATTERN2,
                AuditColumnValueResolvers.from(valueContext)
        );
    }

    private static Integer defaultTaskNumber(LocalDateTime currentDateTime) {
        return Integer.parseInt(TASK_NO_FORMATTER.format(currentDateTime));
    }
}
