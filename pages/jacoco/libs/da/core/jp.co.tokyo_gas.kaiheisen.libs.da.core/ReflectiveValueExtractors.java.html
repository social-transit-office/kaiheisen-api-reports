package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.lang.reflect.Method;
import java.lang.reflect.RecordComponent;
import java.util.EnumMap;
import java.util.HashMap;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;

/**
 * Record コンポーネントと Column enum の名前マッチングにより
 * 値抽出関数マップを自動生成するユーティリティ。
 *
 * <p>Column 名（UPPER_SNAKE_CASE）を lowerCamelCase へ変換し、
 * Record コンポーネント名と照合する。マッチしない Column（監査列等）は
 * マップに含まれない。</p>
 */
public final class ReflectiveValueExtractors {

    private ReflectiveValueExtractors() {
    }

    /**
     * Record クラスのコンポーネントと Column enum を名前照合し、
     * 列→値抽出関数の {@link EnumMap} を構築する。
     *
     * @param <RecordT> Record 型
     * @param <ColumnT> Column enum 型
     * @param columnClass Column enum クラス
     * @param recordClass Record クラス
     * @return 列→値抽出関数マップ（マッチしない列は含まない）
     */
    public static <RecordT extends Record, ColumnT extends Enum<ColumnT> & DataAccessColumn>
        EnumMap<ColumnT, Function<RecordT, Object>> build(
            Class<ColumnT> columnClass,
            Class<RecordT> recordClass
    ) {
        Objects.requireNonNull(columnClass, "columnClass");
        Objects.requireNonNull(recordClass, "recordClass");

        Map<String, Method> accessors = new HashMap<>();
        for (RecordComponent component : recordClass.getRecordComponents()) {
            accessors.put(component.getName(), component.getAccessor());
        }

        EnumMap<ColumnT, Function<RecordT, Object>> extractors = new EnumMap<>(columnClass);
        for (ColumnT column : columnClass.getEnumConstants()) {
            String camelName = snakeToCamel(column.name());
            Method accessor = accessors.get(camelName);
            if (accessor != null) {
                extractors.put(column, createExtractor(accessor));
            }
        }
        return extractors;
    }

    /**
     * UPPER_SNAKE_CASE を lowerCamelCase に変換する。
     *
     * @param snakeCase 変換元文字列（例: {@code GMT_SET_NO}）
     * @return lowerCamelCase 文字列（例: {@code gmtSetNo}）
     */
    public static String snakeToCamel(String snakeCase) {
        Objects.requireNonNull(snakeCase, "snakeCase");
        String[] parts = snakeCase.split("_", -1);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < parts.length; i++) {
            String part = parts[i];
            if (part.isEmpty()) {
                continue;
            }
            if (i == 0) {
                sb.append(part.toLowerCase());
            } else {
                sb.append(Character.toUpperCase(part.charAt(0)));
                if (part.length() > 1) {
                    sb.append(part.substring(1).toLowerCase());
                }
            }
        }
        return sb.toString();
    }

    private static <RecordT> Function<RecordT, Object> createExtractor(Method accessor) {
        accessor.setAccessible(true);
        return record -> {
            try {
                return accessor.invoke(record);
            } catch (ReflectiveOperationException e) {
                throw new IllegalStateException(
                        "Failed to invoke accessor: " + accessor.getName(), e);
            }
        };
    }
}
