package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.time.LocalDate;
import java.util.Objects;
import java.util.Optional;

/**
 * 監査列パターン向けの標準的な値解決を生成するファクトリ。
 */
public final class AuditColumnValueResolvers {

    private AuditColumnValueResolvers() {
    }

    /**
     * 内部監査値コンテキストから値解決を生成する。
     *
     * @param valueContext 内部監査値コンテキスト
     * @return 値解決
     */
    public static AuditColumnValueResolver from(AuditColumnValueContext valueContext) {
        Objects.requireNonNull(valueContext, "valueContext");
        return valueSlot -> switch (valueSlot) {
            case BUSINESS_DATE -> Optional.ofNullable(valueContext.businessDate());
            case CURRENT_DATETIME -> Optional.ofNullable(valueContext.currentDateTime());
            case REGISTER_USER_ID, UPDATE_USER_ID -> Optional.ofNullable(valueContext.userId());
            case TRANSACTION_NUMBER -> Optional.ofNullable(valueContext.transactionNumber());
            case TASK_NUMBER -> Optional.ofNullable(valueContext.taskNumber());
            case UPDATE_SEQUENCE_NUMBER -> Optional.ofNullable(valueContext.updateSequenceNumber());
            case UPDATE_FUNCTION_ID -> Optional.ofNullable(valueContext.updateFunctionId());
        };
    }

    /**
     * Pattern2 向けの値解決を生成する。
     *
     * @param businessDate 業務日付
     * @param updateUserId 更新ユーザーID
     * @param transactionNumber 更新トランザクション番号
     * @param taskNumber 更新タスク番号
     * @param updateSequenceNumber 更新順序番号
     * @return 値解決
     */
    public static AuditColumnValueResolver pattern2(
            LocalDate businessDate,
            String updateUserId,
            String transactionNumber,
            Integer taskNumber,
            Integer updateSequenceNumber
    ) {
        Objects.requireNonNull(businessDate, "businessDate");
        Objects.requireNonNull(updateUserId, "updateUserId");
        Objects.requireNonNull(transactionNumber, "transactionNumber");
        Objects.requireNonNull(taskNumber, "taskNumber");
        Objects.requireNonNull(updateSequenceNumber, "updateSequenceNumber");
        return from(new AuditColumnValueContext(
                businessDate,
                null,
                updateUserId,
                transactionNumber,
                taskNumber,
                updateSequenceNumber,
                null
        ));
    }

    /**
     * Pattern3 向けの値解決を生成する。
     *
     * @param businessDate 業務日付
     * @param updateSequenceNumber 更新順序番号
     * @return 値解決
     */
    public static AuditColumnValueResolver pattern3(LocalDate businessDate, Integer updateSequenceNumber) {
        Objects.requireNonNull(businessDate, "businessDate");
        Objects.requireNonNull(updateSequenceNumber, "updateSequenceNumber");
        return from(new AuditColumnValueContext(
                businessDate,
                null,
                null,
                null,
                null,
                updateSequenceNumber,
                null
        ));
    }
}
