package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Arrays;
import java.util.Map;
import java.util.Objects;
import java.util.function.Function;
import java.util.stream.Collectors;

/**
 * {@link DataAccessColumn} をキーに持つ Map を監査列アクセスへ変換する adapter。
 *
 * @param <ColumnT> 列挙型
 */
public final class MapBackedAuditColumnAccess<ColumnT extends Enum<ColumnT> & DataAccessColumn>
        implements AuditColumnAccess {

    private final Map<String, ColumnT> columnsByName;

    private final Map<ColumnT, Object> values;

    /**
     * adapter を生成する。
     *
     * @param columnType 列挙型クラス
     * @param values 列値マップ
     */
    public MapBackedAuditColumnAccess(Class<ColumnT> columnType, Map<ColumnT, Object> values) {
        Objects.requireNonNull(columnType, "columnType");
        this.values = Objects.requireNonNull(values, "values");
        this.columnsByName = Arrays.stream(columnType.getEnumConstants())
                .collect(Collectors.toUnmodifiableMap(DataAccessColumn::columnName, Function.identity()));
    }

    @Override
    public boolean hasColumn(String columnName) {
        return columnsByName.containsKey(columnName);
    }

    @Override
    public void setValue(String columnName, Object value) {
        ColumnT column = columnsByName.get(columnName);
        if (column == null) {
            throw new IllegalArgumentException("Unknown column: " + columnName);
        }
        values.put(column, value);
    }
}
