package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.List;

/**
 * NCIS テーブルのうち KOSIN 列を持たない 3 列監査パターン。
 *
 * <p>TATEMON 等、REC_REG_YMD / REC_UPD_YMD / REC_UPD_JNJ_NO のみを
 * 監査列として持つテーブルに使用する。</p>
 */
public enum NcisSubsetAuditColumn implements DataAccessColumn {

    REC_REG_YMD("REC_REG_YMD"),
    REC_UPD_YMD("REC_UPD_YMD"),
    REC_UPD_JNJ_NO("REC_UPD_JNJ_NO");

    private static final List<NcisSubsetAuditColumn> INSERT_COLUMNS = List.of(
            REC_REG_YMD,
            REC_UPD_YMD,
            REC_UPD_JNJ_NO
    );

    private static final List<NcisSubsetAuditColumn> UPDATE_COLUMNS = List.of(
            REC_UPD_YMD,
            REC_UPD_JNJ_NO
    );

    private final String columnName;

    NcisSubsetAuditColumn(String columnName) {
        this.columnName = columnName;
    }

    @Override
    public String columnName() {
        return columnName;
    }

    /**
     * 操作種別ごとの監査列一覧を返す。
     *
     * @param operation 操作種別
     * @return 監査列一覧
     */
    public static List<NcisSubsetAuditColumn> columnsFor(DataAccessWriteOperation operation) {
        return switch (operation) {
            case INSERT -> INSERT_COLUMNS;
            case UPDATE -> UPDATE_COLUMNS;
        };
    }
}
