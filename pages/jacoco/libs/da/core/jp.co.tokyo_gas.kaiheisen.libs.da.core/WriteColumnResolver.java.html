package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Comparator;
import java.util.EnumMap;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Objects;
import java.util.Set;

/**
 * write 列メタ情報を扱う共通ユーティリティ。
 *
 * <p>主な責務:</p>
 * <ul>
 *   <li>write record から列値マップの抽出</li>
 *   <li>対象列集合の更新可能列照合と列挙順解決</li>
 *   <li>公開列と managed 列の結合</li>
 * </ul>
 */
public final class WriteColumnResolver {

    private WriteColumnResolver() {
    }

    /**
     * write record から列値マップを抽出する。
     *
     * @param definition テーブル定義
     * @param record write record
     * @param <TRecord> write record 型
     * @param <TColumn> 列型
     * @return 列値マップ
     */
    public static <
            TRecord,
            TColumn extends Enum<TColumn> & DataAccessColumn
            > EnumMap<TColumn, Object> extractValues(
            TableWriteDefinition<TRecord, TColumn> definition,
            TRecord record
    ) {
        Objects.requireNonNull(definition, "definition");
        Objects.requireNonNull(record, "record");
        EnumMap<TColumn, Object> values = Objects.requireNonNull(
                definition.extractValues(record),
                "definition.extractValues(record)"
        );
        return new EnumMap<>(values);
    }

    /**
     * 対象列集合を更新可能列と照合し、列挙順で解決する。
     *
     * @param definition テーブル定義
     * @param targetColumns 対象列集合
     * @param requireNonEmpty 空集合を禁止する場合は true
     * @param <TRecord> write record 型
     * @param <TColumn> 列型
     * @return 列挙順の対象列リスト
     */
    public static <
            TRecord,
            TColumn extends Enum<TColumn> & DataAccessColumn
            > List<TColumn> resolveUpdatableColumns(
            TableWriteDefinition<TRecord, TColumn> definition,
            Set<TColumn> targetColumns,
            boolean requireNonEmpty
    ) {
        Objects.requireNonNull(definition, "definition");

        Set<TColumn> updatable = definition.updatableColumns();
        if (targetColumns != null) {
            for (TColumn col : targetColumns) {
                if (!updatable.contains(col)) {
                    throw new IllegalArgumentException("Column is not updatable: " + col);
                }
            }
        }
        return normalizeColumns(targetColumns, requireNonEmpty);
    }

    /**
     * 公開列と managed 列を結合して重複を除去した順序付きリストを返す。
     *
     * <p>公開列を先頭に配置し、managed 列のうち公開列に含まれないものを末尾に追加する。
     * 同一列名が既に存在する場合は先着を優先する。</p>
     *
     * @param orderedPublicColumns 公開列（順序保持）
     * @param managedColumns managed 列
     * @param <T> 列型
     * @return 結合後の列リスト
     */
    public static <T extends DataAccessColumn> List<T> effectiveColumns(
            List<? extends T> orderedPublicColumns,
            List<? extends T> managedColumns
    ) {
        Objects.requireNonNull(orderedPublicColumns, "orderedPublicColumns");
        Objects.requireNonNull(managedColumns, "managedColumns");
        LinkedHashMap<String, T> columnsByName = new LinkedHashMap<>();
        for (T publicColumn : orderedPublicColumns) {
            T nonNull = Objects.requireNonNull(publicColumn, "orderedPublicColumns contains null");
            columnsByName.put(nonNull.columnName(), nonNull);
        }
        for (T managedColumn : managedColumns) {
            T nonNull = Objects.requireNonNull(managedColumn, "managedColumns contains null");
            columnsByName.putIfAbsent(nonNull.columnName(), nonNull);
        }
        return List.copyOf(columnsByName.values());
    }

    /**
     * 対象列集合を列挙順のリストへ正規化する。
     *
     * @param targetColumns 対象列集合
     * @param requireNonEmpty 空集合を禁止する場合は true
     * @param <TColumn> 列挙型
     * @return 列挙順の対象列リスト
     */
    private static <TColumn extends Enum<TColumn>> List<TColumn> normalizeColumns(
            Set<TColumn> targetColumns,
            boolean requireNonEmpty
    ) {
        Objects.requireNonNull(targetColumns, "targetColumns");
        if (targetColumns.stream().anyMatch(Objects::isNull)) {
            throw new NullPointerException("targetColumns contains null");
        }
        if (requireNonEmpty && targetColumns.isEmpty()) {
            throw new IllegalArgumentException("targetColumns must not be empty");
        }
        return targetColumns.stream()
                .sorted(Comparator.comparingInt(Enum::ordinal))
                .toList();
    }
}
