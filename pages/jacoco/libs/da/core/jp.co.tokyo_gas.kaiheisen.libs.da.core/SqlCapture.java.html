package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentMap;

/**
 * 直近の SQL をスレッド単位で保持する。
 */
public final class SqlCapture {

    private static final ConcurrentMap<Long, String> LAST_SQL_BY_THREAD = new ConcurrentHashMap<>();

    private SqlCapture() {
    }

    /**
     * 直近の SQL を現在のスレッドに保持する。
     *
     * @param sql 保持する SQL 文
     */
    public static void set(String sql) {
        if (sql == null || sql.isBlank()) {
            return;
        }
        LAST_SQL_BY_THREAD.put(Thread.currentThread().threadId(), sql);
    }

    /**
     * 現在のスレッドに保持した SQL を取得し、保持を解除する。
     *
     * @return 保持されていた SQL 文。保持がない場合は {@code null}
     */
    public static String getAndClear() {
        return LAST_SQL_BY_THREAD.remove(Thread.currentThread().threadId());
    }
}
