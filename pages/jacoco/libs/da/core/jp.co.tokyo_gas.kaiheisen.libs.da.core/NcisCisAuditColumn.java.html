package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.List;

/**
 * NCIS/CIS 系テーブルで DA が管理する監査列。
 */
public enum NcisCisAuditColumn implements DataAccessColumn {

    REC_REG_YMD("REC_REG_YMD"),
    REC_UPD_YMD("REC_UPD_YMD"),
    REC_UPD_JNJ_NO("REC_UPD_JNJ_NO"),
    KOSIN_USER_ID("KOSIN_USER_ID"),
    KOSIN_TRAN_NO("KOSIN_TRAN_NO"),
    KOSIN_TASK_NO("KOSIN_TASK_NO");

    private static final List<NcisCisAuditColumn> INSERT_COLUMNS = List.of(
            REC_REG_YMD,
            REC_UPD_YMD,
            REC_UPD_JNJ_NO,
            KOSIN_USER_ID,
            KOSIN_TRAN_NO,
            KOSIN_TASK_NO
    );

    private static final List<NcisCisAuditColumn> UPDATE_COLUMNS = List.of(
            REC_UPD_YMD,
            REC_UPD_JNJ_NO,
            KOSIN_USER_ID,
            KOSIN_TRAN_NO,
            KOSIN_TASK_NO
    );

    private final String columnName;

    NcisCisAuditColumn(String columnName) {
        this.columnName = columnName;
    }

    @Override
    public String columnName() {
        return columnName;
    }

    /**
     * 操作種別ごとの監査列一覧を返す。
     *
     * @param operation 操作種別
     * @return 監査列一覧
     */
    public static List<NcisCisAuditColumn> columnsFor(DataAccessWriteOperation operation) {
        return switch (operation) {
            case INSERT -> INSERT_COLUMNS;
            case UPDATE -> UPDATE_COLUMNS;
        };
    }
}
