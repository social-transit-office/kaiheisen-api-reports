package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogPayloadFormatter;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogPayloadMapper;

/**
 * データアクセス共通ログ出力用の抽象化。
 */
public interface CommonDataAccessLogger {

    /**
     * 情報メッセージをログ出力する。
     *
     * @param message 情報メッセージ
     */
    default void info(String message) {
        // no-op
    }

    /**
     * 構造化された情報ログを出力する。
     *
     * @param event ログイベント
     */
    default void info(LogEvent event) {
        info(LogPayloadFormatter.toJson(LogPayloadMapper.toMap(event)));
    }

    /**
     * エラーメッセージと原因例外をログ出力する。
     *
     * @param message エラーメッセージ
     * @param cause   原因例外
     */
    void error(String message, Throwable cause);

    /**
     * 構造化されたエラーログを出力する。
     *
     * @param event ログイベント
     * @param cause 原因例外
     */
    default void error(LogEvent event, Throwable cause) {
        error(LogPayloadFormatter.toJson(LogPayloadMapper.toMap(event)), cause);
    }

    /**
     * 何も出力しないロガーを返す。
     *
     * @return No-op ロガー
     */
    static CommonDataAccessLogger noop() {
        return new CommonDataAccessLogger() {
            @Override
            public void error(String message, Throwable cause) {
                // no-op
            }
        };
    }

    /**
     * JBoss Logging を使用するロガーを返す。
     *
     * @param owner ログ出力元のクラス
     * @return JBoss Logging ロガー
     */
    static CommonDataAccessLogger jboss(Class<?> owner) {
        return new JbossCommonDataAccessLogger(owner);
    }
}
