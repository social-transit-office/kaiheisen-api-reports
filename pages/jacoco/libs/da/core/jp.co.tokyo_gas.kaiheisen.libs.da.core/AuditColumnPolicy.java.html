package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.Objects;
import java.util.function.Supplier;

/**
 * 監査列パターンと値解決規則を束ねた policy。
 */
public final class AuditColumnPolicy {

    private final AuditColumnPattern pattern;

    private final Supplier<AuditColumnValueResolver> valueResolverSupplier;

    private AuditColumnPolicy(
            AuditColumnPattern pattern,
            Supplier<AuditColumnValueResolver> valueResolverSupplier
    ) {
        this.pattern = Objects.requireNonNull(pattern, "pattern");
        this.valueResolverSupplier = Objects.requireNonNull(valueResolverSupplier, "valueResolverSupplier");
    }

    /**
     * 固定の値解決を使う policy を生成する。
     *
     * @param pattern 監査列パターン
     * @param valueResolver 値解決
     * @return policy
     */
    public static AuditColumnPolicy of(
            AuditColumnPattern pattern,
            AuditColumnValueResolver valueResolver
    ) {
        Objects.requireNonNull(valueResolver, "valueResolver");
        return of(pattern, () -> valueResolver);
    }

    /**
     * 呼び出しごとに値解決を生成する policy を生成する。
     *
     * @param pattern 監査列パターン
     * @param valueResolverSupplier 値解決 supplier
     * @return policy
     */
    public static AuditColumnPolicy of(
            AuditColumnPattern pattern,
            Supplier<AuditColumnValueResolver> valueResolverSupplier
    ) {
        return new AuditColumnPolicy(pattern, valueResolverSupplier);
    }

    /**
     * 監査列パターンを返す。
     *
     * @return 監査列パターン
     */
    public AuditColumnPattern pattern() {
        return pattern;
    }

    /**
     * この policy から値解決を生成する。
     *
     * @return 値解決
     */
    public AuditColumnValueResolver valueResolver() {
        return Objects.requireNonNull(valueResolverSupplier.get(), "valueResolverSupplier.get()");
    }
}
