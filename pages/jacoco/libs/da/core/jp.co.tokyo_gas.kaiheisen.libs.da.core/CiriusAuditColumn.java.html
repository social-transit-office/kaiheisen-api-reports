package jp.co.tokyo_gas.kaiheisen.libs.da.core;

import java.util.List;

/**
 * CIRIUS 系テーブルで DA が管理する監査列。
 */
public enum CiriusAuditColumn implements DataAccessColumn {

    REG_TSMP("REG_TSMP"),
    REG_USR_ID("REG_USR_ID"),
    UPD_TSMP("UPD_TSMP"),
    UPD_USR_ID("UPD_USR_ID"),
    UPD_KTN_KINO_ID("UPD_KTN_KINO_ID"),
    REC_UPD_JNJ_NO("REC_UPD_JNJ_NO");

    private static final List<CiriusAuditColumn> INSERT_COLUMNS = List.of(
            REG_TSMP,
            REG_USR_ID,
            UPD_TSMP,
            UPD_USR_ID,
            UPD_KTN_KINO_ID,
            REC_UPD_JNJ_NO
    );

    private static final List<CiriusAuditColumn> UPDATE_COLUMNS = List.of(
            UPD_TSMP,
            UPD_USR_ID,
            UPD_KTN_KINO_ID,
            REC_UPD_JNJ_NO
    );

    private final String columnName;

    CiriusAuditColumn(String columnName) {
        this.columnName = columnName;
    }

    @Override
    public String columnName() {
        return columnName;
    }

    /**
     * 操作種別ごとの監査列一覧を返す。
     *
     * @param operation 操作種別
     * @return 監査列一覧
     */
    public static List<CiriusAuditColumn> columnsFor(DataAccessWriteOperation operation) {
        return switch (operation) {
            case INSERT -> INSERT_COLUMNS;
            case UPDATE -> UPDATE_COLUMNS;
        };
    }
}
