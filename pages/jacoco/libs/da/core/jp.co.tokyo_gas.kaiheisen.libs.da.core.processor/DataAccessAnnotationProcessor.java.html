package jp.co.tokyo_gas.kaiheisen.libs.da.core.processor;

import java.util.HashMap;
import java.util.Locale;
import java.util.Map;
import java.util.Set;
import javax.annotation.processing.AbstractProcessor;
import javax.annotation.processing.RoundEnvironment;
import javax.annotation.processing.SupportedAnnotationTypes;
import javax.annotation.processing.SupportedSourceVersion;
import javax.lang.model.SourceVersion;
import javax.lang.model.element.Element;
import javax.lang.model.element.ElementKind;
import javax.lang.model.element.ExecutableElement;
import javax.lang.model.element.TypeElement;
import javax.lang.model.util.Elements;
import javax.tools.Diagnostic;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccess;

/**
 * DataAccess アノテーションに対する静的検査を行うアノテーションプロセッサ。
 *
 * <p>目的は以下の 3 点:
 * <ul>
 *   <li>@DataAccess の identifier 未設定/空白をビルド時に検出</li>
 *   <li>@DataAccess の name 未設定/空白をビルド時に検出</li>
 *   <li>@DataAccess の identifier 重複をビルド時に検出</li>
 * </ul>
 *
 * <p>対象は DataAccess API 層のみに限定する。
 * {@code jp.co.tokyo_gas.kaiheisen.libs.da.api.} パッケージ配下の
 * インターフェースメソッドのみを走査することで、実装層やテストの誤検知を避ける。
 *
 * <p>重複チェックはモジュール（コンパイル単位）内に限定される点に注意。
 * 複数モジュール横断の重複検出は別途仕組みが必要。
 */
@SupportedAnnotationTypes("jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccess")
@SupportedSourceVersion(SourceVersion.RELEASE_21)
public class DataAccessAnnotationProcessor extends AbstractProcessor {

    /** DataAccess API 層のクラス判定に使うパッケージ接頭辞。 */
    private static final String API_PACKAGE_PREFIX = "jp.co.tokyo_gas.kaiheisen.libs.da.api.";

    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        // 対象アノテーションが無いラウンドは処理しない。
        if (annotations.isEmpty()) {
            return false;
        }
        Elements elementUtils = processingEnv.getElementUtils();
        // identifier の重複検知用に初出の要素を保持する。
        Map<String, Element> identifiers = new HashMap<>();

        // @DataAccess が付いた要素を走査する。
        for (Element element : roundEnv.getElementsAnnotatedWith(DataAccess.class)) {
            // メソッド以外に付いている場合は無視（想定外の付与を許容しない）。
            if (element.getKind() != ElementKind.METHOD) {
                continue;
            }
            ExecutableElement method = (ExecutableElement) element;
            TypeElement owner = (TypeElement) method.getEnclosingElement();
            String ownerName = elementUtils.getBinaryName(owner).toString();
            // API 層以外は検査対象外（実装層・テスト層の誤検知を避ける）。
            if (!ownerName.startsWith(API_PACKAGE_PREFIX)) {
                continue;
            }

            DataAccess dataAccess = method.getAnnotation(DataAccess.class);
            String identifier = normalize(dataAccess.identifier());
            String name = normalize(dataAccess.name());

            // identifier 未設定（空/空白）をエラー化。
            if (identifier.isEmpty()) {
                error(method, "@DataAccess identifier must not be blank: %s", signature(ownerName, method));
            }
            // name 未設定（空/空白）をエラー化。
            if (name.isEmpty()) {
                error(method, "@DataAccess name must not be blank: %s", signature(ownerName, method));
            }
            // identifier の重複をエラー化（空は対象外）。
            if (!identifier.isEmpty()) {
                Element existing = identifiers.putIfAbsent(identifier, method);
                if (existing != null) {
                    error(method, "@DataAccess identifier must be unique: %s", identifier);
                }
            }
        }
        // 他のプロセッサにも処理を委譲できるよう false を返す。
        return false;
    }

    /**
     * null/空白を空文字に正規化する。
     */
    private static String normalize(String value) {
        if (value == null) {
            return "";
        }
        return value.trim();
    }

    /**
     * コンパイルエラーとして報告する。
     */
    private void error(Element element, String message, Object... args) {
        processingEnv.getMessager().printMessage(
                Diagnostic.Kind.ERROR,
                String.format(Locale.ROOT, message, args),
                element
        );
    }

    /**
     * エラーメッセージ用に "完全修飾名#メソッド名" の簡易シグネチャを返す。
     */
    private static String signature(String owner, ExecutableElement method) {
        return owner + "#" + method.getSimpleName();
    }
}
