package jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1;

import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessCallSpec;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessPort;

/** DB 接続の起動時ウォームアップに使用するデータアクセス。 */
public interface Sysdummy1DataAccess extends DataAccessPort<Sysdummy1Record> {

    /** DA 識別子定義。 */
    final class Identifiers {

        /** DBウォームアップ定数取得。 */
        public static final String FETCH_WARMUP_VALUE = "DA-900-R001";

        private Identifiers() { }
    }

    /** DA 呼出し仕様定義。 */
    final class CallSpecs {

        /** DBウォームアップ定数取得。 */
        public static final DataAccessCallSpec FETCH_WARMUP_VALUE =
                new DataAccessCallSpec(Identifiers.FETCH_WARMUP_VALUE);

        private CallSpecs() { }
    }

    /**
     * DB接続を確立し、定数1を取得する。
     *
     * @return 値1を保持する単一レコード
     */
    @DataAccess(identifier = Identifiers.FETCH_WARMUP_VALUE, name = "DBウォームアップ定数取得")
    Sysdummy1Record fetchWarmupValue();

    /**
     * スキーマ名を返す。
     *
     * @return スキーマ名
     */
    default String schemaName() {
        return "SYSIBM";
    }

    /**
     * テーブル名を返す。
     *
     * @return テーブル名
     */
    @Override
    default String tableName() {
        return "SYSDUMMY1";
    }
}
