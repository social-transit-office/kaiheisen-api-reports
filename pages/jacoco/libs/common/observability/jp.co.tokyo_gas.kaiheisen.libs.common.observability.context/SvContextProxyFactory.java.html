package jp.co.tokyo_gas.kaiheisen.libs.common.observability.context;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.lang.reflect.InvocationHandler;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;
import java.lang.reflect.Parameter;
import java.lang.reflect.Proxy;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogRequestBodyMapper;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.PerformanceObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.sv.core.Service;

/**
 * SV 実行中の識別子を RequestContext へ関連付ける共通プロキシファクトリです.
 */
@ApplicationScoped
public final class SvContextProxyFactory {

    /** 開始時のイベント名. */
    private static final String SERVICE_STARTED_EVENT = "service.started";

    /** 開始時の要約. */
    private static final String SERVICE_STARTED_SUMMARY =
            "Service execution started.";

    /** Successful terminal event name. */
    private static final String SERVICE_COMPLETED_EVENT = "service.completed";

    /** Successful terminal event summary. */
    private static final String SERVICE_COMPLETED_SUMMARY =
            "Service execution completed.";

    /** Failed terminal event name. */
    private static final String SERVICE_FAILED_EVENT = "service.failed";

    /** Failed terminal event summary. */
    private static final String SERVICE_FAILED_SUMMARY =
            "Service execution failed.";

    /** 機能ID属性名. */
    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    /** リクエストコンテキスト. */
    private final RequestContext requestContext;

    /** 構造化ロガー. */
    private final StructuredLogger logger;

    /**
     * コンストラクタ.
     *
     * @param currentRequestContext リクエストコンテキスト
     */
    @Inject
    public SvContextProxyFactory(final RequestContext currentRequestContext) {
        this(
                currentRequestContext,
                new JbossStructuredLogger(SvContextProxyFactory.class));
    }

    /**
     * テスト用にロガーを差し替えて生成します.
     *
     * @param currentRequestContext リクエストコンテキスト
     * @param structuredLogger 構造化ロガー
     */
    SvContextProxyFactory(
            final RequestContext currentRequestContext,
            final StructuredLogger structuredLogger) {
        this.requestContext = Objects.requireNonNull(
                currentRequestContext,
                "requestContext");
        this.logger = Objects.requireNonNull(structuredLogger, "logger");
    }

    /**
     * SV インターフェースに RequestContext 連携用のプロキシを付与します.
     *
     * @param <T> SV 型
     * @param serviceType SV インターフェース型
     * @param delegate 実体
     * @return 必要に応じてラップされた SV
     */
    public <T> T wrap(final Class<T> serviceType, final T delegate) {
        Objects.requireNonNull(serviceType, "serviceType");
        Objects.requireNonNull(delegate, "delegate");

        Service service = serviceType.getAnnotation(Service.class);
        if (service == null || !serviceType.isInterface()) {
            return delegate;
        }

        InvocationHandler handler = new ServiceInvocationHandler<>(
                delegate,
                requestContext,
                logger,
                service);
        return serviceType.cast(Proxy.newProxyInstance(
                serviceType.getClassLoader(),
                new Class<?>[] {serviceType},
                handler));
    }

    private static final class ServiceInvocationHandler<T>
            implements InvocationHandler {

        /** 実処理委譲先. */
        private final T delegate;

        /** リクエストコンテキスト. */
        private final RequestContext requestContext;

        /** 構造化ロガー. */
        private final StructuredLogger logger;

        /** サービス定義メタ情報. */
        private final Service service;

        private ServiceInvocationHandler(
                final T currentDelegate,
                final RequestContext currentRequestContext,
                final StructuredLogger structuredLogger,
                final Service currentService) {
            this.delegate = currentDelegate;
            this.requestContext = currentRequestContext;
            this.logger = structuredLogger;
            this.service = currentService;
        }

        @Override
        public Object invoke(
                final Object proxy,
                final Method method,
                final Object[] args) throws Throwable {
            if (method.getDeclaringClass() == Object.class) {
                if (!method.canAccess(delegate)) {
                    method.setAccessible(true);
                }
                return method.invoke(delegate, args);
            }
            Objects.requireNonNull(proxy, "proxy");
            if (!method.canAccess(delegate)) {
                method.setAccessible(true);
            }
            boolean terminalObservabilityEnabled =
                    PerformanceObservabilitySettings.isTerminalEnabled();
            Long invocationSequence = terminalObservabilityEnabled
                    ? nextServiceInvocationSequence()
                    : null;
            boolean terminalMeasurementEnabled =
                    terminalObservabilityEnabled && invocationSequence != null;
            long startedAtNanos = System.nanoTime();
            requestContext.pushService(service.identifier(), service.name());
            try {
                if (terminalObservabilityEnabled) {
                    safeLogServiceStarted(method, args, invocationSequence);
                } else {
                    logServiceStarted(method, args, invocationSequence);
                }
                Object result = method.invoke(delegate, args);
                if (terminalMeasurementEnabled) {
                    logServiceTerminal(
                            SERVICE_COMPLETED_EVENT,
                            SERVICE_COMPLETED_SUMMARY,
                            invocationSequence,
                            startedAtNanos);
                }
                return result;
            } catch (InvocationTargetException ex) {
                Throwable cause = ex.getCause();
                if (terminalMeasurementEnabled) {
                    logServiceTerminal(
                            SERVICE_FAILED_EVENT,
                            SERVICE_FAILED_SUMMARY,
                            invocationSequence,
                            startedAtNanos);
                }
                throw cause;
            } catch (IllegalAccessException | RuntimeException | Error throwable) {
                if (terminalMeasurementEnabled) {
                    logServiceTerminal(
                            SERVICE_FAILED_EVENT,
                            SERVICE_FAILED_SUMMARY,
                            invocationSequence,
                            startedAtNanos);
                }
                throw throwable;
            } finally {
                requestContext.popService();
            }
        }

        private void logServiceStarted(
                final Method method,
                final Object[] args,
                final Long invocationSequence) {
            Map<String, Object> attributes = new LinkedHashMap<>();
            attributes.put(FUNCTION_ID_ATTRIBUTE, service.identifier());
            if (invocationSequence != null) {
                attributes.put(
                        LogField.SERVICE_INVOCATION_SEQUENCE.key(),
                        invocationSequence);
            }
            String callPath = requestContext.currentCallPath();
            if (callPath != null) {
                attributes.put(LogField.CALL_PATH.key(), callPath);
            }
            Map<String, Object> requestBody = toRequestBody(method, args);
            if (!requestBody.isEmpty()) {
                attributes.put(
                        LogField.REQUEST_BODY.key(),
                        requestBody);
            }
            LogEvent event = new LogEvent(
                    SERVICE_STARTED_EVENT,
                    SERVICE_STARTED_SUMMARY,
                    LogLayer.SV,
                    null,
                    null,
                    attributes);
            logger.info(event);
        }

        private void safeLogServiceStarted(
                final Method method,
                final Object[] args,
                final Long invocationSequence) {
            try {
                logServiceStarted(method, args, invocationSequence);
            } catch (RuntimeException ignored) {
                // Observability must not alter the Service result or failure.
            }
        }

        private void logServiceTerminal(
                final String eventName,
                final String summary,
                final Long invocationSequence,
                final long startedAtNanos) {
            try {
                Map<String, Object> attributes = new LinkedHashMap<>();
                attributes.put(FUNCTION_ID_ATTRIBUTE, service.identifier());
                if (invocationSequence != null) {
                    attributes.put(
                            LogField.SERVICE_INVOCATION_SEQUENCE.key(),
                            invocationSequence);
                }
                String callPath = requestContext.currentCallPath();
                if (callPath != null) {
                    attributes.put(LogField.CALL_PATH.key(), callPath);
                }
                attributes.put(
                        LogField.LATENCY_MS.key(),
                        Math.max(0L, System.nanoTime() - startedAtNanos) / 1_000_000L);
                safeInfo(new LogEvent(
                        eventName,
                        summary,
                        LogLayer.SV,
                        null,
                        null,
                        attributes));
            } catch (RuntimeException ignored) {
                // Observability must not alter the Service result or failure.
            }
        }

        private void safeInfo(final LogEvent event) {
            try {
                logger.info(event);
            } catch (RuntimeException ignored) {
                // Observability must not alter the Service result or failure.
            }
        }

        private Long nextServiceInvocationSequence() {
            try {
                return requestContext.nextServiceInvocationSequence();
            } catch (RuntimeException ignored) {
                return null;
            }
        }

        private Map<String, Object> toRequestBody(
            final Method method,
            final Object[] args) {
            if (args == null || args.length == 0) {
                return Map.of();
            }
            if (args.length == 1 && args[0] != null) {
                Map<String, Object> requestBody =
                    LogRequestBodyMapper.toMap(args[0]);
                if (!requestBody.isEmpty()) {
                    return requestBody;
                }
            }
            Map<String, Object> requestBody = new LinkedHashMap<>();
            Parameter[] parameters = method.getParameters();
            for (int index = 0; index < args.length; index++) {
                String parameterName = resolveParameterName(parameters, index);
                requestBody.put(parameterName, toRequestValue(args[index]));
            }
            return requestBody;
        }

        private String resolveParameterName(
                final Parameter[] parameters,
                final int index) {
            if (parameters.length > index) {
                String parameterName = parameters[index].getName();
                if (parameterName != null
                        && !parameterName.isBlank()) {
                    return parameterName;
                }
            }
            return "arg" + index;
        }

        private Object toRequestValue(final Object arg) {
            if (arg == null) {
                return null;
            }
            Map<String, Object> requestBody = LogRequestBodyMapper.toMap(arg);
            if (!requestBody.isEmpty()) {
                return requestBody;
            }
            return arg;
        }
    }
}
