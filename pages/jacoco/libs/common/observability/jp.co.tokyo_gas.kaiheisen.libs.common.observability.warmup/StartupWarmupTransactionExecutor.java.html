package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import jakarta.transaction.HeuristicMixedException;
import jakarta.transaction.HeuristicRollbackException;
import jakarta.transaction.NotSupportedException;
import jakarta.transaction.RollbackException;
import jakarta.transaction.Status;
import jakarta.transaction.SystemException;
import jakarta.transaction.UserTransaction;
import java.time.Duration;

/** DBウォームアップを新規JTAトランザクションで実行する。 */
@Singleton
public class StartupWarmupTransactionExecutor implements StartupWarmupTransaction {

    private final UserTransaction userTransaction;

    /**
     * コンストラクタ。
     *
     * @param userTransaction JTA UserTransaction
     */
    @Inject
    public StartupWarmupTransactionExecutor(final UserTransaction userTransaction) {
        this.userTransaction = userTransaction;
    }

    @Override
    public void execute(final StartupWarmupStep step, final Duration timeout) {
        boolean began = false;
        try {
            int timeoutSeconds = Math.max(1, Math.toIntExact((timeout.toMillis() + 999L) / 1000L));
            userTransaction.setTransactionTimeout(timeoutSeconds);
            userTransaction.begin();
            began = true;
            step.execute();
            userTransaction.commit();
            began = false;
        } catch (RuntimeException | Error exception) {
            rollbackIfNecessary(began, exception);
            throw exception;
        } catch (NotSupportedException | SystemException | RollbackException
                | HeuristicMixedException | HeuristicRollbackException exception) {
            rollbackIfNecessary(began, exception);
            throw new IllegalStateException("DB warmup transaction failed", exception);
        } finally {
            resetTimeout();
        }
    }

    private void rollbackIfNecessary(final boolean began, final Throwable original) {
        if (!began) {
            return;
        }
        try {
            int status = userTransaction.getStatus();
            if (status != Status.STATUS_NO_TRANSACTION) {
                userTransaction.rollback();
            }
        } catch (RuntimeException | SystemException rollbackFailure) {
            original.addSuppressed(rollbackFailure);
        }
    }

    private void resetTimeout() {
        try {
            userTransaction.setTransactionTimeout(0);
        } catch (SystemException ignored) {
            // 次回トランザクションには実行環境の既定値が適用されるため、起動処理を妨げない。
        }
    }
}
