package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import io.quarkus.runtime.StartupEvent;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.context.control.ActivateRequestContext;
import jakarta.enterprise.event.Observes;
import jakarta.enterprise.inject.Instance;
import jakarta.inject.Inject;
import java.time.Duration;
import java.util.UUID;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;

/** Quarkus起動イベントから共通ウォームアップを同期実行する。 */
@ApplicationScoped
public class StartupWarmupLifecycle {

    private static final String CORRELATION_PREFIX = "startup-warmup-";

    @Inject
    Instance<StartupWarmupContributor> contributors;

    @Inject
    StartupWarmupState state;

    @Inject
    StartupWarmupTransactionExecutor transaction;

    @Inject
    RequestContext requestContext;

    /**
     * HTTP受付開始前にウォームアップする。
     *
     * @param event Quarkus起動イベント
     */
    @ActivateRequestContext
    void onStart(@Observes final StartupEvent event) {
        StartupWarmupSettings settings = StartupWarmupSettings.load();
        StartupWarmupContributor contributor = settings.enabled() ? resolveContributor() : null;
        String correlationId = CORRELATION_PREFIX + UUID.randomUUID();
        requestContext.setCorrelationId(correlationId);
        requestContext.setHttpMethod("STARTUP");
        requestContext.setRequestPath("startup-warmup");
        requestContext.setRootFunctionIdIfAbsent("STARTUP-WARMUP");
        try {
            new StartupWarmupCoordinator(
                    settings,
                    contributor,
                    state,
                    new JbossStructuredLogger(StartupWarmupLifecycle.class),
                    transaction,
                    StartupWarmupLifecycle::sleep)
                    .run(correlationId);
        } finally {
            requestContext.clear();
        }
    }

    private StartupWarmupContributor resolveContributor() {
        if (contributors.isUnsatisfied()) {
            throw new IllegalStateException("StartupWarmupContributor is not configured");
        }
        if (contributors.isAmbiguous()) {
            throw new IllegalStateException("StartupWarmupContributor must be unique");
        }
        return contributors.get();
    }

    private static void sleep(final Duration delay) throws InterruptedException {
        Thread.sleep(delay);
    }
}
