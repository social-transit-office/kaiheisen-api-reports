package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;

/** アプリケーション固有の起動時ウォームアップ処理。 */
public interface StartupWarmupContributor {

    /**
     * アプリケーション名を返す。
     *
     * @return アプリケーション名
     */
    String applicationName();

    /**
     * ログレイヤを返す。
     *
     * @return ログレイヤ
     */
    LogLayer logLayer();

    /** 業務メソッドを呼ばずに代表Resourceと依存グラフを生成する。 */
    void initializeResources();

    /**
     * DBウォームアップDAを実行する。
     *
     * @return 確認値1
     */
    int warmUpDatabase();

    /**
     * アプリ固有の参照DAウォームアップステップを返す。
     *
     * @return 実行順に並んだDAウォームアップステップ
     */
    default List<StartupWarmupStep> businessDataAccessWarmupSteps() {
        return List.of();
    }
}
