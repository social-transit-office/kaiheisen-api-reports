package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import java.time.Duration;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupState.Snapshot;

/** 起動時ウォームアップの状態遷移、リトライ、計測を制御する。 */
final class StartupWarmupCoordinator {

    private static final String DATABASE_WARMUP_FUNCTION_ID = "DA-900-R001";

    private final StartupWarmupSettings settings;

    private final StartupWarmupContributor contributor;

    private final StartupWarmupState state;

    private final StructuredLogger logger;

    private final StartupWarmupTransaction transaction;

    private final Sleeper sleeper;

    StartupWarmupCoordinator(
            final StartupWarmupSettings settings,
            final StartupWarmupContributor contributor,
            final StartupWarmupState state,
            final StructuredLogger logger,
            final StartupWarmupTransaction transaction,
            final Sleeper sleeper) {
        this.settings = settings;
        this.contributor = contributor;
        this.state = state;
        this.logger = logger;
        this.transaction = transaction;
        this.sleeper = sleeper;
    }

    void run(final String correlationId) {
        if (!settings.enabled()) {
            state.update(Snapshot.disabled());
            logger.info(overallEvent(
                    "startup.warmup.disabled",
                    "Startup warmup is disabled",
                    StartupWarmupStatus.DISABLED,
                    correlationId,
                    0,
                    0,
                    0L,
                    0L,
                    0L));
            return;
        }

        settings.validate();
        long startedNanos = System.nanoTime();
        state.update(new Snapshot(
                StartupWarmupStatus.RUNNING, correlationId, 0, 0L, 0L, 0L));
        logger.info(overallEvent(
                "startup.warmup.started",
                "Startup warmup started",
                StartupWarmupStatus.RUNNING,
                correlationId,
                0,
                0,
                0L,
                0L,
                0L));

        long resourceStartedNanos = System.nanoTime();
        try {
            contributor.initializeResources();
        } catch (RuntimeException | Error exception) {
            long resourceLatencyMs = elapsedMillis(resourceStartedNanos);
            long totalLatencyMs = elapsedMillis(startedNanos);
            state.update(new Snapshot(
                    StartupWarmupStatus.FAILED,
                    correlationId,
                    0,
                    resourceLatencyMs,
                    0L,
                    totalLatencyMs));
            logger.error(overallEvent(
                    "startup.warmup.failed",
                    "Startup Resource initialization failed",
                    StartupWarmupStatus.FAILED,
                    correlationId,
                    0,
                    0,
                    resourceLatencyMs,
                    0L,
                    totalLatencyMs), exception);
            throw exception;
        }
        long resourceLatencyMs = elapsedMillis(resourceStartedNanos);
        if (elapsedMillis(startedNanos) >= settings.timeout().toMillis()) {
            IllegalStateException timeout =
                    new IllegalStateException("Startup Resource initialization timed out");
            long totalLatencyMs = elapsedMillis(startedNanos);
            state.update(new Snapshot(
                    StartupWarmupStatus.FAILED,
                    correlationId,
                    0,
                    resourceLatencyMs,
                    0L,
                    totalLatencyMs));
            logger.error(overallEvent(
                    "startup.warmup.failed",
                    "Startup Resource initialization timed out",
                    StartupWarmupStatus.FAILED,
                    correlationId,
                    0,
                    0,
                    resourceLatencyMs,
                    0L,
                    totalLatencyMs), timeout);
            throw timeout;
        }
        logger.info(overallEvent(
                "startup.warmup.resource.completed",
                "Startup Resource initialization completed",
                StartupWarmupStatus.RUNNING,
                correlationId,
                0,
                0,
                resourceLatencyMs,
                0L,
                elapsedMillis(startedNanos)));

        List<StartupWarmupStep> steps = warmupSteps();
        long dataAccessLatencyMs = 0L;
        int totalAttempts = 0;
        for (int stepIndex = 0; stepIndex < steps.size(); stepIndex++) {
            StartupWarmupStep step = steps.get(stepIndex);
            boolean stepSucceeded = false;
            for (int attempt = 1; attempt <= settings.maxAttempts(); attempt++) {
                Duration remaining = remaining(startedNanos);
                if (remaining.isZero() || remaining.isNegative()) {
                    break;
                }
                totalAttempts++;
                logger.info(stepEvent(
                        "startup.warmup.step.started",
                        "Startup DA warmup step started",
                        StartupWarmupStatus.RUNNING,
                        correlationId,
                        step,
                        stepIndex,
                        steps.size(),
                        attempt,
                        0L,
                        resourceLatencyMs,
                        dataAccessLatencyMs,
                        elapsedMillis(startedNanos)));
                long stepStartedNanos = System.nanoTime();
                try {
                    transaction.execute(step, remaining);
                    long stepLatencyMs = elapsedMillis(stepStartedNanos);
                    if (elapsedMillis(startedNanos) >= settings.timeout().toMillis()) {
                        throw new IllegalStateException("Startup DA warmup timed out");
                    }
                    dataAccessLatencyMs += stepLatencyMs;
                    logger.info(stepEvent(
                            "startup.warmup.step.completed",
                            "Startup DA warmup step completed",
                            StartupWarmupStatus.RUNNING,
                            correlationId,
                            step,
                            stepIndex,
                            steps.size(),
                            attempt,
                            stepLatencyMs,
                            resourceLatencyMs,
                            dataAccessLatencyMs,
                            elapsedMillis(startedNanos)));
                    stepSucceeded = true;
                    break;
                } catch (RuntimeException | Error exception) {
                    long stepLatencyMs = elapsedMillis(stepStartedNanos);
                    dataAccessLatencyMs += stepLatencyMs;
                    logger.warn(stepEvent(
                            "startup.warmup.step.attempt.failed",
                            "Startup DA warmup step attempt failed",
                            StartupWarmupStatus.RUNNING,
                            correlationId,
                            step,
                            stepIndex,
                            steps.size(),
                            attempt,
                            stepLatencyMs,
                            resourceLatencyMs,
                            dataAccessLatencyMs,
                            elapsedMillis(startedNanos)));
                }
                if (attempt < settings.maxAttempts() && !sleepBeforeRetry(startedNanos)) {
                    break;
                }
            }
            if (!stepSucceeded) {
                degrade(
                        correlationId,
                        steps.size(),
                        totalAttempts,
                        resourceLatencyMs,
                        dataAccessLatencyMs,
                        startedNanos);
                return;
            }
        }

        long totalLatencyMs = elapsedMillis(startedNanos);
        state.update(new Snapshot(
                StartupWarmupStatus.SUCCEEDED,
                correlationId,
                totalAttempts,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs));
        logger.info(overallEvent(
                "startup.warmup.completed",
                "Startup warmup completed",
                StartupWarmupStatus.SUCCEEDED,
                correlationId,
                steps.size(),
                totalAttempts,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs));
    }

    private List<StartupWarmupStep> warmupSteps() {
        List<StartupWarmupStep> steps = new ArrayList<>();
        steps.add(new StartupWarmupStep(DATABASE_WARMUP_FUNCTION_ID, () -> {
            if (contributor.warmUpDatabase() != 1) {
                throw new IllegalStateException("Startup DB warmup must return the value 1");
            }
        }));
        if (settings.businessDataAccessEnabled()) {
            List<StartupWarmupStep> businessSteps = Objects.requireNonNull(
                    contributor.businessDataAccessWarmupSteps(),
                    "businessDataAccessWarmupSteps");
            for (StartupWarmupStep step : businessSteps) {
                steps.add(Objects.requireNonNull(step, "businessDataAccessWarmupStep"));
            }
        }
        return List.copyOf(steps);
    }

    private void degrade(
            final String correlationId,
            final int stepCount,
            final int totalAttempts,
            final long resourceLatencyMs,
            final long dataAccessLatencyMs,
            final long startedNanos) {
        long totalLatencyMs = elapsedMillis(startedNanos);
        state.update(new Snapshot(
                StartupWarmupStatus.DEGRADED,
                correlationId,
                totalAttempts,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs));
        logger.warn(overallEvent(
                "startup.warmup.degraded",
                "Startup DB warmup failed; application startup continues",
                StartupWarmupStatus.DEGRADED,
                correlationId,
                stepCount,
                totalAttempts,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs));
    }

    private boolean sleepBeforeRetry(final long startedNanos) {
        Duration remaining = remaining(startedNanos);
        Duration delay = settings.retryInitialDelay().compareTo(remaining) < 0
                ? settings.retryInitialDelay()
                : remaining;
        if (delay.isZero() || delay.isNegative()) {
            return false;
        }
        try {
            sleeper.sleep(delay);
            return true;
        } catch (InterruptedException exception) {
            Thread.currentThread().interrupt();
            return false;
        }
    }

    private Duration remaining(final long startedNanos) {
        long remainingNanos = settings.timeout().toNanos()
                - Math.max(0L, System.nanoTime() - startedNanos);
        return remainingNanos <= 0L ? Duration.ZERO : Duration.ofNanos(remainingNanos);
    }

    private LogEvent overallEvent(
            final String event,
            final String summary,
            final StartupWarmupStatus status,
            final String correlationId,
            final int stepCount,
            final int attempts,
            final long resourceLatencyMs,
            final long dataAccessLatencyMs,
            final long totalLatencyMs) {
        return event(
                event,
                summary,
                status,
                correlationId,
                DATABASE_WARMUP_FUNCTION_ID,
                0,
                stepCount,
                attempts,
                0L,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs);
    }

    private LogEvent stepEvent(
            final String event,
            final String summary,
            final StartupWarmupStatus status,
            final String correlationId,
            final StartupWarmupStep step,
            final int zeroBasedStepIndex,
            final int stepCount,
            final int attempt,
            final long stepLatencyMs,
            final long resourceLatencyMs,
            final long dataAccessLatencyMs,
            final long totalLatencyMs) {
        return event(
                event,
                summary,
                status,
                correlationId,
                step.functionId(),
                zeroBasedStepIndex + 1,
                stepCount,
                attempt,
                stepLatencyMs,
                resourceLatencyMs,
                dataAccessLatencyMs,
                totalLatencyMs);
    }

    private LogEvent event(
            final String event,
            final String summary,
            final StartupWarmupStatus status,
            final String correlationId,
            final String functionId,
            final int stepIndex,
            final int stepCount,
            final int attempt,
            final long stepLatencyMs,
            final long resourceLatencyMs,
            final long dataAccessLatencyMs,
            final long totalLatencyMs) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(LogField.CORRELATION_ID.key(), correlationId);
        attributes.put("applicationName", contributor == null ? null : contributor.applicationName());
        attributes.put("state", status.name());
        attributes.put("functionId", functionId);
        attributes.put("stepIndex", stepIndex);
        attributes.put("stepCount", stepCount);
        attributes.put("attempt", attempt);
        attributes.put("maxAttempts", settings.maxAttempts());
        attributes.put("stepLatencyMs", stepLatencyMs);
        attributes.put("resourceLatencyMs", resourceLatencyMs);
        attributes.put("dataAccessLatencyMs", dataAccessLatencyMs);
        attributes.put("totalLatencyMs", totalLatencyMs);
        attributes.put("timeoutMs", settings.timeout().toMillis());
        return new LogEvent(
                event,
                summary,
                contributor == null ? LogLayer.DA : contributor.logLayer(),
                null,
                null,
                attributes);
    }

    private static long elapsedMillis(final long startedNanos) {
        return Math.max(0L, System.nanoTime() - startedNanos) / 1_000_000L;
    }

    @FunctionalInterface
    interface Sleeper {
        void sleep(Duration delay) throws InterruptedException;
    }
}
