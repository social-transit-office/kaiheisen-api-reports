package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import java.util.Objects;

/** 起動時に独立したトランザクションで実行するDAウォームアップステップ。 */
public record StartupWarmupStep(String functionId, Action action) {

    /**
     * コンストラクタ。
     *
     * @param functionId DA識別子
     * @param action パラメータを保持する実行処理
     */
    public StartupWarmupStep {
        if (functionId == null || functionId.isBlank()) {
            throw new IllegalArgumentException("functionId must not be blank");
        }
        Objects.requireNonNull(action, "action");
    }

    /** ウォームアップ処理を実行する。 */
    public void execute() {
        action.execute();
    }

    /** パラメータを外部へ公開せずにDA呼び出しを保持する処理。 */
    @FunctionalInterface
    public interface Action {

        /** DA呼び出しと結果検証を実行する。 */
        void execute();
    }
}
