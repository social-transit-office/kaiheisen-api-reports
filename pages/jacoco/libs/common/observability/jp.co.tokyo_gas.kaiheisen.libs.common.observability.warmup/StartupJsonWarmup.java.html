package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;

/** 業務DTOを列挙せず共通のJSON出力処理を初期化する。 */
public final class StartupJsonWarmup {

    private StartupJsonWarmup() {
    }

    /**
     * アプリケーションが実際に使うMapperで小さな値をJSON化する。
     *
     * <p>個別DTOやHTTP応答ライターの初期化までは保証しない。</p>
     *
     * @param objectMapper アプリケーション共通のObjectMapper
     */
    public static void initialize(ObjectMapper objectMapper) {
        try {
            objectMapper.writeValueAsBytes(new Probe("ready", 1));
        } catch (JsonProcessingException exception) {
            throw new IllegalStateException("Startup JSON warmup failed", exception);
        }
    }

    private record Probe(String state, int value) {
    }
}
