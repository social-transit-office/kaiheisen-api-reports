package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import jakarta.enterprise.context.ApplicationScoped;

/** 起動時ウォームアップの直近状態を保持する。 */
@ApplicationScoped
public class StartupWarmupState {

    private volatile Snapshot snapshot = Snapshot.disabled();

    /**
     * 現在の状態を返す。
     *
     * @return 状態スナップショット
     */
    public Snapshot snapshot() {
        return snapshot;
    }

    void update(final Snapshot next) {
        snapshot = next;
    }

    /**
     * ウォームアップの状態スナップショット。
     *
     * @param status 状態
     * @param correlationId 相関ID
     * @param attempts 全DAウォームアップステップの総試行回数
     * @param resourceLatencyMs Resource初期化時間
     * @param dataAccessLatencyMs 全DAウォームアップ試行の累計時間
     * @param totalLatencyMs 総時間
     */
    public record Snapshot(
            StartupWarmupStatus status,
            String correlationId,
            int attempts,
            long resourceLatencyMs,
            long dataAccessLatencyMs,
            long totalLatencyMs) {

        static Snapshot disabled() {
            return new Snapshot(StartupWarmupStatus.DISABLED, null, 0, 0L, 0L, 0L);
        }
    }
}
