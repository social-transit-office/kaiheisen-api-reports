package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import jakarta.enterprise.context.Dependent;
import jakarta.enterprise.context.spi.Context;
import jakarta.enterprise.context.spi.CreationalContext;
import jakarta.enterprise.inject.Any;
import jakarta.enterprise.inject.spi.Bean;
import jakarta.enterprise.inject.spi.BeanManager;
import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import java.util.Set;

/** CDI Beanのコンストラクタと依存グラフを明示的に初期化する。 */
@Singleton
public class CdiBeanInitializer {

    private final BeanManager beanManager;

    /**
     * コンストラクタ。
     *
     * @param beanManager CDI BeanManager
     */
    @Inject
    public CdiBeanInitializer(final BeanManager beanManager) {
        this.beanManager = beanManager;
    }

    /**
     * 指定Beanを業務メソッドの呼出しなしで生成する。
     *
     * @param beanClasses 初期化対象Beanクラス
     */
    public void initialize(final Class<?>... beanClasses) {
        for (Class<?> beanClass : beanClasses) {
            initialize(beanClass);
        }
    }

    @SuppressWarnings({"rawtypes", "unchecked"})
    private void initialize(final Class<?> beanClass) {
        Set<Bean<?>> beans = beanManager.getBeans(beanClass, Any.Literal.INSTANCE);
        if (beans.isEmpty()) {
            throw new IllegalStateException("Warmup Resource is not a CDI Bean: " + beanClass.getName());
        }
        Bean bean = beanManager.resolve(beans);
        if (bean == null) {
            throw new IllegalStateException("Warmup Resource is ambiguous: " + beanClass.getName());
        }
        CreationalContext context = beanManager.createCreationalContext(bean);
        if (Dependent.class.equals(bean.getScope())) {
            Object instance = bean.create(context);
            if (instance == null) {
                throw new IllegalStateException("Warmup Resource could not be created: " + beanClass.getName());
            }
            bean.destroy(instance, context);
            return;
        }
        Context scopeContext = beanManager.getContext(bean.getScope());
        Object instance = scopeContext.get(bean, context);
        if (instance == null) {
            throw new IllegalStateException("Warmup Resource could not be created: " + beanClass.getName());
        }
    }
}
