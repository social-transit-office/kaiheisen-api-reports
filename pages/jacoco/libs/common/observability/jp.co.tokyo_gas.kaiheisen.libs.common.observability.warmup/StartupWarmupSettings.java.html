package jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup;

import java.time.Duration;
import org.eclipse.microprofile.config.Config;
import org.eclipse.microprofile.config.ConfigProvider;

/** 起動時ウォームアップ設定。 */
public record StartupWarmupSettings(
        boolean enabled,
        boolean businessDataAccessEnabled,
        Duration timeout,
        int maxAttempts,
        Duration retryInitialDelay) {

    /** 有効化設定キー。 */
    public static final String ENABLED_KEY = "kaiheisen.startup-warmup.enabled";

    /** 業務参照DAウォームアップ有効化設定キー。 */
    public static final String BUSINESS_DATA_ACCESS_ENABLED_KEY =
            "kaiheisen.startup-warmup.business-da.enabled";

    /** タイムアウト設定キー。 */
    public static final String TIMEOUT_KEY = "kaiheisen.startup-warmup.timeout";

    /** 最大試行回数設定キー。 */
    public static final String MAX_ATTEMPTS_KEY = "kaiheisen.startup-warmup.max-attempts";

    /** 初回リトライ待機時間設定キー。 */
    public static final String RETRY_INITIAL_DELAY_KEY =
            "kaiheisen.startup-warmup.retry-initial-delay";

    private static final Duration DEFAULT_TIMEOUT = Duration.ofSeconds(60);

    private static final int DEFAULT_MAX_ATTEMPTS = 3;

    private static final Duration DEFAULT_RETRY_DELAY = Duration.ofSeconds(1);

    /**
     * MicroProfile Configから設定を読み込む。
     *
     * @return 検証済み設定
     */
    public static StartupWarmupSettings load() {
        Config config = ConfigProvider.getConfig();
        StartupWarmupSettings settings =
                new StartupWarmupSettings(
                        config.getOptionalValue(ENABLED_KEY, Boolean.class).orElse(false),
                        config.getOptionalValue(BUSINESS_DATA_ACCESS_ENABLED_KEY, Boolean.class)
                                .orElse(false),
                        config.getOptionalValue(TIMEOUT_KEY, Duration.class)
                                .orElse(DEFAULT_TIMEOUT),
                        config.getOptionalValue(MAX_ATTEMPTS_KEY, Integer.class)
                                .orElse(DEFAULT_MAX_ATTEMPTS),
                        config.getOptionalValue(RETRY_INITIAL_DELAY_KEY, Duration.class)
                                .orElse(DEFAULT_RETRY_DELAY));
        settings.validate();
        return settings;
    }

    void validate() {
        if (timeout == null || timeout.isZero() || timeout.isNegative()) {
            throw new IllegalArgumentException(TIMEOUT_KEY + " must be greater than zero");
        }
        if (maxAttempts < 1) {
            throw new IllegalArgumentException(MAX_ATTEMPTS_KEY + " must be at least one");
        }
        if (retryInitialDelay == null || retryInitialDelay.isNegative()) {
            throw new IllegalArgumentException(RETRY_INITIAL_DELAY_KEY + " must not be negative");
        }
    }
}
