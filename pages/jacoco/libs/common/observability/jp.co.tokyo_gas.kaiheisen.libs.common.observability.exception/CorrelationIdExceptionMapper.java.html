package jp.co.tokyo_gas.kaiheisen.libs.common.observability.exception;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.Response;
import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.ext.Provider;
import java.util.UUID;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.DefaultExceptionLogMapper;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.ExceptionLogContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.ExceptionLogMapper;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogDecision;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLevel;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.filter.CorrelationIdFilter;
import org.jboss.logging.MDC;

/**
 * 例外発生時に相関IDヘッダ補完とコンテキスト cleanup を行う共通ハンドラ。
 */
@Provider
@ApplicationScoped
public class CorrelationIdExceptionMapper implements ExceptionMapper<Throwable> {

    private static final String MDC_KEY = "correlationId";

    private final RequestContext requestContext;

    private final StructuredLogger logger;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public CorrelationIdExceptionMapper(RequestContext requestContext) {
        this(requestContext, new JbossStructuredLogger(CorrelationIdExceptionMapper.class));
    }

    CorrelationIdExceptionMapper(RequestContext requestContext, StructuredLogger logger) {
        this.requestContext = requestContext;
        this.logger = logger;
    }

    @Override
    public Response toResponse(Throwable exception) {
        String correlationId = requestContext.getCorrelationId().orElseGet(() -> {
            String generated = UUID.randomUUID().toString();
            requestContext.setCorrelationId(generated);
            return generated;
        });
        MDC.put(MDC_KEY, correlationId);
        logIfNeeded(exception);

        Response base = exception instanceof WebApplicationException webEx
                ? webEx.getResponse()
                : Response.serverError().build();

        Response response = Response.fromResponse(base)
                .header(CorrelationIdFilter.HEADER_NAME, correlationId)
                .build();

        MDC.remove(MDC_KEY);
        requestContext.clear();
        return response;
    }

    private void logIfNeeded(Throwable exception) {
        if (requestContext.isExceptionLogged()) {
            return;
        }
        ExceptionLogMapper exceptionLogMapper = new DefaultExceptionLogMapper(resolveLogLayer());
        LogDecision decision = exceptionLogMapper.map(exception, new ExceptionLogContext(
                requestContext.getCorrelationId().orElse(null),
                requestContext.getHttpMethod().orElse(null),
                requestContext.getRequestPath().orElse(null),
                null,
                requestContext.getServiceId().orElse(null),
                requestContext.getServiceName().orElse(null)));
        if (decision == null || decision.event() == null) {
            return;
        }
        if (decision.level() == LogLevel.ERROR) {
            logger.error(decision.event(), exception);
        } else if (decision.level() == LogLevel.WARN) {
            logger.warn(decision.event());
        } else {
            logger.info(decision.event());
        }
        requestContext.markExceptionLogged();
    }

    // Common fallback infers the app layer from the request path. BFF paths start with /livalit/,
    // and the remaining current public APIs are MA.
    private LogLayer resolveLogLayer() {
        String requestPath = requestContext.getRequestPath().orElse(null);
        if (requestPath != null && requestPath.startsWith("/livalit/")) {
            return LogLayer.BFF;
        }
        return LogLayer.MA;
    }
}
