/**
 * @file AbstractRequestLoggingFilter.java
 * 概要: HTTP 入口/出口ログの共通テンプレートを定義する。
 * 主な仕様:
 * - クエリ/ボディを含むリクエスト情報をログ出力する。
 * - 入口/出口のレイテンシを計測する。
 * 既知の制限事項:
 * - 実際の登録はアプリ側で @Provider を付与して行う。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.observability.logging;

import jakarta.ws.rs.container.ContainerRequestContext;
import jakarta.ws.rs.container.ContainerRequestFilter;
import jakarta.ws.rs.container.ContainerResponseContext;
import jakarta.ws.rs.container.ContainerResponseFilter;
import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.PerformanceObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;

/**
 * HTTP 入口/出口ログの共通テンプレート。
 */
public abstract class AbstractRequestLoggingFilter
        implements ContainerRequestFilter, ContainerResponseFilter {

    private static final String START_TIME_NANOS = "logging.startTimeNanos";

    private static final String REQUEST_METHOD = "logging.httpMethod";

    private static final String REQUEST_PATH = "logging.path";

    private final StructuredLogger logger;

    private final LogLayer layer;

    /**
     * コンストラクタ。
     *
     * @param owner ロガーの所有クラス。型: Class<?>
     * @param layer レイヤ。型: LogLayer
     */
    protected AbstractRequestLoggingFilter(Class<?> owner, LogLayer layer) {
        this(new JbossStructuredLogger(owner), layer);
    }

    /**
     * コンストラクタ。
     *
     * @param logger 構造化ロガー。型: StructuredLogger
     * @param layer レイヤ。型: LogLayer
     */
    protected AbstractRequestLoggingFilter(StructuredLogger logger, LogLayer layer) {
        this.logger = logger;
        this.layer = layer;
    }

    @Override
    public void filter(ContainerRequestContext requestContext) throws IOException {
        String method = requestContext.getMethod();
        String path = requestContext.getUriInfo().getPath();
        long startNanos = System.nanoTime();
        requestContext.setProperty(START_TIME_NANOS, startNanos);
        requestContext.setProperty(REQUEST_METHOD, method);
        requestContext.setProperty(REQUEST_PATH, path);

        onRequestStartNanos(startNanos);

        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(LogField.HTTP_METHOD.key(), method);
        attributes.put(LogField.PATH.key(), path);
        attributes.put(LogField.QUERY_PARAMS.key(), toQueryParams(requestContext));
        attributes.put(LogField.REQUEST_BODY.key(), readRequestBody(requestContext));

        logger.info(new LogEvent(
                "request.in",
                "HTTP request received",
                layer,
                null,
                null,
                attributes));
    }

    /**
     * リクエスト入口時刻をサブクラスへ通知する。
     *
     * @param startNanos {@link System#nanoTime()} で取得した開始時刻
     */
    protected void onRequestStartNanos(final long startNanos) {
        // 既定では何もしない
    }

    /**
     * 詳細性能計測が有効かを返す。
     *
     * @return 有効の場合 {@code true}
     */
    protected boolean performanceObservabilityEnabled() {
        return PerformanceObservabilitySettings.isEnabled();
    }

    @Override
    public void filter(ContainerRequestContext requestContext, ContainerResponseContext responseContext) {
        Long start = (Long) requestContext.getProperty(START_TIME_NANOS);
        long latencyMs = start == null ? 0L : (System.nanoTime() - start) / 1_000_000L;
        String method = (String) requestContext.getProperty(REQUEST_METHOD);
        String path = (String) requestContext.getProperty(REQUEST_PATH);

        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(LogField.HTTP_METHOD.key(), method);
        attributes.put(LogField.PATH.key(), path);
        attributes.put(LogField.LATENCY_MS.key(), latencyMs);

        logger.info(new LogEvent(
                "request.out",
                "HTTP response sent",
                layer,
                responseContext.getStatus(),
                null,
                attributes));
    }

    private static Map<String, List<String>> toQueryParams(ContainerRequestContext requestContext) {
        Map<String, List<String>> params = new LinkedHashMap<>();
        requestContext.getUriInfo().getQueryParameters().forEach(params::put);
        return params;
    }

    private static String readRequestBody(ContainerRequestContext requestContext) throws IOException {
        InputStream inputStream = requestContext.getEntityStream();
        if (inputStream == null) {
            return null;
        }
        byte[] body = readAllBytes(inputStream);
        requestContext.setEntityStream(new ByteArrayInputStream(body));
        if (body.length == 0) {
            return null;
        }
        return new String(body, StandardCharsets.UTF_8);
    }

    private static byte[] readAllBytes(InputStream inputStream) throws IOException {
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        byte[] buffer = new byte[1024];
        int read;
        while ((read = inputStream.read(buffer)) != -1) {
            output.write(buffer, 0, read);
        }
        return output.toByteArray();
    }
}
