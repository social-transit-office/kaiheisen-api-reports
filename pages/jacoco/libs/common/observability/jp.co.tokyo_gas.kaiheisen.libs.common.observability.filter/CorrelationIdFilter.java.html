package jp.co.tokyo_gas.kaiheisen.libs.common.observability.filter;

import jakarta.annotation.Priority;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.Priorities;
import jakarta.ws.rs.container.ContainerRequestContext;
import jakarta.ws.rs.container.ContainerRequestFilter;
import jakarta.ws.rs.container.ContainerResponseContext;
import jakarta.ws.rs.container.ContainerResponseFilter;
import jakarta.ws.rs.ext.Provider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.CorrelationIdResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import org.jboss.logging.MDC;

/**
 * 相関IDをリクエストスコープとレスポンスヘッダーへ反映するフィルタ。
 */
@Provider
@Priority(Priorities.AUTHENTICATION)
@ApplicationScoped
public class CorrelationIdFilter implements ContainerRequestFilter, ContainerResponseFilter {

    public static final String HEADER_NAME = "X-Correlation-Id";

    private static final String MDC_KEY = "correlationId";

    private final RequestContext requestContext;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public CorrelationIdFilter(RequestContext requestContext) {
        this.requestContext = requestContext;
    }

    @Override
    public void filter(ContainerRequestContext requestContext) {
        String correlationId = CorrelationIdResolver.resolve(requestContext.getHeaderString(HEADER_NAME));
        this.requestContext.setCorrelationId(correlationId);
        this.requestContext.setHttpMethod(requestContext.getMethod());
        this.requestContext.setRequestPath(requestContext.getUriInfo().getRequestUri().getPath());
        MDC.put(MDC_KEY, correlationId);
    }

    @Override
    public void filter(ContainerRequestContext requestContext, ContainerResponseContext responseContext) {
        this.requestContext.getCorrelationId().ifPresent(
                cid -> responseContext.getHeaders().putSingle(HEADER_NAME, cid));
        MDC.remove(MDC_KEY);
        this.requestContext.clear();
    }
}
