package jp.co.tokyo_gas.kaiheisen.libs.common.core.context;

import jakarta.enterprise.context.RequestScoped;
import java.time.LocalDateTime;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.List;
import java.util.Optional;

/**
 * リクエストスコープで共有するコンテキスト情報。
 */
@RequestScoped
public class RequestContext {

    private String correlationId;

    private String httpMethod;

    private String requestPath;

    private String hukkSysShbt;

    private LocalDateTime systemDatetime;

    private String rootFunctionId;

    private boolean exceptionLogged;

    private Long requestInNanos;

    private long dataAccessSequence;

    private long externalApiSequence;

    private long serviceSequence;

    private final Deque<DataAccessInvocation> dataAccessStack = new ArrayDeque<>();

    private final Deque<ServiceContext> serviceStack = new ArrayDeque<>();

    /**
     * 相関IDを取得する。
     *
     * @return 相関ID。未設定の場合は空
     */
    public Optional<String> getCorrelationId() {
        return Optional.ofNullable(correlationId);
    }

    /**
     * 相関IDを設定する。
     *
     * <p>MA/BFF が扱う相関IDは常に UUID v4 とする。未指定または不正値は新規 UUID v4 に置換する。</p>
     *
     * @param correlationId 相関IDの候補値
     */
    public void setCorrelationId(String correlationId) {
        this.correlationId = correlationId;
    }

    /**
     * HTTP メソッドを取得する。
     *
     * @return HTTP メソッド。未設定の場合は空
     */
    public Optional<String> getHttpMethod() {
        return Optional.ofNullable(httpMethod);
    }

    /**
     * HTTP メソッドを設定する。
     *
     * @param httpMethod HTTP メソッド
     */
    public void setHttpMethod(String httpMethod) {
        this.httpMethod = httpMethod;
    }

    /**
     * リクエストパスを取得する。
     *
     * @return リクエストパス。未設定の場合は空
     */
    public Optional<String> getRequestPath() {
        return Optional.ofNullable(requestPath);
    }

    /**
     * リクエストパスを設定する。
     *
     * @param requestPath リクエストパス
     */
    public void setRequestPath(String requestPath) {
        this.requestPath = requestPath;
    }

    /**
     * 報告システム種別を取得する。
     *
     * @return 報告システム種別。未設定の場合は空
     */
    public Optional<String> getHukkSysShbt() {
        return Optional.ofNullable(hukkSysShbt);
    }

    /**
     * 報告システム種別を設定する。
     *
     * @param hukkSysShbt 報告システム種別
     */
    public void setHukkSysShbt(String hukkSysShbt) {
        this.hukkSysShbt = hukkSysShbt;
    }

    /**
     * システム日時を取得する。
     *
     * @return システム日時。未設定の場合は空
     */
    public Optional<LocalDateTime> getSystemDatetime() {
        return Optional.ofNullable(systemDatetime);
    }

    /**
     * システム日時を設定する。
     *
     * @param systemDatetime システム日時
     */
    public void setSystemDatetime(LocalDateTime systemDatetime) {
        this.systemDatetime = systemDatetime;
    }

    /**
     * 起点機能 ID を取得する。
     *
     * @return 起点機能 ID。未設定の場合は空
     */
    public Optional<String> getRootFunctionId() {
        return Optional.ofNullable(rootFunctionId);
    }

    /**
     * 起点機能 ID を未設定時のみ設定する。
     *
     * @param rootFunctionId 起点機能 ID
     */
    public void setRootFunctionIdIfAbsent(final String rootFunctionId) {
        if (this.rootFunctionId == null && rootFunctionId != null && !rootFunctionId.isBlank()) {
            this.rootFunctionId = rootFunctionId;
        }
    }

    /**
     * 例外ログ出力済みかを返す。
     *
     * @return 例外ログ出力済みの場合 true
     */
    public boolean isExceptionLogged() {
        return exceptionLogged;
    }

    /**
     * 例外ログ出力済みとして記録する。
     */
    public void markExceptionLogged() {
        this.exceptionLogged = true;
    }

    /**
     * {@code request.in} の計測開始時刻を記録する。
     *
     * @param requestInNanos {@link System#nanoTime()} で取得した開始時刻
     */
    public void setRequestInNanos(final long requestInNanos) {
        this.requestInNanos = requestInNanos;
    }

    /**
     * {@code request.in} から現在までの経過時間をミリ秒で返す。
     *
     * @param nowNanos {@link System#nanoTime()} で取得した現在時刻
     * @return 経過時間。開始時刻が未設定の場合は空
     */
    public Optional<Long> elapsedSinceRequestInMillis(final long nowNanos) {
        if (requestInNanos == null) {
            return Optional.empty();
        }
        return Optional.of(Math.max(0L, nowNanos - requestInNanos) / 1_000_000L);
    }

    /**
     * DataAccess 呼出しを開始し、リクエスト内で一意な連番を採番する。
     *
     * @param functionId DataAccess 機能 ID
     * @return 呼出しコンテキスト
     */
    public DataAccessInvocation beginDataAccess(final String functionId) {
        DataAccessInvocation invocation = new DataAccessInvocation(++dataAccessSequence, functionId);
        dataAccessStack.push(invocation);
        return invocation;
    }

    /**
     * 外部 API 呼び出しへリクエスト内で一意な連番を付与する。
     *
     * @return リクエスト内連番
     */
    public long nextExternalApiInvocationSequence() {
        return ++externalApiSequence;
    }

    /**
     * Assigns a request-local sequence to a Service invocation.
     *
     * @return request-local sequence
     */
    public long nextServiceInvocationSequence() {
        return ++serviceSequence;
    }

    /**
     * 現在実行中の DataAccess 呼出しを返す。
     *
     * @return 呼出しコンテキスト。DataAccess 外の場合は空
     */
    public Optional<DataAccessInvocation> currentDataAccess() {
        return Optional.ofNullable(dataAccessStack.peek());
    }

    /**
     * 指定した DataAccess 呼出しを終了する。
     *
     * @param invocation 終了する呼出しコンテキスト
     */
    public void endDataAccess(final DataAccessInvocation invocation) {
        if (invocation == null || dataAccessStack.isEmpty()) {
            return;
        }
        if (invocation.equals(dataAccessStack.peek())) {
            dataAccessStack.pop();
            return;
        }
        dataAccessStack.remove(invocation);
    }

    /**
     * 実行中サービス ID を取得する。
     *
     * @return サービス ID。未設定の場合は空
     */
    public Optional<String> getServiceId() {
        ServiceContext current = serviceStack.peek();
        return current == null ? Optional.empty() : Optional.ofNullable(current.serviceId());
    }

    /**
     * 実行中サービス名を取得する。
     *
     * @return サービス名。未設定の場合は空
     */
    public Optional<String> getServiceName() {
        ServiceContext current = serviceStack.peek();
        return current == null ? Optional.empty() : Optional.ofNullable(current.serviceName());
    }

    /**
     * 実行中サービスを push する。
     *
     * @param serviceId サービス ID
     * @param serviceName サービス名
     */
    public void pushService(String serviceId, String serviceName) {
        serviceStack.push(new ServiceContext(serviceId, serviceName));
    }

    /**
     * 実行中サービスを pop する。
     */
    public void popService() {
        if (!serviceStack.isEmpty()) {
            serviceStack.pop();
        }
    }

    /**
     * 現在の呼出し経路を返す。
     *
     * @return 呼出し経路。未設定の場合は null
     */
    public String currentCallPath() {
        return buildCallPath(null);
    }

    /**
     * 現在の呼出し経路に末端機能 ID を付与した経路を返す。
     *
     * @param terminalFunctionId 末端機能 ID
     * @return 呼出し経路。未設定の場合は null
     */
    public String buildCallPath(final String terminalFunctionId) {
        List<String> segments = new ArrayList<>();
        if (rootFunctionId != null && !rootFunctionId.isBlank()) {
            segments.add(rootFunctionId);
        }
        serviceStack.descendingIterator().forEachRemaining(serviceContext -> {
            String serviceId = serviceContext.serviceId();
            if (serviceId != null && !serviceId.isBlank()) {
                segments.add(serviceId);
            }
        });
        if (terminalFunctionId != null && !terminalFunctionId.isBlank()) {
            if (segments.isEmpty() || !terminalFunctionId.equals(segments.get(segments.size() - 1))) {
                segments.add(terminalFunctionId);
            }
        }
        if (segments.isEmpty()) {
            return null;
        }
        return String.join("->", segments);
    }

    /**
     * コンテキスト情報をクリアする。
     */
    public void clear() {
        this.correlationId = null;
        this.httpMethod = null;
        this.requestPath = null;
        this.hukkSysShbt = null;
        this.systemDatetime = null;
        this.rootFunctionId = null;
        this.exceptionLogged = false;
        this.requestInNanos = null;
        this.dataAccessSequence = 0L;
        this.externalApiSequence = 0L;
        this.serviceSequence = 0L;
        this.serviceStack.clear();
        this.dataAccessStack.clear();
    }

    /**
     * リクエスト内の DataAccess 呼出しを識別する値。
     *
     * @param sequence リクエスト内連番
     * @param functionId DataAccess 機能 ID
     */
    public record DataAccessInvocation(long sequence, String functionId) {
    }

    private record ServiceContext(String serviceId, String serviceName) {
    }
}
