package jp.co.tokyo_gas.kaiheisen.libs.common.core.context;

import java.util.UUID;

/**
 * 相関IDの正規化と採番を行うユーティリティ。
 */
public final class CorrelationIdResolver {

    /** UUID形式の相関IDの文字数。 */
    public static final int UUID_LENGTH = 36;

    private static final String UUID_V4_PATTERN =
            "^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-4[0-9a-fA-F]{3}-[89aAbB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}$";

    private static final String SHORT_CORRELATION_ID_PATTERN = "^[A-Za-z0-9_-]{1,20}$";

    private CorrelationIdResolver() {
        // ユーティリティクラスのためインスタンス化させない
    }

    /**
     * リクエストヘッダの生値から相関IDを解決する。
     *
     * <p>UUID v4 形式または既存互換の短縮IDであれば受信値をそのまま採用し、
     * それ以外の場合は新規 UUID v4 を採番する。</p>
     *
     * @param rawHeader ヘッダ値
     * @return 正規化済み相関ID
     */
    public static String resolve(String rawHeader) {
        String normalized = normalize(rawHeader);
        if (normalized != null && (isValidUuidV4(normalized) || normalized.matches(SHORT_CORRELATION_ID_PATTERN))) {
            return normalized;
        }
        return UUID.randomUUID().toString();
    }

    /**
     * リクエスト値から UUID v4 形式の相関IDを解決する。
     *
     * <p>UUID v4 形式の文字列であれば受信値をそのまま採用し、それ以外の場合は新規 UUID v4 を採番する。</p>
     *
     * @param rawValue リクエスト値
     * @return UUID v4 形式の相関ID
     */
    public static String resolveUuidV4(String rawValue) {
        String normalized = normalize(rawValue);
        if (normalized != null && isValidUuidV4(normalized)) {
            return normalized;
        }
        return UUID.randomUUID().toString();
    }

    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }

    /**
     * UUID v4形式かを判定する。
     *
     * @param value 判定対象の文字列
     * @return UUID v4形式の場合は {@code true}
     */
    public static boolean isValidUuidV4(String value) {
        if (value == null || !value.matches(UUID_V4_PATTERN)) {
            return false;
        }
        try {
            return UUID.fromString(value).version() == 4;
        } catch (IllegalArgumentException ex) {
            return false;
        }
    }

}
