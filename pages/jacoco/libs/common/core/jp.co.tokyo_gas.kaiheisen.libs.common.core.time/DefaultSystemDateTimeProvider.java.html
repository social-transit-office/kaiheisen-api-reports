/**
 * @file DefaultSystemDateTimeProvider.java
 * 概要: SystemDateTimeProvider のデフォルト実装。
 * 主な仕様:
 * - システムデフォルトのタイムゾーン/クロックを使用する。
 * - テスト用に Clock を注入可能。
 * 既知の制限事項:
 * - なし。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.time;

import jakarta.inject.Singleton;
import java.time.Clock;
import java.time.LocalDate;
import java.time.LocalDateTime;

/**
 * {@link SystemDateTimeProvider} のデフォルト実装。システムデフォルトのタイムゾーン/クロックを使用する。
 *
 * @since 1.0.0
 */
@Singleton
public class DefaultSystemDateTimeProvider implements SystemDateTimeProvider {

    private final Clock clock;

    /**
     * システムデフォルトのタイムゾーンでインスタンスを生成する。
     */
    public DefaultSystemDateTimeProvider() {
        this(Clock.systemDefaultZone());
    }

    /**
     * テスト用コンストラクタ。
     *
     * @param clock 使用するクロック
     */
    DefaultSystemDateTimeProvider(Clock clock) {
        this.clock = clock;
    }

    @Override
    public LocalDate currentDate() {
        return LocalDate.now(clock);
    }

    @Override
    public LocalDateTime currentDateTime() {
        return LocalDateTime.now(clock);
    }
}
