package jp.co.tokyo_gas.kaiheisen.libs.common.core.time;

import java.time.LocalDate;
import java.util.Objects;

/**
 * 日付計算の共通ユーティリティ。
 */
public final class DateCalculationUtil {

    private DateCalculationUtil() {
    }

    /**
     * 基準日から指定月数前の月初日を返却する。
     *
     * @param baseDate 基準日
     * @param monthsAgo 遡る月数
     * @return 指定月数前の月初日
     */
    public static LocalDate firstDayMonthsAgo(LocalDate baseDate, int monthsAgo) {
        LocalDate resolvedBaseDate = Objects.requireNonNull(baseDate, "baseDate");
        return resolvedBaseDate.minusMonths(monthsAgo).withDayOfMonth(1);
    }
}
