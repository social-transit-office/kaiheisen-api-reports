/**
 * @file Rfc3339LocalDateParser.java
 * 概要: RFC3339 日付(yyyy-MM-dd)を LocalDate に変換する。
 * 主な仕様:
 * - ISO_LOCAL_DATE フォーマッタを使用してパースする。
 * 既知の制限事項:
 * - null / 空文字は null を返す。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.parser;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;

/**
 * RFC3339 日付(yyyy-MM-dd)を LocalDate に変換するパーサー。
 *
 * @since 1.0.0
 */
public final class Rfc3339LocalDateParser {

    private static final DateTimeFormatter ISO_DATE = DateTimeFormatter.ISO_LOCAL_DATE;

    private Rfc3339LocalDateParser() {
    }

    /**
     * RFC3339 日付文字列 (yyyy-MM-dd) を {@link LocalDate} に変換する。
     *
     * @param rawValue 日付文字列
     * @return 変換結果。{@code null} または空文字列の場合は {@code null}
     * @throws IllegalArgumentException yyyy-MM-dd 形式に適合しない場合
     */
    public static LocalDate parse(String rawValue) {
        if (rawValue == null) {
            return null;
        }
        String normalized = rawValue.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        try {
            return LocalDate.parse(normalized, ISO_DATE);
        } catch (DateTimeParseException ex) {
            throw new IllegalArgumentException("date must follow yyyy-MM-dd", ex);
        }
    }
}
