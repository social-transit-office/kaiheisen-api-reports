Rfc3339LocalDateTimeParser.java

/**
 * @file Rfc3339LocalDateTimeParser.java
 * 概要: RFC3339 日時を UTC 正規化して LocalDateTime に変換する。
 * 主な仕様:
 * - OffsetDateTime → UTC 正規化 → LocalDateTime に変換する。
 * - OffsetDateTime パースが失敗した場合は LocalDateTime.parse にフォールバックする。
 * 既知の制限事項:
 * - null / 空文字は null を返す。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.parser;

import java.time.LocalDateTime;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.time.format.DateTimeParseException;

/**
 * RFC3339 日時を UTC 正規化して LocalDateTime に変換するパーサー。
 *
 * @since 1.0.0
 */
public final class Rfc3339LocalDateTimeParser {

    private Rfc3339LocalDateTimeParser() {
    }

    /**
     * RFC3339 日時文字列を UTC 正規化した {@link LocalDateTime} に変換する。
     *
     * @param rawValue 日時文字列
     * @return 変換結果。{@code null} または空文字列の場合は {@code null}
     * @throws IllegalArgumentException RFC3339 形式に適合しない場合
     */
    public static LocalDateTime parse(String rawValue) {
        if (rawValue == null) {
            return null;
        }
        String normalized = rawValue.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        try {
            return OffsetDateTime.parse(normalized)
                    .withOffsetSameInstant(ZoneOffset.UTC)
                    .toLocalDateTime();
        } catch (DateTimeParseException ex) {
            try {
                return LocalDateTime.parse(normalized);
            } catch (DateTimeParseException nested) {
                throw new IllegalArgumentException("dateTime must follow RFC3339", ex);
            }
        }
    }
}
