/**
 * @file KaiheisenException.java
 * 概要: 開閉栓共通例外の基底クラスを定義する。
 * 主な仕様:
 * - エラーコードと埋込み文字列リストを保持する。
 * - 例外メッセージに関連パラメータを含める。
 * 既知の制限事項:
 * - 詳細メッセージは呼び出し元が設定する前提とする。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception;

import java.util.Arrays;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 開閉栓共通例外の基底クラス。
 * 型: RuntimeException
 * 主な仕様:
 * - エラーコード群と埋込み文字列リストを保持する。
 * - 例外メッセージに詳細情報を連結する。
 * 既知の制限事項:
 * - 関数名や引数は詳細メッセージに含める前提とする。
 * @since 1.0.0
 */
public class KaiheisenException extends RuntimeException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_990000001;

    private static final int STANDARD_ERROR_CODE_LENGTH = 9;

    /**
     * 大分類エラーコード。
     * @type String
     */
    private final String errorCode1;

    /**
     * 中分類エラーコード。
     * @type String
     */
    private final String errorCode2;

    /**
     * 小分類エラーコード。
     * @type String
     */
    private final String errorCode3;

    /**
     * エラーコード。
     * @type String
     */
    private final String errorCode;

    /**
     * エラーメッセージ埋込み文字列リスト。
     * @type List<String>
     */
    private final List<String> errorMessageEmbeddedStringList;

    /**
     * コンストラクタ。
     */
    public KaiheisenException() {
        this(ERROR_CODE, List.of(), null, null);
    }

    /**
     * コンストラクタ。
     * @param cause 原因例外。型: Throwable
     */
    public KaiheisenException(Throwable cause) {
        this(ERROR_CODE, List.of(), null, cause);
    }

    /**
     * コンストラクタ。
     * @param detailMessage 詳細メッセージ。型: String
     */
    public KaiheisenException(String detailMessage) {
        this(ERROR_CODE, List.of(), detailMessage, null);
    }

    /**
     * コンストラクタ。
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    public KaiheisenException(String detailMessage, Throwable cause) {
        this(ERROR_CODE, List.of(), detailMessage, cause);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected KaiheisenException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String detailMessage,
            Throwable cause) {
        this(
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList,
                detailMessage,
                cause);
    }

    /**
     * コンストラクタ。
     * @param errorCode1 大分類エラーコード。型: String
     * @param errorCode2 中分類エラーコード。型: String
     * @param errorCode3 小分類エラーコード。型: String
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected KaiheisenException(
            String errorCode1,
            String errorCode2,
            String errorCode3,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String detailMessage,
            Throwable cause) {
        this(buildMessage("KaiheisenException", detailMessage, null,
                errorCode1, errorCode2, errorCode3, errorCode, errorMessageEmbeddedStringList),
                cause, errorCode1, errorCode2, errorCode3, errorCode, errorMessageEmbeddedStringList);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode1 大分類エラーコード。型: String
     * @param errorCode2 中分類エラーコード。型: String
     * @param errorCode3 小分類エラーコード。型: String
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     */
    protected KaiheisenException(
            String message,
            Throwable cause,
            String errorCode1,
            String errorCode2,
            String errorCode3,
            String errorCode,
            List<String> errorMessageEmbeddedStringList) {
        super(message, cause);
        this.errorCode1 = errorCode1;
        this.errorCode2 = errorCode2;
        this.errorCode3 = errorCode3;
        this.errorCode = normalizeErrorCode(errorCode);
        this.errorMessageEmbeddedStringList = errorMessageEmbeddedStringList == null
                ? List.of()
                : List.copyOf(errorMessageEmbeddedStringList);
    }

    /**
     * 大分類エラーコードを取得する。
     * @return 大分類エラーコード。型: String
     */
    public String getErrorCode1() {
        return errorCode1;
    }

    /**
     * 中分類エラーコードを取得する。
     * @return 中分類エラーコード。型: String
     */
    public String getErrorCode2() {
        return errorCode2;
    }

    /**
     * 小分類エラーコードを取得する。
     * @return 小分類エラーコード。型: String
     */
    public String getErrorCode3() {
        return errorCode3;
    }

    /**
     * エラーコードを取得する。
     * @return エラーコード。型: String
     */
    public String getErrorCode() {
        return errorCode;
    }

    /**
     * エラーメッセージ埋込み文字列リストを取得する。
     * @return エラーメッセージ埋込み文字列リスト。型: List<String>
     */
    public List<String> getErrorMessageEmbeddedStringList() {
        return errorMessageEmbeddedStringList;
    }

    /**
     * 例外メッセージを組み立てる。
     * @param exceptionName 例外クラス名。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param additionalInfo 追加情報。型: String
     * @param errorCode1 大分類エラーコード。型: String
     * @param errorCode2 中分類エラーコード。型: String
     * @param errorCode3 小分類エラーコード。型: String
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @return 組み立てた例外メッセージ。型: String
     */
    protected static String buildMessage(
            String exceptionName,
            String detailMessage,
            String additionalInfo,
            String errorCode1,
            String errorCode2,
            String errorCode3,
            String errorCode,
            List<String> errorMessageEmbeddedStringList) {
        String mergedDetail = mergeDetailMessage(detailMessage, additionalInfo);
        String baseMessage = exceptionName
                + ": errorCode=" + errorCode
                + ", errorCode1=" + errorCode1
                + ", errorCode2=" + errorCode2
                + ", errorCode3=" + errorCode3
                + ", errorMessageEmbeddedStringList=" + formatList(errorMessageEmbeddedStringList);
        if (mergedDetail == null) {
            return baseMessage;
        }
        return baseMessage + ", detail=" + mergedDetail;
    }

    /**
     * 詳細メッセージを結合する。
     * @param detailMessage 詳細メッセージ。型: String
     * @param additionalInfo 追加情報。型: String
     * @return 結合後の詳細メッセージ。型: String
     */
    protected static String mergeDetailMessage(String detailMessage, String additionalInfo) {
        boolean hasDetail = detailMessage != null && !detailMessage.isBlank();
        boolean hasAdditional = additionalInfo != null && !additionalInfo.isBlank();
        if (!hasDetail && !hasAdditional) {
            return null;
        }
        if (!hasDetail) {
            return additionalInfo;
        }
        if (!hasAdditional) {
            return detailMessage;
        }
        return detailMessage + ", " + additionalInfo;
    }

    /**
     * リストを文字列化する。
     * @param list 対象リスト。型: List<?>
     * @return 文字列表現。型: String
     */
    protected static String formatList(List<?> list) {
        if (list == null) {
            return "null";
        }
        return list.toString();
    }

    /**
     * 配列を文字列化する。
     * @param array 対象配列。型: Object[]
     * @return 文字列表現。型: String
     */
    protected static String formatArray(Object[] array) {
        if (array == null) {
            return "null";
        }
        return Arrays.toString(array);
    }

    /**
     * エラーコードからエラー理由コードを解決する。
     * @param errorCode エラーコード。型: String
     * @return エラー理由コード。型: String
     */
    protected static String resolveReasonCodeFromErrorCode(String errorCode) {
        String trimmed = normalizeErrorCode(errorCode);
        if (trimmed == null) {
            return null;
        }
        if (trimmed.length() < 2) {
            return null;
        }
        String suffix = trimmed.substring(trimmed.length() - 2);
        for (int index = 0; index < suffix.length(); index += 1) {
            if (!Character.isDigit(suffix.charAt(index))) {
                return null;
            }
        }
        return suffix;
    }

    /**
     * エラーコードから大分類コードを解決する。
     * @param errorCode エラーコード。型: String
     * @return 大分類コード。型: String
     */
    protected static String resolveErrorCode1FromErrorCode(String errorCode) {
        String normalized = normalizeErrorCode(errorCode);
        if (normalized == null || normalized.length() < 2) {
            return null;
        }
        return normalized.substring(0, 2);
    }

    /**
     * エラーコードから中分類コードを解決する。
     * @param errorCode エラーコード。型: String
     * @return 中分類コード。型: String
     */
    protected static String resolveErrorCode2FromErrorCode(String errorCode) {
        String normalized = normalizeErrorCode(errorCode);
        if (normalized == null || normalized.length() < 4) {
            return null;
        }
        return normalized.substring(0, 4);
    }

    /**
     * エラーコードから小分類コードを解決する。
     * @param errorCode エラーコード。型: String
     * @return 小分類コード。型: String
     */
    protected static String resolveErrorCode3FromErrorCode(String errorCode) {
        String normalized = normalizeErrorCode(errorCode);
        if (normalized == null) {
            return null;
        }
        if (normalized.length() >= 5) {
            return normalized.substring(normalized.length() - 5);
        }
        return normalized;
    }

    private static String normalizeErrorCode(String errorCode) {
        if (errorCode == null) {
            return null;
        }
        String trimmed = errorCode.trim();
        if (trimmed.isEmpty()) {
            return null;
        }
        if (trimmed.length() < STANDARD_ERROR_CODE_LENGTH) {
            return trimmed;
        }
        return trimmed;
    }
}
