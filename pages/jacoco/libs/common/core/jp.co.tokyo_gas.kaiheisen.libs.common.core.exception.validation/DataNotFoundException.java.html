/**
 * @file DataNotFoundException.java
 * 概要: 処理対象データ無しエラーを表す例外を定義する。
 * 主な仕様:
 * - テーブル名を保持する。
 * 既知の制限事項:
 * - テーブル名の表記は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCatalog;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 処理対象データ無しエラーを表す例外クラス。
 * 型: ValidationException
 * 主な仕様:
 * - テーブル名を保持する。
 * 既知の制限事項:
 * - エラー詳細の粒度は呼び出し元が決定する。
 * @since 1.0.0
 */
public class DataNotFoundException extends ValidationException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_010100001;

    /**
     * テーブル名。
     * @type String
     */
    private final String tableName;

    /**
     * CIRIUSメッセージID。
     * @type String
     */
    private final String ciriusMessageId;

    /**
     * コンストラクタ。
     * @param tableName テーブル名。型: String
     */
    public DataNotFoundException(String tableName) {
        this(List.of(normalizeEmbeddedString(tableName)), List.of(), tableName, resolveCiriusMessageId(), null, null);
    }

    private static String normalizeEmbeddedString(String embeddedString) {
        return embeddedString == null ? "" : embeddedString;
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param errorDetailList エラー詳細リスト。型: List<String>
     * @param tableName テーブル名。型: String
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    private DataNotFoundException(
            List<String> errorMessageEmbeddedStringList,
            List<String> errorDetailList,
            String tableName,
            String ciriusMessageId,
            String detailMessage,
            Throwable cause) {
        super(buildMessage("DataNotFoundException", detailMessage,
                "errorDetailList=" + formatList(errorDetailList)
                        + ", tableName=" + tableName
                        + ", ciriusMessageId=" + ciriusMessageId,
            resolveErrorCode1FromErrorCode(ERROR_CODE),
            resolveErrorCode2FromErrorCode(ERROR_CODE),
            resolveErrorCode3FromErrorCode(ERROR_CODE),
            ERROR_CODE,
                errorMessageEmbeddedStringList),
                cause,
            ERROR_CODE,
                errorMessageEmbeddedStringList,
                errorDetailList);
        this.tableName = tableName;
        this.ciriusMessageId = ciriusMessageId;
    }

    /**
     * テーブル名を取得する。
     * @return テーブル名。型: String
     */
    public String getTableName() {
        return tableName;
    }

    /**
     * CIRIUSメッセージIDを取得する。
     * @return CIRIUSメッセージID。型: String
     */
    public String getCiriusMessageId() {
        return ciriusMessageId;
    }

    private static String resolveCiriusMessageId() {
        return ErrorCatalog.resolveCiriusMessageId(ERROR_CODE);
    }
}
