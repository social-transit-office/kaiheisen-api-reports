/**
 * @file ValidationException.java
 * 概要: バリデーションエラーを表す例外を定義する。
 * 主な仕様:
 * - エラー詳細リストを保持する。
 * 既知の制限事項:
 * - 詳細メッセージは呼び出し元が補足する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation;

import java.util.ArrayList;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCatalog;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorDefinition;

/**
 * バリデーションエラーを表す例外クラス。
 * 型: KaiheisenException
 * 主な仕様:
 * - エラー詳細リストを保持する。
 * 既知の制限事項:
 * - エラー詳細のフォーマットは呼び出し元に依存する。
 * @since 1.0.0
 */
public class ValidationException extends KaiheisenException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_010000001;

    /**
     * エラー詳細リスト。
     * @type List<String>
     */
    private final List<String> errorDetailList;

    /**
     * コンストラクタ。
     * @param errorDetailList エラー詳細リスト。型: List<String>
     */
    public ValidationException(List<String> errorDetailList) {
        this(
                ERROR_CODE,
                List.of(),
                errorDetailList,
                null,
                null);
    }

    /**
     * ValidationError リストからエラー詳細リストを自動解決するコンストラクタ。
     *
     * <p>各 {@link ValidationError} の errorCode と embeddedValues を使い、
     * {@link ErrorCatalog#resolveErrorMessage} でテンプレート埋込済みメッセージを構築し、
     * errorDetailList として保持する。
     *
     * @param validationErrors バリデーションエラーリスト。型: List<ValidationError>
     */
    public ValidationException(List<ValidationError> validationErrors, boolean fromValidationErrors) {
        this(
                ERROR_CODE,
                List.of(),
                resolveErrorDetailList(validationErrors),
                null,
                null);
    }

    /**
     * ValidationError リストから ErrorCatalog テンプレート埋込済みメッセージリストを解決する。
     *
     * @param validationErrors バリデーションエラーリスト
     * @return 解決済みメッセージリスト
     */
    private static List<String> resolveErrorDetailList(List<ValidationError> validationErrors) {
        if (validationErrors == null || validationErrors.isEmpty()) {
            return List.of();
        }
        List<String> resolved = new ArrayList<>();
        for (ValidationError error : validationErrors) {
            ErrorDefinition definition = ErrorCatalog.find(error.errorCode());
            if (definition == null) {
                continue;
            }
            String message = ErrorCatalog.applyPlaceholders(
                    definition.messageTemplate(), error.embeddedValues());
            if (message != null && !message.isBlank()) {
                resolved.add(message);
            }
        }
        return List.copyOf(resolved);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param errorDetailList エラー詳細リスト。型: List<String>
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected ValidationException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            List<String> errorDetailList,
            String detailMessage,
            Throwable cause) {
        this(buildMessage("ValidationException", detailMessage,
                "errorDetailList=" + formatList(errorDetailList),
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList),
                cause, errorCode, errorMessageEmbeddedStringList, errorDetailList);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param errorDetailList エラー詳細リスト。型: List<String>
     */
    protected ValidationException(
            String message,
            Throwable cause,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            List<String> errorDetailList) {
        super(
                message,
                cause,
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList);
        this.errorDetailList = errorDetailList == null ? List.of() : List.copyOf(errorDetailList);
    }

    /**
     * エラー詳細リストを取得する。
     * @return エラー詳細リスト。型: List<String>
     */
    public List<String> getErrorDetailList() {
        return errorDetailList;
    }
}
