/**
 * @file ValidationError.java
 * 概要: バリデーションエラー1件分のエラーコードと埋込値を保持する。
 * 主な仕様:
 * - ErrorCatalog のエラーコードとテンプレート埋込値を不変で保持する。
 * - ValidationException でメッセージ解決に使用される。
 * 既知の制限事項:
 * - embeddedValues のインデックスは ErrorCatalog テンプレートに準拠する。
 *   {1} = フィールド名、{2} 以降 = 追加パラメータ。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation;

import java.util.List;
import java.util.Objects;

/**
 * バリデーションエラー1件分のエラーコードと埋込値を保持する record。
 *
 * <p>ErrorCatalog テンプレートのプレースホルダ規約:
 * <ul>
 *   <li>{@code {0}} — 未使用（空文字）</li>
 *   <li>{@code {1}} — フィールド名</li>
 *   <li>{@code {2}} 以降 — 制約固有の追加パラメータ</li>
 * </ul>
 *
 * @param errorCode エラーコード（例: {@code "010000011"}）
 * @param embeddedValues テンプレート埋込値リスト（{0}=空, {1}=フィールド名, ...）
 * @since 1.0.0
 */
public record ValidationError(String errorCode, List<String> embeddedValues) {

    /**
     * コンパクトコンストラクタ。
     */
    public ValidationError {
        Objects.requireNonNull(errorCode, "errorCode");
        embeddedValues = embeddedValues == null ? List.of() : List.copyOf(embeddedValues);
    }

    /**
     * フィールド名のみを埋込値として持つ ValidationError を生成する。
     *
     * @param errorCode エラーコード
     * @param fieldName フィールド名
     * @return ValidationError
     */
    public static ValidationError ofField(String errorCode, String fieldName) {
        return new ValidationError(errorCode, List.of("", fieldName));
    }

    /**
     * フィールド名と追加パラメータを埋込値として持つ ValidationError を生成する。
     *
     * @param errorCode エラーコード
     * @param fieldName フィールド名
     * @param additionalValues 追加パラメータ
     * @return ValidationError
     */
    public static ValidationError of(String errorCode, String fieldName, String... additionalValues) {
        var values = new java.util.ArrayList<String>();
        values.add(""); // {0} は未使用
        values.add(fieldName); // {1} = フィールド名
        for (String additional : additionalValues) {
            values.add(additional);
        }
        return new ValidationError(errorCode, values);
    }
}
