package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDefinitionNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.KaiheisenConstantItem;

/**
 * 開閉栓API独自コンスタント内部定義実装。
 */
public final class InternalKaiheisenConstantService implements KaiheisenConstantService {

    private static final Comparator<InternalConstantDataLoader.KaiheisenDefinition> KAIHEISEN_ORDER =
            Comparator.comparing(InternalConstantDataLoader.KaiheisenDefinition::value);

    private final Map<String, List<InternalConstantDataLoader.KaiheisenDefinition>> definitionsByType;

    /**
     * リソースパスとクラスローダを指定して内部開閉栓コンスタントサービスを生成する。
     *
     * @param resourcePath TSV リソースパス
     * @param classLoader クラスローダ
     */
    public InternalKaiheisenConstantService(String resourcePath, ClassLoader classLoader) {
        List<InternalConstantDataLoader.KaiheisenDefinition> definitions =
                InternalConstantDataLoader.loadKaiheisenDefinitions(resourcePath, classLoader);

        Map<String, List<InternalConstantDataLoader.KaiheisenDefinition>> mutable = new HashMap<>();
        for (InternalConstantDataLoader.KaiheisenDefinition definition : definitions) {
            mutable.computeIfAbsent(definition.type(), unused -> new ArrayList<>()).add(definition);
        }

        Map<String, List<InternalConstantDataLoader.KaiheisenDefinition>> immutable = new HashMap<>();
        for (Map.Entry<String, List<InternalConstantDataLoader.KaiheisenDefinition>> entry : mutable.entrySet()) {
            List<InternalConstantDataLoader.KaiheisenDefinition> sorted = new ArrayList<>(entry.getValue());
            sorted.sort(KAIHEISEN_ORDER);
            immutable.put(entry.getKey(), List.copyOf(sorted));
        }
        this.definitionsByType = Map.copyOf(immutable);
    }

    @Override
    public List<KaiheisenConstantItem> listActiveByType(String type) {
        String normalizedType = ConstantKeyNormalizer.normalizeRequiredTrimmed(type, "type");

        return definitionsByType.getOrDefault(normalizedType, List.of()).stream()
                .map(definition -> new KaiheisenConstantItem(definition.value(), definition.name()))
                .toList();
    }

    @Override
    public String findNameByTypeAndValue(String type, String value) {
        String normalizedType = ConstantKeyNormalizer.normalizeRequiredTrimmed(type, "type");
        Objects.requireNonNull(value, "value");
        String normalizedValue = value.trim();

        return definitionsByType.getOrDefault(normalizedType, List.of()).stream()
                .filter(definition -> definition.value().equals(normalizedValue))
                .map(InternalConstantDataLoader.KaiheisenDefinition::name)
                .findFirst()
                .orElseThrow(() -> new ConstantDefinitionNotFoundException(
                        "KaiheisenConstantService",
                        "type=" + normalizedType + ", value=" + normalizedValue));
    }
}
