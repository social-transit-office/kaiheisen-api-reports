package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal;

import java.util.Objects;

final class ConstantKeyNormalizer {

    private ConstantKeyNormalizer() {
    }

    static String normalizeRequiredTrimmed(String value, String fieldName) {
        Objects.requireNonNull(value, fieldName);
        String trimmed = value.trim();
        if (trimmed.isEmpty()) {
            throw new IllegalArgumentException(fieldName + " must not be blank");
        }
        return trimmed;
    }

    static String normalizeOptionalTrimmed(String value) {
        if (value == null) {
            return "";
        }
        return value.trim();
    }
}
