package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal;

import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDataFormatException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDataLoadException;

/**
 * ダブルクォートを考慮した TSV パーサ。
 */
public final class TabSeparatedValuesParser {

    private TabSeparatedValuesParser() {
    }

    /**
     * 入力ストリームから TSV を読み込みパースする。
     *
     * @param inputStream 入力ストリーム
     * @param sourceName ソース名（エラーメッセージ用）
     * @return パース済みレコード一覧
     */
    public static List<List<String>> parse(InputStream inputStream, String sourceName) {
        try {
            String content = new String(inputStream.readAllBytes(), StandardCharsets.UTF_8);
            return parse(content, sourceName);
        } catch (IOException ex) {
            throw new ConstantDataLoadException("Failed to read TSV: source=" + sourceName, ex);
        }
    }

    /**
     * 文字列コンテンツから TSV をパースする。
     *
     * @param content TSV 文字列
     * @param sourceName ソース名（エラーメッセージ用）
     * @return パース済みレコード一覧
     */
    public static List<List<String>> parse(String content, String sourceName) {
        List<List<String>> records = new ArrayList<>();
        List<String> row = new ArrayList<>();
        StringBuilder field = new StringBuilder();
        boolean inQuotes = false;

        for (int index = 0; index < content.length(); index++) {
            char ch = content.charAt(index);

            if (ch == '"') {
                if (inQuotes && index + 1 < content.length() && content.charAt(index + 1) == '"') {
                    field.append('"');
                    index++;
                } else {
                    inQuotes = !inQuotes;
                }
                continue;
            }

            if (ch == '\t' && !inQuotes) {
                row.add(field.toString());
                field.setLength(0);
                continue;
            }

            if ((ch == '\n' || ch == '\r') && !inQuotes) {
                if (ch == '\r' && index + 1 < content.length() && content.charAt(index + 1) == '\n') {
                    index++;
                }

                row.add(field.toString());
                field.setLength(0);

                if (!isBlankRow(row)) {
                    records.add(List.copyOf(row));
                }
                row.clear();
                continue;
            }

            field.append(ch);
        }

        if (inQuotes) {
            throw new ConstantDataFormatException("Unclosed quote in TSV: source=" + sourceName);
        }

        row.add(field.toString());
        if (!isBlankRow(row)) {
            records.add(List.copyOf(row));
        }

        if (!records.isEmpty()) {
            List<String> firstRow = new ArrayList<>(records.getFirst());
            if (!firstRow.isEmpty() && firstRow.getFirst().startsWith("\uFEFF")) {
                firstRow.set(0, firstRow.getFirst().substring(1));
                records.set(0, List.copyOf(firstRow));
            }
        }

        return records;
    }

    private static boolean isBlankRow(List<String> row) {
        return row.stream().allMatch(value -> value == null || value.isBlank());
    }
}
