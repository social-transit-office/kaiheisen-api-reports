package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal;

import java.io.IOException;
import java.io.InputStream;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDataDuplicateException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDataFormatException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDataLoadException;

/**
 * TSV からコンスタント定義を読み込むローダ。
 */
final class InternalConstantDataLoader {

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private InternalConstantDataLoader() {
    }

    static List<CiriusDefinition> loadCiriusDefinitions(String resourcePath, ClassLoader classLoader) {
        List<List<String>> rows = loadRows(resourcePath, classLoader);
        if (rows.size() < 2) {
            throw new ConstantDataFormatException("CIRIUS constants TSV has no data: resource=" + resourcePath);
        }

        List<String> header = rows.getFirst();
        validateHeader(header, 8, resourcePath);

        List<CiriusDefinition> result = new ArrayList<>();
        Set<String> duplicateChecker = new HashSet<>();

        for (int i = 1; i < rows.size(); i++) {
            List<String> row = rows.get(i);
            int lineNo = i + 1;

            String typeCode = requiredField(row, 1, "コンスタント種別コード", resourcePath, lineNo);
            String code = requiredField(row, 3, "コンスタントコード", resourcePath, lineNo);
            // 実データではコンスタント名が空欄の行も存在するため、名称は任意として扱う。
            String name = optionalField(row, 4);
            String shortName = optionalField(row, 5);

            int displayOrder = parseIntField(
                    requiredField(row, 6, "表示順番号", resourcePath, lineNo),
                    "表示順番号",
                    resourcePath,
                    lineNo);
            boolean registrationAvailable = parseBooleanFlag(
                    requiredField(row, 7, "登録時利用可否フラグ", resourcePath, lineNo),
                    "登録時利用可否フラグ",
                    resourcePath,
                    lineNo);

            LocalDate startDate = parseOptionalDate(optionalField(row, 9), "適用開始年月日", resourcePath, lineNo);
            LocalDate endDate = parseOptionalDate(optionalField(row, 10), "適用終了年月日", resourcePath, lineNo);

            if (startDate != null && endDate != null && startDate.isAfter(endDate)) {
                throw new ConstantDataFormatException(
                        "Date range is invalid: resource=" + resourcePath
                                + ", line=" + lineNo
                                + ", 適用開始年月日=" + startDate
                                + ", 適用終了年月日=" + endDate);
            }

            String duplicateKey = typeCode + "\u0000" + code;
            if (!duplicateChecker.add(duplicateKey)) {
                throw new ConstantDataDuplicateException(
                        "Duplicate CIRIUS constant key: resource=" + resourcePath
                                + ", line=" + lineNo
                                + ", typeCode=" + typeCode
                                + ", code=" + code);
            }

            result.add(new CiriusDefinition(typeCode, code, name, shortName, displayOrder, registrationAvailable, startDate, endDate));
        }

        return List.copyOf(result);
    }

    static List<KaiheisenDefinition> loadKaiheisenDefinitions(String resourcePath, ClassLoader classLoader) {
        List<List<String>> rows = loadRows(resourcePath, classLoader);
        if (rows.size() < 2) {
            throw new ConstantDataFormatException("Kaiheisen constants TSV has no data: resource=" + resourcePath);
        }

        List<String> header = rows.getFirst();
        validateHeader(header, 4, resourcePath);

        List<KaiheisenDefinition> result = new ArrayList<>();
        Set<String> duplicateChecker = new HashSet<>();

        for (int i = 1; i < rows.size(); i++) {
            List<String> row = rows.get(i);
            int lineNo = i + 1;

            String type = requiredField(row, 0, "コンスタント種別", resourcePath, lineNo);
            String name = requiredField(row, 3, "コンスタント名", resourcePath, lineNo);
            String value = optionalField(row, 2);

            if (value.isEmpty() && !isAllowedBlankKaiheisenValue(type, name)) {
                throw new ConstantDataFormatException(
                        "Required field is blank: resource=" + resourcePath
                                + ", line=" + lineNo
                                + ", field=コンスタント値");
            }

            String duplicateKey = type + "\u0000" + value;
            if (!duplicateChecker.add(duplicateKey)) {
                throw new ConstantDataDuplicateException(
                        "Duplicate Kaiheisen constant key: resource=" + resourcePath
                                + ", line=" + lineNo
                                + ", type=" + type
                                + ", value=" + value);
            }

            result.add(new KaiheisenDefinition(type, value, name));
        }

        return List.copyOf(result);
    }

    private static List<List<String>> loadRows(String resourcePath, ClassLoader classLoader) {
        Objects.requireNonNull(classLoader, "classLoader");
        String normalizedResourcePath = ConstantKeyNormalizer.normalizeRequiredTrimmed(resourcePath, "resourcePath");

        try (InputStream inputStream = classLoader.getResourceAsStream(normalizedResourcePath)) {
            if (inputStream == null) {
                throw new ConstantDataLoadException("Constant resource not found: path=" + normalizedResourcePath);
            }
            return TabSeparatedValuesParser.parse(inputStream, normalizedResourcePath);
        } catch (ConstantDataLoadException ex) {
            throw ex;
        } catch (IOException ex) {
            throw new ConstantDataLoadException(
                    "Failed to load constant resource: path=" + normalizedResourcePath,
                    ex);
        }
    }

    private static void validateHeader(List<String> header, int requiredSize, String resourcePath) {
        if (header.size() < requiredSize) {
            throw new ConstantDataFormatException(
                    "TSV header is invalid: resource=" + resourcePath + ", headerColumns=" + header.size());
        }
    }

    private static boolean isAllowedBlankKaiheisenValue(
            String type,
            String name) {
        return "KH005".equals(type)
                && "ＴＧ小売".equals(name);
    }

    private static String requiredField(
            List<String> row,
            int index,
            String fieldName,
            String resourcePath,
            int lineNo) {
        String value = optionalField(row, index);
        if (value.isEmpty()) {
            throw new ConstantDataFormatException(
                    "Required field is blank: resource=" + resourcePath
                            + ", line=" + lineNo
                            + ", field=" + fieldName);
        }
        return value;
    }

    private static String optionalField(List<String> row, int index) {
        if (index >= row.size()) {
            return "";
        }
        return ConstantKeyNormalizer.normalizeOptionalTrimmed(row.get(index));
    }

    private static int parseIntField(String value, String fieldName, String resourcePath, int lineNo) {
        try {
            return Integer.parseInt(value);
        } catch (NumberFormatException ex) {
            throw new ConstantDataFormatException(
                    "Numeric field is invalid: resource=" + resourcePath
                            + ", line=" + lineNo
                            + ", field=" + fieldName
                            + ", value=" + value,
                    ex);
        }
    }

    private static boolean parseBooleanFlag(String value, String fieldName, String resourcePath, int lineNo) {
        return switch (value) {
            case "1" -> true;
            case "0" -> false;
            default -> throw new ConstantDataFormatException(
                    "Flag field is invalid: resource=" + resourcePath
                            + ", line=" + lineNo
                            + ", field=" + fieldName
                            + ", value=" + value);
        };
    }

    private static LocalDate parseOptionalDate(String value, String fieldName, String resourcePath, int lineNo) {
        if (value.isEmpty()) {
            return null;
        }
        try {
            return LocalDate.parse(value, BASIC_DATE);
        } catch (DateTimeParseException ex) {
            throw new ConstantDataFormatException(
                    "Date field is invalid: resource=" + resourcePath
                            + ", line=" + lineNo
                            + ", field=" + fieldName
                            + ", value=" + value,
                    ex);
        }
    }

    record CiriusDefinition(
            String typeCode,
            String code,
            String name,
            String shortName,
            int displayOrder,
            boolean registrationAvailable,
            LocalDate startDate,
            LocalDate endDate) {

        boolean isActiveOn(LocalDate baseDate) {
            boolean started = startDate == null || !baseDate.isBefore(startDate);
            boolean notEnded = endDate == null || !baseDate.isAfter(endDate);
            return started && notEnded;
        }
    }

    record KaiheisenDefinition(String type, String value, String name) {
    }
}
