package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal;

import java.time.Clock;
import java.time.Instant;
import java.time.LocalDate;
import java.time.ZoneId;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.ConstantDefinitionNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.CiriusConstantDetail;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.CiriusConstantListItem;

/**
 * CIRIUS コンスタント内部定義実装。
 */
public final class InternalCiriusConstantService implements CiriusConstantService {

    private static final ZoneId JST = ZoneId.of("Asia/Tokyo");

    private static final Comparator<InternalConstantDataLoader.CiriusDefinition> CIRIUS_ORDER =
            Comparator.comparingInt(InternalConstantDataLoader.CiriusDefinition::displayOrder)
                    .thenComparing(InternalConstantDataLoader.CiriusDefinition::code);

    private final Map<String, List<InternalConstantDataLoader.CiriusDefinition>> definitionsByTypeCode;

    private final Clock clock;

    /**
     * リソースパスとクラスローダを指定して内部 CIRIUS コンスタントサービスを生成する。
     *
     * @param resourcePath TSV リソースパス
     * @param classLoader クラスローダ
     * @param clock 有効期間判定用クロック
     */
    public InternalCiriusConstantService(String resourcePath, ClassLoader classLoader, Clock clock) {
        this.clock = Objects.requireNonNull(clock, "clock");

        List<InternalConstantDataLoader.CiriusDefinition> definitions =
                InternalConstantDataLoader.loadCiriusDefinitions(resourcePath, classLoader);

        Map<String, List<InternalConstantDataLoader.CiriusDefinition>> mutable = new HashMap<>();
        for (InternalConstantDataLoader.CiriusDefinition definition : definitions) {
            mutable.computeIfAbsent(definition.typeCode(), unused -> new ArrayList<>()).add(definition);
        }

        Map<String, List<InternalConstantDataLoader.CiriusDefinition>> immutable = new HashMap<>();
        for (Map.Entry<String, List<InternalConstantDataLoader.CiriusDefinition>> entry : mutable.entrySet()) {
            List<InternalConstantDataLoader.CiriusDefinition> sorted = new ArrayList<>(entry.getValue());
            sorted.sort(CIRIUS_ORDER);
            immutable.put(entry.getKey(), List.copyOf(sorted));
        }
        this.definitionsByTypeCode = Map.copyOf(immutable);
    }

    @Override
    public List<CiriusConstantListItem> listActiveByTypeCode(String typeCode) {
        String normalizedTypeCode = ConstantKeyNormalizer.normalizeRequiredTrimmed(typeCode, "typeCode");
        LocalDate baseDate = resolveBaseDate();

        return definitionsByTypeCode.getOrDefault(normalizedTypeCode, List.of()).stream()
                .filter(definition -> definition.isActiveOn(baseDate))
                .map(definition -> new CiriusConstantListItem(
                        definition.code(),
                        definition.name(),
                        definition.shortName(),
                        definition.displayOrder(),
                        definition.registrationAvailable()))
                .toList();
    }

    @Override
    public CiriusConstantDetail findActiveByTypeCodeAndCode(String typeCode, String code) {
        String normalizedTypeCode = ConstantKeyNormalizer.normalizeRequiredTrimmed(typeCode, "typeCode");
        String normalizedCode = ConstantKeyNormalizer.normalizeRequiredTrimmed(code, "code");
        LocalDate baseDate = resolveBaseDate();

        return definitionsByTypeCode.getOrDefault(normalizedTypeCode, List.of()).stream()
                .filter(definition -> definition.code().equals(normalizedCode))
                .filter(definition -> definition.isActiveOn(baseDate))
                .findFirst()
                .map(definition -> new CiriusConstantDetail(
                        definition.name(),
                        definition.shortName(),
                        definition.displayOrder(),
                        definition.registrationAvailable()))
                .orElseThrow(() -> new ConstantDefinitionNotFoundException(
                        "CiriusConstantService",
                        "typeCode=" + normalizedTypeCode + ", code=" + normalizedCode));
    }

    private LocalDate resolveBaseDate() {
        return Instant.now(clock).atZone(JST).toLocalDate();
    }
}
