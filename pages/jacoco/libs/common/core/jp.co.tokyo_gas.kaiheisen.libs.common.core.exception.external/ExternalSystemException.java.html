/**
 * @file ExternalSystemException.java
 * 概要: 外部システムエラーを表す例外を定義する。
 * 主な仕様:
 * - 外部システム名を保持する。
 * 既知の制限事項:
 * - 外部システム名の正規化は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 外部システムエラーを表す例外クラス。
 * 型: KaiheisenException
 * 主な仕様:
 * - 外部システム名を保持する。
 * 既知の制限事項:
 * - 外部システム名の表記ゆれは許容する。
 * @since 1.0.0
 */
public class ExternalSystemException extends KaiheisenException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_040000001;

    /**
     * 外部システム名。
     * @type String
     */
    private final String externalSystemName;

    /**
     * コンストラクタ。
     * @param externalSystemName 外部システム名。型: String
     * @param cause 原因例外。型: Throwable
     */
    public ExternalSystemException(String externalSystemName, Throwable cause) {
        this(
                ERROR_CODE,
                List.of(),
                externalSystemName,
                null,
                cause);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param externalSystemName 外部システム名。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected ExternalSystemException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String externalSystemName,
            String detailMessage,
            Throwable cause) {
        this(buildMessage("ExternalSystemException", detailMessage,
            buildExternalAdditionalInfo(externalSystemName),
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList),
                cause, errorCode, errorMessageEmbeddedStringList, externalSystemName);
    }

    /**
     * コンストラクタ。
     * @param exceptionName 例外名。型: String
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param externalSystemName 外部システム名。型: String
     * @param additionalInfo 追加情報。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected ExternalSystemException(
            String exceptionName,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String externalSystemName,
            String additionalInfo,
            String detailMessage,
            Throwable cause) {
        this(buildMessage(
                exceptionName,
                detailMessage,
                mergeAdditionalInfo(buildExternalAdditionalInfo(externalSystemName), additionalInfo),
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList),
            cause,
            errorCode,
            errorMessageEmbeddedStringList,
            externalSystemName);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param externalSystemName 外部システム名。型: String
     */
    protected ExternalSystemException(
            String message,
            Throwable cause,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String externalSystemName) {
        super(
                message,
                cause,
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList);
        this.externalSystemName = externalSystemName;
    }

    /**
     * 外部システム名を取得する。
     * @return 外部システム名。型: String
     */
    public String getExternalSystemName() {
        return externalSystemName;
    }

    /**
     * 外部システム例外の追加情報を組み立てる。
     * @param externalSystemName 外部システム名。型: String
     * @return 追加情報文字列。型: String
     */
    protected static String buildExternalAdditionalInfo(String externalSystemName) {
        return "externalSystemName=" + externalSystemName;
    }

    private static String mergeAdditionalInfo(String baseInfo, String additionalInfo) {
        if (additionalInfo == null || additionalInfo.isBlank()) {
            return baseInfo;
        }
        return baseInfo + ", " + additionalInfo;
    }
}
