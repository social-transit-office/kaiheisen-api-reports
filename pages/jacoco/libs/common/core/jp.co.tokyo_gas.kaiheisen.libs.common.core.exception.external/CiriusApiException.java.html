/**
 * @file CiriusApiException.java
 * 概要: CIRIUS APIエラーを表す例外を定義する。
 * 主な仕様:
 * - リクエスト情報とレスポンス情報を保持する。
 * 既知の制限事項:
 * - リクエスト/レスポンス情報の形式は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * CIRIUS APIエラーを表す例外クラス。
 * 型: ExternalSystemException
 * 主な仕様:
 * - リクエスト情報とレスポンス情報を保持する。
 * 既知の制限事項:
 * - エラーコードは固定値を利用する。
 * @since 1.0.0
 */
public class CiriusApiException extends ExternalSystemException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_040100001;

    public static final String EXTERNAL_SYSTEM_NAME = "CIRIUS API";

    /**
     * リクエスト情報。
     * @type String
     */
    private final String requestInfo;

    /**
     * レスポンス情報。
     * @type String
     */
    private final String responseInfo;

    /**
     * コンストラクタ。
     * @param requestInfo リクエスト情報。型: String
     * @param responseInfo レスポンス情報。型: String
     * @param cause 原因例外。型: Throwable
     */
    public CiriusApiException(String requestInfo, String responseInfo, Throwable cause) {
        this(List.of(), requestInfo, responseInfo, null, cause);
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param requestInfo リクエスト情報。型: String
     * @param responseInfo レスポンス情報。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    private CiriusApiException(
            List<String> errorMessageEmbeddedStringList,
            String requestInfo,
            String responseInfo,
            String detailMessage,
            Throwable cause) {
        super(
                "CiriusApiException",
                ERROR_CODE,
                errorMessageEmbeddedStringList,
                EXTERNAL_SYSTEM_NAME,
                "requestInfo=" + requestInfo + ", responseInfo=" + responseInfo,
                detailMessage,
                cause);
        this.requestInfo = requestInfo;
        this.responseInfo = responseInfo;
    }

    /**
     * リクエスト情報を取得する。
     * @return リクエスト情報。型: String
     */
    public String getRequestInfo() {
        return requestInfo;
    }

    /**
     * レスポンス情報を取得する。
     * @return レスポンス情報。型: String
     */
    public String getResponseInfo() {
        return responseInfo;
    }
}
