/**
 * @file BusinessRuleException.java
 * 概要: 業務エラーを表す例外を定義する。
 * 主な仕様:
 * - CIRIUSメッセージIDと埋込み文字列リストを保持する。
 * 既知の制限事項:
 * - CIRIUSメッセージIDの解決は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCatalog;

/**
 * 業務エラーを表す例外クラス。
 * 型: KaiheisenException
 * 主な仕様:
 * - CIRIUSメッセージ関連情報を保持する。
 * 既知の制限事項:
 * - CIRIUSメッセージ埋込み文字列の内容は呼び出し元が設定する。
 * @since 1.0.0
 */
public class BusinessRuleException extends KaiheisenException {

    /**
     * CIRIUSメッセージID。
     * @type String
     */
    private final String ciriusMessageId;

    /**
     * CIRIUSメッセージ埋込み文字列リスト。
     * @type List<String>
     */
    private final List<String> ciriusMessageEmbeddedStringList;

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     */
    public BusinessRuleException(String errorCode) {
        this(errorCode, List.of(), null, List.of(), null, null);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     */
    public BusinessRuleException(String errorCode, List<String> errorMessageEmbeddedStringList) {
        this(errorCode, errorMessageEmbeddedStringList, null, List.of(), null, null);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     */
    public BusinessRuleException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            List<String> ciriusMessageEmbeddedStringList) {
        this(errorCode, errorMessageEmbeddedStringList, null, ciriusMessageEmbeddedStringList, null, null);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected BusinessRuleException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList,
            String detailMessage,
            Throwable cause) {
        this(
            "BusinessRuleException",
            errorCode,
            errorMessageEmbeddedStringList,
            ciriusMessageId,
            ciriusMessageEmbeddedStringList,
            null,
            detailMessage,
            cause);
    }

    /**
     * コンストラクタ。
     * @param exceptionName 例外名。型: String
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @param additionalInfo 追加情報。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected BusinessRuleException(
            String exceptionName,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList,
            String additionalInfo,
            String detailMessage,
            Throwable cause) {
        this(buildMessage(
                exceptionName,
                detailMessage,
                mergeAdditionalInfo(
                    buildBusinessAdditionalInfo(
                        resolveCiriusMessageId(errorCode, ciriusMessageId),
                        ciriusMessageEmbeddedStringList),
                    additionalInfo),
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList),
            cause,
            errorCode,
            errorMessageEmbeddedStringList,
            resolveCiriusMessageId(errorCode, ciriusMessageId),
            ciriusMessageEmbeddedStringList);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     */
    protected BusinessRuleException(
            String message,
            Throwable cause,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList) {
        super(
                message,
                cause,
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList);
        this.ciriusMessageId = ciriusMessageId;
        this.ciriusMessageEmbeddedStringList = ciriusMessageEmbeddedStringList == null
                ? List.of()
                : List.copyOf(ciriusMessageEmbeddedStringList);
    }

    /**
     * CIRIUSメッセージIDを取得する。
     * @return CIRIUSメッセージID。型: String
     */
    public String getCiriusMessageId() {
        return ciriusMessageId;
    }

    /**
     * CIRIUSメッセージ埋込み文字列リストを取得する。
     * @return CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     */
    public List<String> getCiriusMessageEmbeddedStringList() {
        return ciriusMessageEmbeddedStringList;
    }

    private static String resolveCiriusMessageId(String errorCode, String ciriusMessageId) {
        if (ciriusMessageId != null && !ciriusMessageId.isBlank()) {
            return ciriusMessageId;
        }
        return ErrorCatalog.resolveCiriusMessageId(errorCode);
    }

    /**
     * 業務例外の追加情報を組み立てる。
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @return 追加情報文字列。型: String
     */
    protected static String buildBusinessAdditionalInfo(
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList) {
        return "ciriusMessageId=" + ciriusMessageId
                + ", ciriusMessageEmbeddedStringList=" + formatList(ciriusMessageEmbeddedStringList);
    }

    private static String mergeAdditionalInfo(String baseInfo, String additionalInfo) {
        if (additionalInfo == null || additionalInfo.isBlank()) {
            return baseInfo;
        }
        return baseInfo + ", " + additionalInfo;
    }
}
