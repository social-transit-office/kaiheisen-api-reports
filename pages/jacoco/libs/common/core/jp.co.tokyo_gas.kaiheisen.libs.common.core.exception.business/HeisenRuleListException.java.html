package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 閉栓共通リストエラーを表す例外クラス。
 */
public class HeisenRuleListException extends HeisenRuleException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_020000011;

    private final List<HeisenRuleException> heisenRuleExceptionList;

    /**
     * コンストラクタ。
     * @param heisenRuleExceptionList 閉栓共通エラーリスト
     * @param detailMessage 詳細メッセージ
     * @param cause 原因例外
     */
    private HeisenRuleListException(
            List<HeisenRuleException> heisenRuleExceptionList,
            String detailMessage,
            Throwable cause) {
        super(buildMessage(
                        "HeisenRuleListException",
                        detailMessage,
                        "heisenRuleExceptionList.size="
                                + (heisenRuleExceptionList == null ? 0 : heisenRuleExceptionList.size()),
                        resolveErrorCode1FromErrorCode(ERROR_CODE),
                        resolveErrorCode2FromErrorCode(ERROR_CODE),
                        resolveErrorCode3FromErrorCode(ERROR_CODE),
                        ERROR_CODE,
                        List.of()),
                cause,
                    ERROR_CODE,
                List.of(),
                null,
                List.of(),
                    resolveReasonCodeFromErrorCode(ERROR_CODE));
        this.heisenRuleExceptionList = heisenRuleExceptionList == null ? List.of() : List.copyOf(heisenRuleExceptionList);
    }

    /**
     * コンストラクタ。
     * @param heisenRuleExceptionList 閉栓共通エラーリスト
     */
    public HeisenRuleListException(List<HeisenRuleException> heisenRuleExceptionList) {
        this(heisenRuleExceptionList, null, null);
    }

    /**
     * 閉栓共通エラーリストを取得する。
     * @return 閉栓共通エラーリスト
     */
    public List<HeisenRuleException> getHeisenRuleExceptionList() {
        return heisenRuleExceptionList;
    }
}
