package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 開栓共通リストエラーを表す例外クラス。
 */
public class KaisenRuleListException extends KaisenRuleException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_020000011;

    private final List<KaisenRuleException> kaisenRuleExceptionList;

    /**
     * コンストラクタ。
     * @param kaisenRuleExceptionList 開栓共通エラーリスト
     * @param detailMessage 詳細メッセージ
     * @param cause 原因例外
     */
    private KaisenRuleListException(
            List<KaisenRuleException> kaisenRuleExceptionList,
            String detailMessage,
            Throwable cause) {
        super(buildMessage(
                        "KaisenRuleListException",
                        detailMessage,
                        "kaisenRuleExceptionList.size="
                                + (kaisenRuleExceptionList == null ? 0 : kaisenRuleExceptionList.size()),
                        resolveErrorCode1FromErrorCode(ERROR_CODE),
                        resolveErrorCode2FromErrorCode(ERROR_CODE),
                        resolveErrorCode3FromErrorCode(ERROR_CODE),
                        ERROR_CODE,
                        List.of()),
                cause,
                    ERROR_CODE,
                List.of(),
                null,
                List.of(),
                    resolveReasonCodeFromErrorCode(ERROR_CODE));
        this.kaisenRuleExceptionList = kaisenRuleExceptionList == null ? List.of() : List.copyOf(kaisenRuleExceptionList);
    }

    /**
     * コンストラクタ。
     * @param kaisenRuleExceptionList 開栓共通エラーリスト
     */
    public KaisenRuleListException(List<KaisenRuleException> kaisenRuleExceptionList) {
        this(kaisenRuleExceptionList, null, null);
    }

    /**
     * 開栓共通エラーリストを取得する。
     * @return 開栓共通エラーリスト
     */
    public List<KaisenRuleException> getKaisenRuleExceptionList() {
        return kaisenRuleExceptionList;
    }
}
