/**
 * @file HeisenRuleShiyouryoKyoteichuException.java
 * 概要: 使用量協定中エラーを表す例外を定義する。
 * 主な仕様:
 * - エラーガスメーター設置場所番号を保持する。
 * 既知の制限事項:
 * - エラーコードは固定値を利用する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 使用量協定中エラーを表す例外クラス。
 * 型: HeisenRuleException
 * 主な仕様:
 * - エラーガスメーター設置場所番号を保持する。
 * 既知の制限事項:
 * - 閉栓エラー理由コードは暫定ロジックで解決する。
 * @since 1.0.0
 */
public class HeisenRuleShiyouryoKyoteichuException extends HeisenRuleException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_020200026;

    /**
     * エラーガスメーター設置場所番号。
     * @type String
     */
    private final String errorGasMtstNo;

    /**
     * コンストラクタ。
     * @param errorGasMtstNo エラーガスメーター設置場所番号。型: String
     */
    public HeisenRuleShiyouryoKyoteichuException(String errorGasMtstNo) {
        this(List.of(normalizeEmbeddedString(errorGasMtstNo)), null, List.of(), errorGasMtstNo, null, null);
    }

    private static String normalizeEmbeddedString(String embeddedString) {
        return embeddedString == null ? "" : embeddedString;
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @param errorGasMtstNo エラーガスメーター設置場所番号。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    private HeisenRuleShiyouryoKyoteichuException(
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList,
            String errorGasMtstNo,
            String detailMessage,
            Throwable cause) {
        super(buildMessage("HeisenRuleShiyouryoKyoteichuException", detailMessage,
                "heisenErrorReasonCode=" + resolveReasonCodeFromErrorCode(
                ERROR_CODE)
                        + ", ciriusMessageId=" + ciriusMessageId
                        + ", ciriusMessageEmbeddedStringList=" + formatList(ciriusMessageEmbeddedStringList)
                        + ", errorGasMtstNo=" + errorGasMtstNo,
            resolveErrorCode1FromErrorCode(ERROR_CODE),
            resolveErrorCode2FromErrorCode(ERROR_CODE),
            resolveErrorCode3FromErrorCode(ERROR_CODE),
            ERROR_CODE,
                errorMessageEmbeddedStringList),
                cause,
            ERROR_CODE,
                errorMessageEmbeddedStringList,
                ciriusMessageId,
                ciriusMessageEmbeddedStringList,
                resolveReasonCodeFromErrorCode(
                ERROR_CODE));
        this.errorGasMtstNo = errorGasMtstNo;
    }

    /**
     * エラーガスメーター設置場所番号を取得する。
     * @return エラーガスメーター設置場所番号。型: String
     */
    public String getErrorGasMtstNo() {
        return errorGasMtstNo;
    }
}
