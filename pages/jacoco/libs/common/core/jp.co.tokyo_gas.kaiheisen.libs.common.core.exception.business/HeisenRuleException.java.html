/**
 * @file HeisenRuleException.java
 * 概要: 閉栓共通エラーを表す例外を定義する。
 * 主な仕様:
 * - 閉栓エラー理由コードを保持する。
 * 既知の制限事項:
 * - エラー理由コードのマッピングは暫定実装とする。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business;

import java.util.List;

/**
 * 閉栓共通エラーを表す例外クラス。
 * 型: BusinessRuleException
 * 主な仕様:
 * - 閉栓エラー理由コードを保持する。
 * 既知の制限事項:
 * - エラー理由コードはエラーコード下2桁を優先する暫定実装とする。
 * @since 1.0.0
 */
public class HeisenRuleException extends BusinessRuleException {

    /**
     * 閉栓エラー理由コード。
     * @type String
     */
    private final String heisenErrorReasonCode;

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     */
    public HeisenRuleException(String errorCode) {
        this(errorCode, List.of(), List.of());
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     */
    public HeisenRuleException(String errorCode, List<String> errorMessageEmbeddedStringList) {
        this(errorCode, errorMessageEmbeddedStringList, List.of());
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     */
    public HeisenRuleException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            List<String> ciriusMessageEmbeddedStringList) {
        this(errorCode, errorMessageEmbeddedStringList, null, ciriusMessageEmbeddedStringList, null, null, null);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @param heisenErrorReasonCode 閉栓エラー理由コード。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected HeisenRuleException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList,
            String heisenErrorReasonCode,
            String detailMessage,
            Throwable cause) {
        super(
            "HeisenRuleException",
            errorCode,
            errorMessageEmbeddedStringList,
            ciriusMessageId,
            ciriusMessageEmbeddedStringList,
            "heisenErrorReasonCode=" + resolveReasonCode(errorCode, heisenErrorReasonCode),
            detailMessage,
            cause);
        this.heisenErrorReasonCode = resolveReasonCode(errorCode, heisenErrorReasonCode);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param ciriusMessageId CIRIUSメッセージID。型: String
     * @param ciriusMessageEmbeddedStringList CIRIUSメッセージ埋込み文字列リスト。型: List<String>
     * @param heisenErrorReasonCode 閉栓エラー理由コード。型: String
     */
    protected HeisenRuleException(
            String message,
            Throwable cause,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String ciriusMessageId,
            List<String> ciriusMessageEmbeddedStringList,
            String heisenErrorReasonCode) {
        super(message, cause, errorCode, errorMessageEmbeddedStringList, ciriusMessageId, ciriusMessageEmbeddedStringList);
        this.heisenErrorReasonCode = heisenErrorReasonCode;
    }

    /**
     * 閉栓エラー理由コードを取得する。
     * @return 閉栓エラー理由コード。型: String
     */
    public String getHeisenErrorReasonCode() {
        return heisenErrorReasonCode;
    }

    /**
     * 閉栓エラー理由コードを解決する。
     * @param errorCode エラーコード。型: String
     * @param providedReasonCode 明示指定の理由コード。型: String
     * @return 解決後の理由コード。型: String
     */
    protected static String resolveReasonCode(String errorCode, String providedReasonCode) {
        if (providedReasonCode != null && !providedReasonCode.isBlank()) {
            return providedReasonCode;
        }
        return resolveReasonCodeFromErrorCode(errorCode);
    }
}
