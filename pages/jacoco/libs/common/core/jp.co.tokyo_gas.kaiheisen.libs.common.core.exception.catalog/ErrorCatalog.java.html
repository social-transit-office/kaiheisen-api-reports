    private ErrorCatalog() {
    }

    /**
     * エラーコードに対応するエラー定義を検索する。
     *
     * @param errorCode エラーコード
     * @return エラー定義。見つからない場合は {@code null}
     */
    public static ErrorDefinition find(String errorCode) {
        if (errorCode == null || errorCode.isBlank()) {
            return null;
        }
        return DEFINITIONS.get(errorCode.trim());
    }

    /**
     * ラウンディングを解決した最終エラーコードを返す。
     *
     * @param errorCode エラーコード
     * @return 最終エラーコード
     */
    public static String resolveFinalErrorCode(String errorCode) {
        ErrorDefinition resolved = resolveFinalDefinition(errorCode);
        if (resolved == null) {
            return errorCode;
        }
        return resolved.errorCode();
    }

    /**
     * ラウンディングを解決した最終エラー定義を返す。
     *
     * @param errorCode エラーコード
     * @return 最終エラー定義。見つからない場合は {@code null}
     * @throws IllegalStateException 丸め先エラーコードが循環している場合
     */
    public static ErrorDefinition resolveFinalDefinition(String errorCode) {
        if (errorCode == null || errorCode.isBlank()) {
            return null;
        }
        String normalized = errorCode.trim();
        Set<String> visited = new HashSet<>();
        ArrayDeque<String> history = new ArrayDeque<>();
        while (true) {
            if (!visited.add(normalized)) {
                history.addLast(normalized);
                throw new IllegalStateException("Rounding loop detected: " + String.join(" -> ", history));
            }
            history.addLast(normalized);
            ErrorDefinition current = DEFINITIONS.get(normalized);
            if (current == null) {
                return null;
            }
            String target = current.roundingTargetErrorCode();
            if (target == null || target.isBlank()) {
                return current;
            }
            if (!DEFINITIONS.containsKey(target)) {
                return current;
            }
            normalized = target;
        }
    }

    /**
     * エラーコードと埋め込み値からエラーメッセージを解決する。
     *
     * @param errorCode エラーコード
     * @param embeddedValues 埋め込み値リスト
     * @return エラーメッセージ。定義が見つからない場合は {@code null}
     */
    public static String resolveErrorMessage(String errorCode, List<String> embeddedValues) {
        ErrorDefinition definition = resolveFinalDefinition(errorCode);
        if (definition == null) {
            return null;
        }
        return applyPlaceholders(definition.messageTemplate(), embeddedValues);
    }

    /**
     * エラーコードから CIRIUS メッセージIDを解決する。
     *
     * @param errorCode エラーコード
     * @return CIRIUS メッセージID。未設定または定義が見つからない場合は {@code null}
     */
    public static String resolveCiriusMessageId(String errorCode) {
        ErrorDefinition definition = resolveFinalDefinition(errorCode);
        if (definition == null) {
            return null;
        }
        String messageId = definition.ciriusMessageId();
        if (messageId == null || messageId.isBlank()) {
            return null;
        }
        return messageId;
    }

    /**
     * エラーコードと埋め込み値から CIRIUS メッセージを解決する。
     *
     * @param errorCode エラーコード
     * @param embeddedValues 埋め込み値リスト
     * @return CIRIUS メッセージ。未設定または定義が見つからない場合は {@code null}
     */
    public static String resolveCiriusMessage(String errorCode, List<String> embeddedValues) {
        ErrorDefinition definition = resolveFinalDefinition(errorCode);
        if (definition == null) {
            return null;
        }
        String template = definition.ciriusMessageTemplate();
        if (template == null || template.isBlank()) {
            return null;
        }
        return applyPlaceholders(template, embeddedValues);
    }

    /**
     * メッセージテンプレートに埋め込み値を適用する。
     *
     * @param template メッセージテンプレート
     * @param embeddedValues 埋め込み値リスト
     * @return プレースホルダ適用済み文字列。template が {@code null} の場合は {@code null}
     */
    public static String applyPlaceholders(String template, List<String> embeddedValues) {
        if (template == null) {
            return null;
        }
        List<String> values = embeddedValues == null ? List.of() : new ArrayList<>(embeddedValues);
        String resolved = template;
        boolean zeroBasedTemplate = template.contains("{0}");
        boolean oneBasedValuesWithDummyZero = !zeroBasedTemplate
                && !values.isEmpty()
                && Objects.toString(values.getFirst(), "").isEmpty();
        for (int i = 0; i < values.size(); i++) {
            if (oneBasedValuesWithDummyZero && i == 0) {
                continue;
            }
            int placeholderIndex;
            if (zeroBasedTemplate || oneBasedValuesWithDummyZero) {
                placeholderIndex = i;
            } else {
                placeholderIndex = i + 1;
            }
            String placeholder = "{" + placeholderIndex + "}";
            String value = Objects.toString(values.get(i), "");
            resolved = resolved.replace(placeholder, value);
        }
        return resolved;
    }
