/**
 * @file LogField.java
 * 概要: ログ出力の予約語キーを定義する。
 * 主な仕様:
 * - ログ項目名を固定し、表記揺れを防止する。
 * 既知の制限事項:
 * - 追加項目は規約確定後に追記する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

/**
 * ログ予約語の列挙。
 */
public enum LogField {
    CORRELATION_ID("correlationId"),
    CALL_PATH("callPath"),
    LAYER("layer"),
    EVENT("event"),
    SUMMARY("summary"),
    RESOURCE_CLASS("resourceClass"),
    SERVICE_ID("serviceId"),
    SERVICE_NAME("serviceName"),
    HTTP_METHOD("httpMethod"),
    PATH("path"),
    HTTP_STATUS("httpStatus"),
    LATENCY_MS("latencyMs"),
    REQUEST_SENT_AT("requestSentAt"),
    RESPONSE_RECEIVED_AT("responseReceivedAt"),
    FAILED_AT("failedAt"),
    REQUEST_TO_RESOURCE_LATENCY_MS("requestToResourceLatencyMs"),
    DA_INVOCATION_SEQUENCE("daInvocationSequence"),
    SERVICE_INVOCATION_SEQUENCE("serviceInvocationSequence"),
    EXTERNAL_API_INVOCATION_SEQUENCE("externalApiInvocationSequence"),
    MEASUREMENT_UNAVAILABLE("measurementUnavailable"),
    ERROR_CODE("errorCode"),
    QUERY_PARAMS("queryParams"),
    REQUEST_BODY("requestBody"),
    EXCEPTION_CLASS("exception.class"),
    ROOT_CAUSE_CLASS("exception.rootCause.class"),
    ROOT_CAUSE_MESSAGE("exception.rootCause.message"),
    EXCEPTION_CHAIN("exception.chain"),
    STACK_TRACE("exception.stackTrace"),
    ORIGIN_CLASS("exception.origin.class"),
    ORIGIN_METHOD("exception.origin.method"),
    ORIGIN_FILE("exception.origin.file"),
    ORIGIN_LINE("exception.origin.line"),
    DA_IDENTIFIER("exception.da.identifier"),
    SQL_PRESENT("exception.da.sqlPresent");

    private final String key;

    LogField(String key) {
        this.key = key;
    }

    /**
     * 予約語キーを取得する。
     *
     * @return 予約語キー。型: String
     */
    public String key() {
        return key;
    }
}
