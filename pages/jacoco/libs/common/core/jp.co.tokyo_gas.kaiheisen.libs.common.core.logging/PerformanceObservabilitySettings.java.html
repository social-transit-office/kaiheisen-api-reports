package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import org.eclipse.microprofile.config.Config;
import org.eclipse.microprofile.config.ConfigProvider;

/**
 * 詳細性能計測の有効状態を提供する。
 */
public final class PerformanceObservabilitySettings {

    /** 詳細性能計測の設定キー。 */
    public static final String ENABLED_KEY = "kaiheisen.observability.performance.enabled";

    /** Resource / Service / external API terminal timing log setting key. */
    public static final String TERMINAL_ENABLED_KEY =
            "kaiheisen.observability.performance.terminal.enabled";

    private PerformanceObservabilitySettings() {
    }

    /**
     * 詳細性能計測が有効かを返す。
     *
     * <p>設定取得に失敗した場合は業務処理を優先して無効として扱う。</p>
     *
     * @return 有効の場合 {@code true}
     */
    public static boolean isEnabled() {
        return readBoolean(ENABLED_KEY);
    }

    /**
     * Returns whether terminal timing logs are enabled.
     *
     * @return {@code true} when terminal timing logs are enabled
     */
    public static boolean isTerminalEnabled() {
        return readBoolean(TERMINAL_ENABLED_KEY);
    }

    private static boolean readBoolean(final String key) {
        String systemProperty = System.getProperty(key);
        if (systemProperty != null) {
            return Boolean.parseBoolean(systemProperty);
        }
        try {
            Config config = ConfigProvider.getConfig();
            return config.getOptionalValue(key, Boolean.class).orElse(false);
        } catch (RuntimeException exception) {
            return false;
        }
    }
}
