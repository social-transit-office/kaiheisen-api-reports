/**
 * @file LogPayloadFormatter.java
 * 概要: ログ属性マップを JSON 文字列へ変換する。
 * 主な仕様:
 * - JSON ログを前提に簡易的な変換を行う。
 * 既知の制限事項:
 * - 対象型は String/Number/Boolean/Map/Iterable を想定する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.util.Iterator;
import java.util.Map;

/**
 * ログ属性の JSON 変換ユーティリティ。
 */
public final class LogPayloadFormatter {

    private LogPayloadFormatter() {
    }

    /**
     * 属性マップを JSON 文字列へ変換する。
     *
     * @param payload 属性マップ。型: Map<String, Object>
     * @return JSON 文字列。型: String
     */
    public static String toJson(Map<String, Object> payload) {
        StringBuilder builder = new StringBuilder();
        appendJsonObject(builder, payload);
        return builder.toString();
    }

    private static void appendJsonObject(StringBuilder builder, Map<String, Object> map) {
        builder.append('{');
        if (map != null) {
            Iterator<Map.Entry<String, Object>> iterator = map.entrySet().iterator();
            while (iterator.hasNext()) {
                Map.Entry<String, Object> entry = iterator.next();
                appendJsonString(builder, entry.getKey());
                builder.append(':');
                appendJsonValue(builder, entry.getValue());
                if (iterator.hasNext()) {
                    builder.append(',');
                }
            }
        }
        builder.append('}');
    }

    private static void appendJsonArray(StringBuilder builder, Iterable<?> values) {
        builder.append('[');
        if (values != null) {
            Iterator<?> iterator = values.iterator();
            while (iterator.hasNext()) {
                appendJsonValue(builder, iterator.next());
                if (iterator.hasNext()) {
                    builder.append(',');
                }
            }
        }
        builder.append(']');
    }

    private static void appendJsonValue(StringBuilder builder, Object value) {
        if (value == null) {
            builder.append("null");
            return;
        }
        if (value instanceof String text) {
            appendJsonString(builder, text);
            return;
        }
        if (value instanceof Number || value instanceof Boolean) {
            builder.append(value);
            return;
        }
        if (value instanceof Map<?, ?> map) {
            appendJsonObject(builder, castMap(map));
            return;
        }
        if (value instanceof Iterable<?> iterable) {
            appendJsonArray(builder, iterable);
            return;
        }
        appendJsonString(builder, value.toString());
    }

    @SuppressWarnings("unchecked")
    private static Map<String, Object> castMap(Map<?, ?> map) {
        return (Map<String, Object>) map;
    }

    private static void appendJsonString(StringBuilder builder, String text) {
        builder.append('"');
        if (text != null) {
            for (int i = 0; i < text.length(); i++) {
                char ch = text.charAt(i);
                switch (ch) {
                    case '\\' -> builder.append("\\\\");
                    case '"' -> builder.append("\\\"");
                    case '\n' -> builder.append("\\n");
                    case '\r' -> builder.append("\\r");
                    case '\t' -> builder.append("\\t");
                    default -> builder.append(ch);
                }
            }
        }
        builder.append('"');
    }
}
