/**
 * @file DefaultExceptionLogMapper.java
 * 概要: 例外カタログに基づくログマッピングの既定実装を定義する。
 * 主な仕様:
 * - 例外種別に応じてログレベルと HTTP ステータスを決定する。
 * 既知の制限事項:
 * - 詳細なマッピングは将来の例外カタログ確定後に調整する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DataInconsistencyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DatabaseException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DeadLockException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database.DuplicateKeyException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.ExternalSystemException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationException;

/**
 * 例外→ログ出力情報の既定マッピング。
 */
public final class DefaultExceptionLogMapper implements ExceptionLogMapper {

    private static final int HTTP_BAD_REQUEST = 400;

    private static final int HTTP_INTERNAL_ERROR = 500;

    private final LogLayer layer;

    private final ExceptionLogSettings settings;

    /**
     * コンストラクタ。
     *
     * @param layer レイヤ。型: LogLayer
     */
    public DefaultExceptionLogMapper(LogLayer layer) {
        this(layer, ExceptionLogSettingsProvider.load());
    }

    /**
     * コンストラクタ。
     *
     * @param layer レイヤ。型: LogLayer
     * @param settings 例外ログ設定。型: ExceptionLogSettings
     */
    public DefaultExceptionLogMapper(LogLayer layer, ExceptionLogSettings settings) {
        this.layer = Objects.requireNonNull(layer, "layer");
        this.settings = Objects.requireNonNull(settings, "settings");
    }

    @Override
    public LogDecision map(Throwable throwable) {
        return map(throwable, ExceptionLogContext.empty());
    }

    @Override
    public LogDecision map(Throwable throwable, ExceptionLogContext context) {
        LogLevel level = resolveLevel(throwable);
        Integer httpStatus = resolveHttpStatus(throwable);
        String errorCode = resolveErrorCode(throwable);
        String summary = resolveSummary(throwable);

        Map<String, Object> attributes = buildAttributes(throwable, context == null ? ExceptionLogContext.empty() : context);

        LogEvent event = new LogEvent(
                "exception",
                summary,
                layer,
                httpStatus,
                errorCode,
                attributes);
        return new LogDecision(level, event);
    }

    private Map<String, Object> buildAttributes(Throwable throwable, ExceptionLogContext context) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        if (throwable == null) {
            return attributes;
        }
        attributes.put(LogField.EXCEPTION_CLASS.key(), throwable.getClass().getName());
        putIfPresent(attributes, LogField.CORRELATION_ID, context.correlationId());
        putIfPresent(attributes, LogField.HTTP_METHOD, context.httpMethod());
        putIfPresent(attributes, LogField.PATH, context.requestPath());
        putIfPresent(attributes, LogField.RESOURCE_CLASS, context.resourceClass());
        putIfPresent(attributes, LogField.SERVICE_ID, context.serviceId());
        putIfPresent(attributes, LogField.SERVICE_NAME, context.serviceName());

        Throwable rootCause = resolveRootCause(throwable);
        attributes.put(LogField.ROOT_CAUSE_CLASS.key(), rootCause.getClass().getName());
        putIfPresent(attributes, LogField.ROOT_CAUSE_MESSAGE, normalize(rootCause.getMessage()));

        if (throwable instanceof DatabaseException databaseException) {
            putIfPresent(attributes, LogField.DA_IDENTIFIER, databaseException.getDaIdentifier());
            attributes.put(LogField.SQL_PRESENT.key(), databaseException.getSql() != null);
        }

        if (settings.detailLevel().isDetailed()) {
            StackTraceElement origin = resolveOrigin(rootCause);
            if (origin != null) {
                attributes.put(LogField.ORIGIN_CLASS.key(), origin.getClassName());
                attributes.put(LogField.ORIGIN_METHOD.key(), origin.getMethodName());
                putIfPresent(attributes, LogField.ORIGIN_FILE, origin.getFileName());
                if (origin.getLineNumber() >= 0) {
                    attributes.put(LogField.ORIGIN_LINE.key(), origin.getLineNumber());
                }
            }
            attributes.put(LogField.EXCEPTION_CHAIN.key(), buildExceptionChain(throwable));
            attributes.put(LogField.STACK_TRACE.key(), buildStackTraceText(throwable));
        }
        return attributes;
    }

    private static LogLevel resolveLevel(Throwable throwable) {
        if (throwable instanceof ValidationException) {
            return LogLevel.WARN;
        }
        if (throwable instanceof BusinessRuleException) {
            return LogLevel.WARN;
        }
        if (throwable instanceof DataNotFoundException) {
            return LogLevel.WARN;
        }
        return LogLevel.ERROR;
    }

    private static Integer resolveHttpStatus(Throwable throwable) {
        if (throwable instanceof ValidationException) {
            return HTTP_BAD_REQUEST;
        }
        if (throwable instanceof BusinessRuleException) {
            return HTTP_BAD_REQUEST;
        }
        if (throwable instanceof ExternalSystemException
                || throwable instanceof DatabaseException
                || throwable instanceof DataInconsistencyException
                || throwable instanceof DeadLockException
                || throwable instanceof DuplicateKeyException) {
            return HTTP_INTERNAL_ERROR;
        }
        return HTTP_INTERNAL_ERROR;
    }

    private static String resolveErrorCode(Throwable throwable) {
        if (throwable instanceof KaiheisenException kaiheisenException) {
            return kaiheisenException.getErrorCode();
        }
        return null;
    }

    private static String resolveSummary(Throwable throwable) {
        if (throwable == null) {
            return "exception";
        }
        String message = throwable.getMessage();
        if (message == null || message.isBlank()) {
            return throwable.getClass().getSimpleName();
        }
        return message;
    }

    private static Throwable resolveRootCause(Throwable throwable) {
        Throwable current = throwable;
        while (current != null && current.getCause() != null && current.getCause() != current) {
            current = current.getCause();
        }
        return current == null ? throwable : current;
    }

    private static StackTraceElement resolveOrigin(Throwable throwable) {
        if (throwable == null || throwable.getStackTrace() == null || throwable.getStackTrace().length == 0) {
            return null;
        }
        return throwable.getStackTrace()[0];
    }

    private static String buildExceptionChain(Throwable throwable) {
        StringBuilder builder = new StringBuilder();
        Throwable current = throwable;
        while (current != null) {
            if (builder.length() != 0) {
                builder.append(" <- ");
            }
            builder.append(current.getClass().getSimpleName());
            String message = normalize(current.getMessage());
            if (message != null) {
                builder.append('(').append(message).append(')');
            }
            current = current.getCause();
        }
        return builder.toString();
    }

    private static String buildStackTraceText(Throwable throwable) {
        return Arrays.stream(throwable.getStackTrace())
                .limit(12)
                .map(StackTraceElement::toString)
                .reduce((left, right) -> left + "\n" + right)
                .orElse("");
    }

    private static void putIfPresent(Map<String, Object> attributes, LogField field, String value) {
        String normalized = normalize(value);
        if (normalized != null) {
            attributes.put(field.key(), normalized);
        }
    }

    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String trimmed = value.trim();
        return trimmed.isEmpty() ? null : trimmed;
    }
}
