/**
 * @file ExceptionLogMapper.java
 * 概要: 例外をログ出力情報へ変換するインターフェース。
 * 主な仕様:
 * - 例外カタログに基づくマッピングを実装側で行う。
 * 既知の制限事項:
 * - 具体的なマッピングは別実装で提供する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

/**
 * 例外→ログ出力情報の変換を行う。
 */
public interface ExceptionLogMapper {

    /**
     * 例外をログ出力情報へ変換する。
     *
     * @param throwable 対象例外。型: Throwable
     * @return ログ出力情報。型: LogDecision
     */
    LogDecision map(Throwable throwable);

    /**
     * 例外とログ文脈をログ出力情報へ変換する。
     *
     * @param throwable 対象例外。型: Throwable
     * @param context ログ文脈。型: ExceptionLogContext
     * @return ログ出力情報。型: LogDecision
     */
    default LogDecision map(Throwable throwable, ExceptionLogContext context) {
        return map(throwable);
    }
}
