package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import org.eclipse.microprofile.config.Config;
import org.eclipse.microprofile.config.ConfigProvider;

/** DataAccess 内部フェーズ計測の設定を提供する。 */
public final class DataAccessPhaseObservabilitySettings {

    /** DataAccess 内部フェーズ計測の設定キー。 */
    public static final String ENABLED_KEY = "kaiheisen.observability.da-phase.enabled";

    private DataAccessPhaseObservabilitySettings() {
    }

    /**
     * DataAccess 内部フェーズ計測が有効かを返す。
     *
     * <p>設定取得に失敗した場合は業務処理を優先し、無効として扱う。</p>
     *
     * @return 有効の場合 {@code true}
     */
    public static boolean isEnabled() {
        String systemProperty = System.getProperty(ENABLED_KEY);
        if (systemProperty != null) {
            return Boolean.parseBoolean(systemProperty);
        }
        try {
            Config config = ConfigProvider.getConfig();
            return config.getOptionalValue(ENABLED_KEY, Boolean.class).orElse(false);
        } catch (RuntimeException exception) {
            return false;
        }
    }
}
