/**
 * @file LogEvent.java
 * 概要: 共通ロギングのイベント情報を保持する。
 * 主な仕様:
 * - 予約語と任意属性を組み合わせてログ出力用の構造体を表現する。
 * 既知の制限事項:
 * - 具体的なマスキング処理は別クラスに委譲する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.util.Map;

/**
 * ログ出力用の共通イベント。
 *
 * @param event イベント名。型: String
 * @param summary 要約。型: String
 * @param layer レイヤ。型: LogLayer
 * @param httpStatus HTTP ステータス。型: Integer
 * @param errorCode エラーコード。型: String
 * @param attributes 追加属性。型: Map<String, Object>
 */
public record LogEvent(
        String event,
        String summary,
        LogLayer layer,
        Integer httpStatus,
        String errorCode,
        Map<String, Object> attributes) {
}
