package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.lang.reflect.Method;
import java.lang.reflect.RecordComponent;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * record 型のリクエストをログ出力用の requestBody へ変換する.
 */
public final class LogRequestBodyMapper {

    private LogRequestBodyMapper() {
    }

    /**
     * record を component 宣言順の Map へ変換する.
     *
     * @param request リクエストオブジェクト
     * @return requestBody マップ
     */
    public static Map<String, Object> toMap(final Object request) {
        if (request == null || !request.getClass().isRecord()) {
            return Map.of();
        }
        Map<String, Object> requestBody = new LinkedHashMap<>();
        RecordComponent[] components = request.getClass().getRecordComponents();
        for (RecordComponent component : components) {
            try {
                Method accessor = component.getAccessor();
                accessor.trySetAccessible();
                requestBody.put(
                        component.getName(),
                        accessor.invoke(request));
            } catch (ReflectiveOperationException ex) {
                // Skip only the failing component.
                // Keep other fields visible in the log payload.
            }
        }
        return requestBody;
    }
}
