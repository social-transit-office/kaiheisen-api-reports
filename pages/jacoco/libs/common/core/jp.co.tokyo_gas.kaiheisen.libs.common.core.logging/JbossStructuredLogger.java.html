/**
 * @file JbossStructuredLogger.java
 * 概要: JBoss Logger へ委譲する構造化ログ実装。
 * 主な仕様:
 * - LogEvent を JSON 文字列へ変換して出力する。
 * 既知の制限事項:
 * - マスキングは LogSanitizer 実装を追加後に適用する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.util.Map;
import java.util.Objects;
import org.jboss.logging.Logger;

/**
 * JBoss Logger 連携の構造化ログ実装。
 */
public final class JbossStructuredLogger implements StructuredLogger {

    private final Logger logger;

    private final LogSanitizer sanitizer;

    /**
     * コンストラクタ。
     *
     * @param owner ロガーの所有クラス。型: Class<?>
     */
    public JbossStructuredLogger(Class<?> owner) {
        this(Logger.getLogger(owner), null);
    }

    /**
     * コンストラクタ。
     *
     * @param logger JBoss Logger。型: Logger
     */
    public JbossStructuredLogger(Logger logger) {
        this(logger, null);
    }

    /**
     * コンストラクタ。
     *
     * @param logger JBoss Logger。型: Logger
     * @param sanitizer サニタイザ。型: LogSanitizer
     */
    public JbossStructuredLogger(Logger logger, LogSanitizer sanitizer) {
        this.logger = Objects.requireNonNull(logger, "logger");
        this.sanitizer = sanitizer;
    }

    @Override
    public void info(LogEvent event) {
        logger.info(toPayload(event));
    }

    @Override
    public void warn(LogEvent event) {
        logger.warn(toPayload(event));
    }

    @Override
    public void error(LogEvent event, Throwable throwable) {
        logger.error(toPayload(event), throwable);
    }

    private String toPayload(LogEvent event) {
        Map<String, Object> payload = LogPayloadMapper.toMap(event);
        if (sanitizer != null) {
            payload = sanitizer.sanitize(payload);
        }
        return LogPayloadFormatter.toJson(payload);
    }
}
