package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import org.eclipse.microprofile.config.Config;
import org.eclipse.microprofile.config.ConfigProvider;

/**
 * 例外ログ設定のロードを担当する。
 */
public final class ExceptionLogSettingsProvider {

    static final String DETAIL_LEVEL_KEY = "kaiheisen.logging.exception-detail-level";

    private ExceptionLogSettingsProvider() {
    }

    /**
     * 現在の設定をロードする。
     *
     * @return 例外ログ設定
     */
    public static ExceptionLogSettings load() {
        String raw = readValue(DETAIL_LEVEL_KEY);
        try {
            return new ExceptionLogSettings(ExceptionDetailLevel.from(raw));
        } catch (IllegalArgumentException ex) {
            return ExceptionLogSettings.basic();
        }
    }

    private static String readValue(String key) {
        String systemProperty = System.getProperty(key);
        if (systemProperty != null && !systemProperty.isBlank()) {
            return systemProperty;
        }

        try {
            Config config = ConfigProvider.getConfig();
            return config.getOptionalValue(key, String.class).orElseGet(() -> readEnv(key));
        } catch (RuntimeException ex) {
            return readEnv(key);
        }
    }

    private static String readEnv(String key) {
        String envKey = key.toUpperCase().replace('.', '_').replace('-', '_');
        return System.getenv(envKey);
    }
}
