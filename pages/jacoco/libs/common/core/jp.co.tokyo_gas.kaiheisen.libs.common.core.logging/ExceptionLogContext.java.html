package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

/**
 * 例外ログの文脈情報。
 *
 * @param correlationId 相関 ID
 * @param httpMethod HTTP メソッド
 * @param requestPath リクエストパス
 * @param resourceClass 処理起点クラス名
 * @param serviceId 実行中サービス ID
 * @param serviceName 実行中サービス名
 */
public record ExceptionLogContext(
        String correlationId,
        String httpMethod,
        String requestPath,
     String resourceClass,
     String serviceId,
     String serviceName) {

    /**
     * 空の文脈を返す。
     *
     * @return 空の文脈
     */
    public static ExceptionLogContext empty() {
        return new ExceptionLogContext(null, null, null, null, null, null);
    }
}
