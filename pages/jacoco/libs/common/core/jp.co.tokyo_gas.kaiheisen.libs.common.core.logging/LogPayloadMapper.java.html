/**
 * @file LogPayloadMapper.java
 * 概要: LogEvent をログ出力用のマップへ変換する。
 * 主な仕様:
 * - 予約語を優先し、追加属性をマージする。
 * 既知の制限事項:
 * - 属性のマスキングは LogSanitizer で実施する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.logging;

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import org.jboss.logging.MDC;

/**
 * LogEvent の属性マッピングを行う。
 */
public final class LogPayloadMapper {

    private static final Set<String> RESERVED_ATTRIBUTE_KEYS = Set.of(
            LogField.CORRELATION_ID.key(),
            LogField.EVENT.key(),
            LogField.SUMMARY.key(),
            LogField.LAYER.key(),
            LogField.HTTP_STATUS.key(),
            LogField.ERROR_CODE.key());

    private LogPayloadMapper() {
    }

    /**
     * LogEvent をマップへ変換する。
     *
     * @param event ログイベント。型: LogEvent
     * @return ログ属性マップ。型: Map<String, Object>
     */
    public static Map<String, Object> toMap(LogEvent event) {
        Map<String, Object> payload = new LinkedHashMap<>();
        if (event == null) {
            return payload;
        }
        Object correlationId = MDC.get(LogField.CORRELATION_ID.key());
        if (correlationId != null) {
            payload.put(LogField.CORRELATION_ID.key(), correlationId.toString());
        } else if (event.attributes() != null) {
            Object attributeCorrelationId = event.attributes().get(LogField.CORRELATION_ID.key());
            if (attributeCorrelationId != null) {
                payload.put(LogField.CORRELATION_ID.key(), attributeCorrelationId.toString());
            }
        }
        if (event.event() != null) {
            payload.put(LogField.EVENT.key(), event.event());
        }
        if (event.summary() != null) {
            payload.put(LogField.SUMMARY.key(), event.summary());
        }
        if (event.layer() != null) {
            payload.put(LogField.LAYER.key(), event.layer().name());
        }
        if (event.httpStatus() != null) {
            payload.put(LogField.HTTP_STATUS.key(), event.httpStatus());
        }
        if (event.errorCode() != null) {
            payload.put(LogField.ERROR_CODE.key(), event.errorCode());
        }
        if (event.attributes() != null) {
            for (Map.Entry<String, Object> entry : event.attributes().entrySet()) {
                if (!RESERVED_ATTRIBUTE_KEYS.contains(entry.getKey())) {
                    payload.put(entry.getKey(), entry.getValue());
                }
            }
        }
        return payload;
    }
}
