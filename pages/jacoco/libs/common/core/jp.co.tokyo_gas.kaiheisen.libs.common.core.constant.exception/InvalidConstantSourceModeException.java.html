package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception;

/**
 * コンスタント取得元モード不正例外。
 */
public class InvalidConstantSourceModeException extends RuntimeException {

    /**
     * 不正な取得元モード値を指定してインスタンスを生成する。
     *
     * @param value 不正な取得元モード値
     */
    public InvalidConstantSourceModeException(String value) {
        super("Unsupported constant source mode: value=" + value);
    }
}
