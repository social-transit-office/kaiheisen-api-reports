package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception;

/**
 * 外部コンスタント取得未実装例外。
 */
public class UnsupportedExternalConstantSourceException extends UnsupportedOperationException {

    /**
     * サービス名を指定してインスタンスを生成する。
     *
     * @param serviceName 外部コンスタント取得が未実装のサービス名
     */
    public UnsupportedExternalConstantSourceException(String serviceName) {
        super("External constant source is not implemented yet: service=" + serviceName);
    }
}
