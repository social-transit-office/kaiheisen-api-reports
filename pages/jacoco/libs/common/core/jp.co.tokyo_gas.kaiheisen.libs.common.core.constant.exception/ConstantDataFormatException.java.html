package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception;

/**
 * コンスタント定義フォーマット不正例外。
 */
public class ConstantDataFormatException extends ConstantDataLoadException {

    /**
     * メッセージを指定してインスタンスを生成する。
     *
     * @param message エラーメッセージ
     */
    public ConstantDataFormatException(String message) {
        super(message);
    }

    /**
     * メッセージと原因例外を指定してインスタンスを生成する。
     *
     * @param message エラーメッセージ
     * @param cause 原因例外
     */
    public ConstantDataFormatException(String message, Throwable cause) {
        super(message, cause);
    }
}
