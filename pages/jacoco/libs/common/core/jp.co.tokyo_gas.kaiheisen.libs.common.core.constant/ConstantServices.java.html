package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant;

import java.time.Clock;
import java.time.ZoneId;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.ExternalCiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.ExternalKaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal.InternalCiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.impl.internal.InternalKaiheisenConstantService;

/**
 * コンスタント取得サービス生成ユーティリティ。
 */
public final class ConstantServices {

    private static final ZoneId JST = ZoneId.of("Asia/Tokyo");

    private ConstantServices() {
    }

    /**
     * 既定設定でサービス群を生成する。
     *
     * @return サービス群
     */
    public static Bundle create() {
        return create(ConstantConfiguration.fromSystemProperties());
    }

    /**
     * 指定設定でサービス群を生成する。
     *
     * @param configuration コンスタント取得設定
     * @return サービス群
     */
    public static Bundle create(ConstantConfiguration configuration) {
        return create(configuration, Clock.system(JST), contextClassLoader());
    }

    static Bundle create(ConstantConfiguration configuration, Clock clock, ClassLoader classLoader) {
        Objects.requireNonNull(configuration, "configuration");
        Objects.requireNonNull(clock, "clock");
        Objects.requireNonNull(classLoader, "classLoader");

        if (configuration.sourceMode() == ConstantSourceMode.EXTERNAL) {
            return new Bundle(new ExternalCiriusConstantService(), new ExternalKaiheisenConstantService());
        }

        return new Bundle(
                new InternalCiriusConstantService(
                        configuration.ciriusInternalResourcePath(),
                        classLoader,
                        clock),
                new InternalKaiheisenConstantService(
                        configuration.kaiheisenInternalResourcePath(),
                        classLoader));
    }

    private static ClassLoader contextClassLoader() {
        ClassLoader classLoader = Thread.currentThread().getContextClassLoader();
        return classLoader != null ? classLoader : ConstantServices.class.getClassLoader();
    }

    /**
     * 生成済みコンスタントサービス群。
     */
    public record Bundle(
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        /**
         * 引数の null チェックを行うコンパクトコンストラクタ。
         */
        public Bundle {
            Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
            Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        }
    }
}
