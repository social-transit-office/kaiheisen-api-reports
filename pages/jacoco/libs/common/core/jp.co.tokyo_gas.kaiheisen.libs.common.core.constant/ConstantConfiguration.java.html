package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant;

import java.util.Objects;

/**
 * コンスタント取得設定。
 */
public record ConstantConfiguration(
        ConstantSourceMode sourceMode,
        String ciriusInternalResourcePath,
        String kaiheisenInternalResourcePath) {

    /** 設定キー: 取得元モード。 */
    public static final String SOURCE_MODE_PROPERTY = "kaiheisen.constant.source-mode";

    /** 設定キー: CIRIUS内部リソースパス。 */
    public static final String CIRIUS_INTERNAL_RESOURCE_PROPERTY =
            "kaiheisen.constant.internal.cirius-resource-path";

    /** 設定キー: 開閉栓内部リソースパス。 */
    public static final String KAIHEISEN_INTERNAL_RESOURCE_PROPERTY =
            "kaiheisen.constant.internal.kaiheisen-resource-path";

    /** 既定値: CIRIUS内部リソースパス。 */
    public static final String DEFAULT_CIRIUS_RESOURCE_PATH = "constants/cirius-constants.tsv";

    /** 既定値: 開閉栓内部リソースパス。 */
    public static final String DEFAULT_KAIHEISEN_RESOURCE_PATH = "constants/kaiheisen-constants.tsv";

    /**
     * 設定値の検証を行うコンパクトコンストラクタ。
     */
    public ConstantConfiguration {
        Objects.requireNonNull(sourceMode, "sourceMode");
        ciriusInternalResourcePath = validatePath(ciriusInternalResourcePath, "ciriusInternalResourcePath");
        kaiheisenInternalResourcePath = validatePath(kaiheisenInternalResourcePath, "kaiheisenInternalResourcePath");
    }

    /**
     * システムプロパティ/環境変数から設定を読み込む。
     *
     * @return 読み込み済み設定
     */
    public static ConstantConfiguration fromSystemProperties() {
        String sourceModeValue = readSetting(SOURCE_MODE_PROPERTY, "KAIHEISEN_CONSTANT_SOURCE_MODE", null);
        String ciriusPath = readSetting(
                CIRIUS_INTERNAL_RESOURCE_PROPERTY,
                "KAIHEISEN_CONSTANT_INTERNAL_CIRIUS_RESOURCE_PATH",
                DEFAULT_CIRIUS_RESOURCE_PATH);
        String kaiheisenPath = readSetting(
                KAIHEISEN_INTERNAL_RESOURCE_PROPERTY,
                "KAIHEISEN_CONSTANT_INTERNAL_KAIHEISEN_RESOURCE_PATH",
                DEFAULT_KAIHEISEN_RESOURCE_PATH);

        return new ConstantConfiguration(ConstantSourceMode.fromValue(sourceModeValue), ciriusPath, kaiheisenPath);
    }

    private static String readSetting(String systemPropertyKey, String environmentKey, String defaultValue) {
        String fromProperty = System.getProperty(systemPropertyKey);
        if (fromProperty != null && !fromProperty.isBlank()) {
            return fromProperty;
        }

        String fromEnv = System.getenv(environmentKey);
        if (fromEnv != null && !fromEnv.isBlank()) {
            return fromEnv;
        }

        return defaultValue;
    }

    private static String validatePath(String value, String fieldName) {
        Objects.requireNonNull(value, fieldName);
        String trimmed = value.trim();
        if (trimmed.isEmpty()) {
            throw new IllegalArgumentException(fieldName + " must not be blank");
        }
        return trimmed;
    }
}
