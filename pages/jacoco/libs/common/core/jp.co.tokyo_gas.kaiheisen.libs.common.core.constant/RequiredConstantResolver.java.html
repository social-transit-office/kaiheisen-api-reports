package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant;

import java.util.Objects;

/**
 * 必須コンスタントの存在検証ユーティリティ。
 */
public final class RequiredConstantResolver {

    private RequiredConstantResolver() {
    }

    /**
     * 概要: 必須 CIRIUS 定数の存在を検証してコード値を返却する。
     * 主な仕様:
     * - CIRIUS 定数サービスで typeCode と code の組み合わせを有効値として検索する。
     * - 検索成功時は入力された code をそのまま返却する。
     * 既知の制約:
     * - typeCode/code の妥当性検証は呼び出し先サービスに委譲する。
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param typeCode 定数種別コード
     * @param code 定数コード
     * @return 検証済みの定数コード
     */
    public static String requireCiriusConstant(
            CiriusConstantService ciriusConstantService,
            String typeCode,
            String code) {
        Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        ciriusConstantService.findActiveByTypeCodeAndCode(typeCode, code);
        return code;
    }

    /**
     * 概要: 必須開閉栓定数の存在を検証して値を返却する。
     * 主な仕様:
     * - 開閉栓定数サービスで type と value の組み合わせを検索する。
     * - 検索成功時は入力された value をそのまま返却する。
     * 既知の制約:
     * - type/value の妥当性検証は呼び出し先サービスに委譲する。
     * @param kaiheisenConstantService 開閉栓コンスタント取得サービス
     * @param type 定数種別
     * @param value 定数値
     * @return 検証済みの定数値
     */
    public static String requireKaiheisenConstant(
            KaiheisenConstantService kaiheisenConstantService,
            String type,
            String value) {
        Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        kaiheisenConstantService.findNameByTypeAndValue(type, value);
        return value;
    }
}
