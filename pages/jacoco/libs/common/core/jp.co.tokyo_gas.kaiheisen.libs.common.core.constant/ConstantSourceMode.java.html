package jp.co.tokyo_gas.kaiheisen.libs.common.core.constant;

import java.util.Locale;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.exception.InvalidConstantSourceModeException;

/**
 * コンスタント取得元モード。
 */
public enum ConstantSourceMode {
    INTERNAL,
    EXTERNAL;

    /**
     * 文字列からモードへ変換する。
     *
     * @param value 設定値
     * @return 取得元モード
     */
    public static ConstantSourceMode fromValue(String value) {
        if (value == null || value.isBlank()) {
            return INTERNAL;
        }
        return switch (value.trim().toUpperCase(Locale.ROOT)) {
            case "INTERNAL" -> INTERNAL;
            case "EXTERNAL" -> EXTERNAL;
            default -> throw new InvalidConstantSourceModeException(value);
        };
    }
}
