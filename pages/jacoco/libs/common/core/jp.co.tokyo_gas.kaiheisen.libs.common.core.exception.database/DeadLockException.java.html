/**
 * @file DeadLockException.java
 * 概要: デッドロックエラーを表す例外を定義する。
 * 主な仕様:
 * - DBエラー情報を保持する。
 * 既知の制限事項:
 * - SQLの秘匿化は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * デッドロックエラーを表す例外クラス。
 * 型: DatabaseException
 * 主な仕様:
 * - DBエラーと同等の情報を保持する。
 * 既知の制限事項:
 * - エラーコードは固定値を利用する。
 * @since 1.0.0
 */
public class DeadLockException extends DatabaseException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_030100001;

    /**
     * コンストラクタ。
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param cause 原因例外。型: Throwable
     */
    public DeadLockException(
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            Throwable cause) {
        this(List.of(), daIdentifier, sql, sqlBindParameters, null, cause);
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    private DeadLockException(
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String detailMessage,
            Throwable cause) {
        super(ERROR_CODE, errorMessageEmbeddedStringList, daIdentifier, sql, sqlBindParameters, detailMessage, cause);
    }
}
