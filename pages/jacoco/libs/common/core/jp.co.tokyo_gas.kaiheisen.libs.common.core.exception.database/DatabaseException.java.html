/**
 * @file DatabaseException.java
 * 概要: データベースエラーを表す例外を定義する。
 * 主な仕様:
 * - DA識別子、SQL、SQLバインドパラメータを保持する。
 * 既知の制限事項:
 * - SQLバインドパラメータの型は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * データベースエラーを表す例外クラス。
 * 型: KaiheisenException
 * 主な仕様:
 * - DA識別子とSQL情報を保持する。
 * 既知の制限事項:
 * - SQLの秘匿化は呼び出し元の責務とする。
 * @since 1.0.0
 */
public class DatabaseException extends KaiheisenException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_030000001;

    /**
     * DA識別子。
     * @type String
     */
    private final String daIdentifier;

    /**
     * SQL。
     * @type String
     */
    private final String sql;

    /**
     * SQLバインドパラメータ。
     * @type Object[]
     */
    private final Object[] sqlBindParameters;

    /**
     * コンストラクタ。
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param cause 原因例外。型: Throwable
     */
    public DatabaseException(
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            Throwable cause) {
        this(daIdentifier, sql, sqlBindParameters, extractDetailMessage(cause), cause);
    }

    /**
     * コンストラクタ。
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected DatabaseException(
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String detailMessage,
            Throwable cause) {
        this(
            ERROR_CODE,
                List.of(),
                daIdentifier,
                sql,
                sqlBindParameters,
                detailMessage,
                cause);
    }

    /**
     * コンストラクタ。
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    protected DatabaseException(
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String detailMessage,
            Throwable cause) {
        this(buildMessage("DatabaseException", detailMessage,
            buildDatabaseAdditionalInfo(daIdentifier, sql, sqlBindParameters),
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList),
                cause, errorCode, errorMessageEmbeddedStringList, daIdentifier, sql, sqlBindParameters);
    }

    /**
     * コンストラクタ。
     * @param message 例外メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     * @param errorCode エラーコード。型: String
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     */
    protected DatabaseException(
            String message,
            Throwable cause,
            String errorCode,
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters) {
        super(
                message,
                cause,
                resolveErrorCode1FromErrorCode(errorCode),
                resolveErrorCode2FromErrorCode(errorCode),
                resolveErrorCode3FromErrorCode(errorCode),
                errorCode,
                errorMessageEmbeddedStringList);
        this.daIdentifier = daIdentifier;
        this.sql = sql;
        this.sqlBindParameters = sqlBindParameters == null ? new Object[0] : sqlBindParameters.clone();
    }

    /**
     * DA識別子を取得する。
     * @return DA識別子。型: String
     */
    public String getDaIdentifier() {
        return daIdentifier;
    }

    /**
     * SQLを取得する。
     * @return SQL。型: String
     */
    public String getSql() {
        return sql;
    }

    /**
     * SQLバインドパラメータを取得する。
     * @return SQLバインドパラメータ。型: Object[]
     */
    public Object[] getSqlBindParameters() {
        return sqlBindParameters.clone();
    }

    /**
     * DB 例外向け追加情報を組み立てる。
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @return 追加情報文字列。型: String
     */
    protected static String buildDatabaseAdditionalInfo(String daIdentifier, String sql, Object[] sqlBindParameters) {
        return "daIdentifier=" + daIdentifier
                + ", sql=" + sql
                + ", sqlBindParameters=" + formatArray(sqlBindParameters);
    }

    private static String extractDetailMessage(Throwable cause) {
        if (cause == null) {
            return null;
        }
        return cause.getMessage();
    }
}
