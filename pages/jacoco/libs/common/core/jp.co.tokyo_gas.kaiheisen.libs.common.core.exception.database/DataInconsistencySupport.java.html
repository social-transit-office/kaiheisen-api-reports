package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.Objects;

/**
 * DataInconsistencyException 生成サポート。
 */
public final class DataInconsistencySupport {

    private DataInconsistencySupport() {
    }

    /**
     * DataInconsistencyException を生成する。
     *
     * @param tableName テーブル名
     * @param daIdentifier DA 識別子
     * @param parameterJsonBuilder パラメータ JSON ビルダー
     * @return DataInconsistencyException
     */
    public static DataInconsistencyException create(
            String tableName,
            String daIdentifier,
            FlatJsonBuilder parameterJsonBuilder) {
        Objects.requireNonNull(parameterJsonBuilder, "parameterJsonBuilder");
        return new DataInconsistencyException(tableName, daIdentifier, parameterJsonBuilder.build());
    }

    /**
     * 原因例外を保持した DataInconsistencyException を生成する。
     *
     * @param tableName テーブル名
     * @param daIdentifier DA 識別子
     * @param parameterJsonBuilder パラメータ JSON ビルダー
     * @param cause 原因例外
     * @return DataInconsistencyException
     */
    public static DataInconsistencyException create(
            String tableName,
            String daIdentifier,
            FlatJsonBuilder parameterJsonBuilder,
            Throwable cause) {
        Objects.requireNonNull(parameterJsonBuilder, "parameterJsonBuilder");
        return new DataInconsistencyException(tableName, daIdentifier, parameterJsonBuilder.build(), cause);
    }
}
