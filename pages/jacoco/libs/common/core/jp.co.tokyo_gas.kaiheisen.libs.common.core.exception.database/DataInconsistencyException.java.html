/**
 * @file DataInconsistencyException.java
 * 概要: データ不整合エラーを表す例外を定義する。
 * 主な仕様:
 * - テーブル名とパラメータJSON文字列を保持する。
 * 既知の制限事項:
 * - パラメータJSON文字列の形式は呼び出し元に依存する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * データ不整合エラーを表す例外クラス。
 * 型: DatabaseException
 * 主な仕様:
 * - テーブル名とパラメータJSON文字列を保持する。
 * 既知の制限事項:
 * - エラーコードは固定値を利用する。
 * @since 1.0.0
 */
public class DataInconsistencyException extends DatabaseException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_030200001;

    /**
     * テーブル名。
     * @type String
     */
    private final String tableName;

    /**
     * パラメータJSON文字列。
     * @type String
     */
    private final String parameterJsonString;

    /**
     * コンストラクタ。
     * @param tableName テーブル名。型: String
     * @param daIdentifier DA識別子。型: String
     * @param parameterJsonString パラメータJSON文字列。型: String
     */
    public DataInconsistencyException(String tableName, String daIdentifier, String parameterJsonString) {
        this(createErrorMessageEmbeddedStringList(tableName, daIdentifier, parameterJsonString),
                daIdentifier, null, new Object[0], tableName, parameterJsonString, null, null);
    }

    /**
     * コンストラクタ。
     * @param tableName テーブル名。型: String
     * @param daIdentifier DA識別子。型: String
     * @param parameterJsonString パラメータJSON文字列。型: String
     * @param cause 原因例外。型: Throwable
     */
    public DataInconsistencyException(
            String tableName,
            String daIdentifier,
            String parameterJsonString,
            Throwable cause) {
        this(createErrorMessageEmbeddedStringList(tableName, daIdentifier, parameterJsonString),
                daIdentifier, null, new Object[0], tableName, parameterJsonString, null, cause);
    }

    private static List<String> createErrorMessageEmbeddedStringList(
            String tableName,
            String daIdentifier,
            String parameterJsonString) {
        return List.of(
                normalizeEmbeddedString(tableName),
                normalizeEmbeddedString(daIdentifier),
                normalizeEmbeddedString(parameterJsonString));
    }

    private static String normalizeEmbeddedString(String embeddedString) {
        return embeddedString == null ? "" : embeddedString;
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト。型: List<String>
     * @param daIdentifier DA識別子。型: String
     * @param sql SQL。型: String
     * @param sqlBindParameters SQLバインドパラメータ。型: Object[]
     * @param tableName テーブル名。型: String
     * @param parameterJsonString パラメータJSON文字列。型: String
     * @param detailMessage 詳細メッセージ。型: String
     * @param cause 原因例外。型: Throwable
     */
    private DataInconsistencyException(
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String tableName,
            String parameterJsonString,
            String detailMessage,
            Throwable cause) {
        super(buildMessage("DataInconsistencyException", detailMessage,
            buildDatabaseAdditionalInfo(daIdentifier, sql, sqlBindParameters)
                        + ", tableName=" + tableName
                        + ", parameterJsonString=" + parameterJsonString,
                resolveErrorCode1FromErrorCode(ERROR_CODE),
                resolveErrorCode2FromErrorCode(ERROR_CODE),
                resolveErrorCode3FromErrorCode(ERROR_CODE),
                ERROR_CODE,
                errorMessageEmbeddedStringList),
                cause,
                ERROR_CODE,
                errorMessageEmbeddedStringList,
                daIdentifier,
                sql,
                sqlBindParameters);
        this.tableName = tableName;
        this.parameterJsonString = parameterJsonString;
    }

    /**
     * テーブル名を取得する。
     * @return テーブル名。型: String
     */
    public String getTableName() {
        return tableName;
    }

    /**
     * パラメータJSON文字列を取得する。
     * @return パラメータJSON文字列。型: String
     */
    public String getParameterJsonString() {
        return parameterJsonString;
    }
}
