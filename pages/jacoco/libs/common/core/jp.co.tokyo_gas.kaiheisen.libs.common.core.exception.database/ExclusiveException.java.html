package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 排他エラーを表す例外クラス。
 */
public class ExclusiveException extends DatabaseException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_030400001;

    /**
     * コンストラクタ。
     *
     * @param daIdentifier DA識別子
     * @param sql SQL
     * @param sqlBindParameters SQLバインドパラメータ
     */
    public ExclusiveException(String daIdentifier, String sql, Object[] sqlBindParameters) {
        this(List.of(), daIdentifier, sql, sqlBindParameters, null, null);
    }

    private ExclusiveException(
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String detailMessage,
            Throwable cause) {
        super(
                ERROR_CODE,
                errorMessageEmbeddedStringList,
                daIdentifier,
                sql,
                sqlBindParameters,
                detailMessage,
                cause);
    }
}
