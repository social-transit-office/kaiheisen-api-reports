package jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.database;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;

/**
 * 主キー重複エラーを表す例外クラス。
 */
public class DuplicateKeyException extends DatabaseException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_030300001;

    /**
     * コンストラクタ。
     * @param daIdentifier DA識別子
     * @param sql SQL
     * @param sqlBindParameters SQLバインドパラメータ
     * @param cause 原因例外
     */
    public DuplicateKeyException(
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            Throwable cause) {
        this(List.of(), daIdentifier, sql, sqlBindParameters, null, cause);
    }

    /**
     * コンストラクタ。
     * @param errorMessageEmbeddedStringList エラーメッセージ埋込み文字列リスト
     * @param daIdentifier DA識別子
     * @param sql SQL
     * @param sqlBindParameters SQLバインドパラメータ
     * @param detailMessage 詳細メッセージ
     * @param cause 原因例外
     */
    private DuplicateKeyException(
            List<String> errorMessageEmbeddedStringList,
            String daIdentifier,
            String sql,
            Object[] sqlBindParameters,
            String detailMessage,
            Throwable cause) {
        super(ERROR_CODE, errorMessageEmbeddedStringList, daIdentifier, sql, sqlBindParameters, detailMessage, cause);
    }
}
