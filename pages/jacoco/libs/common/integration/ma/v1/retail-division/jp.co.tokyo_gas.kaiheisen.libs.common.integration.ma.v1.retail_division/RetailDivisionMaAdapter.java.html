/**
 * @file RetailDivisionMaAdapter.java
 * 概要: MA-107 小売判定 API の adapter を定義する。
 * 主な仕様:
 * - AbstractMaAdapter の自動マッピングにより、コンストラクタのみで完結する。
 * - 例外を MaApiException に変換する。
 * 既知の制限事項:
 * - なし。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.ma.v1.retail_division;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.common.retail_common.retail_division.RetailDivisionEndpointContract;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.common.retail_common.retail_division.RetailDivisionRequest;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.common.retail_common.retail_division.RetailDivisionResponse;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.ma.core.AbstractMaAdapter;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * MA-107 小売判定 API の adapter。
 *
 * <p>{@link RetailDivisionMaRestClient} をラップし、
 * Request record → RestClient メソッド引数の自動マッピングと例外変換を行う。</p>
 */
@ApplicationScoped
public class RetailDivisionMaAdapter
        extends AbstractMaAdapter<RetailDivisionMaRestClient, RetailDivisionRequest, RetailDivisionResponse> {

    /** CDI プロキシ用 no-args コンストラクタ。 */
    protected RetailDivisionMaAdapter() {
        super();
    }

    /**
     * コンストラクタ。
     *
     * @param restClient MA-107 REST クライアント。型: RetailDivisionMaRestClient
     */
    @Inject
    public RetailDivisionMaAdapter(@RestClient RetailDivisionMaRestClient restClient) {
        super(RetailDivisionEndpointContract.INSTANCE, restClient);
    }
}
