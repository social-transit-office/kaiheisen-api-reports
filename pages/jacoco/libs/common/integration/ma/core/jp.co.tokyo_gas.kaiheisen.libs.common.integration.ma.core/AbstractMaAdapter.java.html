/**
 * @file AbstractMaAdapter.java
 * 概要: MA API adapter の抽象基底クラスを定義する。
 * 主な仕様:
 * - MaEndpointContract と REST Client インスタンスを保持する。
 * - RestClientInvoker による Request record → RestClient メソッド引数の自動マッピングを提供する。
 * - サブクラスはコンストラクタのみで完結し、invoke メソッドの手動実装は不要。
 * - 特殊ケース向けに手動 invoke(Supplier, String) も提供する。
 * 既知の制限事項:
 * - 自動マッピングは RestClient interface に非 default メソッドが 1 つの場合のみ対応する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.ma.core;

import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.endpoint.MaEndpointContract;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;

/**
 * MA API adapter の抽象基底クラス。
 *
 * <p>{@link MaEndpointContract} と REST Client インスタンスを保持し、
 * {@link RestClientInvoker} による Request record → RestClient メソッド引数の
 * 自動マッピング付き {@link #invoke(Record)} を提供する。
 * サブクラスはコンストラクタのみで完結し、invoke メソッドの手動実装は不要。</p>
 *
 * <p>特殊ケース（複数メソッド等）向けに手動 {@link #invoke(Supplier, String)} も提供する。</p>
 *
 * @param <C> MicroProfile REST Client インターフェースの型
 * @param <REQ> リクエスト型（Record）
 * @param <RES> レスポンス型
 */
public abstract class AbstractMaAdapter<C, REQ extends Record, RES> {

    /**
     * MA エンドポイント契約。
     */
    private final MaEndpointContract<REQ, RES> contract;

    /**
     * REST Client インスタンス。
     */
    private final C restClient;

    /**
     * Request record → RestClient メソッド引数の自動マッピングユーティリティ。
     */
    private final RestClientInvoker invoker;

    /**
     * CDI プロキシ生成用の no-args コンストラクタ。
     * アプリケーションコードからは使用しないこと。
     */
    protected AbstractMaAdapter() {
        this.contract = null;
        this.restClient = null;
        this.invoker = null;
    }

    /**
     * コンストラクタ。
     *
     * <p>{@link MaEndpointContract} から MA 識別子と型情報を取得し、
     * {@link RestClientInvoker} で自動マッピングを構築する。</p>
     *
     * @param contract MA エンドポイント契約。型: MaEndpointContract
     * @param restClient REST Client インスタンス。型: C
     */
    protected AbstractMaAdapter(MaEndpointContract<REQ, RES> contract, C restClient) {
        this.contract = contract;
        this.restClient = restClient;
        this.invoker = new RestClientInvoker(
                RestClientInvoker.findRestClientInterface(restClient), contract.requestType());
    }

    /**
     * REST Client インスタンスを返す。
     *
     * @return REST Client インスタンス。型: C
     */
    protected C restClient() {
        return restClient;
    }

    /**
     * MA API 呼び出しを実行する（自動マッピング版）。
     *
     * <p>Request record の {@code @QueryParam} 付きコンポーネントを RestClient メソッドの
     * パラメータに自動マッピングし、例外を MaApiException に変換する。</p>
     *
     * @param request リクエスト。型: REQ
     * @return レスポンス。型: RES
     */
    public RES invoke(REQ request) {
        return invoke(
                () -> invoker.invoke(restClient, request),
                RequestInfo.of(contract.path(), request));
    }

    /**
     * MA API 呼び出しを実行し、例外を MaApiException に変換する（手動版）。
     *
     * <p>自動マッピングが不適切な特殊ケース向け。</p>
     *
     * @param <T> レスポンス型
     * @param call 呼び出し処理。型: Supplier
     * @param requestInfo リクエスト情報（パラメータ等）。型: String
     * @return 呼び出し結果。型: T
     */
    protected <T> T invoke(Supplier<T> call, String requestInfo) {
        return MaRestClientSupport.invoke(call, contract.maId(), requestInfo);
    }
}
