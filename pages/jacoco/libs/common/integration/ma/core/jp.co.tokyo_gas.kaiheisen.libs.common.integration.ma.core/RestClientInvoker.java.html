/**
 * @file RestClientInvoker.java
 * 概要: RestClient interface のメソッドと Request record を自動マッピングするユーティリティ。
 * 主な仕様:
 * - コンストラクタ時に RestClient interface の非 default 単一メソッドを特定し、マッピングを構築する。
 * - @BeanParam 1引数パターン: Request record をそのまま渡す（推奨）。
 * - @QueryParam 複数引数パターン: Request record の @QueryParam コンポーネントを各引数へ自動マッピングする（レガシー互換）。
 * - invoke 時にキャッシュ済みマッピングで Request record → RestClient メソッド引数を自動変換する。
 * - findRestClientInterface で CDI proxy / テストスタブから @RegisterRestClient 付き interface を探索する。
 * 既知の制限事項:
 * - RestClient interface に非 default メソッドが複数ある場合は IllegalStateException をスローする。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.ma.core;

import jakarta.ws.rs.BeanParam;
import jakarta.ws.rs.QueryParam;
import java.lang.reflect.Method;
import java.lang.reflect.Parameter;
import java.lang.reflect.RecordComponent;
import java.util.Arrays;
import java.util.Map;
import java.util.stream.Collectors;
import org.eclipse.microprofile.rest.client.inject.RegisterRestClient;

/**
 * RestClient interface のメソッドと Request record を自動マッピングするユーティリティ。
 *
 * <p>2つのパターンをサポートする:</p>
 * <ul>
 *   <li>{@code @BeanParam} 1引数パターン（推奨）: Request record をそのまま渡す</li>
 *   <li>{@code @QueryParam} 複数引数パターン（レガシー互換）: Request record の
 *       {@code @QueryParam} コンポーネントを各引数へ名前マッチで自動マッピングする</li>
 * </ul>
 *
 * <p>コンストラクタ時にマッピングを解決し、{@link #invoke} 時に
 * キャッシュ済みマッピングで Request record から RestClient メソッド引数を自動構築する。</p>
 *
 * <p>package-private: {@link AbstractMaAdapter} 内部でのみ使用する。</p>
 */
final class RestClientInvoker {

    /**
     * 解決済みマッピング。
     *
 * <p>{@code accessors} が {@code null} の場合は {@code @BeanParam} 1引数パターンを示し、
     * request をそのまま渡す。{@code null} でない場合は {@code @QueryParam} 複数引数パターンを示し、
     * method のパラメータ順に並んだアクセサで引数を構築する。</p>
     *
     * @param method RestClient interface のメソッド
     * @param accessors Request record のアクセサメソッド（@BeanParam パターンでは null）
     */
    private record Mapping(Method method, Method[] accessors) {
    }

    /**
     * キャッシュ済みマッピング。
     */
    private final Mapping mapping;

    /**
     * コンストラクタ。
     *
     * <p>RestClient interface から非 default 単一メソッドを特定し、
     * パラメータが単一 {@link BeanParam} の場合は request をそのまま渡すマッピングを、
     * それ以外の場合は Request record の {@link QueryParam} コンポーネントとパラメータ名で
     * マッピングを構築する。
     * 不一致がある場合は {@link IllegalStateException} をスローする（フェイルファスト）。</p>
     *
     * @param restClientInterface RestClient interface の {@link Class}。型: Class
     * @param requestType Request record の {@link Class}。型: Class
     */
    RestClientInvoker(Class<?> restClientInterface, Class<? extends Record> requestType) {
        this.mapping = resolve(restClientInterface, requestType);
    }

    /**
     * キャッシュ済みマッピングで RestClient メソッドを呼び出す。
     *
     * @param <RES> レスポンス型
     * @param restClient RestClient インスタンス。型: Object
     * @param request Request record インスタンス。型: Record
     * @return RestClient メソッドの戻り値。型: RES
     */
    @SuppressWarnings("unchecked")
    <RES> RES invoke(Object restClient, Record request) {
        // @BeanParam 1引数パターン: request をそのまま渡す
        if (mapping.accessors == null) {
            try {
                return (RES) mapping.method.invoke(restClient, request);
            } catch (ReflectiveOperationException e) {
                Throwable cause = e.getCause() != null ? e.getCause() : e;
                if (cause instanceof RuntimeException re) {
                    throw re;
                }
                throw new IllegalStateException("RestClient メソッド呼び出しに失敗", cause);
            }
        }
        // @QueryParam 複数引数パターン: accessor で引数を構築
        Object[] args = new Object[mapping.accessors.length];
        for (int i = 0; i < args.length; i++) {
            try {
                args[i] = mapping.accessors[i].invoke(request);
            } catch (ReflectiveOperationException e) {
                throw new IllegalStateException(
                        "Request record コンポーネントのアクセスに失敗: " + mapping.accessors[i].getName(), e);
            }
        }
        try {
            return (RES) mapping.method.invoke(restClient, args);
        } catch (ReflectiveOperationException e) {
            Throwable cause = e.getCause() != null ? e.getCause() : e;
            if (cause instanceof RuntimeException re) {
                throw re;
            }
            throw new IllegalStateException("RestClient メソッド呼び出しに失敗", cause);
        }
    }

    /**
     * RestClient proxy / テストスタブから {@link RegisterRestClient} 付き interface を探索する。
     *
     * <p>CDI proxy / MicroProfile REST Client proxy の場合は {@code getInterfaces()} に
     * {@link RegisterRestClient} 付き interface が含まれる。
     * テスト時の直接実装の場合も同様に探索する。</p>
     *
     * @param restClient RestClient インスタンス。型: Object
     * @return {@link RegisterRestClient} 付き interface の {@link Class}。型: Class
     */
    static Class<?> findRestClientInterface(Object restClient) {
        Class<?> clazz = restClient.getClass();
        while (clazz != null) {
            for (Class<?> iface : clazz.getInterfaces()) {
                if (iface.isAnnotationPresent(RegisterRestClient.class)) {
                    return iface;
                }
            }
            clazz = clazz.getSuperclass();
        }
        throw new IllegalStateException(
                "@RegisterRestClient 付き interface が見つかりません: " + restClient.getClass().getName());
    }

    /**
     * RestClient interface のメソッドと Request record のマッピングを解決する。
     *
     * @param restClientInterface RestClient interface の {@link Class}。型: Class
     * @param requestType Request record の {@link Class}。型: Class
     * @return 解決済みマッピング。型: Mapping
     */
    private static Mapping resolve(Class<?> restClientInterface, Class<? extends Record> requestType) {
        Method targetMethod = findSingleEndpointMethod(restClientInterface);

        // 単一引数パターン: JSON body / @BeanParam ともに request をそのまま渡す
        Parameter[] params = targetMethod.getParameters();
        if (params.length == 1) {
            return new Mapping(targetMethod, null);
        }

        // @QueryParam 複数引数パターン: Request record の RecordComponent から @QueryParam の value → accessor の Map を構築
        Map<String, Method> queryParamAccessors = buildQueryParamAccessorMap(requestType);

        // RestClient メソッドのパラメータ順に accessor を並べる
        Method[] accessors = new Method[params.length];
        for (int i = 0; i < params.length; i++) {
            QueryParam qp = params[i].getAnnotation(QueryParam.class);
            if (qp == null) {
                throw new IllegalStateException(
                        "RestClient メソッドのパラメータに @QueryParam がありません: "
                                + restClientInterface.getSimpleName() + "#"
                                + targetMethod.getName() + " パラメータ[" + i + "]");
            }
            String queryParamName = qp.value();
            Method accessor = queryParamAccessors.get(queryParamName);
            if (accessor == null) {
                throw new IllegalStateException(
                        "Request record に @QueryParam(\"" + queryParamName + "\") が見つかりません: "
                                + requestType.getSimpleName()
                                + " (RestClient: " + restClientInterface.getSimpleName() + "#"
                                + targetMethod.getName() + ")");
            }
            accessors[i] = accessor;
        }

        return new Mapping(targetMethod, accessors);
    }

    /**
     * RestClient interface から非 default・非 static な単一メソッドを特定する。
     *
     * @param restClientInterface RestClient interface の {@link Class}。型: Class
     * @return 対象メソッド。型: Method
     */
    private static Method findSingleEndpointMethod(Class<?> restClientInterface) {
        Method[] candidates = Arrays.stream(restClientInterface.getMethods())
                .filter(m -> !m.isDefault() && !java.lang.reflect.Modifier.isStatic(m.getModifiers()))
                .filter(m -> m.getDeclaringClass() == restClientInterface)
                .toArray(Method[]::new);

        if (candidates.length == 0) {
            throw new IllegalStateException(
                    "RestClient interface にエンドポイントメソッドがありません: "
                            + restClientInterface.getSimpleName());
        }
        if (candidates.length > 1) {
            throw new IllegalStateException(
                    "RestClient interface に複数のエンドポイントメソッドがあります（自動マッピングは単一メソッドのみ対応）: "
                            + restClientInterface.getSimpleName());
        }
        return candidates[0];
    }

    /**
     * Request record の {@link RecordComponent} から {@link QueryParam} の value → accessor メソッド の Map を構築する。
     *
     * @param requestType Request record の {@link Class}。型: Class
     * @return {@link QueryParam} value → accessor の Map。型: Map
     */
    private static Map<String, Method> buildQueryParamAccessorMap(Class<? extends Record> requestType) {
        return Arrays.stream(requestType.getRecordComponents())
                .filter(rc -> rc.getAccessor().isAnnotationPresent(QueryParam.class))
                .collect(Collectors.toMap(
                        rc -> rc.getAccessor().getAnnotation(QueryParam.class).value(),
                        RecordComponent::getAccessor));
    }
}
