package jp.co.tokyo_gas.kaiheisen.libs.common.integration.livalit.core;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.ExternalSystemException;

/**
 * LIVALIT API エラーを表す例外。
 */
public class LivalitApiException extends ExternalSystemException {

    public static final String ERROR_CODE = ErrorCodeConstants.ERROR_CODE_040000001;

    public static final String EXTERNAL_SYSTEM_NAME = "LIVALIT API";

    private final String requestInfo;

    private final String responseInfo;

    /**
     * LIVALIT API 呼び出し失敗時の例外を生成する。
     *
     * @param requestInfo リクエスト情報
     * @param responseInfo レスポンス情報
     * @param cause 原因例外
     */
    public LivalitApiException(String requestInfo, String responseInfo, Throwable cause) {
        this(List.of(), requestInfo, responseInfo, null, cause);
    }

    private LivalitApiException(
            List<String> errorMessageEmbeddedStringList,
            String requestInfo,
            String responseInfo,
            String detailMessage,
            Throwable cause) {
        super(
                "LivalitApiException",
                ERROR_CODE,
                errorMessageEmbeddedStringList,
                EXTERNAL_SYSTEM_NAME,
                "requestInfo=" + requestInfo + ", responseInfo=" + responseInfo,
                detailMessage,
                cause);
        this.requestInfo = requestInfo;
        this.responseInfo = responseInfo;
    }

    /**
     * 失敗時のリクエスト情報を返す。
     *
     * @return リクエスト情報
     */
    public String getRequestInfo() {
        return requestInfo;
    }

    /**
     * 失敗時のレスポンス情報を返す。
     *
     * @return レスポンス情報
     */
    public String getResponseInfo() {
        return responseInfo;
    }
}
