package jp.co.tokyo_gas.kaiheisen.libs.common.integration.livalit.core;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.core.HttpHeaders;
import jakarta.ws.rs.core.MultivaluedMap;
import java.util.NoSuchElementException;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.BasicAuthenticationHeader;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.CorrelationIdClientHeadersFactory;
import org.eclipse.microprofile.config.ConfigProvider;
import org.eclipse.microprofile.rest.client.ext.ClientHeadersFactory;

/**
 * LIVALIT API 呼び出しに Basic 認証と相関IDのヘッダを付与するファクトリ。
 */
@ApplicationScoped
public class LivalitBasicAuthClientHeadersFactory implements ClientHeadersFactory {

    static final String BASIC_AUTH_USERNAME_PROPERTY =
            "kaiheisen.livalit.retail-v1.basic-auth.username";

    static final String BASIC_AUTH_PASSWORD_PROPERTY =
            "kaiheisen.livalit.retail-v1.basic-auth.password";

    private final CorrelationIdClientHeadersFactory correlationIdClientHeadersFactory;

    /**
     * コンストラクタ。
     *
     * @param correlationIdClientHeadersFactory 相関IDヘッダーファクトリ
     */
    @Inject
    public LivalitBasicAuthClientHeadersFactory(
            CorrelationIdClientHeadersFactory correlationIdClientHeadersFactory) {
        this.correlationIdClientHeadersFactory = correlationIdClientHeadersFactory;
    }

    /**
     * 送信ヘッダに Basic 認証と相関IDを付与する。
     *
     * @param incomingHeaders 受信ヘッダ
     * @param clientOutgoingHeaders 送信ヘッダ
     * @return 更新後の送信ヘッダ
     */
    @Override
    public MultivaluedMap<String, String> update(
            MultivaluedMap<String, String> incomingHeaders,
            MultivaluedMap<String, String> clientOutgoingHeaders) {
        MultivaluedMap<String, String> headers =
                correlationIdClientHeadersFactory.update(incomingHeaders, clientOutgoingHeaders);
        headers.putSingle(
                HttpHeaders.AUTHORIZATION,
                BasicAuthenticationHeader.create(
                        configuredValue(BASIC_AUTH_USERNAME_PROPERTY),
                        configuredValue(BASIC_AUTH_PASSWORD_PROPERTY)));
        return headers;
    }

    private static String configuredValue(String propertyName) {
        try {
            return ConfigProvider.getConfig()
                    .getOptionalValue(propertyName, String.class)
                    .filter(value -> !value.isBlank())
                    .orElseThrow(() -> missingProperty(propertyName));
        } catch (NoSuchElementException ignored) {
            throw missingProperty(propertyName);
        }
    }

    private static IllegalStateException missingProperty(String propertyName) {
        return new IllegalStateException(
                "Required LIVALIT Basic authentication property is not configured: "
                        + propertyName);
    }
}
