package jp.co.tokyo_gas.kaiheisen.libs.common.integration.livalit.core;

import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.MpRestClientSupport;

/**
 * LIVALIT API 呼び出し固有の例外変換ヘルパー。
 */
public final class LivalitRestClientSupport {

    private LivalitRestClientSupport() {
    }

    /**
     * LIVALIT API 呼び出しを実行し、共通例外へ変換する。
     *
     * @param call 実行する REST 呼び出し
     * @param apiName API 名
     * @param requestInfo リクエスト情報
     * @param <T> レスポンス型
     * @return API 呼び出し結果
     */
    public static <T> T invoke(Supplier<T> call, String apiName, String requestInfo) {
        String fullRequestInfo = "apiName=" + apiName + ", " + requestInfo;
        return MpRestClientSupport.invoke(call, apiName, fullRequestInfo, LivalitApiException::new);
    }
}
