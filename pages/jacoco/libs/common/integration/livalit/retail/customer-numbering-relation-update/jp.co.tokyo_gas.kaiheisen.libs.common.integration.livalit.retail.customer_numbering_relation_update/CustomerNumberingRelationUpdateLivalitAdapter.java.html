package jp.co.tokyo_gas.kaiheisen.libs.common.integration.livalit.retail.customer_numbering_relation_update;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.livalit.core.AbstractLivalitAdapter;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * LIVALIT カスタマ番号対応表更新 API（他-028）の adapter。
 */
@ApplicationScoped
public class CustomerNumberingRelationUpdateLivalitAdapter
        extends AbstractLivalitAdapter<CustomerNumberingRelationUpdateLivalitRestClient> {

    protected CustomerNumberingRelationUpdateLivalitAdapter() {
        super();
    }

    /**
     * REST クライアントを受け取って adapter を生成する。
     *
     * @param restClient LIVALIT REST クライアント
     */
    @Inject
    public CustomerNumberingRelationUpdateLivalitAdapter(
            @RestClient CustomerNumberingRelationUpdateLivalitRestClient restClient) {
        super("他-028", restClient);
    }

    /**
     * LIVALIT カスタマ番号対応表更新 API を呼び出す。
     *
     * @param request 更新リクエスト
     * @return 更新結果
     */
    public CustomerNumberingRelationUpdateResponse update(CustomerNumberingRelationUpdateRequest request) {
        return invoke(
                () -> restClient().update(request),
                RequestInfo.of(
                        "path",
                        CustomerNumberingRelationUpdateLivalitRestClient.PATH,
                        "userId",
                        String.valueOf(request == null ? null : request.userId()),
                        "gmtSetNo",
                        String.valueOf(request == null ? null : request.gmtSetNo()),
                        "sinStartYmd",
                        String.valueOf(request == null ? null : request.sinStartYmd()),
                        "kyuEndYmd",
                        String.valueOf(request == null ? null : request.kyuEndYmd()),
                        "dgwriUmuFlg",
                        String.valueOf(request == null ? null : request.dgwriUmuFlg()),
                        "skjFlg",
                        String.valueOf(request == null ? null : request.skjFlg()),
                        "custmrNo",
                        String.valueOf(request == null ? null : request.custmrNo())));
    }
}
