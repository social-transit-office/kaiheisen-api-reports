/**
 * @file CiriusRetailV2ClientHeadersFactory.java
 * 概要: CIRIUS 小売 API v2 呼び出しに必要な共通ヘッダを付与する。
 * 主な仕様:
 * - RequestContext から correlationId を取得し X-Correlation-Id ヘッダへ設定する。
 * - 設定値から Version と Ocp-Apim-Subscription-Key ヘッダを設定する。
 * 既知の制限事項:
 * - 必須設定が未設定の場合は呼び出し時に例外を送出する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.core.MultivaluedMap;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import org.eclipse.microprofile.config.inject.ConfigProperty;
import org.eclipse.microprofile.rest.client.ext.ClientHeadersFactory;

/**
 * CIRIUS 小売 API v2 呼び出しに必要な共通ヘッダを付与するファクトリ。
 *
 * <p>相関IDに加えて、CIRIUS/APIM で必須となる {@code Version} と
 * {@code Ocp-Apim-Subscription-Key} を送信ヘッダへ設定する。</p>
 */
@ApplicationScoped
public class CiriusRetailV2ClientHeadersFactory implements ClientHeadersFactory {

    static final String CORRELATION_ID_HEADER_NAME = "X-Correlation-Id";

    static final String VERSION_HEADER_NAME = "Version";

    static final String SUBSCRIPTION_KEY_HEADER_NAME = "Ocp-Apim-Subscription-Key";

    static final String VERSION_PROPERTY_NAME = "kaiheisen.cirius.retail-v2.version";

    static final String SUBSCRIPTION_KEY_PROPERTY_NAME = "kaiheisen.cirius.retail-v2.subscription-key";

    private final RequestContext requestContext;

    private final String version;

    private final String subscriptionKey;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト。型: RequestContext
     * @param version CIRIUS 小売 API v2 の Version ヘッダ値。型: String
     * @param subscriptionKey CIRIUS 小売 API v2 のサブスクリプションキー。型: String
     */
    @Inject
    public CiriusRetailV2ClientHeadersFactory(
            RequestContext requestContext,
            @ConfigProperty(name = VERSION_PROPERTY_NAME, defaultValue = "") String version,
            @ConfigProperty(name = SUBSCRIPTION_KEY_PROPERTY_NAME, defaultValue = "") String subscriptionKey) {
        this.requestContext = requestContext;
        this.version = version;
        this.subscriptionKey = subscriptionKey;
    }

    /**
     * 送信ヘッダを更新し、CIRIUS 小売 API v2 に必要なヘッダを付与する。
     *
     * @param incomingHeaders 受信ヘッダ。型: MultivaluedMap
     * @param clientOutgoingHeaders 送信ヘッダ。型: MultivaluedMap
     * @return 更新後の送信ヘッダ。型: MultivaluedMap
     */
    @Override
    public MultivaluedMap<String, String> update(
            MultivaluedMap<String, String> incomingHeaders,
            MultivaluedMap<String, String> clientOutgoingHeaders) {
        requestContext.getCorrelationId().ifPresent(correlationId ->
                clientOutgoingHeaders.putSingle(CORRELATION_ID_HEADER_NAME, correlationId));
        clientOutgoingHeaders.putSingle(
                VERSION_HEADER_NAME, requireConfigured(VERSION_PROPERTY_NAME, version));
        clientOutgoingHeaders.putSingle(
                SUBSCRIPTION_KEY_HEADER_NAME,
                requireConfigured(SUBSCRIPTION_KEY_PROPERTY_NAME, subscriptionKey));
        return clientOutgoingHeaders;
    }

    private static String requireConfigured(String propertyName, String value) {
        if (value == null || value.isBlank()) {
            throw new IllegalStateException(
                    "Required CIRIUS retail v2 header property is not configured: " + propertyName);
        }
        return value.trim();
    }
}
