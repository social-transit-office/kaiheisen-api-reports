/**
 * @file CiriusRestClientSupport.java
 * 概要: CIRIUS API 呼び出し固有の例外変換ヘルパーを定義する。
 * 主な仕様:
 * - MpRestClientSupport を利用し、例外を CiriusApiException に変換する。
 * - requestInfo に CIRIUS API 名とリクエスト情報を含める。
 * 既知の制限事項:
 * - なし。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core;

import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.CiriusApiException;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.MpRestClientSupport;

/**
 * CIRIUS API 呼び出し固有の例外変換ヘルパー。
 *
 * <p>{@link MpRestClientSupport} を利用し、
 * MP Rest Client 呼び出しの例外を {@link CiriusApiException} に変換する。</p>
 */
public final class CiriusRestClientSupport {

    private CiriusRestClientSupport() {
    }

    /**
     * CIRIUS API 呼び出しを実行し、例外を {@link CiriusApiException} に変換する。
     *
     * @param <T> レスポンス型
     * @param call 呼び出し処理。型: Supplier
     * @param apiName CIRIUS API 名（例: "他-014"）。型: String
     * @param requestInfo リクエスト情報（パラメータ等）。型: String
     * @return 呼び出し結果。型: T
     */
    public static <T> T invoke(Supplier<T> call, String apiName, String requestInfo) {
        String fullRequestInfo = "apiName=" + apiName + ", " + requestInfo;
        return MpRestClientSupport.invoke(call, apiName, fullRequestInfo, CiriusApiException::new);
    }
}
