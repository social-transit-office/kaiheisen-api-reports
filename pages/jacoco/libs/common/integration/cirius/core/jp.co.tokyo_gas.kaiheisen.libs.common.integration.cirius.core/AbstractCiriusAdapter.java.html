/**
 * @file AbstractCiriusAdapter.java
 * 概要: CIRIUS API adapter の抽象基底クラスを定義する。
 * 主な仕様:
 * - REST Client インスタンスと CIRIUS API 名を保持する。
 * - CiriusRestClientSupport への委譲を invoke メソッドで提供する。
 * - サブクラスは restClient() と invoke() を使って adapter メソッドを実装する。
 * 既知の制限事項:
 * - なし。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core;

import java.util.function.Supplier;

/**
 * CIRIUS API adapter の抽象基底クラス。
 *
 * <p>REST Client インスタンスと CIRIUS API 名を保持し、
 * {@link CiriusRestClientSupport#invoke} への委譲を提供する。
 * サブクラスは {@link #restClient()} で REST Client を取得し、
 * {@link #invoke(Supplier, String)} で例外変換付き呼び出しを行う。</p>
 *
 * @param <C> MicroProfile REST Client インターフェースの型
 */
public abstract class AbstractCiriusAdapter<C> {

    /**
     * CIRIUS API 名。
     */
    private final String apiName;

    /**
     * REST Client インスタンス。
     */
    private final C restClient;

    /**
     * CDI プロキシ生成用の no-args コンストラクタ。
     * アプリケーションコードからは使用しないこと。
     */
    protected AbstractCiriusAdapter() {
        this.apiName = null;
        this.restClient = null;
    }

    /**
     * コンストラクタ。
     *
     * @param apiName CIRIUS API 名（例: "他-014"）。型: String
     * @param restClient REST Client インスタンス。型: C
     */
    protected AbstractCiriusAdapter(String apiName, C restClient) {
        this.apiName = apiName;
        this.restClient = restClient;
    }

    /**
     * REST Client インスタンスを返す。
     *
     * @return REST Client インスタンス。型: C
     */
    protected C restClient() {
        return restClient;
    }

    /**
     * CIRIUS API 呼び出しを実行し、例外を CiriusApiException に変換する。
     *
     * @param <T> レスポンス型
     * @param call 呼び出し処理。型: Supplier
     * @param requestInfo リクエスト情報（パラメータ等）。型: String
     * @return 呼び出し結果。型: T
     */
    protected <T> T invoke(Supplier<T> call, String requestInfo) {
        return CiriusRestClientSupport.invoke(call, apiName, requestInfo);
    }
}
