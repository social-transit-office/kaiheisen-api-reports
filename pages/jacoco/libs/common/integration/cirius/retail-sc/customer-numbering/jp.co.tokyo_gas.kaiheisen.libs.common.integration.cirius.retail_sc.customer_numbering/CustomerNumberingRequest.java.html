package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering;

import com.fasterxml.jackson.annotation.JsonInclude;
import java.math.BigDecimal;

/**
 * CIRIUS 顧客番号採番 API（他-048）のリクエスト.
 *
 * @param hatsubanJouhouShutokuJouken 発番情報取得条件
 */
@JsonInclude(JsonInclude.Include.NON_NULL)
public record CustomerNumberingRequest(
        NumberingCondition hatsubanJouhouShutokuJouken) {

    /**
     * 契約お客さま登録番号_導管採番リクエストを返す.
     *
     * @return 契約お客さま登録番号_導管の採番リクエスト
     */
    public static CustomerNumberingRequest forPipelineContractCustomerRegistrationNumber() {
        return new CustomerNumberingRequest(
                new NumberingCondition(
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ONE,
                        BigDecimal.ZERO));
    }

    /**
     * SV-221/SV-222 で利用する契約お客さま登録番号_小売採番リクエストを返す。
     *
     * @return 契約お客さま登録番号_小売の採番リクエスト
     */
    public static CustomerNumberingRequest forRetailContractCustomerRegistrationNumber() {
        return new CustomerNumberingRequest(
                new NumberingCondition(
                        BigDecimal.ZERO,
                        BigDecimal.ONE,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO));
    }

    /**
     * SV-217 で利用する託送サービス契約番号採番リクエストを返す。
     *
     * @return 託送サービス契約番号の採番リクエスト
     */
    public static CustomerNumberingRequest forTransportServiceContractNumber() {
        return new CustomerNumberingRequest(
                new NumberingCondition(
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ZERO,
                        BigDecimal.ONE));
    }

    /**
     * 発番情報取得条件。
     *
     * @param sihKiyNoHbks 支払契約番号発番回数
     * @param kyoTrnoKurHbks 契約お客さま登録番号_小売発番回数
     * @param svkyNoHbks サービス供給番号発番回数
     * @param sikKrnoHbks 使用契約番号発番回数
     * @param sqtiKiyNoHbks 需要家機器番号発番回数
     * @param kyoTrnoDuknHbks 契約お客さま登録番号_導管発番回数
     * @param tksuSvkyNoHbks 託送サービス供給番号発番回数
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record NumberingCondition(
            BigDecimal sihKiyNoHbks,
            BigDecimal kyoTrnoKurHbks,
            BigDecimal svkyNoHbks,
            BigDecimal sikKrnoHbks,
            BigDecimal sqtiKiyNoHbks,
            BigDecimal kyoTrnoDuknHbks,
            BigDecimal tksuSvkyNoHbks) {
    }
}
