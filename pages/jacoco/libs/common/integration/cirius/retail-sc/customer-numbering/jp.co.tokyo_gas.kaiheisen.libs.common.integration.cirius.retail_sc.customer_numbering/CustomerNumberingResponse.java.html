package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering;

import com.fasterxml.jackson.annotation.JsonAlias;
import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonInclude;
import com.fasterxml.jackson.annotation.JsonProperty;
import java.util.List;

/**
 * CIRIUS 顧客番号採番 API（他-048）のレスポンス。
 *
 * <p>外部戻り値仕様 TSV が未整備のため、関連サービスで利用する採番結果とエラー情報のみを保持する。</p>
 */
@JsonInclude(JsonInclude.Include.NON_NULL)
@JsonIgnoreProperties(ignoreUnknown = true)
public record CustomerNumberingResponse(
        @JsonProperty("sihKiyNo")
        @JsonAlias("sih_kiy_no")
        Long sihKiyNo,
        @JsonProperty("kyoTrnoKur")
        @JsonAlias("kyo_trno_kur")
        Long kyoTrnoKur,
        @JsonProperty("svkyNo")
        @JsonAlias("svky_no")
        Long svkyNo,
        @JsonProperty("sikKrno")
        @JsonAlias("sik_krno")
        Long sikKrno,
        @JsonProperty("sqtiKiyNo")
        @JsonAlias("sqti_kiy_no")
        String sqtiKiyNo,
        @JsonProperty("kyoTrnoDukn")
        @JsonAlias("kyo_trno_dukn")
        Long kyoTrnoDukn,
        @JsonProperty("tksuSvkyNo")
        @JsonAlias("tksu_svky_no")
        Long tksuSvkyNo,
        @JsonProperty("errUmFlg")
        String errUmFlg,
        @JsonProperty("errorJouhou")
        List<ErrorJouhou> errorJouhou) {

    /**
     * 概要: 契約お客さま登録番号_導管を返す.
     * 主な仕様: トップレベルの契約お客さま登録番号_導管を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 契約お客さま登録番号_導管
     */
    public Long pipelineContractCustomerRegistrationNumber() {
        return kyoTrnoDukn;
    }

    /**
     * 概要: 契約お客さま登録番号_小売を返す.
     * 主な仕様: トップレベルの契約お客さま登録番号_小売を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 契約お客さま登録番号_小売
     */
    public Long retailContractCustomerRegistrationNumber() {
        return kyoTrnoKur;
    }

    /**
     * 概要: 託送サービス契約番号を返す。
     * 主な仕様: トップレベルの託送サービス契約番号を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 託送サービス契約番号
     */
    public Long transportServiceContractNumber() {
        return tksuSvkyNo;
    }

    /**
     * 概要: サービス契約番号を返す。
     * 主な仕様: トップレベルのサービス契約番号を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return サービス契約番号
     */
    public Long serviceContractNumber() {
        return svkyNo;
    }

    /**
     * 概要: 支払契約番号を返す。
     * 主な仕様: トップレベルの支払契約番号を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 支払契約番号
     */
    public Long paymentContractNumber() {
        return sihKiyNo;
    }

    /**
     * 概要: 請求対象契約番号を返す。
     * 主な仕様: トップレベルの請求対象契約番号を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 請求対象契約番号
     */
    public String billingTargetContractNumber() {
        return sqtiKiyNo;
    }

    /**
     * 概要: 請求管理番号を返す。
     * 主な仕様: トップレベルの請求管理番号を返す。
     * 既知の制約: 他-048 応答に対象項目が存在しない場合は `null` を返す。
     *
     * @return 請求管理番号
     */
    public Long billingManagementNumber() {
        return sikKrno;
    }

    /**
     * 概要: エラー情報。
     * 主な仕様: 他-048 のエラー応答で返却されるメッセージ情報を保持する。
     * 既知の制約: 未知項目は無視される。
     *
     * @param msgId メッセージ ID
     * @param umkmTxt 埋め込みテキスト
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record ErrorJouhou(
            @JsonProperty("msgId")
            String msgId,
            @JsonProperty("umkmTxt")
            String umkmTxt) {
    }
}
