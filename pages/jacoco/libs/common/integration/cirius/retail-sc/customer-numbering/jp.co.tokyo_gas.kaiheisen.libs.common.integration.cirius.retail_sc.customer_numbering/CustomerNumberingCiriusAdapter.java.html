package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * CIRIUS 顧客番号採番 API（他-048）の adapter.
 */
@ApplicationScoped
public class CustomerNumberingCiriusAdapter
        extends AbstractCiriusAdapter<CustomerNumberingCiriusRestClient> {

    /** CDI プロキシ用コンストラクタ. */
    protected CustomerNumberingCiriusAdapter() {
        super();
    }

    /**
     * コンストラクタ.
     *
     * @param restClient CIRIUS REST クライアント
     */
    @Inject
    public CustomerNumberingCiriusAdapter(
            @RestClient final CustomerNumberingCiriusRestClient restClient) {
        super("他-048", restClient);
    }

    /**
     * 顧客番号採番 API を呼び出す.
     *
     * @param request 採番リクエスト
     * @return 採番レスポンス
     */
    public CustomerNumberingResponse number(
            final CustomerNumberingRequest request) {
        return number(request == null ? null : request.hatsubanJouhouShutokuJouken());
    }

    /**
     * 顧客番号採番 API を採番条件直接指定で呼び出す.
     *
     * @param condition 採番条件
     * @return 採番レスポンス
     */
    public CustomerNumberingResponse number(
            final CustomerNumberingRequest.NumberingCondition condition) {
        return invoke(
                () -> restClient().number(condition),
                RequestInfo.of(
                        "path", "/customer-numbering",
                        "sih_kiy_no_hbks", countValue(condition == null ? null : condition.sihKiyNoHbks()),
                        "kyo_trno_kur_hbks", countValue(condition == null ? null : condition.kyoTrnoKurHbks()),
                        "svky_no_hbks", countValue(condition == null ? null : condition.svkyNoHbks()),
                        "sik_krno_hbks", countValue(condition == null ? null : condition.sikKrnoHbks()),
                        "sqti_kiy_no_hbks", countValue(condition == null ? null : condition.sqtiKiyNoHbks()),
                        "kyo_trno_dukn_hbks", countValue(condition == null ? null : condition.kyoTrnoDuknHbks()),
                        "tksu_svky_no_hbks", countValue(condition == null ? null : condition.tksuSvkyNoHbks())));
    }

    private static String countValue(Object value) {
        return value == null ? "" : String.valueOf(value);
    }
}
