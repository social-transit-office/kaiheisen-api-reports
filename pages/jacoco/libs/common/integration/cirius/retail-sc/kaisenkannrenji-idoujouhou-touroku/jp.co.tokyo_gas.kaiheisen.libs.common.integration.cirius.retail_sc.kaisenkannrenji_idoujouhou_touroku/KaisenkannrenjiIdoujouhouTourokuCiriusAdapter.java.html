package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisenkannrenji_idoujouhou_touroku;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * 概要: CIRIUS 開栓関連時異動情報登録 API（他-044）の adapter。
 *
 * 主な仕様:
 * - 外部API（他-044）へのリクエスト送信・例外変換を担う。
 * - 例外発生時はCiriusApiExceptionへ変換し、RequestInfoを付与する。
 *   - RequestInfoにはpath等の非秘匿トレース項目のみを含め、契約番号等のPII/秘匿情報は含めない。
 *
 * 既知の制約:
 * - 外部API仕様変更時はRequestInfo項目の見直しが必要。
 * - PII/秘匿情報をログ・例外へ出力しないこと。
 */
@ApplicationScoped
public class KaisenkannrenjiIdoujouhouTourokuCiriusAdapter
        extends AbstractCiriusAdapter<KaisenkannrenjiIdoujouhouTourokuCiriusRestClient> {

    /**
     * CDI プロキシ用 no-args コンストラクタ。
     */
    protected KaisenkannrenjiIdoujouhouTourokuCiriusAdapter() {
        super();
    }

    /**
     * コンストラクタ。
     *
     * @param restClient CIRIUS REST クライアント
     */
    @Inject
    public KaisenkannrenjiIdoujouhouTourokuCiriusAdapter(
            @RestClient KaisenkannrenjiIdoujouhouTourokuCiriusRestClient restClient) {
        super("他-044", restClient);
    }

    /**
     * 開栓関連時異動情報を登録する。
     *
     * @param request 登録リクエスト
     */
    public void register(KaisenkannrenjiIdoujouhouTourokuRequest request) {
        register(request == null ? null : request.kaisenkannrenjiIdoujouhou());
    }

    /**
     * 開栓関連時異動情報を payload 直接指定で登録する。
     *
     * @param payload 連携 payload
     */
    public void register(KaisenkannrenjiIdoujouhouTourokuRequest.KaisenkannrenjiIdoujouhou payload) {
        // PII/秘匿情報をRequestInfoに含めない（pathのみ）
        invoke(
                () -> {
                    restClient().register(payload);
                    return null;
                },
                RequestInfo.of(
                        "path",
                        "/kaiheisen/kaisen-seikyukaisyu-renkei"
                        // 必要最小限の非秘匿項目のみ付与
                ));
    }
}
