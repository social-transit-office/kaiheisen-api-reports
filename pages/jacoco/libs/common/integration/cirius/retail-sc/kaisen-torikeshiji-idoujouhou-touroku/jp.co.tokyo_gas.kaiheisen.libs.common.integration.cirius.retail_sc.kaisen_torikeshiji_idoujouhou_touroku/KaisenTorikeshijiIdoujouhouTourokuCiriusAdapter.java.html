package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaisen_torikeshiji_idoujouhou_touroku;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * 概要: CIRIUS 開栓取消時異動情報登録 API（他-047）の adapter。
 *
 * 主な仕様:
 * - 外部 API（他-047）へのリクエスト送信と例外変換を担う。
 * - RequestInfo には path などの非秘匿トレース情報のみを付与する。
 *
 * 既知の制約:
 * - 契約番号などの秘匿情報は RequestInfo や例外メッセージへ含めない。
 */
@ApplicationScoped
public class KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter
        extends AbstractCiriusAdapter<KaisenTorikeshijiIdoujouhouTourokuCiriusRestClient> {

    /**
     * 概要: CDI プロキシ用 no-args コンストラクタ。
     */
    protected KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter() {
        super();
    }

    /**
     * 概要: 依存クライアントを受け取って初期化する。
     *
     * @param restClient CIRIUS REST クライアント
     */
    @Inject
    public KaisenTorikeshijiIdoujouhouTourokuCiriusAdapter(
            @RestClient KaisenTorikeshijiIdoujouhouTourokuCiriusRestClient restClient) {
        super("他-047", restClient);
    }

    /**
     * 概要: 開栓取消時異動情報を登録する。
     *
     * @param request 登録リクエスト
     */
    public void register(KaisenTorikeshijiIdoujouhouTourokuRequest request) {
        register(request == null ? null : request.kaisenTorikeshijiIdoujouhou());
    }

    /**
     * 概要: 開栓取消時異動情報を payload 直接指定で登録する。
     *
     * @param payload 連携 payload
     */
    public void register(KaisenTorikeshijiIdoujouhou payload) {
        invoke(
                () -> {
                    restClient().register(payload);
                    return null;
                },
                RequestInfo.of(
                        "path",
                        "/kaiheisen/kaisentorikeshi-seikyukaisyu-renkei"
                ));
    }
}
