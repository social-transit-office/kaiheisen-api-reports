/**

 * 主な仕様:
 * - SV から受け取った request を CIRIUS 小売 API へ橋渡しする。
 * - 例外を CiriusApiException に変換する。
 * 既知の制約:
 * - レスポンスボディを返さない fire-and-forget 型 API として扱う。
 */
@ApplicationScoped
public class HeisenkannrenjiIdoujouhouTourokuCiriusAdapter
        extends AbstractCiriusAdapter<HeisenkannrenjiIdoujouhouTourokuCiriusRestClient> {

    /** CDI プロキシ用 no-args コンストラクタ。 */
    protected HeisenkannrenjiIdoujouhouTourokuCiriusAdapter() {
        super();
    }

    /**
     * 概要: REST クライアントを注入して初期化する。
     *
     * @param restClient CIRIUS REST クライアント
     */
    @Inject
    public HeisenkannrenjiIdoujouhouTourokuCiriusAdapter(
            @RestClient HeisenkannrenjiIdoujouhouTourokuCiriusRestClient restClient) {
        super("他-045", restClient);
    }

    /**
     * 概要: 請求回収基盤（閉栓）API を呼び出す。
     *
     * @param request 登録リクエスト
     */
    public void register(HeisenkannrenjiIdoujouhouTourokuRequest request) {
        register(request == null ? null : request.heisenkannrenjiIdoujouhou());
    }

    /**
     * 概要: 請求回収基盤（閉栓）API を payload 直接指定で呼び出す。
     *
     * @param payload 連携 payload
     */
    public void register(HeisenkannrenjiIdoujouhou payload) {
        invoke(
                () -> {
                    restClient().register(payload);
                    return null;
                },
                buildRequestInfo(payload));
    }

    private String buildRequestInfo(HeisenkannrenjiIdoujouhou payload) {
        return RequestInfo.of(
                "pattern", "①",
                "path", "/kaiheisen/heisen-seikyukaisyu-renkei",
                "skrk_sqkr_jhu", stringify(payload == null ? null : payload.skrkSqkrJhu()),
                "svky_no", stringify(payload == null ? null : payload.svkyNo()));
    }

    private String stringify(Object value) {
        return value == null ? "null" : String.valueOf(value);
    }
}
