import java.math.BigDecimal;

        NotificationInfo kaiheisenHenkouTsuuchiJouhou) {

    /**

    public record NotificationInfo(
            @JsonProperty("tgku_htk_flg")
            String tgkuHtkFlg,
            @JsonProperty("utidx_no")
            String utidxNo,
            @JsonProperty("szsg_sbt")
            String szsgSbt,
            @JsonProperty("mk_sbt")
            String mkSbt,
            @JsonProperty("tdsr_no")
            String tdsrNo,
            @JsonProperty("kiy_no")
            BigDecimal kiyNo,
            @JsonProperty("ttzh")
            String ttzh,
            @JsonProperty("gas_mtst_no_1x")
            BigDecimal gasMtstNo1X,
            @JsonProperty("cst_no_3x")
            BigDecimal cstNo3X,
            @JsonProperty("gssyky_no_4x")
            BigDecimal gssykyNo4X,
            @JsonProperty("kyssnm_sk")
            String kyssnmSk,
            @JsonProperty("kyssnm_mk")
            String kyssnmMk,

            String kyskzSktsCd,

            String gymNyoSybtCd,
            @JsonProperty("tnmi_yago_kn")
            String tnmiYagoKn,
            @JsonProperty("tnmi_yago_ke")
            String tnmiYagoKe,

            String skkKsh,

            String skksy,

            Integer kytkd,
            @JsonProperty("mghk_flg")
            String mghkFlg,
            @JsonProperty("ysk_sbt")
            String yskSbt,
            @JsonProperty("ysk")
            BigDecimal ysk,
            @JsonProperty("sbsky_sbt")
            String sbskySbt,
            @JsonProperty("sbsky_idd")

    }
