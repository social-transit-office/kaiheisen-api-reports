package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.kaiheisen_change_notification;

import com.fasterxml.jackson.annotation.JsonInclude;
import com.fasterxml.jackson.annotation.JsonProperty;

/**
 * 開閉栓変更通知登録 API のレスポンス。
 *
 * @param response レスポンス本文
 */
@JsonInclude(JsonInclude.Include.NON_NULL)
public record KaiheisenChangeNotificationResponse(
        @JsonProperty("response")
        ResponseBody response) {

    /**
     * レスポンス本文。
     *
     * @param errorJouhou エラー情報
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record ResponseBody(
            @JsonProperty("errorJouhou")
            ErrorInfo errorJouhou) {
    }

    /**
     * エラー情報。
     *
     * @param msgId メッセージ ID
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record ErrorInfo(
            @JsonProperty("msgId")
            String msgId) {
    }
}
