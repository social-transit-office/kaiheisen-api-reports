package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.edit_customer;

import com.fasterxml.jackson.annotation.JsonInclude;
import com.fasterxml.jackson.annotation.JsonProperty;

/**
 * お客さま情報編集結果取得 API のリクエスト。
 *
 * @param henshuumaeOkyakusamaJouhou 編集前お客さま情報
 */
@JsonInclude(JsonInclude.Include.NON_NULL)
public record EditCustomerRequest(
        @JsonProperty("henshuumae_okyakusama_jouhou")
        BeforeCustomerInfo henshuumaeOkyakusamaJouhou) {

    /**
     * 編集前お客さま情報。
     *
     * @param henshuumaeOnamaeJouhou 編集前おなまえ情報
     * @param henshuumaeJuushoJouhou 編集前住所情報
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record BeforeCustomerInfo(
            @JsonProperty("henshuumae_onamae_jouhou")
            BeforeNameInfo henshuumaeOnamaeJouhou,
            @JsonInclude(JsonInclude.Include.ALWAYS)
            @JsonProperty("henshuumae_juusho_jouhou")
            BeforeAddressInfo henshuumaeJuushoJouhou) {
    }

    /**
     * 編集前おなまえ情報。
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record BeforeNameInfo(
            @JsonProperty("ctnm_kj1")
            String ctnmKj1,
            @JsonProperty("ctnm_kn1")
            String ctnmKn1,
            @JsonProperty("ctnm1_hujn_shbt")
            String ctnm1HujnShbt,
            @JsonProperty("ctnm1_hshi_shbt")
            String ctnm1HshiShbt,
            @JsonProperty("ctnm_kj2")
            String ctnmKj2,
            @JsonProperty("ctnm_kn2")
            String ctnmKn2,
            @JsonProperty("ctnm2_hujn_shbt")
            String ctnm2HujnShbt,
            @JsonProperty("ctnm2_hshi_shbt")
            String ctnm2HshiShbt,
            @JsonProperty("ctnm_kjhk_flg")
            String ctnmKjhkFlg,
            @JsonProperty("kynm_kj")
            String kynmKj,
            @JsonProperty("kynm_kn")
            String kynmKn,
            @JsonProperty("kynm_hujn_shbt")
            String kynmHujnShbt,
            @JsonProperty("kynm_hshi_shbt")
            String kynmHshiShbt,
            @JsonProperty("kynm_kjhk_flg")
            String kynmKjhkFlg) {
    }

    /**
     * 編集前住所情報。
     */
    @JsonInclude(JsonInclude.Include.NON_NULL)
    public record BeforeAddressInfo(
            @JsonProperty("yubn_no")
            String yubnNo,
            @JsonProperty("gkk_cd")
            String gkkCd,
            @JsonProperty("mch_cd")
            String mchCd,
            @JsonProperty("cmaz_cd")
            String cmazCd,
            @JsonProperty("cbg_suj1")
            String cbgSuj1,
            @JsonProperty("cbg_kn2")
            String cbgKn2,
            @JsonProperty("cbg_kn3")
            String cbgKn3,
            @JsonProperty("cbg_kn4")
            String cbgKn4,
            @JsonProperty("tuno_kn")
            String tunoKn,
            @JsonProperty("hyn_kn")
            String hynKn,
            @JsonProperty("htu_flg")
            String htuFlg,
            @JsonProperty("shki_lv_shbt")
            String shkiLvShbt,
            @JsonProperty("bldn_kj")
            String bldnKj,
            @JsonProperty("bldn_kn")
            String bldnKn,
            @JsonProperty("bldn_kjhk_flg")
            String bldnKjhkFlg,
            @JsonProperty("jut_kbn")
            String jutKbn) {

        /**
         * 空文字で初期化した編集前住所情報を返す。
         *
         * @return 編集前住所情報
         */
        public static BeforeAddressInfo emptyStrings() {
            return new BeforeAddressInfo(
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "",
                    "");
        }
    }
}
