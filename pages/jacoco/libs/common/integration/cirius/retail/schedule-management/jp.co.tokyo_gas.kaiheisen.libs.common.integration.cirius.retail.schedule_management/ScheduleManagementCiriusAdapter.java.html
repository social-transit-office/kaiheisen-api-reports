/**
 * @file ScheduleManagementCiriusAdapter.java
 * 概要: CIRIUS 日程管理情報取得 API（他-014）の adapter を定義する。
 * 主な仕様:
 * - パターン①②③に対応するメソッドを提供する。
 * - 例外を CiriusApiException に変換する。
 * 既知の制限事項:
 * - なし。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.schedule_management;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/**
 * CIRIUS 日程管理情報取得 API（他-014）の adapter。
 *
 * <p>{@link ScheduleManagementCiriusRestClient} をラップし、
 * パターン①②③に対応するメソッドと例外変換を提供する。</p>
 */
@ApplicationScoped
public class ScheduleManagementCiriusAdapter
        extends AbstractCiriusAdapter<ScheduleManagementCiriusRestClient> {

    /** CDI プロキシ用 no-args コンストラクタ。 */
    protected ScheduleManagementCiriusAdapter() {
        super();
    }

    /**
     * コンストラクタ。
     *
     * @param restClient CIRIUS REST クライアント。型: ScheduleManagementCiriusRestClient
     */
    @Inject
    public ScheduleManagementCiriusAdapter(
            @RestClient ScheduleManagementCiriusRestClient restClient) {
        super("他-014", restClient);
    }

    /**
     * パターン①: 対象年月日をキーに日程管理情報を取得する。
     *
     * @param targetYmd 対象年月日（yyyyMMdd 形式）。型: String
     * @return 日程管理情報レスポンス。型: ScheduleManagementResponse
     */
    public ScheduleManagementResponse getByTargetDate(String targetYmd) {
        return invoke(
                () -> restClient().getByTargetDate(targetYmd),
                RequestInfo.of("pattern", "①", "target_ymd", targetYmd));
    }

    /**
     * パターン②: 抽出開始年月日〜抽出終了年月日の範囲で日程管理情報を取得する。
     *
     * @param startYmd 抽出開始年月日（yyyyMMdd 形式）。型: String
     * @param endYmd 抽出終了年月日（yyyyMMdd 形式）。型: String
     * @return 日程管理情報レスポンス。型: ScheduleManagementResponse
     */
    public ScheduleManagementResponse getByDateRange(String startYmd, String endYmd) {
        return invoke(
                () -> restClient().getByDateRange(startYmd, endYmd),
                RequestInfo.of("pattern", "②", "start_ymd", startYmd, "end_ymd", endYmd));
    }

    /**
     * パターン③: 定例検針回分番号・検針方法分類をキーに日程管理情報を取得する。
     *
     * @param startYmd 抽出開始年月日（yyyyMMdd 形式）。型: String
     * @param endYmd 抽出終了年月日（yyyyMMdd 形式）。型: String
     * @param tkkbNo 定例検針回分番号。型: String
     * @param knsHuhuBr 検針方法分類（1:遠隔、2:訪問）。型: String
     * @return 日程管理情報レスポンス。型: ScheduleManagementResponse
     */
    public ScheduleManagementResponse getByDateRangeAndInspection(
            String startYmd, String endYmd, String tkkbNo, String knsHuhuBr) {
        return invoke(
                () -> restClient().getByDateRangeAndInspection(
                        startYmd, endYmd, tkkbNo, knsHuhuBr),
                RequestInfo.of("pattern", "③", "start_ymd", startYmd, "end_ymd", endYmd,
                        "tkkb_no", tkkbNo, "kns_huhu_br", knsHuhuBr));
    }
}
