package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.schedule_management;

import com.fasterxml.jackson.annotation.JsonProperty;

/**
 * CIRIUS 日程管理テーブルの1レコードに対応する DTO。
 *
 * <p>BFF-999 サンプルでは {@code rekiYmd}（暦年月日）と {@code cisKadKaibNo}（CIS稼働回分番号）のみ使用する。
 * その他のフィールドは CIRIUS API 仕様に従い全項目を定義している。</p>
 *
 * @param rekiYmd 暦年月日
 * @param jitenCd 時点コード
 * @param sysEygobiNo システム営業日番号
 * @param cisKadKaibNo CIS稼働回分番号
 * @param kinKaibNo 金融機関営業回分番号
 * @param ssyuManryoYmd 早収満了年月日
 * @param shrKigenYmd 支払期限年月日
 * @param ttKnsnkMinNo 定例電話検針回分（最小）番号
 * @param ttKnsnkMaxNo 定例電話検針回分（最大）番号
 * @param thKnsnKaibNo 定例訪問検針回分番号
 * @param telKnsnYmd 電話検針引抜年月日
 * @param hhcKmKnsnYmd ハコ・紙検針引抜年月日
 * @param ofcnTensoYmd オフコン転送年月日
 * @param hhcTensoYmd ハコ転送年月日
 * @param sokKnsnYmd 即検針年月日
 * @param ofcnUkeireYmd オフコン受入年月日
 * @param hostUkeirYmd ホスト受入年月日
 * @param sokBatRknYmd 即検分バッチ料金計算年月日
 * @param synoKouikiYmd 収納分広域作成年月日
 * @param synoHontenYmd 収納分本店作成年月日
 * @param synoSoksyuYmd 収納分即収集金年月日
 * @param synoKskomiYmd 収納分入金消込年月日
 * @param mtSakusueiYmd MT分作成年月日
 * @param mtFriSiteYmd MT分振替指定年月日
 * @param mtNknKeijYmd MT分入金計上年月日
 * @param mtNknKskmYmd MT分入金消込年月日
 * @param thkSakuseiYmd 帳票交換分作成年月日
 * @param thkFriSitYmd 帳票交換分振替指定年月日
 * @param thkNknKejYmd 帳票交換分入金計上年月日
 * @param thkcNknKjYmd 帳票交換C日程入金計上年月日
 * @param thkNknKesYmd 帳票交換分入金消込年月日
 * @param thkcNknKsYmd 帳票交換C日程入金消込年月日
 * @param sofriSakseYmd 総合振込分作成年月日
 * @param sofriFurisYmd 総合振込分振替指定年月日
 * @param sofriKarbaYmd 総合振込分仮払計上年月日
 * @param sofriKesikYmd 総合振込分返金消込年月日
 * @param sofriMSksYmd 総合振込分毎週用作成年月日
 * @param sofriMFrsYmd 総合振込毎週用振替指定年月日
 * @param sofriMKrbYmd 総合振込毎週用仮払計上年月日
 * @param sofriMKskYmd 総合振込毎週用返金消込年月日
 */
public record ScheduleManagementItem(
        @JsonProperty("reki_ymd") long rekiYmd,
        @JsonProperty("jiten_cd") String jitenCd,
        @JsonProperty("sys_eygobi_no") long sysEygobiNo,
        @JsonProperty("cis_kad_kaib_no") String cisKadKaibNo,
        @JsonProperty("kin_kaib_no") String kinKaibNo,
        @JsonProperty("ssyu_manryo_ymd") long ssyuManryoYmd,
        @JsonProperty("shr_kigen_ymd") long shrKigenYmd,
        @JsonProperty("tt_knsnk_min_no") String ttKnsnkMinNo,
        @JsonProperty("tt_knsnk_max_no") String ttKnsnkMaxNo,
        @JsonProperty("th_knsn_kaib_no") String thKnsnKaibNo,
        @JsonProperty("tel_knsn_ymd") long telKnsnYmd,
        @JsonProperty("hhc_km_knsn_ymd") long hhcKmKnsnYmd,
        @JsonProperty("ofcn_tenso_ymd") long ofcnTensoYmd,
        @JsonProperty("hhc_tenso_ymd") long hhcTensoYmd,
        @JsonProperty("sok_knsn_ymd") long sokKnsnYmd,
        @JsonProperty("ofcn_ukeire_ymd") long ofcnUkeireYmd,
        @JsonProperty("host_ukeir_ymd") long hostUkeirYmd,
        @JsonProperty("sok_bat_rkn_ymd") long sokBatRknYmd,
        @JsonProperty("syno_kouiki_ymd") long synoKouikiYmd,
        @JsonProperty("syno_honten_ymd") long synoHontenYmd,
        @JsonProperty("syno_soksyu_ymd") long synoSoksyuYmd,
        @JsonProperty("syno_kskomi_ymd") long synoKskomiYmd,
        @JsonProperty("mt_sakusuei_ymd") long mtSakusueiYmd,
        @JsonProperty("mt_fri_site_ymd") long mtFriSiteYmd,
        @JsonProperty("mt_nkn_keij_ymd") long mtNknKeijYmd,
        @JsonProperty("mt_nkn_kskm_ymd") long mtNknKskmYmd,
        @JsonProperty("thk_sakusei_ymd") long thkSakuseiYmd,
        @JsonProperty("thk_fri_sit_ymd") long thkFriSitYmd,
        @JsonProperty("thk_nkn_kej_ymd") long thkNknKejYmd,
        @JsonProperty("thkc_nkn_kj_ymd") long thkcNknKjYmd,
        @JsonProperty("thk_nkn_kes_ymd") long thkNknKesYmd,
        @JsonProperty("thkc_nkn_ks_ymd") long thkcNknKsYmd,
        @JsonProperty("sofri_sakse_ymd") long sofriSakseYmd,
        @JsonProperty("sofri_furis_ymd") long sofriFurisYmd,
        @JsonProperty("sofri_karba_ymd") long sofriKarbaYmd,
        @JsonProperty("sofri_kesik_ymd") long sofriKesikYmd,
        @JsonProperty("sofri_m_sks_ymd") long sofriMSksYmd,
        @JsonProperty("sofri_m_frs_ymd") long sofriMFrsYmd,
        @JsonProperty("sofri_m_krb_ymd") long sofriMKrbYmd,
        @JsonProperty("sofri_m_ksk_ymd") long sofriMKskYmd) {
}
