package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.billing;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/** CIRIUS 請求情報取得 API（他-049）の adapter。 */
@ApplicationScoped
public class BillingCiriusAdapter extends AbstractCiriusAdapter<BillingCiriusRestClient> {

    /** CDI プロキシ用コンストラクタ。 */
    protected BillingCiriusAdapter() {
        super();
    }

    /**
     * コンストラクタ。
     *
     * @param restClient REST クライアント
     */
    @Inject
    public BillingCiriusAdapter(@RestClient BillingCiriusRestClient restClient) {
        super("他-049", restClient);
    }

    /**
     * 請求管理番号をキーに請求情報を取得する。
     *
     * @param fromYmd 検索開始年月日
     * @param toYmd 検索終了年月日
     * @param sikKrno 請求管理番号
     * @return 請求情報レスポンス
     */
    public BillingResponse getByBillingManagementNumber(
            String fromYmd,
            String toYmd,
            String sikKrno) {
        return invoke(
                () -> restClient().getByBillingManagementNumber(fromYmd, toYmd, sikKrno),
                RequestInfo.of(
                        "path", "/billing",
                        "from_ymd", fromYmd,
                        "to_ymd", toYmd,
                        "sik_krno", sikKrno));
    }
}
