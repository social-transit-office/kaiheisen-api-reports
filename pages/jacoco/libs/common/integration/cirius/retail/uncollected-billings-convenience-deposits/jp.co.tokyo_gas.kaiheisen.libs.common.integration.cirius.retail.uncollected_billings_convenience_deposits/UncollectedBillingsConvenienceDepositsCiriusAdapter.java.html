package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.uncollected_billings_convenience_deposits;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.core.AbstractCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.core.RequestInfo;
import org.eclipse.microprofile.rest.client.inject.RestClient;

/** 未収請求一覧及び未対応コンビニ入金一覧検索 API（他-050） の adapter。 */
@ApplicationScoped
public class UncollectedBillingsConvenienceDepositsCiriusAdapter extends
        AbstractCiriusAdapter<UncollectedBillingsConvenienceDepositsCiriusRestClient> {

    /** CDI プロキシ用コンストラクタ。 */
    protected UncollectedBillingsConvenienceDepositsCiriusAdapter() {
        super();
    }

    /**
     * コンストラクタ。
     *
     * @param restClient REST クライアント
     */
    @Inject
    public UncollectedBillingsConvenienceDepositsCiriusAdapter(
            @RestClient UncollectedBillingsConvenienceDepositsCiriusRestClient restClient) {
        super("他-050", restClient);
    }

    /**
     * API を呼び出す。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    public UncollectedBillingsConvenienceDepositsResponse search(
            UncollectedBillingsConvenienceDepositsRequest request) {
        UncollectedBillingsConvenienceDepositsRequest.SearchCondition condition =
                request == null ? null : request.seikyuuShiharaiIchiranAndMitaiouKonbiniNyukinIchiran();
        return invoke(
                () -> restClient().search(request),
                RequestInfo.of(
                        "path", "/uncollected-billings-convenience-deposits",
                        "sik_krno", condition == null ? "" : condition.sikKrno(),
                        "max_shtk_knsu",
                        condition == null || condition.maxShtkKnsu() == null
                                ? ""
                                : String.valueOf(condition.maxShtkKnsu())));
    }
}
