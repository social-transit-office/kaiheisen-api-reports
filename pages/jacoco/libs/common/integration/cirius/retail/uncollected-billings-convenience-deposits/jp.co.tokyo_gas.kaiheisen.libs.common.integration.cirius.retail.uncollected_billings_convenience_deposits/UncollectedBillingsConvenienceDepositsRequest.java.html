package jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail.uncollected_billings_convenience_deposits;

import com.fasterxml.jackson.annotation.JsonProperty;

/** 未収請求一覧及び未対応コンビニ入金一覧検索 API のリクエスト。 */
public record UncollectedBillingsConvenienceDepositsRequest(
        @JsonProperty("seikyuu_shiharai_ichiran_and_mitaiou_konbini_nyukin_ichiran")
        SearchCondition seikyuuShiharaiIchiranAndMitaiouKonbiniNyukinIchiran) {

    /**
     * 検索条件を生成する。
     *
     * @param sikKrno 請求管理番号
     * @param maxShtkKnsu 最大取得件数
     * @return リクエスト
     */
    public static UncollectedBillingsConvenienceDepositsRequest of(
            String sikKrno,
            Integer maxShtkKnsu) {
        return new UncollectedBillingsConvenienceDepositsRequest(
                new SearchCondition(sikKrno, maxShtkKnsu));
    }

    /** リクエストボディ本体。 */
    public record SearchCondition(
            @JsonProperty("sik_krno")
            String sikKrno,
            @JsonProperty("max_shtk_knsu")
            Integer maxShtkKnsu) {
    }
}
