/**
 * @file ExternalApiLogTimestamp.java
 * 概要: 外部 API ログの観測時刻を整形する。
 * 主な仕様: UTC・ミリ秒3桁固定の ISO 8601 形式とする。
 * 既知の制限事項: ミリ秒未満は切り捨てる。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import java.time.Instant;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeFormatterBuilder;

/**
 * 外部 API ログの時刻形式を統一する。
 */
final class ExternalApiLogTimestamp {

    private static final DateTimeFormatter FORMATTER =
            new DateTimeFormatterBuilder().appendInstant(3).toFormatter();

    private ExternalApiLogTimestamp() {
    }

    /**
     * 観測済みの時刻を UTC・ミリ秒3桁固定で整形する。
     *
     * @param instant 観測時刻
     * @return ISO 8601 形式の時刻
     */
    static String format(final Instant instant) {
        return FORMATTER.format(instant);
    }
}
