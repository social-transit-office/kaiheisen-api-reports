/**
 * @file ExternalApiRequestLog.java
 * 概要: 外部 API リクエストログの主要項目を保持する。
 * 主な仕様:
 * - full URL、HTTP メソッド、request body JSON 文字列を構造化属性へ変換する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import java.util.LinkedHashMap;
import java.util.Map;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;

/**
 * 外部 API リクエストログ項目。
 *
 * @param url query string を含む full URL
 * @param httpMethod HTTP メソッド
 * @param requestBodyJson request body の JSON 文字列
 */
record ExternalApiRequestLog(String url, String httpMethod, String requestBodyJson) {

    private static final String URL_ATTRIBUTE = "url";

    private static final String REQUEST_BODY_JSON_ATTRIBUTE = "requestBodyJson";

    Map<String, Object> toAttributes() {
        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(URL_ATTRIBUTE, url);
        attributes.put(LogField.HTTP_METHOD.key(), httpMethod);
        attributes.put(REQUEST_BODY_JSON_ATTRIBUTE, requestBodyJson);
        return attributes;
    }
}
