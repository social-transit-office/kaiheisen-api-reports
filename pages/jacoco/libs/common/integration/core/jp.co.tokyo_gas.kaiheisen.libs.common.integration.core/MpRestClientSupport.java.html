/**
 * @file MpRestClientSupport.java
 * 概要: MicroProfile Rest Client 呼び出しの汎用例外変換ヘルパーを定義する。
 * 主な仕様:
 * - WebApplicationException をリクエスト情報・レスポンス情報付きで変換する。
 * - ProcessingException をリクエスト情報付きで通信失敗として変換する。
 * - 例外ファクトリを受け取り、任意の例外型への変換を可能にする。
 * - 既存の失敗ログに例外捕捉時刻 failedAt を付与する。
 * 既知の制限事項:
 * - レスポンスボディが既に消費済みの場合は取得できない。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import jakarta.enterprise.context.ContextNotActiveException;
import jakarta.enterprise.inject.spi.CDI;
import jakarta.ws.rs.ProcessingException;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.Response;
import java.time.Clock;
import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.PerformanceObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;

/**
 * MicroProfile Rest Client 呼び出しの汎用例外変換ヘルパー。
 *
 * <p>MP Rest Client 呼び出しを {@link #invoke} でラップすることで、
 * {@link WebApplicationException} や {@link ProcessingException} を
 * 任意のドメイン例外に変換する。</p>
 */
public final class MpRestClientSupport {

    private static final String EXTERNAL_API_STARTED_EVENT = "externalapi.started";

    private static final String EXTERNAL_API_STARTED_SUMMARY = "External API execution started";

    private static final String EXTERNAL_API_COMPLETED_EVENT = "externalapi.completed";

    private static final String EXTERNAL_API_COMPLETED_SUMMARY = "External API execution completed";

    private static final String EXTERNAL_API_FAILED_EVENT = "externalapi.failed";

    private static final String EXTERNAL_API_FAILED_SUMMARY = "External API execution failed";

    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    private static final String REQUEST_INFO_ATTRIBUTE = "requestInfo";

    private static final String REQUEST_PARAMS_ATTRIBUTE = "requestParams";

    private static final String RESPONSE_INFO_ATTRIBUTE = "responseInfo";

    private static final String RESPONSE_BODY_ATTRIBUTE = "responseBody";

    private static final StructuredLogger LOGGER =
            new JbossStructuredLogger(MpRestClientSupport.class);

    private record ResponseLogInfo(String text, String body) {
    }

    /**
     * 例外変換ファクトリ。
     */
    @FunctionalInterface
    public interface ExceptionFactory {

        /**
         * ドメイン例外を生成する。
         *
         * @param requestInfo リクエスト情報。型: String
         * @param responseInfo レスポンス情報（通信失敗時は null）。型: String
         * @param cause 原因例外。型: Throwable
         * @return ドメイン例外。型: RuntimeException
         */
        RuntimeException create(String requestInfo, String responseInfo, Throwable cause);
    }

    private MpRestClientSupport() {
    }

    /**
     * MP Rest Client 呼び出しを実行し、例外を変換する。
     *
     * @param <T> レスポンス型
     * @param call 呼び出し処理。型: Supplier
     * @param requestInfo リクエスト情報（ログ／例外メッセージ用）。型: String
     * @param exceptionFactory 例外変換ファクトリ。型: ExceptionFactory
     * @return 呼び出し結果。型: T
     */
    public static <T> T invoke(
            Supplier<T> call,
            String requestInfo,
            ExceptionFactory exceptionFactory) {
        return invoke(call, null, requestInfo, exceptionFactory);
    }

    /**
     * MP Rest Client 呼び出しを実行し、例外を変換する。
     *
     * @param <T> レスポンス型
     * @param call 呼び出し処理。型: Supplier
     * @param externalApiId 外部 API 識別子。型: String
     * @param requestInfo リクエスト情報（ログ／例外メッセージ用）。型: String
     * @param exceptionFactory 例外変換ファクトリ。型: ExceptionFactory
     * @return 呼び出し結果。型: T
     */
    public static <T> T invoke(
            Supplier<T> call,
            String externalApiId,
            String requestInfo,
            ExceptionFactory exceptionFactory) {
        return invoke(call, externalApiId, requestInfo, exceptionFactory, resolveRequestContext(), LOGGER);
    }

    static <T> T invoke(
            final Supplier<T> call,
            final String externalApiId,
            final String requestInfo,
            final ExceptionFactory exceptionFactory,
            final RequestContext requestContext,
            final StructuredLogger logger) {
        return invoke(call, externalApiId, requestInfo, exceptionFactory,
                requestContext, logger, Clock.systemUTC());
    }

    static <T> T invoke(
            final Supplier<T> call,
            final String externalApiId,
            final String requestInfo,
            final ExceptionFactory exceptionFactory,
            final RequestContext requestContext,
            final StructuredLogger logger,
            final Clock clock) {
        boolean terminalMeasurementEnabled = PerformanceObservabilitySettings.isTerminalEnabled();
        Long invocationSequence = terminalMeasurementEnabled
                ? safeNextExternalApiInvocationSequence(requestContext) : null;
        long startedAtNanos = System.nanoTime();
        if (terminalMeasurementEnabled) {
            safeInfo(logger, () -> buildStartedEvent(
                    externalApiId, requestInfo, invocationSequence, requestContext));
        } else {
            logger.info(buildStartedEvent(
                    externalApiId, requestInfo, invocationSequence, requestContext));
        }
        ExternalApiCallLogContext previousContext = ExternalApiCallLogContext.push(
                externalApiId, requestInfo, invocationSequence);
        try {
            T result = call.get();
            if (terminalMeasurementEnabled) {
                safeInfo(logger, () -> buildCompletedEvent(
                        externalApiId,
                        invocationSequence,
                        elapsedMillis(startedAtNanos),
                        requestContext));
            }
            return result;
        } catch (WebApplicationException e) {
            Instant failedAt = clock.instant();
            ResponseLogInfo responseInfo = extractResponseInfo(e);
            if (terminalMeasurementEnabled) {
                safeError(
                        logger,
                        () -> buildFailedEvent(
                                externalApiId,
                                requestInfo,
                                responseInfo.text(),
                                extractHttpStatus(e),
                                responseInfo.body(),
                                invocationSequence,
                                elapsedMillis(startedAtNanos),
                                requestContext, failedAt),
                        e);
            } else {
                logger.error(
                        buildFailedEvent(
                                externalApiId,
                                requestInfo,
                                responseInfo.text(),
                                extractHttpStatus(e),
                                responseInfo.body(),
                                invocationSequence,
                                null,
                                requestContext, failedAt),
                        e);
            }
            throw exceptionFactory.create(requestInfo, responseInfo.text(), e);
        } catch (ProcessingException e) {
            Instant failedAt = clock.instant();
            if (terminalMeasurementEnabled) {
                safeError(logger, () -> buildFailedEvent(
                        externalApiId,
                        requestInfo,
                        null,
                        invocationSequence,
                        elapsedMillis(startedAtNanos),
                        requestContext, failedAt), e);
            } else {
                logger.error(buildFailedEvent(
                        externalApiId,
                        requestInfo,
                        null,
                        invocationSequence,
                        null,
                        requestContext, failedAt), e);
            }
            throw exceptionFactory.create(requestInfo, null, e);
        } catch (RuntimeException | Error e) {
            Instant failedAt = clock.instant();
            if (terminalMeasurementEnabled) {
                safeError(logger, () -> buildFailedEvent(
                        externalApiId,
                        null,
                        null,
                        invocationSequence,
                        elapsedMillis(startedAtNanos),
                        requestContext, failedAt), e);
            }
            throw e;
        } finally {
            ExternalApiCallLogContext.restore(previousContext);
        }
    }

    private static LogEvent buildStartedEvent(
            final String externalApiId,
            final String requestInfo,
            final Long invocationSequence,
            final RequestContext requestContext) {
        return new LogEvent(
                EXTERNAL_API_STARTED_EVENT,
                EXTERNAL_API_STARTED_SUMMARY,
                LogLayer.EXTERNAL,
                null,
                null,
                buildAttributes(
                        externalApiId, requestInfo, null, null,
                        invocationSequence, null, requestContext));
    }

    private static LogEvent buildCompletedEvent(
            final String externalApiId,
            final Long invocationSequence,
            final long latencyMs,
            final RequestContext requestContext) {
        return new LogEvent(
                EXTERNAL_API_COMPLETED_EVENT,
                EXTERNAL_API_COMPLETED_SUMMARY,
                LogLayer.EXTERNAL,
                null,
                null,
                buildAttributes(
                        externalApiId, null, null, null,
                        invocationSequence, latencyMs, requestContext));
    }

    private static LogEvent buildFailedEvent(
            final String externalApiId,
            final String requestInfo,
            final String responseInfo,
            final Long invocationSequence,
            final Long latencyMs,
            final RequestContext requestContext,
            final Instant failedAt) {
        return buildFailedEvent(
                externalApiId, requestInfo, responseInfo, null, null,
                invocationSequence, latencyMs, requestContext, failedAt);
    }

    private static LogEvent buildFailedEvent(
            final String externalApiId,
            final String requestInfo,
            final String responseInfo,
            final Integer httpStatus,
            final String responseBody,
            final Long invocationSequence,
            final Long latencyMs,
            final RequestContext requestContext,
            final Instant failedAt) {
        Map<String, Object> attributes = buildAttributes(
                externalApiId, requestInfo, responseInfo, responseBody,
                invocationSequence, latencyMs, requestContext);
        attributes.put(LogField.FAILED_AT.key(), ExternalApiLogTimestamp.format(failedAt));
        return new LogEvent(
                EXTERNAL_API_FAILED_EVENT,
                EXTERNAL_API_FAILED_SUMMARY,
                LogLayer.EXTERNAL,
                httpStatus,
                null,
                attributes);
    }

    private static Map<String, Object> buildAttributes(
            final String externalApiId,
            final String requestInfo,
            final String responseInfo,
            final RequestContext requestContext) {
        return buildAttributes(externalApiId, requestInfo, responseInfo, null, requestContext);
    }

    private static Map<String, Object> buildAttributes(
            final String externalApiId,
            final String requestInfo,
            final String responseInfo,
            final String responseBody,
            final RequestContext requestContext) {
        return buildAttributes(
                externalApiId, requestInfo, responseInfo, responseBody,
                null, null, requestContext);
    }

    private static Map<String, Object> buildAttributes(
            final String externalApiId,
            final String requestInfo,
            final String responseInfo,
            final String responseBody,
            final Long invocationSequence,
            final Long latencyMs,
            final RequestContext requestContext) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        if (externalApiId != null && !externalApiId.isBlank()) {
            attributes.put(FUNCTION_ID_ATTRIBUTE, externalApiId);
        }
        String correlationId = safeGetCorrelationId(requestContext);
        if (correlationId != null) {
            attributes.put(LogField.CORRELATION_ID.key(), correlationId);
        }
        if (requestInfo != null) {
            attributes.put(REQUEST_INFO_ATTRIBUTE, requestInfo);
            String requestParams = extractRequestParams(requestInfo);
            if (requestParams != null) {
                attributes.put(REQUEST_PARAMS_ATTRIBUTE, requestParams);
            }
        }
        String callPath = safeBuildCallPath(requestContext, externalApiId);
        if (callPath != null) {
            attributes.put(LogField.CALL_PATH.key(), callPath);
        }
        if (responseInfo != null) {
            attributes.put(RESPONSE_INFO_ATTRIBUTE, responseInfo);
        }
        if (responseBody != null) {
            attributes.put(RESPONSE_BODY_ATTRIBUTE, responseBody);
        }
        if (invocationSequence != null) {
            attributes.put(LogField.EXTERNAL_API_INVOCATION_SEQUENCE.key(), invocationSequence);
        }
        if (latencyMs != null) {
            attributes.put(LogField.LATENCY_MS.key(), latencyMs);
        }
        return attributes;
    }

    private static Long safeNextExternalApiInvocationSequence(
            final RequestContext requestContext) {
        if (requestContext == null) {
            return null;
        }
        try {
            return requestContext.nextExternalApiInvocationSequence();
        } catch (RuntimeException ex) {
            return null;
        }
    }

    private static long elapsedMillis(final long startedAtNanos) {
        return Math.max(0L, System.nanoTime() - startedAtNanos) / 1_000_000L;
    }

    private static void safeInfo(
            final StructuredLogger logger,
            final Supplier<LogEvent> eventSupplier) {
        try {
            logger.info(eventSupplier.get());
        } catch (RuntimeException ignored) {
            // Observability must not alter the external API result.
        }
    }

    private static void safeError(
            final StructuredLogger logger,
            final Supplier<LogEvent> eventSupplier,
            final Throwable throwable) {
        try {
            logger.error(eventSupplier.get(), throwable);
        } catch (RuntimeException ignored) {
            // Observability must not replace the original failure.
        }
    }

    private static RequestContext resolveRequestContext() {
        try {
            return CDI.current().select(RequestContext.class).get();
        } catch (IllegalStateException | ContextNotActiveException ex) {
            return null;
        }
    }

    private static String safeGetCorrelationId(final RequestContext requestContext) {
        if (requestContext == null) {
            return null;
        }
        try {
            return requestContext.getCorrelationId().orElse(null);
        } catch (ContextNotActiveException ex) {
            return null;
        }
    }

    private static String safeBuildCallPath(
            final RequestContext requestContext,
            final String externalApiId) {
        if (requestContext == null) {
            return null;
        }
        try {
            return requestContext.buildCallPath(externalApiId);
        } catch (ContextNotActiveException ex) {
            return null;
        }
    }

    private static Integer extractHttpStatus(final WebApplicationException exception) {
        Response response = exception.getResponse();
        return response == null ? null : response.getStatus();
    }

    private static String extractRequestParams(final String requestInfo) {
        if (requestInfo == null || requestInfo.isBlank()) {
            return null;
        }
        int pathIndex = requestInfo.indexOf("path=");
        if (pathIndex < 0) {
            return null;
        }
        int firstSeparatorIndex = requestInfo.indexOf(", ", pathIndex);
        if (firstSeparatorIndex < 0 || firstSeparatorIndex + 2 >= requestInfo.length()) {
            return null;
        }
        return requestInfo.substring(firstSeparatorIndex + 2);
    }

    /**
     * WebApplicationException からレスポンス情報を抽出する。
     *
     * @param e 対象例外。型: WebApplicationException
     * @return レスポンス情報文字列。型: String
     */
    private static ResponseLogInfo extractResponseInfo(final WebApplicationException e) {
        Response response = e.getResponse();
        if (response == null) {
            return new ResponseLogInfo("status=unknown", null);
        }
        int status = response.getStatus();
        String body;
        try {
            if (response.hasEntity()) {
                response.bufferEntity();
                body = response.readEntity(String.class);
            } else {
                body = null;
            }
        } catch (RuntimeException suppressed) {
            body = null;
        }
        return new ResponseLogInfo("status=" + status + ", body=" + body, body);
    }
}
