/**
 * @file ExternalApiCallLogContext.java
 * 概要: 外部 API 呼び出し中のログ補助情報をスレッドローカルで保持する。
 * 主な仕様:
 * - MpRestClientSupport の呼び出し範囲内で functionId/requestInfo/呼び出し連番を Rest Client フィルタへ渡す。
 * - ネスト呼び出し時は直前のコンテキストへ復元する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import java.util.Optional;
import org.jboss.logging.MDC;

/**
 * 外部 API 呼び出し中のログ補助情報。
 *
 * @param functionId 外部 API / MA 識別子
 * @param requestInfo 既存のリクエスト情報文字列
 * @param invocationSequence リクエスト内の外部 API 呼び出し連番
 */
record ExternalApiCallLogContext(String functionId, String requestInfo, Long invocationSequence) {

    private static final String FUNCTION_ID_MDC_KEY = "externalApi.functionId";

    private static final String REQUEST_INFO_MDC_KEY = "externalApi.requestInfo";

    private static final String INVOCATION_SEQUENCE_MDC_KEY = "externalApi.invocationSequence";

    static ExternalApiCallLogContext push(
            final String functionId,
            final String requestInfo,
            final Long invocationSequence) {
        ExternalApiCallLogContext previous = current().orElse(null);
        putOrRemove(FUNCTION_ID_MDC_KEY, functionId);
        putOrRemove(REQUEST_INFO_MDC_KEY, requestInfo);
        putOrRemove(INVOCATION_SEQUENCE_MDC_KEY,
                invocationSequence == null ? null : invocationSequence.toString());
        return previous;
    }

    static void restore(final ExternalApiCallLogContext previous) {
        if (previous == null) {
            MDC.remove(FUNCTION_ID_MDC_KEY);
            MDC.remove(REQUEST_INFO_MDC_KEY);
            MDC.remove(INVOCATION_SEQUENCE_MDC_KEY);
        } else {
            putOrRemove(FUNCTION_ID_MDC_KEY, previous.functionId());
            putOrRemove(REQUEST_INFO_MDC_KEY, previous.requestInfo());
            putOrRemove(INVOCATION_SEQUENCE_MDC_KEY,
                    previous.invocationSequence() == null
                            ? null : previous.invocationSequence().toString());
        }
    }

    static Optional<ExternalApiCallLogContext> current() {
        String functionId = toString(MDC.get(FUNCTION_ID_MDC_KEY));
        String requestInfo = toString(MDC.get(REQUEST_INFO_MDC_KEY));
        Long invocationSequence = toLong(MDC.get(INVOCATION_SEQUENCE_MDC_KEY));
        if (functionId == null && requestInfo == null && invocationSequence == null) {
            return Optional.empty();
        }
        return Optional.of(new ExternalApiCallLogContext(functionId, requestInfo, invocationSequence));
    }

    private static void putOrRemove(final String key, final String value) {
        if (value == null) {
            MDC.remove(key);
        } else {
            MDC.put(key, value);
        }
    }

    private static String toString(final Object value) {
        return value == null ? null : value.toString();
    }

    private static Long toLong(final Object value) {
        if (value == null) {
            return null;
        }
        try {
            return Long.valueOf(value.toString());
        } catch (NumberFormatException exception) {
            return null;
        }
    }
}
