package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import java.nio.charset.StandardCharsets;
import java.util.Base64;

/**
 * HTTP Basic 認証の Authorization ヘッダ値を生成する。
 */
public final class BasicAuthenticationHeader {

    private static final String SCHEME = "Basic ";

    private BasicAuthenticationHeader() {
    }

    /**
     * ユーザー名とパスワードから Authorization ヘッダ値を生成する。
     *
     * @param username ユーザー名
     * @param password パスワード
     * @return Basic 認証の Authorization ヘッダ値
     */
    public static String create(String username, String password) {
        String credentials = username + ":" + password;
        return SCHEME
                + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.UTF_8));
    }
}
