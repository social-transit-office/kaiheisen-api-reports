/**
 * @file CorrelationIdClientHeadersFactory.java
 * 概要: 相関IDを外部API呼び出しヘッダに自動伝播するファクトリを定義する。
 * 主な仕様:
 * - RequestContext から correlationId を取得し X-Correlation-Id ヘッダへ設定する。
 * - MicroProfile Rest Client の {@code @RegisterClientHeaders} で指定して利用する。
 * 既知の制限事項:
 * - RequestContext が RequestScope 外で利用された場合はヘッダを付与しない。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.core.MultivaluedMap;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import org.eclipse.microprofile.rest.client.ext.ClientHeadersFactory;

/**
 * 相関IDを外部API呼び出しヘッダに自動伝播するファクトリ。
 *
 * <p>{@link RequestContext} から相関IDを取得し、
 * {@code X-Correlation-Id} ヘッダへ設定する。</p>
 */
@ApplicationScoped
public class CorrelationIdClientHeadersFactory implements ClientHeadersFactory {

    /**
     * 相関IDヘッダ名。
     */
    static final String HEADER_NAME = "X-Correlation-Id";

    /**
     * リクエストコンテキスト。
     */
    private final RequestContext requestContext;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト。型: RequestContext
     */
    @Inject
    public CorrelationIdClientHeadersFactory(RequestContext requestContext) {
        this.requestContext = requestContext;
    }

    /**
     * 送信ヘッダを更新し相関IDを付与する。
     *
     * @param incomingHeaders 受信ヘッダ。型: MultivaluedMap
     * @param clientOutgoingHeaders 送信ヘッダ。型: MultivaluedMap
     * @return 更新後の送信ヘッダ。型: MultivaluedMap
     */
    @Override
    public MultivaluedMap<String, String> update(
            MultivaluedMap<String, String> incomingHeaders,
            MultivaluedMap<String, String> clientOutgoingHeaders) {
        requestContext.getCorrelationId().ifPresent(
                correlationId -> clientOutgoingHeaders.putSingle(HEADER_NAME, correlationId));
        return clientOutgoingHeaders;
    }
}
