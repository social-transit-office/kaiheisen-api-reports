/**
 * @file ExternalApiRestClientLoggingFilter.java
 * 概要: MicroProfile Rest Client の送受信ログを出力するフィルタ。
 * 主な仕様:
 * - 送信時に full URL、HTTP メソッド、request body JSON をログ出力する。
 * - 応答時に status/latency を出力し、エラー応答では response body も出力する。
 * - response body 読み取り後も downstream が再読込できるよう entity stream を復元する。
 * - 送受信フィルタ開始時刻を UTC・ミリ秒3桁固定で出力する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.context.ContextNotActiveException;
import jakarta.inject.Inject;
import jakarta.ws.rs.client.ClientRequestContext;
import jakarta.ws.rs.client.ClientRequestFilter;
import jakarta.ws.rs.client.ClientResponseContext;
import jakarta.ws.rs.client.ClientResponseFilter;
import jakarta.ws.rs.ext.Provider;
import java.io.ByteArrayInputStream;
import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.net.URI;
import java.nio.charset.StandardCharsets;
import java.time.Clock;
import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.Map;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;

/**
 * MicroProfile Rest Client の送受信ログフィルタ。
 */
@Provider
@ApplicationScoped
public class ExternalApiRestClientLoggingFilter implements ClientRequestFilter, ClientResponseFilter {

    private static final String START_TIME_NANOS = "kaiheisen.externalapi.startTimeNanos";

    private static final String REQUEST_SENT_AT = "kaiheisen.externalapi.requestSentAt";

    private static final String REQUEST_URL = "kaiheisen.externalapi.url";

    private static final String REQUEST_METHOD = "kaiheisen.externalapi.httpMethod";

    private static final String REQUEST_EVENT = "externalapi.request";

    private static final String REQUEST_SUMMARY = "External API request sent";

    private static final String RESPONSE_EVENT = "externalapi.response";

    private static final String RESPONSE_SUMMARY = "External API response received";

    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    private static final String URL_ATTRIBUTE = "url";

    private static final String RESPONSE_BODY_ATTRIBUTE = "responseBody";

    private final RequestContext requestContext;

    private final StructuredLogger logger;

    private final Clock clock;

    /**
     * CDI 用コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public ExternalApiRestClientLoggingFilter(final RequestContext requestContext) {
        this(requestContext, new JbossStructuredLogger(ExternalApiRestClientLoggingFilter.class));
    }

    ExternalApiRestClientLoggingFilter(
            final RequestContext requestContext,
            final StructuredLogger logger) {
        this(requestContext, logger, Clock.systemUTC());
    }

    ExternalApiRestClientLoggingFilter(
            final RequestContext requestContext,
            final StructuredLogger logger,
            final Clock clock) {
        this.requestContext = requestContext;
        this.logger = logger;
        this.clock = clock;
    }

    @Override
    public void filter(final ClientRequestContext clientRequestContext) {
        Instant observedAt = clock.instant();
        String url = toUrl(clientRequestContext.getUri());
        String method = clientRequestContext.getMethod();
        clientRequestContext.setProperty(START_TIME_NANOS, System.nanoTime());
        clientRequestContext.setProperty(REQUEST_URL, url);
        clientRequestContext.setProperty(REQUEST_METHOD, method);
        String requestSentAt = ExternalApiLogTimestamp.format(observedAt);
        clientRequestContext.setProperty(REQUEST_SENT_AT, requestSentAt);

        ExternalApiRequestLog requestLog = new ExternalApiRequestLog(
                url,
                method,
                ExternalApiJsonSerializer.toJson(clientRequestContext.getEntity()));
        Map<String, Object> attributes = enrichWithContext(requestLog.toAttributes());
        attributes.put(LogField.REQUEST_SENT_AT.key(), requestSentAt);

        logger.info(new LogEvent(
                REQUEST_EVENT,
                REQUEST_SUMMARY,
                LogLayer.EXTERNAL,
                null,
                null,
                attributes));
    }

    @Override
    public void filter(
            final ClientRequestContext clientRequestContext,
            final ClientResponseContext clientResponseContext) throws IOException {
        Instant observedAt = clock.instant();
        String url = stringProperty(clientRequestContext, REQUEST_URL, toUrl(clientRequestContext.getUri()));
        String method = stringProperty(clientRequestContext, REQUEST_METHOD, clientRequestContext.getMethod());
        long latencyMs = latencyMs(clientRequestContext);

        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(URL_ATTRIBUTE, url);
        attributes.put(LogField.HTTP_METHOD.key(), method);
        attributes.put(LogField.LATENCY_MS.key(), latencyMs);
        String requestSentAt = stringProperty(clientRequestContext, REQUEST_SENT_AT, null);
        if (requestSentAt != null) {
            attributes.put(LogField.REQUEST_SENT_AT.key(), requestSentAt);
        }
        // Quarkus は通信失敗時にも status=0 で受信フィルタを実行する。
        if (clientResponseContext.getStatus() > 0) {
            attributes.put(LogField.RESPONSE_RECEIVED_AT.key(), ExternalApiLogTimestamp.format(observedAt));
        }
        if (clientResponseContext.getStatus() >= 400) {
            String responseBody = readResponseBody(clientResponseContext);
            if (responseBody != null) {
                attributes.put(RESPONSE_BODY_ATTRIBUTE, responseBody);
            }
        }

        logger.info(new LogEvent(
                RESPONSE_EVENT,
                RESPONSE_SUMMARY,
                LogLayer.EXTERNAL,
                clientResponseContext.getStatus(),
                null,
                enrichWithContext(attributes)));
    }

    private Map<String, Object> enrichWithContext(final Map<String, Object> attributes) {
        ExternalApiCallLogContext.current().ifPresent(context -> {
            if (context.functionId() != null && !context.functionId().isBlank()) {
                attributes.put(FUNCTION_ID_ATTRIBUTE, context.functionId());
            }
            if (context.invocationSequence() != null) {
                attributes.put(
                        LogField.EXTERNAL_API_INVOCATION_SEQUENCE.key(),
                        context.invocationSequence());
            }
        });
        String correlationId = safeGetCorrelationId();
        if (correlationId != null) {
            attributes.put(LogField.CORRELATION_ID.key(), correlationId);
        }
        String callPath = safeBuildCallPath();
        if (callPath != null) {
            attributes.put(LogField.CALL_PATH.key(), callPath);
        }
        return attributes;
    }

    private String safeGetCorrelationId() {
        if (requestContext == null) {
            return null;
        }
        try {
            return requestContext.getCorrelationId().orElse(null);
        } catch (ContextNotActiveException exception) {
            return null;
        }
    }

    private String safeBuildCallPath() {
        if (requestContext == null) {
            return null;
        }
        try {
            return ExternalApiCallLogContext.current()
                    .map(context -> requestContext.buildCallPath(context.functionId()))
                    .orElseGet(requestContext::currentCallPath);
        } catch (ContextNotActiveException exception) {
            return null;
        }
    }

    private static String toUrl(final URI uri) {
        return uri == null ? null : uri.toString();
    }

    private static String stringProperty(
            final ClientRequestContext requestContext,
            final String propertyName,
            final String defaultValue) {
        Object value = requestContext.getProperty(propertyName);
        return value == null ? defaultValue : value.toString();
    }

    private static long latencyMs(final ClientRequestContext requestContext) {
        Object start = requestContext.getProperty(START_TIME_NANOS);
        if (start instanceof Long startTimeNanos) {
            return (System.nanoTime() - startTimeNanos) / 1_000_000L;
        }
        return 0L;
    }

    private static String readResponseBody(final ClientResponseContext responseContext) throws IOException {
        InputStream inputStream = responseContext.getEntityStream();
        if (inputStream == null) {
            return null;
        }
        byte[] body = readAllBytes(inputStream);
        responseContext.setEntityStream(new ByteArrayInputStream(body));
        if (body.length == 0) {
            return null;
        }
        return new String(body, StandardCharsets.UTF_8);
    }

    private static byte[] readAllBytes(final InputStream inputStream) throws IOException {
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        byte[] buffer = new byte[1024];
        int read;
        while ((read = inputStream.read(buffer)) != -1) {
            output.write(buffer, 0, read);
        }
        return output.toByteArray();
    }
}
