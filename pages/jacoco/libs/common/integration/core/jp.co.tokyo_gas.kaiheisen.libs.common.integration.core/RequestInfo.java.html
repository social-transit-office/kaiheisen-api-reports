/**
 * @file RequestInfo.java
 * 概要: REST Client 呼び出しのリクエスト情報フォーマットユーティリティを定義する。
 * 主な仕様:
 * - record の {@code @QueryParam} 付きコンポーネントを自動展開してリクエスト情報文字列を生成する。
 * - key-value 可変長引数からリクエスト情報文字列を生成する。
 * 既知の制限事項:
 * - record 版はリフレクションを使用するため、例外パス（ログ用）での利用を想定する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import jakarta.ws.rs.QueryParam;
import java.lang.reflect.RecordComponent;
import java.util.StringJoiner;

/**
 * REST Client 呼び出しのリクエスト情報フォーマットユーティリティ。
 *
 * <p>adapter の例外変換時に渡す requestInfo 文字列を統一フォーマットで生成する。
 * record の {@link QueryParam} 付きコンポーネントを自動展開する版と、
 * key-value 可変長引数版の 2 メソッドを提供する。</p>
 */
public final class RequestInfo {

    private RequestInfo() {
    }

    /**
     * record の {@link QueryParam} 付きコンポーネントを自動展開してリクエスト情報文字列を生成する。
     *
     * <p>{@code @QueryParam} が付与されたコンポーネントのみを収集し、
     * {@code @HeaderParam} などその他のアノテーションが付いたコンポーネントは除外する。
     * コンポーネントの宣言順に {@code path=<path>, key1=val1, key2=val2} 形式で出力する。</p>
     *
     * @param path リクエストパス。型: String
     * @param request リクエスト record。型: Record
     * @return リクエスト情報文字列。型: String
     */
    public static String of(String path, Record request) {
        StringJoiner joiner = new StringJoiner(", ");
        joiner.add("path=" + path);

        RecordComponent[] components = request.getClass().getRecordComponents();
        for (RecordComponent component : components) {
            if (component.getAccessor().isAnnotationPresent(QueryParam.class)) {
                Object value;
                try {
                    value = component.getAccessor().invoke(request);
                } catch (ReflectiveOperationException e) {
                    value = "<error>";
                }
                joiner.add(component.getName() + "=" + value);
            }
        }
        return joiner.toString();
    }

    /**
     * key-value 可変長引数からリクエスト情報文字列を生成する。
     *
     * <p>{@code key1, val1, key2, val2, ...} の順に偶数個の引数を渡す。
     * {@code key1=val1, key2=val2} 形式で出力する。</p>
     *
     * @param keyValues キーと値の交互配列。型: String[]
     * @return リクエスト情報文字列。型: String
     * @throws IllegalArgumentException 引数が奇数個の場合
     */
    public static String of(String... keyValues) {
        if (keyValues.length % 2 != 0) {
            throw new IllegalArgumentException(
                    "keyValues must have an even number of elements, but got " + keyValues.length);
        }
        StringJoiner joiner = new StringJoiner(", ");
        for (int i = 0; i < keyValues.length; i += 2) {
            joiner.add(keyValues[i] + "=" + keyValues[i + 1]);
        }
        return joiner.toString();
    }
}
