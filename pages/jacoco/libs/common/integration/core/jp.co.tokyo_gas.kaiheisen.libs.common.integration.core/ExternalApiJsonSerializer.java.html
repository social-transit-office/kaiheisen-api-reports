/**
 * @file ExternalApiJsonSerializer.java
 * 概要: 外部 API ログ用 JSON シリアライズを提供する。
 * 主な仕様:
 * - Jackson で request entity を JSON 文字列化する。
 * - シリアライズ不能な値でもログ処理自体は継続する。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.integration.core;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.SerializationFeature;

/**
 * 外部 API ログ用 JSON シリアライザ。
 */
final class ExternalApiJsonSerializer {

    private static final ObjectMapper OBJECT_MAPPER = new ObjectMapper()
            .findAndRegisterModules()
            .disable(SerializationFeature.FAIL_ON_EMPTY_BEANS);

    private ExternalApiJsonSerializer() {
    }

    static String toJson(final Object value) {
        try {
            return OBJECT_MAPPER.writeValueAsString(value);
        } catch (JsonProcessingException exception) {
            return String.valueOf(value);
        }
    }
}
