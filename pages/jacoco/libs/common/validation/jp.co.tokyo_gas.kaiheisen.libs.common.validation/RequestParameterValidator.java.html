package jp.co.tokyo_gas.kaiheisen.libs.common.validation;

import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Objects;
import java.util.Set;
import java.util.regex.Pattern;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.parser.Rfc3339LocalDateParser;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.parser.Rfc3339LocalDateTimeParser;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;

/**
 * 複数 API で共通利用するリクエストパラメータのバリデーションユーティリティ。
 */
// TODO: バリデーション周り、全体的に見直し
@Singleton
public class RequestParameterValidator {

    private static final Pattern GAS_METER_PATTERN = Pattern.compile("1\\d{10}");

    private static final DateTimeFormatter BASIC_DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private static final Set<String> SHR_KBN_VALUES = Set.of("1", "2", "3");

    private final SystemDateTimeProvider clock;

    /**
     * コンストラクタ。
     *
     * @param clock システム日時プロバイダ
     */
    @Inject
    public RequestParameterValidator(SystemDateTimeProvider clock) {
        this.clock = Objects.requireNonNull(clock, "clock");
    }

    /**
     * ガスメーター設置場所番号の形式を検証し、long に変換する。
     */
    public long requireValidGasMeterSetNo(String rawValue) {
        if (rawValue == null || rawValue.isBlank()) {
            throw new InvalidRequestParameterException("gmtSetNo must be provided");
        }
        if (!GAS_METER_PATTERN.matcher(rawValue).matches()) {
            throw new InvalidRequestParameterException("gmtSetNo must be an 11-digit number starting with 1");
        }
        return Long.parseLong(rawValue);
    }

    /**
     * 基準日(YYYY-MM-DD)を検証し、yyyyMMdd 形式の int へ正規化する。未指定時はシステム日付を使用する。
     */
    public int resolveBaseDate(String rawValue) {
        LocalDate baseDate;
        if (rawValue == null || rawValue.isBlank()) {
            baseDate = clock.currentDate();
        } else {
            try {
                baseDate = Rfc3339LocalDateParser.parse(rawValue);
            } catch (IllegalArgumentException ex) {
                throw new InvalidRequestParameterException(ex.getMessage(), ex);
            }
        }
        return Integer.parseInt(baseDate.format(BASIC_DATE));
    }

    /**
     * 基準日を検証し、未指定時はシステム日付を使用する。
     */
    public LocalDate resolveBaseDate(LocalDate value) {
        return value == null ? clock.currentDate() : value;
    }

    /**
     * 処理区分(1/2/3)を検証し、トリム後の値を返す。
     */
    public String requireValidShrKbn(String rawValue) {
        if (rawValue == null || rawValue.isBlank()) {
            throw new InvalidRequestParameterException("shrKbn must be provided");
        }
        String normalized = rawValue.trim();
        if (!SHR_KBN_VALUES.contains(normalized)) {
            throw new InvalidRequestParameterException("shrKbn must be 1, 2, or 3");
        }
        return normalized;
    }

    /**
     * RFC3339 形式の日時を検証し、未指定時はシステム日時を返す。
     */
    public LocalDateTime resolveSystemDateTime(String rawValue) {
        if (rawValue == null || rawValue.isBlank()) {
            return clock.currentDateTime();
        }
        try {
            return Rfc3339LocalDateTimeParser.parse(rawValue);
        } catch (IllegalArgumentException ex) {
            throw new InvalidRequestParameterException(ex.getMessage(), ex);
        }
    }

    /**
     * システム日時を検証し、未指定時はシステム日時を使用する。
     */
    public LocalDateTime resolveSystemDateTime(LocalDateTime value) {
        return value == null ? clock.currentDateTime() : value;
    }
}
