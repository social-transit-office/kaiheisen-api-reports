/**
 * カスタムアノテーションを走査して DTO の単一項目バリデーションを行う実装。
 */
@Singleton
public class BeanValidationRequestDtoValidator implements RequestDtoValidator {

    private static final String JAKARTA_VALID_ANNOTATION = "jakarta.validation.Valid";

    private final KaiheisenConstantService kaiheisenConstantService;

    private final CiriusConstantService ciriusConstantService;

    /**
     * コンストラクタ。
     *
     * @param kaiheisenConstantService 開閉栓コンスタントサービス
     * @param ciriusConstantServiceInstance CIRIUS コンスタントサービスのインスタンス
     */
    @Inject
    public BeanValidationRequestDtoValidator(
            KaiheisenConstantService kaiheisenConstantService,
            Instance<CiriusConstantService> ciriusConstantServiceInstance) {
        this(
                kaiheisenConstantService,
                ciriusConstantServiceInstance.isResolvable()
                        ? ciriusConstantServiceInstance.get()
                        : null);
    }

    /**
     * テストや CIRIUS 未使用モジュール向けのコンストラクタ。
     *
     * @param kaiheisenConstantService 開閉栓コンスタントサービス
     */
    public BeanValidationRequestDtoValidator(KaiheisenConstantService kaiheisenConstantService) {
        this(kaiheisenConstantService, (CiriusConstantService) null);
    }

    /**
     * テスト向けのコンストラクタ。
     *
     * @param kaiheisenConstantService 開閉栓コンスタントサービス
     * @param ciriusConstantService CIRIUS コンスタントサービス
     */
    public BeanValidationRequestDtoValidator(
            KaiheisenConstantService kaiheisenConstantService,
            CiriusConstantService ciriusConstantService) {
        this.kaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");
        this.ciriusConstantService = ciriusConstantService;
    }

    @Override
    public List<ValidationError> validate(Object requestDto) {
        Objects.requireNonNull(requestDto, "requestDto");
        List<ValidationError> errors = new ArrayList<>();
        Class<?> requestType = requestDto.getClass();
        if (!requestType.isRecord()) {
            return List.of();
        }
        for (RecordComponent component : requestType.getRecordComponents()) {
            Object value = readComponentValue(component, requestDto);
            String fieldName = component.getName();
            if (hasValidAnnotation(component)) {
                validateNestedValue(value, fieldName, errors);
            }
            validateAnnotations(component.getDeclaredAnnotations(), value, fieldName, errors);
        }
        return List.copyOf(errors);
    }

    private Object readComponentValue(RecordComponent component, Object requestDto) {
        try {
            return component.getAccessor().invoke(requestDto);
        } catch (IllegalAccessException | InvocationTargetException ex) {
            throw new IllegalStateException(
                    "failed to read record component value: " + component.getName(),
                    ex);
        }
    }

    private void validateAnnotations(
            Annotation[] annotations, Object value, String fieldName, List<ValidationError> errors) {
        for (Annotation annotation : annotations) {
            Optional<ValidationError> error = resolveError(annotation, value, fieldName);
            error.ifPresent(errors::add);
        }
    }

    private boolean hasValidAnnotation(RecordComponent component) {
        return containsValidAnnotation(component.getDeclaredAnnotations())
                || containsValidAnnotation(component.getAccessor().getDeclaredAnnotations())
                || containsValidAnnotation(component.getAnnotatedType().getAnnotations());
    }

    private boolean containsValidAnnotation(Annotation[] annotations) {
        for (Annotation annotation : annotations) {
            if (JAKARTA_VALID_ANNOTATION.equals(annotation.annotationType().getName())) {
                return true;
            }
        }
        return false;
    }

    private void validateNestedValue(Object value, String fieldName, List<ValidationError> errors) {
        if (value == null) {
            return;
        }
        if (value instanceof Collection<?> values) {
            int index = 0;
            for (Object element : values) {
                if (element != null) {
                    validateNestedRecord(element, fieldName + "[" + index + "]", errors);
                }
                index++;
            }
            return;
        }
        validateNestedRecord(value, fieldName, errors);
    }

    private void validateNestedRecord(Object value, String fieldName, List<ValidationError> errors) {
        Class<?> nestedType = value.getClass();
        if (!nestedType.isRecord()) {
            return;
        }
        for (RecordComponent component : nestedType.getRecordComponents()) {
            Object nestedValue = readComponentValue(component, value);
            String nestedFieldName = fieldName + "." + component.getName();
            if (hasValidAnnotation(component)) {
                validateNestedValue(nestedValue, nestedFieldName, errors);
            }
            validateAnnotations(
                    component.getDeclaredAnnotations(),
                    nestedValue,
                    nestedFieldName,
                    errors);
        }
    }

    private Optional<ValidationError> resolveError(Annotation annotation, Object value, String fieldName) {
        if (annotation instanceof RequiredText a) {
            return RequiredText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof RequiredValue a) {
            return RequiredValue.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof AllowedValues a) {
            return AllowedValues.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof GasMtstNoFormat a) {
            return GasMtstNoFormat.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof HalfWidthAlphanumericText a) {
            return HalfWidthAlphanumericText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof HalfWidthText a) {
            return HalfWidthText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof FullWidthText a) {
            return FullWidthText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof ArrayTypeValue a) {
            return ArrayTypeValue.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof CollectionMaxSize a) {
            return CollectionMaxSize.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericFixedDigits a) {
            return NumericFixedDigits.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericMaxDigits a) {
            return NumericMaxDigits.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericMinDigits a) {
            return NumericMinDigits.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NonZeroNumericValue a) {
            return NonZeroNumericValue.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericDigitsRange a) {
            return NumericDigitsRange.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericDecimalMaxDigits a) {
            return NumericDecimalMaxDigits.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericHyphenText a) {
            return NumericHyphenText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof NumericText a) {
            return NumericText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof PhoneNumberText a) {
            return PhoneNumberText.Rule.instance().validate(value, a, fieldName);
        }
        Optional<ValidationError> signedDecimalError = resolveSignedNumericDecimalMaxDigits(annotation, value, fieldName);
        if (signedDecimalError.isPresent()) {
            return signedDecimalError;
        }
        if (annotation instanceof TextFixedLength a) {
            return TextFixedLength.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof TextMaxLength a) {
            return TextMaxLength.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof TextMinLength a) {
            return TextMinLength.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof TextLengthRange a) {
            return TextLengthRange.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof ZeroStartNumericFixedDigits a) {
            return ZeroStartNumericFixedDigits.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateText a) {
            return Rfc3339DateText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateTimeText a) {
            return Rfc3339DateTimeText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateOrDateTimeText a) {
            return Rfc3339DateOrDateTimeText.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateMax a) {
            return Rfc3339DateMax.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateMin a) {
            return Rfc3339DateMin.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof Rfc3339DateRange a) {
            return Rfc3339DateRange.Rule.instance().validate(value, a, fieldName);
        }
        if (annotation instanceof ConstantTypeValue a) {
            return ConstantTypeValue.Rule.instance().validate(
                    value,
                    a,
                    fieldName,
                    kaiheisenConstantService,
                    ciriusConstantService);
        }
        return Optional.empty();
    }

    private Optional<ValidationError> resolveSignedNumericDecimalMaxDigits(
            Annotation annotation, Object value, String fieldName) {
        if (!"SignedNumericDecimalMaxDigits".equals(annotation.annotationType().getSimpleName())) {
            return Optional.empty();
        }
        String normalized = normalizeSignedDecimalValue(value).orElse(null);
        if (normalized == null) {
            return Optional.empty();
        }
        if (normalized.isEmpty()) {
            return Optional.empty();
        }

        int integerDigits = readAnnotationInt(annotation, "integerDigits");
        int fractionDigits = readAnnotationInt(annotation, "fractionDigits");
        String errorCode =
                readAnnotationString(annotation, "errorCode")
                        .orElse(ErrorCodeConstants.ERROR_CODE_010000020);
        if (!normalized.matches("^-?\\d+(?:\\.\\d+)?$")) {
            return Optional.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, fieldName));
        }
        String unsigned = normalized.startsWith("-") ? normalized.substring(1) : normalized;
        String[] parts = unsigned.split("\\.", -1);
        int integerLength = parts[0].length();
        int fractionLength = parts.length > 1 ? parts[1].length() : 0;
        if (integerLength <= integerDigits && fractionLength <= fractionDigits) {
            return Optional.empty();
        }
        return Optional.of(ValidationError.of(
                errorCode,
                fieldName,
                String.valueOf(integerDigits),
                String.valueOf(fractionDigits)));
    }

    private Optional<String> normalizeSignedDecimalValue(Object value) {
        if (value instanceof Number number) {
            try {
                return Optional.of(new java.math.BigDecimal(number.toString()).toPlainString());
            } catch (NumberFormatException ex) {
                return Optional.of("__INVALID_NUMBER__");
            }
        }
        if (!(value instanceof String text)) {
            return Optional.empty();
        }
        return Optional.of(text.trim());
    }

    private int readAnnotationInt(Annotation annotation, String methodName) {
        try {
            Method method = annotation.annotationType().getMethod(methodName);
            return (Integer) method.invoke(annotation);
        } catch (NoSuchMethodException | IllegalAccessException | InvocationTargetException ex) {
            throw new IllegalStateException(
                    "failed to read annotation attribute: "
                            + annotation.annotationType().getName()
                            + "."
                            + methodName,
                    ex);
        }
    }

    private Optional<String> readAnnotationString(Annotation annotation, String methodName) {
        try {
            Method method = annotation.annotationType().getMethod(methodName);
            return Optional.ofNullable((String) method.invoke(annotation));
        } catch (NoSuchMethodException ex) {
            return Optional.empty();
        } catch (IllegalAccessException | InvocationTargetException ex) {
            throw new IllegalStateException(
                    "failed to read annotation attribute: "
                            + annotation.annotationType().getName()
                            + "."
                            + methodName,
                    ex);
        }
    }
}
