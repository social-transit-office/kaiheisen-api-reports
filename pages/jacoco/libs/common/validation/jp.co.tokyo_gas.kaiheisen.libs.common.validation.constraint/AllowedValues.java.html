package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.util.Optional;
import java.util.Set;
import java.util.stream.Collectors;
import java.util.stream.Stream;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * 許可値リストを検証するカスタムアノテーション。
 *
 * <p>{@code value} 属性で指定した文字列のいずれかに一致するかを検証する。
 * null・空白は正常扱い（{@link RequiredText} と組み合わせて使用する）。
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface AllowedValues {

    /**
     * 許可する値の一覧。
     */
    String[] value();

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000029;

    /**
     * AllowedValues 検証ロジック。
     */
    final class Rule implements ConstraintRule<AllowedValues> {
        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(Object value, AllowedValues annotation, String fieldName) {
            if (!(value instanceof String text)) {
                return Optional.empty();
            }
            String normalized = text.trim();
            if (normalized.isEmpty()) {
                return Optional.empty();
            }
            Set<String> allowed = Set.of(annotation.value());
            if (allowed.contains(normalized)) {
                return Optional.empty();
            }
            String allowedList = Stream.of(annotation.value()).collect(Collectors.joining(", "));
            return Optional.of(ValidationError.of(annotation.errorCode(), fieldName, "", allowedList));
        }
    }
}
