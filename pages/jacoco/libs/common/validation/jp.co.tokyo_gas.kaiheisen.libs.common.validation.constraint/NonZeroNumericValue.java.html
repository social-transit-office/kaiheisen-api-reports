package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.math.BigDecimal;
import java.util.Optional;
import java.util.regex.Pattern;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * 0 以外の数値を検証するカスタムアノテーション。
 *
 * <p>String は半角数字のみを対象に 0 以外を許可し、Number は 0 のみ拒否する。</p>
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface NonZeroNumericValue {

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000029;

    /**
     * NonZeroNumericValue 検証ロジック。
     */
    final class Rule implements ConstraintRule<NonZeroNumericValue> {
        private static final Pattern NUMERIC = Pattern.compile("^\\d+$");

        private static final Pattern ZERO_ONLY = Pattern.compile("^0+$");

        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(
                Object value,
                NonZeroNumericValue annotation,
                String fieldName) {
            if (value == null) {
                return Optional.empty();
            }
            if (value instanceof String text) {
                String normalized = text.trim();
                if (normalized.isEmpty()) {
                    return Optional.empty();
                }
                if (!NUMERIC.matcher(normalized).matches()) {
                    return Optional.of(
                            ValidationError.ofField(
                                    ErrorCodeConstants.ERROR_CODE_010000012,
                                    fieldName));
                }
                if (ZERO_ONLY.matcher(normalized).matches()) {
                    return Optional.of(ValidationError.of(annotation.errorCode(), fieldName, "0以外"));
                }
                return Optional.empty();
            }
            if (value instanceof Number number) {
                try {
                    if (BigDecimal.ZERO.compareTo(new BigDecimal(number.toString())) == 0) {
                        return Optional.of(ValidationError.of(annotation.errorCode(), fieldName, "0以外"));
                    }
                } catch (NumberFormatException exception) {
                    return Optional.of(
                            ValidationError.ofField(
                                    ErrorCodeConstants.ERROR_CODE_010000012,
                                    fieldName));
                }
            }
            return Optional.empty();
        }
    }
}
