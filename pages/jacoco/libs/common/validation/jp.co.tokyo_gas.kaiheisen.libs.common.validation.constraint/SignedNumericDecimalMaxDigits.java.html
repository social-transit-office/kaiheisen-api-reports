package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.math.BigDecimal;
import java.util.Optional;
import java.util.regex.Pattern;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * 符号付き半角数字の整数部・少数部の最大桁数を検証するカスタムアノテーション。
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface SignedNumericDecimalMaxDigits {

    /**
     * 整数部最大桁数。
     */
    int integerDigits();

    /**
     * 少数部最大桁数。
     */
    int fractionDigits();

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000020;

    /**
     * SignedNumericDecimalMaxDigits 検証ロジック。
     */
    final class Rule implements ConstraintRule<SignedNumericDecimalMaxDigits> {
        private static final Pattern DECIMAL = Pattern.compile("^-?\\d+(?:\\.\\d+)?$");

        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(
                Object value,
                SignedNumericDecimalMaxDigits annotation,
                String fieldName) {
            if (value instanceof Number number) {
                try {
                    return validateNormalized(
                            new BigDecimal(number.toString()).toPlainString(),
                            annotation,
                            fieldName);
                } catch (NumberFormatException exception) {
                    return Optional.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, fieldName));
                }
            }
            if (!(value instanceof String text)) {
                return Optional.empty();
            }

            String normalized = text.trim();
            if (normalized.isEmpty()) {
                return Optional.empty();
            }

            return validateNormalized(normalized, annotation, fieldName);
        }

        private Optional<ValidationError> validateNormalized(
                String normalized,
                SignedNumericDecimalMaxDigits annotation,
                String fieldName) {
            if (!DECIMAL.matcher(normalized).matches()) {
                return Optional.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, fieldName));
            }

            String unsigned = normalized.startsWith("-") ? normalized.substring(1) : normalized;
            String[] parts = unsigned.split("\\.", -1);
            int integerLength = parts[0].length();
            int fractionLength = parts.length > 1 ? parts[1].length() : 0;

            if (integerLength <= annotation.integerDigits() && fractionLength <= annotation.fractionDigits()) {
                return Optional.empty();
            }

            return Optional.of(ValidationError.of(
                    annotation.errorCode(),
                    fieldName,
                    String.valueOf(annotation.integerDigits()),
                    String.valueOf(annotation.fractionDigits())));
        }
    }
}
