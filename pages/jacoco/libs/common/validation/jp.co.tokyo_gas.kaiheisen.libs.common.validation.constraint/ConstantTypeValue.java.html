/**
 * @file ConstantTypeValue.java
 * 概要: コンスタント種別一覧に存在する値であることを検証するカスタムアノテーション。
 * 主な仕様:
 * - type 属性でコンスタント種別コードを指定する。
 * - source 属性に応じて KaiheisenConstantService または CiriusConstantService を用いて許可値を動的に取得・検証する。
 * 既知の制限事項:
 * - BeanValidationRequestDtoValidator が定数参照元に応じたサービスを解決できる前提。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.util.List;
import java.util.Optional;
import java.util.stream.Collectors;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.CiriusConstantListItem;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.model.KaiheisenConstantItem;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * コンスタント種別一覧に存在する値であることを検証するカスタムアノテーション。
 *
 * <p>使用例: {@code @ConstantTypeValue(type = "KH001") String hukkSysShbt}
 *
 * @since 1.0.0
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface ConstantTypeValue {

    /**
     * 定数参照元。
     */
    enum Source {
        /** 開閉栓独自定数。 */
        KAIHEISEN,
        /** CIRIUS 定数。 */
        CIRIUS
    }

    /**
     * コンスタント種別コード。
     */
    String type();

    /**
     * 定数参照元。未指定時は開閉栓独自定数を参照する。
     */
    Source source() default Source.KAIHEISEN;

    /**
     * エラーコード。デフォルトはコード一覧存在チェックバリデーションエラー。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000028;

    /**
     * ConstantTypeValue 検証ロジック。
     */
    final class Rule implements ConstraintRule<ConstantTypeValue> {
        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(Object value, ConstantTypeValue annotation, String fieldName) {
            throw new UnsupportedOperationException(
                "Use validate(Object, ConstantTypeValue, String, KaiheisenConstantService, "
                    + "CiriusConstantService) instead");
        }

        /**
         * コンスタント種別の値存在を検証する。
         *
         * @param value 対象値
         * @param annotation アノテーション
         * @param fieldName フィールド名
         * @param kaiheisenConstantService 開閉栓コンスタントサービス
         * @param ciriusConstantService CIRIUS コンスタントサービス
         * @return 違反時は ValidationError を含む Optional、正常時は empty
         */
        public Optional<ValidationError> validate(
                Object value,
                ConstantTypeValue annotation,
                String fieldName,
                KaiheisenConstantService kaiheisenConstantService,
                CiriusConstantService ciriusConstantService) {
            if (!(value instanceof String text)) {
                return Optional.empty();
            }
            String normalized = text.trim();
            if (normalized.isEmpty()) {
                return Optional.empty();
            }
            return switch (annotation.source()) {
                case KAIHEISEN -> validateKaiheisen(
                        normalized,
                        annotation,
                        fieldName,
                        kaiheisenConstantService);
                case CIRIUS -> validateCirius(
                        normalized,
                        annotation,
                        fieldName,
                        ciriusConstantService);
            };
        }

        private Optional<ValidationError> validateKaiheisen(
                String normalized,
                ConstantTypeValue annotation,
                String fieldName,
                KaiheisenConstantService kaiheisenConstantService) {
            List<KaiheisenConstantItem> items = kaiheisenConstantService.listActiveByType(annotation.type());
            boolean found = items.stream().anyMatch(item -> normalized.equals(item.value()));
            if (found) {
                return Optional.empty();
            }
            String allowedValues = items.stream()
                    .map(KaiheisenConstantItem::value)
                    .collect(Collectors.joining(", "));
            return Optional.of(ValidationError.of(
                    annotation.errorCode(),
                    fieldName,
                    annotation.type(),
                    allowedValues));
        }

        private Optional<ValidationError> validateCirius(
                String normalized,
                ConstantTypeValue annotation,
                String fieldName,
                CiriusConstantService ciriusConstantService) {
            if (ciriusConstantService == null) {
                throw new IllegalStateException(
                        "CiriusConstantService is required for ConstantTypeValue source=CIRIUS");
            }
            List<CiriusConstantListItem> items = ciriusConstantService.listActiveByTypeCode(annotation.type());
            boolean found = items.stream().anyMatch(item -> normalized.equals(item.code()));
            if (found) {
                return Optional.empty();
            }
            String allowedValues = items.stream()
                    .map(CiriusConstantListItem::code)
                    .collect(Collectors.joining(", "));
            return Optional.of(ValidationError.of(
                    annotation.errorCode(),
                    fieldName,
                    annotation.type(),
                    allowedValues));
        }
    }
}
