package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.time.LocalDate;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.parser.Rfc3339LocalDateParser;

/**
 * RFC3339 日付の最大値を検証するカスタムアノテーション。
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface Rfc3339DateMax {

    /**
     * 上限日付（yyyy-MM-dd）。
     */
    String max();

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000025;

    /**
     * Rfc3339DateMax 検証ロジック。
     */
    final class Rule implements ConstraintRule<Rfc3339DateMax> {
        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(Object value, Rfc3339DateMax annotation, String fieldName) {
            if (!(value instanceof String text)) {
                return Optional.empty();
            }
            String normalized = text.trim();
            if (normalized.isEmpty()) {
                return Optional.empty();
            }
            LocalDate targetDate;
            try {
                targetDate = Rfc3339LocalDateParser.parse(normalized);
            } catch (IllegalArgumentException ex) {
                return Optional.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000015, fieldName));
            }
            LocalDate maxDate = parseLimit(annotation.max(), "max");
            if (!targetDate.isAfter(maxDate)) {
                return Optional.empty();
            }
            return Optional.of(ValidationError.of(annotation.errorCode(), fieldName, annotation.max()));
        }

        private LocalDate parseLimit(String rawValue, String attributeName) {
            if (rawValue == null || rawValue.isBlank()) {
                throw new IllegalStateException("Rfc3339DateMax." + attributeName + " must be specified");
            }
            try {
                return Rfc3339LocalDateParser.parse(rawValue);
            } catch (IllegalArgumentException ex) {
                throw new IllegalStateException("Rfc3339DateMax." + attributeName + " must follow yyyy-MM-dd", ex);
            }
        }
    }
}
