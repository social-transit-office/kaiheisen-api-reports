package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.nio.ByteBuffer;
import java.nio.CharBuffer;
import java.nio.charset.Charset;
import java.nio.charset.CharsetEncoder;
import java.nio.charset.CoderResult;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * 全角文字を検証するカスタムアノテーション。
 *
 * <p>本制約では、入力値を {@link String#trim()} で前後空白除去したうえで、
 * Windows-31J で各文字が 2 バイトで表現可能な文字のみを「全角」とみなす。
 * null / blank は本制約では違反とせず、必須判定は別制約で担保する。</p>
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface FullWidthText {

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000030;

    /**
     * FullWidthText 検証ロジック。
     */
    final class Rule implements ConstraintRule<FullWidthText> {

        private static final String WINDOWS_31J_NAME = "windows-31j";

        private static final Charset WINDOWS_31J = resolveWindows31j();

        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(Object value, FullWidthText annotation, String fieldName) {
            if (!(value instanceof String text)) {
                return Optional.empty();
            }
            String normalized = text.trim();
            if (normalized.isEmpty()) {
                return Optional.empty();
            }
            CharsetEncoder encoder = WINDOWS_31J.newEncoder();
            CharBuffer charBuffer = CharBuffer.allocate(2);
            ByteBuffer byteBuffer = ByteBuffer.allocate((int) Math.ceil(encoder.maxBytesPerChar() * 2));
            if (normalized.codePoints().allMatch(
                    codePoint -> isWindows31jFullWidthCodePoint(codePoint, encoder, charBuffer, byteBuffer))) {
                return Optional.empty();
            }
            return Optional.of(ValidationError.ofField(annotation.errorCode(), fieldName));
        }

        private static Charset resolveWindows31j() {
            if (Charset.isSupported(WINDOWS_31J_NAME)) {
                return Charset.forName(WINDOWS_31J_NAME);
            }
            throw new IllegalStateException(WINDOWS_31J_NAME + " charset is required for FullWidthText validation");
        }

        private static boolean isWindows31jFullWidthCodePoint(
                int codePoint,
                CharsetEncoder encoder,
                CharBuffer charBuffer,
                ByteBuffer byteBuffer) {
            charBuffer.clear();
            appendCodePoint(codePoint, charBuffer);
            charBuffer.flip();

            encoder.reset();
            byteBuffer.clear();

            CoderResult coderResult = encoder.encode(charBuffer, byteBuffer, true);
            if (coderResult.isError()) {
                return false;
            }
            coderResult = encoder.flush(byteBuffer);
            if (coderResult.isError()) {
                return false;
            }
            return coderResult.isUnderflow() && byteBuffer.position() == 2;
        }

        private static void appendCodePoint(int codePoint, CharBuffer charBuffer) {
            if (Character.isBmpCodePoint(codePoint)) {
                charBuffer.put((char) codePoint);
                return;
            }
            charBuffer.put(Character.highSurrogate(codePoint));
            charBuffer.put(Character.lowSurrogate(codePoint));
        }
    }
}
