package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.lang.reflect.Array;
import java.util.Collection;
import java.util.Map;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * コレクション・配列・マップの最大件数を検証するカスタムアノテーション。
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface CollectionMaxSize {

    /**
     * 最大件数。
     */
    int max();

    /**
     * エラーコード。
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000032;

    /**
     * CollectionMaxSize 検証ロジック。
     */
    final class Rule implements ConstraintRule<CollectionMaxSize> {

        private Rule() {
        }

        private static final Rule INSTANCE = new Rule();

        /**
         * シングルトンインスタンスを返す。
         *
         * @return Rule インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(Object value, CollectionMaxSize annotation, String fieldName) {
            if (value == null) {
                return Optional.empty();
            }
            Integer size = resolveSize(value);
            if (size == null || size <= annotation.max()) {
                return Optional.empty();
            }
            return Optional.of(ValidationError.of(annotation.errorCode(), fieldName, String.valueOf(annotation.max())));
        }

        private static Integer resolveSize(Object value) {
            if (value instanceof Collection<?> collection) {
                return collection.size();
            }
            if (value instanceof Map<?, ?> map) {
                return map.size();
            }
            if (value.getClass().isArray()) {
                return Array.getLength(value);
            }
            return null;
        }
    }
}
