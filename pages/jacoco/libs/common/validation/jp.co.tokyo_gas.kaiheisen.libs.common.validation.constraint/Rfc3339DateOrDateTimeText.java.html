package jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.OffsetDateTime;
import java.time.format.DateTimeParseException;
import java.util.Optional;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * RFC3339 の日付または日時文字列を検証する。
 */
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.RECORD_COMPONENT, ElementType.FIELD, ElementType.PARAMETER})
public @interface Rfc3339DateOrDateTimeText {

    /**
     * エラーコード。
     *
     * @return エラーコード
     */
    String errorCode() default ErrorCodeConstants.ERROR_CODE_010000015;

    /**
     * RFC3339 の日付または日時文字列検証ルール。
     */
    final class Rule implements ConstraintRule<Rfc3339DateOrDateTimeText> {

        private static final Rule INSTANCE = new Rule();

        private Rule() {
        }

        /**
         * インスタンスを返す。
         *
         * @return インスタンス
         */
        public static Rule instance() {
            return INSTANCE;
        }

        @Override
        public Optional<ValidationError> validate(
                Object value,
                Rfc3339DateOrDateTimeText annotation,
                String fieldName) {
            if (value == null || value.toString().isBlank()) {
                return Optional.empty();
            }
            if (isDateOrDateTime(value.toString())) {
                return Optional.empty();
            }
            return Optional.of(ValidationError.ofField(annotation.errorCode(), fieldName));
        }

        private boolean isDateOrDateTime(String value) {
            try {
                OffsetDateTime.parse(value);
                return true;
            } catch (DateTimeParseException exception) {
                try {
                    LocalDateTime.parse(value);
                    return true;
                } catch (DateTimeParseException ignored) {
                    try {
                        LocalDate.parse(value);
                        return true;
                    } catch (DateTimeParseException dateException) {
                        return false;
                    }
                }
            }
        }
    }
}
