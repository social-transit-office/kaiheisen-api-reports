package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import java.util.concurrent.atomic.AtomicBoolean;

/** A bounded stream is renewed by LIST; a quiet healthy stream does not cause AUTO fallback. */
public final class StateMonitor implements AutoCloseable {
    private final KubeApi api;

    private final ModeState state;

    private final AtomicBoolean stopped = new AtomicBoolean();

    private Thread worker;

    /** Bind one client to the atomic state shared by this process. */
    public StateMonitor(KubeApi api, ModeState state) {
        this.api = api;
        this.state = state;
    }

    /** Start the lifecycle-managed virtual Watch worker. */
    public void start() {
        worker = Thread.ofVirtual().name("service-state-watch").start(this::run);
    }

    private void run() {
        int failures = 0;
        while (!stopped.get()) {
            try {
                KubeApi.Listing listing = api.list(Resources.STATE);
                state.accept(listing.item());
                if (listing.version().isBlank()) {
                    throw new IllegalStateException("MISSING_LIST_VERSION");
                }
                api.watch(
                        Resources.STATE,
                        listing.version(),
                        event -> {
                            String type = event.path("type").asText();
                            switch (type) {
                                case "ERROR" ->
                                        throw new KubeApi.Failure(
                                                event.path("object").path("code").asInt(500));
                                case "ADDED", "MODIFIED" -> state.accept(event.path("object"));
                                case "DELETED" -> state.accept(null);
                                default -> {
                                    /* Bookmarks do not change the state. */
                                }
                            }
                        });
                failures = 0;
            } catch (java.io.IOException | InterruptedException | RuntimeException failure) {
                if (stopped.get()) {
                    break;
                }
                state.unavailable(
                        failure instanceof KubeApi.Failure f
                                ? "HTTP_" + f.status
                                : "ACQUISITION_FAILED");
                try {
                    Thread.sleep(Math.min(5000, 250L << Math.min(failures++, 4)));
                } catch (InterruptedException interrupted) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        }
    }

    @Override
    public void close() {
        stopped.set(true);
        api.closeStream();
        if (worker != null) {
            worker.interrupt();
        }
    }
}
