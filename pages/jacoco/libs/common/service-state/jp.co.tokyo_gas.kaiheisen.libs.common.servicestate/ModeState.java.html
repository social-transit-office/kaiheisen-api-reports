package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import com.fasterxml.jackson.databind.JsonNode;
import java.time.Clock;
import java.time.ZoneId;
import java.time.ZonedDateTime;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.concurrent.atomic.AtomicReference;

/** One immutable publication keeps source identity, mode and acquisition health consistent. */
public final class ModeState {
    /** Persisted operator choices; AUTO uses the business calendar. */
    public enum Mode {
        AUTO,
        FORCE_STOP,
        FORCE_NORMAL
    }

    /** Immutable publication of the saved mode and acquisition health. */
    public record Source(
            String mode,
            String uid,
            String resourceVersion,
            String operationId,
            String health,
            String failure,
            String observedAt) {
    }

    private final Clock clock;

    private final ServicePolicy policy;

    private final AtomicReference<Source> source;

    /** Start in effective AUTO until a valid shared state is acquired. */
    public ModeState(Clock clock, ServicePolicy policy) {
        this.clock = clock;
        this.policy = policy;
        source =
                new AtomicReference<>(
                        new Source("AUTO", "", "", "", "UNAVAILABLE", "NOT_ACQUIRED", now()));
    }

    /** Atomically publish one acquired resource, including its identity and health. */
    public void accept(JsonNode cm) {
        if (cm == null || cm.isNull() || cm.isMissingNode()) {
            source.set(new Source("AUTO", "", "", "", "MISSING", "NOT_FOUND", now()));
            return;
        }
        String mode = cm.path("data").path("mode").asText("");
        boolean valid;
        try {
            Mode.valueOf(mode);
            valid = true;
        } catch (IllegalArgumentException invalid) {
            valid = false;
        }
        String uid = cm.path("metadata").path("uid").asText();
        String rv = cm.path("metadata").path("resourceVersion").asText();
        valid =
                valid
                        && !uid.isBlank()
                        && !rv.isBlank()
                        && !cm.path("data").path("operationId").asText().isBlank();
        source.set(
                new Source(
                        mode,
                        uid,
                        rv,
                        cm.path("data").path("operationId").asText(),
                        valid ? "SYNCED" : "INVALID",
                        valid ? "" : "INVALID_STATE",
                        now()));
    }

    /** Use AUTO locally while retaining the last acquired shared identity. */
    public void unavailable(String failure) {
        source.updateAndGet(
                old ->
                        new Source(
                                old.mode(),
                                old.uid(),
                                old.resourceVersion(),
                                old.operationId(),
                                "UNAVAILABLE",
                                failure,
                                now()));
    }

    /** Evaluate the effective mode and current JST time without network access. */
    public boolean stopped() {
        Source current = source.get();
        String mode = "SYNCED".equals(current.health()) ? current.mode() : "AUTO";
        return "FORCE_STOP".equals(mode)
                || ("AUTO".equals(mode)
                        && !policy.open(
                                ZonedDateTime.now(clock.withZone(ZoneId.of("Asia/Tokyo")))));
    }

    /** Expose this Pod's current configuration and diagnostic state. */
    public Map<String, Object> view(String service, String pod, String podUid, boolean enabled) {
        Source current = source.get();
        String effective = enabled && "SYNCED".equals(current.health()) ? current.mode() : "AUTO";
        boolean normal =
                !enabled
                        || "FORCE_NORMAL".equals(effective)
                        || ("AUTO".equals(effective)
                                && policy.open(
                                        ZonedDateTime.now(
                                                clock.withZone(ZoneId.of("Asia/Tokyo")))));
        Map<String, Object> value = new LinkedHashMap<>();
        value.put("protocolVersion", 1);
        value.put("service", service);
        value.put("pod", pod);
        value.put("podUid", podUid);
        value.put("enabled", enabled);
        value.put("storedMode", current.mode());
        value.put("effectiveMode", effective);
        value.put("state", normal ? "NORMAL" : "STOPPED");
        value.put("sourceUid", current.uid());
        value.put("resourceVersion", current.resourceVersion());
        value.put("operationId", current.operationId());
        value.put("health", enabled ? current.health() : "DISABLED");
        value.put("failure", current.failure());
        value.put("observedAt", current.observedAt());
        value.put("serverTime", now());
        value.put("policy", policy.view());
        return value;
    }

    private String now() {
        return clock.instant().toString();
    }
}
