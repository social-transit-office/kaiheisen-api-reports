package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.net.URI;
import java.net.URLEncoder;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.security.KeyStore;
import java.security.cert.CertificateFactory;
import java.time.Duration;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.function.Consumer;
import java.util.function.Supplier;
import javax.net.ssl.SSLContext;
import javax.net.ssl.TrustManagerFactory;

/** The Kubernetes REST API, using the mounted token (re-read on EVERY request) and CA. */
public final class KubeApi implements AutoCloseable {
    /** A credential-free Kubernetes API failure. */
    public static final class Failure extends RuntimeException {
        public final int status;

        /** Expose only the response status, never a signed URI or response body. */
        public Failure(int status) {
            super("Kubernetes HTTP " + status);
            this.status = status;
        }
    }

    /** A named LIST result and its collection resourceVersion. */
    public record Listing(JsonNode item, String version) {
    }

    private final URI base;

    private final String namespace;

    private final ObjectMapper json;

    private final Supplier<String> token;

    private final HttpClient client;

    private volatile InputStream activeStream;

    /** Create a namespace-scoped client with explicit TLS trust and token source. */
    public KubeApi(String namespace, ObjectMapper json)
            throws java.io.IOException, java.security.GeneralSecurityException {
        this(
                URI.create("https://kubernetes.default.svc"),
                namespace,
                json,
                () -> {
                    try {
                        return Files.readString(
                                        Path.of(
                                                "/var/run/secrets/kubernetes.io/serviceaccount/token"))
                                .strip();
                    } catch (IOException e) {
                        throw new IllegalStateException(
                                "Mounted ServiceAccount token is unavailable");
                    }
                },
                clusterClient());
    }

    /** Create a namespace-scoped client with explicit TLS trust and token source. */
    public KubeApi(
            URI base,
            String namespace,
            ObjectMapper json,
            Supplier<String> token,
            HttpClient client) {
        this.base = base;
        this.namespace = namespace;
        this.json = json;
        this.token = token;
        this.client = client;
    }

    private static HttpClient clusterClient()
            throws java.io.IOException, java.security.GeneralSecurityException {
        var certificates = CertificateFactory.getInstance("X.509");
        KeyStore store = KeyStore.getInstance(KeyStore.getDefaultType());
        store.load(null, null);
        try (InputStream in =
                Files.newInputStream(
                        Path.of("/var/run/secrets/kubernetes.io/serviceaccount/ca.crt"))) {
            int index = 0;
            for (var cert : certificates.generateCertificates(in)) {
                store.setCertificateEntry("cluster-" + index++, cert);
            }
        }
        var trust = TrustManagerFactory.getInstance(TrustManagerFactory.getDefaultAlgorithm());
        trust.init(store);
        SSLContext ssl = SSLContext.getInstance("TLS");
        ssl.init(null, trust.getTrustManagers(), null);
        return HttpClient.newBuilder()
                .sslContext(ssl)
                .connectTimeout(Duration.ofSeconds(5))
                .build();
    }

    /** Return the namespace to which this client is restricted. */
    public String namespace() {
        return namespace;
    }

    /** Read a Deployment or ReplicaSet in the application namespace. */
    public JsonNode apps(String resource) throws java.io.IOException, InterruptedException {
        return call("GET", "/apis/apps/v1/namespaces/" + namespace + "/" + resource, null);
    }

    /** Build a core Kubernetes API path within the application namespace. */
    public String path(String resource) {
        return "/api/v1/namespaces/" + namespace + "/" + resource;
    }

    /** Encode a query value or resource name as UTF-8. */
    public static String encode(String value) {
        return URLEncoder.encode(value, StandardCharsets.UTF_8);
    }

    /** Execute one request, rereading the mounted token and refusing non-success responses. */
    public JsonNode call(String method, String path, JsonNode body)
            throws java.io.IOException, InterruptedException {
        HttpRequest.Builder request =
                request(path)
                        .timeout(Duration.ofSeconds(5))
                        .header("Content-Type", "application/json");
        request.method(
                method,
                body == null
                        ? HttpRequest.BodyPublishers.noBody()
                        : HttpRequest.BodyPublishers.ofString(json.writeValueAsString(body)));
        var response = client.send(request.build(), HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() < 200 || response.statusCode() >= 300) {
            throw new Failure(response.statusCode());
        }
        return response.body().isBlank() ? json.createObjectNode() : json.readTree(response.body());
    }

    private HttpRequest.Builder request(String path) {
        return HttpRequest.newBuilder(base.resolve(path))
                .timeout(Duration.ofSeconds(30))
                .header("Authorization", "Bearer " + token.get())
                .header("Accept", "application/json");
    }

    /** Read one ConfigMap, representing only an actual 404 as missing. */
    public JsonNode get(String name) throws java.io.IOException, InterruptedException {
        try {
            return call("GET", path("configmaps/" + name), null);
        } catch (Failure f) {
            if (f.status == 404) {
                return null;
            }
            throw f;
        }
    }

    /** Read an operation Job after validating its allowed name. */
    public JsonNode job(String name) throws java.io.IOException, InterruptedException {
        if (!name.matches("svc-(stop|start|auto|status)-[a-z0-9-]+")) {
            throw new IllegalStateException("UNEXPECTED_JOB_NAME");
        }
        return call(
                "GET", "/apis/batch/v1/namespaces/" + namespace + "/jobs/" + encode(name), null);
    }

    /** Read one of the four fixed operation CronJobs. */
    public JsonNode cronJob(String name) throws java.io.IOException, InterruptedException {
        if (!name.matches("svc-(stop|start|auto|status)")) {
            throw new IllegalStateException("UNEXPECTED_CRONJOB_NAME");
        }
        return call(
                "GET",
                "/apis/batch/v1/namespaces/" + namespace + "/cronjobs/" + encode(name),
                null);
    }

    /** LIST the named state and retain the collection resourceVersion for Watch. */
    public Listing list(String name) throws java.io.IOException, InterruptedException {
        JsonNode result =
                call(
                        "GET",
                        path("configmaps") + "?fieldSelector=" + encode("metadata.name=" + name),
                        null);
        JsonNode items = result.path("items");
        if (items.size() > 1) {
            throw new IllegalStateException("Named ConfigMap list returned multiple objects");
        }
        return new Listing(
                items.isEmpty() ? null : items.get(0),
                result.path("metadata").path("resourceVersion").asText());
    }

    /** Consume a bounded Watch stream; close stalled streams to permit recovery. */
    public void watch(String name, String version, Consumer<JsonNode> event)
            throws java.io.IOException, InterruptedException {
        String path =
                path("configmaps")
                        + "?watch=true&allowWatchBookmarks=true&timeoutSeconds=20"
                        + "&fieldSelector="
                        + encode("metadata.name=" + name)
                        + "&resourceVersion="
                        + encode(version);
        var response =
                client.send(request(path).GET().build(), HttpResponse.BodyHandlers.ofInputStream());
        if (response.statusCode() != 200) {
            response.body().close();
            throw new Failure(response.statusCode());
        }
        activeStream = response.body();
        AtomicBoolean watchdog = new AtomicBoolean();
        Thread deadline =
                Thread.ofVirtual()
                        .name("service-state-watch-deadline")
                        .start(
                                () -> {
                                    try {
                                        Thread.sleep(25_000);
                                        watchdog.set(true);
                                        closeStream();
                                    } catch (InterruptedException renewed) {
                                        Thread.currentThread().interrupt();
                                    }
                                });
        try (var lines =
                new BufferedReader(
                        new InputStreamReader(response.body(), StandardCharsets.UTF_8))) {
            String line;
            while ((line = lines.readLine()) != null) {
                if (!line.isBlank()) {
                    event.accept(json.readTree(line));
                }
            }
            if (watchdog.get()) {
                throw new IOException("Watch did not renew within its deadline");
            }
        } finally {
            deadline.interrupt();
            activeStream = null;
        }
    }

    /** Create a runtime ConfigMap without overwriting an existing object. */
    public JsonNode createIfAbsent(String name, ObjectNode data)
            throws java.io.IOException, InterruptedException {
        if (!java.util.Set.of(Resources.STATE, Resources.LOCK, Resources.PENDING).contains(name)) {
            throw new IllegalArgumentException("Unexpected runtime ConfigMap");
        }
        ObjectNode resource =
                json.createObjectNode().put("apiVersion", "v1").put("kind", "ConfigMap");
        resource.putObject("metadata")
                .put("name", name)
                .put("namespace", namespace)
                .putObject("labels")
                .put(Resources.LABEL, "v1");
        resource.set("data", data);
        try {
            return call("POST", path("configmaps"), resource);
        } catch (Failure f) {
            if (f.status == 409) {
                return get(name);
            }
            throw f;
        }
    }

    /** Replace a runtime ConfigMap using its resourceVersion for conflict detection. */
    public JsonNode replace(String name, JsonNode resource)
            throws java.io.IOException, InterruptedException {
        if (!java.util.Set.of(Resources.STATE, Resources.LOCK, Resources.PENDING).contains(name)) {
            throw new IllegalArgumentException("Unexpected runtime ConfigMap");
        }
        // resourceVersion makes conflicting operations fail; never blindly retry a change.
        return call("PUT", path("configmaps/" + name), resource);
    }

    /** Delete only a known runtime ConfigMap. */
    public void delete(String name) throws java.io.IOException, InterruptedException {
        if (!java.util.Set.of(Resources.STATE, Resources.LOCK, Resources.PENDING).contains(name)) {
            throw new IllegalArgumentException("Unexpected runtime ConfigMap");
        }
        try {
            call(
                    "DELETE",
                    path("configmaps/" + name),
                    json.createObjectNode().put("apiVersion", "v1").put("kind", "DeleteOptions"));
        } catch (Failure f) {
            if (f.status != 404) {
                throw f;
            }
        }
    }

    /** List namespace-local Pods using the supplied selector. */
    public JsonNode pods(String selector) throws java.io.IOException, InterruptedException {
        return call("GET", path("pods") + "?labelSelector=" + encode(selector), null).path("items");
    }

    /** Cancel the active Watch body before shutting down its client. */
    public void closeStream() {
        InputStream stream = activeStream;
        if (stream != null) {
            try {
                stream.close();
            } catch (IOException ignored) {
                /* reconnection handles this */
            }
        }
    }

    @Override
    public void close() {
        closeStream();
        // A Watch may still be waiting for headers and have no activeStream yet.
        client.shutdownNow();
    }
}
