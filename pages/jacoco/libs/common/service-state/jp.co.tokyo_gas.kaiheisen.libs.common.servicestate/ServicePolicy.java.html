package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.LocalTime;
import java.time.ZonedDateTime;
import java.util.Arrays;
import java.util.Map;
import java.util.Set;
import java.util.function.Function;
import java.util.stream.Collectors;

/** The calendar is immutable deployment configuration. Manual modes always take precedence. */
public record ServicePolicy(
        LocalTime openTime,
        LocalTime closeTime,
        Set<DayOfWeek> closedWeekdays,
        Set<LocalDate> closedDates) {
    /** Validate the configured business interval and immutable calendar sets. */
    public ServicePolicy {
        if (openTime == null
                || closeTime == null
                || !openTime.isBefore(closeTime)
                || openTime.getSecond() != 0
                || closeTime.getSecond() != 0
                || openTime.getNano() != 0
                || closeTime.getNano() != 0) {
            throw new IllegalArgumentException("INVALID_SERVICE_HOURS");
        }
        closedWeekdays = Set.copyOf(closedWeekdays);
        closedDates = Set.copyOf(closedDates);
    }

    /** Parse release configuration; malformed dates, weekdays and intervals are rejected. */
    public static ServicePolicy parse(String open, String close, String weekdays, String dates) {
        return new ServicePolicy(
                LocalTime.parse(open),
                LocalTime.parse(close),
                csv(weekdays, DayOfWeek::valueOf),
                csv(dates, LocalDate::parse));
    }

    private static <T> Set<T> csv(String value, Function<String, T> parser) {
        if (value == null || value.isBlank()) {
            return Set.of();
        }
        return Arrays.stream(value.split(",", -1))
                .map(String::strip)
                .map(parser)
                .collect(Collectors.toUnmodifiableSet());
    }

    /** Evaluate the supplied JST date and time against the business calendar. */
    public boolean open(ZonedDateTime now) {
        return !closedWeekdays.contains(now.getDayOfWeek())
                && !closedDates.contains(now.toLocalDate())
                && !now.toLocalTime().isBefore(openTime)
                && now.toLocalTime().isBefore(closeTime);
    }

    /** Format the fixed service-hours error using the configured interval. */
    public String message() {
        var format = java.time.format.DateTimeFormatter.ofPattern("H:mm");
        return "サービス時間（"
                + openTime.format(format)
                + "～"
                + closeTime.format(format)
                + "）外のため使用できません。";
    }

    /** Expose this Pod's current configuration and diagnostic state. */
    public Map<String, Object> view() {
        return Map.of(
                "timeZone",
                "Asia/Tokyo",
                "openTime",
                openTime.toString(),
                "closeTime",
                closeTime.toString(),
                "closedWeekdays",
                closedWeekdays.stream().map(Enum::name).sorted().toList(),
                "closedDates",
                closedDates.stream().map(LocalDate::toString).sorted().toList());
    }
}
