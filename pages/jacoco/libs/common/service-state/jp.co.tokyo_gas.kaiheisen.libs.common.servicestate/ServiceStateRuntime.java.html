package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import com.fasterxml.jackson.databind.ObjectMapper;
import io.quarkus.runtime.ShutdownEvent;
import io.quarkus.runtime.StartupEvent;
import jakarta.annotation.Priority;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.event.Observes;
import jakarta.inject.Inject;
import java.time.Clock;
import java.util.Map;
import org.eclipse.microprofile.config.ConfigProvider;

/** Lifecycle shared by the BFF and five MA deployments. No request performs network I/O. */
@ApplicationScoped
public class ServiceStateRuntime {
    @Inject ObjectMapper json;

    private boolean enabled;

    private String service;

    private ServicePolicy policy;

    private ModeState state;

    private KubeApi api;

    private StateMonitor monitor;

    private volatile boolean closing;

    private Thread bootstrap;

    private final java.util.concurrent.locks.ReentrantLock lifecycle =
            new java.util.concurrent.locks.ReentrantLock();

    void start(@Observes @Priority(100) StartupEvent event) {
        enabled = Boolean.parseBoolean(setting("enabled", "false"));
        service =
                setting(
                        "service",
                        System.getenv().getOrDefault("SERVICE_STATE_SERVICE", "unknown"));
        policy =
                ServicePolicy.parse(
                        setting("open-time", "07:30"),
                        setting("close-time", "21:00"),
                        setting("closed-weekdays", ""),
                        setting("closed-dates", ""));
        state = new ModeState(Clock.systemUTC(), policy);
        if (enabled) {
            if (!Resources.SERVICES.contains(service)) {
                throw new IllegalArgumentException("UNKNOWN_SERVICE_STATE_SERVICE");
            }
            bootstrap =
                    Thread.ofVirtual()
                            .name("service-state-bootstrap")
                            .start(
                                    () -> {
                                        while (!closing) {
                                            try {
                                                KubeApi candidate =
                                                        new KubeApi(
                                                                System.getenv()
                                                                        .getOrDefault(
                                                                                "POD_NAMESPACE",
                                                                                "kaiheisen"),
                                                                json);
                                                lifecycle.lock();
                                                try {
                                                    if (closing) {
                                                        candidate.close();
                                                        return;
                                                    }
                                                    api = candidate;
                                                    monitor = new StateMonitor(api, state);
                                                    monitor.start();
                                                } finally {
                                                    lifecycle.unlock();
                                                }
                                                return;
                                            } catch (java.io.IOException
                                                    | java.security.GeneralSecurityException
                                                    | RuntimeException unavailable) {
                                                state.unavailable("CLIENT_INITIALIZATION_FAILED");
                                                try {
                                                    Thread.sleep(1000);
                                                } catch (InterruptedException interrupted) {
                                                    Thread.currentThread().interrupt();
                                                    return;
                                                }
                                            }
                                        }
                                    });
        }
    }

    /** Resolve a release setting with its local-development default. */
    public static String setting(String key, String fallback) {
        return ConfigProvider.getConfig()
                .getOptionalValue("kaiheisen.service-state." + key, String.class)
                .orElse(fallback);
    }

    /** Return the current request decision without network access. */
    public boolean stopped() {
        return enabled && state.stopped();
    }

    /** Return the configured fixed error description. */
    public String message() {
        return policy.message();
    }

    /** Return the Deployment service identifier. */
    public String service() {
        return service;
    }

    /** Expose this Pod's current configuration and diagnostic state. */
    public Map<String, Object> view() {
        return state.view(
                service,
                System.getenv().getOrDefault("POD_NAME", "local"),
                System.getenv().getOrDefault("POD_UID", "local"),
                enabled);
    }

    void stop(@Observes ShutdownEvent event) {
        lifecycle.lock();
        try {
            closing = true;
            if (bootstrap != null) {
                bootstrap.interrupt();
            }
            if (monitor != null) {
                monitor.close();
            }
            if (api != null) {
                api.close();
            }
        } finally {
            lifecycle.unlock();
        }
    }
}
