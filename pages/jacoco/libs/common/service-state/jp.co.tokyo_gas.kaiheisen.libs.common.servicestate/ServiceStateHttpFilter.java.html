package jp.co.tokyo_gas.kaiheisen.libs.common.servicestate;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import io.quarkus.vertx.http.runtime.filters.Filters;
import io.vertx.ext.web.RoutingContext;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.event.Observes;
import jakarta.inject.Inject;
import java.util.List;
import java.util.Map;
import java.util.Set;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.CorrelationIdResolver;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;

/** Runs before REST matching, authentication and the CORS preflight shortcut. */
@ApplicationScoped
public class ServiceStateHttpFilter {
    private static final Set<String> HEALTH =
            Set.of("/q/health", "/q/health/ready", "/q/health/live", "/q/health/started");

    @Inject ServiceStateRuntime runtime;

    @Inject ObjectMapper json;

    void register(@Observes Filters filters) {
        filters.register(this::handle, 10_000);
    }

    private void handle(RoutingContext context) {
        String path = context.normalizedPath();
        String method = context.request().method().name();
        boolean read = "GET".equals(method) || "HEAD".equals(method);
        if (read && "/q/service-state".equals(path)) {
            respond(context, 200, runtime.view());
        } else if (read && HEALTH.contains(path)) {
            context.next();
        } else if (runtime.stopped()) {
            String id =
                    respond(
                            context,
                            503,
                            Map.of(
                                    "errorCode",
                                    Resources.ERROR_CODE_990000002,
                                    "errorDescription",
                                    runtime.message(),
                                    "errorDetailList",
                                    List.of()));
            new JbossStructuredLogger(ServiceStateHttpFilter.class)
                    .info(
                            new LogEvent(
                                    "service.state.request.rejected",
                                    "Service unavailable",
                                    "bff-livalit".equals(runtime.service())
                                            ? LogLayer.BFF
                                            : LogLayer.MA,
                                    503,
                                    Resources.ERROR_CODE_990000002,
                                    Map.of(
                                            "httpMethod",
                                            method,
                                            "path",
                                            path,
                                            "correlationId",
                                            id)));
        } else {
            context.next();
        }
    }

    private String respond(RoutingContext context, int status, Object body) {
        String id = CorrelationIdResolver.resolve(context.request().getHeader("X-Correlation-Id"));
        try {
            byte[] encoded = json.writeValueAsBytes(body);
            var response =
                    context.response()
                            .setStatusCode(status)
                            .putHeader("Content-Type", "application/json; charset=UTF-8")
                            .putHeader("Cache-Control", "no-store")
                            .putHeader("X-Correlation-Id", id);
            if ("HEAD".equals(context.request().method().name())) {
                response.putHeader("Content-Length", Integer.toString(encoded.length)).end();
            } else {
                response.end(io.vertx.core.buffer.Buffer.buffer(encoded));
            }
        } catch (JsonProcessingException invalid) {
            context.fail(invalid);
        }
        return id;
    }
}
