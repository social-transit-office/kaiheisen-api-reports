@Path(CustomerInfoGetEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "オンライン開栓報告お客さま情報取得", description = "MA-001 オンライン開栓報告お客さま情報取得 API")
public class CustomerInfoGetResource extends BaseResource<CustomerInfoGetRequest> {
    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final GasmtrdDataAccess gasmtrdDataAccess;

    private final KiycstdDataAccess kiycstdDataAccess;

    private final SuspendedRemovedErrorJudgment suspendedRemovedErrorJudgment;

    private final OpeningClosingInProgressErrorJudgment openingClosingInProgressErrorJudgment;

    private final OriginGasMeterOpeningInProgressConfirmation originGasMeterOpeningInProgressConfirmation;

    private final MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening;

    private final RetailTypeJudgment retailTypeJudgment;

    private final MeterReadingResultInfoAcquisition meterReadingResultInfoAcquisition;

    private final String khsSyoriKbn;

    private final String shrKbn = "2";

    /**

     */

    public CustomerInfoGetResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @CustomerInfoGetConfiguration

            @CustomerInfoGetConfiguration

            @CustomerInfoGetConfiguration
            MeterReadingInfoAcquisitionForOpening meterReadingInfoAcquisitionForOpening,
            @CustomerInfoGetConfiguration

            @CustomerInfoGetConfiguration

            KiycstdDataAccess kiycstdDataAccess,
            KaiheisenConstantService kaiheisenConstantService

        super(requestContext, requestDtoValidator, clock);
        this.suspendedRemovedErrorJudgment = Objects.requireNonNull(

                "suspendedRemovedErrorJudgment");
        this.openingClosingInProgressErrorJudgment = Objects.requireNonNull(

        this.originGasMeterOpeningInProgressConfirmation = Objects.requireNonNull(
                originGasMeterOpeningInProgressConfirmation,
                "originGasMeterOpeningInProgressConfirmation");

                "meterReadingInfoAcquisitionForOpening");

        this.kiycstdDataAccess = Objects.requireNonNull(kiycstdDataAccess, "kiycstdDataAccess");
        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");
        this.khsSyoriKbn = RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.OPENING_PROCESS);

    /**
     * MA-001 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "オンライン開栓お客さま情報取得",
            description = "オンライン開栓お客さま情報を取得する",
            operationId = "getCustomerInfoGet")
    @APIResponse(
            responseCode = "200",
            description = "オンライン開栓お客さま情報取得結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CustomerInfoGetResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public CustomerInfoGetResponse endpoint(@BeanParam CustomerInfoGetRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();
            KiycstdRecord kiycstdRecord = null;
            // DA-033-R006 ガスメーター設置場所＜導管＞情報の取得
            GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, knskKjnYmd);
            // SV-136 休止・撤去済エラー判定
            suspendedRemovedErrorJudgment.execute(gmtsetdRecord.kngmStbJtkb());
            // SV-138 開栓中・閉栓中エラー判定
            openingClosingInProgressErrorJudgment.execute(gmtsetdRecord.gmtKhsJtkb(), this.khsSyoriKbn);
            // SV-269 元ガスメーター開栓中確認
            originGasMeterOpeningInProgressConfirmation.execute(gmtsetdRecord.ckrtGmscJtkb(), gmtsetdRecord.mtgMtstNo());
            // DA-110-R003
            GasmtrdRecord gasmtrdRecord = requireGasmtrd(gasMtstNo, null);
            // SV-027 検針情報(開栓)を取得する
            Optional<MeterReadingInfoAcquisitionForOpeningOutput> meterReadingInfoAcquisitionForOpeningOutput =
                    Optional.of(gmtsetdRecord.gasMtstNo())
                    .map(v -> gmtsetdRecord.gasMtstNo())
                    .map(meterReadingInfoAcquisitionForOpening::execute);
            // SV-001 小売判定用（TG小売/他小売）
            RetailTypeJudgmentOutput retailTypeJudgmentOutput = retailTypeJudgment.execute(shrKbn, gasMtstNo, knskKjnYmd);
            // SV-001にて使用者契約お客さま登録番号が未設定ではない場合
            if (retailTypeJudgmentOutput.tkssvkyInf().syoKyoTrno() != null) {
            // DA-104-R001 契約お客さま情報＜導管＞情報の取得
                kiycstdRecord = requireKiycstd(retailTypeJudgmentOutput.tkssvkyInf().syoKyoTrno());
            }
            // SV-042 検針結果情報取得
            MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput =
                    meterReadingResultInfoAcquisition.execute(gmtsetdRecord.hesnJysb()
                            , gasMtstNo
                            , retailTypeJudgmentOutput.kuHntKbn()
                            , meterReadingInfoAcquisitionForOpeningOutput.get().knsYmd()
                            , meterReadingInfoAcquisitionForOpeningOutput.get().gmtShjSu()
                            , meterReadingInfoAcquisitionForOpeningOutput.get().knkkSsjyShbt());
            return toResponse(
                    gasMtstNo,
                    gmtsetdRecord,
                    gasmtrdRecord,
                    kiycstdRecord,
                    meterReadingInfoAcquisitionForOpeningOutput,
                    retailTypeJudgmentOutput,
                    meterReadingResultInfoAcquisitionOutput);
        });
    }

    /**
     * レスポンス成型する。
     *
     * @param gmtsetdRecord ガスメーター設置場所＜導管＞情報
     * @param gasmtrdRecord ガスメーター情報
     * @param kiycstdRecord 契約お客さま＜導管＞情報
     * @param meterReadingInfoAcquisitionForOpeningOutput 検針情報取得（開栓）の出力
     * @param retailTypeJudgmentOutput 小売判定の出力
     * @param meterReadingResultInfoAcquisitionOutput 検針結果情報の出力
     * @return CustomerInfoGetResponse レスポンス
     */
    private CustomerInfoGetResponse toResponse(
            Long gasMtstNo,
            GmtsetdRecord gmtsetdRecord,
            GasmtrdRecord gasmtrdRecord,
            KiycstdRecord kiycstdRecord,
            Optional<MeterReadingInfoAcquisitionForOpeningOutput> meterReadingInfoAcquisitionForOpeningOutput,
            RetailTypeJudgmentOutput retailTypeJudgmentOutput,
            MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput
    ) {
        return new CustomerInfoGetResponse(
                createGmtsetdInfoResponse(gmtsetdRecord, gasMtstNo),
                createGasmtrdInfoResponse(gasmtrdRecord),
                createKnsKkaInfoResponse(meterReadingResultInfoAcquisitionOutput),
                createTkssvkyInfoResponse(retailTypeJudgmentOutput),
                createKiycstdInfoResponse(kiycstdRecord),
                requestContext.getCorrelationId().orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"))
                );

    }

    /**
     * DA-033-R006取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター＜導管＞の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param gmtsetdNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return gmtsetdRecords
     */
    private GmtsetdRecord requireGmtsetd(Long gmtsetdNo, LocalDate knskKjnYmd) {
        Optional<GmtsetdRecord> gmtsetdRecords =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gmtsetdNo, knskKjnYmd);
        if (gmtsetdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
            gmtsetdDataAccess.tableName(),
            GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                    .put("gasMtstNo", gmtsetdNo)
                    .put("tkysYmd", knskKjnYmd)
                    .put("tyenYmd", knskKjnYmd));
        }
        return gmtsetdRecords.get();
    }

    /**
     * ガスメーター設置場所＜導管＞情報のレスポンスを生成する。
     *
     * @return GmtsetdInfoResponse レスポンス
     */
    private GmtsetdInfoResponse createGmtsetdInfoResponse(GmtsetdRecord gmtsetdRecord, Long gasMtstNo) {
        return new GmtsetdInfoResponse(
                gasMtstNo,
                gmtsetdRecord.kngmStbJtkb(),
                gmtsetdRecord.gmtKhsJtkb(),
                gmtsetdRecord.hesnSdnShbt(),
                gmtsetdRecord.hesnJysb(),
                formatDate(gmtsetdRecord.kngmSbmsYmd()),
                formatDate(gmtsetdRecord.kasYmd()),
                formatDate(gmtsetdRecord.hesnYmd()),
                formatDate(gmtsetdRecord.kusYmd()));
    }

    /**
     * DA-110-R007取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター＜導管＞の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return gasmtrdRecords
     */
    private GasmtrdRecord requireGasmtrd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GasmtrdRecord> gasmtrdRecords =
                gasmtrdDataAccess.findByTtkGmsbNoAndGmtThkgYmd(gasMtstNo, knskKjnYmd);
        if (gasmtrdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
            gasmtrdDataAccess.tableName(),
            GasmtrdDataAccess.CallSpecs.FIND_BY_TTK_GMSB_NO_AND_GMT_THKG_YMD.identifier(),
                new FlatJsonBuilder()
                    .put("ttkGmsbNo", gasMtstNo)
                    .put("gmtThkgkYmd", knskKjnYmd));
        }
        return gasmtrdRecords.get();
    }

    /**
     * ガスメーター情報のレスポンスを生成する。
     *
     * @return GasmtrdInfoResponse レスポンス
     */
    private GasmtrdInfoResponse createGasmtrdInfoResponse(GasmtrdRecord gasmtrdRecord) {
        return new GasmtrdInfoResponse(
                gasmtrdRecord.sjsuMincSu());
    }

    /**
     * 検針結果情報のレスポンスを生成する。
     *
     * @return KnsKkaInfoResponse レスポンス
     */
    private KnsKkaInfoResponse createKnsKkaInfoResponse(MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput) {
        return new KnsKkaInfoResponse(
                meterReadingResultInfoAcquisitionOutput.knkkSsjyShbt(),
                formatDate(meterReadingResultInfoAcquisitionOutput.sokKnsnYmd()),
                meterReadingResultInfoAcquisitionOutput.gmtShjSu()
                );
    }

    /**
     * 託送サービス契約情報のレスポンスを生成する。
     *
     * @return KiycstInfoResponse レスポンス
     */
    private TkssvkyInfoResponse createTkssvkyInfoResponse(RetailTypeJudgmentOutput retailTypeJudgmentOutput) {
        return new TkssvkyInfoResponse(
                retailTypeJudgmentOutput.tkssvkyInf().tksuSvkyJtkb(),
                retailTypeJudgmentOutput.tkssvkyInf().tksuKijsCd());
    }

    /**
     * DA-104-R001取得処理
     * 主な仕様:
     * - 契約お客さま登録番号を引数に契約お客さま＜導管＞の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param kiyTrkNo 契約お客さま登録番号
     * @return kiycstdRecords
     */
    private KiycstdRecord requireKiycstd(Long kiyTrkNo) {
        Optional<KiycstdRecord> kiycstdRecords =
                kiycstdDataAccess.findByKyoTrkNo(kiyTrkNo);
        if (kiycstdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
            kiycstdDataAccess.tableName(),
            KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                new FlatJsonBuilder()
                    .put("kyoTrkNo", kiyTrkNo));
        }
        return kiycstdRecords.get();
    }

    /**
     * お客さま情報1のレスポンスを生成する。
     *
     * @return CstInf1Response レスポンス
     */
    private KiycstdInfoResponse createKiycstdInfoResponse(KiycstdRecord kiycstdRecord) {
        if (kiycstdRecord == null) {
            return new KiycstdInfoResponse(
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null,
                    null);
        } else {
            return new KiycstdInfoResponse(
                kiycstdRecord.ctnmKn1(),
                kiycstdRecord.ctnmKj1(),
                kiycstdRecord.ctnmKjhkFlg(),
                kiycstdRecord.ctnmKn2(),
                kiycstdRecord.ctnmKj2(),
                kiycstdRecord.ctnm2HujnShbt(),
                kiycstdRecord.ctnm2HshiShbt(),
                kiycstdRecord.ctnm1HujnShbt(),
                kiycstdRecord.ctnm1HshiShbt(),
                kiycstdRecord.kyoTrkNo());
        }
    }

    /**
     * LocalDate型の日付をString型に変換
     *
     * @return value String型の日付
     */
    private static String formatDate(LocalDate value) {
        return value == null ? null : value.toString();
    }
