    private OpeningCommonRecoveryResponse handle(OpeningCommonRecoveryRequest request) {
        LocalDate kasgYmd = parseRequestDate(request.kasgYmd());

        PreOpeningState preOpeningState = loadPreOpeningState(request, kasgYmd);
        CustomerState customerState = loadCustomerState(request);

        if (SECURITY_RESTORATION_OPENING_REASON.equals(request.kasJysb())
                && ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE.equals(
                        preOpeningState.hesnJysb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000094);
        }

        boolean hasInvalidTransportContract = tkssvkyDataAccess
                .findBySyoKyoTrnoAndTksuSvkyJtkb(
                        request.syoKyoTrno(),
                        invalidTransportServiceContractStatus)
                .isPresent();
        boolean wkSyoKyoTrnoUmFlg = hasInvalidTransportContract;
        JudgmentState judgmentState = processJudgmentState(
                request,
                preOpeningState,
                customerState,
                wkSyoKyoTrnoUmFlg);

        openingInProgressCheckAtOpening.execute(preOpeningState.gmtKhsJtkb());
        openingDateCheckSeriesInstallationStatus.execute(
                preOpeningState.systemDateCkrtGmscJtkb(),
                preOpeningState.ksenYmdCkrtGmscJtkb());
        originMeterClosingInProgressCheck.execute(
                preOpeningState.ksenYmdCkrtGmscJtkb(),
                preOpeningState.originGmtKhsJtkb());
        openingDateCheckFutureDate.execute(kasgYmd);
        if (preOpeningState.tusoGmtkYmd() != null) {
            openingDateCheckInstallationDate.execute(kasgYmd, preOpeningState.tusoGmtkYmd());
        }
        if (preOpeningState.ksenHkkKnuKigenYmd() != null) {
            openingDateCheckRecentMeterReadingDate.execute(
                    kasgYmd,
                    preOpeningState.ksenHkkKnuKigenYmd());
        }
        if (preOpeningState.kasYmd() != null) {
            openingDateCheckSeriesDedicatedMeterOpeningDate.execute(
                    preOpeningState.ksenYmdCkrtGmscJtkb(),
                    preOpeningState.kasYmd(),
                    kasgYmd);
        }
        openingDateCheckEndDate.execute(
                preOpeningState.tksuKijsCd(),
                preOpeningState.rkkiKnrnJtkb(),
                request.kasJysb(),
                preOpeningState.rkkiEndYmd(),
                kasgYmd);
        openingDateCheckSwitchingCompletion.execute(preOpeningState.swKnsUmFlg());
        sameDayOpeningDateCheckMultipleOpening.execute(preOpeningState.kasYmd(), kasgYmd);
        otherRetailTransportContractCheckAtOpening.execute(
                judgmentState.kiyUmFlg(),
                preOpeningState.taksmtrHntrUmFlg(),
                preOpeningState.taksmtrKtrUmFlg(),
                preOpeningState.tksuKijsCd(),
                preOpeningState.tkykCisjCd());

        if (SECURITY_RESTORATION_OPENING_REASON.equals(request.kasJysb())) {
            openingReasonCheck.execute(
                    judgmentState.kiyUmFlg(),
                    preOpeningState.ctnmKn1Hsen(),
                    judgmentState.chkzKsenCtnmKn(),
                    request.kasJysb(),
                    preOpeningState.ksenYmdTskgmstUmFlg(),
                    preOpeningState.hesnJysb());
        }

        openingReadingCheckPreviousReading.execute(
                preOpeningState.hesnJysb(),
                request.gmtShjSu(),
                preOpeningState.rawOutput() != null && preOpeningState.rawOutput().knsKkaKsenInf() != null
                        ? preOpeningState.rawOutput().knsKkaKsenInf().skinGmtShjSu()
                        : null);
        tgPipelineReadingJudgment.execute(
                OPENING_PROCESSING_DIVISION,
                preOpeningState.sjskHticSu(),
                preOpeningState.sjsuMincSu(),
                request.gmtShjSu());
