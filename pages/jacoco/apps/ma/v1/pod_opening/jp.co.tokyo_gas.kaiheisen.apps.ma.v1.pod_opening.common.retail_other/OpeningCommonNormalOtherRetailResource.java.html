    private static List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SnyInf> toAlternativePipelineSnyInfList(
            PreOpeningInfoAcquisitionOutput preOpeningInfo,
            TransportServiceContractNumberConfirmationAtOpeningOutput transportOutput) {
        if (preOpeningInfo == null || preOpeningInfo.snyInfList() == null) {
            return List.of();
        }
        List<PreOpeningSnyInf> preOpeningDedicatedMeters = preOpeningInfo.snyInfList();
        List<TransportServiceContractConfirmationDedicatedMeterInfo> confirmedDedicatedMeters =
                transportOutput == null || transportOutput.snyInfList() == null
                        ? List.of()
                        : transportOutput.snyInfList();
        List<jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SnyInf> result = new ArrayList<>();
        for (int index = 0; index < preOpeningDedicatedMeters.size(); index++) {
            PreOpeningSnyInf preOpeningDedicatedMeter = preOpeningDedicatedMeters.get(index);
            result.add(new jp.co.tokyo_gas.kaiheisen.libs.sv.api.opening.model.SnyInf(

                    resolveConfirmedDedicatedTransportServiceContractNumber(
                            preOpeningDedicatedMeter.tksuSvkyNo(),
                            confirmedDedicatedMeters,
                            index)));
        }
        return List.copyOf(result);
    }

    private static Long resolveConfirmedDedicatedTransportServiceContractNumber(
            Long transportServiceContractNumber,
            List<TransportServiceContractConfirmationDedicatedMeterInfo> confirmedDedicatedMeters,
            int index) {
        if (index < 0 || index >= confirmedDedicatedMeters.size()) {
            return transportServiceContractNumber;
        }
        TransportServiceContractConfirmationDedicatedMeterInfo confirmedDedicatedMeter =
                confirmedDedicatedMeters.get(index);
        if (confirmedDedicatedMeter == null || confirmedDedicatedMeter.kktTksuSvkyNo() == null) {
            return transportServiceContractNumber;
        }
        return confirmedDedicatedMeter.kktTksuSvkyNo();
    }
