    public OpeningCommonForceTgRetailResponse endpoint(OpeningCommonForceTgRetailRequest request) {
        return execute(request, () -> {
            AreaInfo areaInfo = resolveAreaInfo(request);
            CustomerInfo customerInfo = resolveCustomerInfo(request);
            String decodedFunctionId = decodeFunctionId(request.hukkSysShbt());
            AuditContext auditContext = AuditContext.of(UPD_USER_ID, decodedFunctionId);

            kseflDataAccess.insert(
                    toKseflWriteRecord(request, areaInfo, customerInfo),
                    auditContext);

            KasnhkRecord kasnhkRecord = kasnhkDataAccess
                    .findByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndKasHukkJtkb(
                            request.gmb(),
                            parseRequestDate(request.kasgYmd()),
                            parseRequestDate(request.kasHukkYmd()),
                            toLvltAnknNo(request.sgyAnknNo()),
                            openingReportStatusReporting)
                    .orElse(null);

            if (kasnhkRecord != null) {
                kasnhkDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndRecUpdJnjNo(
                        toKasnhkUpdateRecord(kasnhkRecord),
                        java.util.Set.of(KasnhkUpdatableColumn.KAS_HUKK_JTKB),
                        new KasnhkU002UpdateKey(
                                kasnhkRecord.gmb(),
                                kasnhkRecord.kasgYmd(),
                                kasnhkRecord.kasHukkYmd(),
                                kasnhkRecord.lvltAnknNo(),
                                kasnhkRecord.recUpdJnjNo()),
                        auditContext);
            }

            return new OpeningCommonForceTgRetailResponse(
                    resolveResponseCorrelationId());
        });
    }

    private AreaInfo resolveAreaInfo(OpeningCommonForceTgRetailRequest request) {
        String regionCode = normalizeOptional(request.cktkCd());
        String baseCode = null;
        Long gasMtstNo = request.gasMtstNo();
        if (gasMtstNo == null || gasMtstNo == 0L) {
            return new AreaInfo(regionCode, baseCode);
        }

        LocalDate criteriaDate = requestContext.getSystemDatetime()
                .map(java.time.LocalDateTime::toLocalDate)
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"));
        KbagmstRecord kbagmstRecord = kbagmstDataAccess
                .findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, toYearMonth(criteriaDate))
                .orElse(null);
        if (kbagmstRecord != null) {
            return new AreaInfo(kbagmstRecord.kbabCktkCd(), kbagmstRecord.kbabKtnCd());
        }
        if (regionCode == null) {
            throw DataInconsistencySupport.create(
                    kbagmstDataAccess.tableName(),
                    KbagmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", gasMtstNo)
                            .put("knskKjnYm", toYearMonth(criteriaDate).toString()));
        }
        return new AreaInfo(regionCode, baseCode);
    }

    private CustomerInfo resolveCustomerInfo(OpeningCommonForceTgRetailRequest request) {
        if (normalizeOptional(request.ctnmKn1()) != null) {
            return toRequestCustomerInfo(request, request.bldnKn(), request.bldnKj());
        }

        KiycstkRecord kiycstkRecord = kiycstkDataAccess.findByKyoTrkNo(request.syoKyoTrno())
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kiycstkDataAccess.tableName(),
                        KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                        new FlatJsonBuilder().put("kyoTrkNo", request.syoKyoTrno())));

        String buildingNo = kiycstkRecord.bldNo();
        if (buildingNo == null) {
            return toCustomerInfo(kiycstkRecord, null);
        }

        TatemonRecord tatemonRecord = tatemonDataAccess.findByBldNo(buildingNo)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        tatemonDataAccess.tableName(),
                        TatemonDataAccess.CallSpecs.FIND_BY_BLD_NO.identifier(),
                        new FlatJsonBuilder().put("bldNo", buildingNo)));

        return toCustomerInfo(kiycstkRecord, tatemonRecord);
    }

        String normalized = value == null ? null : value.trim();

            return null;
        }

            return LocalDate.parse(normalized);
        } catch (DateTimeParseException ignored) {
            try {
                return OffsetDateTime.parse(normalized).toLocalDate();

                throw new IllegalArgumentException("date must follow RFC3339", ex);
            }
        }
    }
