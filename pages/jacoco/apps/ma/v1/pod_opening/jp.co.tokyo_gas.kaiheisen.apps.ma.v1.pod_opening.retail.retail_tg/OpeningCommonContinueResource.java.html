    private OpeningCommonContinueRequest normalizeRequest(OpeningCommonContinueRequest request) {
        String normalizedGmb = stripTrailingSpaces(request.gmb());
        if (Objects.equals(normalizedGmb, request.gmb())) {
            return request;
        }
        return mapRecord(
                request,
                OpeningCommonContinueRequest.class,

    private static String stripTrailingSpaces(String value) {
        if (value == null || value.isEmpty()) {
            return value;
        }
        int endIndex = value.length();
        while (endIndex > 0 && value.charAt(endIndex - 1) == ' ') {
            endIndex--;
        }
        return value.substring(0, endIndex);
    }

    private static LocalDate localDate(Object value) {
        if (value instanceof LocalDate localDate) {
            return localDate;
        }
        if (value instanceof LocalDateTime localDateTime) {
            return localDateTime.toLocalDate();
        }
        if (value instanceof OffsetDateTime offsetDateTime) {
            return offsetDateTime.toLocalDate();
        }

    private static Object read(Object source, String property) {
        if (source == null || property == null || property.isBlank()) {
            return null;
        }
        try {
            Method method = source.getClass().getMethod(property);
            return method.invoke(source);
        } catch (NoSuchMethodException ex) {
            return null;
        } catch (IllegalAccessException | InvocationTargetException ex) {
            throw new IllegalStateException("record property read failed: " + property, ex);
        }
    }

    private static <T> T mapRecord(Object source, Class<T> targetType) {

    private static <T> T mapRecord(Object source, Class<T> targetType, Map<String, Object> overrides) {
        if (source == null) {
            if (overrides == null || overrides.isEmpty()) {
                return null;
            }
        }
        if (overrides == null) {
            overrides = Map.of();
        }
        if (source != null && overrides.isEmpty() && targetType.isInstance(source)) {
            return targetType.cast(source);
        }
        if (!targetType.isRecord()) {
            return null;
        }
        try {
            RecordComponent[] components = targetType.getRecordComponents();
            Class<?>[] parameterTypes = new Class<?>[components.length];
            Object[] values = new Object[components.length];
            for (int index = 0; index < components.length; index++) {
                RecordComponent component = components[index];
                parameterTypes[index] = component.getType();
                Object value = overrides.containsKey(component.getName())
                        ? overrides.get(component.getName())
                        : read(source, component.getName());
                values[index] = convert(value, component.getGenericType(), component.getType());
            }
            Constructor<T> constructor = targetType.getDeclaredConstructor(parameterTypes);
            constructor.setAccessible(true);
            return constructor.newInstance(values);
        } catch (ReflectiveOperationException ex) {
            throw new IllegalStateException("record mapping failed: " + targetType.getName(), ex);
        }
    }

    private static Map<String, Object> overrides(Object... entries) {
        if (entries.length % 2 != 0) {
            throw new IllegalArgumentException("override entries must be key-value pairs");
        }
        Map<String, Object> values = new LinkedHashMap<>();
        for (int index = 0; index < entries.length; index += 2) {
            values.put((String) entries[index], entries[index + 1]);
        }
        return values;
    }

    private static Object convert(Object value, Type genericType, Class<?> targetType) {
        if (value == null) {
            return primitiveDefault(targetType);
        }
        if (targetType.isInstance(value)) {
            return value;
        }
        if (targetType == LocalDate.class) {
            return localDate(value);
        }
        if (targetType == Long.class || targetType == long.class) {
            return longValue(value);
        }
        if (targetType == Integer.class || targetType == int.class) {
            return integer(value);
        }
        if (targetType == BigDecimal.class) {
            return bigDecimal(value);
        }
        if (targetType == String.class) {
            return string(value);
        }
        if (targetType == Boolean.class || targetType == boolean.class) {
            return bool(value);
        }
        if (List.class.isAssignableFrom(targetType) && value instanceof List<?> listValue) {
            Class<?> elementType = resolveElementType(genericType);
            return mapList(listValue, elementType);
        }
        if (targetType.isRecord()) {
            return mapRecord(value, targetType);
        }
        return null;
    }

    private static Object primitiveDefault(Class<?> targetType) {
        if (!targetType.isPrimitive()) {
            return null;
        }
        if (targetType == boolean.class) {
            return false;
        }
        if (targetType == int.class) {
            return 0;
        }
        if (targetType == long.class) {
            return 0L;
        }
        if (targetType == double.class) {
            return 0D;
        }
        if (targetType == float.class) {
            return 0F;
        }
        if (targetType == short.class) {
            return (short) 0;
        }
        if (targetType == byte.class) {
            return (byte) 0;
        }
        if (targetType == char.class) {
            return '\0';
        }
        return null;
    }

    private static Class<?> resolveElementType(Type genericType) {
        if (genericType instanceof ParameterizedType parameterizedType
                && parameterizedType.getActualTypeArguments()[0] instanceof Class<?> elementType) {
            return elementType;
        }
        return Object.class;
    }

    private static <T> List<T> mapList(List<?> source, Class<T> elementType) {
        ArrayList<T> result = new ArrayList<>();
        if (source == null) {
            return result;
        }
        for (Object element : source) {
            result.add(elementType == Object.class ? elementType.cast(element) : mapRecord(element, elementType));
        }
        return result;
    }
