    /**
     * MA-003 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @POST
    @Transactional(TxType.NOT_SUPPORTED)
    @Operation(
            summary = "オンライン開栓を報告",
            description = "オンライン開栓を報告する",
            operationId = "postOpeningReportRetailResource")
    @APIResponse(
            responseCode = "200",
            description = "正常応答",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OpeningReportRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningReportRetailResponse endpoint(OpeningReportRetailRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
            Long kyoTrkNo = request.kyoTrkNo() != null
                    ? Long.valueOf(request.kyoTrkNo().trim())
                    : null;
            Boolean namTisiFlg = Boolean.parseBoolean(request.namTisiUmFlg());
            Boolean kseflKknnFlg = Boolean.parseBoolean(request.kseflKknnFlg());
            Boolean ctnmKjhkFlg = Boolean.parseBoolean(request.kiycstInf().ctnmKjhkFlg());
            LocalDate kasgYmd = Rfc3339LocalDateParser.parse(request.kasYmd());
            String kasJysb = request.knsKkaInf().kasJysb();
            String correlationId = requestContext.getCorrelationId().orElseThrow(
                    () -> new IllegalStateException("correlationId is not set in RequestContext"));
            LocalDate gmtThkgYmd = null;
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();
            // SV-027 検針結果（開栓）情報を取得する
            MeterReadingInfoAcquisitionForOpeningOutput meterReadingInfoAcquisitionForOpeningOutput =
                    meterReadingInfoAcquisitionForOpening.execute(gasMtstNo);

            // SV-182 開栓報告可能期限日チェック
            openingDateCheckRecentMeterReadingDate.execute(
                    kasgYmd,
                    meterReadingInfoAcquisitionForOpeningOutput.ksenHkkKnuKigenYmd());

            // ガスメーター設置場所＜導管＞情報取得（DA-033-R005)
            GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, knskKjnYmd);

            // 開栓時お客さま番号採番対象判定（SV-276)
            CustomerNumberAssignmentTargetJudgmentAtOpeningOutput customerNumberAssignmentTargetJudgmentAtOpeningOutput =
                    customerNumberAssignmentTargetJudgmentAtOpening.execute(namTisiFlg, gmtsetdRecord.ckrtGmscJtkb(), kasJysb);

           //住所情報取得(DA-002-R001)
            JusyomRecord jusyomRecord = jusyomDataAccess.findByGkkCdAndMchCdAndCmazCd(
                    gmtsetdRecord.szcGkkCd(),
                    gmtsetdRecord.szcMchCd(),
                    gmtsetdRecord.szcCmazCd())
                    .orElseThrow(() -> DataInconsistencySupport.create(
                            jusyomDataAccess.tableName(),
                            JusyomDataAccess.CallSpecs.FIND_BY_GKK_CD_AND_MCH_CD_AND_CMAZ_CD.identifier(),
                            new FlatJsonBuilder()
                                    .put("gkkCd", gmtsetdRecord.szcGkkCd())
                                    .put("mchCd", gmtsetdRecord.szcMchCd())
                                    .put("cmazCd", gmtsetdRecord.szcCmazCd())));

            // ガスメーター＜導管＞情報取得(DA-110-R004)
            GasmtrdRecord gasmtrdRecords = requireGasmtrd(gasMtstNo, knskKjnYmd, gmtThkgYmd);

            // 開栓エラーフォロー件名情報存在チェック
            if (kseflKknnFlg.equals(Boolean.FALSE)) {
                List<KseflRecord> kseflRecord =
                        kseflDataAccess.findAllByGmbAndFlwStCd(
                                gasmtrdRecords.gmb(),
                                followStatusCodeN2NotYetStarted);
                // 開栓エラーフォロー件名が0件でない場合
                if (!kseflRecord.isEmpty()) {
                    return (new OpeningReportRetailResponse(ERROR_CODE_020000070,
                            ErrorCatalog.resolveErrorMessage(ERROR_CODE_020000070, null),
                            null,
                            null,
                            null,
                            ErrorCatalog.resolveCiriusMessageId(ERROR_CODE_020000070),
                            null,
                            correlationId));
                }
            }

            // SV-275実行
            ContractCustomerAcquisitionRegistrationAtOpeningOutput contractCustomerAcquisitionRegistrationAtOpeningOutput =
                this.callContractCustomerAcquisitionRegistrationAtOpening(
                    customerNumberAssignmentTargetJudgmentAtOpeningOutput, request, kyoTrkNo, ctnmKjhkFlg, jusyomRecord, gmtsetdRecord);

            try {
                this.callOpeningOperation(kasJysb, kyoTrkNo, gasmtrdRecords.gmb(), request,
                        contractCustomerAcquisitionRegistrationAtOpeningOutput, knskKjnYmd, correlationId);
            } catch (MaApiException ex) {
                throw new MaPassthroughMaApiException(ex, this::handleError);
            }

            return (new OpeningReportRetailResponse(null, null, null, null, null, null, null, correlationId));
        });
    }

                            request,
                            contractCustomerAcquisitionRegistrationAtOpeningOutput.kyoTrkNo(),
                            gmb,
                            knskKjnYmd,
                            correlationId
                    ));
        }

    }

    /**
     * MAAPI呼出例外の独自メソッド定義
     *
     * @param exception MAAPI呼出時の透過例外
     * @return MaPassthroughMaApiException MAAPI呼出時の透過例外
     */
    private MaPassthroughMaApiException handleError(MaPassthroughMaApiException exception) {
        String body = exception.getResponseBody();
        String ErrorCode_020100048 = ErrorCodeConstants.ERROR_CODE_020100048;
        String correlationId = requestContext.getCorrelationId().orElseThrow(
                () -> new IllegalStateException("correlationId is not set in RequestContext"));
        String mediaType = exception.getMediaType();
        int status = exception.getStatus();
        if (body == null) {
            return exception;
        }

        if (mediaType == null || ! mediaType.contains("json")) {
            return exception;
        }

        ObjectMapper mapper = new ObjectMapper();
        mapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

        try {
            MaPassthroughMaApiErrorResponse originErrDto =
                    mapper.readValue(body, MaPassthroughMaApiErrorResponse.class);
            boolean isWarning =
                  ErrorCode_020100048.equals(originErrDto.errorCode());

            String crsMsgId =
                    Optional.ofNullable(originErrDto.ciriusMessageId())
                            .orElse(ErrorCatalog.resolveCiriusMessageId(originErrDto.errorCode()));

            List<String> crsEmbeddedStringList =
                    ErrorCode_020100048.equals(originErrDto.errorCode())
                            ? originErrDto.ciriusEmbeddedStringList()
                                    .stream()
                                    .map(this::formatNumber).toList()
                            : originErrDto.ciriusEmbeddedStringList();

            OpeningReportRetailResponse entityDto =
                    isWarning
                            ? new OpeningReportRetailResponse(
                                    originErrDto.errorCode(),
                                    originErrDto.errorDescription(),
                                    originErrDto.errorDetailList(),
                                    null,
                                    null,
                                    crsMsgId,
                                    crsEmbeddedStringList,
                                    correlationId)
                            : new OpeningReportRetailResponse(
                                    originErrDto.errorCode(),
                                    originErrDto.errorDescription(),
                                    originErrDto.errorDetailList(),
                                    crsMsgId,
                                    crsEmbeddedStringList,
                                    null,
                                    null,
                                    correlationId);

            String entityStr = mapper.writeValueAsString(entityDto);

            return isWarning
                    ? new MaPassthroughMaApiException(
                            Response.Status.OK.getStatusCode(),
                            entityStr,
                            mediaType,
                            null,
                            null)
                    : new MaPassthroughMaApiException(
                            status,
                            entityStr,
                            mediaType,
                            null,
                            null);
        } catch (IOException e) {
            return exception;
        }
    }

    /**
     * 1xのハイフン編集(xxxx-xxx-xxxx)
     *
     * @param input 修正対象文字列
     * @return String 修正後文字列
     */
    private String formatNumber(String input) {
        if (input == null || input.length() != 11) {
            return input;
        }

        return input.substring(0, 4) + "-" +
               input.substring(4, 7) + "-" +
               input.substring(7, 11);
    }
