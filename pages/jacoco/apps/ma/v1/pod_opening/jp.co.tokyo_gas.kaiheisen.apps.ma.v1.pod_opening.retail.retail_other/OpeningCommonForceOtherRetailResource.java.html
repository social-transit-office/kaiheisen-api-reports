    /**
     * MA-058 の業務処理本体を実行する。
     *
     * @param request リクエスト DTO
     * @return 実行結果レスポンス
     */
    OpeningCommonForceOtherRetailResponse handle(OpeningCommonForceOtherRetailRequest request) {
        LocalDateTime systemDatetime = requiredSystemDatetime();
        String hukkSysShbt = requiredHukkSysShbt();
        CustomerNameKanjiDisplayFlagAcquisitionOutput flagOutput =
                customerNameKanjiDisplayFlagAcquisition.execute(request.ctnmKj1());
        KbagmstRecord kbagmstRecord = resolveKbagmstRecord(request, systemDatetime);
        kseflDataAccess.insert(
            buildWriteRecord(request, flagOutput, kbagmstRecord, hukkSysShbt),
                AuditContext.of(
                        SERVICE_USER_ID,
                        decodeOriginFunctionId(hukkSysShbt)));
        return new OpeningCommonForceOtherRetailResponse(
                resolveResponseCorrelationId(),
                null,
                null,
                null);
    }

    /**
     * フォロー担当の地域統括コードと拠点コードを解決するため KBAGMST を取得する。
     *
     * <p>ガスメーター設置場所番号が null または 0 の場合、または KBAGMST を取得できない場合は、
     * リクエスト値を用いた疑似レコードを返す。</p>
     *
     * @param request リクエスト DTO
     * @param systemDatetime システム日時
     * @return KBAGMST レコード、または地域統括コードのみ補完した疑似レコード
     */
    private KbagmstRecord resolveKbagmstRecord(
            OpeningCommonForceOtherRetailRequest request,
            LocalDateTime systemDatetime) {
        YearMonth knskKjnYm = YearMonth.of(systemDatetime.getYear(), systemDatetime.getMonthValue());
        String normalizedCktkCd = normalizeOptional(request.cktkCd());
        if (request.gasMtstNo() == null || request.gasMtstNo() == 0L) {
            return fallbackKbagmstRecord(request.gasMtstNo(), knskKjnYm, normalizedCktkCd);
        }
        return kbagmstDataAccess.findByGasMtstNoAndTkysYmAndTyenYm(request.gasMtstNo(), knskKjnYm)
                .orElseGet(() -> fallbackKbagmstRecord(request.gasMtstNo(), knskKjnYm, normalizedCktkCd));
    }

    private FollowBaseInfo resolveFollowBaseInfo(String value) {
        String normalized = normalizeSingleCharacterCode(value);
        if (normalized == null) {
            return new FollowBaseInfo(UNRESOLVED_BASE_CODE, TKJ_MEM_BASE_UNIDENTIFIED);
        }
        return new FollowBaseInfo(normalized, null);
    }

    private String normalizeSingleCharacterCode(String value) {
        String normalized = normalizeOptional(value);
        if (normalized == null) {
            return null;
        }
        return normalized.substring(0, 1);
    }

    /**
     * 報告システム種別から最終更新起点機能 ID を解決する。
     *
     * @param hukkSysShbt 報告システム種別
     * @return 最終更新起点機能 ID
     */
    private String decodeOriginFunctionId(String hukkSysShbt) {
        String normalizedHukkSysShbt = normalizeOptional(hukkSysShbt);
        try {
            return kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                    normalizedHukkSysShbt);
        } catch (ConstantDefinitionNotFoundException ex) {
            throw new KaiheisenException("報告システム種別の定数定義が見つかりません", ex);
        }
    }

    /**
     * 日付文字列を LocalDate へ変換する。
     *
     * @param value RFC3339 日付文字列
     * @return 変換結果。未設定時は {@code null}
     */
    private static LocalDate parseDate(String value) {
        String normalized = value == null ? null : value.trim();
        return normalized == null || normalized.isEmpty() ? null : LocalDate.parse(normalized);
    }

    /**
     * 電話更新情報リストから KSEFL の 1 レコード分の電話情報を取り出す。
     *
     * @param telnoUpdInfList 電話更新情報リスト
     * @param index 取り出し対象の添字
     * @return 抽出結果。存在しない場合は空オブジェクト
     */
    private static TelProjection telProjection(List<TelnoUpdInf> telnoUpdInfList, int index) {
        if (telnoUpdInfList == null || telnoUpdInfList.size() <= index || telnoUpdInfList.get(index) == null) {
            return TelProjection.empty();
        }
        TelnoUpdInf telnoUpdInf = telnoUpdInfList.get(index);
        NewTelInf newTelInf = telnoUpdInf.newTelInf();
        if (newTelInf == null) {
            return new TelProjection(null, null, null, null, null, null, null, null, null, telnoUpdInf.oldKyoTelNo());
        }
        return new TelProjection(
                newTelInf.kyoTelNo(),
                newTelInf.telShbt(),
                newTelInf.telRrskShbt(),
                newTelInf.telKnzShbt(),
                newTelInf.telStsdShbt(),
                parseDate(newTelInf.telShtkYmd()),
                newTelInf.tetsCtnmKj(),
                newTelInf.telZkgrShbt(),
                newTelInf.telMusdKbn(),
                telnoUpdInf.oldKyoTelNo());
    }

    private record TelProjection(

            String telShbt,
            String telRrskShbt,

        /**
         * 値未設定を表す空の投影値を返す。
         *
         * @return 空の投影値
         */
        private static TelProjection empty() {
            return new TelProjection(null, null, null, null, null, null, null, null, null, null);
        }
    }
