package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_opening.config;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Produces;
import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantServices;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.integration.cirius.retail_sc.customer_numbering.CustomerNumberingCiriusAdapter;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.context.SvContextProxyFactory;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tkssvky.TkssvkyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.RetailTypeJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening.CiriusCustomerNumberingAllInOneSupport;
import jp.co.tokyo_gas.kaiheisen.libs.sv.impl.opening.NumberingAllInOneSupport;
import jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation.RetailTypeJudgmentImpl;

/**
 * pod_opening で利用する共通基盤 Bean を定義する設定クラス。
 */
@ApplicationScoped
public class PodOpeningLibsConfiguration {

    @Inject
    SvContextProxyFactory svContextProxyFactory;

    @Produces
    @Singleton
    CiriusConstantService ciriusConstantService() {
        return ConstantServices.create().ciriusConstantService();
    }

    @Produces
    @Singleton
    KaiheisenConstantService kaiheisenConstantService() {
        return ConstantServices.create().kaiheisenConstantService();
    }

    @Produces
    @Singleton
    RetailTypeJudgment retailTypeJudgment(
            TskgmstDataAccess tskgmstDataAccess,
            TkssvkyDataAccess tkssvkyDataAccess,
            CiriusConstantService ciriusConstantService) {
        return wrapService(
                RetailTypeJudgment.class,
                new RetailTypeJudgmentImpl(
                        tskgmstDataAccess,
                        tkssvkyDataAccess,
                        ciriusConstantService));
    }

    @Produces
    @Singleton
    NumberingAllInOneSupport numberingAllInOneSupport(
            CustomerNumberingCiriusAdapter customerNumberingCiriusAdapter) {
        return new CiriusCustomerNumberingAllInOneSupport(customerNumberingCiriusAdapter);
    }

    private <T> T wrapService(Class<T> serviceType, T service) {
        return svContextProxyFactory.wrap(serviceType, service);
    }
}
