    /**
     * MA-013 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "開栓エラーフォロー表示項目取得",
            description = "TG小売の開栓エラーフォロー表示項目を取得する",
            operationId = "getOpeningErrorFollowDisplayGetTgRetail")
    @APIResponse(
            responseCode = "200",
            description = "開栓エラーフォロー表示項目取得結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OpeningErrorFollowDisplayGetTgRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningErrorFollowDisplayGetTgRetailResponse endpoint(@BeanParam OpeningErrorFollowDisplayGetTgRetailRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo());
            String tksuSvkyJtkb = request.tksuSvkyJtkb();
            LocalDate tyenYmd = Rfc3339LocalDateParser.parse(request.tyenYmd());
            String tksuKijsCd = request.tksuKijsCd();
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();
            Long syoKyoTrno = Long.valueOf(request.syoKyoTrno());

            // DA-033-R006 ガスメーター設置場所＜導管＞情報の取得
            GmtsetdRecord gmtsetdRecord = gmtsetdDataAccess
                    .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                    .orElseThrow(() -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000029));

            if (gmtsetdRecord.tusoGmtkYmd() == null) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020100012);
            }

            // DA-110-R007 ガスメーター＜導管＞情報の取得
            Optional<GasmtrdRecord> gasmtrdRecord = gasmtrdDataAccess
                    .findByTtkGmsbNoAndGmtkYmd(gasMtstNo, knskKjnYmd);

            // SV-027検針情報(開栓)を取得する
            Optional<MeterReadingInfoAcquisitionForOpeningOutput> meterReadingInfoAcquisitionForOpeningOutput =
                    Optional.ofNullable(gmtsetdRecord.kngmStbJtkb())
                            .filter(kngmStbJtkb -> !kngmStbJtkb.isBlank())
                            .filter(meterPossession::equals)
                            .map(v -> gmtsetdRecord.gasMtstNo())
                            .map(meterReadingInfoAcquisitionForOpening::execute);

            // DA-007-R001 建物情報を取得する
            Optional<TatemonRecord> tatemonRecord =
                    Optional.ofNullable(gmtsetdRecord.bldNo())
                            .filter(v -> !v.isBlank())
                            .map(bldNo -> tatemonDataAccess.findByBldNo(bldNo)
                                    .orElseThrow(() -> DataInconsistencySupport.create(
                                            tatemonDataAccess.tableName(),
                                            TatemonDataAccess.CallSpecs.FIND_BY_BLD_NO.identifier(),
                                            new FlatJsonBuilder()
                                                    .put("bldNo", bldNo))));

            Optional<SvckyaRecord> svckyaRecord = Optional.empty();

            if (contractActive.equals(tksuSvkyJtkb)) {
                // SV-003 サービス契約番号を取得する
                ServiceContractNumberAcquisitionOutput serviceContractNumberAcquisitionOutput
                        = serviceContractNumberAcquisition.execute(
                                gasMtstNo, gmtsetdRecord.mtgMtstNo(),
                                ServiceContractNumberAcquisitionConstants.ShrKbn.NEAREST,
                                gmtsetdRecord.ckrtGmscJtkb(), knskKjnYmd);

                // DA-038-R007 サービス契約情報を取得する（サービス契約状態区分どうする）
                svckyaRecord =
                        svckyaDataAccess.findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNot(
                                serviceContractNumberAcquisitionOutput.svkyNo(), knskKjnYmd, kykTkytyu);
            }

            // SV-042 検針結果情報を取得する
            MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput =
                    meterReadingResultInfoAcquisition.execute(
                            gmtsetdRecord.hesnJysb(), gmtsetdRecord.gasMtstNo(),
                            RetailTypeJudgmentCode.TOKYO_GAS.code(),
                            meterReadingInfoAcquisitionForOpeningOutput
                                    .map(MeterReadingInfoAcquisitionForOpeningOutput::knsYmd)
                                    .orElse(null),
                            meterReadingInfoAcquisitionForOpeningOutput
                                    .map(MeterReadingInfoAcquisitionForOpeningOutput::gmtShjSu)
                                    .orElse(null),
                            meterReadingInfoAcquisitionForOpeningOutput
                                    .map(MeterReadingInfoAcquisitionForOpeningOutput::knkkSsjyShbt)
                                    .orElse(null));

            // DA-104-R001 契約お客さま＜導管＞情報を取得する
            KiycstdRecord kiycstdRecord =
                    kiycstdDataAccess.findByKyoTrkNo(syoKyoTrno)
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    kiycstdDataAccess.tableName(),
                                    KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                                    new FlatJsonBuilder()
                                            .put("syoKyoTrno", syoKyoTrno)));

            // DA-113-R001 メーターグループ＜導管＞情報を取得する
            Optional<MtrgrpdRecord> mtrgrpdRecord =
                    mtrgrpdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gmtsetdRecord.gasMtstNo(), knskKjnYmd);

            // SV-079 照会レベル種別を編集する
            InquiryLevelCodeEditOutput inquiryLevelCodeEditOutput =
                    inquiryLevelCodeEdit.execute(kiycstdRecord.shkiLvShbt());

            // SV-081 供給停止精算状態区分を取得する
            SupplySuspensionSettlementStatusAcquisitionOutput supplySuspensionSettlementStatusAcquisitionOutput =
                    supplySuspensionSettlementStatusAcquisition.execute(
                            gmtsetdRecord.hesnJysb(),
                            TKSSVKYUMFLG_U,
                            tksuKijsCd,
                            meterReadingResultInfoAcquisitionOutput.kkysSisnJtkb());

            return toResponse(tatemonRecord,
                    meterReadingResultInfoAcquisitionOutput,
                    inquiryLevelCodeEditOutput,
                    supplySuspensionSettlementStatusAcquisitionOutput,
                    gmtsetdRecord,
                    gasmtrdRecord,
                    mtrgrpdRecord,
                    svckyaRecord,
                    kiycstdRecord,
                    tksuSvkyJtkb,
                    tyenYmd,
                    tksuKijsCd);
        });
    }
