package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_error_follow.retail.retail_common.opening_error_follow_update.model;

import jakarta.validation.Valid;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateText;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * 判定住所情報DTO。

 * @param ksenErrInf 開栓エラー情報
 * @param cbgSuj1 丁目－番地－号数字１

 * @param kassCsttNo 開栓先お客さま登録番号

@Schema(description = "判定住所情報")
public record HtiJsInfoRequest(
        @Valid
        @Schema(description = "ガスメーター情報")
        GmtInfoRequest gmtInf,
        @Valid
        @Schema(description = "検針結果情報")
        KnsKkaInfoRequest knsKkaInf,
        @Valid

        KsenErrInfoRequest ksenErrInf,
        @Schema(description = "丁目－番地－号数字１")
        String cbgSuj1,
        @Schema(description = "丁目－番地－号カナ２")
        String cbgKn2,
        @Schema(description = "丁目－番地－号カナ３")
        String cbgKn3,
        @Schema(description = "所在地町コード")
        String szcMchCd,
        @Schema(description = "所在地行政区画コード")
        String szcGkkCd,
        @Schema(description = "所在地丁目・字コード")
        String szcCmazCd,
        @Schema(description = "棟番号カナ")
        String tunoKn,
        @Schema(description = "部屋番号カナ")
        String hynKn,
        @Schema(description = "部屋番号登録有無フラグ")
        String htuFlg,
        @Rfc3339DateText
        @Schema(description = "開栓作業年月日")
        String kasgYmd,
        @Schema(description = "ガスメーター設置場所番号")
        String gasMtstNo,
        @Schema(description = "開栓先お客さま登録番号")
        String kassCsttNo
) {
}
