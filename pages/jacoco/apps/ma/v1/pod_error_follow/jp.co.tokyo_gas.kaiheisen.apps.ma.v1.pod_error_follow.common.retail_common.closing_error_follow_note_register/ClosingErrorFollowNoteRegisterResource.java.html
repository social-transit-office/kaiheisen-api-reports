/**
 * MA-032 閉栓エラーフォロー特記事項登録-共通-共通 のリソース実装。
 */
@Path(ClosingErrorFollowNoteRegisterEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "閉栓エラーフォロー特記事項登録", description = "MA-032 閉栓エラーフォロー特記事項登録 API")
public class ClosingErrorFollowNoteRegisterResource extends BaseResource<ClosingErrorFollowNoteRegisterRequest> {

    private final HseflDataAccess hseflDataAccess;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String usrId = "KHSM032";

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム時刻プロバイダ
     * @param hseflDataAccess     閉栓テーブルDA
     */
    @Inject
    public ClosingErrorFollowNoteRegisterResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            HseflDataAccess hseflDataAccess,
            KaiheisenConstantService kaiheisenConstantService) {
        super(requestContext, requestDtoValidator, clock);

        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");

        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.GASTA);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TAXIS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.KRAKEN);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.EAGLE);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.HURRY);
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
    }

    /**
     * MA-032 閉栓エラーフォロー特記事項登録 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 閉栓エラーフォロー特記事項登録レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓エラーフォロー特記事項登録",
            description = "開栓エラーフォロー特記事項登録を行う",
            operationId = "postClosingErrorFollowNoteRegister")
    @APIResponse(
            responseCode = "200",
            description = "閉栓エラーフォロー特記事項登録の結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingErrorFollowNoteRegisterResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingErrorFollowNoteRegisterResponse endpoint(ClosingErrorFollowNoteRegisterRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo());
            LocalDate hssgYmd = Rfc3339LocalDateParser.parse(request.hssgYmd());
            LocalDate hesnHukkYmd = Rfc3339LocalDateParser.parse(request.hesnHukkYmd());
            Integer regTsno = Integer.valueOf(request.regTsno());
            String tkjMem = request.tkjMem();
            Integer recUpdJnjNo = Integer.valueOf(request.recUpdJnjNo());
            // DA-141-U001パラメータ設定
            HseflWriteRecord record = new HseflWriteRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, tkjMem);
            Set<HseflColumn> targetColumns = Set.of(HseflColumn.TKJ_MEM);
            HseflU001UpdateKey key = new HseflU001UpdateKey(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno, recUpdJnjNo);
            String decodedHkkSysSyb = kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE, this.requestContext.getHukkSysShbt()
                            .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
            AuditContext context = new AuditContext(usrId, decodedHkkSysSyb);
            // DA-141-U001:GMB・KASG_YMD・KAS_HUKK_YMD・SGY_ANKN_NO・REC_UPD_JNJ_NOによる更新。
            int count = hseflDataAccess.updateByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsnoAndRecUpdJnjNo(
                    record, targetColumns, key, context);
            if (count == 0) {
                throw DataInconsistencySupport.create(
                        hseflDataAccess.tableName(),
                        HseflDataAccess.CallSpecs
                        .UPDATE_BY_GAS_MTST_NO_AND_HSSG_YMD_AND_HESN_HUKK_YMD_AND_REG_TSNO_AND_REC_UPD_JNJ_NO
                        .identifier(),
                        new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("hssgYmd", hssgYmd)
                        .put("hesnHukkYmd", hesnHukkYmd)
                        .put("regTsno", regTsno)
                        .put("recUpdJnjNo", recUpdJnjNo)
                        .put("tkjMem", tkjMem)
                        .put("updUsrId", usrId)
                        .put("updKtnKinoId", decodedHkkSysSyb));
            }
            return new ClosingErrorFollowNoteRegisterResponse(requestContext.getCorrelationId()
                    .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext")));
        });
    }
}
