    /**
     * MA-029 閉栓エラーフォロー更新 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 閉栓エラーフォロー更新 レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓エラーフォロー更新",
            description = "閉栓エラーフォロー画面で閉栓登録をおこなう",
            operationId = "postClosingRegisterResource")
    @APIResponse(
            responseCode = "200",
            description = "正常応答",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingRegisterResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingRegisterResponse endpoint(ClosingRegisterRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.hseflInf().gasMtstNo());
            LocalDate hssgYmd = Rfc3339LocalDateParser.parse(request.hseflKeyInf().hssgYmd());
            LocalDate hesnHukkYmd = Rfc3339LocalDateParser.parse(request.hseflKeyInf().hesnHukkYmd());
            Integer regTsno = Integer.valueOf(request.hseflKeyInf().regTsno());
            // DA-141-R001 閉栓エラーフォロー件名 検索
            HseflRecord hseflRecord = this.findHseflRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);

            // SV-001 TG小売判定
            String shrKbn = allPeriods;
            LocalDateTime systemDatetime = this.requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"));
            LocalDate systemLocalDate = systemDatetime.toLocalDate();
            RetailTypeJudgmentOutput retailTypeJudgmentOutput  = null;
            retailTypeJudgmentOutput = retailTypeJudgment.execute(shrKbn, gasMtstNo, systemLocalDate);

            String correlationId = this.requestContext.getCorrelationId()
                    .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"));
            try {
                // 呼出先MAの判定
                this.callClosingOperation(request, hseflRecord, retailTypeJudgmentOutput);
            } catch (MaApiException ex) {
                throw new MaPassthroughMaApiException(ex, this::handleError);
            }
            // 出力伝文編集処理
            return new ClosingRegisterResponse(
                    null, null, null, null, null, null, null, correlationId);
        });
    }

    private HseflRecord findHseflRecord(Long gasMtstNo, LocalDate hssgYmd, LocalDate hesnHukkYmd, Integer regTsno) {

        Optional<HseflRecord> record = hseflDataAccess.findByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsno(
                gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
        if (!record.isEmpty()) {
            return record.get();
        }
        // 取得件数が0件の場合 例外スロー
        throw DataInconsistencySupport.create(
                hseflDataAccess.tableName(),
                HseflDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_HSSG_YMD_AND_HESN_HUKK_YMD_AND_REG_TSNO.identifier(),
                new FlatJsonBuilder()
                .put("gasMtstNo", gasMtstNo)
                .put("hssgYmd", hssgYmd)
                .put("hesnHukkYmd", hesnHukkYmd)
                .put("regTsno", regTsno));

    }

    /**
     * 外部MA呼出処理。
     *
     * @param request リクエストDTO
     * @param hseflRecord 閉栓エラーフォロー情報
     * @param RetailTypeJudgmentOutput 小売判定
     */
    private void callClosingOperation(ClosingRegisterRequest request,
            HseflRecord hseflRecord,
            RetailTypeJudgmentOutput retailTypeJudgmentOutput) {

        if (kuHntKbnTokyoGas.equals(retailTypeJudgmentOutput.kuHntKbn()) && !closReasType4.equals(hseflRecord.hesnJysb())) {
            // MA-059
            closingCommonNormalTgRetailMaAdapter.invoke(
                    createClosingCommonNormalTgRetailRequest(request, hseflRecord, retailTypeJudgmentOutput));
        // SV-001で取得した判定結果コード = "1" (TG小売) かつ、閉栓事由＝【J0273　4：供給停止】の場合
        } else if (kuHntKbnTokyoGas.equals(retailTypeJudgmentOutput.kuHntKbn())) {
            // MA-067
            closingCommonSupplyStopMaAdapter.invoke(
                    createClosingCommonSupplyStopRequest(request, hseflRecord, retailTypeJudgmentOutput));
        } else if (kuHntKbnOther.equals(retailTypeJudgmentOutput.kuHntKbn())) {
            // MA-068
            closingCommonNormalOtherRetailMaAdapter.invoke(
                    createClosingCommonNormalOtherRetailRequest(request, hseflRecord, retailTypeJudgmentOutput));
        }

    }

    /**
     * MAAPI呼出例外の独自メソッド定義
     *
     * @param exception MAAPI呼出時の透過例外
     * @return MaPassthroughMaApiException MAAPI呼出時の透過例外
     */
    private MaPassthroughMaApiException handleError(MaPassthroughMaApiException exception) {
        String body = exception.getResponseBody();
        String correlationId = requestContext.getCorrelationId().orElseThrow(
                () -> new IllegalStateException("correlationId is not set in RequestContext"));
        String mediaType = exception.getMediaType();
        int status = exception.getStatus();
        if (body == null) {
            return exception;
        }

        if (mediaType == null || ! mediaType.contains("json")) {
            return exception;
        }

        ObjectMapper mapper = new ObjectMapper();
        mapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

        try {
            MaPassthroughMaApiErrorResponse originErrDto =
                    mapper.readValue(body, MaPassthroughMaApiErrorResponse.class);
            //　ワーニング判定フラグ（閉栓登録結果.エラーコード＝ 【定数.020000011】（"020000011"））
            boolean isWarning =
                    ErrorCodeConstants.ERROR_CODE_020000011.equals(originErrDto.errorCode());

            SyryWarnInfoResponse syryWarnInf =
                    isWarning
                            ? new SyryWarnInfoResponse(
                                    originErrDto.syryWarnInf() != null
                                    ? originErrDto.syryWarnInf().crsMsgIdList()
                                    : Collections.emptyList())
                            : null;

            IppnWarnInfoResponse ippnWarnInf =
                    isWarning
                            ? new IppnWarnInfoResponse(
                                    originErrDto.ippnWarnInf() != null
                                    ? originErrDto.ippnWarnInf().crsMsgIdList()
                                    : Collections.emptyList())
                            : null;

            String crsMsgId =
                    Optional.ofNullable(originErrDto.ciriusMessageId())
                            .orElse(ErrorCatalog.resolveCiriusMessageId(originErrDto.errorCode()));

            List<String> crsEmbeddedStringList;
            // エラーガスメーター設置場所番号 を保持している場合、ハイフン編集
            if (originErrDto.errGmsbNo() != null) {
                crsEmbeddedStringList = List.of(this.formatNumber(originErrDto.errGmsbNo().toString()));
            //　算出使用量　を保持している場合
            } else if (originErrDto.snstSyoRyo() != null) {
                crsEmbeddedStringList = List.of(originErrDto.snstSyoRyo().toString());
            } else {
                crsEmbeddedStringList = originErrDto.ciriusEmbeddedStringList();
            }

            ClosingRegisterResponse entityDto =
                    isWarning
                            ? new ClosingRegisterResponse(
                                    syryWarnInf,
                                    ippnWarnInf,
                                    originErrDto.errorCode(),
                                    null,
                                    null,
                                    null,
                                    null,
                                    correlationId)
                            : new ClosingRegisterResponse(
                                    null,
                                    null,
                                    originErrDto.errorCode(),
                                    originErrDto.errorDescription(),
                                    originErrDto.errorDetailList(),
                                    crsMsgId,
                                    crsEmbeddedStringList,
                                    correlationId);

            String entityStr = mapper.writeValueAsString(entityDto);

            return isWarning
                    // ワーニング時、ステータスコード200
                    ? new MaPassthroughMaApiException(
                            Response.Status.OK.getStatusCode(),
                            entityStr,
                            mediaType,
                            null,
                            null)
                    : new MaPassthroughMaApiException(
                            status,
                            entityStr,
                            mediaType,
                            null,
                            null);
        } catch (IOException e) {
            return exception;
        }
    }

    /**
     * 1xのハイフン編集(xxxx-xxx-xxxx)
     *
     * @param input 修正対象文字列
     * @return String 修正後文字列
     */
    private String formatNumber(String input) {
        if (input.length() != 11) {
            return input;
        }

        return input.substring(0, 4) + "-" +
               input.substring(4, 7) + "-" +
               input.substring(7, 11);
    }
