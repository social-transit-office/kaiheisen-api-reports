package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_error_follow.config;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_error_follow.common.retail_common.opening_error_follow_display_get.OpeningErrorFollowDisplayGetResource;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.CdiBeanInitializer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupContributor;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;

/** エラー後処理MAの起動時ウォームアップ定義。 */
@ApplicationScoped
public class PodErrorFollowStartupWarmupContributor implements StartupWarmupContributor {

    private final CdiBeanInitializer beanInitializer;

    private final Sysdummy1DataAccess dataAccess;

    /**
     * コンストラクタ。
     *
     * @param beanInitializer CDI Bean初期化処理
     * @param dataAccess DBウォームアップDA
     */
    @Inject
    public PodErrorFollowStartupWarmupContributor(
            final CdiBeanInitializer beanInitializer,
            final Sysdummy1DataAccess dataAccess) {
        this.beanInitializer = beanInitializer;
        this.dataAccess = dataAccess;
    }

    @Override
    public String applicationName() {
        return "ma-error-follow";
    }

    @Override
    public LogLayer logLayer() {
        return LogLayer.MA;
    }

    @Override
    public void initializeResources() {
        beanInitializer.initialize(OpeningErrorFollowDisplayGetResource.class);
    }

    @Override
    public int warmUpDatabase() {
        return dataAccess.fetchWarmupValue().value();
    }
}
