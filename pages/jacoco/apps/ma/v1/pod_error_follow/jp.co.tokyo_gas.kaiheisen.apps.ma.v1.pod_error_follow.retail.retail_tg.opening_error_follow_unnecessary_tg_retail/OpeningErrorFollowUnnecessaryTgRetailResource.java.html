/**
 * MA-016 開栓エラーフォロー不要-TG小売 のリソース実装。
 */
@Path(OpeningErrorFollowUnnecessaryTgRetailEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "開栓エラーフォロー不要", description = "MA-016 開栓エラーフォロー不要 API")
public class OpeningErrorFollowUnnecessaryTgRetailResource extends BaseResource<OpeningErrorFollowUnnecessaryTgRetailRequest> {

    private final KasnhkDataAccess kasnhkDataAccess;

    private final KseflDataAccess kseflDataAccess;

    private final NwkseigDataAccess nwkseigDataAccess;

    private final ApplicationTypeAcquisition applicationTypeAcquisition;

    private final ReceptionInfoChangeNotificationApiIntegration receptionInfoChangeNotificationApiIntegration;

    private final OpeningErrorPfFollowUnnecessaryUpdate openingErrorPfFollowUnnecessaryUpdate;

    private final KaiheisenConstantService kaiheisenConstantService;

    private final String ksenJtiKbnFlwf;

    private final String customerServiceHistoryWorkDetailsOthers;

    private final String procedureStatusCancel;

    private static final String USER_ID = "KHSM016";

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム時刻プロバイダ
     */
    @Inject
    public OpeningErrorFollowUnnecessaryTgRetailResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @OpeningErrorFollowUnnecessaryTgRetailConfiguration
            ApplicationTypeAcquisition applicationTypeAcquisition,
            @OpeningErrorFollowUnnecessaryTgRetailConfiguration
            ReceptionInfoChangeNotificationApiIntegration receptionInfoChangeNotificationApiIntegration,
            @OpeningErrorFollowUnnecessaryTgRetailConfiguration
            OpeningErrorPfFollowUnnecessaryUpdate openingErrorPfFollowUnnecessaryUpdate,
            KasnhkDataAccess kasnhkDataAccess,
            KseflDataAccess kseflDataAccess,
            NwkseigDataAccess nwkseigDataAccess,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        super(requestContext, requestDtoValidator, clock);
        this.kasnhkDataAccess = Objects.requireNonNull(kasnhkDataAccess, "kasnhkDataAccess");
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");
        this.nwkseigDataAccess = Objects.requireNonNull(nwkseigDataAccess, "nwkseigDataAccess");
        this.applicationTypeAcquisition = Objects.requireNonNull(applicationTypeAcquisition, "applicationTypeAcquisition");
        this.receptionInfoChangeNotificationApiIntegration = Objects.requireNonNull(
                receptionInfoChangeNotificationApiIntegration, "receptionInfoChangeNotificationApiIntegration");
        this.openingErrorPfFollowUnnecessaryUpdate = Objects.requireNonNull(
                openingErrorPfFollowUnnecessaryUpdate, "openingErrorPfFollowUnnecessaryUpdate");

        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");

        this.ksenJtiKbnFlwf = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpeningReportStatus.TYPE,
                ConstantLookupKeys.Cirius.OpeningReportStatus.NO_NEED_TO_FOLLOW_UP_ON_OPENING);

        this.customerServiceHistoryWorkDetailsOthers = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkDetailsCode.TYPE,
                ConstantLookupKeys.Cirius.CustomerServiceHistoryWorkDetailsCode.OTHERS);

        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");

        this.kaiheisenConstantService = resolvedKaiheisenConstantService;

        this.procedureStatusCancel = RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ProcedureStatus.TYPE,
                ConstantLookupKeys.Kaiheisen.ProcedureStatus.CANCELLATION_OF_PROCEDURE);
    }

    /**
     * MA-016 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "開栓エラーフォロー不要",
            description = "開栓エラーフォロー不要として更新する",
            operationId = "postOpeningErrorFollowUnnecessaryTgRetailResource")
    @APIResponse(
            responseCode = "200",
            description = "開栓エラーフォロー不要結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OpeningErrorFollowUnnecessaryTgRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningErrorFollowUnnecessaryTgRetailResponse endpoint(OpeningErrorFollowUnnecessaryTgRetailRequest request) {
        return execute(request, () -> {
            String gmb = request.gmb();
            LocalDate kasgYmd = Rfc3339LocalDateParser.parse(request.kasgYmd());
            LocalDate kasHukkYmd = Rfc3339LocalDateParser.parse(request.kasHukkYmd());
            Integer recUpdJnjNo = Integer.valueOf(request.recUpdJnjNo());
            String sgyAnknNo = request.sgyAnknNo();
            String hukkSysShbt = this.requestContext.getHukkSysShbt()
                    .orElseThrow(() ->
                        new IllegalStateException(
                            "hukkSysShbt is not set in RequestContext"));
            String correlationId = this.requestContext.getCorrelationId()
                    .orElseThrow(() ->
                    new IllegalStateException(
                        "correlationId is not set in RequestContext"));
            LocalDateTime systemDateTime = this.requestContext.getSystemDatetime()
                    .orElseThrow(() ->
                    new IllegalStateException(
                        "systemDateTime is not set in RequestContext"));

            // SV-117 開栓エラーフォロー件名を「フォロー不要」として更新する
            openingErrorPfFollowUnnecessaryUpdate.execute(
                    gmb, kasgYmd, kasHukkYmd, sgyAnknNo, recUpdJnjNo);

            // DA-118-U002 開栓報告情報を更新する
            KasnhkWriteRecord record = new KasnhkWriteRecord(
                    gmb, kasgYmd, kasHukkYmd, null, sgyAnknNo, ksenJtiKbnFlwf, systemDateTime.toLocalDate(),
                    null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null);

            Set<KasnhkUpdatableColumn> targetColumns =
                    Set.of(KasnhkUpdatableColumn.KAS_HUKK_JTKB, KasnhkUpdatableColumn.KAS_HUKR_YMD);

            KasnhkU002UpdateKey updateKey =
                    new KasnhkU002UpdateKey(gmb, kasgYmd, kasHukkYmd, sgyAnknNo, null);

            String kinoId = kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                    hukkSysShbt);

            AuditContext context = new AuditContext(USER_ID, kinoId);

            kasnhkDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNoAndRecUpdJnjNo(
                    record, targetColumns, updateKey, context);

            // DA-139-R004 開栓エラーフォロー件名の情報を取得する
            KseflRecord kseflRecord =
                    kseflDataAccess.findByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNo(gmb, kasgYmd, kasHukkYmd, sgyAnknNo)
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    kseflDataAccess.tableName(),
                                    KseflDataAccess.CallSpecs.FIND_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO.identifier(),
                                    new FlatJsonBuilder()
                                            .put("gmb", gmb)
                                            .put("kasgYmd", kasgYmd)
                                            .put("kasHukkYmd", kasHukkYmd)
                                            .put("sgyAnknNo", sgyAnknNo)));

            // DA-135-R001 ＮＷ箇所名を取得する
            String nwkCd = kseflRecord.kasgNwkCd();
            NwkseigRecord nwkseigRecord =
                    nwkseigDataAccess.findByNwkCdAndTkysYmdAndTyenYmd(nwkCd, kasgYmd)
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    nwkseigDataAccess.tableName(),
                                    NwkseigDataAccess.CallSpecs.FIND_BY_NWK_CD_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                                    new FlatJsonBuilder()
                                            .put("nwkCd", nwkCd)
                                            .put("kasgYmd", kasgYmd)));

            // SV-087 申込種別取得
            String kasJysb = kseflRecord.kasJysb();
            String htyoSysShbtCd = kseflRecord.hukkSysShbt();
            ApplicationTypeAcquisitionOutput applicationTypeAcquisitionOutput =
                    applicationTypeAcquisition.execute(
                    kasJysb, null, htyoSysShbtCd);

            // SV-092 受付情報変更通知API連携
            String utidxNo = kseflRecord.utidxNo();
            String uketkSeiriNo = kseflRecord.sgyAnknNo();
            Long gasMtstNo = kseflRecord.gasMtstNo();
            Long cstRegNo = kseflRecord.kassKyoTrno();
            String nwknKj = nwkseigRecord.nwknKj();
            String sagyoSyaName = kseflRecord.sgytSinnKj();
            receptionInfoChangeNotificationApiIntegration.execute(
                    utidxNo, uketkSeiriNo, null, procedureStatusCancel, gasMtstNo, cstRegNo,
                    customerServiceHistoryWorkDetailsOthers, nwknKj, sagyoSyaName,
                    applicationTypeAcquisitionOutput.mkSbt(), null, LocalDate.of(1900, 1, 1),
                    null, null);

            return new OpeningErrorFollowUnnecessaryTgRetailResponse(correlationId);
        });
    }
