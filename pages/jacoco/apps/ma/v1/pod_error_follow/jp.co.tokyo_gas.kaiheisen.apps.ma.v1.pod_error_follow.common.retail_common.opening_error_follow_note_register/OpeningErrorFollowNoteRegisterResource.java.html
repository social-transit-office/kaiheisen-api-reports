/**
 * MA-018 開栓エラーフォロー特記事項登録-共通-共通 のリソース実装。
 */
@Path(OpeningErrorFollowNoteRegisterEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "開栓エラーフォロー特記事項登録", description = "MA-018 開栓エラーフォロー特記事項登録 API")
public class OpeningErrorFollowNoteRegisterResource extends BaseResource<OpeningErrorFollowNoteRegisterRequest> {

    private final KseflDataAccess kseflDataAccess;

    private final KaiheisenConstantService kaiheisenConstantService;

    private static final String USER_ID = "KHSM018";

    /**
     * コンストラクタ。
     *
     * @param requestContext      リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock               システム時刻プロバイダ
     * @param kseflDataAccess     開閉栓テーブルDA
     */
    @Inject
    public OpeningErrorFollowNoteRegisterResource(RequestContext requestContext,
            RequestDtoValidator requestDtoValidator, SystemDateTimeProvider clock, KseflDataAccess kseflDataAccess,
            KaiheisenConstantService kaiheisenConstantService) {
        super(requestContext, requestDtoValidator, clock);

        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(kaiheisenConstantService,
                "kaiheisenConstantService");

        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.CIRIUS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.GASTA);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TAXIS);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.KRAKEN);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.EAGLE);
        RequiredConstantResolver.requireKaiheisenConstant(resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.HURRY);
        this.kaiheisenConstantService = resolvedKaiheisenConstantService;
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");
    }

    /**
     * MA-018 開栓エラーフォロー特記事項登録 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 開栓エラーフォロー特記事項登録レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "開栓エラーフォロー特記事項登録",
            description = "開栓エラーフォロー特記事項登録を行う",
            operationId = "postOpeningErrorFollowNoteRegister")
    @APIResponse(
            responseCode = "200",
            description = "開栓エラーフォロー特記事項登録の結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OpeningErrorFollowNoteRegisterResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningErrorFollowNoteRegisterResponse endpoint(OpeningErrorFollowNoteRegisterRequest request) {
        return execute(request, () -> {
            String gmb = request.gmb();
            LocalDate kasgYmd = Rfc3339LocalDateParser.parse(request.kasgYmd());
            LocalDate kasHukkYmd = Rfc3339LocalDateParser.parse(request.kasHukkYmd());
            String sgyAnknNo = request.sgyAnknNo();
            String tkjkMemo = request.tkjMem();
            Integer recUpdJnjNo = Integer.valueOf(request.recUpdJnjNo());

            // DA-139-U002パラメータ設定
            KseflWriteRecord record = new KseflWriteRecord(gmb, kasgYmd, kasHukkYmd, null, sgyAnknNo, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, tkjkMemo);
            Set<KseflColumn> targetColumns = Set.of(KseflColumn.TKJ_MEM);
            KseflU002UpdateKey key = new KseflU002UpdateKey(gmb, kasgYmd, kasHukkYmd, sgyAnknNo, recUpdJnjNo);
            String decodedHkkSysSyb = kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE, this.requestContext.getHukkSysShbt()
                            .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required")));
            AuditContext context = new AuditContext(USER_ID, decodedHkkSysSyb);

            // DA-139-U002:GMB・KASG_YMD・KAS_HUKK_YMD・SGY_ANKN_NO・REC_UPD_JNJ_NOによる更新。
            int updatedCount = kseflDataAccess.updateByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNoAndRecUpdJnjNo(
                    record, targetColumns, key, context);

            if (updatedCount == 0) {
                throw DataInconsistencySupport.create(
                        kseflDataAccess.tableName(),
                        KseflDataAccess.CallSpecs
                                .UPDATE_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO_AND_REC_UPD_JNJ_NO
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("gmb", gmb)
                                .put("kasgYmd", kasgYmd)
                                .put("kasHukkYmd", kasHukkYmd)
                                .put("sgyAnknNo", sgyAnknNo)
                                .put("recUpdJnjNo", recUpdJnjNo)
                                .put("recUpdJnjNo", recUpdJnjNo)
                                .put("tkjkMemo", tkjkMemo));
            }

            return new OpeningErrorFollowNoteRegisterResponse(
                    requestContext.getCorrelationId()
                            .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext")));
        });
    }
}
