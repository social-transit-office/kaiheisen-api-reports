/**
 * MA-009 フォロー対象一覧取得-メーター取付待ち のリソース実装。
 */
@Path(ErrorFollowTargetListEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "フォロー対象一覧取得", description = "MA-009 フォロー対象一覧取得 API")
public class ErrorFollowTargetListResource extends BaseResource<ErrorFollowTargetListRequest> {

    private final KseflDataAccess kseflDataAccess;

    private String openErrorReasonCodeMeterError;

    private String followStatusCodeN2NotYetStarted;

    private final MixFieldEdit mixFieldEdit;

    private static final String CIRIUS_MSG_ID = "CXXM00037W";

    private String meterTrtkMt = "1";

    /** 定数：フォロー担当拠点コード.支社("0") */
    private static final String SHISHA = "0";

    /**
     * コンストラクタ
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム時刻プロバイダ
     * @param kseflDataAccess 開栓エラーフォロー件名データアクセス
     */
    @Inject
    public ErrorFollowTargetListResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            KseflDataAccess kseflDataAccess,
            @ErrorFollowTargetListConfiguration
            MixFieldEdit mixFieldEdit,
            CiriusConstantService ciriusConstantService) {
        super(requestContext, requestDtoValidator, clock);
        this.kseflDataAccess = Objects.requireNonNull(kseflDataAccess, "kseflDataAccess");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.openErrorReasonCodeMeterError = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.OpenErrorReasonCode.TYPE,
                ConstantLookupKeys.Cirius.OpenErrorReasonCode.WAITING_FOR_METER_INSTALLATION_ERROR);

        this.followStatusCodeN2NotYetStarted = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);

        this.mixFieldEdit = Objects.requireNonNull(
                mixFieldEdit,
                "mixFieldEdit");
    }

    /**
     * MA-009 フォロー対象一覧取得-メーター取付待ち API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "フォロー対象一覧取得",
            description = "フォロー対象一覧取得-メーター取付待ち",
            operationId = "getErrorFollowTargetList")
    @APIResponse(
            responseCode = "200",
                    description = "フォロー対象一覧取得",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ErrorFollowTargetListResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ErrorFollowTargetListResponse endpoint(@BeanParam ErrorFollowTargetListRequest request) {
        return execute(request, () -> {
            String flttCktkCd = request.flttCktkCd();
            String flttKtnCd = SHISHA.equals(request.flttKtnCd()) ? null : request.flttKtnCd();
            String errBnriCd = request.errBnriCd();
            LocalDate knskKjnYmd = LocalDate.parse(request.knskKjnYmd().trim());
            List<KseflRecord> KseflRecords = new ArrayList<>();
            String ciriusMsgId = "";
            List<String> ciriusMsgUmkm = List.of();

            if (errBnriCd.equals(meterTrtkMt)) {
                List<KseflRecord> kseflRecord =
                    kseflDataAccess.findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdAndKasgYmdAndFlwStCd(
                            flttCktkCd,
                            flttCktkCd,
                            flttKtnCd,
                            openErrorReasonCodeMeterError,
                            knskKjnYmd,
                            followStatusCodeN2NotYetStarted);
                KseflRecords = kseflRecord;
            } else {
                List<KseflRecord> kseflRecord =
                    kseflDataAccess.findAllByFlttCktkCdAndFlwKtnCdAndKasErryCdOrKasErryCdAndKasgYmdAndFlwStCd(
                            flttCktkCd,
                            flttCktkCd,
                            flttKtnCd,
                            openErrorReasonCodeMeterError,
                            knskKjnYmd,
                            followStatusCodeN2NotYetStarted);
                KseflRecords = kseflRecord;
            }

            //DA-139-R006取得件数分繰返し、レスポンスパラメーターを設定する
            List<KsenflIrResponse> ksenflIrResponse = loopedRetrieval(KseflRecords);

            //DA-139-R001の取得件数=0件の場合
            if (KseflRecords.size() == 0) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000093);
            }

            //DAの取得件数によって判定
            if (KseflRecords.size() > 500) {
                ciriusMsgId = CIRIUS_MSG_ID;
                ciriusMsgUmkm = List.of("500");
            }

            return toResponse(ksenflIrResponse, ciriusMsgId, ciriusMsgUmkm);
        });
    }

    /**
     * 概要：DA-139-R006取得件数分繰返しレスポンスパラメーターを設定する
     * - 主な仕様
     * DA-139の取得件数≠0件」且つ「繰り返し項目の処理データ項番<=500件の場合SV-017取得処理を行いレスポンスに設定する
     * DA-139の取得件数≠0件」且つ「繰り返し項目の処理データ項番<=500件以外の場合は処理なしで終了
     * @param kseflRecords 開栓エラーフォロー件名レコード
     * @param hyjPageNo 表示頁番号
     * @return 開栓エラーフォロー件名一覧
     */
    private List<KsenflIrResponse> loopedRetrieval(List<KseflRecord> kseflRecords) {

        List<KsenflIrResponse> ksenflIrResponse = new ArrayList<>();

        if (0 < kseflRecords.size()) {
            for (int i = 0; i < kseflRecords.size(); i++) {
                if (i == 500) {
                    break;
                }
                KseflRecord kseflRecord = kseflRecords.get(i);

                //SV-017取得処理
                MixFieldEditOutput mixFieldEditOutput =
                    mixFieldEdit.execute(
                        kseflRecord.ctnmKjhkFlg(),
                        kseflRecord.ctnmKj1(),
                        null,
                        kseflRecord.ctnmKn1(),
                        null,
                        null,
                        null,
                        null);
                ksenflIrResponse.add(createKsenflIrResponse(kseflRecord, mixFieldEditOutput));
            }
        } else {
            return null;
        }
        return ksenflIrResponse;
    }

    /**
     * レスポンス成型する。
     *
     * @param kkisuInf 繰返し回数情報
     * @param totalKnsu 総件数
     * @param ksenflIr 開栓エラーフォロー件名一覧
     * @param ciriusMsgId ＣＩＲＩＵＳメッセージＩＤ（ワーニング）
     * @param ciriusMsgUmkm ＣＩＲＩＵＳメッセージ埋込み文字列リスト（ワーニング）
     * @return ErrorFollowTargetListGetMeterWaitResponse レスポンス
     */
    private ErrorFollowTargetListResponse toResponse(
        List<KsenflIrResponse> ksenflIr,
        String ciriusMsgId,
        List<String> ciriusMsgUmkm) {
        return new ErrorFollowTargetListResponse(
                        ksenflIr,
                        ciriusMsgId,
                        ciriusMsgUmkm,
                        requestContext.getCorrelationId()
                            .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"))
                );
    }

    /**
     * 開栓エラーフォロー件名一覧情報のレスポンスを生成する
     *
     * @param kseflRecord 開栓エラーフォロー件名record
     * @param mixFieldEditOutput SV-017取得結果
     * @return KsenflIrResponse 開栓エラーフォロー件名一覧情報
     */
    private KsenflIrResponse createKsenflIrResponse(KseflRecord kseflRecord, MixFieldEditOutput mixFieldEditOutput) {
        return new KsenflIrResponse(
            kseflRecord.gasMtstNo(),
            kseflRecord.kasErryCd(),
            kseflRecord.flttCktkCd(),
            kseflRecord.gmb(),
            formatDate(kseflRecord.kasgYmd()),
            formatDate(kseflRecord.kasHukkYmd()),
            kseflRecord.tsgkKyoTrno(),
            kseflRecord.kasgNwkCd(),
            mixFieldEditOutput.ctnm1(),
            kseflRecord.sgyAnknNo(),
            kseflRecord.tkjMem(),
            kseflRecord.recUpdJnjNo());
    }

    /**
     * 日付を ISO-8601 文字列へ変換する。
     *
     * @param value 変換対象日付
     * @return 日付文字列。{@code value} が {@code null} の場合は {@code null}
     */
    private String formatDate(LocalDate value) {
        return Objects.toString(value, null);
    }
}
