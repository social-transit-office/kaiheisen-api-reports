            operationId = "getOpeningErrorFollowDisplayGetOtherRetail")
    @APIResponse(

                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(

                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningErrorFollowDisplayGetOtherRetailResponse endpoint(@BeanParam OpeningErrorFollowDisplayGetOtherRetailRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo());
            String tksuSvkyJtkb = request.tksuSvkyJtkb();
            LocalDate tyenYmd = Rfc3339LocalDateParser.parse(request.tyenYmd());
            String tksuKijsCd = request.tksuKijsCd();
            Long syoKyoTrno = toLong(request.syoKyoTrno());
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();

            TatemonRecord tatemonRecord = null;
            KiycstdRecord kiycstdRecord = null;

            //DA-033-R006取得処理
            GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, knskKjnYmd);

            if (gmtsetdRecord.tusoGmtkYmd() == null) {
                throw businessRuleException(ErrorCodeConstants.ERROR_CODE_020100012);
            }

            //DA-110-R007取得処理
            GasmtrdRecord gasmtrdRecord = requireGasmtrd(gasMtstNo, knskKjnYmd);

            // SV-027検針情報(開栓)を取得する
            Optional<MeterReadingInfoAcquisitionForOpeningOutput> meterReadingInfoAcquisitionForOpeningOutput =
                    Optional.ofNullable(gmtsetdRecord.kngmStbJtkb())
                            .filter(kngmStbJtkb -> !kngmStbJtkb.isBlank())
                            .filter(meterPossession::equals)
                            .map(v -> gmtsetdRecord.gasMtstNo())
                            .map(meterReadingInfoAcquisitionForOpening::execute);

            //DA-007-R001取得処理
            if (gmtsetdRecord.bldNo() != null) {
                tatemonRecord = requireTatemon(gmtsetdRecord.bldNo());
            }

            //DA-104-R001取得処理
            if (syoKyoTrno != null) {
                kiycstdRecord = requireKiycstd(syoKyoTrno);
            }

            //DA-113-R001取得処理
            MtrgrpdRecord mtrgrpdRecord = requireMtrgrpd(gmtsetdRecord.gasMtstNo(), knskKjnYmd);

            //SV-079（照会レベルコード編集）を取得する
            Optional<InquiryLevelCodeEditOutput> inquiryLevelCodeEditOutput =
                    Optional.ofNullable(inquiryLevelCodeEdit.execute(
                Optional.ofNullable(kiycstdRecord)
                    .map(kiycstdRecords -> kiycstdRecords.shkiLvShbt())
                    .orElse(null)));

            return toResponse(
                    tatemonRecord,
                    meterReadingInfoAcquisitionForOpeningOutput,
                    inquiryLevelCodeEditOutput,
                    gmtsetdRecord,
                    gasmtrdRecord,
                    kiycstdRecord,
                    mtrgrpdRecord,
                    tksuSvkyJtkb,
                    tyenYmd,
                    tksuKijsCd);
        });
    }

    /**
     * DA-033-R006取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター設置場所番号＜導管＞の検索を行う
     * - 取得件数が0件の場合、業務エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return GmtsetdRecord
     */
    private GmtsetdRecord requireGmtsetd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GmtsetdRecord> gmtsetdRecords =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd);
        if (!gmtsetdRecords.isEmpty()) {
            return gmtsetdRecords.get();
        } else {
            throw businessRuleException(ErrorCodeConstants.ERROR_CODE_020000029);
        }
    }

    /**
     * DA-110-R007取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター＜導管＞の検索を行う
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return GasmtrdRecord
     */
    private GasmtrdRecord requireGasmtrd(Long gasMtstNo, LocalDate knskKjnYmd) {
        GasmtrdRecord gasmtrdRecords =
                gasmtrdDataAccess.findByTtkGmsbNoAndGmtkYmd(gasMtstNo, knskKjnYmd).orElse(null);
        return gasmtrdRecords;
    }

    /**
     * DA-007-R001取得処理
     * 主な仕様:
     * - 建物番号を引数に建物の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param bldNo 建物番号
     * @return TatemonRecord
     */
    private TatemonRecord requireTatemon(String bldNo) {
        Optional<TatemonRecord> tatemonRecords =
                tatemonDataAccess.findByBldNo(bldNo);
        if (tatemonRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    tatemonDataAccess.tableName(),
                    TatemonDataAccess.CallSpecs.FIND_BY_BLD_NO.identifier(),
                    new FlatJsonBuilder()
                            .put("bldNo", bldNo));
        }
        return tatemonRecords.get();
    }

    /**
     * DA-104-R001取得処理
     * 主な仕様:
     * - 使用者契約お客さま登録番号を引数に契約お客さまの検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param syoKyoTrno 使用者契約お客さま登録番号
     * @return KiycstdRecord
     */
    private KiycstdRecord requireKiycstd(Long syoKyoTrno) {
        Optional<KiycstdRecord> kiycstdRecords =
                kiycstdDataAccess.findByKyoTrkNo(syoKyoTrno);
        if (kiycstdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    kiycstdDataAccess.tableName(),
                    KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                    new FlatJsonBuilder()
                            .put("syoKyoTrno", syoKyoTrno));
        }
        return kiycstdRecords.get();
    }

     * @return Long型。{@code value} が {@code null} の場合は {@code null}
