    /**
     * MA-028 閉栓エラーフォロー詳細取得 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 閉栓エラーフォロー詳細取得 レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓エラーフォロー詳細取得",
            description = "閉栓エラーフォロー詳細画面の表示に必要な情報を取得する",
            operationId = "getClosingErrorFollowDetailGetResource")
    @APIResponse(
            responseCode = "200",
            description = "閉栓エラーフォロー詳細取得結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingErrorFollowDetailGetResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingErrorFollowDetailGetResponse endpoint(@BeanParam ClosingErrorFollowDetailGetRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo());
            LocalDate hssgYmd = Rfc3339LocalDateParser.parse(request.hssgYmd());
            LocalDate hesnHukkYmd = Rfc3339LocalDateParser.parse(request.hesnHukkYmd());
            Integer regTsno = Integer.valueOf(request.regTsno());
            // DA-141-R001 閉栓エラーフォロー件名 検索
            HseflRecord hseflRecord = this.findHseflRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
            // フォロー状態の判定
            if (!flStCdNotYetStarted.equals(hseflRecord.flwStCd())) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000043);
            }

            // DA-110-R007 ガスメーター＜導管＞情報 検索
            LocalDateTime systemDatetime = this.requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"));
            LocalDate systemLocalDate = systemDatetime.toLocalDate();
            GasmtrdRecord gasmtrdRecord = this.findGasmtrdRecord(gasMtstNo, systemLocalDate);

            // DA-113-R001 メーターグループ＜導管＞　検索
            Optional<MtrgrpdRecord> mtrgrpdRecord =
                    mtrgrpdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemLocalDate);

            // DA-033-R006 ガスメーター設置場所＜導管＞ 検索
            GmtsetdRecord gmtsetdRecord = this.findGmtsetdRecord(gasMtstNo, systemLocalDate);

            // DA-108-R005 託送サービス契約・ガスメーター設置 検索
            TskgmstRecord tskgmstRecord = this.findTskgmstRecord(gasMtstNo, systemLocalDate, tranSvConStInvalid);

            // DA-099-R005 託送サービス契約 検索
            TkssvkyRecord tkssvkyRecord = this.findTkssvkyRecord(tskgmstRecord.tksuSvkyNo(), systemLocalDate, tranSvConStInvalid);

            // DA-104-R001 契約お客さま＜導管＞ 検索
            KiycstdRecord kiycstdRecord = this.findKiycstdRecord(tkssvkyRecord.syoKyoTrno());

            // 建物番号 ≠ 未設定の場合
            TatemonRecord tatemonRecord = null;
            Optional<String> bldNo = Optional.ofNullable(gmtsetdRecord.bldNo());
            if (!bldNo.isEmpty()) {
                // DA-007-R001 建物 検索
                tatemonRecord = this.findTatemonRecord(gmtsetdRecord.bldNo());
            }
            // 供内管・ガスメーター設備状態区分 ＝【05800 1：取付中】の場合
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput = null;
            if (svPipeGmEquStMetPossSt.equals(gmtsetdRecord.kngmStbJtkb())) {
                // SV-0162 検針結果取得処理（閉栓）
                meterReadingResultAcquisitionForClosingOutput =
                        meterReadingResultAcquisitionForClosing.execute(gasMtstNo);
            }
            // SV-017 お客さま名ミクス、建物名ミクス項目編集
            MixFieldEditOutput mixFieldEditOutput = null;
            mixFieldEditOutput =
                    mixFieldEdit.execute(
                            kiycstdRecord.ctnmKjhkFlg(),
                            kiycstdRecord.ctnmKj1(),
                            kiycstdRecord.ctnmKj2(),
                            kiycstdRecord.ctnmKn1(),
                            kiycstdRecord.ctnmKn2(),
                            tatemonRecord != null ? tatemonRecord.bldKjHyojiCd() : null,
                            tatemonRecord != null ? tatemonRecord.bldNamKj() : null,
                            tatemonRecord != null ? tatemonRecord.bldNamKn() : null);

            // SV-010 ＧＨＰメータ設置コード編集
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput = null;
            ghpMeterInstallationCodeEditOutput =
                    ghpMeterInstallationCodeEdit.execute(
                            gmtsetdRecord.ckrtGmscJtkb(),
                            gmtsetdRecord.kkyuGmtKbn());

            // DA-113-R001の取得件数が0件（戻り値 ＝ 未設定）の場合
            Integer hrtGrpNo =
                    mtrgrpdRecord
                            .map(MtrgrpdRecord::hrtGrpNo)
                            .orElse(null);

            // レスポンスを返却
            return this.toResponse(
                    hseflRecord,
                    gasmtrdRecord,
                    gmtsetdRecord,
                    tkssvkyRecord,
                    kiycstdRecord,
                    tatemonRecord,
                    meterReadingResultAcquisitionForClosingOutput,
                    mixFieldEditOutput,
                    ghpMeterInstallationCodeEditOutput,
                    hrtGrpNo
                    );
        });
    }

    private HseflRecord findHseflRecord(Long gasMtstNo, LocalDate hssgYmd, LocalDate hesnHukkYmd, Integer regTsno) {

        Optional<HseflRecord> record = hseflDataAccess.findByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsno(
                gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
        if (!record.isEmpty()) {
            return record.get();
        }
        // 取得件数が0件の場合 例外スロー
        throw DataInconsistencySupport.create(
                hseflDataAccess.tableName(),
                HseflDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_HSSG_YMD_AND_HESN_HUKK_YMD_AND_REG_TSNO.identifier(),
                new FlatJsonBuilder()
                .put("gasMtstNo", gasMtstNo)
                .put("hssgYmd", hssgYmd)
                .put("hesnHukkYmd", hesnHukkYmd)
                .put("regTsno", regTsno));

    }

     * @return ClosingErrorFollowDetailGetResponse レスポンス
     */
    private ClosingErrorFollowDetailGetResponse toResponse(
            HseflRecord hseflRecord,
            GasmtrdRecord gasmtrdRecord,
            GmtsetdRecord gmtsetdRecord,
            TkssvkyRecord tkssvkyRecord,
            KiycstdRecord kiycstdRecord,
            TatemonRecord tatemonRecord,
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput,
            MixFieldEditOutput mixFieldEditOutput,
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput,
            Integer hrtGrpNo) {
        return new ClosingErrorFollowDetailGetResponse(
                this.createKiycstInfoResponse(
                        kiycstdRecord, mixFieldEditOutput),
                this.createTatemonInfoResponse(
                        tatemonRecord, mixFieldEditOutput),
                this.createGmtsetdInfoResponse(
                        gmtsetdRecord, ghpMeterInstallationCodeEditOutput),
                this.createGmtInfoResponse(gasmtrdRecord),
                this.createKnsKkaInfoResponse(meterReadingResultAcquisitionForClosingOutput),
                this.createHseflInfoResponse(hseflRecord),
                this.createMtrgrpdInfoResponse(hrtGrpNo),
                this.createTksuKijsCd(tkssvkyRecord),
                requestContext.getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"))
                );

    }
