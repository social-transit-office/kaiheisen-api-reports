    /**
     * MA-030 閉栓エラーフォロー不要登録　TG小売 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 閉栓エラーフォロー不要登録　TG小売 レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓エラーフォロー不要登録　TG小売",
            description = "TG小売客番に閉栓エラーフォロー不要登録を行う",
            operationId = "postClosingErrorUnnecessaryRegisterTgRetailResource")
    @APIResponse(
            responseCode = "200",
            description = "閉栓エラーフォロー不要後の結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingErrorUnnecessaryRegisterTgRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingErrorUnnecessaryRegisterTgRetailResponse endpoint(ClosingErrorUnnecessaryRegisterTgRetailRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo());
            LocalDate hssgYmd = Rfc3339LocalDateParser.parse(request.hssgYmd());
            LocalDate hesnHukkYmd = Rfc3339LocalDateParser.parse(request.hesnHukkYmd());
            Integer regTsno = Integer.valueOf(request.regTsno());
            Integer recUpdJnjNo = Integer.valueOf(request.recUpdJnjNo());
            LocalDateTime systemDatetime = this.requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"));
            // DA-141-U001パラメータ設定
            LocalDate systemLocalDate = systemDatetime.toLocalDate();
            HseflWriteRecord record = new HseflWriteRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno, null, null,
                    flStCdCompletedNoFollowRequired, systemLocalDate, null, null, null, null, null, null, null, null, null,
                    null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null, null);
            Set<HseflColumn> targetColumns = Set.of(HseflColumn.FLW_ST_CD, HseflColumn.FLTK_YMD);
            HseflU001UpdateKey key = new HseflU001UpdateKey(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno, recUpdJnjNo);
            String hukkSysShbt = this.requestContext.getHukkSysShbt()
                    .orElseThrow(() -> new IllegalStateException("hukkSysShbt is not set in RequestContext"));
            String decodedHkkSysSyb = kaiheisenConstantService.findNameByTypeAndValue(
                    ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE, hukkSysShbt);

            AuditContext context = new AuditContext(usrId, decodedHkkSysSyb);

            // DA-141-U001:GMB・KASG_YMD・KAS_HUKK_YMD・SGY_ANKN_NO・REC_UPD_JNJ_NOによる更新
            try {
                hseflDataAccess.updateByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsnoAndRecUpdJnjNo(
                        record, targetColumns, key, context);
            } catch (DatabaseException e) {
                String daIdentifier = e.getDaIdentifier();
                String sql = e.getSql();
                Object[] sqlBindPar = e.getSqlBindParameters();
                Throwable cause = e.getCause();
                // DA-141-U001で取得したエラーコード  = "030400001 " (排他エラー)が返却された場合
                if (cause.getCause() instanceof ExclusiveException) {
                    // DA-141-R001 による取得
                    HseflRecord hseflRecord1 = findHseflRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
                    // フォロー状態の判定
                    if (!flStCdNotYetStarted.equals(hseflRecord1.flwStCd())) {
                        throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000043);
                    } else {
                        throw new DatabaseException(
                                daIdentifier,
                                sql,
                                sqlBindPar,
                                cause
                                );
                    }
                } else {
                    throw e ;
                }
            }
            // DA-141-U001で取得したエラーコード  =　未設定　の場合
            // DA-141-R001 による取得
            HseflRecord hseflRecord2 = findHseflRecord(gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
            // SV-087 申込種別取得
            String correlationId = this.requestContext.getCorrelationId()
                    .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"));
            String hesnJysb = hseflRecord2.hesnJysb();
            String htyoSysShbtCd = hseflRecord2.hukkSysShbt();
            ApplicationTypeAcquisitionOutput applicationTypeAcquisitionOutput =
                    applicationTypeAcquisition.execute(null, hesnJysb, htyoSysShbtCd);

            // DA-141-R001の受付インデックス番号 = 未設定 以外、かつSV-087の申込種別 = 未設定 以外の場合
            Optional<String> utidxNo = Optional.ofNullable(hseflRecord2.utidxNo());
            Optional<String> mkSbt = Optional.ofNullable(applicationTypeAcquisitionOutput.mkSbt());
            if (!utidxNo.isEmpty() && !mkSbt.isEmpty() ) {
                // DA-135-R001 ＮＷ箇所名を取得する
                String nwkCd = hseflRecord2.hssgNwkCd();
                NwkseigRecord nwkseigRecord = findNwkseigRecord(nwkCd, hssgYmd);
                // SV-092 受付情報変更通知API連携
                String utidxNo2 = hseflRecord2.utidxNo();
                String uketkSeiriNo = hseflRecord2.sgyAnknNo();
                Long cstRegNo = hseflRecord2.syoKyoTrno();
                String nwknKj = nwkseigRecord.nwknKj();
                String sagyoSyaName = hseflRecord2.sgytSinnKj();
                receptionInfoChangeNotificationApiIntegration.execute(
                        utidxNo2, uketkSeiriNo, null, procedureStatusCancel, gasMtstNo, cstRegNo,
                        cusSvHstConOther, nwknKj, sagyoSyaName,
                        applicationTypeAcquisitionOutput.mkSbt(), hssgYmd, LocalDate.of(1900, 1, 1),
                        null, null);
            }
            return new ClosingErrorUnnecessaryRegisterTgRetailResponse(correlationId);
        });
    }

    private HseflRecord findHseflRecord(Long gasMtstNo, LocalDate hssgYmd, LocalDate hesnHukkYmd, Integer regTsno) {

        Optional<HseflRecord> record = hseflDataAccess.findByGasMtstNoAndHssgYmdAndHesnHukkYmdAndRegTsno(
                gasMtstNo, hssgYmd, hesnHukkYmd, regTsno);
        if (!record.isEmpty()) {
            return record.get();
        }
        throw DataInconsistencySupport.create(
                hseflDataAccess.tableName(),
                HseflDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_HSSG_YMD_AND_HESN_HUKK_YMD_AND_REG_TSNO.identifier(),
                new FlatJsonBuilder()
                .put("gasMtstNo", gasMtstNo)
                .put("hssgYmd", hssgYmd)
                .put("hesnHukkYmd", hesnHukkYmd)
                .put("regTsno", regTsno));

    }

    private NwkseigRecord findNwkseigRecord(String nwkCd, LocalDate hssgYmd) {

        Optional<NwkseigRecord> record = nwkseigDataAccess.findByNwkCdAndTkysYmdAndTyenYmd(nwkCd, hssgYmd);
        if (!record.isEmpty()) {
            return record.get();
        }
        throw DataInconsistencySupport.create(
                nwkseigDataAccess.tableName(),
                NwkseigDataAccess.CallSpecs.FIND_BY_NWK_CD_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                .put("nwkCd", nwkCd)
                .put("hssgYmd", hssgYmd));

    }
