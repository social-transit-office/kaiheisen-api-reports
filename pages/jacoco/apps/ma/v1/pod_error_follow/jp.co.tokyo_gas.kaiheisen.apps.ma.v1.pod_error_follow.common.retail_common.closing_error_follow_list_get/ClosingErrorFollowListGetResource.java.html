/**
 * MA-027 閉栓エラーフォロー一覧取得-共通-共通
 */
@Path(ClosingErrorFollowListGetEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "閉栓エラーフォロー一覧取得", description = "MA-027 閉栓エラーフォロー一覧取得 API")
public class ClosingErrorFollowListGetResource extends BaseResource<ClosingErrorFollowListGetRequest> {

    private final HseflDataAccess hseflDataAccess;

    private String followStatusCodeNotFollowed;

    private List<String> hesnJysbListIsEmpty;

    private List<String> hesnJysbListSupplyOutage;

    private List<String> hesnJysbListOtherThanSupplyOutage;

    private final MixFieldEdit mixFieldEdit;

    /** 定数：配列用数字 (501) */
    private static final int GET_LIMIT_NUM = 500;

    /** 定数：CIRIUSメッセージID（CXXM00037:表示上限を超えたため、{0}件のみ表示します。)*/
    private static final String CIRIUS_MSG_ID_CXXM00037W = "CXXM00037W";

    /** 定数：CXXM00037用埋め込み文字 */
    private static final String UMKM_MOJI_500 = "500";

    /** 定数：フォロー担当拠点コード.支社("0") */
    private static final String SHISHA = "0";

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム時刻プロバイダ
     */
    @Inject
    public ClosingErrorFollowListGetResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @ClosingErrorFollowListGetConfiguration
            MixFieldEdit mixFieldEdit,
            CiriusConstantService ciriusConstantService,
            HseflDataAccess hseflDataAccess) {
        super(requestContext, requestDtoValidator, clock);
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        this.followStatusCodeNotFollowed = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);
        this.hesnJysbListIsEmpty = List.of();
        this.hesnJysbListSupplyOutage = List.of(RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE));
        this.hesnJysbListOtherThanSupplyOutage = List.of(RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.RELOCATION_CLOSURE),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.UNUSED_CLOSURE),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.FIRED_AND_LOST_CAP),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE),
                RequiredConstantResolver.requireCiriusConstant(
                        resolvedCiriusConstantService,
                        ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                        ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER));
        this.mixFieldEdit = Objects.requireNonNull(
                mixFieldEdit,
                "mixFieldEdit");
    }

    /**
     * MA-027 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓エラーフォロー一覧取得",
            description = "閉栓フォロー対象一覧を表示するため閉栓エラーフォロー件名を検索し、情報を取得する",
            operationId = "getClosingErrorFollowListGet")
    @APIResponse(
            responseCode = "200",
            description = "閉栓エラーフォロー一覧取得結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingErrorFollowListGetResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingErrorFollowListGetResponse endpoint(@BeanParam ClosingErrorFollowListGetRequest request) {
        return execute(request, () -> {
            String flttCktkCd = request.flttCktkCd();
            String flttKtnCd = request.flttKtnCd();
            String hesnJysbKbn = request.hesnJysbKbn();
            String ciriusMsgId = "";
            List<String> ciriusMsgUmkm = List.of();
            List<HseflRecord> hseflRecords = SHISHA.equals(flttKtnCd)
            // リクエストパラメーター.フォロー担当拠点コード=0(支社)の場合
                ? switch (hesnJysbKbn) {
                case "1" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        null,
                        followStatusCodeNotFollowed,
                        hesnJysbListIsEmpty);
                case "2" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        null,
                        followStatusCodeNotFollowed,
                        hesnJysbListSupplyOutage);
                case "3" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        null,
                        followStatusCodeNotFollowed,
                        hesnJysbListOtherThanSupplyOutage);
                default -> List.of();
            }
                // リクエストパラメーター.フォロー担当拠点コード≠0(支社)の場合
                : switch (hesnJysbKbn) {
                case "1" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        flttKtnCd,
                        followStatusCodeNotFollowed,
                            hesnJysbListIsEmpty);
                case "2" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        flttKtnCd,
                        followStatusCodeNotFollowed,
                        hesnJysbListSupplyOutage);
                case "3" ->
                hseflDataAccess.findAllByFlttCktkCdAndFlttKtnCdAndFlwStCd(
                        flttCktkCd,
                        flttKtnCd,
                        followStatusCodeNotFollowed,
                        hesnJysbListOtherThanSupplyOutage);
                default -> List.of();
            };
            //DA-141-R004取得件数分繰返しレスポンスパラメーターを設定する
            List<HseflIrResponse> hseflIrResponse = loopedRetrieval(hseflRecords);

            // DA-141-R004の取得件数=0の場合
            if (0 == hseflRecords.size()) {
                // BusinessRuleException（該当データなし）
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000093);
            }
            // DA-141-R004の取得件数＞500件の場合
            if (GET_LIMIT_NUM < hseflRecords.size()) {
                // CXXM00037:表示上限を超えたため、{0}件のみ表示します。
                ciriusMsgId = CIRIUS_MSG_ID_CXXM00037W;
                // {0}:表示上限件数
                ciriusMsgUmkm = List.of(UMKM_MOJI_500);
            }
            return toResponse(hseflIrResponse, ciriusMsgId, ciriusMsgUmkm);
        });
    }

    private List<HseflIrResponse> loopedRetrieval(List<HseflRecord> hseflRecords) {
        List<HseflIrResponse> hsenflIrResponse = new ArrayList<>();
        if (0 < hseflRecords.size()) {
            for (int i = 0; i < hseflRecords.size() && i < GET_LIMIT_NUM; i++) {
                HseflRecord hseflRecord = hseflRecords.get(i);
                //SV-017(ミクス項目編集)処理呼出し
                MixFieldEditOutput mixFieldEditOutput =
                    mixFieldEdit.execute(
                        hseflRecord.ctnmKjhkFlg(),
                        hseflRecord.ctnmKj1(),
                        null,
                        hseflRecord.ctnmKn1(),
                        null,
                        null,
                        null,
                        null);
                hsenflIrResponse.add(createHseflIrResponse(hseflRecord, mixFieldEditOutput));
            }
        } else {
            return null;
        }
        return hsenflIrResponse;
    }

    /**
     * レスポンス成型する。
     *
     * @param hsenflIr 開栓エラーフォロー件名一覧
     * @param ciriusMsgId ＣＩＲＩＵＳメッセージＩＤ（ワーニング）
     * @param ciriusMsgUmkm ＣＩＲＩＵＳメッセージ埋込み文字列リスト（ワーニング）
     * @return ClosingErrorFollowListGetResponse レスポンス
     */
    private ClosingErrorFollowListGetResponse toResponse(
        List<HseflIrResponse> hsenflIr,
        String ciriusMsgId,
        List<String> ciriusMsgUmkm) {
        return new ClosingErrorFollowListGetResponse(
                        hsenflIr,
                        ciriusMsgId,
                        ciriusMsgUmkm,
                        requestContext.getCorrelationId()
                            .orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"))
                );
    }

    /**
     * 閉栓エラーフォロー件名一覧情報のレスポンスを生成する
     *
     * @param hseflRecord 閉栓エラーフォロー件名record
     * @param mixFieldEditOutput SV-017取得結果
     * @return HseflIrResponse 閉栓エラーフォロー件名一覧情報
     */
    private HseflIrResponse createHseflIrResponse(HseflRecord hseflRecord, MixFieldEditOutput mixFieldEditOutput) {
        return new HseflIrResponse(
            hseflRecord.gasMtstNo(),
            formatDate(hseflRecord.hssgYmd()),
            formatDate(hseflRecord.hesnHukkYmd()),
            Long.valueOf(hseflRecord.regTsno()),
            hseflRecord.hssgNwkCd(),
            hseflRecord.hesnErryCd(),
            hseflRecord.hesnJysb(),
            mixFieldEditOutput.ctnm1(),
            hseflRecord.ctnm1HshiShbt(),
            hseflRecord.ctnm1HujnShbt(),
            hseflRecord.tkjMem(),
            hseflRecord.recUpdJnjNo());
    }

    /**
     * 日付を ISO-8601 文字列へ変換する。
     *
     * @param value 変換対象日付
     * @return 日付文字列。{@code value} が {@code null} の場合は {@code null}
     */
    private String formatDate(LocalDate value) {
        return Objects.toString(value, null);
    }
}
