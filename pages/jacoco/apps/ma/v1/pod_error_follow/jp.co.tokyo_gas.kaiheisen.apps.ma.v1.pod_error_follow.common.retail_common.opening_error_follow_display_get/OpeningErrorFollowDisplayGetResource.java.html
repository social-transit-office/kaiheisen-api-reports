        return execute(request, () -> {
            String gmb = request.gmb();
            LocalDate kasgYmd = Rfc3339LocalDateParser.parse(request.kasgYmd());
            LocalDate kasHukkYmd = Rfc3339LocalDateParser.parse(request.kasHukkYmd());
            String sgyAnknNo = request.sgyAnknNo();
            LocalDate systemDate = requestContext.getSystemDatetime().map(LocalDateTime::toLocalDate)
                    .orElseThrow(
                            () -> new IllegalStateException("systemDatetime is not set in RequestContext"));
            // DA-139-R004 開栓エラーフォロー件名取得
            KseflRecord kseflRecord =
                    kseflDataAccess.findByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNo(
                            gmb, kasgYmd, kasHukkYmd, sgyAnknNo)
                    .orElseThrow(() -> DataInconsistencySupport.create(
                            kseflDataAccess.tableName(),
                            KseflDataAccess.CallSpecs.FIND_BY_GMB_AND_KASG_YMD_AND_KAS_HUKK_YMD_AND_SGY_ANKN_NO.identifier(),
                            new FlatJsonBuilder()
                                    .put("gmb", gmb)
                                    .put("kasgYmd", kasgYmd)
                                    .put("kasHukkYmd", kasHukkYmd)
                                    .put("sgyAnknNo", sgyAnknNo)));
            if (!flwMityksyu.equals(kseflRecord.flwStCd())) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000043);
            }

            // 照会レベル種別
            String sykiLevelSbt;

            if (otherRetail.equals(kseflRecord.tgTkurKbn())) {
                // 契約お客さま＜導管＞情報を取得する
                KiycstdRecord kiycstdRecordForSykiLevelSbt =
                        kiycstdDataAccess.findByKyoTrkNo(kseflRecord.kassKyoTrno())
                                .orElseThrow(() -> DataInconsistencySupport.create(
                                        kiycstdDataAccess.tableName(),
                                        KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                                        new FlatJsonBuilder()
                                                .put("kyoTrkNo", kseflRecord.kassKyoTrno())));
                // 照会レベル種別に値を設定
                sykiLevelSbt = kiycstdRecordForSykiLevelSbt.shkiLvShbt();
            } else {
                // 契約お客さま＜小売＞情報を取得する
                KiycstkRecord kiycstkRecordForSykiLevelSbt =
                        kiycstkDataAccess.findByKyoTrkNo(kseflRecord.kassKyoTrno())
                                .orElseThrow(() -> DataInconsistencySupport.create(
                                        kiycstkDataAccess.tableName(),
                                        KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                                        new FlatJsonBuilder()
                                                .put("kyoTrkNo", kseflRecord.kassKyoTrno())));
                // 照会レベル種別に値を設定

            }

                    .orElseGet(() -> gasmtrdDataAccess.findByGmbAndGmtkYmd(kseflRecord.gmb(), systemDate)

                                    .orElse(null));

            // DA-033-R006 ガスメーター設置場所＜導管＞情報を取得する
            Optional<GmtsetdRecord> gmtsetdRecord = Optional.ofNullable(gasMtstNo)
                    .map( gmtNo -> gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gmtNo, systemDate))
                    .orElse(Optional.empty());

            // ガスメーター指示数
            Long gmtShjSu = null;
            // 検針日
            LocalDate knsYmd = null;
            // 検針結果作成事由種別
            String knkkSsjyShbt = null;
            // 建物情報
            TatemonRecord tatemonRecord = null;
            // SV-001 小売判定用情報取得
            RetailTypeJudgmentOutput retailTypeJudgmentOutput = null;
            //SV-027 検針結果情報
            MeterReadingInfoAcquisitionForOpeningOutput meterReadingInfoAcquisitionForOpeningOutput = null;
            // SV-042 検針結果情報
            MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput = null;
            // DA-104-R001 契約お客さま＜導管＞情報
            KiycstdRecord kiycstdRecord = null;
            // DA-110-R007 ガスメーター＜導管＞情報
            Optional<GasmtrdRecord> gasmtrdRecord = java.util.Optional.empty();
            // DA-113-R001 メーターグループ＜導管＞情報
            Optional<MtrgrpdRecord> mtrgrpdRecord = java.util.Optional.empty();
            // DA-038-R007 サービス契約情報
            SvckyaRecord svckyaRecord = null;

            if (gmtsetdRecord.isPresent()) {
                if (meterPossession.equals(gmtsetdRecord.map(GmtsetdRecord::kngmStbJtkb).orElse(null))) {
                    // SV-027 検針結果（開栓）情報を取得する
                    meterReadingInfoAcquisitionForOpeningOutput =
                            meterReadingInfoAcquisitionForOpening.execute(gmtsetdRecord.map(GmtsetdRecord::gasMtstNo).orElse(null));
                    gmtShjSu = meterReadingInfoAcquisitionForOpeningOutput.gmtShjSu();
                    knsYmd = meterReadingInfoAcquisitionForOpeningOutput.knsYmd();
                    knkkSsjyShbt = meterReadingInfoAcquisitionForOpeningOutput.knkkSsjyShbt();
                }

                // DA-007-R001 建物情報を取得する
                tatemonRecord = gmtsetdRecord
                        .filter(record -> record.bldNo() != null)
                        .map(GmtsetdRecord::bldNo)
                        .map(bldNo -> tatemonDataAccess.findByBldNo(bldNo)
                                .orElseThrow(() -> DataInconsistencySupport.create(
                                        tatemonDataAccess.tableName(),
                                        TatemonDataAccess.CallSpecs.FIND_BY_BLD_NO.identifier(),

                        .orElse(null);

                // SV-001 小売判定用情報取得
                retailTypeJudgmentOutput =
                        retailTypeJudgment.execute(SHRKBN, gasMtstNo, systemDate);

                if (retailTypeJudgmentCodeTokyoGas.equals(retailTypeJudgmentOutput.kuHntKbn())) {
                    if (contractActive.equals(retailTypeJudgmentOutput.tkssvkyInf().tksuSvkyJtkb())) {
                        // SV-003 サービス契約番号情報を取得する
                        ServiceContractNumberAcquisitionOutput serviceContractNumberAcquisitionOutput =
                                serviceContractNumberAcquisition.execute(
                                    gmtsetdRecord.map(GmtsetdRecord::gasMtstNo).orElse(null),
                                    gmtsetdRecord.map(GmtsetdRecord::mtgMtstNo).orElse(null),
                                    serviceContractNumberAcquisitionConstantsNearest,
                                    gmtsetdRecord.map(GmtsetdRecord::ckrtGmscJtkb).orElse(null),
                                    systemDate);
                        // DA-038-R007 サービス契約情報を取得する
                        svckyaRecord = svckyaDataAccess.findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNot(
                                serviceContractNumberAcquisitionOutput.svkyNo(),
                                systemDate,
                                contractActive)
                                .orElseThrow(() -> DataInconsistencySupport.create(
                                        svckyaDataAccess.tableName(),
                                        SvckyaDataAccess.CallSpecs.FIND_LATEST_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_NOT.identifier(),
                                                                new FlatJsonBuilder()
                                                                        .put("svkyNo", serviceContractNumberAcquisitionOutput.svkyNo())
                                                                        .put("knskKjnYmd", systemDate)
                                                                        .put("svkyJtkb", contractActive)));
                    }

                    // SV-042 検針結果情報を取得する
                    meterReadingResultInfoAcquisitionOutput =
                            meterReadingResultInfoAcquisition.execute(
                                    gmtsetdRecord.map(GmtsetdRecord::hesnJysb).orElse(null),
                                    gmtsetdRecord.map(GmtsetdRecord::gasMtstNo).orElse(null),
                                    retailTypeJudgmentOutput.kuHntKbn(),
                                    knsYmd,
                                    gmtShjSu,
                                    knkkSsjyShbt);
                }

                if (retailTypeJudgmentOutput.tskgmstInf().tksuSvkyNo() != null) {
                    // DA104-R001 契約お客さま＜導管＞情報取得
                    kiycstdRecord = kiycstdDataAccess.findByKyoTrkNo(retailTypeJudgmentOutput.tkssvkyInf().syoKyoTrno())
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    kiycstdDataAccess.tableName(),
                                    KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                                    new FlatJsonBuilder()
                                            .put("kyoTrkNo", kseflRecord.kassKyoTrno())));
                }

                // DA-110-R007 ガスメーター＜導管＞情報取得
                gasmtrdRecord = gasmtrdDataAccess.findByTtkGmsbNoAndGmtkYmd(
                        gmtsetdRecord.map(GmtsetdRecord::gasMtstNo).orElse(null),
                        systemDate);
                // DA-113-R001
                mtrgrpdRecord = mtrgrpdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(
                        gmtsetdRecord.map(GmtsetdRecord::gasMtstNo).orElse(null),
                        systemDate);
            }

            // SV-076 建物漢字表示可否フラグ取得
            BuildingKanjiDisplayFlagAcquisitionOutput buildingKanjiDisplayFlagAcquisitionOutput =
                    buildingKanjiDisplayFlagAcquisition.execute(kseflRecord.bldnKj());
            // SV-077 住宅区分コード取得
            HousingCategoryCodeAcquisitionOutput housingCategoryCodeAcquisitionOutput =
                    housingCategoryCodeAcquisition.execute(kseflRecord.bldnKn());
            // SV-079 照会レベルコード編集
            InquiryLevelCodeEditOutput inquiryLevelCodeEditOutput =
                    inquiryLevelCodeEdit.execute(
                            kiycstdRecord != null ? kiycstdRecord.shkiLvShbt() : null);
            // SV-081供給停止精算状態区分を取得する
            SupplySuspensionSettlementStatusAcquisitionOutput supplySuspensionSettlementStatusAcquisitionOutput =
                    supplySuspensionSettlementStatusAcquisition.execute(
                            gmtsetdRecord.map(GmtsetdRecord::hesnJysb).orElse(null),
                            retailTypeJudgmentOutput != null
                                ? retailTypeJudgmentOutput.tskgmstInf().tksuSvkyNo() != null
                                : false,
                            retailTypeJudgmentOutput != null
                                ? retailTypeJudgmentOutput.tkssvkyInf().tksuKijsCd()
                                : null,
                            meterReadingResultInfoAcquisitionOutput != null
                                ? meterReadingResultInfoAcquisitionOutput.kkysSisnJtkb()
                                : null);

            return toResponse(
                    kseflRecord,
                    kiycstdRecord,
                    tatemonRecord,
                    gmtsetdRecord,
                    gasmtrdRecord,
                    svckyaRecord,
                    buildingKanjiDisplayFlagAcquisitionOutput,
                    housingCategoryCodeAcquisitionOutput,
                    inquiryLevelCodeEditOutput,
                    supplySuspensionSettlementStatusAcquisitionOutput,
                    meterReadingInfoAcquisitionForOpeningOutput,
                    meterReadingResultInfoAcquisitionOutput,
                    retailTypeJudgmentOutput,
                    mtrgrpdRecord,
                    sykiLevelSbt
            );
        });
    }

    /**
     * ガス使用先情報を生成する。
     *
     * @param tatemonRecord 建物情報
     * @param gmtsetdRecord ガスメーター設置場所情報
     * @param gasmtrdRecord ガスメーター情報
     * @param meterReadingInfoAcquisitionForOpeningOutput 検針情報(開栓)
     * @param meterReadingResultInfoAcquisitionOutput 検針結果情報
     * @param svckyaRecord サービス契約情報
     * @param kiycstdRecord 契約お客さま情報
     * @param retailTypeJudgmentOutput 小売判定結果
     * @param inquiryLevelCodeEditOutput 照会レベルコード編集結果
     * @return ガス使用先情報
     */
    private GasSyskInfoResponse createGasSyskInfoResponse(
            TatemonRecord tatemonRecord,
            Optional<GmtsetdRecord> gmtsetdRecord,
            Optional<GasmtrdRecord> gasmtrdRecord,
            MeterReadingInfoAcquisitionForOpeningOutput meterReadingInfoAcquisitionForOpeningOutput,
            MeterReadingResultInfoAcquisitionOutput meterReadingResultInfoAcquisitionOutput,
            SvckyaRecord svckyaRecord,
            KiycstdRecord kiycstdRecord,
            RetailTypeJudgmentOutput retailTypeJudgmentOutput,
            InquiryLevelCodeEditOutput inquiryLevelCodeEditOutput) {

        TatemonInfoResponse tatemonInfoResponse =
                tatemonRecord != null
                ? new TatemonInfoResponse(
                        tatemonRecord.bldNamKn(),
                        tatemonRecord.bldNamKj(),
                        tatemonRecord.bldKjHyojiCd(),
                        tatemonRecord.jyutakuKbnCd()
                )
                : new TatemonInfoResponse(
                        null,
                        null,
                        null,
                        null
                );

        GmtsetdInfo2Response gmtsetdInfo2Response = gmtsetdRecord.map(x ->
                new GmtsetdInfo2Response(
                        x.gasMtstNo(),
                        x.szcGkkCd(),
                        x.szcCmazCd(),
                        x.szcMchCd(),
                        x.gmtKhsJtkb(),
                        x.kngmStbJtkb(),
                        x.cbgSuj1(),
                        x.cbgKn2(),
                        x.cbgKn3(),
                        x.tunoKn(),
                        x.hynKn(),
                        x.htuFlg(),
                        x.mtgMtstNo(),
                        Optional.ofNullable(x.kngmSbmsYmd()).map(LocalDate::toString).orElse(null),
                        Optional.ofNullable(x.kasYmd()).map(LocalDate::toString).orElse(null),
                        Optional.ofNullable(x.hesnYmd()).map(LocalDate::toString).orElse(null),
                        Optional.ofNullable(x.kusYmd()).map(LocalDate::toString).orElse(null),
                        x.ckrtGmscJtkb(),
                        x.kkyuGmtKbn(),
                        x.hesnJysb()
                )).orElse(
                        new GmtsetdInfo2Response(
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null,
                                null
                        )
                );

        GasmtrdInfoResponse gasmtrdInfoResponse =
                gasmtrdRecord.map(x -> new GasmtrdInfoResponse(x.gmb()))
                        .orElse(new GasmtrdInfoResponse(null));

        KnsKkaInfoResponse knsKkaInfoResponse;

        if (retailTypeJudgmentOutput != null
                && retailTypeJudgmentCodeTokyoGas.equals(
                        retailTypeJudgmentOutput.kuHntKbn())) {

            // SV-042
            knsKkaInfoResponse =
                    meterReadingResultInfoAcquisitionOutput != null
                            ? new KnsKkaInfoResponse(
                                    meterReadingResultInfoAcquisitionOutput.sokKnsnYmd() != null
                                            ? meterReadingResultInfoAcquisitionOutput.sokKnsnYmd().toString()
                                            : null,
                                    meterReadingResultInfoAcquisitionOutput.gmtShjSu(),
                                    meterReadingResultInfoAcquisitionOutput.knkkSsjyShbt())
                            : new KnsKkaInfoResponse(
                                    null,
                                    null,
                                    null);

        } else {

            // SV-027
            knsKkaInfoResponse =
                    meterReadingInfoAcquisitionForOpeningOutput != null
                            ? new KnsKkaInfoResponse(
                                    meterReadingInfoAcquisitionForOpeningOutput.knsYmd() != null
                                            ? meterReadingInfoAcquisitionForOpeningOutput.knsYmd().toString()
                                            : null,
                                    meterReadingInfoAcquisitionForOpeningOutput.gmtShjSu(),
                                    meterReadingInfoAcquisitionForOpeningOutput.knkkSsjyShbt())
                            : new KnsKkaInfoResponse(
                                    null,
                                    null,
                                    null);
        }

        Long svkyNo = null;

        if (retailTypeJudgmentOutput != null) {
            if (retailTypeJudgmentCodeTokyoGas.equals(retailTypeJudgmentOutput.kuHntKbn())) {
                // DA-038-R007 のサービス契約番号
                svkyNo = svckyaRecord != null ? svckyaRecord.svkyNo() : null;
            } else {
                // SV-001 の託送サービス契約番号
                svkyNo = retailTypeJudgmentOutput.tkssvkyInf().tksuSvkyNo();
            }
        }

        String sziKziKbn = svckyaRecord != null ? svckyaRecord.sziKziKbn() : null;

        SvckyaInfoResponse svckyaInfoResponse =
                new SvckyaInfoResponse(
                        svkyNo,
                        sziKziKbn
                );


        KiycstdInfoResponse kiycstdInfoResponse =
                kiycstdRecord != null
                        ? new KiycstdInfoResponse(
                                kiycstdRecord.ctnmKn1(),
                                kiycstdRecord.ctnm1HujnShbt(),
                                kiycstdRecord.ctnm1HshiShbt(),
                                inquiryLevelCodeEditOutput.syokaiLevelCd()
                        )
                        :  new KiycstdInfoResponse(
                                null,
                                null,
                                null,
                                null
                        );

        return new GasSyskInfoResponse(
                tatemonInfoResponse,
                gmtsetdInfo2Response,
                gasmtrdInfoResponse,
                knsKkaInfoResponse,
                svckyaInfoResponse,
                kiycstdInfoResponse
        );
    }
