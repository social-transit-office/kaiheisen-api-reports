    /**
     * MA-015 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "開栓エラーフォロー更新",
            description = "開栓エラーフォローを更新する",
            operationId = "postOpeningErrorFollowUpdateResource")
    @APIResponse(
            responseCode = "200",
            description = "正常応答",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OpeningErrorFollowUpdateResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OpeningErrorFollowUpdateResponse endpoint(OpeningErrorFollowUpdateRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.htiJsInf().gasMtstNo());
            String gkkCd = request.htiJsInf().szcGkkCd();
            String mchCd = request.htiJsInf().szcMchCd();
            String cmazCd = request.htiJsInf().szcCmazCd();
            String gmb = request.htiJsInf().ksenErrInf().gmb();
            LocalDate kasgYmd = Rfc3339LocalDateParser.parse(request.htiJsInf().ksenErrInf().kasgYmd());
            LocalDate kasHukkYmd = Rfc3339LocalDateParser.parse(request.htiJsInf().ksenErrInf().kasHukkYmd());
            String sgyAnknNo = request.htiJsInf().ksenErrInf().sgyAnknNo();
            LocalDate kasYmd = Rfc3339LocalDateParser.parse(request.htiJsInf().kasgYmd());
            String tkourKbn = request.tkourKbn();
            String correlationId = requestContext.getCorrelationId().orElseThrow(
                    () -> new IllegalStateException("correlationId is not set in RequestContext"));
            LocalDate knskKjnYmd =
                    requestContext.getSystemDatetime()
                            .map(LocalDateTime::toLocalDate)
                            .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"));

            // DA-033-R006 ガスメーター設置場所情報を取得
            GmtsetdRecord gmtsetdRecord =
                    gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    gmtsetdDataAccess.tableName(),
                                    GmtsetdDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                                    new FlatJsonBuilder()
                                            .put("gasMtstNo", gasMtstNo)
                                            .put("knskKjnYmd", knskKjnYmd)));

            // SV-085 開栓日チェック（開栓専用）
            openingDateCheckForOpening.execute(
                    kasYmd,
                    gmtsetdRecord.kngmStbJtkb(),
                    gmtsetdRecord.gasMtstNo());

            // DA-002-R-001 住所情報を取得する
            JusyomRecord jusyomRecord =
                    jusyomDataAccess.findByGkkCdAndMchCdAndCmazCd(gkkCd, mchCd, cmazCd)
                            .orElseThrow(() -> DataInconsistencySupport.create(
                                    jusyomDataAccess.tableName(),
                                    JusyomDataAccess.CallSpecs.FIND_BY_GKK_CD_AND_MCH_CD_AND_CMAZ_CD.identifier(),
                                    new FlatJsonBuilder()
                                            .put("gkkCd", gkkCd)
                                            .put("mchCd", mchCd)
                                            .put("cmazCd", cmazCd)));

            // DA-139-R004 開栓エラーフォロー件名情報を取得する
            Optional<KseflRecord> kseflRecord =
                    kseflDataAccess.findByGmbAndKasgYmdAndKasHukkYmdAndSgyAnknNo(
                            gmb, kasgYmd, kasHukkYmd, sgyAnknNo);

            // 開栓共通エラー判定
            kseflRecord.ifPresentOrElse(
                    record -> {
                        if (!flwMcs.equals(record.flwStCd())) {
                            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_030000001);
                        }
                    },
                    () -> {
                        throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_030000001);
                    }
            );

            String kasJysb = kseflRecord.map(KseflRecord::kasJysb).orElse(null);

            try {
                this.callOpeningOperation(kasJysb, tkourKbn, request, kseflRecord, jusyomRecord);
            } catch (MaApiException ex) {
                throw new MaPassthroughMaApiException(ex, this::handleError);
            }

            return new OpeningErrorFollowUpdateResponse(
                    null, null, null, null, null, null, null, null, correlationId);
        });
    }

    /**
     * 外部MA呼出処理。
     *
     * @param kasJysb 開栓事由種別
     * @param tkourKbn 他小売区分
     * @param request リクエストDTO
     * @param kseflRecord 開栓エラーフォロー情報
     * @param jusyomRecord 住所マスター情報
     */
    private void callOpeningOperation(
            String kasJysb,
            String tkourKbn,
            OpeningErrorFollowUpdateRequest request,
            Optional<KseflRecord> kseflRecord,
            JusyomRecord jusyomRecord) {
        if (alternativeOpening.equals(kasJysb) || newlyOpenedTap.equals(kasJysb)) {
            if (ohterRetail.equals(tkourKbn)) {
                // MA-054
                openingCommonNormalOtherRetailMaAdapter.invoke(
                        createOpeningCommonNormalOtherRetailRequest(request, kseflRecord, jusyomRecord));
            } else {
                // MA-050
                openingCommonNormalTgRetailMaAdapter.invoke(
                        createOpeningCommonNormalTgRetailRequest(request, kseflRecord, jusyomRecord));
            }
        } else if (continuedUse.equals(kasJysb)) {
            // MA-051
            openingCommonContinueMaAdapter.invoke(
                    createOpeningCommonContinueRequest(request, kseflRecord, jusyomRecord));
        } else if (supplyResumed.equals(kasJysb)) {

    /**
     * MAAPI呼出例外の独自メソッド定義
     *
     * @param exception MAAPI呼出時の透過例外
     * @return MaPassthroughMaApiException MAAPI呼出時の透過例外
     */
    private MaPassthroughMaApiException handleError(MaPassthroughMaApiException exception) {
        String body = exception.getResponseBody();
        String correlationId = requestContext.getCorrelationId().orElseThrow(
                () -> new IllegalStateException("correlationId is not set in RequestContext"));
        String mediaType = exception.getMediaType();
        int status = exception.getStatus();
        if (body == null) {
            return exception;
        }

        if (mediaType == null || ! mediaType.contains("json")) {
            return exception;
        }

        ObjectMapper mapper = new ObjectMapper();
        mapper.configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

        try {
            MaPassthroughMaApiErrorResponse originErrDto =
                    mapper.readValue(body, MaPassthroughMaApiErrorResponse.class);
            boolean isWarning =
                    warningCodeList.stream().anyMatch(x -> x.equals(originErrDto.kasErryCd()));

            String crsMsgId =
                    Optional.ofNullable(originErrDto.ciriusMessageId())
                            .orElse(ErrorCatalog.resolveCiriusMessageId(originErrDto.errorCode()));

            List<String> crsEmbeddedStringList =
                    generalSettingsNotCompletedError.equals(originErrDto.kasErryCd())
                            ? originErrDto.ciriusEmbeddedStringList()
                                    .stream()
                                    .map(this::formatNumber).toList()
                            : originErrDto.ciriusEmbeddedStringList();

            OpeningErrorFollowUpdateResponse entityDto =
                    isWarning
                            ? new OpeningErrorFollowUpdateResponse(
                                    originErrDto.errorCode(),
                                    originErrDto.errorDescription(),
                                    originErrDto.errorDetailList(),
                                    null,
                                    null,
                                    originErrDto.kasErryCd(),
                                    crsMsgId,
                                    crsEmbeddedStringList,
                                    correlationId)
                            : new OpeningErrorFollowUpdateResponse(
                                    originErrDto.errorCode(),
                                    originErrDto.errorDescription(),
                                    originErrDto.errorDetailList(),
                                    crsMsgId,
                                    crsEmbeddedStringList,
                                    originErrDto.kasErryCd(),
                                    null,
                                    null,
                                    correlationId);

            String entityStr = mapper.writeValueAsString(entityDto);

            return isWarning
                    ? new MaPassthroughMaApiException(
                            Response.Status.OK.getStatusCode(),
                            entityStr,
                            mediaType,
                            null,
                            null)
                    : new MaPassthroughMaApiException(
                            status,
                            entityStr,
                            mediaType,
                            null,
                            null);
        } catch (IOException e) {
            return exception;
        }
    }

    /**
     * 1xのハイフン編集(xxxx-xxx-xxxx)
     *
     * @param input 修正対象文字列
     * @return String 修正後文字列
     */
    private String formatNumber(String input) {
        if (input == null || input.length() != 11) {
            return input;
        }

        return input.substring(0, 4) + "-" +
               input.substring(4, 7) + "-" +
               input.substring(7, 11);
    }
