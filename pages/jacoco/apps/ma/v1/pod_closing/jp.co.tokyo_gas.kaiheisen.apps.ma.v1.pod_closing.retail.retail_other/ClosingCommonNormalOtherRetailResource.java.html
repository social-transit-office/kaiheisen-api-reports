    private BigDecimal resolveUsageAmountCurrentUsage(
            ClosingCommonNormalOtherRetailRequest request,

    }

    private Long resolveUsageAmountJudgmentThreshold(
            PreClosingInfoAcquisitionOutput preClosingInfo) {
        if (preClosingInfo.gasmtrdInf() == null
                || preClosingInfo.gasmtrdInf().sjskHticSu() == null
                || preClosingInfo.gasmtrdInf().sjsuMincSu() == null) {

                        .gasmtrdInf()
                        .sjsuMincSu()
                        .multiply(BigDecimal.valueOf(preClosingInfo.gasmtrdInf().sjskHticSu())));
    }

    private Long truncateToInteger12Digits(BigDecimal value) {
        BigDecimal integerValue = value.setScale(0, RoundingMode.DOWN);
        int overflowDigits = integerValue.precision() - 12;
        if (overflowDigits > 0) {
            integerValue =
                    integerValue.movePointLeft(overflowDigits).setScale(0, RoundingMode.DOWN);
        }
        return integerValue.longValueExact();
    }
