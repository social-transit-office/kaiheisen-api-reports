/**
 * MA-069 閉栓共通処理 強制入力 他小売 API のリソース実装。
 */
@Path(ClosingCommonForceOtherRetailEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-069", name = "閉栓共通処理 強制入力 他小売 API")
@Tag(name = "閉栓共通処理 強制入力 他小売", description = "MA-069 閉栓共通処理 強制入力 他小売 API")
public class ClosingCommonForceOtherRetailResource
        extends BaseResource<ClosingCommonForceOtherRetailRequest> {

    private static final char HALF_WIDTH_SPACE = ' ';

    private static final char FULL_WIDTH_SPACE = '　';

    /** サービスユーザー ID。 */
    private static final String SERVICE_USER_ID = "KHSM069";

    /** フォロー担当地域統括コードの空白値。 */
    private static final String DEFAULT_FOLLOW_CKTK_CD = repeat(HALF_WIDTH_SPACE, 3);

    /** フォロー担当拠点コードの空白値。 */
    private static final String DEFAULT_FOLLOW_KTN_CD = repeat(HALF_WIDTH_SPACE, 1);

    /** お客さま名カナ1の空白値。 */
    private static final String DEFAULT_CUSTOMER_NAME_KANA = repeat(HALF_WIDTH_SPACE, 30);

    /** お客さま名漢字1の空白値。 */
    private static final String DEFAULT_CUSTOMER_NAME_KANJI = repeat(FULL_WIDTH_SPACE, 15);

    /** お客さま名漢字表示可否フラグの空白値。 */
    private static final String DEFAULT_CUSTOMER_NAME_DISPLAY_FLAG = repeat(HALF_WIDTH_SPACE, 1);

    /** お客さま名1法人種別の空白値。 */
    private static final String DEFAULT_CUSTOMER_CORPORATE_TYPE = repeat(HALF_WIDTH_SPACE, 2);

    /** お客さま名1法人種別表示位置種別の空白値。 */
    private static final String DEFAULT_CUSTOMER_CORPORATE_POSITION_TYPE = repeat(HALF_WIDTH_SPACE, 1);

    /** 閉栓エラーフォロー DataAccess。 */
    private final HseflDataAccess hseflDataAccess;

    /** 契約お客さま DataAccess。 */
    private final KiycstdDataAccess kiycstdDataAccess;

    /** 区分別エリアマスタ DataAccess。 */
    private final KbagmstDataAccess kbagmstDataAccess;

    /** フォロー状況コード。 */
    private final String followStatusCode;

    /** 開閉栓定数サービス。 */
    private final KaiheisenConstantService kaiheisenConstantService;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator リクエストバリデータ
     * @param clock システム日時プロバイダ
     * @param hseflDataAccess HSEFL DataAccess
     * @param kiycstdDataAccess KIYCSTD DataAccess
     * @param kbagmstDataAccess KBAGMST DataAccess
     * @param ciriusConstantService CIRIUS 定数サービス
     * @param kaiheisenConstantService 開閉栓定数サービス
     */
    @Inject
    public ClosingCommonForceOtherRetailResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            HseflDataAccess hseflDataAccess,
            KiycstdDataAccess kiycstdDataAccess,
            KbagmstDataAccess kbagmstDataAccess,
            CiriusConstantService ciriusConstantService,
                KaiheisenConstantService kaiheisenConstantService) {
        super(requestContext, requestDtoValidator, clock);
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
        this.kiycstdDataAccess = Objects.requireNonNull(kiycstdDataAccess, "kiycstdDataAccess");
        this.kbagmstDataAccess = Objects.requireNonNull(kbagmstDataAccess, "kbagmstDataAccess");
        this.followStatusCode = RequiredConstantResolver.requireCiriusConstant(
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService"),
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);
        this.kaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
    }

    /**
     * MA-069 を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "閉栓共通処理-強制入力-小売-他小売",
            description = "閉栓エラーフォロー件名へ強制処理データを登録する。",
            operationId = "postClosingCommonForceOtherRetail")
    @RequestBody(required = true, content = @Content(
            mediaType = MediaType.APPLICATION_JSON,
            schema = @Schema(implementation = ClosingCommonForceOtherRetailRequest.class)))
    @APIResponse(
            responseCode = "200",
            description = "強制処理データの登録結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = ClosingCommonForceOtherRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public ClosingCommonForceOtherRetailResponse endpoint(ClosingCommonForceOtherRetailRequest request) {
        return execute(request, () -> handle(request));
    }

    ClosingCommonForceOtherRetailResponse handle(ClosingCommonForceOtherRetailRequest request) {
        LocalDate hsenHkkYmd = Rfc3339LocalDateParser.parse(request.hsenHkkYmd().trim());
        LocalDate hssgYmd = Rfc3339LocalDateParser.parse(request.hssgYmd().trim());
        String decodedOriginFunctionId = decodeOriginFunctionId();
        Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
        Long syoKyoTrno = Long.valueOf(request.syoKyoTrno().trim());
        Integer nextRegTsno = resolveNextRegTsno(gasMtstNo, hssgYmd, hsenHkkYmd);
        KiycstdRecord customer = findKiycstd(syoKyoTrno);
        KbagmstRecord area = findKbagmst(gasMtstNo, hssgYmd);

        HseflWriteRecord writeRecord = new HseflWriteRecord(
                gasMtstNo,
                hssgYmd,
                hsenHkkYmd,
                nextRegTsno,
                requestContext.getHukkSysShbt().orElse(null),
                request.sgyAnknNo(),
                followStatusCode,
                null,
                syoKyoTrno,
                request.gmb(),
                request.hesnJysb(),
                request.hsenErrRyCd(),
                resolveFollowAreaChiefCode(area),
                resolveFollowAreaBranchCode(area),
                request.sgytNwkCd(),
                request.sagyTntEmpId(),
                request.sagyTntEmpNmKj(),
                Long.valueOf(request.gmtShjSu().trim()),
                new BigDecimal(request.gasSyry().trim()),
                request.sjskKbn(),
                null,
                null,
                null,
                request.utidxNo(),
                customer != null ? customer.ctnmKn1() : DEFAULT_CUSTOMER_NAME_KANA,
                customer != null ? customer.ctnmKj1() : DEFAULT_CUSTOMER_NAME_KANJI,
                customer != null
                    ? toFlagOrDefault(customer.ctnmKjhkFlg())
                    : DEFAULT_CUSTOMER_NAME_DISPLAY_FLAG,
                customer != null ? customer.ctnm1HujnShbt() : DEFAULT_CUSTOMER_CORPORATE_TYPE,
                customer != null ? customer.ctnm1HshiShbt() : DEFAULT_CUSTOMER_CORPORATE_POSITION_TYPE,
                null,
                null,
                null,
                null,
                null);

        hseflDataAccess.insert(
                writeRecord,
                AuditContext.of(SERVICE_USER_ID, decodedOriginFunctionId));

        String correlationId = requestContext
                .getCorrelationId()
                .orElseThrow(() -> new IllegalStateException("RequestContext.correlationId is required"));
        return new ClosingCommonForceOtherRetailResponse(correlationId);
    }

    private Integer resolveNextRegTsno(Long gasMtstNo, LocalDate hssgYmd, LocalDate hsenHkkYmd) {
        return hseflDataAccess.findAllByGasMtstNoAndHssgYmdAndHesnHukkYmd(gasMtstNo, hssgYmd, hsenHkkYmd)
                .stream()
                .map(HseflRecord::regTsno)
                .filter(Objects::nonNull)
                .max(Comparator.naturalOrder())
                .map(value -> value + 1)
                .orElse(1);
    }

    private KiycstdRecord findKiycstd(Long syoKyoTrno) {
        return kiycstdDataAccess.findByKyoTrkNo(syoKyoTrno).orElse(null);
    }

    private KbagmstRecord findKbagmst(Long gasMtstNo, LocalDate hssgYmd) {
        YearMonth knskKjnYm = YearMonth.of(hssgYmd.getYear(), hssgYmd.getMonthValue());
        return kbagmstDataAccess.findByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, knskKjnYm)
                .orElse(null);
    }

    private String toFlag(Boolean value) {
        if (Boolean.TRUE.equals(value)) {
            return "1";
        }
        if (Boolean.FALSE.equals(value)) {
            return "0";
        }
        return null;
    }

    private String toFlagOrDefault(Boolean value) {
        return Objects.requireNonNullElse(toFlag(value), DEFAULT_CUSTOMER_NAME_DISPLAY_FLAG);
    }

    private String resolveFollowAreaChiefCode(KbagmstRecord area) {
        if (area == null) {
            return DEFAULT_FOLLOW_CKTK_CD;
        }
        return Objects.requireNonNullElse(area.kbabCktkCd(), DEFAULT_FOLLOW_CKTK_CD);
    }

    private String resolveFollowAreaBranchCode(KbagmstRecord area) {
        if (area == null) {
            return DEFAULT_FOLLOW_KTN_CD;
        }
        return Objects.requireNonNullElse(area.kbabKtnCd(), DEFAULT_FOLLOW_KTN_CD);
    }

    private static String repeat(char character, int count) {
        return String.valueOf(character).repeat(count);
    }

    private String decodeOriginFunctionId() {
        String hukkSysShbt = requestContext.getHukkSysShbt()
                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String reportingSystemType = ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE;
        try {
            return kaiheisenConstantService.findNameByTypeAndValue(reportingSystemType, hukkSysShbt);
        } catch (ConstantDefinitionNotFoundException ex) {
            throw new KaiheisenException("報告システム種別の定数定義が見つかりません", ex);
        }
    }
}
