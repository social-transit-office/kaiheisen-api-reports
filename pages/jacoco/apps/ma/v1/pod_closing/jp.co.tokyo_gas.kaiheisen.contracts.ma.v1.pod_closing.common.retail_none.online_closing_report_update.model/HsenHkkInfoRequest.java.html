package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_closing.common.retail_none.online_closing_report_update.model;

import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.NumericDecimalMaxDigits;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateText;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * 閉栓報告情報DTO。

 * @param hesnJysb 閉栓事由種別
 * @param hesnYmd 閉栓年月日

 * @param tkurYteKknFlg 他小売閉栓予定日確認フラグ

@Schema(description = "閉栓報告情報")
public record HsenHkkInfoRequest(
        @Schema(description = "閉栓事由種別")
        String hesnJysb,
        @Rfc3339DateText
        @Schema(description = "閉栓年月日")
        String hesnYmd,
        @Schema(description = "ガスメーター指示数")
        String gmtShjSu,
        @NumericDecimalMaxDigits(integerDigits = 11, fractionDigits = 4)

        String gasSyry,
        @Schema(description = "確認フラグ")
        String kknnFlg,
        @Schema(description = "指示数回帰区分")

        String rshhSjsb,
        @Schema(description = "履歴メモ")
        String rirekiMemo,
        @Schema(description = "他小売閉栓予定日確認フラグ")
        String tkurYteKknFlg
