 * MA-020 オンライン閉栓報告情報取得-導管-共通 のリソース実装。

@Path(OnlineClosingReportInfoGetSupplyEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "オンライン閉栓報告情報取得", description = "MA-020:オンライン閉栓報告情報取得-導管-共通")
public class OnlineClosingReportInfoGetSupplyResource extends BaseResource<OnlineClosingReportInfoGetSupplyRequest> {

    private final OpeningClosingInProgressErrorJudgment openingClosingInProgressErrorJudgment;

    private final MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing;

    private GhpMeterInstallationCodeEdit ghpMeterInstallationCodeEdit;

    private final GmtsetdDataAccess gmtsetdDataAccess;

    private final GasmtrdDataAccess gasmtrdDataAccess;

    private final HseflDataAccess hseflDataAccess;

    /**

     * @param clock システム時刻プロバイダ
     * @param gmtsetdDataAccess ガスメーター設置場所＜導管＞データアクセス
     * @param gasmtrdDataAccess ガスメーター＜導管＞データアクセス
     * @param hseflDataAccess 閉栓エラーフォロー件名データアクセス
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     * @param kaiheisenConstantService 開閉栓API独自コンスタント取得サービス
     *
     */
    @Inject
    public OnlineClosingReportInfoGetSupplyResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @OnlineClosingReportInfoGetSupplyConfiguration
            OpeningClosingInProgressErrorJudgment openingClosingInProgressErrorJudgment,
            @OnlineClosingReportInfoGetSupplyConfiguration
            MeterReadingResultAcquisitionForClosing meterReadingResultAcquisitionForClosing,
            @OnlineClosingReportInfoGetSupplyConfiguration
            GhpMeterInstallationCodeEdit ghpMeterInstallationCodeEdit,
            GmtsetdDataAccess gmtsetdDataAccess,
            GasmtrdDataAccess gasmtrdDataAccess,
            HseflDataAccess hseflDataAccess,
            CiriusConstantService ciriusConstantService,
            KaiheisenConstantService kaiheisenConstantService) {
        super(requestContext, requestDtoValidator, clock);

        CiriusConstantService resolvedCiriusConstantService = Objects.requireNonNull(
                ciriusConstantService,
                "ciriusConstantService");

        KaiheisenConstantService resolvedKaiheisenConstantService = Objects.requireNonNull(
                kaiheisenConstantService,
                "kaiheisenConstantService");

        this.followStatusCodeN2NotYetStarted = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.TYPE,
                ConstantLookupKeys.Cirius.FollowStatusCode__N2.NOT_YET_STARTED);

        this.openingClosingProcessingClosingProcess = RequiredConstantResolver.requireKaiheisenConstant(
                resolvedKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.TYPE,
                ConstantLookupKeys.Kaiheisen.OpeningClosingProcessing.CLOSING_PROCESS);

        this.openingClosingInProgressErrorJudgment = Objects.requireNonNull(
                openingClosingInProgressErrorJudgment,
                "openingClosingInProgressErrorJudgment");

        this.meterReadingResultAcquisitionForClosing = Objects.requireNonNull(
                meterReadingResultAcquisitionForClosing,
                "meterReadingResultAcquisitionForClosing");

        this.ghpMeterInstallationCodeEdit = Objects.requireNonNull(
                ghpMeterInstallationCodeEdit,
                "ghpMeterInstallationCodeEdit");

        this.gmtsetdDataAccess = Objects.requireNonNull(gmtsetdDataAccess, "gmtsetdDataAccess");
        this.gasmtrdDataAccess = Objects.requireNonNull(gasmtrdDataAccess, "gasmtrdDataAccess");
        this.hseflDataAccess = Objects.requireNonNull(hseflDataAccess, "hseflDataAccess");
    }

    /**
     *  MA-020 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "オンライン閉栓報告情報取得",
            description = "オンライン閉栓報告情報取得を取得する",
            operationId = "OnlineClosingReportInfoGetSupply")
    @APIResponse(
            responseCode = "200",
            description = "オンライン閉栓報告情報取得取得結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OnlineClosingReportInfoGetSupplyResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OnlineClosingReportInfoGetSupplyResponse endpoint(@BeanParam OnlineClosingReportInfoGetSupplyRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();

            //DA-033-R006取得処理
            GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, knskKjnYmd);

            try {
                //SV-138処理
                openingClosingInProgressErrorJudgment.execute(
                        gmtsetdRecord.gmtKhsJtkb(),
                        openingClosingProcessingClosingProcess);
            } catch (HeisenRuleException exception) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000073);
            }

            //DA-110-R007取得処理
            GasmtrdRecord gasmtrdRecord = requireGasmtrd(gasMtstNo, knskKjnYmd);

            //SV-162処理
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput =
                meterReadingResultAcquisitionForClosing.execute(
                    gasMtstNo);

            //DA-141-R003処理
            HseflRecord hseflRecord = requireHsefl(gasMtstNo);

            //SV-010処理
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput =
                ghpMeterInstallationCodeEdit.execute(
                    gmtsetdRecord.ckrtGmscJtkb(),
                    gmtsetdRecord.kkyuGmtKbn());

            return toResponse(
                gmtsetdRecord,
                gasmtrdRecord,
                hseflRecord,
                openingClosingInProgressErrorJudgment,
                meterReadingResultAcquisitionForClosingOutput,
                ghpMeterInstallationCodeEditOutput);
        });
    }

    /**
     * レスポンスを成形する
     *
     * @param gmtsetdRecord ガスメーター設置場所＜導管＞取得結果
     * @param gasmtrdRecord ガスメーター＜導管＞取得結果
     * @param hseflRecord 閉栓エラーフォロー件名取得結果
     * @param openingClosingInProgressErrorJudgmentOutput SV-138取得結果
     * @param meterReadingResultAcquisitionForClosingOutput SV-162取得結果
     * @param ghpMeterInstallationCodeEditOutput SV-010取得結果
     * @return　OnlineClosingReportInfoGetRetailResponse レスポンス
     */
    private OnlineClosingReportInfoGetSupplyResponse toResponse(
            GmtsetdRecord gmtsetdRecord,
            GasmtrdRecord gasmtrdRecord,
            HseflRecord hseflRecord,
            OpeningClosingInProgressErrorJudgment openingClosingInProgressErrorJudgment,
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput,
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput
    ) {
        return new OnlineClosingReportInfoGetSupplyResponse(
            createGmtsetdInfoResponse(gmtsetdRecord, ghpMeterInstallationCodeEditOutput),
            createGmtInfoResponse(gasmtrdRecord),
            createKnsKkaInfoResponse(meterReadingResultAcquisitionForClosingOutput),
            createHseflTishFlgResponse(hseflRecord),
            requestContext.getCorrelationId().orElseThrow(() -> new IllegalStateException("correlationId is not set in RequestContext"))
            );
    }

    /**
     * DA-038-R006取得処理
     * 主な仕様:
     * - サービス契約番号、検索基準年月日、サービス契約状態区分を引数にサービス契約の検索を行う
     * @param svkyNo サービス契約番号
     * @param knskKjnYmd 検索基準年月日
     * @return SvckyaRecord
     */
    private GmtsetdRecord requireGmtsetd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GmtsetdRecord> gmtsetdRecords =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd);
        if (!gmtsetdRecords.isEmpty()) {
            return gmtsetdRecords.get();
        } else {
            throw businessRuleException(ErrorCodeConstants.ERROR_CODE_020000030);
        }
    }

    /**
     * DA-110-R007取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター＜導管＞の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return GasmtrdRecord
     */
    private GasmtrdRecord requireGasmtrd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GasmtrdRecord> gasmtrdRecords =
                gasmtrdDataAccess.findByTtkGmsbNoAndGmtkYmd(gasMtstNo, knskKjnYmd);
        if (gasmtrdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    gasmtrdDataAccess.tableName(),
                    GasmtrdDataAccess.CallSpecs.FIND_BY_TTK_GMSB_NO_AND_GMTK_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("ttkGmsbNo", gasMtstNo)
                            .put("gmtkYmd", knskKjnYmd));
        }
        return gasmtrdRecords.get();
    }

    /**
     * DA-141-R003取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、フォローステータスコードを引数に閉栓エラーフォロー件名の検索を行う
     * - 取得件数が0件の場合、nullを返却する
     * @param gasMtstNo ガスメーター設置場所番号
     * @return HseflRecord
     */
    private HseflRecord requireHsefl(Long gasMtstNo) {
        List<HseflRecord> hseflrecords =
                hseflDataAccess.findAllByGasMtstNoAndFlwStCd(gasMtstNo, followStatusCodeN2NotYetStarted);

        if (hseflrecords.isEmpty()) {
            return null;
        }
        return hseflrecords.getFirst();
    }

    /**
      * ガスメーター設置場所＜導管＞DTOを成形する
      *
      * @param gmtsetdRecord ガスメーター設置場所＜導管＞取得結果
      * @param ghpMeterInstallationCodeEditOutput SV-010取得結果
      * @return GmtsetdInfoResponse レスポンス
      */
    private GmtsetdInfoResponse createGmtsetdInfoResponse(
            GmtsetdRecord gmtsetdRecord,
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput) {
        return new GmtsetdInfoResponse(
            gmtsetdRecord.gasMtstNo(),
            ghpMeterInstallationCodeEditOutput.ghpmSchCd(),
            gmtsetdRecord.gmtKhsJtkb(),
            gmtsetdRecord.kngmStbJtkb(),
            formatDate(gmtsetdRecord.kngmSbmsYmd()),
            formatDate(gmtsetdRecord.kasYmd()),
            formatDate(gmtsetdRecord.hesnYmd()),
            formatDate(gmtsetdRecord.kusYmd()),
            gmtsetdRecord.hesnJysb(),
            gmtsetdRecord.kasJysb());
    }

    /**
     * ガスメーター情報DTOを成形する
     *
     * @param gasmtrdRecord ガスメーター＜導管＞取得結果
     * @return GmtInfoResponse レスポンス
     */
    private GmtInfoResponse createGmtInfoResponse(GasmtrdRecord gasmtrdRecord) {
        return new GmtInfoResponse(
            gasmtrdRecord.gmb(),
            gasmtrdRecord.gmkShbt(),
            gasmtrdRecord.gmtGusu(),
            gasmtrdRecord.sjsuMincSu());
    }

    /**
     * 検針結果情報DTOを成形する
     *
     * @param meterReadingResultAcquisitionForClosingOutput SV-162取得結果
     * @return KnsKkaInfoResponse レスポンス
     */
    private KnsKkaInfoResponse createKnsKkaInfoResponse(
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput) {
        return new KnsKkaInfoResponse(
            meterReadingResultAcquisitionForClosingOutput.skinKnkkInf().gmtShjSu(),
            formatDate(meterReadingResultAcquisitionForClosingOutput.znkiKnkkInf().knsYmd()));
    }

    /**
     * 閉栓エラーフォロー件名対象フラグレスポンスを判定する
     *
     * @param hseflRecord 閉栓エラーフォロー件名取得結果
     * @return HseflTishFlgResponse レスポンス
     */
    private Boolean createHseflTishFlgResponse(HseflRecord hseflRecord) {
        if (hseflRecord != null) {
            return true;
        } else {
            return false;
        }
    }

    /**
     * 日付を ISO-8601 文字列へ変換する。
     *
     * @param value 変換対象日付
     * @return 日付文字列。{@code value} が {@code null} の場合は {@code null}
     */
    private String formatDate(LocalDate value) {
        return Objects.toString(value, null);
    }

    /**
     * BusinessRuleExceptionを定義する
     *
     * @param errorCode
     * @return
     */
    private static BusinessRuleException businessRuleException(String errorCode) {
        return new BusinessRuleException(errorCode);
    }
