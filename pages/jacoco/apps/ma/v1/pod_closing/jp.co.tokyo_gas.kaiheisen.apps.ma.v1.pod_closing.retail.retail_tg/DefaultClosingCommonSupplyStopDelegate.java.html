package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_tg;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.util.Objects;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_closing.retail.retail_tg.ClosingCommonSupplyStopRequest;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_closing.retail.retail_tg.ClosingCommonSupplyStopResponse;

/**
 * 本番用の MA-067 呼び出し委譲実装。
 */
@ApplicationScoped
public class DefaultClosingCommonSupplyStopDelegate implements ClosingCommonSupplyStopDelegate {

    private final ClosingCommonSupplyStopResource closingCommonSupplyStopResource;

    /**
     * 子 resource を受け取る。
     *
     * @param closingCommonSupplyStopResource MA-067 resource
     */
    @Inject
    public DefaultClosingCommonSupplyStopDelegate(
            ClosingCommonSupplyStopResource closingCommonSupplyStopResource) {
        this.closingCommonSupplyStopResource =
                Objects.requireNonNull(closingCommonSupplyStopResource, "closingCommonSupplyStopResource");
    }

    @Override
    public ClosingCommonSupplyStopResponse execute(ClosingCommonSupplyStopRequest request) {
        return closingCommonSupplyStopResource.delegateFromParent(request);
    }
}
