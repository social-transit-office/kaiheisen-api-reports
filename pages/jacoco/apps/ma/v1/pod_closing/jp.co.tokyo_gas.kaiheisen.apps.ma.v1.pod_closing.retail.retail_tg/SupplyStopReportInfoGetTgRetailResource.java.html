    /**
     * 概要: MA-023 の実処理を順序どおり実行する。
     * 主な仕様:
     * - TSV step の順序どおりに DA / SV を実行し、取得結果をレスポンスへ整形する。
     * - 0 件時の業務エラー、データなし、データ不整合エラーは設計どおりに送出する。
     * 既知の制約:
     * - systemDatetime は BaseResource 経由で RequestContext に設定済みである前提とする。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    SupplyStopReportInfoGetTgRetailResponse handle(SupplyStopReportInfoGetTgRetailRequest request) {
        Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
        LocalDate systemDate =
                requestContext.getSystemDatetime()
                        .orElseThrow(() -> new IllegalStateException("systemDatetime is not set"))
                        .toLocalDate();
        YearMonth systemYearMonth = YearMonth.of(systemDate.getYear(), systemDate.getMonthValue());

        GmtsetdRecord gmtsetdRecord =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemDate)
                        .orElseThrow(
                                () -> new BusinessRuleException(
                                        ErrorCodeConstants.ERROR_CODE_020000012));

        executeOpeningClosingInProgressErrorJudgment(gmtsetdRecord);

        ClosingTargetGasMeterRetailCategoryJudgmentOutput retailCategory =
                closingTargetGasMeterRetailCategoryJudgment.execute(
                        gasMtstNo, RETAIL_STOCK_DIVISION);

        ServiceContractNumberAcquisitionOutput serviceContract =
                serviceContractNumberAcquisition.execute(
                        gasMtstNo,
                        gmtsetdRecord.mtgMtstNo(),
                        SERVICE_CONTRACT_ACQUISITION_DIVISION,
                        gmtsetdRecord.ckrtGmscJtkb(),
                        systemDate);

        ClosingTargetGasMeterCompanyUseCategoryJudgmentOutput companyUseCategory =
                closingTargetGasMeterCompanyUseCategoryJudgment.execute(serviceContract.svkyNo());

        MeterReadingResultAcquisitionForClosingOutput meterReadingResult =
                meterReadingResultAcquisitionForClosing.execute(gasMtstNo);

        GasmtrdRecord gasmtrdRecord =
                gasmtrdDataAccess
                        .findByTtkGmsbNoAndGmtkYmd(gasMtstNo, systemDate)
                        .orElseThrow(() -> new DataNotFoundException(gasmtrdDataAccess.tableName()));

        KnsytiRecord knsytiRecord =
                knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, systemYearMonth)
                        .stream()
                        .findFirst()
                        .orElseThrow(() -> new DataNotFoundException(knsytiDataAccess.tableName()));

        ComprehensiveJudgmentProcessOutput comprehensiveJudgment =
                comprehensiveJudgmentProcess.execute(
                        gasMtstNo, null, comprehensiveJudgmentProcessingDivision);

        SvsihkrRecord svsihkrRecord =
                svsihkrDataAccess
                        .findBySvkyNoAndTkysYmdAndTyenYmdAndNotSihKiyJtkb(
                                serviceContract.svkyNo(),
                                systemDate,
                                invalidPaymentContractStatus)
                        .orElseThrow(
                                () -> createSvsihkrDataInconsistency(
                                        serviceContract.svkyNo(), systemDate));

        SihakyRecord sihakyRecord =
                sihakyDataAccess
                        .findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDesc(
                                svsihkrRecord.sihKiyNo(),
                                systemDate,
                                invalidPaymentContractStatus)
                        .orElseThrow(
                                () -> createSihakyDataInconsistency(
                                        svsihkrRecord.sihKiyNo(), systemDate));

        GhpMeterInstallationCodeEditOutput ghpCode =
                ghpMeterInstallationCodeEdit.execute(
                        gmtsetdRecord.ckrtGmscJtkb(), gmtsetdRecord.kkyuGmtKbn());

        return toResponse(
                gmtsetdRecord,
                gasmtrdRecord,
                knsytiRecord,
                sihakyRecord,
                retailCategory,
                serviceContract,
                companyUseCategory,
                meterReadingResult,
                comprehensiveJudgment,
                ghpCode);
    }

                    gmtsetdRecord.gmtKhsJtkb(), closingInProgressJudgmentDivision);
        } catch (HeisenRuleException exception) {
            String errorCode = ErrorCodeConstants.ERROR_CODE_020000073;
            SupplyStopReportInfoGetTgRetailResponse.ErrorResponse body =
                    new SupplyStopReportInfoGetTgRetailResponse.ErrorResponse(
                            requestContext.getCorrelationId().orElseThrow(
                                    () -> new IllegalStateException("correlationId is not set")),
                            errorCode,
                            ErrorCatalog.resolveErrorMessage(errorCode, List.of()),
                            List.of(),
                            ErrorCatalog.resolveCiriusMessageId(errorCode),
                            List.of());
            throw new WebApplicationException(exception, Response.status(Response.Status.BAD_REQUEST)
                    .type(MediaType.APPLICATION_JSON)
                    .entity(body)
                    .build());
        }
    }
