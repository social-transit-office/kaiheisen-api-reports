    SupplyStopReportUpdateTgRetailResponse handle(
            SupplyStopReportUpdateTgRetailRequest request) {
        Long paymentContractNo = resolvePaymentContractNo(request.svkyNo());
        Long previousPaymentContractNo = resolvePreviousPaymentContractNo(request, paymentContractNo);
        PaidDataConfirmationOutput paidDataConfirmationOutput =
                paidDataConfirmation.execute(paymentContractNo, previousPaymentContractNo);

        try {
            ClosingCommonSupplyStopResponse childResponse =
                    closingCommonSupplyStopDelegate.execute(
                            toChildRequest(request));
            return toResponse(childResponse, paidDataConfirmationOutput.nyukinZmFlg());
        } catch (WebApplicationException exception) {
            ClosingCommonSupplyStopResponse childResponse = asChildResponse(exception);
            if (childResponse == null) {
                throw exception;
            }
            if (HeisenRuleListException.ERROR_CODE.equals(childResponse.errorCode())) {
                return toResponse(childResponse, paidDataConfirmationOutput.nyukinZmFlg());
            }
            throw badRequestResponse(toResponse(childResponse, null));
        }
    }

    private Long resolvePaymentContractNo(Long serviceContractNo) {
        return svsihkrDataAccess
                .findAllBySvkyNoAndSihKiyJtkb(serviceContractNo, activePaymentContractStatus)
                .stream()
                .findFirst()
                .map(record -> record.sihKiyNo())
                .orElseThrow(() -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000017));
    }

    private Long resolvePreviousPaymentContractNo(
            SupplyStopReportUpdateTgRetailRequest request,
            Long paymentContractNo) {
        SupplyStopReportUpdateTgRetailRequest.KkysHkkInf reportInfo = request.kkysHkkInf();
        Long previousGasMeterLocationNumber = reportInfo == null ? null : reportInfo.mejsGmsbNo();
        if (previousGasMeterLocationNumber == null) {
            return null;
        }

        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set"))
                .toLocalDate();
        gmtsetdDataAccess.findByGasMtstNoAndTkysYmdForR003(previousGasMeterLocationNumber, systemDate)
                .orElseThrow(() -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000012));

        List<jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.ncis.itensak.ItensakRecord> transferRecords =
                itensakDataAccess.findAllByGmtSetNoAndItenSShrkyNoAndItenTksCd(
                        previousGasMeterLocationNumber,
                        paymentContractNo,
                        transferDestinationCancellationNoConcept);
        return transferRecords.isEmpty() ? null : transferRecords.get(0).itenZShrkyNo();
    }

    private boolean requiresPreviousAddressGasMeterLocationNumber(String supplySuspensionReasonType) {

    private String resolveCiriusMessageId(ClosingCommonSupplyStopResponse response) {
        if (response.errorCode() == null || response.errorCode().isBlank()) {
            return null;

    }

    private List<String> resolveCiriusEmbeddedStringList(ClosingCommonSupplyStopResponse response) {
        if (response.errorCode() == null || response.errorCode().isBlank()) {
            return null;
        }
        if (ErrorCodeConstants.ERROR_CODE_020200026.equals(response.errorCode())
                && response.errGmsbNo() != null) {
            return List.of(formatGasMeterLocationNumber(response.errGmsbNo()));
        }
        if (ErrorCodeConstants.ERROR_CODE_020200042.equals(response.errorCode())
                && response.snstSyoRyo() != null) {
            return List.of(formatUsageAmount(response.snstSyoRyo()));
        }
        return null;
    }

    private String formatGasMeterLocationNumber(Long gasMeterLocationNumber) {
        String zeroFilled = String.format("%011d", gasMeterLocationNumber);
        return zeroFilled.substring(0, 4)
                + "-"
                + zeroFilled.substring(4, 7)
                + "-"
                + zeroFilled.substring(7);
    }

    private String formatUsageAmount(Double calculatedUsageAmount) {
        return BigDecimal.valueOf(calculatedUsageAmount)
                .setScale(4, RoundingMode.HALF_UP)
                .toPlainString();
    }

    private ClosingCommonSupplyStopResponse asChildResponse(WebApplicationException exception) {
        Response response = exception.getResponse();
        if (response == null) {
            return null;
        }
        Object entity = response.getEntity();
        return entity instanceof ClosingCommonSupplyStopResponse childResponse
                ? childResponse
                : null;
    }

    private WebApplicationException badRequestResponse(

        return new WebApplicationException(
                Response.status(Response.Status.BAD_REQUEST)
                        .type(MediaType.APPLICATION_JSON_TYPE)
                        .entity(response)

    }
