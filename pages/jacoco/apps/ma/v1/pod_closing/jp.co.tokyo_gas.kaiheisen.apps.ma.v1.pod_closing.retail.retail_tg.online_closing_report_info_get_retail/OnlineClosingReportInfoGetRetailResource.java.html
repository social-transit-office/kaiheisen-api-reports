    @APIResponse(

            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OnlineClosingReportInfoGetRetailResponse endpoint(@BeanParam OnlineClosingReportInfoGetRetailRequest request) {
        return execute(request, () -> {
            Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
            LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                    .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                    .toLocalDate();

            //DA-033-R006取得処理
            GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, knskKjnYmd);

            try {
                //SV-138処理
                openingClosingInProgressErrorJudgment.execute(
                        gmtsetdRecord.gmtKhsJtkb(),
                        openingClosingProcessingClosingProcess);
            } catch (HeisenRuleException exception) {
                throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000073);
            }

            //SV-003処理
            ServiceContractNumberAcquisitionOutput serviceContractNumberAcquisitionOutput =
                serviceContractNumberAcquisition.execute(
                    gasMtstNo,
                    gmtsetdRecord.mtgMtstNo(),
                    ShrKbn.NEAREST,
                    gmtsetdRecord.ckrtGmscJtkb(),
                    knskKjnYmd);

            //DA-038-R006取得処理
            SvckyaRecord svckyaRecord = requireSvckya(serviceContractNumberAcquisitionOutput.svkyNo(), knskKjnYmd);

            //DA-110-R007取得処理
            GasmtrdRecord gasmtrdRecord = requireGasmtrd(gasMtstNo, knskKjnYmd);

            //SV-162処理
            MeterReadingResultAcquisitionForClosingOutput meterReadingResultAcquisitionForClosingOutput =
                meterReadingResultAcquisitionForClosing.execute(
                    gasMtstNo);

            //DA-141-R003処理
            HseflRecord hseflRecord = requireHsefl(gasMtstNo);

            //SV-049処理
            Optional<ComprehensiveJudgmentProcessOutput> comprehensiveJudgmentProcessOutput =
                    Optional.ofNullable(comprehensiveJudgmentProcess.execute(
                    gasMtstNo,
                    null,
                    comprehensiveJudgmentProcessingEnd));

            //DA-044-R007取得処理
            SvsihkrRecord svsihkrRecord = requireSvsihkr(serviceContractNumberAcquisitionOutput.svkyNo(), knskKjnYmd);

            //DA-075-R005取得処理
            SihakyRecord sihakyRecord = requireSihaky(svsihkrRecord.sihKiyNo(), knskKjnYmd);

            //SV-010処理
            GhpMeterInstallationCodeEditOutput ghpMeterInstallationCodeEditOutput =
                    ghpMeterInstallationCodeEdit.execute(
                    gmtsetdRecord.ckrtGmscJtkb(),
                    gmtsetdRecord.kkyuGmtKbn());

            return toResponse(
                gmtsetdRecord,
                svckyaRecord,
                gasmtrdRecord,
                hseflRecord,
                svsihkrRecord,
                sihakyRecord,
                openingClosingInProgressErrorJudgment,
                serviceContractNumberAcquisitionOutput,
                meterReadingResultAcquisitionForClosingOutput,
                comprehensiveJudgmentProcessOutput,
                ghpMeterInstallationCodeEditOutput);
        });
    }

    /**
     * DA-033-R006取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター設置場所番号＜導管＞の検索を行う
     * - 取得件数が0件の場合、業務エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return GmtsetdRecord
     */
    private GmtsetdRecord requireGmtsetd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GmtsetdRecord> gmtsetdRecords =
                gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd);
        if (!gmtsetdRecords.isEmpty()) {
            return gmtsetdRecords.get();
        } else {
            throw businessRuleException(ErrorCodeConstants.ERROR_CODE_020000030);
        }
    }

    /**
     * DA-038-R006取得処理
     * 主な仕様:
     * - サービス契約番号、検索基準年月日、サービス契約状態区分を引数にサービス契約の検索を行う
     * @param svkyNo サービス契約番号
     * @param knskKjnYmd 検索基準年月日
     * @return SvckyaRecord
     */
    private SvckyaRecord requireSvckya(Long svkyNo, LocalDate knskKjnYmd) {
        List<SvckyaRecord> svckyaRecords =
            svckyaDataAccess.findAllBySvkyNoAtKnskKjnYmdAndSvkyJtkbNot(svkyNo, knskKjnYmd, serviceContractStatusInvalid);

        return svckyaRecords.getFirst();
    }

    /**
     * DA-110-R007取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、検索基準年月日を引数にガスメーター＜導管＞の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return GasmtrdRecord
     */
    private GasmtrdRecord requireGasmtrd(Long gasMtstNo, LocalDate knskKjnYmd) {
        Optional<GasmtrdRecord> gasmtrdRecords =
                gasmtrdDataAccess.findByTtkGmsbNoAndGmtkYmd(gasMtstNo, knskKjnYmd);
        if (gasmtrdRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    gasmtrdDataAccess.tableName(),
                    GasmtrdDataAccess.CallSpecs.FIND_BY_TTK_GMSB_NO_AND_GMTK_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("ttkGmsbNo", gasMtstNo)
                            .put("gmtkYmd", knskKjnYmd));
        }
        return gasmtrdRecords.get();
    }

    /**
     * DA-141-R003取得処理
     * 主な仕様:
     * - ガスメーター設置場所番号、フォローステータスコードを引数に閉栓エラーフォロー件名の検索を行う
     * - 取得件数が0件の場合、nullを返却する
     * @param gasMtstNo ガスメーター設置場所番号
     * @return HseflRecord
     */

        List<HseflRecord> hseflrecords =
                hseflDataAccess.findAllByGasMtstNoAndFlwStCd(gasMtstNo, followStatusCodeN2NotYetStarted);
        if (hseflrecords.isEmpty()) {
            return null;
        }
        return hseflrecords.getFirst();
    }

    /**
     * DA-044-R007取得処理
     * 主な仕様:
     * - サービス契約番号、検索基準年月日、支払契約状態区分を引数にサービス契約・支払契約関連の検索を行う
     * - 取得件数が0件の場合、データ不整合エラーを送出する
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検索基準年月日
     * @return SvsihkrRecord
     */
    private SvsihkrRecord requireSvsihkr(Long svryNo, LocalDate knskKjnYmd) {
        Optional<SvsihkrRecord> svsihkrRecords =
                svsihkrDataAccess.findBySvkyNoAndTkysYmdAndTyenYmdAndNotSihKiyJtkb(svryNo, knskKjnYmd, paymentContractStatusInvalid);
        if (svsihkrRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svsihkrDataAccess.tableName(),
                    SvsihkrDataAccess.CallSpecs.FIND_BY_SVKY_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_NOT_SIH_KIY_JTKB.identifier(),
                    new FlatJsonBuilder()
                            .put("svryNo", svryNo)
                            .put("knskKjnYmd", knskKjnYmd)
                            .put("sihKiyJtkb", paymentContractStatusInvalid));
        }
        return svsihkrRecords.get();
    }

      * @param knskKjnYmd 検索基準年月日
      * @return SihakyRecord
      */
    private SihakyRecord requireSihaky(Long svryNo, LocalDate knskKjnYmd) {
        Optional<SihakyRecord> sihakyRecords =
                sihakyDataAccess.findBySihKiyNoAndKnskKjnYmdAndNotSihKiyJtkbOrderByTkysYmdDesc(
                        svryNo,
                        knskKjnYmd,
                        paymentContractStatusInvalid);
        return sihakyRecords.get();
    }
