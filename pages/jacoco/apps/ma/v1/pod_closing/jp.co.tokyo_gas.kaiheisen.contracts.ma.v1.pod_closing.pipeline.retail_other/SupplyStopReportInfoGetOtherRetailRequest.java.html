package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_closing.pipeline.retail_other;

import jakarta.ws.rs.QueryParam;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.request.BaseRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.GasMtstNoFormat;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.RequiredText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateTimeText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.TextMaxLength;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * MA-024 供給停止報告情報取得-導管-他小売 API のリクエスト。
 *
 * <p>YAML 正本は GET + body だが、Java 実装では規約に合わせて query で受ける。
 * 相関 ID は HTTP ヘッダで管理し、DTO には含めない。</p>
 *
 * @param gasMtstNo ガスメーター設置場所番号
 * @param hukkSysShbt 報告システム種別
 * @param systemDatetime システム日時
 */
public record SupplyStopReportInfoGetOtherRetailRequest(
        @QueryParam("gasMtstNo")
        @RequiredText
        @GasMtstNoFormat
        @Schema(description = "ガスメーター設置場所番号", required = true, pattern = "^1\\d{10}$")
        String gasMtstNo,
        @QueryParam("hukkSysShbt")
        @TextMaxLength(max = 1)
        @Schema(description = "報告システム種別", maxLength = 1)
        String hukkSysShbt,
        @QueryParam("systemDatetime")
        @Rfc3339DateTimeText
        @Schema(description = "システム日時 (RFC3339)", format = "date-time")
        String systemDatetime) implements BaseRequest {
}
