            GasmtrdDataAccess gasmtrdDataAccess,
            KnsytiDataAccess knsytiDataAccess,

    /**
     * MA-024 供給停止報告情報取得-導管-他小売 API を実行する。

    @APIResponse(

    public SupplyStopReportInfoGetOtherRetailResponse endpoint(

        return execute(request, () -> handle(request));
    }

    SupplyStopReportInfoGetOtherRetailResponse handle(
            SupplyStopReportInfoGetOtherRetailRequest request) {
        Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
        LocalDate systemDate =
                requestContext.getSystemDatetime()
                        .orElseThrow(
                                () -> new IllegalStateException(
                                        "systemDatetime is not set"))
                        .toLocalDate();
        YearMonth systemYearMonth = YearMonth.of(
                systemDate.getYear(),
                systemDate.getMonthValue());

        GmtsetdRecord gmtsetdRecord =
                gmtsetdDataAccess
                        .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemDate)
                        .orElseThrow(() -> new BusinessRuleException(
                                ErrorCodeConstants.ERROR_CODE_020000012));

        executeOpeningClosingInProgressErrorJudgment(gmtsetdRecord);

        ClosingTargetGasMeterRetailCategoryJudgmentOutput retailCategory =
                closingTargetGasMeterRetailCategoryJudgment.execute(
                        gasMtstNo,
                        pipelineStockDivision);

        MeterReadingResultAcquisitionForClosingOutput meterReadingResult =
                meterReadingResultAcquisitionForClosing.execute(gasMtstNo);

        GasmtrdRecord gasmtrdRecord =
                gasmtrdDataAccess
                        .findByTtkGmsbNoAndGmtkYmd(gasMtstNo, systemDate)
                        .orElseThrow(() -> new DataNotFoundException(
                                gasmtrdDataAccess.tableName()));

        KnsytiRecord knsytiRecord =
                knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(
                                gasMtstNo,
                                systemYearMonth)
                        .stream()
                        .findFirst()
                        .orElseThrow(() -> new DataNotFoundException(
                                knsytiDataAccess.tableName()));

        GhpMeterInstallationCodeEditOutput ghpCode =
                ghpMeterInstallationCodeEdit.execute(
                        gmtsetdRecord.ckrtGmscJtkb(),
                        gmtsetdRecord.kkyuGmtKbn());

        String correlationId =
                requestContext.getCorrelationId()
                        .orElseThrow(
                                () -> new IllegalStateException(
                                        "correlationId is not set"));

        return toResponse(
                gmtsetdRecord,
                gasmtrdRecord,
                knsytiRecord,
                retailCategory,
                meterReadingResult,
                ghpCode,
                correlationId);
    }

    private SupplyStopReportInfoGetOtherRetailResponse toResponse(
            GmtsetdRecord gmtsetdRecord,
            GasmtrdRecord gasmtrdRecord,
            KnsytiRecord knsytiRecord,
            ClosingTargetGasMeterRetailCategoryJudgmentOutput retailCategory,
            MeterReadingResultAcquisitionForClosingOutput meterReadingResult,
            GhpMeterInstallationCodeEditOutput ghpCode,
            String correlationId) {
        return new SupplyStopReportInfoGetOtherRetailResponse(
                new SupplyStopReportInfoGetOtherRetailResponse.GmtsetInfo(
                        gmtsetdRecord.gasMtstNo(),
                        gmtsetdRecord.gmtKhsJtkb(),
                        gmtsetdRecord.kngmStbJtkb(),
                        formatDate(gmtsetdRecord.kasYmd()),
                        formatDate(gmtsetdRecord.hesnYmd()),
                        formatDate(gmtsetdRecord.kusYmd()),
                        formatDate(gmtsetdRecord.kngmSbmsYmd()),
                        gmtsetdRecord.hesnJysb(),
                        gmtsetdRecord.kasJysb(),
                        ghpCode == null ? null : ghpCode.ghpmSchCd()),
                new SupplyStopReportInfoGetOtherRetailResponse.KnsytiInfo(
                        knsytiRecord.knsHuhuKbn()),
                new SupplyStopReportInfoGetOtherRetailResponse.GmtInfo(
                        gasmtrdRecord.gmb(),
                        gasmtrdRecord.gmkShbt(),
                        MaResponseNumericValues.oneDecimalOrZero(gasmtrdRecord.gmtGusu()),
                        MaResponseNumericValues.oneDecimalOrZero(gasmtrdRecord.sjsuMincSu())),
                new SupplyStopReportInfoGetOtherRetailResponse.KnsKkaInfo(
                        meterReadingResult == null || meterReadingResult.skinKnkkInf() == null
                                ? null
                                : meterReadingResult.skinKnkkInf().gmtShjSu(),
                        meterReadingResult == null || meterReadingResult.skinKnkkInf() == null
                                ? null
                                : formatDate(meterReadingResult.skinKnkkInf().knsYmd())),
                new SupplyStopReportInfoGetOtherRetailResponse.TkssvkyInfo(
                        retailCategory == null ? null : retailCategory.tksuKijsCd(),
                        retailCategory == null ? null : retailCategory.syoKyoTrno()),
                correlationId,
                null,
                null,
                null,
                null,
                null);
    }

    private static String formatDate(LocalDate date) {
        return date == null ? null : date.toString();
    }

    /**
     * 概要: SV-138 の閉栓共通エラーを MA-024 の 400 レスポンスへ変換する。
     * 主な仕様:
     * - 閉栓共通エラーは元のエラーコードにかかわらず 020000073 として返す。
     * - 相関 ID は共通フィルタが RequestContext に設定した値を返す。
     * 既知の制約:
     * - SV-138 以外で発生した例外は本処理の変換対象外とする。
     *
     * @param gmtsetdRecord ガスメーター設置場所レコード
     */
    private void executeOpeningClosingInProgressErrorJudgment(GmtsetdRecord gmtsetdRecord) {
        try {
            openingClosingInProgressErrorJudgment.execute(
                    gmtsetdRecord.gmtKhsJtkb(), closingInProgressJudgmentDivision);
        } catch (HeisenRuleException exception) {
            String errorCode = ErrorCodeConstants.ERROR_CODE_020000073;
            SupplyStopReportInfoGetOtherRetailResponse.ErrorResponse body =
                    new SupplyStopReportInfoGetOtherRetailResponse.ErrorResponse(
                            requestContext.getCorrelationId().orElseThrow(
                                    () -> new IllegalStateException("correlationId is not set")),
                            errorCode,
                            ErrorCatalog.resolveErrorMessage(errorCode, List.of()),
                            List.of(),
                            ErrorCatalog.resolveCiriusMessageId(errorCode),
                            List.of());
            throw new WebApplicationException(exception, Response.status(Response.Status.BAD_REQUEST)
                    .type(MediaType.APPLICATION_JSON)
                    .entity(body)
                    .build());
        }
    }
