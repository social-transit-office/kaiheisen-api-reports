/**
 * MA-026 供給停止報告更新-導管-他小売 API のリソース実装。
 */
@MainService(identifier = "MA-026", name = "供給停止報告更新-導管-他小売 API")
@Path(SupplyStopReportUpdateOtherRetailEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "供給停止報告更新-導管-他小売", description = "MA-026 供給停止報告更新-導管-他小売 API")
public class SupplyStopReportUpdateOtherRetailResource
        extends BaseResource<SupplyStopReportUpdateOtherRetailRequest> {

    private static final String MDC_CORRELATION_ID_KEY = "correlationId";

    private final ClosingCommonSafeOtherRetailResource closingCommonSafeOtherRetailResource;

    private final String childBusinessCode;

    private final String childBusinessDetailCode;

    private final String childReportRouteDivision;

    private final String childClosingReason;

    /**
     * MA-026 用の依存コンポーネントを受け取る。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator リクエスト DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param closingCommonSafeOtherRetailResource MA-074 リソース
     * @param ciriusConstantService CIRIUS コンスタント取得サービス
     */
    @Inject
    public SupplyStopReportUpdateOtherRetailResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            ClosingCommonSafeOtherRetailResource closingCommonSafeOtherRetailResource,
            CiriusConstantService ciriusConstantService) {
        super(requestContext, requestDtoValidator, clock);
        this.closingCommonSafeOtherRetailResource = Objects.requireNonNull(
                closingCommonSafeOtherRetailResource,
                "closingCommonSafeOtherRetailResource");
        CiriusConstantService nonNullCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.childBusinessCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.SUPPLY_OUTAGE_REPORT_CONDUIT);
        this.childBusinessDetailCode = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.SUPPLY_OUTAGE_CCJG1201);
        this.childReportRouteDivision = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.TYPE,
                ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.ONLINE_REPORTING_USUALLY);
        this.childClosingReason = RequiredConstantResolver.requireCiriusConstant(
                nonNullCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.SUPPLY_OUTAGE);
    }

    /**
     * MA-026 供給停止報告更新-導管-他小売 API を実行する。
     *
     * @param request リクエスト DTO
     * @return レスポンス DTO
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "供給停止報告更新-導管-他小売",
            description = "供給停止報告情報を MA-074 へ委譲し、MA-026 の応答形式に整形する。",
            operationId = "postSupplyStopReportUpdateOtherRetail")
    @RequestBody(required = true, content = @Content(
            mediaType = MediaType.APPLICATION_JSON,
            schema = @Schema(implementation = SupplyStopReportUpdateOtherRetailRequest.class)))
    @APIResponse(
            responseCode = "200",
            description = "供給停止報告更新結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = SupplyStopReportUpdateOtherRetailResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            anyOf = {
                                CommonErrorResponse.class,
                                SupplyStopReportUpdateOtherRetailResponse.class
                            })))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public SupplyStopReportUpdateOtherRetailResponse endpoint(
            SupplyStopReportUpdateOtherRetailRequest request) {
        bindUuidCorrelationIdForResponse();
        return execute(request, () -> handle(request));
    }

    private void bindUuidCorrelationIdForResponse() {
        String correlationId = CorrelationIdResolver.resolveUuidV4(
                requestContext.getCorrelationId().orElse(null));
        requestContext.setCorrelationId(correlationId);
        MDC.put(MDC_CORRELATION_ID_KEY, correlationId);
    }

    SupplyStopReportUpdateOtherRetailResponse handle(
            SupplyStopReportUpdateOtherRetailRequest request) {
        try {
            ClosingCommonSafeOtherRetailResponse response =
                    closingCommonSafeOtherRetailResource.delegateFromParent(toChildRequest(request));
            return toResponse(response);
        } catch (WebApplicationException exception) {
            ClosingCommonSafeOtherRetailResponse childResponse = asChildResponse(exception);
            if (childResponse == null) {
                throw exception;
            }
            if (HeisenRuleListException.ERROR_CODE.equals(childResponse.errorCode())) {
                return toResponse(childResponse);
            }
            throw badRequestResponse(toResponse(childResponse));
        }
    }

    private ClosingCommonSafeOtherRetailRequest toChildRequest(
            SupplyStopReportUpdateOtherRetailRequest request) {
        SupplyStopReportUpdateOtherRetailRequest.KkysHkkInf reportInfo = request.kkysHkkInf();
        return new ClosingCommonSafeOtherRetailRequest(
                childBusinessCode,
                childBusinessDetailCode,
                childReportRouteDivision,
                resolveReportDate(),
                reportInfo == null ? null : reportInfo.hesnYmd(),
                stringify(request.gasMtstNo()),
                stringify(request.syoKyoTrno()),
                request.gmb(),
                childClosingReason,
                reportInfo == null ? null : stringify(reportInfo.gmtShjSu()),
                reportInfo == null ? null : stringify(reportInfo.gasSyry()),
                reportInfo == null ? null : reportInfo.sjskKbn(),
                reportInfo == null ? null : reportInfo.hukkUsrId(),
                reportInfo == null ? null : reportInfo.hukkUnmKj(),
                reportInfo == null ? null : reportInfo.rirekiMemo(),
                reportInfo == null
                        ? null
                        : new ClosingCommonSafeOtherRetailRequest.WarnKknInf(reportInfo.kknnFlg()),
                request.hukkSysShbt(),
                request.systemDatetime());
    }

    private String stringify(Object value) {
        if (value == null) {
            return null;
        }
        if (value instanceof BigDecimal decimalValue) {
            return decimalValue.toPlainString();
        }
        return value.toString();
    }

    private String resolveReportDate() {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set"));
        return systemDatetime.toLocalDate().toString();
    }

    private SupplyStopReportUpdateOtherRetailResponse toResponse(
            ClosingCommonSafeOtherRetailResponse response) {
        List<String> warningMessageIds = normalizeWarningMessageIds(response);
        String errorCode = resolveErrorCode(response, warningMessageIds);
        return new SupplyStopReportUpdateOtherRetailResponse(
                warningMessageIds == null
                        ? null
                        : new SupplyStopReportUpdateOtherRetailResponse.SyryWarnInf(warningMessageIds),
                response.correlationId(),
                errorCode,
                response.errorDescription(),
                normalizeErrorDetailList(response, errorCode),
                resolveCiriusMessageId(response),
                resolveCiriusEmbeddedStringList(response));
    }

    private String resolveErrorCode(
            ClosingCommonSafeOtherRetailResponse response,
            List<String> warningMessageIds) {
        if (response.errorCode() != null && !response.errorCode().isBlank()) {
            return response.errorCode();
        }
        if (warningMessageIds != null) {
            return HeisenRuleListException.ERROR_CODE;
        }
        return null;
    }

    private List<String> normalizeErrorDetailList(
            ClosingCommonSafeOtherRetailResponse response,
            String errorCode) {
        if (errorCode == null || errorCode.isBlank()) {
            return null;
        }
        return response.errorDetailList() == null ? List.of() : response.errorDetailList();
    }

    private List<String> normalizeWarningMessageIds(ClosingCommonSafeOtherRetailResponse response) {
        if (response.syryWarnInf() == null || response.syryWarnInf().crsMsgIdList() == null) {
            return null;
        }
        return response.syryWarnInf().crsMsgIdList().isEmpty()
                ? null
                : response.syryWarnInf().crsMsgIdList();
    }

    private String resolveCiriusMessageId(ClosingCommonSafeOtherRetailResponse response) {
        if (response.errorCode() == null || response.errorCode().isBlank()) {
            return null;
        }
        return ErrorCatalog.resolveCiriusMessageId(response.errorCode());
    }

    private List<String> resolveCiriusEmbeddedStringList(ClosingCommonSafeOtherRetailResponse response) {
        if (response.errorCode() == null || response.errorCode().isBlank()) {
            return null;
        }
        if (ErrorCodeConstants.ERROR_CODE_020200026.equals(response.errorCode())
                && response.errGmsbNo() != null) {
            return List.of(formatGasMeterLocationNumber(response.errGmsbNo()));
        }
        if (ErrorCodeConstants.ERROR_CODE_020200042.equals(response.errorCode())
                && response.snstSyoRyo() != null) {
            return List.of(formatUsageAmount(response.snstSyoRyo()));
        }
        return null;
    }

    private String formatGasMeterLocationNumber(Long gasMeterLocationNumber) {
        String zeroFilled = String.format("%011d", gasMeterLocationNumber);
        return zeroFilled.substring(0, 4)
                + "-"
                + zeroFilled.substring(4, 7)
                + "-"
                + zeroFilled.substring(7);
    }

    private String formatUsageAmount(Double calculatedUsageAmount) {
        return BigDecimal.valueOf(calculatedUsageAmount)
                .setScale(4, RoundingMode.HALF_UP)
                .toPlainString();
    }

    private ClosingCommonSafeOtherRetailResponse asChildResponse(WebApplicationException exception) {
        Response response = exception.getResponse();
        if (response == null) {
            return null;
        }
        Object entity = response.getEntity();
        return entity instanceof ClosingCommonSafeOtherRetailResponse childResponse
                ? childResponse
                : null;
    }

    private WebApplicationException badRequestResponse(
            SupplyStopReportUpdateOtherRetailResponse response) {
        return new WebApplicationException(
                Response.status(Response.Status.BAD_REQUEST)
                        .type(MediaType.APPLICATION_JSON_TYPE)
                        .entity(response)
                        .build());
    }
}
