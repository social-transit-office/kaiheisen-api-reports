    private boolean isMa026CompatibleNegativeGasUsage(String gasSyry) {
        if (gasSyry == null) {
            return false;
        }
        String normalized = gasSyry.trim();
        if (!normalized.startsWith("-") || !SIGNED_DECIMAL_PATTERN.matcher(normalized).matches()) {
            return false;
        }
        String unsigned = normalized.substring(1);
        String[] parts = unsigned.split("\\.", -1);
        int integerLength = parts[0].length();
        int fractionLength = parts.length > 1 ? parts[1].length() : 0;
        return integerLength <= MA026_GAS_USAGE_INTEGER_DIGITS
                && fractionLength <= MA026_GAS_USAGE_FRACTION_DIGITS;
    }
