package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_closing.retail.retail_tg;

import jakarta.validation.ConstraintValidator;
import jakarta.validation.ConstraintValidatorContext;

/**
 * {@link SignedNumericDecimalMaxDigits} の Bean Validation 実装です。
 */
public class SignedNumericDecimalMaxDigitsValidator
        implements ConstraintValidator<SignedNumericDecimalMaxDigits, String> {

    private int integerDigits;

    private int fractionDigits;

    @Override
    public void initialize(SignedNumericDecimalMaxDigits annotation) {
        this.integerDigits = annotation.integerDigits();
        this.fractionDigits = annotation.fractionDigits();
    }

    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        if (value == null || value.isBlank()) {
            return true;
        }

        String regex = "^-?\\d{1," + integerDigits + "}(\\.\\d{1," + fractionDigits + "})?$";
        return value.matches(regex);
    }
}
