package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.config;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Produces;
import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantServices;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.context.SvContextProxyFactory;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tkssvky.TkssvkyDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.OtherRetailClosingScheduledDateDiscrepancyErrorJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.RetailTypeJudgment;
import jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation.RetailTypeJudgmentImpl;

/**
 * pod_closing で利用する共通基盤 Bean を定義する設定クラス。
 */
@ApplicationScoped
public class PodClosingLibsConfiguration {

    @Inject
    SvContextProxyFactory svContextProxyFactory;

    /**
     * 定数サービス一式をアプリケーション内で1回だけ生成する。
     *
     * <p>主な仕様: 2つの producer が同じサービス一式を共有し、各辞書の二重読込を防ぐ。
     * 既知の制約: 取得元設定の変更にはアプリケーションの再起動が必要。</p>
     *
     * @return 共有する定数サービス一式
     */
    @Produces
    @Singleton
    ConstantServices.Bundle constantServices() {
        return ConstantServices.create();
    }

    /**
     * 共有する定数サービス一式から CIRIUS 定数サービスを取得する。
     *
     * @param constantServices アプリケーション内で共有する定数サービス一式
     * @return CIRIUS 定数サービス
     */
    @Produces
    @Singleton
    CiriusConstantService ciriusConstantService(ConstantServices.Bundle constantServices) {
        return constantServices.ciriusConstantService();
    }

    /**
     * 共有する定数サービス一式から開閉栓定数サービスを取得する。
     *
     * @param constantServices アプリケーション内で共有する定数サービス一式
     * @return 開閉栓定数サービス
     */
    @Produces
    @Singleton
    KaiheisenConstantService kaiheisenConstantService(ConstantServices.Bundle constantServices) {
        return constantServices.kaiheisenConstantService();
    }

    /**
     * MA-068 用の小売判定サービスを生成する。
     *
     * @param tskgmstDataAccess TSKGMST DataAccess
     * @param tkssvkyDataAccess TKSSVKY DataAccess
     * @param ciriusConstantService CIRIUS 定数サービス
     * @return 小売判定サービス
     */
    @Produces
    @Singleton
    RetailTypeJudgment retailTypeJudgment(
            TskgmstDataAccess tskgmstDataAccess,
            TkssvkyDataAccess tkssvkyDataAccess,
            CiriusConstantService ciriusConstantService) {
        return wrapService(
                RetailTypeJudgment.class,
                new RetailTypeJudgmentImpl(
                        tskgmstDataAccess,
                        tkssvkyDataAccess,
                        ciriusConstantService));
    }

    /**
     * MA-068 用の他小売閉栓予定日相違エラー判定サービスを生成する。
     *
     * @param ciriusConstantService CIRIUS 定数サービス
     * @return 他小売閉栓予定日相違エラー判定サービス
     */
    @Produces
    @Singleton
    OtherRetailClosingScheduledDateDiscrepancyErrorJudgment
            otherRetailClosingScheduledDateDiscrepancyErrorJudgment(
                    CiriusConstantService ciriusConstantService) {
        return wrapService(
                OtherRetailClosingScheduledDateDiscrepancyErrorJudgment.class,
                new jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation
                        .OtherRetailClosingScheduledDateDiscrepancyErrorJudgmentImpl(
                                ciriusConstantService));
    }

    private <T> T wrapService(Class<T> serviceType, T service) {
        return svContextProxyFactory.wrap(serviceType, service);
    }
}
