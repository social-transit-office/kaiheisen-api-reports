package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.config;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.time.LocalDate;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_other.ClosingCommonForceOtherRetailResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_other.ClosingCommonNormalOtherRetailResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_tg.ClosingCommonForceTgRetailResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_tg.ClosingCommonNormalTgRetailResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_closing.retail.retail_tg.ClosingCommonSupplyStopResource;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.CdiBeanInitializer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupContributor;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupStep;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gmtsetd.GmtsetdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;

/** 閉栓MAの起動時ウォームアップ定義。 */
@ApplicationScoped
public class PodClosingStartupWarmupContributor implements StartupWarmupContributor {

    private static final long WARMUP_GAS_MTST_NO = 99_999_999_999L;

    private static final LocalDate WARMUP_KNSK_KJN_YMD = LocalDate.of(9999, 12, 31);

    private final CdiBeanInitializer beanInitializer;

    private final Sysdummy1DataAccess databaseWarmupDataAccess;

    private final GmtsetdDataAccess businessDataAccess;

    /**
     * コンストラクタ。
     *
     * @param beanInitializer CDI Bean初期化処理
     * @param databaseWarmupDataAccess DBウォームアップDA
     * @param businessDataAccess 業務参照DA
     */
    @Inject
    public PodClosingStartupWarmupContributor(
            final CdiBeanInitializer beanInitializer,
            final Sysdummy1DataAccess databaseWarmupDataAccess,
            final GmtsetdDataAccess businessDataAccess) {
        this.beanInitializer = beanInitializer;
        this.databaseWarmupDataAccess = databaseWarmupDataAccess;
        this.businessDataAccess = businessDataAccess;
    }

    @Override
    public String applicationName() {
        return "ma-closing";
    }

    @Override
    public LogLayer logLayer() {
        return LogLayer.MA;
    }

    @Override
    public void initializeResources() {
        beanInitializer.initialize(
                ClosingCommonNormalTgRetailResource.class,
                ClosingCommonSupplyStopResource.class,
                ClosingCommonNormalOtherRetailResource.class,
                ClosingCommonForceTgRetailResource.class,
                ClosingCommonForceOtherRetailResource.class);
    }

    @Override
    public int warmUpDatabase() {
        return databaseWarmupDataAccess.fetchWarmupValue().value();
    }

    @Override
    public List<StartupWarmupStep> businessDataAccessWarmupSteps() {
        return List.of(new StartupWarmupStep(
                GmtsetdDataAccess.Identifiers.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD,
                () -> {
                    if (businessDataAccess
                            .findByGasMtstNoAndTkysYmdAndTyenYmd(
                                    WARMUP_GAS_MTST_NO, WARMUP_KNSK_KJN_YMD)
                            .isPresent()) {
                        throw new IllegalStateException(
                                "Startup business DA warmup dummy value collision");
                    }
                }));
    }
}
