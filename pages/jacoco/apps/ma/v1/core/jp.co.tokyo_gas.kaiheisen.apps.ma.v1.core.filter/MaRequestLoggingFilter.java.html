/**
 * @file MaRequestLoggingFilter.java
 * 概要: MA リソース向けのHTTP 入口/出口ログフィルタ。
 * 主な仕様:
 * - AbstractRequestLoggingFilter を利用して共通ログを出力する。
 * 既知の制限事項:
 * - PII/秘匿のマスキングは未実装。
 * 型: Javaソース
 */

package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.filter;

import jakarta.annotation.Priority;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import jakarta.ws.rs.Priorities;
import jakarta.ws.rs.ext.Provider;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.logging.AbstractRequestLoggingFilter;

/**
 * MA 共通のリクエストログフィルタ。
 */
@Provider
@Priority(Priorities.USER)
@ApplicationScoped
public class MaRequestLoggingFilter extends AbstractRequestLoggingFilter {

    @Inject private RequestContext requestContext;

    /**
     * MA レイヤでのリクエストログフィルタを初期化する。
     */
    public MaRequestLoggingFilter() {
        super(MaRequestLoggingFilter.class, LogLayer.MA);
    }

    @Override
    protected void onRequestStartNanos(final long startNanos) {
        if (!performanceObservabilityEnabled()) {
            return;
        }
        try {
            requestContext.setRequestInNanos(startNanos);
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }
}
