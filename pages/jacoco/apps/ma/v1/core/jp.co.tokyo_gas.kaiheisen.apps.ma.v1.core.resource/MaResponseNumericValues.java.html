package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource;

import java.math.BigDecimal;

/**
 * MAレスポンス向けの数値null補完ユーティリティ.
 *
 * <p>主な仕様: Long、Integer、BigDecimal の値がnullの場合に0へ補完する.
 *
 * <p>既知の制約: BigDecimalはnullの場合のみ指定スケールの0を返し、nullでない値は変更しない.
 */
public final class MaResponseNumericValues {

    /** 小数第1位のスケール. */
    private static final int ONE_DECIMAL_SCALE = 1;

    /** 小数第4位のスケール. */
    private static final int FOUR_DECIMAL_SCALE = 4;

    private MaResponseNumericValues() {
    }

    /**
     * Long値がnullの場合に0へ補完する.
     *
     * @param value 変換対象の値
     * @return 引数がnullの場合は0L、nullでない場合は元の値
     */
    public static Long orZero(final Long value) {
        return value == null ? 0L : value;
    }

    /**
     * Integer値がnullの場合に0へ補完する.
     *
     * @param value 変換対象の値
     * @return 引数がnullの場合は0、nullでない場合は元の値
     */
    public static Integer orZero(final Integer value) {
        return value == null ? 0 : value;
    }

    /**
     * BigDecimal値がnullの場合に指定スケールの0へ補完する.
     *
     * @param value 変換対象の値
     * @param scale null時に補完する0のスケール
     * @return 引数がnullの場合は指定スケールの0、nullでない場合は元の値
     */
    public static BigDecimal orZeroWithScale(
            final BigDecimal value,
            final int scale) {
        return value == null ? BigDecimal.ZERO.setScale(scale) : value;
    }

    /**
     * BigDecimal値がnullの場合に小数第1位の0へ補完する.
     *
     * @param value 変換対象の値
     * @return 引数がnullの場合は0.0、nullでない場合は元の値
     */
    public static BigDecimal oneDecimalOrZero(final BigDecimal value) {
        return orZeroWithScale(value, ONE_DECIMAL_SCALE);
    }

    /**
     * BigDecimal値がnullの場合に小数第4位の0へ補完する.
     *
     * @param value 変換対象の値
     * @return 引数がnullの場合は0.0000、nullでない場合は元の値
     */
    public static BigDecimal fourDecimalOrZero(final BigDecimal value) {
        return orZeroWithScale(value, FOUR_DECIMAL_SCALE);
    }
}
