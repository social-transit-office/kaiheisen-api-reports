package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource;

import java.time.OffsetDateTime;
import java.time.format.DateTimeParseException;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.request.BaseRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.DefaultExceptionLogMapper;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.ExceptionLogMapper;

import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.time.SystemDateTimeProvider;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.RequestDtoValidator;

/**
 * MA リクエスト共通パラメータの RequestContext バインドを担う基底クラス。
 *
 * @param <T> 共通パラメータを提供するリクエスト DTO 型
 */
public abstract class BaseResource<T extends BaseRequest> {

    /** 開始時のイベント名。 */
    private static final String RESOURCE_STARTED_EVENT = "resource.started";

    /** 開始時の要約。 */
    private static final String RESOURCE_STARTED_SUMMARY = "MA resource action started";

    /** Successful terminal event name. */
    private static final String RESOURCE_COMPLETED_EVENT = "resource.completed";

    /** Successful terminal event summary. */
    private static final String RESOURCE_COMPLETED_SUMMARY = "MA resource action completed";

    /** Failed terminal event name. */
    private static final String RESOURCE_FAILED_EVENT = "resource.failed";

    /** Failed terminal event summary. */
    private static final String RESOURCE_FAILED_SUMMARY = "MA resource action failed";

    /** 機能ID属性名。 */
    private static final String FUNCTION_ID_ATTRIBUTE = "functionId";

    /** 既定の例外ログマッパー。 */
    private static final ExceptionLogMapper DEFAULT_EXCEPTION_LOG_MAPPER =
            new DefaultExceptionLogMapper(LogLayer.MA);

    protected final RequestContext requestContext;

    private final RequestDtoValidator requestDtoValidator;

    private final SystemDateTimeProvider clock;

    private final StructuredLogger logger;

    protected BaseResource(
            final RequestContext requestContext,
            final RequestDtoValidator requestDtoValidator,
            final SystemDateTimeProvider clock) {
        this.requestContext = Objects.requireNonNull(requestContext, "requestContext");
        this.requestDtoValidator = Objects.requireNonNull(requestDtoValidator, "requestDtoValidator");
        this.clock = Objects.requireNonNull(clock, "clock");
        this.logger = new JbossStructuredLogger(getClass());
    }

    protected final <R> R execute(T request, Supplier<R> action) {
        Objects.requireNonNull(action, "action");
        bindRequestContext(request);
        try {
            validate(request);
            String functionId = resolveFunctionId();
            boolean terminalMeasurementEnabled =
                    PerformanceObservabilitySettings.isTerminalEnabled();
            long startedAtNanos = System.nanoTime();
            if (terminalMeasurementEnabled) {
                safeLogResourceStarted(request, functionId);
            } else {
                logResourceStarted(request, functionId);
            }
            try {
                R result = action.get();
                if (terminalMeasurementEnabled) {
                    logResourceTerminal(
                            RESOURCE_COMPLETED_EVENT,
                            RESOURCE_COMPLETED_SUMMARY,
                            functionId,
                            startedAtNanos);
                }
                return result;
            } catch (RuntimeException | Error throwable) {
                if (terminalMeasurementEnabled) {
                    logResourceTerminal(
                            RESOURCE_FAILED_EVENT,
                            RESOURCE_FAILED_SUMMARY,
                            functionId,
                            startedAtNanos);
                }
                throw throwable;
            }
        } catch (RuntimeException ex) {
            logException(ex, logger(), exceptionLogMapper());
            throw ex;
        }
    }

    protected final void executeVoid(T request, Runnable action) {
        Objects.requireNonNull(action, "action");
        bindRequestContext(request);
        try {
            validate(request);
            String functionId = resolveFunctionId();
            boolean terminalMeasurementEnabled =
                    PerformanceObservabilitySettings.isTerminalEnabled();
            long startedAtNanos = System.nanoTime();
            if (terminalMeasurementEnabled) {
                safeLogResourceStarted(request, functionId);
            } else {
                logResourceStarted(request, functionId);
            }
            try {
                action.run();
                if (terminalMeasurementEnabled) {
                    logResourceTerminal(
                            RESOURCE_COMPLETED_EVENT,
                            RESOURCE_COMPLETED_SUMMARY,
                            functionId,
                            startedAtNanos);
                }
            } catch (RuntimeException | Error throwable) {
                if (terminalMeasurementEnabled) {
                    logResourceTerminal(
                            RESOURCE_FAILED_EVENT,
                            RESOURCE_FAILED_SUMMARY,
                            functionId,
                            startedAtNanos);
                }
                throw throwable;
            }
        } catch (RuntimeException ex) {
            logException(ex, logger(), exceptionLogMapper());
            throw ex;
        }
    }

    protected final void bindRequestContext(T request) {
        Objects.requireNonNull(request, "request");
        // 相関IDは CorrelationIdFilter が RequestContext へ設定済みのため、ここでは扱わない
        requestContext.setHukkSysShbt(normalizeOptional(request.hukkSysShbt()));
        requestContext.setSystemDatetime(resolveSystemDatetime(request.systemDatetime()));
    }

    /**
     * 単一項目バリデーションと項目間バリデーションを実行する。
     *
     * <p>サブクラスが独自の追加検証を行う場合も、既定の DTO 検証と
     * {@link #interFieldValidation()} の結果を維持すること。</p>
     *
     * @param request リクエスト DTO
     */
    protected void validate(T request) {
        List<ValidationError> errorDetailList = new ArrayList<>();
        appendErrors(errorDetailList, requestDtoValidator.validate(request));
        appendErrors(errorDetailList, interFieldValidation().validate(request));
        if (!errorDetailList.isEmpty()) {
            throw new ValidationException(errorDetailList, true);
        }
    }

    /**
     * ログ出力に使用するロガーを返す。デフォルトではサブクラス名をカテゴリとする。
     *
     * @return ロガー
     */
    protected StructuredLogger logger() {
        return logger;
    }

    /**
     * 例外ログマッパーを返す。デフォルトでは MA レイヤ用の既定マッパーを使用する。
     *
     * @return 例外ログマッパー
     */
    protected ExceptionLogMapper exceptionLogMapper() {
        return DEFAULT_EXCEPTION_LOG_MAPPER;
    }

    /**
     * 項目間バリデーションを返す。
     *
     * <p>サブクラスで上書きする場合は、副作用を持たない検証実装を返し、
     * 検証エラーは {@link ValidationError} のリストで返すこと。</p>
     *
     * @return 項目間バリデーション
     */
    protected InterFieldValidation<T> interFieldValidation() {
        return InterFieldValidation.noOp();
    }

    protected final void logException(Throwable throwable, StructuredLogger logger, ExceptionLogMapper mapper) {
        Objects.requireNonNull(logger, "logger");
        Objects.requireNonNull(mapper, "mapper");
        LogDecision decision = mapper.map(throwable, exceptionLogContext());
        if (decision == null || decision.event() == null) {
            return;
        }
        if (decision.level() == LogLevel.ERROR) {
            logger.error(decision.event(), throwable);
        } else if (decision.level() == LogLevel.WARN) {
            logger.warn(decision.event());
        } else {
            logger.info(decision.event());
        }
        requestContext.markExceptionLogged();
    }

    private ExceptionLogContext exceptionLogContext() {
        return new ExceptionLogContext(
                requestContext.getCorrelationId().orElse(null),
                requestContext.getHttpMethod().orElse(null),
                requestContext.getRequestPath().orElse(null),
                getClass().getName(),
                requestContext.getServiceId().orElse(null),
                requestContext.getServiceName().orElse(null));
    }

    protected final String normalizeOptional(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }

    private void logResourceStarted(T request, final String functionId) {
        requestContext.setRootFunctionIdIfAbsent(functionId);
        Map<String, Object> attributes = new LinkedHashMap<>();
        attributes.put(FUNCTION_ID_ATTRIBUTE, functionId);
        String callPath = requestContext.currentCallPath();
        if (callPath != null) {
            attributes.put(LogField.CALL_PATH.key(), callPath);
        }
        Map<String, Object> requestBody = toRequestBody(request);
        if (!requestBody.isEmpty()) {
            attributes.put(LogField.REQUEST_BODY.key(), requestBody);
        }
        appendRequestToResourceLatency(attributes);
        logger().info(new LogEvent(
                RESOURCE_STARTED_EVENT,
                RESOURCE_STARTED_SUMMARY,
                LogLayer.MA,
                null,
                null,
                attributes));
    }

    private void safeLogResourceStarted(final T request, final String functionId) {
        try {
            logResourceStarted(request, functionId);
        } catch (RuntimeException ignored) {
            // Observability must not alter the resource action result.
        }
    }

    private void logResourceTerminal(
            final String eventName,
            final String summary,
            final String functionId,
            final long startedAtNanos) {
        try {
            Map<String, Object> attributes = new LinkedHashMap<>();
            attributes.put(FUNCTION_ID_ATTRIBUTE, functionId);
            String callPath = requestContext.currentCallPath();
            if (callPath != null) {
                attributes.put(LogField.CALL_PATH.key(), callPath);
            }
            attributes.put(
                    LogField.LATENCY_MS.key(),
                    Math.max(0L, System.nanoTime() - startedAtNanos) / 1_000_000L);
            safeInfo(new LogEvent(
                    eventName,
                    summary,
                    LogLayer.MA,
                    null,
                    null,
                    attributes));
        } catch (RuntimeException ignored) {
            // Observability must not alter the resource action result.
        }
    }

    private void safeInfo(final LogEvent event) {
        try {
            logger().info(event);
        } catch (RuntimeException ignored) {
            // Observability must not alter the resource action result.
        }
    }

    private void appendRequestToResourceLatency(final Map<String, Object> attributes) {
        if (!PerformanceObservabilitySettings.isEnabled()) {
            return;
        }
        try {
            requestContext.elapsedSinceRequestInMillis(System.nanoTime())
                    .ifPresentOrElse(
                            latency -> attributes.put(
                                    LogField.REQUEST_TO_RESOURCE_LATENCY_MS.key(),
                                    latency),
                            () -> attributes.put(LogField.MEASUREMENT_UNAVAILABLE.key(), true));
        } catch (RuntimeException exception) {
            attributes.put(LogField.MEASUREMENT_UNAVAILABLE.key(), true);
        }
    }

    private Map<String, Object> toRequestBody(T request) {
        return LogRequestBodyMapper.toMap(request);
    }

    private String resolveFunctionId() {
        Class<?> resourceClass = getClass();
        while (resourceClass != null) {
            MainService mainService = resourceClass.getAnnotation(MainService.class);
            if (mainService != null) {
                return mainService.identifier();
            }
            resourceClass = resourceClass.getSuperclass();
        }
        return null;
    }

    private void appendErrors(List<ValidationError> collector, List<ValidationError> errors) {
        if (errors == null || errors.isEmpty()) {
            return;
        }
        collector.addAll(errors);
    }

    private LocalDateTime resolveSystemDatetime(String rawSystemDatetime) {
        String normalized = normalizeOptional(rawSystemDatetime);
        if (normalized == null) {
            return clock.currentDateTime();
        }
        try {
            return OffsetDateTime.parse(normalized).toLocalDateTime();
        } catch (DateTimeParseException ex) {
            try {
                return LocalDateTime.parse(normalized);
            } catch (DateTimeParseException nested) {
                throw new ValidationException(
                        List.of(ValidationError.ofField(
                                ErrorCodeConstants.ERROR_CODE_010000015,
                                "systemDatetime")),
                        true);
            }
        }
    }
}
