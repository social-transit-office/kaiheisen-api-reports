package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationError;

/**
 * MA リクエストの項目間バリデーションを提供する拡張ポイント。
 *
 * @param <T> リクエスト DTO 型
 */
@FunctionalInterface
public interface InterFieldValidation<T> {

    /**
     * 項目間バリデーションを実行する。
     *
     * @param request リクエスト DTO
     * @return バリデーションエラーリスト。エラーがない場合は空配列
     */
    List<ValidationError> validate(T request);

    /**
     * 項目間バリデーションを行わない no-op 実装を返す。
     *
     * @param <T> リクエスト DTO 型
     * @return no-op 実装
     */
    static <T> InterFieldValidation<T> noOp() {
        return request -> List.of();
    }
}
