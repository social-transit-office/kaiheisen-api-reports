package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource.exception_mapper;

import jakarta.inject.Inject;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.ext.Provider;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.response.KaiheisenErrorResponse;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.DefaultExceptionLogMapper;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.ExceptionLogContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.ExceptionLogMapper;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogDecision;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLevel;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;

/**
 * 未捕捉のアプリケーション例外を MA 共通の 500 エラーレスポンスへ変換するマッパー。
 */
@Provider
public class RuntimeExceptionMapper implements ExceptionMapper<Exception> {

    private final RequestContext requestContext;

    private final StructuredLogger logger;

    private final ExceptionLogMapper exceptionLogMapper;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public RuntimeExceptionMapper(RequestContext requestContext) {
        this(
                requestContext,
                new JbossStructuredLogger(RuntimeExceptionMapper.class),
                new DefaultExceptionLogMapper(LogLayer.MA));
    }

    RuntimeExceptionMapper(
            RequestContext requestContext,
            StructuredLogger logger,
            ExceptionLogMapper exceptionLogMapper) {
        this.requestContext = requestContext;
        this.logger = logger;
        this.exceptionLogMapper = exceptionLogMapper;
    }

    @Override
    public Response toResponse(Exception exception) {
        if (exception instanceof WebApplicationException webApplicationException) {
            return Response.fromResponse(webApplicationException.getResponse()).build();
        }
        logIfNeeded(exception);

        KaiheisenErrorResponse body = MaErrorResponseResolver.fromUnhandledException(exception);
        return Response.serverError()
                .type(MediaType.APPLICATION_JSON)
                .entity(body)
                .build();
    }

    private void logIfNeeded(Exception exception) {
        if (requestContext.isExceptionLogged()) {
            return;
        }
        LogDecision decision = exceptionLogMapper.map(exception, new ExceptionLogContext(
                requestContext.getCorrelationId().orElse(null),
                requestContext.getHttpMethod().orElse(null),
                requestContext.getRequestPath().orElse(null),
                null,
                requestContext.getServiceId().orElse(null),
                requestContext.getServiceName().orElse(null)));
        if (decision == null || decision.event() == null) {
            return;
        }
        if (decision.level() == LogLevel.ERROR) {
            logger.error(decision.event(), exception);
        } else if (decision.level() == LogLevel.WARN) {
            logger.warn(decision.event());
        } else {
            logger.info(decision.event());
        }
        requestContext.markExceptionLogged();
    }
}
