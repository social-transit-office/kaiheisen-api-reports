package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource.exception_mapper;

import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.ext.Provider;
import java.util.Optional;
import java.util.function.Function;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.response.ErrorResponseBody;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.MaApiException;

/**
 * MA から返された HTTP エラーを MA からそのまま返却するマッパー。
 * ただし、呼出し元MAの個別仕様が存在する場合は、それを取り込む。
 */
@Provider
public class MaPassthroughMaApiExceptionMapper implements ExceptionMapper<MaPassthroughMaApiException> {

    /**
     * MA から返された HTTP エラーをレスポンスへ変換する。
     * 呼出し元MAの個別仕様が存在する場合はそれを取り込む。
     *
     * @param exception MA エラー
     * @return 変換後レスポンス
     */
    @Override
    public Response toResponse(MaPassthroughMaApiException exception) {

        Function<MaPassthroughMaApiException, MaPassthroughMaApiException> safeCustomizer =
                Optional.ofNullable(exception.getCustomizer()).orElse(Function.identity());

        return MaPassthroughMaApiException.from(exception.getMaApiException())
                .map(safeCustomizer)
                .map(this::toPassthroughResponse)
                .orElseGet(() -> toFallbackResponse(exception.getMaApiException()));
    }

    private Response toPassthroughResponse(MaPassthroughMaApiException exception) {
        Response.ResponseBuilder builder = Response.status(exception.getStatus());
        String mediaType = exception.getMediaType();
        builder.type(mediaType == null ? MediaType.APPLICATION_JSON : mediaType);

        if (exception.getResponseBody() != null) {
            builder.entity(exception.getResponseBody());
        }
        return builder.build();
    }

    private Response toFallbackResponse(MaApiException exception) {
        ErrorResponseBody body = MaErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(MaErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(body)
                .build();
    }
}
