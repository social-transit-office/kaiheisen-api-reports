package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource.exception_mapper;

import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import java.util.Optional;
import java.util.function.Function;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.MaApiException;

/**
 * MA が MA エラーを HTTP ステータスとボディごと透過返却するための例外。
 */
public class MaPassthroughMaApiException extends RuntimeException {

    private static final String STATUS_PREFIX = "status=";

    private static final String BODY_PREFIX = ", body=";

    private static final String NULL_BODY = "null";

    private final int status;

    private final String responseBody;

    private final String mediaType;

    private final MaApiException maApiException;

    private final Function<MaPassthroughMaApiException, MaPassthroughMaApiException> customizer;

    /**
     * コンストラクタ。
     *
     * @param status HTTP ステータス
     * @param responseBody レスポンスボディ
     * @param mediaType レスポンスの Content-Type
     * @param cause 原因例外
     */
    public MaPassthroughMaApiException(int status, String responseBody, String mediaType, Throwable cause,
            Function<MaPassthroughMaApiException, MaPassthroughMaApiException> customizer) {
        super(cause);
        this.status = status;
        this.responseBody = responseBody;
        this.mediaType = mediaType;
        this.maApiException = (MaApiException) cause;
        this.customizer = customizer;
    }

    /**
     * コンストラクタ。
     *
     * @param status HTTP ステータス
     * @param responseBody レスポンスボディ
     * @param mediaType レスポンスの Content-Type
     * @param cause 原因例外
     */
    public MaPassthroughMaApiException(Throwable cause, Function<MaPassthroughMaApiException, MaPassthroughMaApiException> customizer) {
        this(0, null, null, cause, customizer);
    }

    /**
     * MaApiException から透過用例外を生成する。
     *
     * @param exception MA API 例外
     * @return HTTP レスポンスを保持できる場合の透過用例外
     */
    public static Optional<MaPassthroughMaApiException> from(MaApiException exception) {
        if (!(exception.getCause() instanceof WebApplicationException webApplicationException)) {
            return Optional.empty();
        }

        Response response = webApplicationException.getResponse();
        Integer status = extractStatus(exception.getResponseInfo());
        if (status == null && response == null) {
            return Optional.empty();
        }

        String body = extractBody(exception.getResponseInfo());
        if (body == null) {
            body = readResponseBody(response);
        }

        MediaType responseMediaType = response == null ? null : response.getMediaType();
        return Optional.of(new MaPassthroughMaApiException(
                status == null ? response.getStatus() : status,
                body,
                responseMediaType == null ? null : responseMediaType.toString(),
                exception, null));
    }

    private static Integer extractStatus(String responseInfo) {
        if (responseInfo == null || !responseInfo.startsWith(STATUS_PREFIX)) {
            return null;
        }

        int bodyPrefixIndex = responseInfo.indexOf(BODY_PREFIX);
        String statusText = bodyPrefixIndex >= 0
                ? responseInfo.substring(STATUS_PREFIX.length(), bodyPrefixIndex)
                : responseInfo.substring(STATUS_PREFIX.length());
        try {
            return Integer.valueOf(statusText.trim());
        } catch (NumberFormatException ignored) {
            return null;
        }
    }

    private static String extractBody(String responseInfo) {
        if (responseInfo == null) {
            return null;
        }

        int bodyPrefixIndex = responseInfo.indexOf(BODY_PREFIX);
        if (bodyPrefixIndex < 0) {
            return null;
        }

        String body = responseInfo.substring(bodyPrefixIndex + BODY_PREFIX.length());
        return NULL_BODY.equals(body) ? null : body;
    }

    private static String readResponseBody(Response response) {
        if (response == null || !response.hasEntity()) {
            return null;
        }

        try {
            response.bufferEntity();
            return response.readEntity(String.class);
        } catch (RuntimeException suppressed) {
            return null;
        }
    }

    /**
     * HTTP ステータスを返す。
     *
     * @return HTTP ステータス
     */
    public int getStatus() {
        return status;
    }

    /**
     * レスポンスボディを返す。
     *
     * @return レスポンスボディ
     */
    public String getResponseBody() {
        return responseBody;
    }

    /**
     * レスポンスの Content-Type を返す。
     *
     * @return レスポンスの Content-Type
     */
    public String getMediaType() {
        return mediaType;
    }

    /**
     * レスポンスの MaApiException を返す。
     *
     * @return レスポンスの MaApiException
     */
    public MaApiException getMaApiException() {
        return maApiException;
    }

    /**
     * レスポンスの Customizer を返す。
     *
     * @return レスポンスの Customizer
     */
    public Function<MaPassthroughMaApiException, MaPassthroughMaApiException> getCustomizer() {
        return customizer;
    }
}
