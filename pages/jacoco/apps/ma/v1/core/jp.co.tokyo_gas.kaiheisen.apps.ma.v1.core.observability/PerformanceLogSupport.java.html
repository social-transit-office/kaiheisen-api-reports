package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.observability;

import jakarta.enterprise.context.ContextNotActiveException;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.Map;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext.DataAccessInvocation;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;

/**
 * MA 詳細性能ログへ相関コンテキストを安全に付与する。
 */
final class PerformanceLogSupport {

    static final String DA_INVOCATION_SEQUENCE = LogField.DA_INVOCATION_SEQUENCE.key();

    static final String MEASUREMENT_UNAVAILABLE = LogField.MEASUREMENT_UNAVAILABLE.key();

    private PerformanceLogSupport() {
    }

    static Map<String, Object> contextAttributes(final RequestContext requestContext) {
        Map<String, Object> attributes = new LinkedHashMap<>();
        if (requestContext == null) {
            attributes.put(MEASUREMENT_UNAVAILABLE, true);
            return attributes;
        }
        try {
            requestContext.getCorrelationId().ifPresent(correlationId ->
                    attributes.put(LogField.CORRELATION_ID.key(), correlationId));
            requestContext.currentDataAccess().ifPresent(invocation ->
                    appendDataAccess(attributes, requestContext, invocation));
        } catch (ContextNotActiveException exception) {
            attributes.put(MEASUREMENT_UNAVAILABLE, true);
        } catch (RuntimeException exception) {
            attributes.put(MEASUREMENT_UNAVAILABLE, true);
        }
        return attributes;
    }

    static Map<String, Object> poolContextAttributes(final RequestContext requestContext) {
        try {
            if (requestContext == null) {
                return Collections.emptyMap();
            }
            boolean hasCorrelationId = requestContext.getCorrelationId().isPresent();
            boolean hasDataAccess = requestContext.currentDataAccess().isPresent();
            if (!hasCorrelationId && !hasDataAccess) {
                return Collections.emptyMap();
            }
            return contextAttributes(requestContext);
        } catch (ContextNotActiveException exception) {
            return Collections.emptyMap();
        } catch (RuntimeException exception) {
            return Collections.emptyMap();
        }
    }

    private static void appendDataAccess(
            final Map<String, Object> attributes,
            final RequestContext requestContext,
            final DataAccessInvocation invocation) {
        attributes.put(DA_INVOCATION_SEQUENCE, invocation.sequence());
        String callPath = requestContext.buildCallPath(invocation.functionId());
        if (callPath != null) {
            attributes.put(LogField.CALL_PATH.key(), callPath);
        }
    }
}
