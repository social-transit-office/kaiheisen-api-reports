package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.observability;

import io.agroal.api.AgroalDataSource;
import io.agroal.api.AgroalDataSourceMetrics;
import io.agroal.api.AgroalPoolInterceptor;
import io.quarkus.arc.Arc;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.sql.Connection;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.function.Supplier;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.PerformanceObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;

/**
 * Agroal の接続生成・取得・返却・破棄とプール統計を記録する。
 */
@ApplicationScoped
public class AgroalPerformanceInterceptor implements AgroalPoolInterceptor {

    private final RequestContext requestContext;

    private final Supplier<AgroalDataSourceMetrics> metricsSupplier;

    private final StructuredLogger logger;

    /**
     * CDI 用コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     */
    @Inject
    public AgroalPerformanceInterceptor(final RequestContext requestContext) {
        this(
                requestContext,
                AgroalPerformanceInterceptor::currentMetrics,
                new JbossStructuredLogger(AgroalPerformanceInterceptor.class));
    }

    AgroalPerformanceInterceptor(
            final RequestContext requestContext,
            final Supplier<AgroalDataSourceMetrics> metricsSupplier,
            final StructuredLogger logger) {
        this.requestContext = requestContext;
        this.metricsSupplier = metricsSupplier;
        this.logger = logger;
    }

    @Override
    public void onConnectionCreate(final Connection connection) {
        log("db.pool.connection.created", "Agroal connection created");
    }

    @Override
    public void onConnectionAcquire(final Connection connection) {
        log("db.pool.connection.acquired", "Agroal connection acquired");
    }

    @Override
    public void onConnectionReturn(final Connection connection) {
        log("db.pool.connection.returned", "Agroal connection returned");
    }

    @Override
    public void onConnectionDestroy(final Connection connection) {
        log("db.pool.connection.destroyed", "Agroal connection destroyed");
    }

    private void log(final String event, final String summary) {
        if (!PerformanceObservabilitySettings.isEnabled()) {
            return;
        }
        Map<String, Object> attributes;
        try {
            attributes = new LinkedHashMap<>(
                    PerformanceLogSupport.poolContextAttributes(requestContext));
        } catch (RuntimeException exception) {
            attributes = new LinkedHashMap<>();
            attributes.put(PerformanceLogSupport.MEASUREMENT_UNAVAILABLE, true);
        }
        try {
            appendMetrics(attributes, metricsSupplier.get());
        } catch (RuntimeException exception) {
            attributes.put(PerformanceLogSupport.MEASUREMENT_UNAVAILABLE, true);
        }
        try {
            logger.info(new LogEvent(event, summary, LogLayer.DA, null, null, attributes));
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }

    private static void appendMetrics(
            final Map<String, Object> attributes,
            final AgroalDataSourceMetrics metrics) {
        if (metrics == null) {
            attributes.put(PerformanceLogSupport.MEASUREMENT_UNAVAILABLE, true);
            return;
        }
        attributes.put("pool.activeCount", metrics.activeCount());
        attributes.put("pool.availableCount", metrics.availableCount());
        attributes.put("pool.awaitingCount", metrics.awaitingCount());
        attributes.put("pool.acquireCount", metrics.acquireCount());
        attributes.put("pool.creationCount", metrics.creationCount());
        attributes.put("pool.maxUsedCount", metrics.maxUsedCount());
        attributes.put("pool.blockingTimeAverageMs", toMillis(metrics.blockingTimeAverage()));
        attributes.put("pool.blockingTimeMaxMs", toMillis(metrics.blockingTimeMax()));
        attributes.put("pool.creationTimeAverageMs", toMillis(metrics.creationTimeAverage()));
        attributes.put("pool.creationTimeMaxMs", toMillis(metrics.creationTimeMax()));
    }

    private static Long toMillis(final java.time.Duration duration) {
        return duration == null ? null : duration.toMillis();
    }

    private static AgroalDataSourceMetrics currentMetrics() {
        return Arc.container().instance(AgroalDataSource.class).get().getMetrics();
    }
}
