package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.observability;

import io.quarkus.arc.Arc;
import java.util.Map;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.JbossStructuredLogger;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogEvent;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogField;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.PerformanceObservabilitySettings;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.StructuredLogger;
import org.hibernate.BaseSessionEventListener;

/**
 * Hibernate の JDBC 接続取得開始／終了を計測するリスナー。
 */
public class DbConnectionSessionEventListener extends BaseSessionEventListener {

    private static final String ACQUIRE_STARTED_EVENT = "db.connection.acquire.started";

    private static final String ACQUIRE_COMPLETED_EVENT = "db.connection.acquire.completed";

    private static final StructuredLogger LOGGER =
            new JbossStructuredLogger(DbConnectionSessionEventListener.class);

    private Long acquireStartNanos;

    @Override
    public void jdbcConnectionAcquisitionStart() {
        if (!PerformanceObservabilitySettings.isEnabled()) {
            return;
        }
        long startNanos = System.nanoTime();
        acquireStartNanos = startNanos;
        Map<String, Object> attributes = contextAttributes();
        logSafely(new LogEvent(
                ACQUIRE_STARTED_EVENT,
                "Hibernate JDBC connection acquisition started",
                LogLayer.DA,
                null,
                null,
                attributes));
    }

    @Override
    public void jdbcConnectionAcquisitionEnd() {
        Long startNanos = acquireStartNanos;
        acquireStartNanos = null;
        if (!PerformanceObservabilitySettings.isEnabled()) {
            return;
        }
        Map<String, Object> attributes = contextAttributes();
        if (startNanos == null) {
            attributes.put(PerformanceLogSupport.MEASUREMENT_UNAVAILABLE, true);
        } else {
            attributes.put(
                    LogField.LATENCY_MS.key(),
                    Math.max(0L, System.nanoTime() - startNanos) / 1_000_000L);
        }
        logSafely(new LogEvent(
                ACQUIRE_COMPLETED_EVENT,
                "Hibernate JDBC connection acquisition completed",
                LogLayer.DA,
                null,
                null,
                attributes));
    }

    private static Map<String, Object> contextAttributes() {
        try {
            return PerformanceLogSupport.contextAttributes(
                    Arc.container().instance(RequestContext.class).get());
        } catch (RuntimeException exception) {
            return PerformanceLogSupport.contextAttributes(null);
        }
    }

    private static void logSafely(final LogEvent event) {
        try {
            LOGGER.info(event);
        } catch (RuntimeException exception) {
            // 計測失敗は業務処理へ伝播させない
        }
    }
}
