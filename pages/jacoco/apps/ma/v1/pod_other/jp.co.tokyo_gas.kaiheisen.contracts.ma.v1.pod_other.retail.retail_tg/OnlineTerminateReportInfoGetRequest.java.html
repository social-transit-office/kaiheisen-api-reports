package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.retail.retail_tg;

import jakarta.ws.rs.QueryParam;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.request.BaseRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantLookupKeys;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.ConstantTypeValue;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.GasMtstNoFormat;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.RequiredText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateTimeText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.TextMaxLength;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * MA-033 オンライン解約報告情報取得 API のリクエスト。
 *
 * <p>相関IDは `X-Correlation-Id` ヘッダを基本としつつ、既存 TSV 互換のため
 * クエリパラメータ指定も受け付ける。</p>
 *
 * @param gasMtstNo ガスメーター設置場所番号
 * @param hukkSysShbt 報告システム種別
 * @param correlationId 相関ID
 * @param systemDatetime システム日時
 */
@Schema(description = "MA-033 オンライン解約報告情報取得リクエスト")
public record OnlineTerminateReportInfoGetRequest(
        @QueryParam("gasMtstNo")
        @RequiredText
        @GasMtstNoFormat
        @Schema(description = "ガスメーター設置場所番号", required = true, pattern = "^1\\d{10}$")
        String gasMtstNo,

        @QueryParam("hukkSysShbt")
        @TextMaxLength(max = 1)
        @ConstantTypeValue(type = ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE)
        @Schema(description = "報告システム種別", maxLength = 1, nullable = true)
        String hukkSysShbt,

        @QueryParam("correlationId")
        @TextMaxLength(max = 20)
        @Schema(description = "相関ID", maxLength = 20, nullable = true)
        String correlationId,

        @QueryParam("systemDatetime")
        @Rfc3339DateTimeText
        @Schema(description = "システム日時 (RFC3339)", format = "date-time", nullable = true)
        String systemDatetime) implements BaseRequest {
}
