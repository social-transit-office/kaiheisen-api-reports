package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.retail.retail_tg;

import jakarta.ws.rs.QueryParam;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.request.BaseRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.NonZeroNumericValue;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.NumericFixedDigits;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.RequiredText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateTimeText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.TextMaxLength;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * MA-047 閉栓作業票情報取得 API のリクエスト。
 */
public record LivalitClosingWorkTicketInfoGetRequest(
        @QueryParam("gasMtstNo")
        @RequiredText
        @NumericFixedDigits(digits = 11)
        @NonZeroNumericValue
        @Schema(description = "ガスメーター設置場所番号", pattern = "^(?!0{11})\\d{11}$")
        String gasMtstNo,
        @QueryParam("hsenKiboYmd")
        @RequiredText
        @Rfc3339DateText
        @Schema(description = "閉栓希望年月日", format = "date")
        String hsenKiboYmd,
        @QueryParam("syoKyoTrno")
        @RequiredText
        @NumericFixedDigits(digits = 11)
        @Schema(description = "使用者契約お客さま登録番号", pattern = "^\\d{11}$")
        String syoKyoTrno,
        @QueryParam("tksuSvkyNo")
        @RequiredText
        @NumericFixedDigits(digits = 11)
        @Schema(description = "託送サービス契約番号", pattern = "^\\d{11}$")
        String tksuSvkyNo,
        @QueryParam("hukkSysShbt")
        @TextMaxLength(max = 1)
        @Schema(description = "報告システム種別", maxLength = 1)
        String hukkSysShbt,
        @QueryParam("systemDatetime")
        @Rfc3339DateTimeText
        @Schema(description = "システム日時 (RFC3339)", format = "date-time")
        String systemDatetime) implements BaseRequest {

    /**
     * ガスメーター設置場所番号を返します。
     *
     * @return ガスメーター設置場所番号
     */
    public String getGasMtstNo() {
        return gasMtstNo;
    }

    /**
     * 閉栓希望年月日を返します。
     *
     * @return 閉栓希望年月日
     */
    public String getHsenKiboYmd() {
        return hsenKiboYmd;
    }

    /**
     * 使用者契約お客さま登録番号を返します。
     *
     * @return 使用者契約お客さま登録番号
     */
    public String getSyoKyoTrno() {
        return syoKyoTrno;
    }

    /**
     * 託送サービス契約番号を返します。
     *
     * @return 託送サービス契約番号
     */
    public String getTksuSvkyNo() {
        return tksuSvkyNo;
    }

    /**
     * 報告システム種別を返します。
     *
     * @return 報告システム種別
     */
    public String getHukkSysShbt() {
        return hukkSysShbt;
    }

    /**
     * システム日時を返します。
     *
     * @return システム日時
     */
    public String getSystemDatetime() {
        return systemDatetime;
    }
}
