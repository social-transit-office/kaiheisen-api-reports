package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_other.config;

import com.fasterxml.jackson.databind.ObjectMapper;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.inject.Inject;
import java.util.List;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_other.common.retail_common.retail_division.RetailDivisionResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_other.retail.retail_tg.LivalitClosingWorkTicketInfoGetResource;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.CdiBeanInitializer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupJsonWarmup;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupContributor;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupStep;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.tskgmst.TskgmstDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;

/** その他MAの起動時ウォームアップ定義。 */
@ApplicationScoped
public class PodOtherStartupWarmupContributor implements StartupWarmupContributor {

    private static final long WARMUP_GAS_MTST_NO = 99_999_999_999L;

    private static final List<String> WARMUP_TKSU_SVKK_JTKB_LIST = List.of("98", "99");

    private final CdiBeanInitializer beanInitializer;

    private final ObjectMapper objectMapper;

    private final Sysdummy1DataAccess databaseWarmupDataAccess;

    private final TskgmstDataAccess businessDataAccess;

    /**
     * コンストラクタ。
     *
     * @param beanInitializer CDI Bean初期化処理
     * @param objectMapper アプリケーション共通のJSON変換処理
     * @param databaseWarmupDataAccess DBウォームアップDA
     * @param businessDataAccess 業務参照DA
     */
    @Inject
    public PodOtherStartupWarmupContributor(
            final CdiBeanInitializer beanInitializer,
            final ObjectMapper objectMapper,
            final Sysdummy1DataAccess databaseWarmupDataAccess,
            final TskgmstDataAccess businessDataAccess) {
        this.beanInitializer = beanInitializer;
        this.objectMapper = objectMapper;
        this.databaseWarmupDataAccess = databaseWarmupDataAccess;
        this.businessDataAccess = businessDataAccess;
    }

    @Override
    public String applicationName() {
        return "ma-other";
    }

    @Override
    public LogLayer logLayer() {
        return LogLayer.MA;
    }

    @Override
    public void initializeResources() {
        beanInitializer.initialize(LivalitClosingWorkTicketInfoGetResource.class, RetailDivisionResource.class);
        StartupJsonWarmup.initialize(objectMapper);
    }

    @Override
    public int warmUpDatabase() {
        return databaseWarmupDataAccess.fetchWarmupValue().value();
    }

    @Override
    public List<StartupWarmupStep> businessDataAccessWarmupSteps() {
        return List.of(new StartupWarmupStep(
                TskgmstDataAccess.Identifiers.FIND_BY_GAS_MTST_NO_AND_TKSU_SVKK_JTKB_LIST,
                () -> {
                    if (businessDataAccess
                            .findByGasMtstNoAndTksuSvkkJtkbList(
                                    WARMUP_GAS_MTST_NO, WARMUP_TKSU_SVKK_JTKB_LIST)
                            .isPresent()) {
                        throw new IllegalStateException(
                                "Startup business DA warmup dummy value collision");
                    }
                }));
    }
}
