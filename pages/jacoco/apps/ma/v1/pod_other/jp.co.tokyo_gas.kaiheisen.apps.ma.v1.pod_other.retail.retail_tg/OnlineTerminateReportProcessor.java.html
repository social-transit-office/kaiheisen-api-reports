    /**
     * オンライン解約報告の業務処理を実行する。
     *
     * @param request オンライン解約報告リクエスト
     */
    public void process(OnlineTerminateReportRequest request) {
        LocalDate tkysYmd = LocalDate.parse(request.gmtsetkInf().tkysYmd());
        LocalDate kykYmd = LocalDate.parse(request.kykYmd());
        LocalDateTime systemDatetime = requestContext
                .getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("RequestContext.systemDatetime is required"));
        Long gasMtstNo = request.gmtsetkInf().gasMtstNo();
        Long svkyNo = request.svckyaInf().svkyNo();
        Integer skno = request.svckyaInf().skno();

        GmtsetkRecord gmtsetk = gmtsetkDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, systemDatetime.toLocalDate())
                .orElseThrow(() -> dataInconsistency(
                        TABLE_GMTSETK,
                        GmtsetkDataAccess.Identifiers.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD,
                        "{\"gasMtstNo\":" + gasMtstNo
                                + ",\"knskKjnYmd\":\"" + systemDatetime.toLocalDate() + "\"}"));

        executeOpeningClosingInProgressErrorJudgment(gmtsetk);

        OriginalMeterCancellationDedicatedContractCheckOutput originalMeterCheckOutput =
                originalMeterCancellationDedicatedContractCheck.execute(
                        gmtsetk.ckrtGmscJtkb(),
                        gasMtstNo,
                        tkysYmd);

        SvsihkrRecord svsihkr = svsihkrDataAccess
                .findBySvkyNoAndTkysYmdAndTyenYmdAndNotSihKiyJtkb(
                        svkyNo,
                        systemDatetime.toLocalDate(),
                        ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID)
                .orElseThrow(() -> dataInconsistency(
                        TABLE_SVSIHKR,
                        "DA-044-R007",
                        "{\"svkyNo\":" + svkyNo
                                + ",\"knskKjnYmd\":\"" + systemDatetime.toLocalDate()
                                + "\",\"sihKiyJtkb\":\""
                                + ConstantLookupKeys.Cirius.PaymentContractStatus.INVALID
                                + "\"}"));

        SihakyRecord sihaky = sihakyDataAccess
                .findBySihKiyNoAndSihKiyJtkb(
                        svsihkr.sihKiyNo(),
                        ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING)
                .orElseThrow(() -> dataInconsistency(
                        TABLE_SIHAKY,
                        "DA-075-R001",
                        "{\"sihKiyNo\":" + svsihkr.sihKiyNo()
                                + ",\"sihKiyJtkb\":\""
                                + ConstantLookupKeys.Cirius.PaymentContractStatus.APPLYING
                                + "\"}"));

        ComprehensiveJudgmentProcessOutput comprehensiveJudgmentOutput =
                comprehensiveJudgmentProcess.execute(
                        gasMtstNo,
                        systemDatetime.toLocalDate(),
                        ConstantLookupKeys.Kaiheisen.ComprehensiveJudgmentProcessing
                                .OVERALL_JUDGMENT_RELATED_END);

        if (Boolean.TRUE.equals(comprehensiveJudgmentOutput.suguHntFlg())) {
            throw new BusinessRuleException(
                    ErrorCodeConstants.ERROR_CODE_020000036,
                    List.of(),
                    List.of(formatGasMeterLocationNumber(gasMtstNo)));
        }

        SvckyaRecord svckya = svckyaDataAccess
                .findBySvkyNoAndSkno(svkyNo, skno)
                .orElseThrow(() -> dataInconsistency(
                        TABLE_SVCKYA,
                        "DA-038-R010",
                        "{\"svkyNo\":" + svkyNo + ",\"skno\":" + skno + "}"));

        SvmstkrRecord relatedSvmstkr = svmstkrDataAccess
                .findByMtstNoAndRkkiKnrnJtkb(
                        gasMtstNo,
                        List.of(
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID,
                                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03))
                .orElseThrow(() -> dataInconsistency(
                        TABLE_SVMSTKR,
                        "DA-041-R009",
                        "{\"mtstNo\":" + gasMtstNo
                                + ",\"rkkiKnrnJtkbList\":[\""
                                + ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID
                                + "\",\""
                                + ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.BLANK_03
                                + "\"]}"));

        GasFeeContractTypeSortingContentAcquisitionOutput gasFeeContractTypeOutput =
                gasFeeContractTypeSortingContentAcquisition.execute(
                        svckya.kiyShbt(),
                        List.of(new GrkKsbHwsb(
                                ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionType
                                        .POSSIBILITY_OF_ONLINE_CANCELLATION,
                                null)),
                        systemDatetime.toLocalDate());

        boolean onlineCancellationAvailable = gasFeeContractTypeOutput.grkKsbHnsbList().stream()
                .map(value -> value.grkKsbHnsb())
                .anyMatch(ConstantLookupKeys.Cirius.GasFeeContractTypeDistributionContentType
                        .POSSIBILITY_OF_ONLINE_CANCELLATION_POSSIBLE::equals);
        if (!onlineCancellationAvailable) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000037);
        }

        if (svckyaDataAccess.findFirstBySvkyNoAndSvkyJtkb(
                svkyNo,
                ConstantLookupKeys.Cirius.ServiceContractStatus.WAITING_FOR_CONTRACT_APPLICATION)
                .isPresent()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000038);
        }

        TskgmstRecord tskgmst = tskgmstDataAccess
                .findByGasMtstNoAndTksuSvkkJtkb(
                        gasMtstNo,
                        ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING)
                .orElseThrow(() -> dataInconsistency(
                        TABLE_TSKGMST,
                        "DA-108-R002",
                        "{\"gasMtstNo\":" + gasMtstNo
                                + ",\"tksuSvkkJtkb\":\""
                                + ConstantLookupKeys.Cirius.TransportationServiceContractRelatedStatus.APPLYING
                                + "\"}"));

        TkssvkyRecord tkssvky = tkssvkyDataAccess
                .findAllByTksuSvkyNoAndTksuSvkyJtkb(
                        tskgmst.tksuSvkyNo(),
                        ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT)
                .stream()
                .sorted(Comparator
                        .comparing(
                                TkssvkyRecord::tkysYmd,
                                Comparator.nullsLast(Comparator.reverseOrder()))
                        .thenComparing(
                                TkssvkyRecord::skno,
                                Comparator.nullsLast(Comparator.reverseOrder())))
                .findFirst()
                .orElseThrow(() -> dataInconsistency(
                        TABLE_TKSSVKY,
                        "DA-099-R001",
                        "{\"tksuSvkyNo\":" + tskgmst.tksuSvkyNo()
                                + ",\"tksuSvkyJtkb\":\""
                                + ConstantLookupKeys.Cirius.TransportationServiceContractStatus.UNDER_CONTRACT
                                + "\"}"));

        KnsnkrnRecord knsnkrn = knsnkrnDataAccess
                .findByGasMtstNo(gasMtstNo)
                .orElseThrow(() -> dataInconsistency(
                        TABLE_KNSNKRN,
                        "DA-063-R001",
                        "{\"gasMtstNo\":" + gasMtstNo + "}"));

        ConstantMasterInfoAcquisitionOutput contractTypeConstant =
                constantMasterInfoAcquisition.execute(
                        ConstantLookupKeys.Cirius.ContractType.TYPE,
                        svckya.kiyShbt());
        ConstantMasterInfoAcquisitionOutput transportationContractTypeConstant =
                constantMasterInfoAcquisition.execute(
                        ConstantLookupKeys.Cirius.ConsignmentContractType.TYPE,
                        tkssvky.tkykShbt());

        if (!ConstantLookupKeys.Cirius.ClosureReasonType.SECURITY_CLOSURE.equals(gmtsetk.hesnJysb())) {
            supplyStopUpdateAndMeterReadingResultRegistrationAtCancellation.execute(
                    gasMtstNo,
                    svckya.kkysSisnJtkb(),
                    new SupplyStopUpdateAndMeterReadingResultRegistrationAtCancellationGmtsetkInf(
                            gmtsetk.gmtKhsJtkb(),
                            gmtsetk.hesnYmd(),
                            gmtsetk.ckrtGmscJtkb(),
                            gmtsetk.mtgMtstNo()),
                    knsnkrn.knsHuhuKbn(),
                    tkssvky.tksuSvkyNo());
        }

        TransportServiceContractTerminationReasonTypeAcquisitionOutput transportTerminationReason =
                transportServiceContractTerminationReasonTypeAcquisition.execute(request.svkyEndJysb());

        relocationDestinationManagementAdditionAtRelocationCancellation.execute(
                request.svkyEndJysb(),
                gasMtstNo,
                YearMonth.of(kykYmd.getYear(), kykYmd.getMonthValue()),
                svkyNo,
                svsihkr.sihKiyNo());

        cancellationCommonProcessCommon.execute(
                gasMtstNo,
                kykYmd,
                gmtsetk.hesnJysb(),
                transportTerminationReason.tksuSvkeJysb(),
                request.svkyEndJysb(),
                toCancellationGmtsetdInf(gmtsetk, originalMeterCheckOutput),
                toCancellationTskgmstInf(tskgmst),
                toCancellationTkssvkyInf(tkssvky),
                toCancellationSvmstkrInf(relatedSvmstkr),
                toCancellationSvckyaInf(svckya),
                toCancellationSvsihkrInf(svsihkr),
                toCancellationSihakyInf(sihaky),
                toCancellationSnyInfList(originalMeterCheckOutput),
                List.of(),
                ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail
                        .ONLINE_CANCELLATION_REPORT,
                request.hukkUsrId(),
                request.hukkUnmKj(),
                ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute
                        .ONLINE_REPORTING_USUALLY);

        TransferWorkHistoryDetailCodePipelineAcquisitionOutput conduitDetailCode =
                transferWorkHistoryDetailCodePipelineAcquisition.execute(request.svkyEndJysb());
        doukanAdapter.register(toDoukanRequest(
                request,
                systemDatetime,
                kykYmd,
                gmtsetk.hesnYmd(),
                conduitDetailCode.duknIsrsCd(),
                transportationContractTypeConstant.cntn()));

        TransferWorkHistoryDetailCodeRetailAcquisition2Output retailDetailCode =
                transferWorkHistoryDetailCodeRetailAcquisition2.execute(request.svkyEndJysb());
        kouriAdapter.register(toKouriRequest(
                request,
                systemDatetime,
                kykYmd,
                gmtsetk.hesnYmd(),
                retailDetailCode.isrrGmssCd(),
                contractTypeConstant.cntn()));

        String kkysSisnJtkb = ConstantLookupKeys.Cirius.SupplyStopPaymentStatus
                .SUPPLY_STOP_PAYMENT_PENDING.equals(svckya.kkysSisnJtkb())
                        ? ""
                        : ConstantLookupKeys.Cirius.SupplyStopPaymentStatus.SUPPLY_OUTAGE;

    private String formatGasMeterLocationNumber(Long gasMtstNo) {
        String digits = String.format("%011d", gasMtstNo);
        return digits.substring(0, 4) + "-" + digits.substring(4, 7) + "-" + digits.substring(7);
    }

    private int toHms(LocalDateTime systemDatetime) {
        return systemDatetime.getHour() * 10000
                + systemDatetime.getMinute() * 100
                + systemDatetime.getSecond();
    }

    private String toTimeText(LocalDateTime systemDatetime) {
        return String.format(
                "%02d%02d%02d",
                systemDatetime.getHour(),
                systemDatetime.getMinute(),
                systemDatetime.getSecond());
    }

    private String toDateText(LocalDate value) {
        return value == null ? "" : DATE_FORMATTER.format(value);
    }

    private String toSlashDateText(LocalDate value) {
        return value == null ? "" : SLASH_DATE_FORMATTER.format(value);
    }

    private String blankToEmpty(String value) {
        return value == null ? "" : value;
    }
