        return execute(request, () -> {
            Long gasMtstNoValue = Long.valueOf(request.gasMtstNo().trim());
            LocalDate knskKjnYmdValue = Rfc3339LocalDateParser.parse(request.knskKjnYmd());
            OpeningWorkLocationInfoAcquisitionOutput output =
                    openingWorkLocationInfoAcquisition.execute(gasMtstNoValue, knskKjnYmdValue);
            ServiceContractSupplementalInfo supplementalInfo = svmstkrDataAccess
                    .findByMtstNoAndRkkiSttYmdAndRkkiKnrnJtkb(
                            gasMtstNoValue,
                            knskKjnYmdValue,
                            invalidChargeCalculationRelatedStatus)
                    .map(record -> findServiceContractSupplementalInfo(record.svkyNo(), knskKjnYmdValue))
                    .orElseGet(ServiceContractSupplementalInfo::empty);
            return toResponse(
                    gasMtstNoValue,
                    output,
                    supplementalInfo,
                    requestContext.getCorrelationId().orElseThrow(
                            () -> new IllegalStateException(
                                    "correlationId is not set in RequestContext")));
        });
    }

    private ServiceContractSupplementalInfo findServiceContractSupplementalInfo(Long svkyNo, LocalDate knskKjnYmd) {
        SvckyaRecord svckyaRecord = svckyaDataAccess
                .findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNotOrderByRkkiSttYmd(
                        svkyNo,
                        knskKjnYmd,
                        invalidServiceContractStatus)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        svckyaDataAccess.tableName(),
                        SvckyaDataAccess.CallSpecs
                                .FIND_LATEST_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_NOT_ORDER_BY_RKKI_STT_YMD
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("svkyNo", svkyNo)
                                .put("knskKjnYmd", knskKjnYmd)
                                .put("svkyJtkb", invalidServiceContractStatus)));
        return new ServiceContractSupplementalInfo(
                svckyaRecord.kiyShbt(),
                svckyaRecord.svkyEndJysb(),
                svckyaRecord.kkysSisnJtkb());
    }
