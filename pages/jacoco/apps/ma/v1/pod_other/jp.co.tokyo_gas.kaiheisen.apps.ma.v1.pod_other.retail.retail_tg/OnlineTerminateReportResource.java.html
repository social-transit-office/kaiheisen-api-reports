package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_other.retail.retail_tg;

import jakarta.transaction.Transactional;
import jakarta.transaction.Transactional.TxType;
import jakarta.validation.Valid;
import jakarta.ws.rs.Consumes;
import jakarta.ws.rs.POST;
import jakarta.ws.rs.Path;

import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource.BaseResource;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.core.resource.MainService;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.response.CommonErrorResponse;

/**
 * MA-034 オンライン解約報告 API のリソース実装。
 */
@Path(OnlineTerminateReportEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-034", name = "オンライン解約報告-小売-TG小売")
@Tag(name = "オンライン解約報告", description = "MA-034 オンライン解約報告 API")
public class OnlineTerminateReportResource extends BaseResource<OnlineTerminateReportRequest> {

    private static final String CORRELATION_ID_PATTERN =
            "^([0-9a-fA-F]{8}-[0-9a-fA-F]{4}-4[0-9a-fA-F]{3}-[89aAbB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}|[A-Za-z0-9_-]{1,20})$";

    private final OnlineTerminateReportProcessor processor;

    /**
     * コンストラクタ。
     */
    @Inject
    public OnlineTerminateReportResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            OnlineTerminateReportProcessor processor) {
        super(
                Objects.requireNonNull(requestContext, "requestContext"),
                Objects.requireNonNull(requestDtoValidator, "requestDtoValidator"),
                Objects.requireNonNull(clock, "clock"));
        this.processor = Objects.requireNonNull(processor, "processor");
    }

    /**
     * MA-034 を実行する。
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "オンライン解約報告",
            description = "オンライン解約報告の登録処理を実行する。",
            operationId = "postOnlineTerminateReport")
    @Parameter(
            name = CorrelationIdFilter.HEADER_NAME,
            in = ParameterIn.HEADER,
            description = "相関ID。未指定時はサーバで採番する。",
            schema = @Schema(
                    type = SchemaType.STRING,
                    maxLength = 36,
                    pattern = CORRELATION_ID_PATTERN))
    @APIResponse(
            responseCode = "200",
            description = "オンライン解約報告完了",
            headers = @Header(
                    name = CorrelationIdFilter.HEADER_NAME,
                    description = "相関ID",
                    schema = @Schema(
                            type = SchemaType.STRING,
                            maxLength = 36,
                            pattern = CORRELATION_ID_PATTERN)),
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OnlineTerminateReportResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            headers = @Header(
                    name = CorrelationIdFilter.HEADER_NAME,
                    description = "相関ID",
                    schema = @Schema(
                            type = SchemaType.STRING,
                            maxLength = 36,
                            pattern = CORRELATION_ID_PATTERN)),
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            headers = @Header(
                    name = CorrelationIdFilter.HEADER_NAME,
                    description = "相関ID",
                    schema = @Schema(
                            type = SchemaType.STRING,
                            maxLength = 36,
                            pattern = CORRELATION_ID_PATTERN)),
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OnlineTerminateReportResponse endpoint(@Valid OnlineTerminateReportRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        return execute(request, () -> handle(request));
    }

    OnlineTerminateReportResponse handle(OnlineTerminateReportRequest request) {
        processor.process(request);
        return new OnlineTerminateReportResponse(requestContext.getCorrelationId().orElse(null));
    }

    private void bindCorrelationIdFromBodyIfPresent(OnlineTerminateReportRequest request) {
        if (request == null) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId != null) {
            requestContext.setCorrelationId(CorrelationIdResolver.resolve(correlationId));
        }
    }
}
