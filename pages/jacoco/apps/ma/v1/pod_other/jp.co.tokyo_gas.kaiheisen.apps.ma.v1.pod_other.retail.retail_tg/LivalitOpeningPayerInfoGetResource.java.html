    @Override
    protected InterFieldValidation<LivalitOpeningPayerInfoGetRequest> interFieldValidation() {
        return request -> {
            List<ValidationError> errors = new ArrayList<>();
            if (request == null || request.shrsTelnoInf() == null) {
                return errors;
            }

            Integer telnoCnt = request.shrsTelnoInf().telnoCnt();
            List<LivalitOpeningPayerInfoGetPayerTelephoneEntry> entries =
                    request.shrsTelnoInf().telnoInf() == null ? List.of() : request.shrsTelnoInf().telnoInf();

            if (telnoCnt != null && telnoCnt > 2) {
                errors.add(ValidationError.ofField(ARRAY_ERROR_CODE, "shrsTelnoInf.telnoCnt"));
            }
            if (entries.size() > 2) {
                errors.add(ValidationError.ofField(ARRAY_ERROR_CODE, "shrsTelnoInf.telnoInf"));
            }
            if (!errors.isEmpty()) {
                return errors;
            }
            if (telnoCnt != null && telnoCnt != entries.size()) {
                errors.add(ValidationError.ofField(ARRAY_ERROR_CODE, "shrsTelnoInf.telnoCnt"));
                return errors;
            }
            for (int index = 0; index < entries.size(); index++) {
                LivalitOpeningPayerInfoGetPayerTelephoneEntry entry = entries.get(index);
                if (entry == null || normalizeOptional(entry.shsyTelNo()) == null) {
                    continue;
                }
                addRequiredTextError(errors, fieldName(index, "shsyTelShbt"), entry.shsyTelShbt());
                addRequiredTextError(errors, fieldName(index, "shsyTrrsShbt"), entry.shsyTrrsShbt());
                addRequiredTextError(errors, fieldName(index, "shsyTknzShbt"), entry.shsyTknzShbt());
                addRequiredTextError(errors, fieldName(index, "shsyTssdShbt"), entry.shsyTssdShbt());
                addRequiredTextError(errors, fieldName(index, "shsyTstkYmd"), entry.shsyTstkYmd());
                addRequiredTextError(errors, fieldName(index, "shsyTtsnKj"), entry.shsyTtsnKj());
                addRequiredTextError(errors, fieldName(index, "shsyTezgShbt"), entry.shsyTezgShbt());
            }
            return errors;
        };
    }

    LivalitOpeningPayerInfoGetResponse handle(LivalitOpeningPayerInfoGetRequest request) {
        LocalDate kasgYmd = parseRequestDate(request.kaisenUktkInf().kasgYmd());
        LocalDate kasHukkYmd = parseRequestDate(request.kaisenUktkInf().kasHukkYmd());
        String gmb = request.kaisenUktkInf().gmb();
        String sgyAnknNo = request.kaisenUktkInf().sgyAnknNo();

        kasnhkDataAccess.deleteByGmbAndKasgYmdAndKasHukkYmdAndLvltAnknNo(
                gmb,
                kasgYmd,
                kasHukkYmd,
                sgyAnknNo,
                null);

        ResolvedKanjiFlags resolvedKanjiFlags = resolveKanjiFlags(request);
        String buildingNameKana = normalizeOptional(buildingInfo(request).sbdnKn());
        HousingCategoryCodeAcquisitionOutput housingCategory =
                housingCategoryCodeAcquisition.execute(buildingNameKana);

        String resolvedHukkSysShbt = normalizeOptional(request.hukkSysShbt());
        if (resolvedHukkSysShbt == null) {
            resolvedHukkSysShbt = ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT;
        }

        kasnhkDataAccess.insert(
                toKasnhkWriteRecord(
                        request,
                        kasgYmd,
                        kasHukkYmd,
                        resolvedKanjiFlags,
                        housingCategory.jutKbn()),
                AuditContext.of(UPD_USER_ID, decodeFunctionId(resolvedHukkSysShbt)));
        return new LivalitOpeningPayerInfoGetResponse(resolveResponseCorrelationId());
    }

        Boolean sonKjhkFlg = request.shrsCustmrInf().sonKjhkFlg();

            return new ResolvedKanjiFlags(sonKjhkFlg, sbdnKjhkFlg);

            request.shrsCustmrInf().sonKj1(),

                sonKjhkFlg != null ? sonKjhkFlg : output.khFlg1(),
                sbdnKjhkFlg != null ? sbdnKjhkFlg : output.khFlg2());
    }

    private PhoneSlot phoneSlot(LivalitOpeningPayerInfoGetRequest request, int index) {
        if (request.shrsTelnoInf() == null || request.shrsTelnoInf().telnoInf() == null) {
            return PhoneSlot.empty();
        }
        List<LivalitOpeningPayerInfoGetPayerTelephoneEntry> entries = request.shrsTelnoInf().telnoInf();
        if (index >= entries.size() || entries.get(index) == null) {
            return PhoneSlot.empty();
        }
        LivalitOpeningPayerInfoGetPayerTelephoneEntry entry = entries.get(index);
        String telNo = normalizeOptional(entry.shsyTelNo());
        if (telNo == null) {
            return PhoneSlot.empty();
        }
        return new PhoneSlot(
                telNo,
                normalizeOptional(entry.shsyTelShbt()),
                normalizeOptional(entry.shsyTrrsShbt()),
                normalizeOptional(entry.shsyTknzShbt()),
                normalizeOptional(entry.shsyTssdShbt()),
                parseOptionalRequestDate(entry.shsyTstkYmd()),
                normalizeOptional(entry.shsyTtsnKj()),
                normalizeOptional(entry.shsyTezgShbt()),
                normalizeOptional(entry.shsyTemdKbn()));
    }

    private String decodeFunctionId(String hukkSysShbt) {
        String normalized = normalizeOptional(hukkSysShbt);
        if (normalized == null) {
            return null;
        }
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                normalized);
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }

    private LocalDate parseRequestDate(String rawDate) {
        String normalized = Objects.requireNonNull(normalizeOptional(rawDate), "rawDate");
        try {
            return LocalDate.parse(normalized);
        } catch (DateTimeParseException ignored) {
            try {
                return OffsetDateTime.parse(normalized).toLocalDate();
            } catch (DateTimeParseException ex) {
                throw new IllegalArgumentException("date must follow RFC3339", ex);
            }
        }
    }

    private LocalDate parseOptionalRequestDate(String rawDate) {
        String normalized = normalizeOptional(rawDate);
        return normalized == null ? null : parseRequestDate(normalized);
    }

    private static void addRequiredTextError(List<ValidationError> errors, String field, String value) {
        if (value == null || value.isBlank()) {
            errors.add(ValidationError.ofField(REQUIRED_ERROR_CODE, field));
        }
    }

    private static String fieldName(int index, String field) {
        return "shrsTelnoInf.telnoInf[" + index + "]." + field;
