    /**
     * 閉栓作業票情報を組み立てて返却する。
     *
     * @param request リクエスト DTO
     * @return 閉栓作業票情報レスポンス
     */
    LivalitClosingWorkTicketInfoGetResponse handle(LivalitClosingWorkTicketInfoGetRequest request) {
        Long gasMtstNo = parseLong(request.gasMtstNo());
        Long kyoTrkNo = parseLong(request.syoKyoTrno());
        Long tksuSvkyNo = parseLong(request.tksuSvkyNo());
        LocalDate knskKjnYmd = parseDate(request.hsenKiboYmd());
        String correlationId =
            requestContext
                .getCorrelationId()
                .orElseThrow(
                    () ->
                        new IllegalStateException(
                            "correlationId is not set in RequestContext"));
        GmtsetdRecord gmtsetdRecord =
            findGmtsetd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> new DataNotFoundException(gmtsetdDataAccess.tableName()));
        SeriesOriginMeterLocationNumberAcquisitionOutput seriesOriginOutput =
            seriesOriginMeterLocationNumberAcquisition.execute(
                gasMtstNo,
                gmtsetdRecord.mtgMtstNo(),
                gmtsetdRecord.gmtKhsJtkb(),
                gmtsetdRecord.ckrtGmscJtkb());
        GasmtrkRecord gasmtrkRecord =
            findGasMeter(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> new DataNotFoundException(gasmtrkDataAccess.tableName()));
        KnsytiRecord meterReadingSchedule =
            findMeterReadingSchedule(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> new DataNotFoundException(knsytiDataAccess.tableName()));
        Long svkyNo =
            resolveServiceContractNo(
                gasMtstNo, seriesOriginOutput.gasMtstNo(), gmtsetdRecord, knskKjnYmd);
        SvckyaRecord svckyaRecord =
            findServiceContract(svkyNo, knskKjnYmd)
                .orElseThrow(() -> new DataNotFoundException(svckyaDataAccess.tableName()));
        SvsihkrRecord svsihkrRecord =
            findPaymentContract(svkyNo)
                .orElseThrow(() -> new DataNotFoundException(svsihkrDataAccess.tableName()));
        KiycstkRecord customer =
            findCustomer(kyoTrkNo)
                .orElseThrow(() -> new DataNotFoundException(kiycstkDataAccess.tableName()));
        MeterReadingSupportInfoAcquisitionOutput meterReadingSupport =
            gasMtstNo == null ? null : meterReadingSupportInfoAcquisition.execute(gasMtstNo);
        RecentThreeMeterReadingResultsAcquisitionOutput recentThreeMeterReadingResults =
            buildRecentThreeMeterReadingResults(gasMtstNo, knskKjnYmd);
        OneYearAgoMeterReadingResultAcquisitionOutput oneYearAgoMeterReadingResult =
            buildOneYearAgoMeterReadingResult(gasMtstNo, knskKjnYmd, tksuSvkyNo);
        RelatedDedicatedMeterInfoAcquisitionOutput relatedDedicatedMeterInfo =
            buildRelatedDedicatedMeterInfo(gasMtstNo, gmtsetdRecord, knskKjnYmd);

        return new LivalitClosingWorkTicketInfoGetResponse(
            MaResponseNumericValues.orZero(svkyNo),
            MaResponseNumericValues.orZero(svsihkrRecord.sihKiyNo()),
            MaResponseNumericValues.orZero(kyoTrkNo),
            formatDate(gmtsetdRecord.hesnYmd()),
            formatDate(gmtsetdRecord.kasYmd()),
            formatYearMonth(gmtsetdRecord.snstGkjYm()),
            gmtsetdRecord.gmicDbriShbt(),
            gmtsetdRecord.knsKhFlg(),
            gmtsetdRecord.gmicSbriShbt(),
            gmtsetdRecord.gmtKhsJtkb(),
            gmtsetdRecord.ckrtGmscJtkb(),
            gmtsetdRecord.kkyuGmtKbn(),
            gmtsetdRecord.kngmStbJtkb(),
            meterReadingSchedule.knsHuhuKbn(),
            gmtsetdRecord.hesnSdnShbt(),
            meterReadingSupport == null ? null : meterReadingSupport.gmtItiInf(),
            gasmtrkRecord.gmb(),
            gasmtrkRecord.gmkShbt(),
            MaResponseNumericValues.oneDecimalOrZero(gasmtrkRecord.gmtGusu()),
            MaResponseNumericValues.orZero(gasmtrkRecord.sjskHticSu()),
            MaResponseNumericValues.oneDecimalOrZero(gasmtrkRecord.sjsuMincSu()),
            formatYearMonth(gasmtrkRecord.ktyYm()),
            svckyaRecord.kiyShbt(),
            gmtsetdRecord.kasJysb(),
            customer.ctnmKn1(),
            customer.ctnmKj1(),
            customer.ctnmKn2(),
            customer.ctnmKj2(),
            customer.ctnm2HujnShbt(),
            customer.ctnm2HshiShbt(),
            customer.ctnm1HujnShbt(),
            customer.ctnm1HshiShbt(),
            customer.ctnmKjhkFlg(),
            toMeterReadingInfoResponses(recentThreeMeterReadingResults),
            toGasUsageFluctuationCheckResponse(oneYearAgoMeterReadingResult),
            buildMeterReadingSupportList(meterReadingSupport),
            toRelatedDedicatedMeterInfoResponses(relatedDedicatedMeterInfo),
            correlationId);
    }

    /**
     * 指定日時点の検針予定を取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検針基準日
     * @return 検針予定
     */
    private Optional<KnsytiRecord> findMeterReadingSchedule(Long gasMtstNo, LocalDate knskKjnYmd) {
        if (gasMtstNo == null || knskKjnYmd == null) {
            return Optional.empty();
        }
        return knsytiDataAccess
            .findAllByGasMtstNoAndTkysYmAndTyenYm(
                gasMtstNo, YearMonth.of(knskKjnYmd.getYear(), knskKjnYmd.getMonthValue()))
            .stream()
            .findFirst();
    }

    /**

     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検針基準日
     * @return ガスメーター設置場所＜導管＞
     */
    private Optional<GmtsetdRecord> findGmtsetd(Long gasMtstNo, LocalDate knskKjnYmd) {
        if (gasMtstNo == null || knskKjnYmd == null) {
            return Optional.empty();
        }
        return gmtsetdDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd);
    }

    /**
     * 指定日時点のガスメーター情報を取得する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param knskKjnYmd 検針基準日
     * @return ガスメーター情報
     */
    private Optional<GasmtrkRecord> findGasMeter(Long gasMtstNo, LocalDate knskKjnYmd) {
        if (gasMtstNo == null || knskKjnYmd == null) {
            return Optional.empty();
        }
        return gasmtrkDataAccess
            .findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(gasMtstNo, knskKjnYmd)
            .stream()
            .findFirst();
    }

    /**
     * サービス契約番号を導出する。
     *
     * @param gasMtstNo ガスメーター設置場所番号
     * @param seriesOriginGasMtstNo 直列元のガスメーター設置場所番号
     * @param gmtsetdRecord ガスメーター設置場所＜導管＞
     * @param knskKjnYmd 検針基準日
     * @return サービス契約番号
     */
    private Long resolveServiceContractNo(
        Long gasMtstNo,
        Long seriesOriginGasMtstNo,
        GmtsetdRecord gmtsetdRecord,
        LocalDate knskKjnYmd) {
        if (gasMtstNo == null
            || gmtsetdRecord == null
            || knskKjnYmd == null) {
            return null;
        }
        return serviceContractNumberAcquisition
            .execute(
                gasMtstNo,
                seriesOriginGasMtstNo,
                ServiceContractNumberAcquisitionConstants.ShrKbn.ACTIVE_OR_END,
                gmtsetdRecord.ckrtGmscJtkb(),
                knskKjnYmd)
            .svkyNo();
    }

    /**
     * 有効なサービス契約を取得する。
     *
     * @param svkyNo サービス契約番号
     * @param knskKjnYmd 検針基準日
     * @return サービス契約
     */
    private Optional<SvckyaRecord> findServiceContract(Long svkyNo, LocalDate knskKjnYmd) {
        if (svkyNo == null || knskKjnYmd == null) {
            return Optional.empty();
        }
        return svckyaDataAccess.findLatestBySvkyNoAtKnskKjnYmdAndSvkyJtkbNot(
            svkyNo, knskKjnYmd, invalidServiceContractStatus);
    }

    /**
     * 適用中の支払契約を取得する。
     *
     * @param svkyNo サービス契約番号
     * @return 支払契約
     */
    private Optional<SvsihkrRecord> findPaymentContract(Long svkyNo) {
        if (svkyNo == null) {
            return Optional.empty();
        }
        return svsihkrDataAccess.findBySvkyNoAndSihKiyJtkb(
            svkyNo, applicablePaymentContractStatus);
    }

    /**
     * 契約お客さま情報を取得する。
     *
     * @param kyoTrkNo 契約お客さま番号
     * @return 契約お客さま情報
     */
    private Optional<KiycstkRecord> findCustomer(Long kyoTrkNo) {
        if (kyoTrkNo == null) {
            return Optional.empty();
        }
        return kiycstkDataAccess.findByKyoTrkNo(kyoTrkNo);
    }

    /**

    private RelatedDedicatedMeterInfoAcquisitionOutput buildRelatedDedicatedMeterInfo(
        Long gasMtstNo,

        LocalDate knskKjnYmd) {
        if (gasMtstNo == null
                || gmtsetdRecord == null
                || knskKjnYmd == null) {
            return new RelatedDedicatedMeterInfoAcquisitionOutput(List.of(), List.of());

        return relatedDedicatedMeterInfoAcquisition.execute(
            gasMtstNo,
            gmtsetdRecord.mtgMtstNo(),
            gmtsetdRecord.ckrtGmscJtkb(),
            knskKjnYmd,
            gmtsetdRecord.szcGkkCd(),
            gmtsetdRecord.szcCmazCd(),
            gmtsetdRecord.szcMchCd());
    }

    /**
     * 関連専用メーター情報取得結果をレスポンス DTO 一覧へ変換する。
     *
     * @param output 関連専用メーター情報取得結果
     * @return 関連専用メーター情報レスポンス一覧
     */
    private List<RelatedDedicatedMeterInfoResponse> toRelatedDedicatedMeterInfoResponses(
        RelatedDedicatedMeterInfoAcquisitionOutput output) {
        if (output == null || output.gasMtstNo() == null || output.gmb() == null) {
            return List.of();
        }
        int size = Math.min(output.gasMtstNo().size(), output.gmb().size());
        ArrayList<RelatedDedicatedMeterInfoResponse> responses = new ArrayList<>(size);
        for (int index = 0; index < size; index++) {
            responses.add(
                new RelatedDedicatedMeterInfoResponse(
                    MaResponseNumericValues.orZero(output.gasMtstNo().get(index)),
                    output.gmb().get(index)));
        }
        return List.copyOf(responses);
    }

    /**
     * 文字列を {@link Long} へ変換する。
     *
     * @param value 変換対象文字列
     * @return 数値。空文字または {@code null} の場合は {@code null}
     */
    private Long parseLong(String value) {
        String normalized = normalize(value);
        if (normalized == null) {
            return null;
        }
        return Long.valueOf(normalized);
    }

    /**
     * 文字列を {@link LocalDate} へ変換する。
     *
     * @param value 変換対象文字列
     * @return 日付。空文字または {@code null} の場合は {@code null}
     */
    private LocalDate parseDate(String value) {
        String normalized = normalize(value);
        if (normalized == null) {
            return null;
        }
        return LocalDate.parse(normalized);
    }

    /**
     * 日付を ISO-8601 文字列へ変換する。
     *
     * @param value 変換対象日付
     * @return 日付文字列。{@code value} が {@code null} の場合は {@code null}
     */
    private String formatDate(LocalDate value) {
        return value == null ? null : value.toString();
    }

     * 年月を文字列へ変換する。
     *
     * @param value 変換対象年月
     * @return 年月文字列。{@code value} が {@code null} の場合は {@code null}
     */
    private String formatYearMonth(YearMonth value) {
        return value == null ? null : value.toString();
    }

    /**
     * 文字列を正規化する。
     *
     * @param value 正規化対象文字列
     * @return trim 後の文字列。空文字または {@code null} の場合は {@code null}
     */
    private String normalize(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }
