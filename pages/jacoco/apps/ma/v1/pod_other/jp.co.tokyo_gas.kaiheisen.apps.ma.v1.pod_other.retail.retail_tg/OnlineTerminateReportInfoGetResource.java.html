/**
 * MA-033 オンライン解約報告情報取得 API のリソース実装。
 */
@MainService(identifier = "MA-033", name = "オンライン解約報告情報取得-小売-TG小売")
@Path(OnlineTerminateReportInfoGetEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "オンライン解約報告情報取得", description = "MA-033 オンライン解約報告情報取得 API")
public class OnlineTerminateReportInfoGetResource extends BaseResource<OnlineTerminateReportInfoGetRequest> {

    private final MeterReadingResultAcquisitionForCancellation meterReadingResultAcquisitionForCancellation;

    private final SvmstkrDataAccess svmstkrDataAccess;

    private final SvckyaDataAccess svckyaDataAccess;

    private final GmtsetkDataAccess gmtsetkDataAccess;

    private final TaksmtrDataAccess taksmtrDataAccess;

    private final String relatedValidStatus;

    private final String applyingServiceContractStatus;

    private final String openingStatus;

    private final String disasterClosureReason;

    private final String temporaryRegistrationStatus;

    private final String switchApplicationCisCode;

    /**
     * コンストラクタ。
     */
    @Inject
    public OnlineTerminateReportInfoGetResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            MeterReadingResultAcquisitionForCancellation meterReadingResultAcquisitionForCancellation,
            SvmstkrDataAccess svmstkrDataAccess,
            SvckyaDataAccess svckyaDataAccess,
            GmtsetkDataAccess gmtsetkDataAccess,
            TaksmtrDataAccess taksmtrDataAccess,
            CiriusConstantService ciriusConstantService) {
        super(requestContext, requestDtoValidator, clock);
        this.meterReadingResultAcquisitionForCancellation = Objects.requireNonNull(
                meterReadingResultAcquisitionForCancellation,
                "meterReadingResultAcquisitionForCancellation");
        this.svmstkrDataAccess = Objects.requireNonNull(svmstkrDataAccess, "svmstkrDataAccess");
        this.svckyaDataAccess = Objects.requireNonNull(svckyaDataAccess, "svckyaDataAccess");
        this.gmtsetkDataAccess = Objects.requireNonNull(gmtsetkDataAccess, "gmtsetkDataAccess");
        this.taksmtrDataAccess = Objects.requireNonNull(taksmtrDataAccess, "taksmtrDataAccess");

        CiriusConstantService resolvedCiriusConstantService =
                Objects.requireNonNull(ciriusConstantService, "ciriusConstantService");
        this.relatedValidStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.TYPE,
                ConstantLookupKeys.Cirius.ChargeCalculationRelatedStatus.RELATED_VALID);
        this.applyingServiceContractStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ServiceContractStatus.TYPE,
                ConstantLookupKeys.Cirius.ServiceContractStatus.UNDER_CONTRACT);
        this.openingStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.TYPE,
                ConstantLookupKeys.Cirius.GasMeterOpenCloseValveStatus.OPENING);
        this.disasterClosureReason = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ClosureReasonType.TYPE,
                ConstantLookupKeys.Cirius.ClosureReasonType.CLOSED_IN_CASE_OF_DISASTER);
        this.temporaryRegistrationStatus = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractStatusCode.TEMPORARY_REGISTRATION);
        this.switchApplicationCisCode = RequiredConstantResolver.requireCiriusConstant(
                resolvedCiriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractCisJudgmentCode.SWITCH_APPLICATION);
    }

    /**
     * MA-033 を実行する。
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "オンライン解約報告情報取得",
            description = "オンライン解約報告情報を返却する。",
            operationId = "getOnlineTerminateReportInfo")
    @APIResponse(
            responseCode = "200",
            description = "オンライン解約報告情報",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = OnlineTerminateReportInfoGetResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public OnlineTerminateReportInfoGetResponse endpoint(@BeanParam OnlineTerminateReportInfoGetRequest request) {
        bindCorrelationIdFromQueryIfPresent(request);
        return execute(request, () -> handle(request));
    }

    OnlineTerminateReportInfoGetResponse handle(OnlineTerminateReportInfoGetRequest request) {
        Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
        MeterReadingResultAcquisitionForCancellationOutput meterReading =
                meterReadingResultAcquisitionForCancellation.execute(gasMtstNo);

        SvmstkrRecord svmstkrRecord = svmstkrDataAccess
                .findByMtstNoAndRkkiKnrnJtkb(gasMtstNo, relatedValidStatus)
                .orElseThrow(() -> new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000027));

        List<SvckyaRecord> svckyaRecords =
                svckyaDataAccess.findAllBySvkyNoAndSvkyJtkb(svmstkrRecord.svkyNo(), applyingServiceContractStatus);
        if (svckyaRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs.FIND_ALL_BY_SVKY_NO_AND_SVKY_JTKB.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", svmstkrRecord.svkyNo())
                            .put("svkyJtkb", applyingServiceContractStatus));
        }
        SvckyaRecord svckyaRecord = svckyaRecords.get(0);

        LocalDate knskKjnYmd = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set in RequestContext"))
                .toLocalDate();
        GmtsetkRecord gmtsetkRecord = gmtsetkDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, knskKjnYmd)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        gmtsetkDataAccess.tableName(),
                        GmtsetkDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYmd", knskKjnYmd)));

        validateBusinessRules(meterReading, gmtsetkRecord);

        if (!taksmtrDataAccess.findAllByGmtSetNoAndTaksKyJtCdListAndTaksCisHntCd(
                gasMtstNo,
                List.of(temporaryRegistrationStatus),
                switchApplicationCisCode).isEmpty()) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000042);
        }

        String correlationId = resolveResponseCorrelationId();

        return new OnlineTerminateReportInfoGetResponse(
                new GmtsetkInfoResponse(
                        gmtsetkRecord.gasMtstNo(),
                        formatDate(gmtsetkRecord.tkysYmd()),
                        formatDate(gmtsetkRecord.hesnYmd()),
                        gmtsetkRecord.hesnJysb(),
                        gmtsetkRecord.hesnSdnShbt()),
                new SvckyaInfoResponse(
                        svckyaRecord.svkyNo(),
                        svckyaRecord.skno(),
                        svckyaRecord.kiyShbt(),
                        svckyaRecord.recUpdJnjNo()),
                new SvmstkrInfoResponse(svmstkrRecord.skno()),
                new SkinKnkkInfoResponse(formatDate(meterReading.knsYmd())),
                new KjnKnsKkaInfoResponse(formatDate(meterReading.knsYmd())),
                correlationId);
    }

    private void bindCorrelationIdFromQueryIfPresent(OnlineTerminateReportInfoGetRequest request) {
        if (request == null) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId != null) {
            requestContext.setCorrelationId(CorrelationIdResolver.resolve(correlationId));
        }
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }

    private void validateBusinessRules(
            MeterReadingResultAcquisitionForCancellationOutput meterReading,
            GmtsetkRecord gmtsetkRecord) {
        if (openingStatus.equals(gmtsetkRecord.gmtKhsJtkb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000067);
        }
        if (disasterClosureReason.equals(gmtsetkRecord.hesnJysb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000034);
        }
        if (meterReading.knsYmd() == null || gmtsetkRecord.hesnYmd() == null) {
            return;
        }
        if (meterReading.knsYmd().isBefore(gmtsetkRecord.hesnYmd())
                || (meterReading.knsYmd().isEqual(gmtsetkRecord.hesnYmd())
                && openingStatus.equals(meterReading.gmtKhsJtkb()))) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000035);
        }
    }

    private static String formatDate(LocalDate value) {
        return value == null ? null : value.toString();
    }
}
