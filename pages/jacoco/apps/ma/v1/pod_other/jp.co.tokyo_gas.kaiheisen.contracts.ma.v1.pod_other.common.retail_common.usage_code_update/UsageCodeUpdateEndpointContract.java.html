package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.common.retail_common.usage_code_update;

import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.endpoint.MaEndpointContract;

/**
 * MA-108 のエンドポイント定数を定義する契約クラス。
 */
public final class UsageCodeUpdateEndpointContract
        implements MaEndpointContract<UsageCodeUpdateRequest, UsageCodeUpdateResponse> {

    public static final UsageCodeUpdateEndpointContract INSTANCE = new UsageCodeUpdateEndpointContract();

    public static final String MA_ID = "MA-108";

    public static final String PATH = "/v1/usage-code";

    public static final String HTTP_METHOD = "POST";

    private UsageCodeUpdateEndpointContract() {
    }

    @Override
    public String maId() {
        return MA_ID;
    }

    @Override
    public String httpMethod() {
        return HTTP_METHOD;
    }

    @Override
    public String path() {
        return PATH;
    }

    @Override
    public Class<UsageCodeUpdateRequest> requestType() {
        return UsageCodeUpdateRequest.class;
    }

    @Override
    public Class<UsageCodeUpdateResponse> responseType() {
        return UsageCodeUpdateResponse.class;
    }
}
