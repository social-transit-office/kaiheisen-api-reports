/**
 * MA-108 用途コード更新機能-共通 リクエスト。
 */
@Schema(description = "MA-108 用途コード更新機能-共通 リクエスト")
public record UsageCodeUpdateRequest(
        @RequiredValue
        @NumericMaxDigits(max = 11)
        @NonZeroNumericValue

        String gasYtCd,

        @RequiredText
        @TextMaxLength(max = 1)
        @ConstantTypeValue(type = ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE)
        @Schema(description = "報告システム種別", required = true, maxLength = 1)
        String hukkSysShbt,

        @TextMaxLength(max = 20)
        @Schema(description = "相関ID", maxLength = 20, nullable = true)
        String correlationId,

        @Rfc3339DateTimeText
        @Schema(description = "システム日時", format = "date-time", nullable = true)
        String systemDatetime) implements BaseRequest {
