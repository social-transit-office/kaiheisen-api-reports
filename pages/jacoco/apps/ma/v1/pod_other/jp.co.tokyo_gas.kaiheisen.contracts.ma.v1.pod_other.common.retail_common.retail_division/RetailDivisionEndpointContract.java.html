/**
 * MA-107 のエンドポイント定数を定義する契約クラス。
 */
public final class RetailDivisionEndpointContract
        implements MaEndpointContract<RetailDivisionRequest, RetailDivisionResponse> {

    /** シングルトンインスタンス。 */
    public static final RetailDivisionEndpointContract INSTANCE = new RetailDivisionEndpointContract();

    /** MA 識別子。 */
    public static final String MA_ID = "MA-107";

    /** MA-107 のパス。 */
    public static final String PATH = "/v1/retail-division";

    /** MA-107 の HTTP メソッド。 */
    public static final String HTTP_METHOD = "GET";

    private RetailDivisionEndpointContract() {
    }

    @Override
    public String maId() {
        return MA_ID;
    }

    @Override
    public String httpMethod() {
        return HTTP_METHOD;
    }

    @Override
    public String path() {
        return PATH;
    }

    @Override
    public Class<RetailDivisionRequest> requestType() {
        return RetailDivisionRequest.class;
    }

    @Override
    public Class<RetailDivisionResponse> responseType() {
        return RetailDivisionResponse.class;
    }
}
