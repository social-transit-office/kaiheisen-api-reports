package jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.common.retail_common.retail_division;

import jakarta.ws.rs.QueryParam;
import jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.core.request.BaseRequest;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.AllowedValues;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.NonZeroNumericValue;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.NumericMaxDigits;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.RequiredText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.Rfc3339DateTimeText;
import jp.co.tokyo_gas.kaiheisen.libs.common.validation.constraint.TextMaxLength;
import org.eclipse.microprofile.openapi.annotations.media.Schema;

/**
 * MA-107 小売判定 API のリクエスト。
 *
 * <p>相関IDは HTTP ヘッダで管理し、リクエストボディやクエリパラメータ
 * （本 DTO を含む）には含めない。</p>
 *
 * @param shrKbn 処理区分
 * @param gasMtstNo ガスメーター設置場所番号
 * @param knskKjnYmd 検索基準年月日
 * @param hukkSysShbt 報告システム種別
 * @param systemDatetime システム日時
 */
public record RetailDivisionRequest(
        @QueryParam("shrKbn")
        @RequiredText
        @AllowedValues(value = {"1", "2", "3"})
        @Schema(description = "処理区分", enumeration = {"1", "2", "3"}, maxLength = 1)
        String shrKbn,
        @QueryParam("gasMtstNo")
        @RequiredText
        @NumericMaxDigits(max = 11)
        @NonZeroNumericValue
        @Schema(description = "ガスメーター設置場所番号", pattern = "^(?!0+$)\\d{1,11}$", maxLength = 11)
        String gasMtstNo,
        @QueryParam("knskKjnYmd")
        @RequiredText
        @Rfc3339DateText
        @Schema(description = "検索基準年月日 (RFC3339 日付: yyyy-MM-dd)", format = "date")
        String knskKjnYmd,
        @QueryParam("hukkSysShbt")
        @TextMaxLength(max = 1)
        @Schema(description = "報告システム種別", maxLength = 1)
        String hukkSysShbt,
        @QueryParam("systemDatetime")
        @Rfc3339DateTimeText
        @Schema(description = "システム日時 (RFC3339)", format = "date-time")
        String systemDatetime) implements BaseRequest {

    /**
     * 処理区分を返します。
     *
     * @return 処理区分
     */
    public String getShrKbn() {
        return shrKbn;
    }

    /**
     * ガスメーター設置場所番号を返します。
     *
     * @return ガスメーター設置場所番号
     */
    public String getGasMtstNo() {
        return gasMtstNo;
    }

    /**
     * 検索基準年月日を返します。
     *
     * @return 検索基準年月日
     */
    public String getKnskKjnYmd() {
        return knskKjnYmd;
    }

    /**
     * 報告システム種別を返します。
     *
     * @return 報告システム種別
     */
    public String getHukkSysShbt() {
        return hukkSysShbt;
    }

    /**
     * システム日時を返します。
     *
     * @return システム日時
     */
    public String getSystemDatetime() {
        return systemDatetime;
    }
}
