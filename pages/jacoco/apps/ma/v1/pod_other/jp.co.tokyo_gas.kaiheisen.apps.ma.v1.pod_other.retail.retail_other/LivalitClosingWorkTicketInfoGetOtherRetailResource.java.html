    LivalitClosingWorkTicketInfoGetOtherRetailResponse handle(
            LivalitClosingWorkTicketInfoGetOtherRetailRequest request) {
        LocalDate hsenKiboYmd = Rfc3339LocalDateParser.parse(request.hsenKiboYmd());
        Long gasMtstNo = Long.valueOf(request.gasMtstNo().trim());
        Long syoKyoTrno = parseOptionalLong(request.syoKyoTrno());
        Long tksuSvkyNo = parseOptionalLong(request.tksuSvkyNo());
        String correlationId = requestContext
                .getCorrelationId()
                .orElseThrow(
                        () -> new IllegalStateException("correlationId is not set in RequestContext"));

        GmtsetdRecord gmtsetdRecord = requireGmtsetd(gasMtstNo, hsenKiboYmd);
        if (areContractNumbersOmitted(syoKyoTrno, tksuSvkyNo)
                && !openingStatus.equals(gmtsetdRecord.gmtKhsJtkb())) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000068);
        }

        SeriesOriginMeterLocationNumberAcquisitionOutput seriesOriginOutput =
            seriesOriginMeterLocationNumberAcquisition.execute(
                gasMtstNo,
                gmtsetdRecord.mtgMtstNo(),
                gmtsetdRecord.gmtKhsJtkb(),
                gmtsetdRecord.ckrtGmscJtkb());

        GasmtrkRecord gasmtrkRecord = requireGasmtrk(gasMtstNo, hsenKiboYmd);
        KnsytiRecord knsytiRecord = requireKnsyti(gasMtstNo, hsenKiboYmd);
        Optional<KiycstdRecord> kiycstdRecord = findKiycstd(syoKyoTrno);

        RecentThreeMeterReadingResultsAcquisitionOutput recentThreeOutput =
            recentThreeMeterReadingResultsAcquisition.execute(gasMtstNo, hsenKiboYmd);
        MeterReadingSupportInfoAcquisitionOutput supportInfoOutput =
            meterReadingSupportInfoAcquisition.execute(gasMtstNo);
        OneYearAgoMeterReadingResultAcquisitionOutput oneYearAgoOutput =
            oneYearAgoMeterReadingResultAcquisition.execute(gasMtstNo, hsenKiboYmd, tksuSvkyNo);
        RelatedDedicatedMeterInfoAcquisitionOutput relatedMeterOutput =
            relatedDedicatedMeterInfoAcquisition.execute(
                gasMtstNo,
                gmtsetdRecord.mtgMtstNo(),
                gmtsetdRecord.ckrtGmscJtkb(),
                hsenKiboYmd,
                gmtsetdRecord.szcGkkCd(),
                gmtsetdRecord.szcCmazCd(),
                gmtsetdRecord.szcMchCd());

        return new LivalitClosingWorkTicketInfoGetOtherRetailResponse(
            MaResponseNumericValues.orZero(tksuSvkyNo),
            MaResponseNumericValues.orZero(kiycstdRecord.map(KiycstdRecord::kyoTrkNo).orElse(null)),
            formatDate(gmtsetdRecord.hesnYmd()),
            formatDate(gmtsetdRecord.kasYmd()),
            formatYearMonth(gmtsetdRecord.snstGkjYm()),
            gmtsetdRecord.gmicDbriShbt(),

            gmtsetdRecord.hesnSdnShbt(),
            supportInfoOutput == null ? null : supportInfoOutput.gmtItiInf(),
            toFirstNineCharacters(gasmtrkRecord.gmb()),
            gasmtrkRecord.gmkShbt(),
            MaResponseNumericValues.oneDecimalOrZero(gasmtrkRecord.gmtGusu()),
            toLower11Digits(MaResponseNumericValues.orZero(gasmtrkRecord.sjskHticSu())),
            MaResponseNumericValues.oneDecimalOrZero(gasmtrkRecord.sjsuMincSu()),
            formatYearMonth(gasmtrkRecord.ktyYm()),
            takusoKeiyakuShubetsu,
            gmtsetdRecord.kasJysb(),
            kiycstdRecord.map(KiycstdRecord::ctnmKn1).orElse(null),
            kiycstdRecord.map(KiycstdRecord::ctnmKj1).orElse(null),

    private GmtsetdRecord requireGmtsetd(Long gasMtstNo, LocalDate hsenKiboYmd) {
        List<GmtsetdRecord> records =
                gmtsetdDataAccess.findAllByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, hsenKiboYmd);
        if (!records.isEmpty()) {
            return records.getFirst();
        }
        throw DataInconsistencySupport.create(
                gmtsetdDataAccess.tableName(),
                GmtsetdDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                        .put("knskKjnYmd", hsenKiboYmd));
    }

    private GasmtrkRecord requireGasmtrk(Long gasMtstNo, LocalDate hsenKiboYmd) {
        List<GasmtrkRecord> records =
                gasmtrkDataAccess.findAllByTtkGmsbNoAndGmtkYmdAndGmtThkgYmd(gasMtstNo, hsenKiboYmd);
        if (!records.isEmpty()) {
            return records.getFirst();
        }
        throw DataInconsistencySupport.create(
                gasmtrkDataAccess.tableName(),
                GasmtrkDataAccess.CallSpecs.FIND_ALL_BY_TTK_GMSB_NO_AND_GMTK_YMD_AND_GMT_THKG_YMD.identifier(),
                new FlatJsonBuilder()
                        .put("ttkGmsbNo", gasMtstNo)
                        .put("knskKjnYmd", hsenKiboYmd));
    }

    private KnsytiRecord requireKnsyti(Long gasMtstNo, LocalDate hsenKiboYmd) {
        YearMonth month = YearMonth.of(hsenKiboYmd.getYear(), hsenKiboYmd.getMonthValue());
        List<KnsytiRecord> records = knsytiDataAccess.findAllByGasMtstNoAndTkysYmAndTyenYm(gasMtstNo, month);
        if (!records.isEmpty()) {
            return records.getFirst();
        }
        throw DataInconsistencySupport.create(
            knsytiDataAccess.tableName(),
            KnsytiDataAccess.CallSpecs.FIND_ALL_BY_GAS_MTST_NO_AND_TKYS_YM_AND_TYEN_YM.identifier(),
                new FlatJsonBuilder()
                        .put("gasMtstNo", gasMtstNo)
                .put("knskKjnYm", month.toString()));
    }

    private Optional<KiycstdRecord> findKiycstd(Long kyoTrkNo) {
        if (kyoTrkNo == null) {
            return Optional.empty();
        }
        Optional<KiycstdRecord> record = kiycstdDataAccess.findByKyoTrkNo(kyoTrkNo);
        if (record.isPresent()) {
            return record;
        }
        throw DataInconsistencySupport.create(
            kiycstdDataAccess.tableName(),
            KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
            new FlatJsonBuilder().put("kyoTrkNo", kyoTrkNo));
    }

    private static List<MeterReadingInfoResponse> mapKnsnInf(
            RecentThreeMeterReadingResultsAcquisitionOutput output) {
        if (output == null || output.knsnInf() == null || output.knsnInf().isEmpty()) {
            return null;
        }
        return output.knsnInf().stream()
                .map(LivalitClosingWorkTicketInfoGetOtherRetailResource::mapKnsnInfItem)
                .toList();
    }

    private static MeterReadingInfoResponse mapKnsnInfItem(KnsnInf source) {
        if (source == null) {

        }
        return new MeterReadingInfoResponse(

    }

            OneYearAgoMeterReadingResultAcquisitionOutput output) {
        if (output == null) {

    }

    private static List<String> mapKnsnSienInf(MeterReadingSupportInfoAcquisitionOutput output) {
        if (output == null) {
            return List.of();
        }
        List<String> values = new ArrayList<>();
        addIfPresent(values, output.knsnSienInf1());
        addIfPresent(values, output.knsnSienInf2());
        return values.isEmpty() ? List.of() : List.copyOf(values);
    }

    private static void addIfPresent(List<String> collector, String value) {
        if (value == null) {
            return;
        }
        collector.add(value);
    }

    private static List<RelatedDedicatedMeterInfoResponse> mapCkrtGmtInf(
            RelatedDedicatedMeterInfoAcquisitionOutput output) {
        if (output == null || output.gasMtstNo() == null || output.gasMtstNo().isEmpty()) {
            return List.of();
        }
        List<Long> gasMtstNoList = output.gasMtstNo();
        List<String> gmbList = output.gmb() == null ? List.of() : output.gmb();
        List<RelatedDedicatedMeterInfoResponse> result = new ArrayList<>();
        for (int index = 0; index < gasMtstNoList.size(); index++) {
            String gmb = index < gmbList.size() ? gmbList.get(index) : null;
            result.add(new RelatedDedicatedMeterInfoResponse(
                    MaResponseNumericValues.orZero(gasMtstNoList.get(index)), gmb));
        }
        return List.copyOf(result);
    }

    private static String formatDate(LocalDate date) {
        return date == null ? null : date.toString();
    }

    private static String formatYearMonth(YearMonth yearMonth) {
        return yearMonth == null ? null : yearMonth.toString();
    }

    private static String toFirstNineCharacters(String value) {
        if (value == null || value.isEmpty()) {
            return value;
        }
        return value.substring(0, Math.min(9, value.length()));
    }

    private static Long toLower11Digits(Long value) {
        if (value == null) {
            return null;
        }
        return Math.floorMod(value, 100_000_000_000L);
    }

    private static Long parseOptionalLong(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        if (normalized.isEmpty()) {
            return null;
        }
        return Long.valueOf(normalized);
    }

    private static boolean areContractNumbersOmitted(Long syoKyoTrno, Long tksuSvkyNo) {
        return syoKyoTrno == null && tksuSvkyNo == null;
    }

    private static String resolveTakusoKeiyakuShubetsu(CiriusConstantService ciriusConstantService) {
        return RequiredConstantResolver.requireCiriusConstant(
                ciriusConstantService,
                ConstantLookupKeys.Cirius.ConsignmentContractType.TYPE,
                ConstantLookupKeys.Cirius.ConsignmentContractType.CONSIGNMENT_CONTRACT);
    }
