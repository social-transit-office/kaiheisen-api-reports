/**
 * MA-108 用途コード更新機能-共通 API のリソース実装。
 */
@Path(UsageCodeUpdateEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-108", name = "用途コード更新機能-共通")
@Tag(name = "用途コード更新機能-共通", description = "MA-108 用途コード更新機能-共通 API")
public final class UsageCodeUpdateResource extends BaseResource<UsageCodeUpdateRequest> {

    static final long MIN_GAS_MTST_NO = 10_000_000_000L;

    static final long MAX_GAS_MTST_NO = 19_999_999_999L;

    static final String CASE_HISTORY_DIVISION = ConstantLookupKeys.Cirius.CaseHistory.WORK;

    static final String CASE_HISTORY_REGISTRATION_DESTINATION =
            ConstantLookupKeys.Cirius.ItemHistoryRegistrationDestinationType.PLACE;

    static final String RETAIL_WORK_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.REGULAR_SECURITY_PATROL_S00002;

    static final String RETAIL_WORK_DETAIL_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailCodeRetail.CHANGE_USAGE_CODE_S0000204;

    static final String PIPELINE_WORK_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryWorkCodeConduit.REGULAR_SECURITY_PATROL_S00002;

    static final String PIPELINE_WORK_DETAIL_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryWorkDetailsCodeConduit.CHANGE_USAGE_CODE_S0000202;

    static final String REPORTING_ROUTE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryReportingRoute.OTHER_SYSTEMS;

    static final String INQUIRY_TARGET_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryInquiryCode.SUBJECT;

    private final UsageCodeUpdateFunctionPipeline usageCodeUpdateFunctionPipeline;

    private final UsageCodeUpdateFunctionRetail usageCodeUpdateFunctionRetail;

    private final TransactionSynchronizationRegistry transactionSynchronizationRegistry;

    /**
     * コンストラクタ。
     */
    @Inject
    public UsageCodeUpdateResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @UsageCodeUpdateFunctionPipelineConfiguration
            UsageCodeUpdateFunctionPipeline usageCodeUpdateFunctionPipeline,
            @UsageCodeUpdateFunctionRetailConfiguration
            UsageCodeUpdateFunctionRetail usageCodeUpdateFunctionRetail,
            TransactionSynchronizationRegistry transactionSynchronizationRegistry) {
        super(requestContext, requestDtoValidator, clock);
        this.usageCodeUpdateFunctionPipeline = Objects.requireNonNull(
                usageCodeUpdateFunctionPipeline,
                "usageCodeUpdateFunctionPipeline");
        this.usageCodeUpdateFunctionRetail = Objects.requireNonNull(
                usageCodeUpdateFunctionRetail,
                "usageCodeUpdateFunctionRetail");
        this.transactionSynchronizationRegistry = transactionSynchronizationRegistry;
    }

    UsageCodeUpdateResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            UsageCodeUpdateFunctionPipeline usageCodeUpdateFunctionPipeline,
            UsageCodeUpdateFunctionRetail usageCodeUpdateFunctionRetail) {
        super(requestContext, requestDtoValidator, clock);
        this.usageCodeUpdateFunctionPipeline = Objects.requireNonNull(
                usageCodeUpdateFunctionPipeline,
                "usageCodeUpdateFunctionPipeline");
        this.usageCodeUpdateFunctionRetail = Objects.requireNonNull(
                usageCodeUpdateFunctionRetail,
                "usageCodeUpdateFunctionRetail");
        this.transactionSynchronizationRegistry = null;
    }

    /**
     * MA-108 を実行する。
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(summary = "用途コード更新機能-共通", description = "TSV 設計に従い用途コード更新を実行する。")
    @APIResponse(
            responseCode = "200",
            description = "用途コード更新完了",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = UsageCodeUpdateResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = UsageCodeUpdateErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public Response endpoint(@RequestBody UsageCodeUpdateRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        try {
            return Response.ok(execute(request, () -> handle(request)))
                    .type(MediaType.APPLICATION_JSON)
                    .build();
        } catch (BusinessRuleException | DataNotFoundException exception) {
            return usageCodeUpdateErrorResponse(exception);
        }
    }

    private UsageCodeUpdateResponse handle(UsageCodeUpdateRequest request) {
        Long gasMtstNo = Objects.requireNonNull(request.gasMtstNo(), "gasMtstNo");
        if (gasMtstNo < MIN_GAS_MTST_NO || gasMtstNo > MAX_GAS_MTST_NO) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000089);
        }

        InternalVariables internalVariables = resolveInternalVariables(normalizeOptional(request.hukkSysShbt()));
        LocalDate shrYmd = Rfc3339LocalDateParser.parse(request.shrYmd());
        String gasYtCd = normalizeOptional(request.gasYtCd());

        usageCodeUpdateFunctionPipeline.execute(
                gasMtstNo,
                shrYmd,
                gasYtCd,
                internalVariables.akrkKbn(),
                internalVariables.akrkTrskShbt(),
                internalVariables.duknIsrgCd(),
                internalVariables.duknIsrsCd(),
                internalVariables.isrrHkkrKbn(),
                internalVariables.isrrSktCd());

        usageCodeUpdateFunctionRetail.execute(
                gasMtstNo,
                shrYmd,
                gasYtCd,
                internalVariables.akrkKbn(),
                internalVariables.akrkTrskShbt(),
                internalVariables.isrrGymCd(),
                internalVariables.isrrGmssCd(),
                internalVariables.isrrHkkrKbn(),
                internalVariables.isrrSktCd());

        return new UsageCodeUpdateResponse(resolveResponseCorrelationId());
    }

    private Response usageCodeUpdateErrorResponse(KaiheisenException exception) {
        markTransactionRollbackOnly();
        ErrorResponseBody body = MaErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(MaErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(new UsageCodeUpdateErrorResponse(
                        body.errorCode(),
                        body.errorDescription(),
                        body.errorDetailList()))
                .build();
    }

    private void markTransactionRollbackOnly() {
        if (transactionSynchronizationRegistry != null) {
            transactionSynchronizationRegistry.setRollbackOnly();
        }
    }

    private InternalVariables resolveInternalVariables(String hukkSysShbt) {
        if (ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT.equals(hukkSysShbt)) {
            return new InternalVariables(
                    CASE_HISTORY_DIVISION,
                    CASE_HISTORY_REGISTRATION_DESTINATION,
                    RETAIL_WORK_CODE,
                    RETAIL_WORK_DETAIL_CODE,
                    PIPELINE_WORK_CODE,
                    PIPELINE_WORK_DETAIL_CODE,
                    REPORTING_ROUTE,
                    INQUIRY_TARGET_CODE);
        }
        return new InternalVariables(null, null, null, null, null, null, null, null);
    }

    private void bindCorrelationIdFromBodyIfPresent(UsageCodeUpdateRequest request) {
        if (request == null) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId != null) {
            requestContext.setCorrelationId(CorrelationIdResolver.resolve(correlationId));
        }
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }

    record InternalVariables(
            String akrkKbn,
            String akrkTrskShbt,
            String isrrGymCd,
            String isrrGmssCd,
            String duknIsrgCd,
            String duknIsrsCd,
            String isrrHkkrKbn,
            String isrrSktCd) {
    }
}
