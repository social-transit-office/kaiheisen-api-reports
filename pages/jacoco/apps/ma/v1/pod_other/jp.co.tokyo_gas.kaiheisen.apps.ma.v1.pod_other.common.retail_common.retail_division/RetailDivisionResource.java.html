/**
 * MA-107 小売判定 API のリソース実装。
 */
@MainService(identifier = "MA-107", name = "小売判定 API")
@Path(RetailDivisionEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
@Tag(name = "小売他小売判定", description = "MA-107 小売／他小売判定 API")
public class RetailDivisionResource extends BaseResource<RetailDivisionRequest> {

    private final RetailTypeJudgment retailTypeJudgment;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム時刻プロバイダ
     * @param retailTypeJudgment 小売判定サービス
     */
    @Inject
    public RetailDivisionResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            @RetailDivisionConfiguration
            RetailTypeJudgment retailTypeJudgment) {
        super(requestContext, requestDtoValidator, clock);
        this.retailTypeJudgment = retailTypeJudgment;
    }

    /**
     * MA-107 小売判定 API を実行する。
     *
     * @param request リクエスト DTO
     * @return 小売判定レスポンス
     */
    @GET
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "小売／他小売判定",
            description = "小売／他小売判定を行い、判定結果と託送サービス契約関連情報を返却する。",
            operationId = "getRetailDivision")
    @APIResponse(
            responseCode = "200",
            description = "小売／他小売判定の結果",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = RetailDivisionResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public RetailDivisionResponse endpoint(@BeanParam RetailDivisionRequest request) {
        return execute(request, () -> {
            Long gasMtstNoValue = Long.valueOf(request.gasMtstNo().trim());
            LocalDate knskKjnYmdValue = Rfc3339LocalDateParser.parse(request.knskKjnYmd());

            RetailTypeJudgmentOutput output = retailTypeJudgment.execute(
                    request.shrKbn(),
                    gasMtstNoValue,
                    knskKjnYmdValue);
            return toResponse(output);
        });
    }

    /**
     * SV-001 出力をレスポンス DTO へ変換する。
     *
     * @param output SV-001 出力
     * @return レスポンス DTO
     */
    private RetailDivisionResponse toResponse(
            RetailTypeJudgmentOutput output) {
        return new RetailDivisionResponse(
                output.kuHntKbn(),
                mapTskgmstInf(output.tskgmstInf()),
                mapTkssvkyInf(output.tkssvkyInf()),
                requestContext.getCorrelationId().orElseThrow(
                        () -> new IllegalStateException(
                                "correlationId is not set in RequestContext")));
    }

    /**
     * 託送サービス契約・ガスメーター設置場所関連情報を レスポンス DTO へ変換する。
     *
     * @param source SV-001 出力の託送サービス契約・ガスメーター設置場所関連情報
     * @return レスポンス DTO。{@code source} が {@code null} の場合は {@code null}
     */
    private static TskgmstInfoResponse mapTskgmstInf(TskgmstInfo source) {
        if (source == null) {
            return null;
        }
        return new TskgmstInfoResponse(
            MaResponseNumericValues.orZero(source.gasMtstNo()),
            MaResponseNumericValues.orZero(source.tksuSvkyNo()),
            MaResponseNumericValues.orZero(source.skno()),
                source.tksuSvkkJtkb(),
                formatDate(source.tkysYmd()),
                formatDate(source.tyenYmd()),
            MaResponseNumericValues.orZero(source.recUpdJnjNo()));
    }

    /**
     * 託送サービス契約情報をレスポンス DTO へ変換する。
     *
     * @param source SV-001 出力の託送サービス契約情報
     * @return レスポンス DTO。{@code source} が {@code null} の場合は {@code null}
     */
    private static TkssvkyInfoResponse mapTkssvkyInf(TkssvkyInfo source) {
        if (source == null) {
            return null;
        }
        return new TkssvkyInfoResponse(
            MaResponseNumericValues.orZero(source.tksuSvkyNo()),
            MaResponseNumericValues.orZero(source.skno()),
                source.tksuSvkyJtkb(),
                source.tksuSvksJysb(),
                source.tksuSvkeJysb(),
                formatDate(source.tkysYmd()),
                formatDate(source.tyenYmd()),
                formatDate(source.hesnYtiYmd()),
                source.tkykShbt(),
                source.tksuKijsCd(),
                source.sutiKskhFlg(),
                MaResponseNumericValues.orZero(source.syoKyoTrno()),
                MaResponseNumericValues.orZero(source.recUpdJnjNo()));
    }

    /**
     * {@link LocalDate} を ISO-8601 文字列（{@code yyyy-MM-dd}）へ変換する。
     *
     * @param date 変換対象日付
     * @return 文字列表現。{@code date} が {@code null} の場合は {@code null}
     */
    private static String formatDate(LocalDate date) {
        if (date == null) {
            return null;
        }
        return date.toString();
    }
}
