package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_cancel.config;

import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Produces;
import jakarta.inject.Singleton;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantServices;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;

/**
 * pod_cancel の共通基盤 Bean 定義。
 */
@ApplicationScoped
public class PodCancelLibsConfiguration {

    @Produces
    @Singleton
    CiriusConstantService ciriusConstantService() {
        return ConstantServices.create().ciriusConstantService();
    }

    @Produces
    @Singleton
    KaiheisenConstantService kaiheisenConstantService() {
        return ConstantServices.create().kaiheisenConstantService();
    }
}
