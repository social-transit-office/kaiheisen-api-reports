package jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_cancel.config;

import com.blazebit.persistence.CriteriaBuilderFactory;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Instance;
import jakarta.inject.Inject;
import jakarta.persistence.EntityManager;
import jp.co.tokyo_gas.kaiheisen.apps.ma.v1.pod_cancel.retail.retail_tg.ClosingCancelClosingErrorFollowCancelTgRetailResource;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.logging.LogLayer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.CdiBeanInitializer;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.warmup.StartupWarmupContributor;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.jpa.core.JpaStartupQueryWarmup;

/** 取消MAの起動時ウォームアップ定義。 */
@ApplicationScoped
public class PodCancelStartupWarmupContributor implements StartupWarmupContributor {

    private final CdiBeanInitializer beanInitializer;

    private final EntityManager entityManager;

    private final Instance<CriteriaBuilderFactory> criteriaBuilderFactory;

    private final Sysdummy1DataAccess dataAccess;

    /**
     * コンストラクタ。
     *
     * @param beanInitializer CDI Bean初期化処理
     * @param entityManager JPAエンティティマネージャ
     * @param criteriaBuilderFactory 実行時に取得するBlazeの問い合わせファクトリ
     * @param dataAccess DBウォームアップDA
     */
    @Inject
    public PodCancelStartupWarmupContributor(
            final CdiBeanInitializer beanInitializer,
            final EntityManager entityManager,
            final Instance<CriteriaBuilderFactory> criteriaBuilderFactory,
            final Sysdummy1DataAccess dataAccess) {
        this.beanInitializer = beanInitializer;
        this.entityManager = entityManager;
        this.criteriaBuilderFactory = criteriaBuilderFactory;
        this.dataAccess = dataAccess;
    }

    @Override
    public String applicationName() {
        return "ma-cancel";
    }

    @Override
    public LogLayer logLayer() {
        return LogLayer.MA;
    }

    @Override
    public void initializeResources() {
        beanInitializer.initialize(ClosingCancelClosingErrorFollowCancelTgRetailResource.class);
    }

    @Override
    public String databaseWarmupFunctionId() {
        return JpaStartupQueryWarmup.FUNCTION_ID;
    }

    @Override
    public int warmUpDatabase() {
        int databaseValue = dataAccess.fetchWarmupValue().value();
        if (databaseValue != 1) {
            return databaseValue;
        }
        return JpaStartupQueryWarmup.execute(entityManager, criteriaBuilderFactory.get());
    }
}
