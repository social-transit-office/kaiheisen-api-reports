    SamedayReportCancelTargetGetClosingTgRetailResponse handle(
            SamedayReportCancelTargetGetClosingTgRetailRequest request) {
        LocalDate targetMeterReadingDate = Rfc3339LocalDateParser.parse(request.trkKnsYmd());
        validateNotFuture(targetMeterReadingDate);

        long gasMtstNo = Long.parseLong(request.gasMtstNo());
        long contractCustomerNo = Long.parseLong(request.syoKyoTrno());

        GmtsetdRecord gmtsetdRecord = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, targetMeterReadingDate)
                .orElseThrow(() -> new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND));

        try {
            openingClosingInProgressErrorJudgment.execute(
                    gmtsetdRecord.gmtKhsJtkb(),
                    openingInProgressErrorJudgmentDivision);
        } catch (KaisenRuleException exception) {
            throw new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND);
        }

        KiycstkRecord kiycstkRecord = kiycstkDataAccess.findByKyoTrkNo(contractCustomerNo)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kiycstkDataAccess.tableName(),
                        KiycstkDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                        new FlatJsonBuilder().put("kyoTrkNo", contractCustomerNo)));

        ServiceContractNumberAcquisitionOutput serviceContractOutput =
                serviceContractNumberAcquisition.execute(
                        gasMtstNo,
                        gmtsetdRecord.mtgMtstNo(),
                        SHR_KBN_RELATED_ACTIVE_OR_END_WAITING_OR_END,
                        gmtsetdRecord.ckrtGmscJtkb(),
                        targetMeterReadingDate);

        Long serviceContractNumber = serviceContractOutput.svkyNo();
        if (serviceContractNumber == null || serviceContractNumber == 0L) {
            throw new KaiheisenException("ガスメーター設置場所番号の値が不正のため、処理を終了しました。");
        }

        List<SvckyaRecord> serviceContractRecords = svckyaDataAccess.findAllBySvkyNoAtKnskKjnYmdAndSvkyJtkbList(
                serviceContractNumber,
                targetMeterReadingDate,
                List.of(underContractStatus, endOfContractStatus));
        if (serviceContractRecords.isEmpty()) {
            throw DataInconsistencySupport.create(
                    svckyaDataAccess.tableName(),
                    SvckyaDataAccess.CallSpecs.FIND_ALL_BY_SVKY_NO_AT_KNSK_KJN_YMD_AND_SVKY_JTKB_LIST.identifier(),
                    new FlatJsonBuilder()
                            .put("svkyNo", serviceContractNumber)
                            .put("knskKjnYmd", targetMeterReadingDate)
                            .put("svkyJtkbList", List.of(underContractStatus, endOfContractStatus)));
        }
        SvckyaRecord serviceContractRecord = serviceContractRecords.getFirst();

        MeterReadingRelatedInfoEditProcessOutput meterReadingOutput =
                meterReadingRelatedInfoEditProcess.execute(
                        gasMtstNo,
                        targetMeterReadingDate,
                        gmtsetdRecord.hesnJysb(),
                        ConstantLookupKeys.Kaiheisen.MeterReadingRelatedInfoEditProcessingDivision
                                .CLOSING_METER_READING_RELATED_INFO_ACQUISITION);

        return toResponse(
                kiycstkRecord,
                gmtsetdRecord,
                serviceContractRecord,
                meterReadingOutput,
                requestContext.getCorrelationId().orElseThrow(
                        () -> new IllegalStateException("correlationId is not set")));
    }

    private void validateNotFuture(LocalDate targetMeterReadingDate) {
        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set"))
                .toLocalDate();
        if (targetMeterReadingDate.isAfter(systemDate)) {
            throw new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND);
        }
    }

            GmtsetdRecord gmtsetdRecord,

                new ServiceContractInfo(

                        formatDate(svckyaRecord.rkkiEndYmd())),

                meterReadingOutput.gasSyry() == null ? null : meterReadingOutput.gasSyry().doubleValue(),

                correlationId,
                null,
                null,
                null,
                null,

    }
