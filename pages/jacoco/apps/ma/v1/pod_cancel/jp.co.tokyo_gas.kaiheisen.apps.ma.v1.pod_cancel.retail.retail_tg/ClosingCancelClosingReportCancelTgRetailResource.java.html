        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd().trim());
        ClosingCancelClosingReportCancelTgRetailRequest.IsrrkrUpdInf isrrkrUpdInf =
                request.isrrkrUpdInf();

                followStatusJudgment.execute(
                        request.trkIsrrGymCd(),
                        request.gasMtstNo(),
                        trkKnsYmd,

        HseflInf hseflInf = sv096.hseflInf();

        String receptionNo;

        Long customerRegistrationNo;
        String workerName;

                    Objects.requireNonNull(hseflInf, "SV-096 hseflInf is required for not-yet-started branch");
            deleteHseflOrThrow(requiredHseflInf);
            ServiceContractNumberAcquisitionOutput sv003 =
                    serviceContractNumberAcquisition.execute(
                            request.gasMtstNo(),
                            null,
                            ServiceContractNumberAcquisitionConstants.ShrKbn.ACTIVE,
                            null,
                            trkKnsYmd);
            receptionIndexNo = requiredHseflInf.utidxNo();
            receptionNo = null;
            serviceContractNo = sv003.svkyNo();
            customerRegistrationNo = requiredHseflInf.syoKyoTrno();
