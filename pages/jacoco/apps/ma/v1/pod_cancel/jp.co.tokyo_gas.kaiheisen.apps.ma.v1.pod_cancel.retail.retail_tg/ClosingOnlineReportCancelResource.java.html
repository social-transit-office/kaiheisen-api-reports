    ClosingOnlineReportCancelResponse handle(ClosingOnlineReportCancelRequest request) {
        Long gasMtstNo = Long.valueOf(request.gasMtstNo());
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd());
        ClosingOnlineReportCancelRequest.IsrrkrUpdInf isrrkrUpdInf = request.isrrkrUpdInf();

        followStatusJudgment.execute(request.trkIsrrGymCd(), gasMtstNo, trkKnsYmd, SHR_KBN_CANCEL);
        ClosingCancelFeasibilityJudgmentEquipmentStatusOutput sv101 =
                equipmentStatusJudgment.execute(gasMtstNo, trkKnsYmd);
        dedicatedMeterLocationJudgment.execute(
                trkKnsYmd,
                new DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentGmtsetdInf(
                        sv101.ckrtGmscJtkb(),
                        sv101.mtgMtstNo()));
        ClosingCancelFeasibilityJudgmentRetailInfoOutput sv103 = retailInfoJudgment.execute(
                trkKnsYmd,
                new jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.GmtsetdInf(
                        gasMtstNo,
                        sv101.ckrtGmscJtkb(),
                        sv101.mtgMtstNo()),
                isrrkrUpdInf.duknIsrsCd());
        ClosingCancelFeasibilityJudgmentMeterReadingDataOutput sv099 =
                meterReadingDataJudgment.execute(isrrkrUpdInf.duknIsrsCd(), gasMtstNo, trkKnsYmd);
        gasMeterLocationInfoUpdateForRevival.execute(gasMtstNo, sv101.kasYmd(), sv101.hesnYmd());
        relocationDestinationManagementDeletion.execute(
                isrrkrUpdInf.isrrGmssCd(),
                gasMtstNo,
                primarySvckya(sv103).svkyNo());
        ServiceContractRevivalTgRetailOutput sv106 = executeServiceContractRevival(trkKnsYmd, sv101, sv103);
        PaymentContractRevivalOutput sv112 = executePaymentContractRevivalIfNeeded(sv103, sv106);
        executeMeterReadingDelete(sv099);
        executeTransferWorkHistoryCreation(request, trkKnsYmd, sv103, sv106);
        executeBillingCollectionPlatformIntegration(trkKnsYmd, sv103, sv106, sv112);

        return new ClosingOnlineReportCancelResponse(
                requestContext.getCorrelationId().orElseThrow(() -> new IllegalStateException("correlationId is not set")),
                null,
                null,
                null);
    }

    private PaymentContractRevivalOutput executePaymentContractRevivalIfNeeded(
            ClosingCancelFeasibilityJudgmentRetailInfoOutput sv103,
            ServiceContractRevivalTgRetailOutput sv106) {
        if (!Boolean.TRUE.equals(sv106.sihakyFkktFlg())) {
            return null;
        }
        jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.SvckyaInf primarySvckya = primarySvckya(sv103);
        return paymentContractRevival.execute(primarySvckya.svkyNo(), primarySvckya.syaKbn());
    }

    private void executeMeterReadingDelete(ClosingCancelFeasibilityJudgmentMeterReadingDataOutput sv099) {
        int deleteCount = ktmgmkkDataAccess.deleteByGasMtstNoAndKnsYmdAndKnsKkaTsno(
                sv099.gasMtstNo(),
                sv099.knsYmd(),
                sv099.knsKkaTsno(),
                null);
        if (deleteCount == 0) {
            throw DataInconsistencySupport.create(
                    ktmgmkkDataAccess.tableName(),
                    KtmgmkkDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_KNS_YMD_AND_KNS_KKA_TSNO.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", sv099.gasMtstNo())
                            .put("knsYmd", sv099.knsYmd())
                            .put("knsKkaTsno", sv099.knsKkaTsno()));
        }
    }

    private void executeBillingCollectionPlatformIntegration(

                                sv112.sihakyInf().sikSvcShbt(),

    }

    private void bindCorrelationIdFromBodyIfPresent(ClosingOnlineReportCancelRequest request) {
        if (request == null) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId != null) {
            requestContext.setCorrelationId(CorrelationIdResolver.resolveUuidV4(correlationId));
        }
    }

    private void bindUuidCorrelationIdForResponse() {
        String correlationId = CorrelationIdResolver.resolveUuidV4(
                requestContext.getCorrelationId().orElse(null));
        requestContext.setCorrelationId(correlationId);
        MDC.put(MDC_CORRELATION_ID_KEY, correlationId);
    }

    private jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.model.SvckyaInf primarySvckya(
            ClosingCancelFeasibilityJudgmentRetailInfoOutput sv103) {
        return sv103.svckyaInfList().get(0);
    }
