    ClosingSupplyStopReportCancelResponse handle(ClosingSupplyStopReportCancelRequest request) {
        long gasMtstNo = request.gasMtstNo();
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd());
        ClosingSupplyStopReportCancelRequest.IsrrkrUpdInf requestIsrrkrUpdInf =
                Objects.requireNonNull(request.isrrkrUpdInf(), "isrrkrUpdInf");

        followStatusJudgment.execute(
                request.trkIsrrGymCd(),
                gasMtstNo,
                trkKnsYmd,
                processDivisionClosingCancel);

        ClosingCancelFeasibilityJudgmentEquipmentStatusOutput equipmentStatus =
                equipmentStatusJudgment.execute(gasMtstNo, trkKnsYmd);

        dedicatedMeterLocationJudgment.execute(
                trkKnsYmd,
                new DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentGmtsetdInf(
                        equipmentStatus.ckrtGmscJtkb(),
                        equipmentStatus.mtgMtstNo()));

        ClosingCancelFeasibilityJudgmentRetailInfoOutput retailInfo = retailInfoJudgment.execute(
                trkKnsYmd,
                new GmtsetdInf(
                        gasMtstNo,
                        equipmentStatus.ckrtGmscJtkb(),
                        equipmentStatus.mtgMtstNo()),
                requestIsrrkrUpdInf.duknIsrsCd());

        meterReadingDataJudgment.execute(
                requestIsrrkrUpdInf.duknIsrsCd(),
                gasMtstNo,
                trkKnsYmd);

        gasMeterLocationInfoUpdateForRevival.execute(
                gasMtstNo,
                equipmentStatus.kasYmd(),
                equipmentStatus.hesnYmd());

        deleteSupplyStopManagement(retailInfo);
        clearSupplyStopSettlementStatuses(retailInfo.svckyaInfList(), request.hukkSysShbt());
        executeCancelTransferWorkHistory(request, gasMtstNo, trkKnsYmd, requestIsrrkrUpdInf);
        executeBillingCollectionPlatformIntegration(retailInfo, trkKnsYmd);

        return new ClosingSupplyStopReportCancelResponse(
                requestContext.getCorrelationId().orElseThrow(
                        () -> new IllegalStateException("correlationId is not set")),
                null,
                null,
                null);
    }

    private void deleteSupplyStopManagement(ClosingCancelFeasibilityJudgmentRetailInfoOutput retailInfo) {
        if (retailInfo.kkytskrInf() == null) {
            throw DataInconsistencySupport.create(
                    kkytskrDataAccess.tableName(),
                    KkytskrDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_HESN_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", null)
                            .put("hesnYmd", null)
                            .put("recUpdJnjNo", null));
        }

        int deleted = kkytskrDataAccess.deleteByGasMtstNoAndHesnYmd(
                retailInfo.kkytskrInf().gasMtstNo(),
                retailInfo.kkytskrInf().hesnYmd(),
                null);
        if (deleted == 0) {
            throw DataInconsistencySupport.create(
                    kkytskrDataAccess.tableName(),
                    KkytskrDataAccess.CallSpecs.DELETE_BY_GAS_MTST_NO_AND_HESN_YMD.identifier(),
                    new FlatJsonBuilder()
                            .put("gasMtstNo", retailInfo.kkytskrInf().gasMtstNo())
                            .put("hesnYmd", retailInfo.kkytskrInf().hesnYmd())
                            .put("recUpdJnjNo", null));
        }
    }

                                .put("updUsrId", SERVICE_USER_ID)
                                .put("updKtnKinoId", decodedOriginFunctionId));
            }
        }

    private void executeCancelTransferWorkHistory(

    private void executeBillingCollectionPlatformIntegration(
                ClosingCancelFeasibilityJudgmentRetailInfoOutput retailInfo,
                LocalDate trkKnsYmd) {
        List<SvckyaInf> serviceContracts = retailInfo.svckyaInfList();
        if (serviceContracts == null || serviceContracts.isEmpty()) {
            billingCollectionPlatformIntegrationAtClosingCancel.execute(
                    Boolean.FALSE,
                    Boolean.FALSE,
                    new BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf(null, null, null),
                    new BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf(
                            null,
                            null,
                            null,
                            null,
                            null,
                            null),
                    retailInfo.svmstkrInfList() != null && retailInfo.svmstkrInfList().size() > 1,
                    trkKnsYmd,
                    retailWorkCode);
            return;
        }

        SvckyaInf firstServiceContract = serviceContracts.getFirst();
        billingCollectionPlatformIntegrationAtClosingCancel.execute(
                Boolean.FALSE,
                Boolean.FALSE,
                new BillingCollectionPlatformIntegrationAtClosingCancelSihakyInf(null, null, null),
                new BillingCollectionPlatformIntegrationAtClosingCancelSvckyaInf(
                        firstServiceContract.svkyNo(),
                        firstServiceContract.skno(),
                        firstServiceContract.rkkiSttYmd(),
                        firstServiceContract.rkkiEndYmd(),
                        firstServiceContract.sqtiKiyNo(),
                        firstServiceContract.kkysSisnJtkb()),
                retailInfo.svmstkrInfList() != null && retailInfo.svmstkrInfList().size() > 1,
                trkKnsYmd,
                retailWorkCode);
    }

    private String decodeOriginFunctionId(String hukkSysShbt) {
        return kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
    }

    private void bindCorrelationIdFromBodyIfPresent(ClosingSupplyStopReportCancelRequest request) {
        if (request == null) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId != null) {
            requestContext.setCorrelationId(correlationId);
        }
    }
