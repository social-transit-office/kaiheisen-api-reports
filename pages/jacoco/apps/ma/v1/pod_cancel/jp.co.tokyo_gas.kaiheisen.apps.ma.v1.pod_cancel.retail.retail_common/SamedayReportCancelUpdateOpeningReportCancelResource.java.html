@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-041", name = "当日報告取消更新-オンライン開栓報告取消-小売-共通 API")

    private static final String CORRELATION_ID_HEADER_NAME = "X-Correlation-Id";

    private static final String SHR_KBN_APPLIED = "1";

    @Context
    HttpHeaders httpHeaders;

    /**

    @Inject

        super(requestContext, requestDtoValidator, clock);
        this.openingCancelBffCallErrorJudgment = Objects.requireNonNull(
                openingCancelBffCallErrorJudgment,
                "openingCancelBffCallErrorJudgment");
        this.retailTypeJudgment = Objects.requireNonNull(retailTypeJudgment, "retailTypeJudgment");
        this.onlineReportTgRetailService = Objects.requireNonNull(
                onlineReportTgRetailService,
                "onlineReportTgRetailService");
    }

    /**
     * MA-041 を実行する。

    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(summary = "当日報告取消更新-オンライン開栓報告取消-小売-共通")
    @APIResponse(
            responseCode = "200",
            description = "正常終了",

                    schema = @Schema(

            description = "入力不正・業務エラー",

            SamedayReportCancelUpdateOpeningReportCancelRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        try {
            return Response.ok(execute(request, () -> handle(request)))
                    .type(MediaType.APPLICATION_JSON)
                    .build();
        } catch (ValidationException | BusinessRuleException exception) {
            return badRequestErrorResponse(exception);
        }
    }

    /**
     * MA-041 の実処理を実行する。
     */
    public SamedayReportCancelUpdateOpeningReportCancelResponse handle(
            SamedayReportCancelUpdateOpeningReportCancelRequest request) {
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd().trim());

        OpeningCancelBffCallErrorJudgmentOutput historyDetailCode =
                openingCancelBffCallErrorJudgment.execute(request.gasMtstNo(), request.trkIsrrGymCd());
        RetailTypeJudgmentOutput retailType =
                retailTypeJudgment.execute(SHR_KBN_APPLIED, request.gasMtstNo(), trkKnsYmd);

        onlineReportTgRetailService.execute(
                request.gasMtstNo(),
                trkKnsYmd,
                request.trkIsrrGymCd(),
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.rirekiMemo(),
                toOnlineReportTkssvkyInfo(retailType),
                toHistoryInfo(historyDetailCode));

        return response();
    }

    private SamedayReportCancelUpdateOpeningReportCancelResponse response() {
        return new SamedayReportCancelUpdateOpeningReportCancelResponse(
                resolveResponseCorrelationId(),
                null,
                null,
                null,
                null,
                null);
    }

    private Response badRequestErrorResponse(KaiheisenException exception) {
        ErrorResponseBody body = MaErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(MaErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(new SamedayReportCancelUpdateOpeningReportCancelResponse(
                        resolveResponseCorrelationId(),
                        body.errorCode(),
                        body.errorDescription(),
                        body.errorDetailList(),
                        body.crsMsgId(),
                        body.crsMsgEmbStrList()))
                .build();
    }

    private static OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailTkssvkyInf
            toOnlineReportTkssvkyInfo(RetailTypeJudgmentOutput retailType) {
        TkssvkyInfo tkssvkyInfo = retailType.tkssvkyInf();
        if (tkssvkyInfo == null) {
            return null;
        }
        return new OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailTkssvkyInf(
                tkssvkyInfo.tksuSvkyNo(),
                tkssvkyInfo.tkysYmd(),
                tkssvkyInfo.tksuSvksJysb(),
                tkssvkyInfo.tksuKijsCd());
    }

    private static OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailIsrrkrUpdInf
            toHistoryInfo(OpeningCancelBffCallErrorJudgmentOutput output) {
        return new OpeningCancelReportCancelAvailabilityCheckOnlineReportTgRetailIsrrkrUpdInf(
                output.duknIsrgCd(),
                output.duknIsrsCd(),
                output.isrrGymCd(),
                output.isrrGmssCd());
    }

    private void bindCorrelationIdFromBodyIfPresent(
            SamedayReportCancelUpdateOpeningReportCancelRequest request) {
        if (request == null) {
            return;
        }
        if (hasRequestCorrelationIdHeader()) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId == null) {
            return;
        }
        requestContext.setCorrelationId(correlationId);
    }

    void setHttpHeaders(HttpHeaders httpHeaders) {
        this.httpHeaders = httpHeaders;
    }

    private boolean hasRequestCorrelationIdHeader() {
        if (httpHeaders == null) {
            return false;
        }
        return normalizeOptional(httpHeaders.getHeaderString(CORRELATION_ID_HEADER_NAME)) != null;
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }
