/**
 * MA-042 当日報告取消更新-閉栓エラーフォロー取消-小売-共通 API のリソース実装。
 */
@Path(SamedayReportCancelUpdateClosingErrorFollowCancelEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-042", name = "当日報告取消更新-閉栓エラーフォロー取消-小売-共通 API")
@Tag(name = "当日報告取消更新-閉栓エラーフォロー取消-小売-共通", description = "MA-042")
public class SamedayReportCancelUpdateClosingErrorFollowCancelResource
        extends BaseResource<SamedayReportCancelUpdateClosingErrorFollowCancelRequest> {

    private static final String SHR_KBN_NEAREST = "2";

    private static final String INVALID_RETAIL_TYPE_MESSAGE =
            "ガスメーター設置場所番号の値が不正のため、処理を終了しました";

    private final ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeJudgment;

    private final RetailTypeJudgment retailTypeJudgment;

    private final String retailTypeTokyoGas;

    private final String retailTypeOther;

    private final ClosingCancelClosingErrorFollowCancelTgRetailResource tgRetailResource;

    private final ClosingCancelClosingErrorFollowCancelOtherRetailResource otherRetailResource;

    /**
     * MA-042 Resource を生成する。
     */
    @Inject
    public SamedayReportCancelUpdateClosingErrorFollowCancelResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            KaiheisenConstantService kaiheisenConstantService,
            ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeJudgment,
            RetailTypeJudgment retailTypeJudgment,
            ClosingCancelClosingErrorFollowCancelTgRetailResource tgRetailResource,
            ClosingCancelClosingErrorFollowCancelOtherRetailResource otherRetailResource) {
        super(requestContext, requestDtoValidator, clock);
        this.historyDetailCodeJudgment = historyDetailCodeJudgment;
        this.retailTypeJudgment = retailTypeJudgment;
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.retailTypeTokyoGas = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TG_RETAIL);
        this.retailTypeOther = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.OTHER_RETAIL);
        this.tgRetailResource = tgRetailResource;
        this.otherRetailResource = otherRetailResource;
    }

    /**
     * MA-042 を実行する。
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(summary = "当日報告取消更新-閉栓エラーフォロー取消-小売-共通")
    @APIResponse(
            responseCode = "200",
            description = "正常終了",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            implementation = SamedayReportCancelUpdateClosingErrorFollowCancelResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            implementation = SamedayReportCancelUpdateClosingErrorFollowCancelResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public Response endpoint(
            @RequestBody(required = true,
                    content = @Content(
                            mediaType = MediaType.APPLICATION_JSON,
                            schema = @Schema(
                                    implementation =
                                            SamedayReportCancelUpdateClosingErrorFollowCancelRequest.class)))
            SamedayReportCancelUpdateClosingErrorFollowCancelRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        try {
            return Response.ok(execute(request, () -> handle(request)))
                    .type(MediaType.APPLICATION_JSON)
                    .build();
        } catch (BusinessRuleException exception) {
            return businessRuleErrorResponse(exception);
        }
    }

    /**
     * MA-042 の実処理を実行する。
     */
    public SamedayReportCancelUpdateClosingErrorFollowCancelResponse handle(
            SamedayReportCancelUpdateClosingErrorFollowCancelRequest request) {
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd().trim());

        ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode =
                historyDetailCodeJudgment.execute(request.gasMtstNo(), request.trkIsrrGymCd());
        RetailTypeJudgmentOutput retailType =
                retailTypeJudgment.execute(SHR_KBN_NEAREST, request.gasMtstNo(), trkKnsYmd);

        String judgmentCode = retailType.kuHntKbn();
        if (!retailTypeTokyoGas.equals(judgmentCode) && !retailTypeOther.equals(judgmentCode)) {
            throw new KaiheisenException(INVALID_RETAIL_TYPE_MESSAGE);
        }

        if (retailTypeTokyoGas.equals(judgmentCode)) {
            tgRetailResource.endpoint(toTgRetailRequest(request, historyDetailCode));
        } else {
            otherRetailResource.endpoint(toOtherRetailRequest(request, historyDetailCode, retailType));
        }

        return response();
    }

    private SamedayReportCancelUpdateClosingErrorFollowCancelResponse response() {
        return new SamedayReportCancelUpdateClosingErrorFollowCancelResponse(
                resolveResponseCorrelationId(),
                null,
                null,
                null,
                null,
                null);
    }

    private Response businessRuleErrorResponse(BusinessRuleException exception) {
        ErrorResponseBody body = MaErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(MaErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(new SamedayReportCancelUpdateClosingErrorFollowCancelResponse(
                        resolveResponseCorrelationId(),
                        body.errorCode(),
                        body.errorDescription(),
                        body.errorDetailList(),
                        body.crsMsgId(),
                        body.crsMsgEmbStrList()))
                .build();
    }

    private static ClosingCancelClosingErrorFollowCancelTgRetailRequest toTgRetailRequest(
            SamedayReportCancelUpdateClosingErrorFollowCancelRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode) {
        return new ClosingCancelClosingErrorFollowCancelTgRetailRequest(
                request.gasMtstNo(),
                request.trkKnsYmd(),
                request.trkIsrrGymCd(),
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.rirekiMemo(),
                new ClosingCancelClosingErrorFollowCancelTgRetailRequest.IsrrkrUpdInf(
                        historyDetailCode.duknIsrgCd(),
                        historyDetailCode.duknIsrsCd(),
                        historyDetailCode.isrrGymCd(),
                        historyDetailCode.isrrGmssCd()),
                request.hukkSysShbt(),
                request.correlationId(),
                request.systemDatetime());
    }

    private static ClosingCancelClosingErrorFollowCancelOtherRetailRequest toOtherRetailRequest(
            SamedayReportCancelUpdateClosingErrorFollowCancelRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode,
            RetailTypeJudgmentOutput retailType) {
        return new ClosingCancelClosingErrorFollowCancelOtherRetailRequest(
                request.gasMtstNo(),
                request.trkKnsYmd(),
                request.trkIsrrGymCd(),
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.rirekiMemo(),
                new ClosingCancelClosingErrorFollowCancelOtherRetailRequest.TskgmstInf(
                        retailType.tskgmstInf().tksuSvkyNo(),
                        retailType.tskgmstInf().skno()),
                new ClosingCancelClosingErrorFollowCancelOtherRetailRequest.TkssvkyInf(
                        retailType.tkssvkyInf().tksuSvkyNo(),
                        retailType.tkssvkyInf().skno()),
                new ClosingCancelClosingErrorFollowCancelOtherRetailRequest.IsrrkrUpdInf(
                        historyDetailCode.duknIsrgCd(),
                        historyDetailCode.duknIsrsCd(),
                        historyDetailCode.isrrGymCd(),
                        historyDetailCode.isrrGmssCd()),
                request.hukkSysShbt(),
                request.systemDatetime());
    }

    private void bindCorrelationIdFromBodyIfPresent(SamedayReportCancelUpdateClosingErrorFollowCancelRequest request) {
        if (request == null || request.correlationId() == null || request.correlationId().isBlank()) {
            return;
        }
        requestContext.setCorrelationId(request.correlationId());
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }
}
