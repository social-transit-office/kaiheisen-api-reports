/**
 * MA-043 当日報告取消更新-オンライン閉栓報告取消-小売-共通 API のリソース実装。
 */
@Path(SamedayReportCancelUpdateClosingReportCancelEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-043", name = "当日報告取消更新-オンライン閉栓報告取消-小売-共通 API")
@Tag(name = "当日報告取消更新-オンライン閉栓報告取消-小売-共通", description = "MA-043")
public class SamedayReportCancelUpdateClosingReportCancelResource
        extends BaseResource<SamedayReportCancelUpdateClosingReportCancelRequest> {

    private static final String SUPPLY_STOP_REPORT_RETAIL =
            ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.SUPPLY_OUTAGE_REPORT_RETAIL;

    private static final String ONLINE_CLOSURE_REPORT_RETAIL =
            ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.ONLINE_CLOSURE_REPORT_RETAIL;

    private final ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeJudgment;

    private final ClosingSupplyStopReportCancelResource closingSupplyStopReportCancelResource;

    private final ClosingOnlineReportCancelResource closingOnlineReportCancelResource;

    /**
     * MA-043 Resource を生成する。
     */
    @Inject
    public SamedayReportCancelUpdateClosingReportCancelResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeJudgment,
            ClosingSupplyStopReportCancelResource closingSupplyStopReportCancelResource,
            ClosingOnlineReportCancelResource closingOnlineReportCancelResource) {
        super(requestContext, requestDtoValidator, clock);
        this.historyDetailCodeJudgment = Objects.requireNonNull(
                historyDetailCodeJudgment,
                "historyDetailCodeJudgment");
        this.closingSupplyStopReportCancelResource = Objects.requireNonNull(
                closingSupplyStopReportCancelResource,
                "closingSupplyStopReportCancelResource");
        this.closingOnlineReportCancelResource = Objects.requireNonNull(
                closingOnlineReportCancelResource,
                "closingOnlineReportCancelResource");
    }

    /**
     * MA-043 を実行する。
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(summary = "当日報告取消更新-オンライン閉栓報告取消-小売-共通")
    @APIResponse(
            responseCode = "200",
            description = "正常終了",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            implementation = SamedayReportCancelUpdateClosingReportCancelResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CommonErrorResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public SamedayReportCancelUpdateClosingReportCancelResponse endpoint(
            @RequestBody(required = true,
                    content = @Content(
                            mediaType = MediaType.APPLICATION_JSON,
                            schema = @Schema(
                                    implementation =
                                            SamedayReportCancelUpdateClosingReportCancelRequest.class)))
            SamedayReportCancelUpdateClosingReportCancelRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        return execute(request, () -> handle(request));
    }

    /**
     * MA-043 の実処理を実行する。
     */
    public SamedayReportCancelUpdateClosingReportCancelResponse handle(
            SamedayReportCancelUpdateClosingReportCancelRequest request) {
        ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode =
                historyDetailCodeJudgment.execute(request.gasMtstNo(), request.trkIsrrGymCd());

        String retailBusinessCode = historyDetailCode.isrrGymCd();
        if (SUPPLY_STOP_REPORT_RETAIL.equals(retailBusinessCode)) {
            closingSupplyStopReportCancelResource.endpoint(toClosingSupplyStopReportCancelRequest(request, historyDetailCode));
        } else if (ONLINE_CLOSURE_REPORT_RETAIL.equals(retailBusinessCode)) {
            closingOnlineReportCancelResource.endpoint(toClosingOnlineReportCancelRequest(request, historyDetailCode));
        }

        return response();
    }

    private SamedayReportCancelUpdateClosingReportCancelResponse response() {
        return new SamedayReportCancelUpdateClosingReportCancelResponse(
                resolveResponseCorrelationId(),
                null,
                null,
                null,
                null,
                null);
    }

    private static ClosingSupplyStopReportCancelRequest toClosingSupplyStopReportCancelRequest(
            SamedayReportCancelUpdateClosingReportCancelRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode) {
        return new ClosingSupplyStopReportCancelRequest(
                request.gasMtstNo(),
                request.trkKnsYmd(),
                historyDetailCode.isrrGymCd(),
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.rirekiMemo(),
                new ClosingSupplyStopReportCancelRequest.IsrrkrUpdInf(
                        historyDetailCode.duknIsrgCd(),
                        historyDetailCode.duknIsrsCd(),
                        historyDetailCode.isrrGymCd(),
                        historyDetailCode.isrrGmssCd()),
                request.hukkSysShbt(),
                request.correlationId(),
                request.systemDatetime());
    }

    private static ClosingOnlineReportCancelRequest toClosingOnlineReportCancelRequest(
            SamedayReportCancelUpdateClosingReportCancelRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCode) {
        return new ClosingOnlineReportCancelRequest(
                String.valueOf(request.gasMtstNo()),
                request.trkKnsYmd(),
                historyDetailCode.isrrGymCd(),
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.rirekiMemo(),
                new ClosingOnlineReportCancelRequest.IsrrkrUpdInf(
                        historyDetailCode.duknIsrgCd(),
                        historyDetailCode.duknIsrsCd(),
                        historyDetailCode.isrrGymCd(),
                        historyDetailCode.isrrGmssCd()),
                request.hukkSysShbt(),
                request.correlationId(),
                request.systemDatetime());
    }

    private void bindCorrelationIdFromBodyIfPresent(SamedayReportCancelUpdateClosingReportCancelRequest request) {
        if (request == null || request.correlationId() == null || request.correlationId().isBlank()) {
            return;
        }
        requestContext.setCorrelationId(request.correlationId());
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }
}
