/**
 * MA-040 当日報告取消更新-開栓エラーフォロー取消-小売-共通 API のリソース実装。
 * 400 系は MA-040 個別契約に合わせて本クラスでレスポンス DTO を組み立て、500 系は共通 ExceptionMapper へ委譲する。
 */
@Path(SamedayReportCancelUpdateOpeningErrorFollowCancelEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
@MainService(identifier = "MA-040", name = "当日報告取消更新-開栓エラーフォロー取消-小売-共通 API")
@Tag(name = "当日報告取消更新-開栓エラーフォロー取消-小売-共通", description = "MA-040")
public class SamedayReportCancelUpdateOpeningErrorFollowCancelResource
        extends BaseResource<SamedayReportCancelUpdateOpeningErrorFollowCancelRequest> {

    private static final String SHR_KBN_APPLIED = "1";

    private static final String INVALID_RETAIL_TYPE_MESSAGE =
            "ガスメーター設置場所番号の値が不正のため、処理を終了しました";

    private final OpeningCancelBffCallErrorJudgment openingCancelBffCallErrorJudgment;

    private final RetailTypeJudgment retailTypeJudgment;

    private final String retailTypeTokyoGas;

    private final String retailTypeOther;

    private final OpeningCancelReportCancelAvailabilityCheckErrorFollowTgRetail tgRetailService;

    private final OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetail otherRetailService;

    /**
     * MA-040 Resource を生成する。
     */
    @Inject
    SamedayReportCancelUpdateOpeningErrorFollowCancelResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            KaiheisenConstantService kaiheisenConstantService,
            OpeningCancelBffCallErrorJudgment openingCancelBffCallErrorJudgment,
            RetailTypeJudgment retailTypeJudgment,
            OpeningCancelReportCancelAvailabilityCheckErrorFollowTgRetail tgRetailService,
            OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetail otherRetailService) {
        super(requestContext, requestDtoValidator, clock);
        this.openingCancelBffCallErrorJudgment = openingCancelBffCallErrorJudgment;
        this.retailTypeJudgment = retailTypeJudgment;
        KaiheisenConstantService nonNullKaiheisenConstantService =
                Objects.requireNonNull(kaiheisenConstantService, "kaiheisenConstantService");
        this.retailTypeTokyoGas = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TG_RETAIL);
        this.retailTypeOther = RequiredConstantResolver.requireKaiheisenConstant(
                nonNullKaiheisenConstantService,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.TYPE,
                ConstantLookupKeys.Kaiheisen.RetailTypeJudgmentDivision.OTHER_RETAIL);
        this.tgRetailService = tgRetailService;
        this.otherRetailService = otherRetailService;
    }

    /**
     * MA-040 を実行する。
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(summary = "当日報告取消更新-開栓エラーフォロー取消-小売-共通")
    @APIResponse(
            responseCode = "200",
            description = "正常終了",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            implementation = SamedayReportCancelUpdateOpeningErrorFollowCancelResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(
                            implementation = SamedayReportCancelUpdateOpeningErrorFollowCancelResponse.class)))
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = KaiheisenErrorResponse.class)))
    public Response endpoint(
            @RequestBody(required = true,
                    content = @Content(
                            mediaType = MediaType.APPLICATION_JSON,
                            schema = @Schema(
                                    implementation =
                                            SamedayReportCancelUpdateOpeningErrorFollowCancelRequest.class)))
            SamedayReportCancelUpdateOpeningErrorFollowCancelRequest request) {
        bindCorrelationIdFromBodyIfPresent(request);
        try {
            return Response.ok(execute(request, () -> handle(request)))
                    .type(MediaType.APPLICATION_JSON)
                    .build();
        } catch (ValidationException | BusinessRuleException exception) {
            return badRequestErrorResponse(exception);
        }
    }

    /**
     * MA-040 の実処理を実行する。
     */
    public SamedayReportCancelUpdateOpeningErrorFollowCancelResponse handle(
            SamedayReportCancelUpdateOpeningErrorFollowCancelRequest request) {
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd().trim());

        OpeningCancelBffCallErrorJudgmentOutput historyDetailCode =
                openingCancelBffCallErrorJudgment.execute(request.gasMtstNo(), request.trkIsrrGymCd());
        RetailTypeJudgmentOutput retailType =
                retailTypeJudgment.execute(SHR_KBN_APPLIED, request.gasMtstNo(), trkKnsYmd);

        String judgmentCode = retailType.kuHntKbn();
        if (!retailTypeTokyoGas.equals(judgmentCode) && !retailTypeOther.equals(judgmentCode)) {
            throw new KaiheisenException(INVALID_RETAIL_TYPE_MESSAGE);
        }

        if (retailTypeTokyoGas.equals(judgmentCode)) {
            tgRetailService.execute(
                            request.gasMtstNo(),
                            trkKnsYmd,
                            request.trkIsrrGymCd(),
                            request.hukkUsrId(),
                            request.hukkUnmKj(),
                            request.rirekiMemo(),
                            toTkssvkyInfo(retailType),
                            toTgRetailHistoryInfo(historyDetailCode));
        } else {
            otherRetailService.execute(
                            request.gasMtstNo(),
                            trkKnsYmd,
                            request.trkIsrrGymCd(),
                            request.hukkUsrId(),
                            request.hukkUnmKj(),
                            request.rirekiMemo(),
                            toOtherRetailTkssvkyInfo(retailType),
                            toOtherRetailHistoryInfo(historyDetailCode));
        }

        return response();
    }

    private SamedayReportCancelUpdateOpeningErrorFollowCancelResponse response() {
        return new SamedayReportCancelUpdateOpeningErrorFollowCancelResponse(
                resolveResponseCorrelationId(),
                null,
                null,
                null,
                null,
                null);
    }

    private Response badRequestErrorResponse(KaiheisenException exception) {
        ErrorResponseBody body = MaErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(MaErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(new SamedayReportCancelUpdateOpeningErrorFollowCancelResponse(
                        resolveResponseCorrelationId(),
                        body.errorCode(),
                        body.errorDescription(),
                        body.errorDetailList(),
                        body.crsMsgId(),
                        body.crsMsgEmbStrList()))
                .build();
    }

    private static TkssvkyInfo toTkssvkyInfo(RetailTypeJudgmentOutput retailType) {
        TkssvkyInfo tkssvkyInfo = retailType.tkssvkyInf();
        if (tkssvkyInfo == null) {
            return null;
        }
        return new TkssvkyInfo(
                tkssvkyInfo.tksuSvkyNo(),
                tkssvkyInfo.skno(),
                tkssvkyInfo.tksuSvkyJtkb(),
                tkssvkyInfo.tksuSvksJysb(),
                tkssvkyInfo.tksuSvkeJysb(),
                tkssvkyInfo.tkysYmd(),
                tkssvkyInfo.tyenYmd(),
                tkssvkyInfo.hesnYtiYmd(),
                tkssvkyInfo.tkykShbt(),
                tkssvkyInfo.tksuKijsCd(),
                tkssvkyInfo.sutiKskhFlg(),
                tkssvkyInfo.syoKyoTrno(),
                tkssvkyInfo.recUpdJnjNo());
    }

    private static IsrrkrUpdInf toTgRetailHistoryInfo(OpeningCancelBffCallErrorJudgmentOutput output) {
        return new IsrrkrUpdInf(
                output.duknIsrgCd(),
                output.duknIsrsCd(),
                output.isrrGymCd(),
                output.isrrGmssCd());
    }

    private static OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetailTkssvkyInf
            toOtherRetailTkssvkyInfo(RetailTypeJudgmentOutput retailType) {
        TkssvkyInfo tkssvkyInfo = retailType.tkssvkyInf();
        if (tkssvkyInfo == null) {
            return null;
        }
        return new OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetailTkssvkyInf(
                tkssvkyInfo.tksuSvkyNo(),
                tkssvkyInfo.tkysYmd(),
                tkssvkyInfo.tksuSvksJysb(),
                tkssvkyInfo.tksuKijsCd(),
                tkssvkyInfo.syoKyoTrno());
    }

    private static OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetailIsrrkrUpdInf
            toOtherRetailHistoryInfo(OpeningCancelBffCallErrorJudgmentOutput output) {
        return new OpeningCancelReportCancelAvailabilityCheckErrorFollowOtherRetailIsrrkrUpdInf(
                output.duknIsrgCd(),
                output.duknIsrsCd(),
                output.isrrGymCd(),
                output.isrrGmssCd());
    }

    private void bindCorrelationIdFromBodyIfPresent(
            SamedayReportCancelUpdateOpeningErrorFollowCancelRequest request) {
        if (request == null) {
            return;
        }
        if (requestContext.getCorrelationId().isPresent()) {
            return;
        }
        String correlationId = normalizeOptional(request.correlationId());
        if (correlationId == null) {
            return;
        }
        requestContext.setCorrelationId(correlationId);
    }

    private String resolveResponseCorrelationId() {
        return requestContext.getCorrelationId().orElseGet(() -> {
            String generated = CorrelationIdResolver.resolve(null);
            requestContext.setCorrelationId(generated);
            return generated;
        });
    }
