    SamedayReportCancelTargetGetClosingOtherRetailResponse handle(
            SamedayReportCancelTargetGetClosingOtherRetailRequest request) {
        LocalDate targetMeterReadingDate = Rfc3339LocalDateParser.parse(request.trkKnsYmd());
        validateNotFuture(targetMeterReadingDate);

        long gasMtstNo = Long.parseLong(request.gasMtstNo());
        long contractCustomerNo = Long.parseLong(request.syoKyoTrno());

        GmtsetdRecord gmtsetdRecord = gmtsetdDataAccess
                .findByGasMtstNoAndTkysYmdAndTyenYmd(gasMtstNo, targetMeterReadingDate)
                .orElseThrow(() -> new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND));

        try {
            openingClosingInProgressErrorJudgment.execute(
                    gmtsetdRecord.gmtKhsJtkb(),
                    openingClosingErrorJudgmentDivision);
        } catch (KaisenRuleException exception) {
            throw new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND);
        }

        KiycstdRecord kiycstdRecord = kiycstdDataAccess.findByKyoTrkNo(contractCustomerNo)
                .orElseThrow(() -> DataInconsistencySupport.create(
                        kiycstdDataAccess.tableName(),
                        KiycstdDataAccess.CallSpecs.FIND_BY_KYO_TRK_NO.identifier(),
                        new FlatJsonBuilder().put("kyoTrkNo", contractCustomerNo)));

        MeterReadingRelatedInfoEditProcessOutput meterReadingOutput =
                meterReadingRelatedInfoEditProcess.execute(
                        gasMtstNo,
                        targetMeterReadingDate,
                        gmtsetdRecord.hesnJysb(),
                        meterReadingEditDivision);

        return toResponse(
                kiycstdRecord,
                gmtsetdRecord,
                meterReadingOutput,
                resolveResponseCorrelationId());
    }

    private void validateNotFuture(LocalDate targetMeterReadingDate) {
        LocalDate systemDate = requestContext.getSystemDatetime()
                .orElseThrow(() -> new IllegalStateException("systemDatetime is not set"))
                .toLocalDate();
        if (targetMeterReadingDate.isAfter(systemDate)) {
            throw new BusinessRuleException(ERROR_CODE_TARGET_NOT_FOUND);
        }
    }
