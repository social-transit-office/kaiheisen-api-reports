    /**
     * 閉栓取消処理-閉栓報告取消-小売-他小売 API の業務処理を実行する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    public ClosingCancelClosingReportCancelOtherRetailResponse handle(
            ClosingCancelClosingReportCancelOtherRetailRequest request) {
        LocalDate trkKnsYmd = Rfc3339LocalDateParser.parse(request.trkKnsYmd().trim());
        ClosingCancelClosingReportCancelOtherRetailRequest.IsrrkrUpdInf isrrkrUpdInf =
                request.isrrkrUpdInf();

        ClosingCancelFeasibilityJudgmentFollowStatusOutput followStatusOutput =
                closingCancelFeasibilityJudgmentFollowStatus.execute(
                        request.trkIsrrGymCd(),
                        request.gasMtstNo(),
                        trkKnsYmd,
                        SHR_KBN_CANCEL);

        Long customerRegistrationNo;
        String workerEmployeeNo;
        String workerName;
        String workplaceCode;

        if (FOLLOW_NOT_YET_STARTED.equals(followStatusOutput.hseflFlwMcksKbn())) {
            HseflInf hseflInf = followStatusOutput.hseflInf();
            deleteHseflOrThrow(hseflInf);
            customerRegistrationNo = hseflInf.syoKyoTrno();
            workerEmployeeNo = hseflInf.sgytSinNo();
            workerName = hseflInf.sgytSinnKj();
            workplaceCode = hseflInf.hssgNwkCd();
        } else {
            ClosingCancelFeasibilityJudgmentTransportContractStatusOutput transportContractStatusOutput =
                    closingCancelFeasibilityJudgmentTransportContractStatus.execute(request.gasMtstNo());

            ClosingCancelFeasibilityJudgmentEquipmentStatusOutput equipmentStatusOutput =
                    closingCancelFeasibilityJudgmentEquipmentStatus.execute(
                            request.gasMtstNo(),
                            trkKnsYmd);

            dedicatedMeterLocationAcquisitionAndClosingCancelJudgment.execute(
                    trkKnsYmd,
                    new DedicatedMeterLocationAcquisitionAndClosingCancelJudgmentGmtsetdInf(
                            equipmentStatusOutput.ckrtGmscJtkb(),
                            equipmentStatusOutput.mtgMtstNo()));

            ClosingCancelFeasibilityJudgmentMeterReadingDataOutput meterReadingDataOutput =
                    closingCancelFeasibilityJudgmentMeterReadingData.execute(
                            isrrkrUpdInf == null ? null : isrrkrUpdInf.duknIsrsCd(),
                            request.gasMtstNo(),
                            trkKnsYmd);

            gasMeterLocationInfoUpdateForRevival.execute(
                    request.gasMtstNo(),
                    equipmentStatusOutput.kasYmd(),
                    equipmentStatusOutput.hesnYmd());

            TskgmstRecord tskgmstRecord = findTskgmstOrThrow(request.gasMtstNo(), trkKnsYmd);
            TkssvkyRecord tkssvkyRecord = findTkssvkyOrThrow(tskgmstRecord.tksuSvkyNo(), trkKnsYmd);

            transportServiceContractRevivalOtherRetail.execute(
                    new GmtsetdInf(
                            equipmentStatusOutput.gasMtstNo(),
                            equipmentStatusOutput.szcGkkCd(),
                            equipmentStatusOutput.szcMchCd(),
                            equipmentStatusOutput.szcCmazCd(),
                            equipmentStatusOutput.ckrtGmscJtkb()),
                    new TkssvkyInf(
                            tkssvkyRecord.tksuSvkyNo(),
                            tkssvkyRecord.skno()),
                    new TskgmstInf(
                            tskgmstRecord.gasMtstNo(),
                            tskgmstRecord.tksuSvkyNo(),
                            tskgmstRecord.skno()),
                    trkKnsYmd);

            deleteKtmgmkkOrThrow(meterReadingDataOutput);
            updateTaksmtrOrThrow(transportContractStatusOutput);

            cancelTransferWorkHistoryCreation.execute(
                    cancelTransferWorkHistoryProcessingDivision,
                    request.gasMtstNo(),
                    closingRegisterConduitWorkCode,
                    closingRegisterConduitWorkDetailCode,
                    closingRegisterRetailWorkCode,
                    closingRegisterRetailWorkDetailCode,
                    onlineReportRoute,
                    trkKnsYmd,
                    request.hukkUsrId(),
                    request.hukkUnmKj(),
                    request.rirekiMemo(),
                    null,
                    Boolean.FALSE,
                    Boolean.FALSE,
                    Boolean.FALSE,
                    null,
                    isrrkrUpdInf == null
                            ? null
                            : new jp.co.tokyo_gas.kaiheisen.libs.sv.api.register.model.IsrrkrUpdInf(
                                    isrrkrUpdInf.duknIsrgCd(),
                                    isrrkrUpdInf.duknIsrsCd(),
                                    isrrkrUpdInf.isrrGymCd(),
                                    isrrkrUpdInf.isrrGmssCd()));

            customerRegistrationNo = tkssvkyRecord.syoKyoTrno();
            workerEmployeeNo = request.sgttUsrId();
            workerName = request.sgttUsrKj();
            workplaceCode = request.sgytNwkCd();
        }

        customerServiceHistoryApiIntegration.execute(
                customerRegistrationNo,
                workerEmployeeNo,
                workerName,
                workplaceCode,
                request.gasMtstNo(),
                customerServiceHistoryWorkTypeCode,
                customerServiceHistoryApiIntegrationProcessingDivision);

        return new ClosingCancelClosingReportCancelOtherRetailResponse(
                requestContext.getCorrelationId()
                        .orElseThrow(() -> new IllegalStateException("correlationId is not set")),
                null,
                null,
                null);
    }

    private TskgmstRecord findTskgmstOrThrow(Long gasMtstNo, LocalDate trkKnsYmd) {
        return tskgmstDataAccess.findByGasMtstNoAndTkysYmdAndTyenYmdAndTksuSvkkJtkb(
                        gasMtstNo,
                        trkKnsYmd,
                        List.of(
                                transportationServiceContractRelatedStatusApplying,
                                transportationServiceContractRelatedStatusApplicableEnds))
                .orElseThrow(() -> DataInconsistencySupport.create(
                        tskgmstDataAccess.tableName(),
                        TskgmstDataAccess.CallSpecs.FIND_BY_GAS_MTST_NO_AND_TKYS_YMD_AND_TYEN_YMD_AND_TKSU_SVKK_JTKB
                                .identifier(),
                        new FlatJsonBuilder()
                                .put("gasMtstNo", gasMtstNo)
                                .put("knskKjnYmd", trkKnsYmd)
                                .put("tksuSvkkJtkbList", List.of(
                                        transportationServiceContractRelatedStatusApplying,
                                        transportationServiceContractRelatedStatusApplicableEnds))));
    }

                .orElseThrow(() -> new IllegalStateException("RequestContext.hukkSysShbt is required"));
        String decoded = kaiheisenConstantService.findNameByTypeAndValue(
                ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE,
                hukkSysShbt);
        if (decoded == null) {
            throw new ConstantDefinitionNotFoundException(
                    "KaiheisenConstantService",
                    "type=" + ConstantLookupKeys.Kaiheisen.ReportingSystemType.TYPE + ", value=" + hukkSysShbt);
        }
        return decoded;
