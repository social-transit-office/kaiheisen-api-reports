package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.config;

import com.blazebit.persistence.Criteria;
import com.blazebit.persistence.CriteriaBuilderFactory;
import io.quarkus.arc.properties.IfBuildProperty;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.enterprise.inject.Produces;
import jakarta.inject.Inject;
import jakarta.inject.Singleton;
import jakarta.persistence.EntityManager;
import jakarta.persistence.EntityManagerFactory;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.CiriusConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.ConstantServices;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.constant.KaiheisenConstantService;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.context.RequestContext;
import jp.co.tokyo_gas.kaiheisen.libs.common.observability.context.SvContextProxyFactory;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.gasmtrk.GasmtrkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkd.IsrkdDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.cirius.isrkk.IsrkkDataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.api.ciriusdb.sysibm.sysdummy1.Sysdummy1DataAccess;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.CommonDataAccessLogger;
import jp.co.tokyo_gas.kaiheisen.libs.da.core.DataAccessProxyFactory;
import jp.co.tokyo_gas.kaiheisen.libs.sv.api.validation.ClosingCancelFeasibilityJudgmentHistoryDetailCode;
import jp.co.tokyo_gas.kaiheisen.libs.sv.impl.validation.ClosingCancelFeasibilityJudgmentHistoryDetailCodeImpl;

/**
 * BFF livalit/v1 で利用する共通基盤 Bean を定義する設定クラス。
 */
@ApplicationScoped
public class BffLibsConfiguration {

    @Inject
    RequestContext requestContext;

    @Inject
    SvContextProxyFactory svContextProxyFactory;

    /**
     * 定数サービス一式をアプリケーション内で1回だけ生成する。
     *
     * <p>主な仕様: 2つの producer が同じサービス一式を共有し、各辞書の二重読込を防ぐ。
     * 既知の制約: 取得元設定の変更にはアプリケーションの再起動が必要。</p>
     *
     * @return 共有する定数サービス一式
     */
    @Produces
    @Singleton
    ConstantServices.Bundle constantServices() {
        return ConstantServices.create();
    }

    /**
     * 共有する定数サービス一式から CIRIUS 定数サービスを取得する。
     *
     * @param constantServices アプリケーション内で共有する定数サービス一式
     * @return CIRIUS 定数サービス
     */
    @Produces
    @Singleton
    CiriusConstantService ciriusConstantService(ConstantServices.Bundle constantServices) {
        return constantServices.ciriusConstantService();
    }

    /**
     * 共有する定数サービス一式から開閉栓定数サービスを取得する。
     *
     * @param constantServices アプリケーション内で共有する定数サービス一式
     * @return 開閉栓定数サービス
     */
    @Produces
    @Singleton
    KaiheisenConstantService kaiheisenConstantService(ConstantServices.Bundle constantServices) {
        return constantServices.kaiheisenConstantService();
    }

    /**
     * CriteriaBuilderFactory を生成する。
     *
     * @param entityManagerFactory EntityManagerFactory
     * @return CriteriaBuilderFactory
     */
    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "jpa", enableIfMissing = true)
    CriteriaBuilderFactory criteriaBuilderFactory(EntityManagerFactory entityManagerFactory) {
        return Criteria.getDefault().createCriteriaBuilderFactory(entityManagerFactory);
    }

    /**
     * GASMTRK DataAccess を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @return GASMTRK DataAccess
     */
    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "jpa", enableIfMissing = true)
    GasmtrkDataAccess gasmtrkDataAccess(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        return proxy(
                GasmtrkDataAccess.class,
                new jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.gasmtrk.GasmtrkDataAccessImpl(
                        entityManager,
                        criteriaBuilderFactory));
    }

    /**
     * ISRKK DataAccess を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @return ISRKK DataAccess
     */
    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "jpa", enableIfMissing = true)
    IsrkkDataAccess isrkkDataAccessForDefaultProfile(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        return proxy(
                IsrkkDataAccess.class,
                new jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrkk.IsrkkDataAccessImpl(
                        entityManager,
                        criteriaBuilderFactory));
    }

    /**
     * ISRKD DataAccess を生成する。
     *
     * @param entityManager EntityManager
     * @param criteriaBuilderFactory CriteriaBuilderFactory
     * @return ISRKD DataAccess
     */
    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "jpa", enableIfMissing = true)
    IsrkdDataAccess isrkdDataAccessForDefaultProfile(
            EntityManager entityManager,
            CriteriaBuilderFactory criteriaBuilderFactory) {
        return proxy(
                IsrkdDataAccess.class,
                new jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.cirius.isrkd.IsrkdDataAccessImpl(
                        entityManager,
                        criteriaBuilderFactory));
    }

    /**
     * SV-097 を生成する。
     *
     * @param isrkkDataAccess ISRKK DataAccess
     * @param isrkdDataAccess ISRKD DataAccess
     * @param requestContext リクエストコンテキスト
     * @return SV-097
     */
    @Produces
    @Singleton
    ClosingCancelFeasibilityJudgmentHistoryDetailCode
            closingCancelFeasibilityJudgmentHistoryDetailCode(
                    IsrkkDataAccess isrkkDataAccess,
                    IsrkdDataAccess isrkdDataAccess,
                    RequestContext requestContext) {
        return wrapService(
                ClosingCancelFeasibilityJudgmentHistoryDetailCode.class,
                new ClosingCancelFeasibilityJudgmentHistoryDetailCodeImpl(
                        isrkkDataAccess,
                        isrkdDataAccess,
                        requestContext));
    }

    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "inmemory")
    Sysdummy1DataAccess sysdummy1DataAccessForTestProfile() {
        return proxy(
                Sysdummy1DataAccess.class,
                new jp.co.tokyo_gas.kaiheisen.libs.da.inmemory.ciriusdb.sysibm.sysdummy1
                        .Sysdummy1DataAccessImpl());
    }

    @Produces
    @Singleton
    @IfBuildProperty(name = "kaiheisen.da.impl", stringValue = "jpa", enableIfMissing = true)
    Sysdummy1DataAccess sysdummy1DataAccessForDefaultProfile(final EntityManager entityManager) {
        return proxy(
                Sysdummy1DataAccess.class,
                new jp.co.tokyo_gas.kaiheisen.libs.da.jpa.ciriusdb.sysibm.sysdummy1
                        .Sysdummy1DataAccessImpl(entityManager));
    }

    private <T> T proxy(final Class<T> type, final T delegate) {
        return DataAccessProxyFactory.create(
                type,
                delegate,
                CommonDataAccessLogger.jboss(BffLibsConfiguration.class),
                requestContext);
    }

    private <T> T wrapService(final Class<T> serviceType, final T service) {
        return svContextProxyFactory.wrap(serviceType, service);
    }
}
