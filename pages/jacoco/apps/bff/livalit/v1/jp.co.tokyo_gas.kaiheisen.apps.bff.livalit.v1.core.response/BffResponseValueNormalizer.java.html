package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.response;

import java.math.BigDecimal;

/**
 * BFF 正常系レスポンスの null 値を型別の既定値へ正規化する utility。
 */
public final class BffResponseValueNormalizer {

    private static final String EMPTY = "";

    private static final Long ZERO_LONG = 0L;

    private static final Integer ZERO_INTEGER = 0;

    private static final BigDecimal ZERO_DECIMAL = BigDecimal.ZERO;

    private BffResponseValueNormalizer() {
    }

    /**
     * 任意文字列を正規化します。
     *
     * <p>主な仕様:
     * 入力が null の場合は空文字を返却し、
     * それ以外は入力値をそのまま返却します。
     * <p>既知の制約: 空白除去や trim は行いません。
     *
     * @param value 正規化対象の文字列
     * @return 正規化後の文字列
     */
    public static String stringOrEmpty(final String value) {
        return value == null ? EMPTY : value;
    }

    /**
     * Long 値を正規化します。
     *
     * <p>主な仕様:
     * 入力が null の場合は 0 を返却し、
     * それ以外は入力値をそのまま返却します。
     * <p>既知の制約: 範囲補正や丸めは行いません。
     *
     * @param value 正規化対象の値
     * @return 正規化後の値
     */
    public static Long longOrZero(final Long value) {
        return value == null ? ZERO_LONG : value;
    }

    /**
     * Integer 値を正規化します。
     *
     * <p>主な仕様:
     * 入力が null の場合は 0 を返却し、
     * それ以外は入力値をそのまま返却します。
     * <p>既知の制約: 範囲補正や丸めは行いません。
     *
     * @param value 正規化対象の値
     * @return 正規化後の値
     */
    public static Integer integerOrZero(final Integer value) {
        return value == null ? ZERO_INTEGER : value;
    }

    /**
     * BigDecimal 値を正規化します。
     *
     * <p>主な仕様:
     * 入力が null の場合は BigDecimal.ZERO を返却し、
     * それ以外は入力値をそのまま返却します。
     * <p>既知の制約: scale や丸めモードの変更は行いません。
     *
     * @param value 正規化対象の値
     * @return 正規化後の値
     */
    public static BigDecimal decimalOrZero(final BigDecimal value) {
        return value == null ? ZERO_DECIMAL : value;
    }

    /**
     * Boolean 値を正規化します。
     *
     * <p>主な仕様: 入力が null の場合は false を返却し、
     * それ以外は入力値をそのまま返却します。
     * <p>既知の制約: true / false の反転は行いません。
     *
     * @param value 正規化対象の値
     * @return 正規化後の値
     */
    public static Boolean booleanOrFalse(final Boolean value) {
        return value != null && value;
    }

    /**
     * Boolean 値を正規化します。
     *
     * @param value 正規化対象の値
     * @return 正規化後の値
     * @deprecated {@link #booleanOrFalse(Boolean)} を使用してください。
     */
    @Deprecated
    public static Boolean booleanOrNull(final Boolean value) {
        return booleanOrFalse(value);
    }
}
