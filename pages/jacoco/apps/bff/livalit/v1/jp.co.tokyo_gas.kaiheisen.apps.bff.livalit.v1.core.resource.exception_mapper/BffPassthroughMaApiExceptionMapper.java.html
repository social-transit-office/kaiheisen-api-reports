package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.exception_mapper;

import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import jakarta.ws.rs.ext.ExceptionMapper;
import jakarta.ws.rs.ext.Provider;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.ErrorResponseBody;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.MaApiException;

/**
 * MA から返された HTTP エラーを BFF からそのまま返却するマッパー。
 */
@Provider
public class BffPassthroughMaApiExceptionMapper implements ExceptionMapper<MaApiException> {

    /**
     * MA から返された HTTP エラーをレスポンスへ変換する。
     *
     * @param exception MA エラー
     * @return 変換後レスポンス
     */
    @Override
    public Response toResponse(MaApiException exception) {
        return BffPassthroughMaApiException.from(exception)
                .map(this::toPassthroughResponse)
                .orElseGet(() -> toFallbackResponse(exception));
    }

    private Response toPassthroughResponse(BffPassthroughMaApiException exception) {
        Response.ResponseBuilder builder = Response.status(exception.getStatus());
        String mediaType = exception.getMediaType();
        builder.type(mediaType == null ? MediaType.APPLICATION_JSON : mediaType);

        if (exception.getResponseBody() != null) {
            builder.entity(exception.getResponseBody());
        }
        return builder.build();
    }

    private Response toFallbackResponse(MaApiException exception) {
        ErrorResponseBody body = BffErrorResponseResolver.fromKaiheisenException(exception);
        return Response.status(BffErrorResponseResolver.resolveHttpStatus(exception))
                .type(MediaType.APPLICATION_JSON)
                .entity(body)
                .build();
    }
}
