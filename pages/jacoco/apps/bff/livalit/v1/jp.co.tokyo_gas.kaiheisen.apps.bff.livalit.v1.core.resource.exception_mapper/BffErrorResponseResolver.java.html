package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.exception_mapper;

import java.util.List;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.BadRequestErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.BusinessRuleErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.ConstraintViolationErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.DataNotFoundErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.ErrorResponseBody;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.HeisenRuleErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.KaiheisenErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.KaisenRuleErrorResponse;
import jp.co.tokyo_gas.kaiheisen.contracts.bff.livalit.v1.core.response.ValidationExceptionErrorResponse;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.KaiheisenException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.BusinessRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.HeisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.business.KaisenRuleException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCatalog;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.catalog.ErrorCodeConstants;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.DataNotFoundException;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.validation.ValidationException;

/**
 * BFF 例外を BFF 契約エラーレスポンスへ変換するユーティリティ。
 */
public final class BffErrorResponseResolver {

    private static final int HTTP_BAD_REQUEST = 400;

    private static final int HTTP_INTERNAL_SERVER_ERROR = 500;

    private BffErrorResponseResolver() {
    }

    /**
     * 開閉栓共通例外から HTTP ステータスを解決する。
     *
     * @param exception 開閉栓共通例外
     * @return HTTPステータスコード
     */
    public static int resolveHttpStatus(KaiheisenException exception) {
        if (isSystemErrorCode(resolveResponseErrorCode(normalize(exception.getErrorCode())))) {
            return HTTP_INTERNAL_SERVER_ERROR;
        }
        if (exception instanceof ValidationException) {
            return HTTP_BAD_REQUEST;
        }
        if (exception instanceof BusinessRuleException) {
            return HTTP_BAD_REQUEST;
        }
        return HTTP_INTERNAL_SERVER_ERROR;
    }

    /**
     * 開閉栓共通例外を BFF 共通エラーレスポンスへ変換する。
     *
     * @param exception 開閉栓共通例外
     * @return エラーレスポンス
     */
    public static ErrorResponseBody fromKaiheisenException(KaiheisenException exception) {
        String originalCode = normalize(exception.getErrorCode());
        String resolvedCode = resolveResponseErrorCode(originalCode);

        if (isSystemErrorCode(resolvedCode)) {
            return systemErrorResponse();
        }

        List<String> errorEmbedded = copyOrEmpty(exception.getErrorMessageEmbeddedStringList());
        List<String> errorDetailList = resolveErrorDetailList(exception, errorEmbedded);
        String errorDescription = normalize(ErrorCatalog.resolveErrorMessage(originalCode, errorEmbedded));
        if (errorDescription == null) {
            errorDescription = normalize(exception.getMessage());
        }
        if (errorDescription == null) {
            errorDescription = "error";
        }

        if (exception instanceof DataNotFoundException dataNotFoundException) {
            return new DataNotFoundErrorResponse(
                    resolvedCode,
                    errorDescription,
                    errorDetailList,
                    normalize(dataNotFoundException.getCiriusMessageId()),
                    List.of());
        }

        if (exception instanceof ValidationException) {
            return new ValidationExceptionErrorResponse(
                    resolvedCode,
                    errorDescription,
                    errorDetailList);
        }

        List<String> crsMsgEmbStrList = List.of();
        String crsMsgIdByException = null;
        if (exception instanceof BusinessRuleException businessRuleException) {
            crsMsgEmbStrList = copyOrEmpty(businessRuleException.getCiriusMessageEmbeddedStringList());
            crsMsgIdByException = normalize(businessRuleException.getCiriusMessageId());
        }

        String crsMsgId = normalize(ErrorCatalog.resolveCiriusMessageId(originalCode));
        if (crsMsgId == null) {
            crsMsgId = crsMsgIdByException;
        }

        if (exception instanceof KaisenRuleException kaisenRuleException) {
            return new KaisenRuleErrorResponse(
                    resolvedCode,
                    errorDescription,
                    errorDetailList,
                    crsMsgId,
                    crsMsgEmbStrList,
                    normalize(kaisenRuleException.getKaisenErrorReasonCode()));
        }

        if (exception instanceof HeisenRuleException heisenRuleException) {
            return new HeisenRuleErrorResponse(
                    resolvedCode,
                    errorDescription,
                    errorDetailList,
                    crsMsgId,
                    crsMsgEmbStrList,
                    normalize(heisenRuleException.getHeisenErrorReasonCode()));
        }

        if (exception instanceof BusinessRuleException) {
            return new BusinessRuleErrorResponse(
                    resolvedCode,
                    errorDescription,
                    errorDetailList,
                    crsMsgId,
                    crsMsgEmbStrList);
        }

        return new KaiheisenErrorResponse(
                resolvedCode,
                errorDescription,
                List.of());
    }

    /**
     * BadRequest 例外向けレスポンスを生成する。
     *
     * @param description エラー概要
     * @param detailList エラー詳細
     * @return BadRequest エラーレスポンス
     */
    public static BadRequestErrorResponse fromBadRequest(String description, List<String> detailList) {
        List<String> details = copyOrEmpty(detailList);
        return new BadRequestErrorResponse(
                ErrorCodeConstants.ERROR_CODE_010000001,
                resolveValidationErrorDescription(description, details),
                details);
    }

    /**
     * 制約違反例外向けレスポンスを生成する。
     *
     * @param description エラー概要
     * @param detailList エラー詳細
     * @return 制約違反エラーレスポンス
     */
    public static ConstraintViolationErrorResponse fromConstraintViolation(String description, List<String> detailList) {
        List<String> details = copyOrEmpty(detailList);
        return new ConstraintViolationErrorResponse(
                ErrorCodeConstants.ERROR_CODE_010000001,
                resolveValidationErrorDescription(description, details),
                details);
    }

    /**
     * BFF 未捕捉例外向けの 500 エラーレスポンスを生成する。
     *
     * @param exception 未捕捉例外
     * @return 500 エラーレスポンス
     */
    public static KaiheisenErrorResponse fromUnhandledException(Exception exception) {
        String errorCode = ErrorCodeConstants.ERROR_CODE_990000001;
        String resolvedCode = normalize(ErrorCatalog.resolveFinalErrorCode(errorCode));
        if (resolvedCode == null) {
            resolvedCode = errorCode;
        }

        String errorDescription = normalize(ErrorCatalog.resolveErrorMessage(errorCode, List.of()));
        if (errorDescription == null) {
            errorDescription = normalize(exception.getMessage());
        }
        if (errorDescription == null) {
            errorDescription = "error";
        }

        return new KaiheisenErrorResponse(resolvedCode, errorDescription, List.of());
    }

    private static List<String> copyOrEmpty(List<String> values) {
        return values == null ? List.of() : List.copyOf(values);
    }

    private static List<String> resolveErrorDetailList(
            KaiheisenException exception,
            List<String> defaultDetailList) {
        if (exception instanceof ValidationException validationException) {
            return copyOrEmpty(validationException.getErrorDetailList());
        }
        return defaultDetailList;
    }

    private static String resolveValidationErrorDescription(String description, List<String> detailList) {
        String normalizedDescription = normalize(description);
        if (normalizedDescription == null) {
            normalizedDescription = detailList.isEmpty() ? "validation error" : detailList.getFirst();
        }
        String catalogMessage = normalize(
                ErrorCatalog.resolveErrorMessage(ErrorCodeConstants.ERROR_CODE_010000001, List.of()));
        return catalogMessage == null ? normalizedDescription : catalogMessage;
    }

    private static String normalize(String value) {
        if (value == null) {
            return null;
        }
        String trimmed = value.trim();
        return trimmed.isEmpty() ? null : trimmed;
    }

    private static String resolveResponseErrorCode(String originalCode) {
        String resolvedCode = normalize(ErrorCatalog.resolveFinalErrorCode(originalCode));
        String candidate = resolvedCode == null ? originalCode : resolvedCode;
        if (candidate != null && candidate.matches("^[0-9]{9}$")) {
            return candidate;
        }
        return ErrorCodeConstants.ERROR_CODE_990000001;
    }

    private static boolean isSystemErrorCode(String errorCode) {
        return ErrorCodeConstants.ERROR_CODE_990000001.equals(errorCode);
    }

    private static KaiheisenErrorResponse systemErrorResponse() {
        String errorCode = ErrorCodeConstants.ERROR_CODE_990000001;
        String errorDescription = normalize(ErrorCatalog.resolveErrorMessage(errorCode, List.of()));
        return new KaiheisenErrorResponse(
                errorCode,
                errorDescription == null ? "error" : errorDescription,
                List.of());
    }
}
