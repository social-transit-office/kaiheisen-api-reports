package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.exception_mapper;

import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import java.util.Optional;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import jp.co.tokyo_gas.kaiheisen.libs.common.core.exception.external.MaApiException;

/**
 * BFF が MA エラーを HTTP ステータスとボディごと透過返却するための例外。
 */
public class BffPassthroughMaApiException extends RuntimeException {

    private static final Pattern STATUS_PATTERN = Pattern.compile("status\\s*=\\s*(\\d{3})");

    private static final Pattern BODY_PATTERN = Pattern.compile("body\\s*=\\s*(.*)", Pattern.DOTALL);

    private static final String NULL_BODY = "null";

    private final int status;

    private final String responseBody;

    private final String mediaType;

    /**
     * コンストラクタ。
     *
     * @param status HTTP ステータス
     * @param responseBody レスポンスボディ
     * @param mediaType レスポンスの Content-Type
     * @param cause 原因例外
     */
    public BffPassthroughMaApiException(int status, String responseBody, String mediaType, Throwable cause) {
        super(cause);
        this.status = status;
        this.responseBody = responseBody;
        this.mediaType = mediaType;
    }

    /**
     * MaApiException から透過用例外を生成する。
     *
     * @param exception MA API 例外
     * @return HTTP レスポンスを保持できる場合の透過用例外
     */
    public static Optional<BffPassthroughMaApiException> from(MaApiException exception) {

        WebApplicationException webApplicationException =
                exception.getCause() instanceof WebApplicationException cause
                        ? cause
                        : null;

        Response response =
                webApplicationException == null
                        ? null
                        : webApplicationException.getResponse();

        Integer status = extractStatus(exception.getResponseInfo());

        if (status == null && response == null) {
            return Optional.empty();
        }

        String body = extractBody(exception.getResponseInfo());

        if (body == null) {
            body = readResponseBody(response);
        }

        MediaType responseMediaType =
                response == null
                        ? null
                        : response.getMediaType();

        return Optional.of(
                new BffPassthroughMaApiException(
                        status == null ? response.getStatus() : status,
                        body,
                        responseMediaType == null ? null : responseMediaType.toString(),
                        exception));
    }

    private static WebApplicationException findWebApplicationException(Throwable throwable) {
        Throwable current = throwable;
        while (current != null) {
            if (current instanceof WebApplicationException webApplicationException) {
                return webApplicationException;
            }
            current = current.getCause();
        }
        return null;
    }

    private static Integer extractStatus(String responseInfo) {
        if (responseInfo == null) {
            return null;
        }
        Matcher matcher = STATUS_PATTERN.matcher(responseInfo);
        if (!matcher.find()) {
            return null;
        }
        String statusText = matcher.group(1);
        try {
            return Integer.valueOf(statusText.trim());
        } catch (NumberFormatException ignored) {
            return null;
        }
    }

    private static String extractBody(String responseInfo) {
        if (responseInfo == null) {
            return null;
        }
        Matcher matcher = BODY_PATTERN.matcher(responseInfo);
        if (!matcher.find()) {
            return null;
        }
        String body = matcher.group(1).stripLeading();
        return NULL_BODY.equals(body) ? null : body;
    }

    private static String readResponseBody(Response response) {
        if (response == null || !response.hasEntity()) {
            return null;
        }

        try {
            response.bufferEntity();
            return response.readEntity(String.class);
        } catch (RuntimeException suppressed) {
            return null;
        }
    }

    /**
     * HTTP ステータスを返す。
     *
     * @return HTTP ステータス
     */
    public int getStatus() {
        return status;
    }

    /**
     * レスポンスボディを返す。
     *
     * @return レスポンスボディ
     */
    public String getResponseBody() {
        return responseBody;
    }

    /**
     * レスポンスの Content-Type を返す。
     *
     * @return レスポンスの Content-Type
     */
    public String getMediaType() {
        return mediaType;
    }
}
