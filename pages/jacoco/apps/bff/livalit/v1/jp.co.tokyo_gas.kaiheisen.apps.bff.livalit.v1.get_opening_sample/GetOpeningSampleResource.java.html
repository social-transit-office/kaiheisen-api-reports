/**
 * BFF-999 サンプルBFF: 閉栓希望年月日チェック API。
 *
 * <p>処理概要:</p>
 * <ol>
 *   <li>システム日時の年月を取得し、前月を算出する。</li>
 *   <li>閉栓希望年月日の年月が前月より前の場合は業務エラー（400）を返す。</li>
 *   <li>CIRIUS 日程管理情報取得 API（他-014）を呼び出し、2営業日後の暦年月日を取得する。</li>
 *   <li>閉栓希望年月日が2営業日後の暦年月日を超過している場合は業務エラー（400）を返す。</li>
 *   <li>MA-107 小売判定 API を呼び出し、TG 小売か他小売かを判定する。</li>
 *   <li>正常レスポンス（200）を返す。</li>
 * </ol>
 */
@Bff(identifier = "BFF-999", name = "サンプルBFF: 閉栓希望年月日チェック")
@Tag(name = "sample", description = "BFF-999 サンプルBFF")
@Path(GetOpeningSampleEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
public class GetOpeningSampleResource
        extends BaseResource<GetOpeningSampleRequest> {

    private static final DateTimeFormatter YYYYMMDD = DateTimeFormatter.ofPattern("yyyyMMdd");

    private final RetailDivisionMaAdapter maAdapter;

    private final ScheduleManagementCiriusAdapter ciriusAdapter;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator リクエスト DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param maAdapter MA-107 小売判定 adapter
     * @param ciriusAdapter CIRIUS 日程管理 adapter
     */
    @Inject
    public GetOpeningSampleResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            RetailDivisionMaAdapter maAdapter,
            ScheduleManagementCiriusAdapter ciriusAdapter) {
        super(requestContext, requestDtoValidator, clock);
        this.maAdapter = maAdapter;
        this.ciriusAdapter = ciriusAdapter;
    }

    /**
     * 閉栓希望年月日チェックを実行する。
     *
     * @param request リクエストパラメータ
     * @return レスポンス
     */
    @GET
    @Operation(
            summary = "閉栓希望年月日チェック",
            description = "閉栓希望年月日の妥当性を検証し、小売判定結果を返却するサンプルBFF API。",
            operationId = "getOpeningSample")
    @APIResponse(
            responseCode = "200",
            description = "正常レスポンス",
            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = GetOpeningSampleResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(anyOf = {CommonErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(anyOf = {KaiheisenErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "default",
            description = "MA API エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response getOpeningSample(@BeanParam GetOpeningSampleRequest request) {
        return execute(request, () -> doProcess(request));
    }

    private Response doProcess(GetOpeningSampleRequest request) {
        String correlationId = requestContext.getCorrelationId().orElseThrow(
                () -> new IllegalStateException("correlationId is not set in RequestContext"));
        String systemDatetime = boundSystemDatetimeText();
        LocalDate systemDate = requestContext.getSystemDatetime()
                .map(dt -> dt.toLocalDate())
                .orElseGet(LocalDate::now);
        LocalDate hsenKiboYmd = LocalDate.parse(request.hsenKiboYmd());

        // システム日時の年月の前月と閉栓希望年月日の年月を比較する
        YearMonth prevMonth = YearMonth.from(systemDate).minusMonths(1);
        YearMonth hsenMonth = YearMonth.from(hsenKiboYmd);
        if (hsenMonth.isBefore(prevMonth)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000020);
        }

        // CIRIUS 日程管理情報取得 API（他-014）を呼び出す
        String startYmd = systemDate.format(YYYYMMDD);
        String endYmd = systemDate.plusMonths(1).format(YYYYMMDD);
        ScheduleManagementResponse ciriusResponse = ciriusAdapter.getByDateRange(startYmd, endYmd);

        // CIS稼働回分番号が設定されている2件目のデータの暦年月日を取得する（2営業日後）
        LocalDate secondBusinessDay = extractSecondCisWorkDay(ciriusResponse.scheduleManagementList());

        // 閉栓希望年月日が2営業日後を超える場合は業務エラー
        if (hsenKiboYmd.isAfter(secondBusinessDay)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000020);
        }

        // MA-107 小売判定 API を呼び出す
        RetailDivisionRequest maRequest = new RetailDivisionRequest(
                "1",
                String.valueOf(request.gasMtstNo()),
                request.hsenKiboYmd(),
                "L",
                systemDatetime);
        RetailDivisionResponse maResponse = maAdapter.invoke(maRequest);

        // 正常レスポンスを返却する
        Long svkyNo = extractSvkyNo(maResponse);
        return Response.ok(
                new GetOpeningSampleResponse(
                BffResponseValueNormalizer.longOrZero(svkyNo),
                        List.of(new CkrtGmtInfo(request.gasMtstNo())),
                BffResponseValueNormalizer.stringOrEmpty(correlationId)))
                .build();
    }

    /**
     * CIRIUS レスポンスから CIS稼働回分番号が設定されている2件目の暦年月日を取得する。
     *
     * @param items 日程管理情報リスト
     * @return 2営業日後の日付
     */
    private LocalDate extractSecondCisWorkDay(List<ScheduleManagementItem> items) {
        int count = 0;
        for (ScheduleManagementItem item : items) {
            if (item.cisKadKaibNo() != null && !item.cisKadKaibNo().isBlank()) {
                count++;
                if (count == 2) {
                    return LocalDate.parse(String.valueOf(item.rekiYmd()), YYYYMMDD);
                }
            }
        }
        throw new KaiheisenException(
                "CIRIUS日程管理情報からCIS稼働回分番号が設定された2件目のデータが取得できません");
    }

    /**
     * MA-107 レスポンスから託送サービス契約番号を取得する。
     *
     * @param maResponse MA-107 レスポンス
     * @return 託送サービス契約番号
     */
    private Long extractSvkyNo(RetailDivisionResponse maResponse) {
        if (maResponse.tkssvkyInf() != null) {
            return maResponse.tkssvkyInf().tksuSvkyNo();
        }
        return null;
    }
}
