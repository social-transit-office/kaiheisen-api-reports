    /**
     * 閉栓作業票情報を取得します。
     *
     * @param request BFF-002 リクエスト
     * @return 閉栓作業票情報レスポンス
     */
    public GetClosingResponse handle(GetClosingRequest request) {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime().orElseGet(clock::currentDateTime);
        LocalDate systemDate = systemDatetime.toLocalDate();
        String normalizedHsenKiboYmd = normalizeOptional(request.hsenKiboYmd());
        LocalDate hsenKiboYmd = Rfc3339LocalDateParser.parse(normalizedHsenKiboYmd);

        validateClosingRequestMonth(systemDate, hsenKiboYmd);

        ScheduleManagementResponse ciriusResponse = scheduleManagementCiriusAdapter.getByDateRange(
                systemDate.format(YYYYMMDD),
                systemDate.plusMonths(1).format(YYYYMMDD));
        LocalDate secondBusinessDay = extractSecondCisWorkDay(ciriusResponse.scheduleManagementList());
        if (hsenKiboYmd.isAfter(secondBusinessDay)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000020);
        }

        String downstreamSystemDatetime = resolveSystemDatetimeForDownstream(request, systemDatetime);
        RetailDivisionResponse retailDivisionResponse = retailDivisionMaAdapter.invoke(new RetailDivisionRequest(
                "1",
                format11Digits(request.gasMtstNo()),
                normalizedHsenKiboYmd,
                REPORTING_SYSTEM_TYPE,
                downstreamSystemDatetime));

        String retailDivision = retailDivisionResponse.kuHntKbn();
        if (RetailTypeJudgmentCode.TOKYO_GAS.code().equals(retailDivision)) {
            LivalitClosingWorkTicketInfoGetRequest tgRequest =
                buildTgRetailRequest(
                request,
                normalizedHsenKiboYmd,
                retailDivisionResponse,
                downstreamSystemDatetime);
            LivalitClosingWorkTicketInfoGetResponse tgResponse =
                tgRetailMaAdapter.invoke(tgRequest);
            return mapTgResponse(tgResponse);
        }

        LivalitClosingWorkTicketInfoGetOtherRetailRequest otherRequest =
            buildOtherRetailRequest(
                request,
                normalizedHsenKiboYmd,
                retailDivisionResponse,
                downstreamSystemDatetime);
        LivalitClosingWorkTicketInfoGetOtherRetailResponse otherResponse =
            otherRetailMaAdapter.invoke(otherRequest);
        return mapOtherRetailResponse(otherResponse);
    }

    /**
     * 閉栓希望年月日の月が許容範囲内かを検証します。
     *
     * @param systemDate システム日付
     * @param hsenKiboYmd 閉栓希望年月日
     */
    private void validateClosingRequestMonth(LocalDate systemDate, LocalDate hsenKiboYmd) {
        YearMonth prevMonth = YearMonth.from(systemDate).minusMonths(1);
        YearMonth hsenMonth = YearMonth.from(hsenKiboYmd);
        if (hsenMonth.isBefore(prevMonth)) {
            throw new BusinessRuleException(ErrorCodeConstants.ERROR_CODE_020000020);
        }
    }

    /**
     * CIRIUS 日程管理情報から 2 営業日後の日付を抽出します。
     *
     * @param items 日程管理一覧
     * @return 2 営業日後の日付
     */
    private LocalDate extractSecondCisWorkDay(List<ScheduleManagementItem> items) {
        int count = 0;
        for (ScheduleManagementItem item : items) {
            if (item.cisKadKaibNo() != null && !item.cisKadKaibNo().isBlank()) {
                count++;
                if (count == 2) {
                    return LocalDate.parse(String.valueOf(item.rekiYmd()), YYYYMMDD);
                }
            }
        }
        throw new KaiheisenException(
                "CIRIUS日程管理情報からCIS稼働回分番号が設定された2件目のデータが取得できません");
    }

    private boolean isMissingContractNumber(Long value) {
        return value == null || value == 0L;
    }

    /**
     * 下流 API へ渡す systemDatetime を決定します。
     *
     * @param request BFF-002 リクエスト
     * @param systemDatetime RequestContext 上のシステム日時
     * @return 下流連携用 systemDatetime
     */
    private String resolveSystemDatetimeForDownstream(GetClosingRequest request, LocalDateTime systemDatetime) {
        String requestedSystemDatetime = normalizeOptional(request.systemDatetime());
        if (requestedSystemDatetime != null) {
            return toRfc3339DateTimeText(requestedSystemDatetime);
        }
        return systemDatetime.atOffset(ZoneOffset.UTC)
                .truncatedTo(ChronoUnit.MICROS)
                .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
    }

    private String toRfc3339DateTimeText(String rawDateTime) {
        String normalized = normalizeOptional(rawDateTime);
        if (normalized == null) {
            return null;
        }
        try {
            return OffsetDateTime.parse(normalized)
                    .truncatedTo(ChronoUnit.MICROS)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        } catch (DateTimeParseException exception) {
            return LocalDateTime.parse(normalized)
                    .truncatedTo(ChronoUnit.MICROS)
                    .atOffset(ZoneOffset.UTC)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        }
    }

    /**
     * 数値を 11 桁のゼロ埋め文字列へ変換します。
     *
     * @param value 変換対象の値
     * @return 11 桁のゼロ埋め文字列
     */
    private String format11Digits(Long value) {
        if (value == null) {
            throw new KaiheisenException("必須の11桁数値項目が取得できません");
        }
        return String.format("%011d", value);
    }

    private String readString(Object target, String accessorName) {
        return readValue(target, accessorName, String.class);
    }

    private Long readLong(Object target, String accessorName) {
        return readValue(target, accessorName, Long.class);
    }

    private Integer readInteger(Object target, String accessorName) {
        return readValue(target, accessorName, Integer.class);
    }

    private BigDecimal readBigDecimal(Object target, String accessorName) {
        return readValue(target, accessorName, BigDecimal.class);
    }

    private <T> T readValue(Object target, String accessorName, Class<T> expectedType) {
        if (target == null) {
            return null;
        }
        try {
            Method accessor = target.getClass().getMethod(accessorName);
            Object value = accessor.invoke(target);
            return expectedType.cast(value);
        } catch (NoSuchMethodException | IllegalAccessException | InvocationTargetException exception) {
            throw new KaiheisenException(
                    "他小売応答の項目取得に失敗しました: " + target.getClass().getName() + "." + accessorName,
                    exception);
        }
    }
