package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.get_closing;

import jakarta.ws.rs.Path;
import jakarta.ws.rs.Produces;

/**
 * BFF-002 閉栓作業票情報取得 API。
 */
@Bff(identifier = "BFF-002", name = "閉栓作業票情報取得")
@Tag(name = "closing", description = "BFF-002 閉栓作業票情報取得")
@Path(GetClosingEndpointContract.PATH)
@Produces(MediaType.APPLICATION_JSON)
public class GetClosingResource extends BaseResource<GetClosingRequest> {

    /** 閉栓作業票情報取得のユースケースを実行するサービス。 */
    private final GetClosingService getClosingService;

    /**
     * 閉栓作業票情報取得 resource を生成します。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator リクエスト DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param getClosingService 閉栓作業票情報取得サービス
     */
    @Inject
    public GetClosingResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            GetClosingService getClosingService) {
        super(requestContext, requestDtoValidator, clock);
        this.getClosingService = getClosingService;
    }

    /**
     * 閉栓作業票情報を取得します。
     *
     * @param request BFF-002 リクエスト
     * @return レスポンス
     */
    @GET
    @Operation(
            summary = "閉栓作業票情報取得",
            description = "閉栓希望年月日の妥当性を検証し、CIRIUS と MA API を呼び出して閉栓作業票情報を返却する。",
            operationId = "getClosing")
    @APIResponse(
            responseCode = "200",
            description = "正常レスポンス",
            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                schema = @Schema(implementation = GetClosingResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(anyOf = {CommonErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(anyOf = {KaiheisenErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "default",
            description = "MA API エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response getClosing(@BeanParam GetClosingRequest request) {
        return execute(
                request,
                () -> Response.ok(getClosingService.handle(request)).build());
    }

    @Override
    protected InterFieldValidation<GetClosingRequest> interFieldValidation() {
        return request -> {
            if (request.gasMtstNo() == null || isFixed11Digits(request.gasMtstNo())) {
                return List.of();
            }
            return List.of(ValidationError.of(ErrorCodeConstants.ERROR_CODE_010000016, "gasMtstNo", "11"));
        };
    }

    private boolean isFixed11Digits(Long value) {
        return String.valueOf(value).length() == 11;
    }
}
