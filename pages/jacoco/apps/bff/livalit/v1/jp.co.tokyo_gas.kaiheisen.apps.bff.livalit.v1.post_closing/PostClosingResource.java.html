package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.post_closing;

import jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.BaseResource;
import jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.Bff;
import jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.core.resource.InterFieldValidation;

/**
 * BFF-004 LIVALIT閉栓落成 API。
 */
@Bff(identifier = "BFF-004", name = "LIVALIT閉栓落成")
@Tag(name = "closing", description = "BFF-004 LIVALIT閉栓落成")
@Path(PostClosingEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
public class PostClosingResource extends BaseResource<PostClosingRequest> {

    private final PostClosingService postClosingService;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param postClosingService 閉栓落成サービス
     */
    @Inject
    public PostClosingResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            PostClosingService postClosingService) {
        super(requestContext, requestDtoValidator, clock);
        this.postClosingService = postClosingService;
    }

    /**
     * 閉栓落成を実施する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    @POST
    @Operation(
            summary = "LIVALIT閉栓落成",
            description = "小売判定の結果に応じて閉栓落成の downstream MA を呼び分ける。",
            operationId = "postClosing")
    @RequestBody(required = true, content = @Content(
            mediaType = MediaType.APPLICATION_JSON,
            schema = @Schema(implementation = PostClosingRequest.class)))
    @APIResponse(
            responseCode = "200",
            description = "正常レスポンス",
            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = PostClosingResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {CommonErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {KaiheisenErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "default",
            description = "MA API エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response postClosing(PostClosingRequest request) {
        if (request == null) {
            throw new ValidationException(
                List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "requestBody")),
                true);
        }
        return execute(request, () -> Response.ok(postClosingService.handle(request)).build());
    }

    @Override
    protected InterFieldValidation<PostClosingRequest> interFieldValidation() {
        return request -> {
            List<ValidationError> errors = new ArrayList<>();
            if (!isBlank(request.sgyAnknNo()) && !request.sgyAnknNo().startsWith("0")) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, "sgyAnknNo"));
            }
            if (!isGasUsageWithinScale(request.gasSyry())) {
                errors.add(ValidationError.of(ErrorCodeConstants.ERROR_CODE_010000020, "gasSyry", "11", "4"));
            }
            if ("4".equals(request.hesnJysb()) && isBlank(request.hesnSdnShbt())) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "hesnSdnShbt"));
            }
            if (!"4".equals(request.hesnJysb()) && isBlank(request.utidxNo())) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "utidxNo"));
            }
            if ("3".equals(request.syoriKbn()) && isBlank(request.hsenErrRyCd())) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "hsenErrRyCd"));
            }
            return List.copyOf(errors);
        };
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }

    private boolean isGasUsageWithinScale(BigDecimal gasSyry) {
        if (gasSyry == null) {
            return true;
        }

        String text = gasSyry.abs().toPlainString();
        String[] parts = text.split("\\.", 2);

        String integerPart = parts[0].replaceFirst("^0+(?!$)", "");
        String fractionPart = parts.length == 2 ? parts[1] : "";

        return integerPart.length() <= 11 && fractionPart.length() <= 4;
    }
}
