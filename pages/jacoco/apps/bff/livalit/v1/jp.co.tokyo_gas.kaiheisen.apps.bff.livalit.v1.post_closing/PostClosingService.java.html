    /**
     * BFF-004 の閉栓落成を処理する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    public PostClosingResponse handle(PostClosingRequest request) {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime().orElseGet(clock::currentDateTime);
        String downstreamSystemDatetime = resolveSystemDatetimeForDownstream(request, systemDatetime);
        String searchDate = resolveRetailDivisionSearchDate(request, systemDatetime);

        RetailDivisionResponse retailDivisionResponse = retailDivisionMaAdapter.invoke(new RetailDivisionRequest(
                "3",
                request.gasMtstNo(),
                searchDate,
                reportingSystemType,
                downstreamSystemDatetime));

        String retailDivision = retailDivisionResponse.kuHntKbn();
        if (PROCESS_UPDATE.equals(request.syoriKbn())) {
            if (RetailTypeJudgmentCode.TOKYO_GAS.code().equals(retailDivision)) {
                if (CLOSING_REASON_SUPPLY_STOP.equals(request.hesnJysb())) {
                    ClosingCommonSupplyStopResponse response = closingCommonSupplyStopTgRetailMaAdapter.invoke(
                            buildSupplyStopTgRetailRequest(request, downstreamSystemDatetime));
                    return mapTgSupplyStopResponse(response);
                }
                ClosingCommonNormalTgRetailResponse response = closingCommonNormalTgRetailMaAdapter.invoke(
                        buildNormalTgRetailRequest(request, retailDivisionResponse, downstreamSystemDatetime));
                return mapNormalTgResponse(response);
            }
            if (RetailTypeJudgmentCode.OTHER.code().equals(retailDivision)) {
                return handleUpdateOtherRetail(request, retailDivisionResponse, downstreamSystemDatetime);
            }
            throw new KaiheisenException("小売判定区分の値が不正のため、処理を終了しました");
        }

        if (PROCESS_FORCE.equals(request.syoriKbn())) {
            if (RetailTypeJudgmentCode.TOKYO_GAS.code().equals(retailDivision)) {
                ClosingCommonForceTgRetailResponse response = closingCommonForceTgRetailMaAdapter.invoke(
                        buildForceTgRetailRequest(request, downstreamSystemDatetime));
                return toForceResponse(response.correlationId());
            }
            if (RetailTypeJudgmentCode.OTHER.code().equals(retailDivision)) {
                ClosingCommonForceOtherRetailResponse response = closingCommonForceOtherRetailMaAdapter.invoke(
                        buildForceOtherRetailRequest(request, downstreamSystemDatetime));
                return toForceResponse(response.correlationId());
            }
            throw new KaiheisenException("小売判定区分の値が不正のため、処理を終了しました");
        }

        throw new KaiheisenException("処理区分の値が不正のため、処理を終了しました");
    }

    private String resolveSystemDatetimeForDownstream(
            PostClosingRequest request,
            LocalDateTime systemDatetime) {
        String requestedSystemDatetime = normalizeOptional(request.systemDatetime());
        if (requestedSystemDatetime != null) {
            return toRfc3339DateTimeText(requestedSystemDatetime);
        }
        return systemDatetime
                .atOffset(ZoneOffset.UTC)
                .truncatedTo(ChronoUnit.MICROS)
                .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
    }

    private PostClosingResponse handleUpdateOtherRetail(
            PostClosingRequest request,
            RetailDivisionResponse retailDivisionResponse,
            String downstreamSystemDatetime) {
        ClosingCommonNormalOtherRetailResponse response = closingCommonNormalOtherRetailMaAdapter.invoke(
                buildNormalOtherRetailRequest(request, retailDivisionResponse, downstreamSystemDatetime));
        return mapNormalOtherRetailResponse(response);
    }

    private String toRfc3339DateTimeText(String rawDateTime) {
        try {
            return OffsetDateTime.parse(rawDateTime)
                    .truncatedTo(ChronoUnit.MICROS)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        } catch (DateTimeParseException exception) {
            return LocalDateTime.parse(rawDateTime)
                    .truncatedTo(ChronoUnit.MICROS)
                    .atOffset(ZoneOffset.UTC)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        }
    }

        return systemDatetime.toLocalDate().toString();
    }

    private String resolveCorrelationId(String responseCorrelationId) {
        String requestCorrelationId = requestContext.getCorrelationId().orElse(null);
        if (requestCorrelationId != null) {
            return requestCorrelationId;
        }
        String normalizedResponseCorrelationId = normalizeOptional(responseCorrelationId);
        if (normalizedResponseCorrelationId != null) {
            return normalizedResponseCorrelationId;
        }
        return null;
    }

    private String toRfc3339DateText(String rawDateTime) {
        try {
            return OffsetDateTime.parse(rawDateTime)
                    .toLocalDate()
                    .toString();
        } catch (DateTimeParseException exception) {
            try {
                return LocalDateTime.parse(rawDateTime)
                        .toLocalDate()
                        .toString();
            } catch (DateTimeParseException ignored) {
                return LocalDate.parse(rawDateTime).toString();
            }
        }
    }

    private String format11Digits(Long value) {
        if (value == null) {
            throw new KaiheisenException("必須の11桁数値項目が取得できません");
        }
        return String.format("%011d", value);
    }

    private String nullable11Digits(Number value) {
        if (value == null) {
            return null;
        }
        return String.format("%011d", value.longValue());
    }

    private String stringify(Long value) {
        if (value == null) {
            throw new KaiheisenException("必須の数値項目が取得できません");
        }
        return value.toString();
    }

    private Long parseRequiredLong(String value, String fieldName) {
        String normalized = normalizeOptional(value);
        if (normalized == null) {
            throw new KaiheisenException("必須の数値項目が取得できません: " + fieldName);
        }
        try {
            return Long.valueOf(normalized);
        } catch (NumberFormatException exception) {
            throw new KaiheisenException("数値項目の形式が不正です: " + fieldName, exception);
        }
    }

    private String formatDecimal(BigDecimal value) {
        if (value == null) {
            throw new KaiheisenException("必須の小数項目が取得できません");
        }
        return value.stripTrailingZeros().toPlainString();
    }

    private List<String> copyOrNull(List<String> values) {
        if (values == null) {
            return null;
        }
        return List.copyOf(values);
    }

    private String normalizeOptional(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }

    private static String stripTrailingSpaces(String value) {
        if (value == null || value.isEmpty()) {
            return value;
        }
        int endIndex = value.length();
        while (endIndex > 0 && value.charAt(endIndex - 1) == ' ') {
            endIndex--;
        }
        return value.substring(0, endIndex);
    }
