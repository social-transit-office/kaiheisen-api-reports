@Bff(identifier = "BFF-001", name = "LIVALIT用：開栓作業関連情報取得")

    /**

     * @param request リクエスト
     * @return レスポンス

    @GET
    @Operation(
            summary = "LIVALIT用：開栓作業関連情報取得",
            description = "処理区分に応じてガスメーター設置場所番号を確定し、開栓作業関連情報を返却する。",
            operationId = "getOpening")
    @APIResponse(
            responseCode = "200",

            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = GetOpeningResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(anyOf = {CommonErrorResponse.class, DownstreamErrorResponse.class})),

    @APIResponse(
            responseCode = "500",

            responseCode = "default",
            description = "MA API エラー",

                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response getOpening(@BeanParam GetOpeningRequest request) {
        return execute(request, () -> doProcess(request));
    }

    @Override
    protected InterFieldValidation<GetOpeningRequest> interFieldValidation() {
        return request -> {
            if ("1".equals(request.syoriKbn()) && request.gasMtstNo() == null) {
                return List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "gasMtstNo"));
            }
            if ("2".equals(request.syoriKbn()) && (request.gmb() == null || request.gmb().trim().isEmpty())) {
                return List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "gmb"));
            }
            return List.of();
        };
    }

    private Response doProcess(GetOpeningRequest request) {
        Long resolvedGasMtstNo = resolveGasMtstNo(request);
        if (resolvedGasMtstNo == null) {
            return temporaryBusinessError();
        }

        String systemDatetime = boundSystemDatetimeText();

        RetailDivisionResponse retailDivisionResponse = retailDivisionMaAdapter.invoke(new RetailDivisionRequest(
                "2",
                String.valueOf(resolvedGasMtstNo),
                request.knskKjnYmd(),
                reportingSystemType,
                systemDatetime));

        String kuHntKbn = retailDivisionResponse.kuHntKbn();
        if (RetailTypeJudgmentCode.TOKYO_GAS.code().equals(kuHntKbn)) {
            jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.retail.retail_tg.LivalitOpeningWorkInfoGetResponse maResponse =
                    tgRetailMaAdapter.invoke(
                            new jp.co.tokyo_gas.kaiheisen.contracts.ma.v1.pod_other.retail.retail_tg.LivalitOpeningWorkInfoGetRequest(
                                    String.valueOf(resolvedGasMtstNo),
                                    request.knskKjnYmd(),
                                    reportingSystemType,
                                    systemDatetime));
            return Response.ok(toResponse(maResponse)).build();
        }
        if (RetailTypeJudgmentCode.OTHER.code().equals(kuHntKbn)) {
            LivalitOpeningWorkInfoGetOtherRetailResponse maResponse = otherRetailMaAdapter.invoke(
                    new LivalitOpeningWorkInfoGetOtherRetailRequest(
                            String.valueOf(resolvedGasMtstNo),
                            request.knskKjnYmd(),
                            reportingSystemType,
                            systemDatetime));
            return Response.ok(toResponse(maResponse)).build();
        }
        if (RetailTypeJudgmentCode.NO_DATA.code().equals(kuHntKbn)) {
            return invokeRetailNoneMa(resolvedGasMtstNo, request.knskKjnYmd(), systemDatetime);
        }

        // 既存挙動に合わせ、KH031 以外の値も対象データなし経路へフォールバックする。
        return invokeRetailNoneMa(resolvedGasMtstNo, request.knskKjnYmd(), systemDatetime);
    }

    private Response invokeRetailNoneMa(
            Long resolvedGasMtstNo,
            String knskKjnYmd,
            String systemDatetime) {
        LivalitOpeningWorkInfoGetResponse maResponse = retailNoneMaAdapter.invoke(
                new LivalitOpeningWorkInfoGetRequest(
                        String.valueOf(resolvedGasMtstNo),
                        knskKjnYmd,
                        reportingSystemType,
                        systemDatetime));
        return Response.ok(toResponse(maResponse)).build();
    }

    private Long resolveGasMtstNo(GetOpeningRequest request) {
        if ("1".equals(request.syoriKbn())) {
            return request.gasMtstNo();
        }
        LocalDate knskKjnYmd = Rfc3339LocalDateParser.parse(request.knskKjnYmd());
        List<GasmtrkRecord> gasmtrkRecords = gasmtrkDataAccess.findAllByGmbAndGmtkYmdAndGmtThkgYmd(request.gmb(), knskKjnYmd);
        if (gasmtrkRecords.isEmpty()) {
            return null;
        }
        return gasmtrkRecords.getFirst().ttkGmsbNo();
    }

    private Response temporaryBusinessError() {
        return Response.status(Response.Status.BAD_REQUEST)
            .entity(new KaisenRuleErrorResponse(
                ErrorCodeConstants.ERROR_CODE_020100012,
                "ガス未使用のため受付できません。正しいお客さまを特定しなおしてください。",
                List.of(),
                "CXXM20859E",
                List.of(),
                "12"))
            .build();
    }
