/**
 * BFF-003 LIVALIT開栓落成 API。
 */
@Bff(identifier = "BFF-003", name = "LIVALIT開栓落成")
@Tag(name = "opening", description = "BFF-003 LIVALIT開栓落成")
@Path(PostOpeningEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
public class PostOpeningResource extends BaseResource<PostOpeningRequest> {

    private final PostOpeningService postOpeningService;

    private final ObjectMapper objectMapper;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param postOpeningService 開栓落成サービス
     * @param objectMapper JSON マッパー
     */
    @Inject
    public PostOpeningResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            PostOpeningService postOpeningService,
            ObjectMapper objectMapper) {
        super(requestContext, requestDtoValidator, clock);
        this.postOpeningService = postOpeningService;
        this.objectMapper = objectMapper;
    }

    /**
     * 開栓落成を実施する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    @POST
    @Transactional(TxType.REQUIRED)
    @Operation(
            summary = "LIVALIT開栓落成",
            description = "処理区分・開栓事由種別・ＴＧ他小売区分に応じて downstream MA を呼び分ける。",
            operationId = "postOpening")
    @RequestBody(
            required = true,
            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = PostOpeningRequest.class)))
    @APIResponse(
            responseCode = "200",
            description = "正常レスポンス",
            content = @Content(mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = PostOpeningResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {PostOpeningBadRequestErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {KaiheisenErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "default",
            description = "MA API エラー",
            content = {
                @Content(mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response postOpening(PostOpeningRequest request) {
        try {
            if (request == null) {
                throw new ValidationException(
                        List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "requestBody")),
                        true);
            }
            return execute(request, () -> Response.ok(postOpeningService.handle(request)).build());
        } catch (ValidationException | BusinessRuleException exception) {
            return toBadRequestResponse(exception);
        } catch (MaApiException exception) {
            return toMaBadRequestResponse(exception);
        }
    }

    @Override
    protected InterFieldValidation<PostOpeningRequest> interFieldValidation() {
        return request -> {
            List<ValidationError> errors = new ArrayList<>();
            if (!isBlank(request.sgyAnknNo()) && !request.sgyAnknNo().startsWith("0")) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, "sgyAnknNo"));
            }
            if ("1".equals(normalizeOptional(request.tgTkourKbn())) && isBlank(request.tkykNo())) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "tkykNo"));
            }
            if ("3".equals(request.syoriKbn()) && isBlank(request.ksenErrRyCd())) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "ksenErrRyCd"));
            }
            if (("21".equals(request.kasJysb()) || "31".equals(request.kasJysb()))
                    && normalizeOptional(request.tgTkourKbn()) != null) {
                errors.add(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000012, "tgTkourKbn"));
            }
            return List.copyOf(errors);
        };
    }

    private boolean isBlank(String value) {
        return value == null || value.trim().isEmpty();
    }

    private Response toBadRequestResponse(KaiheisenException exception) {
        ErrorResponseBody responseBody = BffErrorResponseResolver.fromKaiheisenException(exception);
        PostOpeningBadRequestErrorResponse body = new PostOpeningBadRequestErrorResponse(
                null,
                responseBody.errorCode(),
                responseBody.errorDescription(),
                responseBody.errorDetailList());
        return Response.status(Response.Status.BAD_REQUEST)
                .type(MediaType.APPLICATION_JSON)
                .entity(body)
                .build();
    }

    private Response toMaBadRequestResponse(MaApiException exception) {
        return BffPassthroughMaApiException.from(exception)
                .filter(maResponse -> maResponse.getStatus() == Response.Status.BAD_REQUEST.getStatusCode())
                .map(maResponse -> toPostOpeningBadRequestResponse(maResponse))
                .orElseThrow(() -> exception);
    }

    private Response toPostOpeningBadRequestResponse(BffPassthroughMaApiException exception) {
        String responseBody = exception.getResponseBody();
        if (responseBody == null || responseBody.isBlank()) {
            throw exception;
        }

        try {
            JsonNode root = objectMapper.readTree(responseBody);

            PostOpeningBadRequestErrorResponse body = new PostOpeningBadRequestErrorResponse(
                    textOrNull(root, "ksenErrRyCd"),
                    textOrNull(root, "errorCode"),
                    textOrNull(root, "errorDescription"),
                    stringListOrNull(root, "errorDetailList"));

            return Response.status(Response.Status.BAD_REQUEST)
                    .type(MediaType.APPLICATION_JSON)
                    .entity(body)
                    .build();
        } catch (IOException | RuntimeException e) {
            throw exception;
        }
    }

    private String textOrNull(JsonNode root, String fieldName) {
        JsonNode value = root.get(fieldName);
        return value == null || value.isNull() ? null : value.asText();
    }

    private List<String> stringListOrNull(JsonNode root, String fieldName) {
        JsonNode value = root.get(fieldName);
        if (value == null || value.isNull()) {
            return null;
        }

        List<String> result = new ArrayList<>();
        value.forEach(item -> result.add(item.asText()));
        return result;
    }
}
