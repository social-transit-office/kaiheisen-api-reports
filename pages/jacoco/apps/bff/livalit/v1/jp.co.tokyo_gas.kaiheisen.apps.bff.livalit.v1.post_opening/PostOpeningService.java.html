    /**
     * BFF-003 の開栓落成を処理する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    public PostOpeningResponse handle(PostOpeningRequest request) {
        LocalDateTime systemDatetime = requestContext.getSystemDatetime().orElseGet(clock::currentDateTime);
        String downstreamSystemDatetime = resolveSystemDatetimeForDownstream(request, systemDatetime);
        WorkHistoryBusinessDetailCodes workHistoryBusinessDetailCodes =
                resolveWorkHistoryBusinessDetailCodes(request.kasJysb());

        if (PROCESS_UPDATE.equals(request.syoriKbn())) {
            return handleUpdate(request, workHistoryBusinessDetailCodes, downstreamSystemDatetime);
        }
        if (PROCESS_FORCE.equals(request.syoriKbn())) {
            return handleForce(request, workHistoryBusinessDetailCodes, downstreamSystemDatetime);
        }
        throw new KaiheisenException("処理区分の値が不正のため、処理を終了しました");
    }

    private PostOpeningResponse handleUpdate(
            PostOpeningRequest request,
            WorkHistoryBusinessDetailCodes workHistoryBusinessDetailCodes,
            String downstreamSystemDatetime) {
        if (openingReasonAlternativeOpening.equals(request.kasJysb())
                || openingReasonNewlyOpenedTap.equals(request.kasJysb())) {
            if (RETAIL_DIVISION_OTHER.equals(normalizeOptional(request.tgTkourKbn()))) {
                OpeningCommonNormalOtherRetailResponse response =
                        openingCommonNormalOtherRetailMaAdapter.invoke(
                                buildNormalOtherRetailRequest(
                                        request,
                                        workHistoryBusinessDetailCodes,
                                        downstreamSystemDatetime));
                return toNormalResponse(
                        response.ksenErrRyCd(),
                        response.syoKyoTrno(),
                        null,
                        response.svkyNo(),
                        response.custmrNo(),
                        response.correlationId(),
                        response.errorCode(),
                        response.errorDescription(),
                        response.errorDetailList());
            }
            OpeningCommonNormalTgRetailResponse response =
                    openingCommonNormalTgRetailMaAdapter.invoke(
                            buildNormalTgRetailRequest(
                                    request,
                                    workHistoryBusinessDetailCodes,
                                    downstreamSystemDatetime));
            return toNormalResponse(
                    response.ksenErrRyCd(),
                    response.syoKyoTrno(),
                    response.sihKiyNo(),
                    response.svkyNo(),
                    response.custmrNo(),
                    response.correlationId(),
                    response.errorCode(),
                    response.errorDescription(),
                    response.errorDetailList());
        }

        if (openingReasonContinuedUse.equals(request.kasJysb())) {
            OpeningCommonContinueResponse response =
                    openingCommonContinueMaAdapter.invoke(buildContinueRequest(
                            request,
                            workHistoryBusinessDetailCodes,
                            downstreamSystemDatetime));
            return toNormalResponse(
                    response.ksenErrRyCd(),
                    response.syoKyoTrno(),
                    response.sihKiyNo(),
                    response.svkyNo(),
                    response.custmrNo(),
                    response.correlationId(),
                    response.errorCode(),
                    response.errorDescription(),
                    response.errorDetailList());
        }

        if (!openingReasonSupplyResumed.equals(request.kasJysb())) {
            throw new KaiheisenException("開栓事由種別の値が不正のため、処理を終了しました");
        }

        OpeningCommonResumeTgRetailResponse response =
                openingCommonResumeTgRetailMaAdapter.invoke(buildResumeTgRetailRequest(
                        request,
                        workHistoryBusinessDetailCodes,
                        downstreamSystemDatetime));
        return toNormalResponse(
                response.ksenErrRyCd(),
                response.syoKyoTrno(),
                response.sihKiyNo(),
                response.svkyNo(),
                response.custmrNo(),
                response.correlationId(),
                response.errorCode(),
                response.errorDescription(),
                response.errorDetailList());
    }

    private PostOpeningResponse handleForce(
            PostOpeningRequest request,
            WorkHistoryBusinessDetailCodes workHistoryBusinessDetailCodes,
            String downstreamSystemDatetime) {
        if (RETAIL_DIVISION_OTHER.equals(normalizeOptional(request.tgTkourKbn()))) {
            OpeningCommonForceOtherRetailResponse response =
                    openingCommonForceOtherRetailMaAdapter.invoke(
                            buildForceOtherRetailRequest(
                                    request,
                                    workHistoryBusinessDetailCodes,
                                    downstreamSystemDatetime));
            return toForceResponse(response.correlationId());
        }
        OpeningCommonForceTgRetailResponse response =
                openingCommonForceTgRetailMaAdapter.invoke(buildForceTgRetailRequest(
                        request,
                        workHistoryBusinessDetailCodes,
                        downstreamSystemDatetime));
        return toForceResponse(response.correlationId());
    }

    private String resolveCorrelationId(String responseCorrelationId) {
        String requestCorrelationId = requestContext.getCorrelationId().orElse(null);
        if (requestCorrelationId != null) {
            return requestCorrelationId;
        }
        String normalizedResponseCorrelationId = normalizeOptional(responseCorrelationId);
        if (normalizedResponseCorrelationId != null) {
            return normalizedResponseCorrelationId;
        }
        return null;
    }

    private String resolveSystemDatetimeForDownstream(PostOpeningRequest request, LocalDateTime systemDatetime) {
        String requestedSystemDatetime = normalizeOptional(request.systemDatetime());
        if (requestedSystemDatetime != null) {
            return toRfc3339DateTimeText(requestedSystemDatetime);
        }
        return systemDatetime.atOffset(ZoneOffset.UTC)
                .truncatedTo(ChronoUnit.MICROS)
                .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
    }

    private WorkHistoryBusinessDetailCodes resolveWorkHistoryBusinessDetailCodes(String kasJysb) {
        if (openingReasonAlternativeOpening.equals(kasJysb)) {
            return alternativeOpeningCodes;
        }
        if (openingReasonNewlyOpenedTap.equals(kasJysb)) {
            return newlyOpenedTapCodes;
        }
        if (openingReasonContinuedUse.equals(kasJysb)) {
            return continuedUseCodes;
        }
        if (openingReasonSupplyResumed.equals(kasJysb)) {
            return supplyResumedCodes;
        }
        throw new KaiheisenException("開栓事由種別の値が不正のため、処理を終了しました");
    }

    private String normalizeOptional(String value) {
        if (value == null) {
            return null;
        }
        String trimmed = value.trim();
        return trimmed.isEmpty() ? null : trimmed;
    }

    private String optionalText(String value) {
        return normalizeOptional(value);
    }

    private static String stripTrailingSpaces(String value) {
        return value == null ? null : value.stripTrailing();
    }

    private String optionalAddressKana(String value, String fieldName, int maxLength) {
        if (value == null) {
            return null;
        }
        String trimmed = value.trim();
        if (trimmed.isEmpty()) {
            return null;
        }
        String normalized = value.stripTrailing();
        List<ValidationError> errors = new ArrayList<>();
        if (trimmed.length() > maxLength) {
            errors.add(ValidationError.of(
                    ErrorCodeConstants.ERROR_CODE_010000022,
                    fieldName,
                    String.valueOf(maxLength)));
        }
        if (!HALF_WIDTH_TEXT_PATTERN.matcher(normalized).matches()) {
            errors.add(ValidationError.ofField(
                    ErrorCodeConstants.ERROR_CODE_010000031,
                    fieldName));
        }
        if (!errors.isEmpty()) {
            throw new ValidationException(errors, true);
        }
        return normalized;
    }

    private Long nullableIfZero(Long value) {
        return value != null && value.longValue() == 0L ? null : value;
    }

    private String toRfc3339DateText(String raw) {
        String normalized = normalizeOptional(raw);
        if (normalized == null) {
            return null;
        }
        try {
            return OffsetDateTime.parse(normalized).toLocalDate().toString();
        } catch (DateTimeParseException exception) {
            try {
                return LocalDateTime.parse(normalized).toLocalDate().toString();
            } catch (DateTimeParseException ignored) {
                return LocalDate.parse(normalized).toString();
            }
        }
    }

    private String toRfc3339DateTimeText(String rawDateTime) {
        String normalized = normalizeOptional(rawDateTime);
        if (normalized == null) {
            return null;
        }
        try {
            return OffsetDateTime.parse(normalized)
                    .truncatedTo(ChronoUnit.MICROS)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        } catch (DateTimeParseException exception) {
            return LocalDateTime.parse(normalized)
                    .truncatedTo(ChronoUnit.MICROS)
                    .atOffset(ZoneOffset.UTC)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        }
    }
