package jp.co.tokyo_gas.kaiheisen.apps.bff.livalit.v1.cancel_closing;

/**
 * BFF-005 LIVALIT閉栓取消 API。
 */
@Bff(identifier = "BFF-005", name = "LIVALIT閉栓取消")
@Tag(name = "closing", description = "BFF-005 LIVALIT閉栓取消")
@Path(CancelClosingEndpointContract.PATH)
@Consumes(MediaType.APPLICATION_JSON)
@Produces(MediaType.APPLICATION_JSON)
public class CancelClosingResource extends BaseResource<CancelClosingRequest> {

    private static final String MDC_CORRELATION_ID_KEY = "correlationId";

    private final CancelClosingService cancelClosingService;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param requestDtoValidator DTO バリデータ
     * @param clock システム日時プロバイダ
     * @param cancelClosingService 閉栓取消サービス
     */
    @Inject
    public CancelClosingResource(
            RequestContext requestContext,
            RequestDtoValidator requestDtoValidator,
            SystemDateTimeProvider clock,
            CancelClosingService cancelClosingService) {
        super(requestContext, requestDtoValidator, clock);
        this.cancelClosingService = cancelClosingService;
    }

    /**
     * 閉栓取消を実施する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    @POST
    @Operation(
            summary = "LIVALIT閉栓取消",
            description = "SV-097 と MA-107 の結果に応じて閉栓取消の downstream MA を呼び分ける。",
            operationId = "cancelClosing")
    @RequestBody(
            required = true,
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CancelClosingRequest.class)))
    @APIResponse(
            responseCode = "200",
            description = "正常レスポンス",
            content = @Content(
                    mediaType = MediaType.APPLICATION_JSON,
                    schema = @Schema(implementation = CancelClosingResponse.class)))
    @APIResponse(
            responseCode = "400",
            description = "入力不正・業務エラー",
            content = {
                @Content(
                        mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {CommonErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "500",
            description = "サーバエラー",
            content = {
                @Content(
                        mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(anyOf = {KaiheisenErrorResponse.class, DownstreamErrorResponse.class})),
                @Content(mediaType = "*/*")
            })
    @APIResponse(
            responseCode = "default",
            description = "MA API エラー",
            content = {
                @Content(
                        mediaType = MediaType.APPLICATION_JSON,
                        schema = @Schema(implementation = DownstreamErrorResponse.class)),
                @Content(mediaType = "*/*")
            })
    public Response cancelClosing(CancelClosingRequest request) {
        if (request == null) {
            throw new ValidationException(
                    List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "requestBody")),
                    true);
        }
        return execute(
                request,
                () -> Response.ok(cancelClosingService.handle(request)).build());
    }

    /**
     * RequestContext バインド後に相関IDをUUID v4へ正規化する。
     *
     * @param request リクエスト
     */
    @Override
    protected void afterBindRequestContext(CancelClosingRequest request) {
        bindUuidCorrelationIdForResponse();
    }

    private void bindUuidCorrelationIdForResponse() {
        String correlationId = CorrelationIdResolver.resolveUuidV4(
                requestContext.getCorrelationId().orElse(null));
        requestContext.setCorrelationId(correlationId);
        MDC.put(MDC_CORRELATION_ID_KEY, correlationId);
    }
}
