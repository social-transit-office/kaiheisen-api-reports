/**
 * BFF-005 LIVALIT閉栓取消のアプリケーションサービス。
 */
@ApplicationScoped
public class CancelClosingService {

    static final String REPORTING_SYSTEM_TYPE = ConstantLookupKeys.Kaiheisen.ReportingSystemType.LIVALIT;

    static final String RETAIL_DIVISION_SHR_KBN = "2";

    static final String CLOSING_REGISTRATION_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.CLOSING_REGISTRATION;

    static final String SUPPLY_STOP_REPORT_CODE =
            ConstantLookupKeys.Cirius.TransferWorkHistoryBusinessCodeRetail.SUPPLY_OUTAGE_REPORT_RETAIL;

    private final RequestContext requestContext;

    private final ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeService;

    private final RetailDivisionMaAdapter retailDivisionMaAdapter;

    private final ClosingCancelClosingReportCancelTgRetailMaAdapter closingCancelClosingReportCancelTgRetailMaAdapter;

    private final ClosingSupplyStopReportCancelMaAdapter closingSupplyStopReportCancelMaAdapter;

    private final ClosingCancelClosingReportCancelOtherRetailMaAdapter closingCancelClosingReportCancelOtherRetailMaAdapter;

    private final SystemDateTimeProvider clock;

    /**
     * コンストラクタ。
     *
     * @param requestContext リクエストコンテキスト
     * @param historyDetailCodeService SV-097
     * @param retailDivisionMaAdapter MA-107 adapter
     * @param closingCancelClosingReportCancelTgRetailMaAdapter MA-099 adapter
     * @param closingSupplyStopReportCancelMaAdapter MA-100 adapter
     * @param closingCancelClosingReportCancelOtherRetailMaAdapter MA-103 adapter
     * @param clock システム日時プロバイダ
     */
    @Inject
    public CancelClosingService(
            RequestContext requestContext,
            ClosingCancelFeasibilityJudgmentHistoryDetailCode historyDetailCodeService,
            RetailDivisionMaAdapter retailDivisionMaAdapter,
            ClosingCancelClosingReportCancelTgRetailMaAdapter closingCancelClosingReportCancelTgRetailMaAdapter,
            ClosingSupplyStopReportCancelMaAdapter closingSupplyStopReportCancelMaAdapter,
            ClosingCancelClosingReportCancelOtherRetailMaAdapter closingCancelClosingReportCancelOtherRetailMaAdapter,
            SystemDateTimeProvider clock) {
        this.requestContext = requestContext;
        this.historyDetailCodeService = historyDetailCodeService;
        this.retailDivisionMaAdapter = retailDivisionMaAdapter;
        this.closingCancelClosingReportCancelTgRetailMaAdapter = closingCancelClosingReportCancelTgRetailMaAdapter;
        this.closingSupplyStopReportCancelMaAdapter = closingSupplyStopReportCancelMaAdapter;
        this.closingCancelClosingReportCancelOtherRetailMaAdapter =
                closingCancelClosingReportCancelOtherRetailMaAdapter;
        this.clock = clock;
    }

    /**
     * BFF-005 の閉栓取消を処理する。
     *
     * @param request リクエスト
     * @return レスポンス
     */
    public CancelClosingResponse handle(CancelClosingRequest request) {
        requestContext.setHukkSysShbt(REPORTING_SYSTEM_TYPE);

        LocalDateTime systemDatetime = requestContext.getSystemDatetime().orElseGet(clock::currentDateTime);
        String downstreamSystemDatetime = resolveSystemDatetimeForDownstream(request, systemDatetime);
        String trkKnsYmd = toRfc3339DateText(request.trkKnsYmd());

        ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput =
                historyDetailCodeService.execute(request.gasMtstNo(), CLOSING_REGISTRATION_CODE);

        RetailDivisionResponse retailDivisionResponse = retailDivisionMaAdapter.invoke(new RetailDivisionRequest(
                RETAIL_DIVISION_SHR_KBN,
                request.gasMtstNo().toString(),
                trkKnsYmd,
                REPORTING_SYSTEM_TYPE,
                downstreamSystemDatetime));

        String requestedRetailerCode = retailDivisionResponse.tkssvkyInf() == null
                ? null
                : retailDivisionResponse.tkssvkyInf().tksuKijsCd();
        if (ConstantLookupKeys.Cirius.RetailerCode.TOKYO_GAS_CO_LTD.equals(requestedRetailerCode)) {
            if (SUPPLY_STOP_REPORT_CODE.equals(historyDetailCodeOutput.isrrGymCd())) {
                ClosingSupplyStopReportCancelResponse response = closingSupplyStopReportCancelMaAdapter.invoke(
                        buildSupplyStopRequest(request, historyDetailCodeOutput, trkKnsYmd, downstreamSystemDatetime));
                return mapResponse(response.correlationId(), response.errorCode(), response.errorDescription(),
                        response.errorDetailList());
            }

            ClosingCancelClosingReportCancelTgRetailResponse response =
                    closingCancelClosingReportCancelTgRetailMaAdapter.invoke(
                            buildTgRetailRequest(
                                    request,
                                    retailDivisionResponse,
                                    historyDetailCodeOutput,
                                    trkKnsYmd,
                                    downstreamSystemDatetime));
            return mapResponse(response.correlationId(), response.errorCode(), response.errorDescription(),
                    response.errorDetailList());
        } else {
            ClosingCancelClosingReportCancelOtherRetailResponse response =
                    closingCancelClosingReportCancelOtherRetailMaAdapter.invoke(
                            buildOtherRetailRequest(
                                    request,
                                    historyDetailCodeOutput,
                                    trkKnsYmd,
                                    downstreamSystemDatetime));
            return mapResponse(response.correlationId(), response.errorCode(), response.errorDescription(),
                    response.errorDetailList());
        }
    }

    private ClosingCancelClosingReportCancelTgRetailRequest buildTgRetailRequest(
            CancelClosingRequest request,
            RetailDivisionResponse retailDivisionResponse,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput,
            String trkKnsYmd,
            String downstreamSystemDatetime) {
        TkssvkyInfoResponse tkssvkyInfo = requiredTkssvkyInfo(retailDivisionResponse);
        String ukeIndexNo = requiredUkeIndexNo(request);
        return new ClosingCancelClosingReportCancelTgRetailRequest(
                request.gasMtstNo(),
                trkKnsYmd,
                CLOSING_REGISTRATION_CODE,
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.sgytNwkCd(),
                request.sgttUsrKj(),
                normalizeOptional(request.rirekiMemo()),
                ukeIndexNo,
                new ClosingCancelClosingReportCancelTgRetailRequest.TkssvkyInf(tkssvkyInfo.syoKyoTrno()),
                toTgRetailHistoryUpdateInfo(historyDetailCodeOutput),
                REPORTING_SYSTEM_TYPE,
                downstreamSystemDatetime);
    }

    private ClosingSupplyStopReportCancelRequest buildSupplyStopRequest(
            CancelClosingRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput,
            String trkKnsYmd,
            String downstreamSystemDatetime) {
        return new ClosingSupplyStopReportCancelRequest(
                request.gasMtstNo(),
                trkKnsYmd,
                SUPPLY_STOP_REPORT_CODE,
                request.hukkUsrId(),
                request.hukkUnmKj(),
                normalizeOptional(request.rirekiMemo()),
                toSupplyStopHistoryUpdateInfo(historyDetailCodeOutput),
                REPORTING_SYSTEM_TYPE,
                normalizeOptional(request.correlationId()),
                downstreamSystemDatetime);
    }

    private ClosingCancelClosingReportCancelOtherRetailRequest buildOtherRetailRequest(
            CancelClosingRequest request,
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput,
            String trkKnsYmd,
            String downstreamSystemDatetime) {
        return new ClosingCancelClosingReportCancelOtherRetailRequest(
                request.gasMtstNo(),
                trkKnsYmd,
                CLOSING_REGISTRATION_CODE,
                request.hukkUsrId(),
                request.hukkUnmKj(),
                request.sgytNwkCd(),
                request.sgttUsrId(),
                request.sgttUsrKj(),
                normalizeOptional(request.rirekiMemo()),
                toOtherRetailHistoryUpdateInfo(historyDetailCodeOutput),
                REPORTING_SYSTEM_TYPE,
                downstreamSystemDatetime);
    }

    private ClosingCancelClosingReportCancelTgRetailRequest.IsrrkrUpdInf toTgRetailHistoryUpdateInfo(
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput) {
        return new ClosingCancelClosingReportCancelTgRetailRequest.IsrrkrUpdInf(
                historyDetailCodeOutput.duknIsrgCd(),
                historyDetailCodeOutput.duknIsrsCd(),
                historyDetailCodeOutput.isrrGymCd(),
                historyDetailCodeOutput.isrrGmssCd());
    }

    private ClosingSupplyStopReportCancelRequest.IsrrkrUpdInf toSupplyStopHistoryUpdateInfo(
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput) {
        return new ClosingSupplyStopReportCancelRequest.IsrrkrUpdInf(
                historyDetailCodeOutput.duknIsrgCd(),
                historyDetailCodeOutput.duknIsrsCd(),
                historyDetailCodeOutput.isrrGymCd(),
                historyDetailCodeOutput.isrrGmssCd());
    }

    private ClosingCancelClosingReportCancelOtherRetailRequest.IsrrkrUpdInf toOtherRetailHistoryUpdateInfo(
            ClosingCancelFeasibilityJudgmentHistoryDetailCodeOutput historyDetailCodeOutput) {
        return new ClosingCancelClosingReportCancelOtherRetailRequest.IsrrkrUpdInf(
                historyDetailCodeOutput.duknIsrgCd(),
                historyDetailCodeOutput.duknIsrsCd(),
                historyDetailCodeOutput.isrrGymCd(),
                historyDetailCodeOutput.isrrGmssCd());
    }

    private CancelClosingResponse mapResponse(
            String downstreamCorrelationId,
            String errorCode,
            String errorDescription,
            List<String> errorDetailList) {
        return new CancelClosingResponse(
                requestContext.getCorrelationId().orElse(downstreamCorrelationId),
                errorCode,
                errorDescription,
                errorDetailList);
    }

    private TkssvkyInfoResponse requiredTkssvkyInfo(RetailDivisionResponse retailDivisionResponse) {
        if (retailDivisionResponse == null || retailDivisionResponse.tkssvkyInf() == null
                || retailDivisionResponse.tkssvkyInf().syoKyoTrno() == null) {
            throw new KaiheisenException("MA-107 の託送サービス契約情報が不正のため、処理を終了しました");
        }
        return retailDivisionResponse.tkssvkyInf();
    }

    private String requiredUkeIndexNo(CancelClosingRequest request) {
        String ukeIndexNo = normalizeOptional(request.utidxNo());
        if (ukeIndexNo != null) {
            return ukeIndexNo;
        }
        throw new ValidationException(
                List.of(ValidationError.ofField(ErrorCodeConstants.ERROR_CODE_010000011, "utidxNo")),
                true);
    }

    private String resolveSystemDatetimeForDownstream(
            CancelClosingRequest request,
            LocalDateTime systemDatetime) {
        String requestedSystemDatetime = normalizeOptional(request.systemDatetime());
        if (requestedSystemDatetime != null) {
            return toRfc3339DateTimeText(requestedSystemDatetime);
        }
        return systemDatetime
                .atOffset(ZoneOffset.UTC)
                .truncatedTo(ChronoUnit.MICROS)
                .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
    }

    private String toRfc3339DateText(String rawDate) {
        try {
            return LocalDate.parse(rawDate).format(DateTimeFormatter.ISO_LOCAL_DATE);
        } catch (DateTimeParseException ignored) {
            try {
                return OffsetDateTime.parse(rawDate).toLocalDate().format(DateTimeFormatter.ISO_LOCAL_DATE);
            } catch (DateTimeParseException secondary) {
                return LocalDateTime.parse(rawDate).toLocalDate().format(DateTimeFormatter.ISO_LOCAL_DATE);
            }
        }
    }

    private String toRfc3339DateTimeText(String rawDateTime) {
        try {
            return OffsetDateTime.parse(rawDateTime)
                    .truncatedTo(ChronoUnit.MICROS)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        } catch (DateTimeParseException exception) {
            return LocalDateTime.parse(rawDateTime)
                    .truncatedTo(ChronoUnit.MICROS)
                    .atOffset(ZoneOffset.UTC)
                    .format(DateTimeFormatter.ISO_OFFSET_DATE_TIME);
        }
    }

    private String normalizeOptional(String value) {
        if (value == null) {
            return null;
        }
        String normalized = value.trim();
        return normalized.isEmpty() ? null : normalized;
    }
